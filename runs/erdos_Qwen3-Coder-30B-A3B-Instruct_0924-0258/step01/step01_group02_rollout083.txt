import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        initial_h = np.full(45, 1.0/45)
        
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial_h satisfies the constraint
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def mutate_individual(h_vec, mutation_rate=0.1):
        h_new = h_vec.copy()
        for i in range(len(h_new)):
            if np.random.random() < mutation_rate:
                h_new[i] += np.random.normal(0, 0.05)
        return project_h(h_new)
    
    def crossover_individuals(h1, h2):
        # Uniform crossover
        mask = np.random.random(len(h1)) > 0.5
        h_new = np.where(mask, h1, h2)
        return project_h(h_new)
    
    def tournament_selection(population, fitnesses, tournament_size=3):
        selected = []
        for _ in range(len(population) // 2):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitness)]
            selected.append(population[winner_idx])
        return selected
    
    # Initialize population - include the initial solution
    population_size = 50
    population = [initial_h.copy()]
    
    while len(population) < population_size:
        h = np.random.random(n_points)
        h = project_h(h)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = [compute_c5(h) for h in population]
    
    best_h = population[np.argmin(fitnesses)].copy()
    best_c5 = min(fitnesses)
    
    start_time = time.time()
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Crossover
        offspring = []
        for i in range(0, len(selected), 2):
            if i + 1 < len(selected):
                child1 = crossover_individuals(selected[i], selected[i+1])
                child2 = crossover_individuals(selected[i], selected[i+1])
                offspring.extend([child1, child2])
            else:
                # Single individual: mutate
                offspring.append(mutate_individual(selected[i]))
        
        # Mutation
        for i in range(len(offspring)):
            offspring[i] = mutate_individual(offspring[i])
        
        # Evaluate offspring
        offspring_fitnesses = [compute_c5(h) for h in offspring]
        
        # Update population (replace worst 25%)
        combined_pop = population + offspring
        combined_fitnesses = fitnesses + offspring_fitnesses
        
        # Sort by fitness (ascending - since we want to minimize)
        sorted_indices = np.argsort(combined_fitnesses)
        population = [combined_pop[i] for i in sorted_indices[:population_size]]
        fitnesses = [combined_fitnesses[i] for i in sorted_indices[:population_size]]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
    return (best_h.tolist(), best_c5, n_points)
