import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Used for feasibility projection
    def project_feasible(h_vec, target_sum):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Compute C5 value for given h
    def compute_C5(h_vec, dx):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Grid search with adaptive refinement
    best_h = None
    best_C5 = float('inf')
    
    # Levels to search
    grid_levels = [5, 10, 15, 20, 25, 30, 45]
    n_levels = len(grid_levels)
    
    # Start with initial solution if available, otherwise use a default
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values, dtype=float)
        n_points = len(h_start)
        # Project to ensure feasibility
        h_start = project_feasible(h_start, n_points / 2.0)
    else:
        # Default initialization
        n_points = 45
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1.0
        h_start = project_feasible(h_start, n_points / 2.0)
    
    # Initialize tracking
    best_h = h_start.copy()
    best_C5 = compute_C5(best_h, 2.0 / n_points)
    
    # Iterate over grid levels and refine
    for level_idx in range(n_levels):
        n_points = grid_levels[level_idx]
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        
        # Generate candidates
        num_candidates = 100 if level_idx < 3 else 20
        candidates = []
        
        for _ in range(num_candidates):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Random feasible point
            h = np.random.rand(n_points)
            h = project_feasible(h, target_sum)
            candidates.append(h)
        
        # Evaluate candidates
        for h in candidates:
            if time.time() - start_time > budget_s - 0.1:
                break
            C5 = compute_C5(h, dx)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
        
        # Refine best candidates at this level
        top_candidates = []
        for h in candidates:
            C5 = compute_C5(h, dx)
            top_candidates.append((C5, h))
        
        top_candidates.sort(key=lambda x: x[0])
        refined_candidates = []
        num_refine = min(20, len(top_candidates) // 2)
        for _, h in top_candidates[:num_refine]:
            # Perturb each value slightly
            h_refined = h + np.random.normal(0, 0.02, n_points)
            h_refined = project_feasible(h_refined, target_sum)
            refined_candidates.append(h_refined)
        
        # Evaluate refined ones
        for h_refined in refined_candidates:
            if time.time() - start_time > budget_s - 0.1:
                break
            C5 = compute_C5(h_refined, dx)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h_refined.copy()
                
        # Early exit if time is running out
        if time.time() - start_time > budget_s - 0.1:
            break
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n_points / 2.0)
    C5_final = compute_C5(best_h, 2.0 / n_points)
    
    return list(best_h), C5_final, n_points
