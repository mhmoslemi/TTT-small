import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    def compute_C5(h_vec, dx):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def latin_hypercube_samples(n, n_samples):
        samples = np.zeros((n_samples, n))
        for i in range(n):
            samples[:, i] = np.random.permutation(np.linspace(0, 1, n_samples))
        return samples
    
    def coordinate_descent_step(h_vec, step_size=0.05):
        h_new = h_vec.copy()
        best_C5 = compute_C5(h_vec, dx)
        current_C5 = best_C5
        best_h_new = h_new.copy()
        
        for i in range(len(h_vec)):
            original = h_vec[i]
            for delta in [-step_size, step_size]:
                h_try = h_vec.copy()
                h_try[i] = original + delta
                h_try = project_feasible(h_try)
                new_C5 = compute_C5(h_try, dx)
                if new_C5 < current_C5:
                    current_C5 = new_C5
                    best_h_new = h_try.copy()
                    
        return best_h_new, current_C5
    
    def swap_intervals(h_vec, num_swaps=10):
        h_new = h_vec.copy()
        best_C5 = compute_C5(h_vec, dx)
        
        for _ in range(num_swaps):
            if time.time() - start_time > budget_s - 0.1:
                break
            i, j = np.random.randint(0, len(h_vec), 2)
            h_try = h_vec.copy()
            h_try[i], h_try[j] = h_try[j], h_try[i]
            h_try = project_feasible(h_try)
            new_C5 = compute_C5(h_try, dx)
            if new_C5 < best_C5:
                best_C5 = new_C5
                h_new = h_try.copy()
                
        return h_new, best_C5
    
    # Generate initial population of candidates
    initial_candidates = []
    lh_samples = latin_hypercube_samples(n_points, 5)
    
    for sample in lh_samples:
        h_sample = project_feasible(sample * target_sum)
        initial_candidates.append(h_sample)
        
    # Add some perturbed versions of the parent solution
    if 'initial_h_values' in globals():
        for _ in range(2):
            h_pert = h.copy() + np.random.normal(0, 0.02, n_points)
            h_pert = project_feasible(h_pert)
            initial_candidates.append(h_pert)
    
    for candidate in initial_candidates:
        candidate_C5 = compute_C5(candidate, dx)
        if candidate_C5 < best_C5:
            best_C5 = candidate_C5
            best_h = candidate.copy()
    
    # Optimization loop
    remaining_time = budget_s - (time.time() - start_time)
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        
        # Pick a random candidate from our pool
        random_idx = np.random.randint(len(initial_candidates))
        candidate = initial_candidates[random_idx].copy()
        
        # Try coordinate descent
        h_cd, c5_cd = coordinate_descent_step(candidate, step_size=0.05)
        
        # Try swaps
        h_swap, c5_swap = swap_intervals(candidate, num_swaps=10)
        
        # Select best among all attempts
        best_candidate = candidate
        best_c5 = compute_C5(candidate, dx)
        
        if c5_cd < best_c5:
            best_candidate = h_cd
            best_c5 = c5_cd
            
        if c5_swap < best_c5:
            best_candidate = h_swap
            best_c5 = c5_swap
        
        # If we're improving, update our best solution
        if best_c5 < best_C5:
            best_C5 = best_c5
            best_h = best_candidate.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Add new candidate to the mix by trying a small random walk
        if iteration % 10 == 0:
            h_random = best_h + np.random.normal(0, 0.02, n_points)
            h_random = project_feasible(h_random)
            initial_candidates.append(h_random)
            
        # Remove oldest candidate to keep population size manageable
        if len(initial_candidates) > 10:
            initial_candidates.pop(0)
            
        if no_improvement_count > 30:
            # Reset with a new random population
            new_samples = latin_hypercube_samples(n_points, 3)
            initial_candidates = []
            for sample in new_samples:
                h_sample = project_feasible(sample * target_sum)
                initial_candidates.append(h_sample)
            no_improvement_count = 0
    
    # Final adjustment
    final_h = project_feasible(best_h)
    final_C5 = compute_C5(final_h, dx)
    
    return list(final_h), final_C5, n_points
