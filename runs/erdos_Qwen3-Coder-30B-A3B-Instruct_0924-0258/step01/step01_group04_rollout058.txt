import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Particle Swarm Optimization
    n_particles = 100
    w_start = 0.7
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    max_speed = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        # Start near initial solution with small perturbation
        h_init = h + np.random.normal(0, 0.05, n_points)
        h_init = project_h(h_init)
        particles.append(h_init.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h_init.copy())
        pbest_vals.append(compute_C5(h_init))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        w = w_start - (w_start - w_end) * iteration / 1000
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest[i] - particles[i]) +
                c2 * r2 * (gbest - particles[i])
            )
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_speed, max_speed)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_h(particles[i])
            
            # Evaluate and update personal best
            new_val = compute_C5(particles[i])
            if new_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = new_val
                
                # Check if it's new global best
                if new_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = new_val
        
        # Periodic check
        if iteration % 50 == 0:
            current_C5 = compute_C5(gbest)
            if current_C5 < gbest_val:
                gbest_val = current_C5
    
    # Final validation
    final_C5 = compute_C5(gbest)
    if final_C5 < gbest_val:
        gbest_val = final_C5
    
    return (gbest.tolist(), gbest_val, n_points)
