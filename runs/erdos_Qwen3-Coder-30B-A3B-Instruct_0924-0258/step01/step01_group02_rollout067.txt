import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():

    else:
        # Default initialization
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def loss_and_grad(h_vec, k_max=None):
        # Compute loss as max correlation and gradient
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        max_k = np.argmax(corr) if k_max is None else k_max
        loss = corr[max_k]
        
        # Gradient computation
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            eps = 1e-6
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            grad[i] = (corr_plus[max_k] - loss) / eps
            
        return loss, grad
    
    # Optimizer: gradient descent with adaptive learning rate
    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Momentum and adaptive step size control
    momentum = 0.9
    lr_adapt = 0.01
    velocity = np.zeros_like(h)
    
    max_iter = 10000
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute current max overlap and gradient
        corr = np.correlate(h, 1 - h, mode='full')
        corr *= dx
        max_k = np.argmax(corr)
        current_loss, grad = loss_and_grad(h, max_k)
        
        # Update velocity and step
        velocity = momentum * velocity - lr_adapt * grad
        h = h + velocity
        h = project_h(h)
        
        # Update learning rate if not improving
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            lr_adapt *= 1.02  # Increase learning rate if improving
        else:
            lr_adapt *= 0.99  # Decrease if not improving
            
        iter_count += 1
        
        # Occasionally restart with best solution
        if iter_count % 500 == 0 and iter_count > 0:
            h = best_h.copy()
            
    return (best_h.tolist(), best_c5, n_points)
