import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with 0<=h<=1 and sum=h."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-10:
        h = np.ones_like(h) * (target_sum / len(h))
    else:
        h = h * target_sum / current_sum
        h = np.clip(h, 0, 1)
        # If still not exact, fix with small adjustments
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-10:
            # Distribute difference
            idx = np.argsort(h)
            for i in range(len(idx)-1, -1, -1):
                if diff == 0:
                    break
                j = idx[i]
                if h[j] < 1.0 and diff > 0:
                    update = min(diff, 1.0 - h[j])
                    h[j] += update
                    diff -= update
                elif h[j] > 0.0 and diff < 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    return h

    lr = 0.01
    patience = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    for i in range(max_iter):
        elapsed = time.time() - start_time

        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode="full")
        # Find shift k that maximizes overlap
        k_max = np.argmax(np.abs(corr))
        
        # Compute gradient with respect to this shift
        grad = np.zeros_like(h)
        if k_max >= n_points - 1:
            # Corresponds to negative shift
            shift = k_max - (n_points - 1)  # Negative shift
            for j in range(n_points):
                idx = (j - shift) % n_points
                grad[j] = (1 - 2 * h[idx])
        else:
            # Positive shift
            shift = k_max
            for j in range(n_points):
                idx = (j + shift) % n_points
                grad[j] = (1 - 2 * h[idx])
        
        # Update
        h = h - lr * grad
        
        # Project to feasible region
        h = project_to_feasible(h, target_sum)
        
        # Evaluate
        c5 = evaluate(h, dx)
        if c5 < best_c5:
            best_c5 = c5
