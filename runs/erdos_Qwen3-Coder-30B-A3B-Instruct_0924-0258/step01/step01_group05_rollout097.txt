import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx

    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec

    def compute_gradient(h_vec, k_max):
        # Compute gradient for the k_max shift
        n = len(h_vec)
        grad = np.zeros(n)
        for i in range(n):
            if (i + k_max) < n:
                grad[i] -= h_vec[i + k_max]
            if (i - k_max) >= 0:
                grad[i] -= h_vec[i - k_max]
            grad[i] += 1  # For the (1-h) term
        return grad

    def optimize_step(h_vec, max_iter=50):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        alpha = 0.1
        stagnation_count = 0
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Compute all overlaps
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            # Find maximum lag
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (len(h_current) - 1)
            
            # Compute gradient
            grad = compute_gradient(h_current, max_lag)
            
            # Gradient descent step
            h_new = h_current - alpha * grad
            h_new = project_feasible(h_new)
            
            new_C5 = compute_C5(h_new)
            
            if new_C5 < best_C5:
                h_current = h_new.copy()
                best_C5 = new_C5
                stagnation_count = 0
                if alpha > 0.01:
                    alpha *= 0.95  # Slowly decrease step size if improving
            else:
                stagnation_count += 1
                if stagnation_count > 10:
                    # Perturb and restart
                    h_current = h_current + np.random.normal(0, 0.01, n_points)
                    h_current = project_feasible(h_current)
                    stagnation_count = 0
                    alpha = 0.1  # Reset step size
            
        return h_current, best_C5

    h_best = h.copy()
    C5_best = compute_C5(h_best)

    # Local optimization loop
    i = 0
    while time.time() - start_time < budget_s - 0.1:
        # Try several local optimizations from current point
        for _ in range(3):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Small random perturbation
            h_perturbed = h_best + np.random.normal(0, 0.01, n_points)
            h_perturbed = project_feasible(h_perturbed)
            
            # Perform optimization from this point
            h_opt, C5_opt = optimize_step(h_perturbed, max_iter=50)
            if C5_opt < C5_best:
                C5_best = C5_opt
                h_best = h_opt.copy()
        i += 1

    # Final check
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
