import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to target sum
    if current_sum > 0:
        h = h * target_sum / current_sum
    # Clamp again after scaling
    h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, k_max, dx):
    """Compute gradient of the overlap w.r.t. h for the shift that maximizes it."""
    n = len(h)
    grad = np.zeros(n)
    # For shift k_max, contribution to overlap is h[i] * (1 - h[i + k_max])
    # So gradient is -1 * (1 - h[i + k_max]) from h[i] term
    # and -1 * h[i] from h[i + k_max] term
    for i in range(n - k_max):
        j = i + k_max
        grad[i] -= (1 - h[j])
        grad[j] -= h[i]
    return grad * dx  # scale by dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5  # Simple uniform start
    
    # Normalize to desired sum
    h = project_to_feasible(h0, n_points / 2.0)
    
    # Set dx
    dx = 2.0 / n_points
    
    # Initialize hyperparameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 500
    best_c5 = float('inf')
    best_h = None
    no_improvement_count = 0
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Evaluate current solution
        c5 = evaluate(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early termination if no improvement
        if no_improvement_count > patience:
            # Simple perturbation to escape local minima
            h += np.random.normal(0, 0.001, n_points)
            h = project_to_feasible(h, n_points / 2.0)
            no_improvement_count = 0
            continue
            
        # Compute correlations to find dominant shift
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        
        # Compute gradient
        grad = compute_gradient(h, k_max, dx)
        
        # Update h with gradient descent
        h = h - learning_rate * grad
        
        # Project to feasible region (box + sum constraint)
        h = project_to_feasible(h, n_points / 2.0)
        
        # Adaptive learning rate
        if i % 100 == 0 and i > 0:
            # Reduce learning rate if no significant improvement
            if no_improvement_count > 50:
                learning_rate *= 0.99
            # Increase if improving fast
            elif no_improvement_count < 20:
                learning_rate *= 1.01
                
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        # Fallback to current h
        final_c5 = evaluate(h, dx)
        return (h.tolist(), float(final_c5), n_points)
