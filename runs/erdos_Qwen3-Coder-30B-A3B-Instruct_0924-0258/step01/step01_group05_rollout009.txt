import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        # Compute cross-correlation for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Genetic Algorithm parameters
    pop_size = 50
    tournament_size = 10
    elitism_rate = 0.2
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_init = np.random.rand(n_points)
        h_init = project_feasible(h_init)
        population.append(h_init)
    
    best_h = None
    best_C5 = float('inf')
    
    # Evaluation of initial population
    fitness_scores = []
    for h_ind in population:
        C5 = compute_C5(h_ind)
        fitness_scores.append(C5)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h_ind.copy()
    
    generation = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 0.1 and generation < max_generations:
        # Sort population by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if fitness_scores[0] < best_C5:
            best_C5 = fitness_scores[0]
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > patience:
            # Reset worst performer if no improvement
            worst_idx = len(population) - 1
            h_reset = np.random.rand(n_points)
            h_reset = project_feasible(h_reset)
            population[worst_idx] = h_reset
            fitness_scores[worst_idx] = compute_C5(h_reset)
            no_improvement = 0
        
        # Selection (tournament-based)
        new_population = []
        # Elitism: keep best individuals
        num_elite = int(elitism_rate * pop_size)
        new_population.extend(population[:num_elite])
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[winner_idx]
            
            # Another parent
            if np.random.rand() < crossover_rate:
                tournament_indices2 = np.random.choice(len(population), tournament_size, replace=False)
                tournament_fitness2 = [fitness_scores[i] for i in tournament_indices2]
                winner_idx2 = tournament_indices2[np.argmin(tournament_fitness2)]
                parent2 = population[winner_idx2]
            else:
                parent2 = parent1
            
            # Crossover
            if np.random.rand() < 0.5:
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                offspring = np.where(mask, parent1, parent2)
            else:
                # Two-point crossover
                point1, point2 = sorted(np.random.choice(n_points, 2, replace=False))
                offspring = np.concatenate([parent1[:point1], parent2[point1:point2], parent1[point2:]])
                
            # Apply feasibility projection
            offspring = project_feasible(offspring)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Gaussian mutation
                noise = np.random.normal(0, 0.05, n_points)
                offspring += noise
                offspring = project_feasible(offspring)
                
            # Reset mutation
            if np.random.rand() < 0.1:
                reset_genes = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                for idx in reset_genes:
                    offspring[idx] = np.random.choice([0, 1])
                offspring = project_feasible(offspring)
            
            new_population.append(offspring)
        
        population = new_population[:pop_size]
        fitness_scores = []
        for h_ind in population:
            C5 = compute_C5(h_ind)
            fitness_scores.append(C5)
        
        generation += 1
    
    # Final check
    best_C5 = compute_C5(best_h)
    
    return list(best_h), best_C5, n_points
