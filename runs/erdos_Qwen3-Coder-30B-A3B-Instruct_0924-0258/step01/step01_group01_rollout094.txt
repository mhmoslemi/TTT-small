import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    # Project h to satisfy sum(h) * dx = 1
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Scale to have the right sum
    h = h * target_sum / current_sum
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust again to ensure correct sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find largest element and increment it
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with low resolution
    n_points = 10
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        # Random initial
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Track best ever
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient-based refinement + SA
    max_iters = 2000
    iter_count = 0
    T = 0.05
    cooling = 0.999
    iter_per_temp = 100
    
    while True:
        if time.time() - start_time > budget_s * 0.98:
            break
            
        for _ in range(iter_per_temp):
            if time.time() - start_time > budget_s:
                break
                
            # Try gradient-like update
            h_new = h.copy()
            # Small random perturbations in gradient direction
            gradient = np.zeros_like(h)
            # Approximate gradient by finite differences
            eps = 1e-4
            for i in range(n_points):
                h_pert = h.copy()
                h_pert[i] += eps
                h_pert = project_to_constraints(h_pert, dx)
                c5_pert = evaluate(h_pert, dx)
                
                h_pert[i] -= 2*eps
                h_pert = project_to_constraints(h_pert, dx)
                c5_pert2 = evaluate(h_pert, dx)
                
                gradient[i] = (c5_pert - c5_pert2) / (2*eps)
            
            # Apply gradient
            h_new = h - 0.1 * gradient
            h_new = project_to_constraints(h_new, dx)
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_constraints(h_new, dx)
            
            c5_new = evaluate(h_new, dx)
            
            # Accept with probability
            delta = c5_new - best_c5
            if delta < 0 or np.random.uniform() < np.exp(-delta / T):
                h = h_new.copy()
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
                    
            iter_count += 1
            
        # Cool down
        T *= cooling
        
        # Adaptive resolution
        if n_points < 100 and iter_count > 1000:
            # Interpolate to next resolution
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            T = 0.05
            
        if iter_count >= max_iters:
            break
    
    return best_h.tolist(), best_c5, len(best_h)
