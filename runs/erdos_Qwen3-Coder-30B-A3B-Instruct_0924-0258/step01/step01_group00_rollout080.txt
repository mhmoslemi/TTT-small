import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def perturb_h(h, noise_level=0.1):
    """Add controlled noise to the function"""
    noise = np.random.normal(0, noise_level, len(h))
    return project_to_feasible(h + noise, len(h))

def augmented_lagrangian_optimize(h_start, n_points, dx, time_limit):
    """Optimize using augmented Lagrangian method"""
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Try different optimization strategies
    for iteration in range(500):
        if time() - start_time > time_limit:
            break
            
        # Perturb the current solution
        h_perturbed = perturb_h(h_best, 0.05)
        
        # Simple local optimization step
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        # Use a few gradient steps with SLSQP
        try:
            from scipy.optimize import minimize
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(obj, h_perturbed, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 20, 'ftol': 1e-6})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < c5_best:
                    h_best = h_opt.copy()
                    c5_best = c5_opt
        except:
            pass
            
        # Occasionally do a more global search
        if iteration % 50 == 0:
            h_global = perturb_h(h_best, 0.2)
            c5_global = evaluate_erdos_solution(h_global, dx)
            if c5_global < c5_best:
                h_best = h_global
                c5_best = c5_global

    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with smaller resolutions for exploration
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Multiple random restarts
        h_restart = h_start.copy()
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            h_opt, c5_opt = augmented_lagrangian_optimize(h_restart, n_points, dx, time_per_res / 5)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
            h_restart = perturb_h(h_opt, 0.1)
    
    if best_h is None:
        # Fallback to simplest approach
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = augmented_lagrangian_optimize(h_start, n_points, dx, budget_s)
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
