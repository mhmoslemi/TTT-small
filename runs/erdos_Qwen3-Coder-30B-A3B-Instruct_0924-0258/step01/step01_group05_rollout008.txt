import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec

    # Genetic Algorithm parameters
    pop_size = 50
    tournament_size = 10
    elitism_count = int(0.2 * pop_size)
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_init = np.random.rand(n_points)
        h_init = project_feasible(h_init)
        population.append(h_init)
        
    best_solution = None
    best_C5 = float('inf')
    no_improvement_count = 0
    
    def evaluate_population(pop):
        fitness = []
        for h in pop:
            c5 = compute_C5(h)
            fitness.append(c5)
            if c5 < best_C5:
                best_C5 = c5
                best_solution = h.copy()
                no_improvement_count = 0
        return fitness
    
    generation = 0
    while time.time() - start_time < budget_s - 0.1:
        generation += 1
        # Evaluate population
        fitness = evaluate_population(population)
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if sorted_fitness[0] < best_C5:
            best_C5 = sorted_fitness[0]
            best_solution = sorted_pop[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > patience:
            break
            
        # Elitism - keep best individuals
        new_pop = sorted_pop[:elitism_count]
        
        # Create offspring
        while len(new_pop) < pop_size:
            # Tournament selection
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            parent1_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            parent2_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
            else:
                # Two-point crossover
                point1, point2 = sorted(np.random.choice(n_points, 2, replace=False))
                child = np.concatenate([parent1[:point1], parent2[point1:point2], parent1[point2:]])
                
            # Mutation
            if np.random.rand() < mutation_rate:
                # Gaussian mutation
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = project_feasible(child)
            elif np.random.rand() < 0.5:  # 5% reset mutation
                # Reset some genes
                reset_indices = np.random.choice(n_points, int(0.05 * n_points), replace=False)
                child[reset_indices] = np.random.choice([0, 1], len(reset_indices))
                child = project_feasible(child)
                
            new_pop.append(child)
            
        population = new_pop[:pop_size]
        
    if best_solution is None:
        # Evaluate final population
        final_fitness = evaluate_population(population)
        final_best_idx = np.argmin(final_fitness)
        best_solution = population[final_best_idx]
        best_C5 = final_fitness[final_best_idx]
        
    return list(best_solution), best_C5, n_points
