```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def gaussian_mutation(h, sigma=0.05):
    """Add Gaussian noise to h for genetic diversity"""
    return h + np.random.normal(0, sigma, len(h))

def polynomial_mutation(h, order=2):
    """Apply polynomial adjustment to h"""
    x = np.linspace(-1, 1, len(h))
    coeffs = np.random.normal(0, 0.1, order + 1)
    poly = np.polyval(coeffs, x)
    return h + 0.1 * poly

def initialize_h(n_points):
    """Initialize h values"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), 
                         np.linspace(0, 2, len(h)), h)
    else:
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def optimize_at_resolution(h_start, n_points, dx, time_limit):
    """Optimize at a given resolution with time constraint"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a modified genetic algorithm with gradient-based refinement
    population_size = 20
    generations = 200
    elite_size = int(0.1 * population_size)
    mutation_rate = 0.3
    
    # Initialize population
    population = [project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points) 
                  for _ in range(population_size)]
    
    for gen in range(generations):
        if time() - start_time > time_limit:
            break
            
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new population
        new_population = population[:elite_size].copy()
        
        # Crossover and mutation
        for _ in range(population_size - elite_size):
            # Tournament selection
            idx1, idx2 = np.random.choice(elite_size, 2, replace=False)
            parent1, parent2 = population[idx1], population[idx2]
            
            # Single-point crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            if np.random.rand() < mutation_rate:
                child = gaussian_mutation(child)
            
            # Polynomial mutation
            if np.random.rand() < mutation_rate:
                child = polynomial_mutation(child)
            
            new_population.append(project_to_feasible(child, n_points))
        
        population = new_population
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Multi-resolution approach
    resolutions = [30, 60, 90, 120]  # Keep at reasonable size
    
    best_h = initialize_h(resolutions[0])
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / len(best_h))
    best_n_points = len(best_h)
    
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if necessary
        if len(best_h) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(best_h)), 
                                best_h)
            h_start = project_to_feasible(h_interp, n_points)
        else:
            h_start = best_h.copy()
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Optimize
        h_opt, c5_opt = optimize_at_resolution(h_start, n_points, dx, time_per_res)
        
        # Update best
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n_points = n_points
        
        # Make a better start for next resolution
        best_h = project_to_feasible(h_opt, n_points)
    
    return (list(best_h), best_c5, best_n_points)
```
