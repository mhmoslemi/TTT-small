import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    n_particles = 100
    inertia_start = 0.7
    inertia_end = 0.4
    c1 = 2.0
    c2 = 2.0
    max_speed = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for _ in range(n_particles):
        # Start near initial solution with small random perturbation
        particle_h = h + np.random.normal(0, 0.05, n_points)
        particle_h = project_h(particle_h)
        particles.append(particle_h.copy())
        velocities.append(np.zeros(n_points))
        
        # Evaluate initial position
        fitness = compute_C5(particle_h)
        pbest_positions.append(particle_h.copy())
        pbest_values.append(fitness)
    
    # Global best
    best_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[best_idx].copy()
    gbest_value = pbest_values[best_idx]
    
    # PSO main loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * iteration / 1000
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocity = (
                inertia * velocities[i] +
                c1 * r1 * (pbest_positions[i] - particles[i]) +
                c2 * r2 * (gbest_position - particles[i])
            )
            
            # Clip velocity
            velocity = np.clip(velocity, -max_speed, max_speed)
            velocities[i] = velocity
            
            # Update position
            particles[i] += velocity
            particles[i] = project_h(particles[i])
            
            # Evaluate fitness
            fitness = compute_C5(particles[i])
            
            # Update personal best
            if fitness < pbest_values[i]:
                pbest_values[i] = fitness
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if fitness < gbest_value:
                    gbest_value = fitness
                    gbest_position = particles[i].copy()
                    
    # Return best solution found
    final_h = gbest_position.copy()
    final_C5 = compute_C5(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
