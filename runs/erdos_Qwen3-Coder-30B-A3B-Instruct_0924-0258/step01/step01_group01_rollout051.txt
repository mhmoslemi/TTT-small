import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution schedule
    resolutions = [10, 20, 40, 70]
    best_h = None
    best_c5 = float('inf')
    
    for n_points in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Initialize with initial_h_values or random
        if 'initial_h_values' in globals() and n_points == 10:
            h = np.array(initial_h_values)
            h = h[:n_points]
        else:
            h = np.random.uniform(0, 1, n_points)
            h = project_to_constraints(h, dx)
            
        # Particle Swarm Optimization
        n_particles = 100
        particles = []
        velocities = []
        pbest = []
        pbest_values = []
        for _ in range(n_particles):
            if 'initial_h_values' in globals() and n_points == 10:
                h_part = np.array(initial_h_values)
                h_part = h_part[:n_points]
            else:
                h_part = np.random.uniform(0, 1, n_points)
                h_part = project_to_constraints(h_part, dx)
            particles.append(h_part.copy())
            velocities.append(np.zeros(n_points))
            pbest.append(h_part.copy())
            pbest_values.append(evaluate(h_part, dx))
            
        # Global best
        gbest_idx = np.argmin(pbest_values)
        gbest = pbest[gbest_idx].copy()
        gbest_value = pbest_values[gbest_idx]
        
        # PSO parameters
        omega = 0.7
        phi_p = 0.5
        phi_g = 0.5
        v_max = 0.05
        
        max_iter = min(5000, int(150000 / n_points))
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.random(2)
                velocities[i] = (omega * velocities[i] + 
                                phi_p * r1 * (pbest[i] - particles[i]) +
                                phi_g * r2 * (gbest - particles[i]))
                # Clip velocity
                velocities[i] = np.clip(velocities[i], -v_max, v_max)
                # Update position
                particles[i] += velocities[i]
                # Boundary constraints
                particles[i] = np.clip(particles[i], 0, 1)
                # Reproject
                particles[i] = project_to_constraints(particles[i], dx)
                
                # Evaluate
                value = evaluate(particles[i], dx)
                # Update personal best
                if value < pbest_values[i]:
                    pbest[i] = particles[i].copy()
                    pbest_values[i] = value
                    # Update global best
                    if value < gbest_value:
                        gbest = particles[i].copy()
                        gbest_value = value
                        
        # Gradient descent refinement
        h_refined = gbest.copy()
        learning_rate = 0.01
        max_grad_iter = 100
        for _ in range(max_grad_iter):
            if time.time() - start_time > budget_s:
                break
            # Approximate gradient
            eps = 1e-5
            grad = np.zeros(n_points)
            base_val = evaluate(h_refined, dx)
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = project_to_constraints(h_plus, dx)
                val_plus = evaluate(h_plus, dx)
                grad[i] = (val_plus - base_val) / eps
                
            # Update with gradient
            h_refined -= learning_rate * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = project_to_constraints(h_refined, dx)
            
            # Early stopping if improvement is small
            new_val = evaluate(h_refined, dx)
            if abs(new_val - base_val) < 1e-8:
                break
                
        # Update global best
        final_val = evaluate(h_refined, dx)
        if final_val < best_c5:
            best_h = h_refined.copy()
            best_c5 = final_val
            
    if best_h is None:
        # Fallback to initial if no progress
        n_points = 10
        dx = 2.0 / n_points
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
            best_h = best_h[:n_points]
        else:
            best_h = np.random.uniform(0, 1, n_points)
            best_h = project_to_constraints(best_h, dx)
        best_c5 = evaluate(best_h, dx)
        
    return best_h.tolist(), best_c5, len(best_h)
