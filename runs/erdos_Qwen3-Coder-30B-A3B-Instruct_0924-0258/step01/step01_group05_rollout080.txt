import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations: h * (1 - h[k:]) for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Genetic Algorithm
    pop_size = 50
    tournament_size = 10
    elitism = int(0.2 * pop_size)
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_rand = np.random.rand(n_points)
        h_rand = project_feasible(h_rand)
        population.append(h_rand)
    
    def evaluate_population(pop):
        fitness = []
        for h in pop:
            fit = compute_C5(h)
            fitness.append(fit)
        return np.array(fitness)
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    generation = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 0.1:
        generation += 1
        if no_improvement > patience:
            # Reset some population
            for i in range(0, pop_size, 5):
                population[i] = np.random.rand(n_points)
                population[i] = project_feasible(population[i])
            no_improvement = 0
        
        # Evaluate fitness
        fitness = evaluate_population(population)
        best_in_gen = np.min(fitness)
        best_idx = np.argmin(fitness)
        
        if best_in_gen < best_C5:
            best_C5 = best_in_gen
            best_h = population[best_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size)
            tournament_fitness = fitness[tournament]
            winner = tournament[np.argmin(tournament_fitness)]
            selected.append(population[winner].copy())
        
        # Elitism
        elite_idx = np.argsort(fitness)[:elitism]
        new_pop = [population[i].copy() for i in elite_idx]
        
        # Crossover and mutation
        while len(new_pop) < pop_size:
            if np.random.rand() < crossover_rate:
                p1, p2 = np.random.choice(pop_size, 2, replace=False)
                h1, h2 = selected[p1], selected[p2]
                
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, h1, h2)
                
                # Apply feasibility
                child = project_feasible(child)
                new_pop.append(child)
            else:
                # Mutation only
                parent = selected[np.random.randint(0, pop_size)]
                child = parent.copy()
                
                # Gaussian mutation
                mask = np.random.rand(n_points) < mutation_rate
                noise = np.random.normal(0, 0.05, n_points)
                child[mask] += noise[mask]
                
                # Apply feasibility
                child = project_feasible(child)
                new_pop.append(child)
        
        population = new_pop[:pop_size]  # Keep pop_size members
        
    # Return best solution
    return list(best_h), best_C5, n_points
