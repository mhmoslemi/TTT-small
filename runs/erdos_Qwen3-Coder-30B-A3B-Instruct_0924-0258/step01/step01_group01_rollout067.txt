import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 10
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Use a more structured refinement approach
    max_iter = 10000
    iter_count = 0
    improvement_threshold = 0.01
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        iter_count += 1
        # Perturb h slightly to explore neighborhood
        noise = np.random.normal(0, 0.01, n_points)
        h_new = h + noise
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_constraints(h_new, dx)
        
        c5_new = evaluate(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Accept with some probability to escape local minima
            accept_prob = np.exp(-(c5_new - best_c5) / (0.01 + 0.01 * best_c5))
            if np.random.random() < accept_prob:
                h = h_new.copy()
                
    return best_h.tolist(), best_c5, len(best_h)
