import numpy as np
import time
from scipy.signal import correlate

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def compute_gradient(h, dx, k_max):
    # Compute the correlation with (1-h) and identify the gradient direction
    # based on maximal contribution
    corr = correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    # Find which lag k contributes most
    n = len(h)
    # corr is of length 2*n - 1
    k = max_idx - (n - 1)
    # Invert k to get the actual lag from the beginning
    return k

def update_h(h, lr, dx, k_max):
    # Update h in direction of gradient (-k_max) to reduce C5
    n = len(h)
    # The gradient is based on cross-correlation at k_max
    corr = correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    
    # Derivative w.r.t. h_i is (1-h_{i+k}) when k>0
    # We do gradient descent to reduce the maximum correlation
    
    grad = np.zeros(n)
    # Compute explicit gradient
    for i in range(n):
        for j in range(n):
            if j == (i + k) % n:
                grad[i] += -(1 - h[j])
    
    # Simple gradient update, with clipping and projection
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0, 1)
    h_new = project_to_constraints(h_new, dx)
    return h_new

def compute_c5(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_low = 10
    n_high = 70
    n_points = n_low
    dx = 2.0 / n_points
    max_grad_iterations = 100
    
    if 'initial_h_values' in globals():

        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Learning rate for gradient descent
    lr = 0.02
    iter_per_temp = 200
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for _ in range(iter_per_temp):
            if time.time() - start_time > budget_s:
                break
                
            h_candidate = update_h(h, lr, dx, None)
            c5_candidate = compute_c5(h_candidate, dx)
            
            if c5_candidate < best_c5:
                h = h_candidate
                best_c5 = c5_candidate
                best_h = h.copy()
            elif np.random.rand() < 0.1:
                # Occasionally accept worse moves sometimes to avoid local minima
                h = h_candidate
                best_c5 = c5_candidate
                
        # Adaptive learning rate
        if best_c5 < 0.381:
            lr *= 0.95
        if lr < 1e-5:
            lr = 1e-5
            
        # If reached high resolution, break
        if n_points >= n_high:
            break
            
        # Check for resolution upgrade
        if n_points < n_high:
            # Interpolate to higher resolution
            new_n = 2 * n_points
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, new_n)
            h_new = np.interp(x_new, x_old, h)
            n_points = new_n
            dx = 2.0 / n_points
            h = project_to_constraints(h_new, dx)
    
    return best_h.tolist(), best_c5, len(best_h)
