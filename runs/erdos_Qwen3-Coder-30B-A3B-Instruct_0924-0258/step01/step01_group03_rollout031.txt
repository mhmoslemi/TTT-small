import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint while keeping values in [0,1]."""
    h = np.array(h, dtype=float)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to target sum
    h = h * target_sum / current_sum
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Re-normalize
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

    h = h0.copy()
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Parameters
    learning_rate = 0.01
    patience = 50
    patience_counter = 0
    best_c5 = float('inf')
    best_h = None
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s:
        # Evaluate current C5
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Check for recovery
        if patience_counter >= patience:
            # Add small random noise
            noise = np.random.uniform(-1e-3, 1e-3, n_points)
            h += noise
            h = project_h(h, target_sum)
            patience_counter = 0
        
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        
        # Compute gradient for the active shift
        grad = np.zeros(n_points)
        # The active shift k corresponds to the maximum in correlation
        # We must compute the gradient for that particular shift
        # We find the indices involved: for shift k, h[i] contributes to correlation at position i+k
        # The gradient element j is (1 - 2 * h[j-k]) where we use wrap-around
        # Create the gradient vector
        for i in range(n_points):
            j = (i - max_k) % n_points
            grad[i] = 1 - 2 * h[j]
        
        # Update h
        h -= learning_rate * grad
        
        # Project back to feasible set
        h = project_h(h, target_sum)
        
        # Adaptive learning rate
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.9
        
        iteration += 1
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        # Fallback to initial
        final_h = project_h(h0, target_sum)
        final_c5 = evaluate(final_h, dx)
        return (final_h.tolist(), float(final_c5), n_points)
