import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():

        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Initialize
    h = h0.copy().astype(float)
    dx = 2.0 / n_points
    
    # Ensure sum constraint
    h = h * (n_points / 2.0) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 10000
    improvement_patience = 50
    gradient_smoothing = 5
    
    # Gradient tracking
    grad_history = []
    
    for iter_count in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Evaluate overlaps for all shifts
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(np.abs(corr))
        max_val = corr[k_max]
        
        # Compute gradient for the maximizing shift
        gradient = np.zeros_like(h)
        if k_max < len(h):
            # Positive shift
            for i in range(len(h)):
                j = i + k_max
                if j < len(h):
                    gradient[i] += (1 - h[j])
        else:
            # Negative shift (should be handled differently)
            shift = k_max - (2 * len(h) - 2)
            for i in range(len(h)):
                j = i + shift
                if j >= 0 and j < len(h):
                    gradient[i] += (1 - h[j])
        
        # Smooth gradient
        if len(grad_history) < gradient_smoothing:
            grad_history.append(gradient)
        else:
            grad_history = grad_history[1:] + [gradient]
        
        avg_grad = np.mean(grad_history, axis=0)
        
        # Update h using gradient
        h = h - learning_rate * avg_grad
        
        # Project to feasible set
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h = h * (n_points / 2.0) / current_sum
        
        # Evaluate current solution
        current_c5 = evaluate(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            improvement_patience = 50
        
        # Adaptive learning rate
        improvement_patience -= 1
        if improvement_patience <= 0:
            learning_rate *= 0.9
            improvement_patience = 50
            
            # Add noise to escape local minima
            noise = np.random.uniform(-1e-3, 1e-3, len(h))
            h = h + noise
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            h = h * (n_points / 2.0) / current_sum
    
    return (best_h.tolist(), float(best_c5), n_points)
