import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def adaptive_perturb(h, sigma):
    """Perturb h with adaptive noise"""
    return project_to_feasible(h + np.random.normal(0, sigma, len(h)), len(h))

def smooth_objective(h, dx):
    """Smooth objective approximating the maximum overlap"""
    corr = np.correlate(h, 1 - h, mode='full')
    # Make it smooth but still approximates the max
    return np.sum(corr**2) * dx**2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with the initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        # Default initialization
        n0 = 30
        h_start = np.ones(n0) / 2
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Try different resolutions
    resolutions = [25, 30, 35, 40, 45]
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n_points
        
        # Interpolate if needed
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Use a more aggressive optimization, multiple restarts
        best_h_level = h_start.copy()
        best_c5_level = evaluate_erdos_solution(h_start, dx)
        
        # Multiple restarts, each with SLSQP
        for attempt in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            # Perturb the starting point
            h_perturbed = adaptive_perturb(h_start, 0.05)
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            
            # Define objective
            def obj(h):
                return smooth_objective(h, dx)
            
            # Define gradient (approximate)
            def grad(h):
                eps = 1e-6
                g = np.zeros_like(h)
                base_val = obj(h)
                for i in range(len(h)):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    g[i] = (obj(h_plus) - base_val) / eps
                return g
            
            # Constraints
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(obj, h_perturbed, method='SLSQP', 
                               bounds=bounds, constraints=cons, jac=grad, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5_level:
                        best_c5_level = c5_opt
                        best_h_level = h_opt.copy()
            except:
                pass
            
            # Early termination
            if time() - start_time > budget_s * 0.9:
                break
        
        if best_c5_level < best_c5:
            best_h = best_h_level.copy()
            best_c5 = best_c5_level
            best_n_points = n_points
            
        h_start = best_h_level.copy()
    
    return (list(best_h), best_c5, best_n_points)
