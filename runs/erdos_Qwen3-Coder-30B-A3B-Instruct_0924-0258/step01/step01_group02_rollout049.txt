import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    T = 10 * best_c5  # Initial temperature
    cooling_rate = 0.95
    min_T = 1e-6
    iterations_per_T = 500
    
    iteration = 0
    while T > min_T:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # For each temperature, run multiple iterations
        for _ in range(iterations_per_T):
            # Make a random perturbation
            h_test = h.copy()
            
            # Select random indices to perturb
            n_perturb = max(1, n_points // 10)  # Perturb ~10% of points
            indices = np.random.choice(n_points, size=n_perturb, replace=False)
            
            # Add small random changes
            for i in indices:
                perturbation = np.random.normal(0, 0.05)
                h_test[i] += perturbation
            
            # Project back to feasible region
            h_test = project_h(h_test)
            
            # Compute new energy
            new_c5 = compute_c5(h_test)
            
            # Accept or reject the move
            delta = new_c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_test
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
        
        # Cool down
        T *= cooling_rate
        iteration += 1
        
        # Occasionally suppress temperature to avoid stagnation
        if iteration % 10 == 0 and T > 1e-4:
            T = max(T, 1e-4)
            
    return (best_h.tolist(), best_c5, n_points)
