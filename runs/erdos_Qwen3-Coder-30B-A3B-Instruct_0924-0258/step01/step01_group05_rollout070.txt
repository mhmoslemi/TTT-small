import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_gradients(h_vec, k_max):
        # Compute gradient for specific lag k_max
        grad = np.zeros_like(h_vec)
        n = len(h_vec)
        for i in range(n):
            # Gradient contribution from h[i] affecting correlation at lag k_max
            # h[i] contributes to term h[i] * (1 - h[i + k_max])  
            # Particle derivative: (1 - h[i + k_max]) if i + k_max < n, else (1 - h[i + k_max - n])
            j = (i + k_max) % n
            grad[i] = -(1 - h_vec[j])
            # h[i] also contributes to term h[i] * (1 - h[i - k_max])
            j_neg = (i - k_max) % n
            grad[i] -= h_vec[j_neg]
        return grad
        
    def optimize_step(h_vec, alpha=0.1, max_iter=100):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        unchanged_count = 0
        alpha_decay = 0.5
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Compute all correlations and get the maximum lag
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_lag_idx = np.argmax(corr)
            k_max = max_lag_idx - (len(h_current) - 1)
            
            # Compute gradient with respect to k_max
            grad = compute_gradients(h_current, k_max)
            
            # Try gradient descent
            new_h = h_current - alpha * grad
            new_h = project_feasible(new_h)
            new_C5 = compute_C5(new_h)
            
            if new_C5 < best_C5:
                h_current = new_h.copy()
                best_C5 = new_C5
                unchanged_count = 0
            else:
                unchanged_count += 1
                if unchanged_count > 10:
                    alpha *= alpha_decay
                    unchanged_count = 0
                    
        return h_current, best_C5
    
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Local optimization loop
    while time.time() - start_time < budget_s - 0.1:
        # Try several local optimizations with different step sizes
        alpha_options = [0.1, 0.05, 0.01]
        
        for alpha in alpha_options:
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Perturb current point slightly and optimize
            h_perturbed = h_best + np.random.normal(0, 0.02, n_points)
            h_perturbed = project_feasible(h_perturbed)
            
            # Perform gradient descent from this point
            h_opt, C5_opt = optimize_step(h_perturbed, alpha=alpha, max_iter=50)
            
            if C5_opt < C5_best:
                C5_best = C5_opt
                h_best = h_opt.copy()
                
    # Final check
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
