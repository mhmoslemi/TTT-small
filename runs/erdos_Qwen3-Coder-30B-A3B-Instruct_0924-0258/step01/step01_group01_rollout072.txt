import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def compute_gradients(h, dx):
    # We compute gradients for the maximum correlation value
    # and update h to decrease it
    corr = correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    # The correlation is of length 2*n-1
    n = len(h)
    k = max_idx - (n - 1)  # actual lag
    # compute gradient w.r.t. each h[i]
    # The derivative of correlation w.r.t. h[i] at lag k
    grad = np.zeros_like(h)
    # For any position i, if we look at the correlation at lag k:
    # corr[k] = sum_{j} h[j] * (1 - h[j+k])
    # But the correlation is symmetric, so we just consider positive lags:
    for i in range(n):
        if 0 <= i + k < n:
            grad[i] += (1 - h[i + k])
        if 0 <= i - k < n:
            grad[i] += (1 - h[i - k])
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_low = 10
    n_high = 70
    n_points = n_low
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
        
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    iter_count = 0
    max_iter_per_res = 1000
    
    # Optimizer parameters
    learning_rate = 0.05
    patience = 20
    patience_counter = 0
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for i in range(max_iter_per_res):
            if time.time() - start_time > budget_s:
                break
                
            iter_count += 1
            
            # Compute current max correlation (C5)
            c5_old = evaluate(h, dx)
            
            # Compute gradient
            grad = compute_gradients(h, dx)
            
            # Update h
            h_new = h - learning_rate * grad
            
            # Clip to [0, 1]
            h_new = np.clip(h_new, 0, 1)
            
            # Project back to correct integral
            h_new = project_to_constraints(h_new, dx)
            
            # Evaluate new point
            c5_new = evaluate(h_new, dx)
            
            # Accept move if it's better or with some probability
            if c5_new < c5_old:
                h = h_new
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
                    patience_counter = 0
                else:
                    patience_counter += 1
            else:
                patience_counter += 1
            
            if patience_counter >= patience:
                break
                
        # Possibly increase resolution
        if n_points < n_high and time.time() - start_time < budget_s * 0.8:
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), best_h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            best_h = h.copy()
            best_c5 = evaluate(h, dx)
            patience_counter = 0
        else:
            break
    
    return best_h.tolist(), best_c5, len(best_h)
