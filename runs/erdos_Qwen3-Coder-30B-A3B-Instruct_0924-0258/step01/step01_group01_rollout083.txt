import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def smooth_max(x, beta=100):
    # Smooth approximation to max
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))

def compute_smooth_c5(h, dx, beta=100):
    # Compute smooth maximum of correlations
    corr = correlate(h, 1-h, mode='full')
    # Flatten correlation (middle part is the one we care about)
    n = len(h)
    # Take middle elements corresponding to valid overlaps
    start_idx = n - 1
    end_idx = 2 * n - 2
    corr_vals = corr[start_idx:end_idx]
    return smooth_max(corr_vals, beta) * dx

def grad_smooth_c5(h, dx, beta=100):
    # Gradient of smooth max approximation
    n = len(h)
    # We use FFT; get convolution result and its gradient
    # Using computational gradient from correlation of h with (1-h)
    # This is a placeholder for numerical gradient - better to compute analytically
    # But for right now we'll use finite difference approximation
    eps = 1e-6
    grad = np.zeros_like(h)
    
    # Compute correlation for original h  
    corr_full = correlate(h, 1-h, mode='full')
    corr_vals = corr_full[n-1:2*n-1]
    
    # Find max index
    max_idx = np.argmax(corr_vals)
    max_val = corr_vals[max_idx]
    
    # Approximate gradient using finite differences
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_constraints(h_plus, dx)
        corr_plus = correlate(h_plus, 1-h_plus, mode='full')
        corr_plus_vals = corr_plus[n-1:2*n-1]
        val_plus = smooth_max(corr_plus_vals, beta) * dx
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_constraints(h_minus, dx)
        corr_minus = correlate(h_minus, 1-h_minus, mode='full')
        corr_minus_vals = corr_minus[n-1:2*n-1]
        val_minus = smooth_max(corr_minus_vals, beta) * dx
        
        grad[i] = (val_plus - val_minus) / (2 * eps)
        
    return grad

def optimize_step(h, dx, lr=1e-3, max_iter=20):
    # Coordinate descent with projection
    n = len(h)
    for _ in range(max_iter):
        grad = grad_smooth_c5(h, dx)
        h_new = h - lr * grad
        h_new = project_to_constraints(h_new, dx)
        # Check if we improved
        old_val = compute_smooth_c5(h, dx)
        new_val = compute_smooth_c5(h_new, dx)
        if new_val < old_val:
            h = h_new
        else:
            break
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with resolution consistent with parent construction
    n_low = 20
    n_high = 150
    n_points = n_low
    dx = 2.0 / n_points
    
    # Initialize from parent if available, otherwise random
    if 'initial_h_values' in globals():
        # Interpolate parent construction to current resolution
        h_parent = np.array(initial_h_values)
        x_parent = np.linspace(0, 2, len(h_parent))
        x_current = np.linspace(0, 2, n_points)
        h = np.interp(x_current, x_parent, h_parent)
        h = project_to_constraints(h, dx)
    else:
        # Random initialization
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    iter_count = 0
    lr = 5e-3
    
    while True:
        if time.time() - start_time > budget_s * 0.98:
            break
            
        # Do optimization step
        h = optimize_step(h, dx, lr=lr, max_iter=5)
        c5 = evaluate(h, dx)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        iter_count += 1
        
        # Adjust learning rate
        if iter_count % 100 == 0:
            lr *= 0.95
            if lr < 1e-5:
                lr = 1e-5
                
        # Possibly increase resolution
        if iter_count % 300 == 0 and n_points < n_high:
            n_points *= 2
            if n_points > n_high:
                n_points = n_high
            dx = 2.0 / n_points
            h_interp = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points//2), h)
            h = h_interp
            h = project_to_constraints(h, dx)
            best_h = h.copy()
            best_c5 = evaluate(h, dx)
        
        # Break if too much time
        if time.time() - start_time > budget_s:
            break
    
    return best_h.tolist(), best_c5, len(best_h)
