import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def convex_relaxation(h, n_points, dx):
    """Convex relaxation for the optimization problem"""
    # Create a new variable for the maximum overlap
    max_val = cp.Variable()
    # Convolution constraints
    correlations = []
    for k in range(2 * n_points - 1):
        corr_sum = 0
        for i in range(n_points):
            if 0 <= i + k < n_points:
                corr_sum += h[i] * (1 - h[i + k])
        correlations.append(corr_sum * dx)
    
    # Constraint: max_val >= all correlations
    constraints = [max_val >= corr for corr in correlations]
    objective = cp.Minimize(max_val)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        return prob.value
    except:
        return float('inf')

def genetic_mutation(h, n_points):
    """Apply polynomial fit mutation"""
    # Create polynomial fit to the current function
    x = np.linspace(0, 2, n_points)
    # Fit a polynomial of degree 3 to current function
    coeffs = np.polyfit(x, h, 3)
    # Generate new curve using polynomial + perturbation
    h_new = np.polyval(coeffs, x) + np.random.normal(0, 0.01, n_points)
    return h_new

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    # Multi-scale approach
    scales = [15, 30, 45, 60]
    scales = [s for s in scales if s <= n_points]
    if not scales:
        scales = [n_points]
    
    for res in scales:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Adjust resolution
        if res != n_points:
            # Interpolate
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            h_interp = np.interp(x_new, x_old, best_h)
            h_current = project_to_feasible(h_interp, res)
            n_points = res
        else:
            h_current = best_h.copy()
            
        dx = 2.0 / n_points
        candidates = []
        start_candidates = []
        
        # Generate candidates via genetic approach
        for _ in range(10):
            if time() - start_time > budget_s * 0.95:
                break
            # Clone and perturb
            h_mutated = genetic_mutation(h_current, n_points)
            h_projected = project_to_feasible(h_mutated, n_points)
            start_candidates.append(h_projected.copy())
        
        # Improve each candidate via optimization (SLSQP)
        for h_init in start_candidates:
            try:
                # Use local optimization to improve each
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_func(h):
                    return np.sum(h) - n_points / 2
                
                cons = {'type': 'eq', 'fun': constraint_func}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(obj, h_init, method='SLSQP',

                if res.success:

                    c5 = evaluate_erdos_solution(h_opt, dx)
                    candidates.append((h_opt.copy(), c5))

        # Select best from candidates
        if candidates:
            best_candidate = min(candidates, key=lambda x: x[1])
            if best_candidate[1] < best_c5:

                best_c5 = best_candidate[1]
