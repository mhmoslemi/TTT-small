import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    n_particles = 100
    w_start = 0.7
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    max_speed = 0.1
    
    # Initialize particles
    particles = np.array([h + np.random.normal(0, 0.01, n_points) for _ in range(n_particles)])
    particles = np.array([project_h(p) for p in particles])
    
    # Personal bests
    pbest = particles.copy()
    pbest_scores = np.array([compute_C5(p) for p in particles])
    
    # Global best
    global_best_idx = np.argmin(pbest_scores)
    gbest = pbest[global_best_idx].copy()
    gbest_score = pbest_scores[global_best_idx]
    
    # Velocity initialization
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Update inertia weight linearly
        w = w_start - (w_start - w_end) * iteration / 1000
        
        # Update particles
        for i in range(n_particles):
            # Random numbers
            r1, r2 = np.random.random(2)
            
            # Update velocity
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_speed, max_speed)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_h(particles[i])
            
            # Evaluate
            score = compute_C5(particles[i])
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = particles[i].copy()
                
                # Update global best
                if score < gbest_score:
                    gbest_score = score
                    gbest = particles[i].copy()
        
    return (gbest.tolist(), gbest_score, n_points)
