import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Set up
    dx = 2.0 / n_points
    h = h0.copy().astype(float)
    h = h * (n_points / 2) / np.sum(h)  # Normalize to sum = n_points/2
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent with adaptive learning rate
    eta = 0.01
    patience = 0
    patience_limit = 50
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Compute cross-correlations
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(np.abs(corr))
        
        # Determine the shift that maximizes overlap
        # We compute both positive and negative shifts relative to center
        shift = k_max - (len(corr) - 1) // 2
        
        # Compute gradient
        gradient = np.zeros_like(h)
        if shift >= 0:
            # h[i] contributes to (1 - h[i+shift]) in the correlation
            for i in range(len(h)):
                j = i + shift
                if j < len(h):
                    gradient[i] += (1 - h[j])
        else:
            shift = -shift
            # h[i] contributes to (1 - h[i-shift]) in the correlation
            for i in range(len(h)):
                j = i - shift
                if j >= 0:
                    gradient[i] += (1 - h[j])
        
        # Update h
        h = h - eta * gradient
        
        # Projection: clip to [0,1]
        h = np.clip(h, 0, 1)
        
        # Projection: normalize to sum = n_points/2
        h_sum = np.sum(h)
        if abs(h_sum) > 1e-12:
            h = h * (n_points / 2) / h_sum
        
        # Evaluate
        c5 = evaluate(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
            last_improvement = iteration
        else:
            patience += 1
            
        # Adaptive learning rate
        if patience > 10 and iteration > last_improvement + 20:
            eta *= 0.9
            patience = 0
            
        # Occasionally add noise to escape local minima
        if patience >= patience_limit:
            noise = np.random.uniform(-1e-3, 1e-3, len(h))
            h = h + noise
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum) > 1e-12:
                h = h * (n_points / 2) / h_sum
            patience = 0
        
    return (best_h.tolist(), float(best_c5), n_points)
