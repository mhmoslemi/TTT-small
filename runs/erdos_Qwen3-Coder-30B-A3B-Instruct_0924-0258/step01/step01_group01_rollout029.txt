import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def gradient_step(h, dx, lr=0.01):
    # Analytical gradient of C5 with respect to h
    # C5 = max_k int h(x)(1 - h(x+k)) dx
    # We compute gradient at the current best k
    corr = correlate(h, 1-h, mode='full')
    best_k = np.argmax(corr)
    # The gradient of correlation at k is approximated by
    # d/dh_i [ sum_j h_j (1 - h_{j+k}) ] = (1 - 2 h_i) * delta_{j, i} * dx
    
    # Simplified approach: since it's a complicated derivative, we just backpropagate
    # Propagate the derivative from the point of maximum correlation with 1-h
    n = len(h)
    grad = np.zeros(n)
    # Find where the max correlation occurs
    max_corr_idx = np.argmax(corr)
    
    # Create a step function for 1-h
    one_minus_h = 1 - h
    # Correlation with 1-h maps to indices in overlap
    # For indices j, i such that j = i + k, k = max_corr_idx - n + 1, add -1
    # In full mode, we are looking at -n+1 to n-1 as shifts
    
    # Compute more accurate approx of gradient
    # Taking one step of gradient by computing 1-d gradient of 1-h convolution
    idx = max_corr_idx
    # Forward and backward truncations:
    for i in range(n):
        # gradient contribution from the maximum correlation at idx
        # derivative of h[i] with respect to overlap involving h[i] and 1-h[i+k] 
        # essay for i, check where h[i] contributes in index 
        if idx - n + 1 <= 0 and 0 <= idx - n + 1 + i < n:  # direct shift
            grad[i] += (-1.0) * dx
        elif idx - n + 1 + i >= 0 and idx - n + 1 + i < n:
            grad[i] += (-1.0) * dx
    
    # Simple right now, diff might be more complex
    h_new = h - lr * grad
    return h_new

def optimize_with_pso(h, dx, n_particles=100, max_iter=1000):
    # Particle Swarm Optimization for h
    n = len(h)
    # Initialize particles
    particles = []
    velocities = []
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            p = np.array(initial_h_values[:n]).copy()
        else:
            p = np.random.uniform(0, 1, n)
        
        p = project_to_constraints(p, dx)
        particles.append(p)
        velocities.append(np.random.uniform(-0.05, 0.05, n))
    
    # Global and particle bests
    gbest = h.copy()
    gbest_val = evaluate(h, dx)
    pbest = [p.copy() for p in particles]
    pbest_val = [evaluate(p, dx) for p in particles]
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.05
    
    for it in range(max_iter):
        for i in range(n_particles):
            # Update velocity with bounds
            r1, r2 = np.random.rand(n), np.random.rand(n)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = project_to_constraints(particles[i], dx)
            
            # Evaluate and update pbest if needed
            val = evaluate(particles[i], dx)
            if val < pbest_val[i]:
                pbest_val[i] = val
                pbest[i] = particles[i].copy()
                if val < gbest_val:
                    gbest_val = val
                    gbest = particles[i].copy()
    
    return gbest, gbest_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    resolution_schedule = [10, 20, 40, 70]
    n_points = resolution_schedule[0]
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values[:n_points])
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Track best
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Multi-resolution optimization with PSO & gradient descent
    for res in resolution_schedule:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to current resolution
        if res > n_points:
            h_long = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = res
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
        
        # PSO optimization
        pso_h, pso_c5 = optimize_with_pso(h, dx, n_particles=100, max_iter=500)
        
        # Gradient descent refinement
        h_refined = pso_h.copy()
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            h_refined = gradient_step(h_refined, dx, lr=0.008)
            h_refined = project_to_constraints(h_refined, dx)
            val = evaluate(h_refined, dx)
            if val < best_c5:
                best_c5 = val
                best_h = h_refined.copy()
        
        h = h_refined.copy()
        
    # Final evaluation with full PSO if space
    if time.time() - start_time < budget_s * 0.95:
        final_h, final_c5 = optimize_with_pso(h, dx, n_particles=100, max_iter=300)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()
    
    return best_h.tolist(), best_c5, len(best_h)
