import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with small resolution
    n_points = 10
    dx = 2.0 / n_points
    
    # Initialize from previous construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
    
    h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # ADMM parameters
    rho = 1.0
    max_iter = 1000
    tol = 1e-6
    
    # Resize refinement
    resolutions = [10, 20, 40, 70]
    res_idx = 0
    iter_per_res = 100
    
    for res_idx in range(len(resolutions)):
        n_points = resolutions[res_idx]
        dx = 2.0 / n_points
        
        if res_idx > 0:
            # Interpolate to new resolution
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, len(h)), h)
            h = h_interp
            h = project_to_constraints(h, dx)
        
        # Precompute some constants
        full_corr_len = 2 * n_points - 1
        
        # ADMM iterations
        for iter_count in range(iter_per_res):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Update h using gradient descent on cross-correlation
            # Compute correlation of h with (1-h) 
            corr_h_1h = correlate(h, 1-h, mode='full')
            # Approximate gradient of correlation w.r.t h
            # We can get an approximate gradient by finite differences
            # Using subgradient techniques for max operator
            max_k = np.argmax(corr_h_1h)
            # Essentially, we want gradient of the maximum correlation term
            # Use subgradient approach: take derivative of correlation at max_k
            # Compute gradient w.r.t. h (approximate)
            
            # Simple gradient approximation: this is a heuristic based 
            # on cómo the correlation changes when we slightly perturb h[i]
            grad = np.zeros_like(h)
            for i in range(n_points):
                eps = 1e-6  # perturbation
                h_plus = h.copy()
                h_minus = h.copy()
                h_plus[i] += eps
                h_minus[i] -= eps  
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                corr_plus = correlate(h_plus, 1-h_plus, mode='full')
                corr_minus = correlate(h_minus, 1-h_minus, mode='full')
                grad[i] = (corr_plus[max_k] - corr_minus[max_k]) / (2*eps)
            
            # Update h
            step_size = 0.001
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_constraints(h_new, dx)
            h = h_new
            
            # Check convergence
            c5_new = evaluate(h, dx)
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
                
        if time.time() - start_time > budget_s:
            break
            
    return best_h.tolist(), best_c5, len(best_h)
