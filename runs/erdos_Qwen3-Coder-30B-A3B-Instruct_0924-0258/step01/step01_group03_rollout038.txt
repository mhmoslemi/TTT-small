import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        idx = np.argsort(h)
        for i in range(len(idx)-1, -1, -1):
            if diff <= 0:
                break
            j = idx[i]
            if h[j] < 1:
                incr = min(diff, 1.0 - h[j])
                h[j] += incr
                diff -= incr
    else:
        # Need to decrease values
        idx = np.argsort(h)
        for i in range(len(idx)):
            if diff >= 0:
                break
            j = idx[i]
            if h[j] > 0:
                decr = min(-diff, h[j])
                h[j] -= decr
                diff += decr

    return h

    h0 = project_to_feasible(h0, target_sum)
    
    # Fine-tune h0 to improve it
    def objective(h):
        return evaluate(h, dx)
    
    # Nelder-Mead optimization with feasibility
    best_h = h0.copy()
    best_c5 = evaluate(h0, dx)
    
    # Initialize simplex
    simplex = [h0]
    for _ in range(n_points + 1):
        pert = np.random.normal(0, 0.05, n_points)
        h_trial = h0 + pert
        h_trial = project_to_feasible(h_trial, target_sum)
        simplex.append(h_trial)
    
    simplex = np.array(simplex)
    
    # Nelder-Mead loop
    iteration = 0
    max_iter = 5000
    
    while iteration < max_iter and (time.time() - start_time) < budget_s:
        # Evaluate all points
        values = []
        for h in simplex:
            val = evaluate(h, dx)
            values.append(val)
        values = np.array(values)
        
        # Sort simplex by values
        idxs = np.argsort(values)
        simplex = simplex[idxs]
        values = values[idxs]
        
        # Update best
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
        
        # Check termination condition
        if np.std(values) < 1e-8:
            break
            
        # Nelder-Mead operations on simplex[0] to simplex[-2]
        x0 = simplex[0]  # lowest
        x1 = simplex[1]  # second lowest

        x3 = simplex[-1]  # highest
        
        # Compute centroid (excluding worst point)
        xc = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        xr = xc + (xc - x3)  # reflection point
        xr = project_to_feasible(xr, target_sum)
        value_r = evaluate(xr, dx)
        
        # Expansion
        if value_r < values[0]:
            xe = xc + 2 * (xr - xc)
            xe = project_to_feasible(xe, target_sum)
