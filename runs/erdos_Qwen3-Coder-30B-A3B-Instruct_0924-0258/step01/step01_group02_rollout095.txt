import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    T = 10.0 * best_c5
    cooling_rate = 0.95
    min_temp = 1e-6
    iterations_per_temp = 500
    iteration = 0
    max_iter = int(budget_s * 1000)  # Rough estimate
    
    while T > min_temp and iteration < max_iter:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        for _ in range(iterations_per_temp):
            # Create neighbor solution
            h_new = h.copy()
            # Perturb a few random indices
            num_perturb = max(1, n_points // 10)
            indices = np.random.choice(n_points, size=num_perturb, replace=False)
            for idx in indices:
                h_new[idx] += np.random.normal(0, 0.01)
            
            # Project to feasible region
            h_new = project_h(h_new)
            
            # Compute new energy
            new_c5 = compute_c5(h_new)
            
            # Accept or reject
            delta = new_c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_new
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            
            iteration += 1
            
            if iteration >= max_iter:
                break
                
        # Cool down
        T *= cooling_rate
        
        # Occasionally reset if stuck
        if iteration % 1000 == 0 and T > 1e-4:
            # Restart with best solution
            h = best_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
