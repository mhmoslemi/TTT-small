import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def objective_with_smooth_max(h, n_points, dx):
    """Objective using smooth max to approximate the maximum"""
    corr = np.correlate(h, 1 - h, mode='full')
    # Smooth max of the correlation values
    smooth_max_val = smooth_max(corr, alpha=100)
    # Penalize if too large to encourage minimizing the true maximum
    penalty = 0.1 * (smooth_max_val - 0.3)**2
    return smooth_max_val + penalty

def gradient_smooth_max(h, n_points, dx, eps=1e-8):
    """Gradient of the smooth max approximation"""
    corr = np.correlate(h, 1 - h, mode='full')
    alpha = 100
    exp_corr = np.exp(alpha * corr)
    sum_exp = np.sum(exp_corr)
    grad_corr = alpha * exp_corr / sum_exp
    
    # Gradient of the convolution with respect to h
    # For simplicity, we use finite difference
    grad = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        obj_plus = objective_with_smooth_max(h_plus, n_points, dx)
        obj_base = objective_with_smooth_max(h, n_points, dx)
        grad[i] = (obj_plus - obj_base) / eps
    return grad

def optimize_with_custom_method(h_start, n_points, dx, time_limit):
    """Optimize using custom gradient descent with simulated annealing"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Custom objective and gradient functions
    def obj(h):
        return objective_with_smooth_max(h, n_points, dx)
    
    def grad(h):
        return gradient_smooth_max(h, n_points, dx)
    
    # Constraints
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Simulated annealing inspired cooldown
    temp = 1.0
    cooling_rate = 0.95
    for i in range(200):  # maximum iterations
        if time() - start_time > time_limit:
            break
            
        # Perturb with temperature-dependent noise
        noise = np.random.normal(0, temp, n_points)
        h_perturbed = project_to_feasible(h_start + noise, n_points)
        
        # Optimization using SLSQP
        try:
            res = minimize(obj, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons, jac=grad, 
                           options={'maxiter': 30, 'ftol': 1e-6})
            if res.success:
                h_res = project_to_feasible(res.x, n_points)
                c5 = evaluate_erdos_solution(h_res, dx)
                if c5 < best_c5:
                    best_h = h_res.copy()
                    best_c5 = c5
        except:
            pass
        
        # Update temperature
        temp *= cooling_rate
        h_start = best_h.copy()
    
    return best_h, best_c5

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    # Try different resolutions
    resolutions = [25, 30, 35, 40, 45]
    
    for i, n_points_try in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
        
        # Run optimization with time constraints  
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 80)
        
        # Multiple restarts
        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Small perturbation around current best
            h_restart = project_to_feasible(h_start + np.random.normal(0, 0.03, n_points), n_points)
            h_opt, c5_opt = optimize_with_custom_method(h_restart, n_points, dx_try, time_per_res/3)
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
                
        h_start = best_h.copy()
        
    return (list(best_h), best_c5, best_n_points)
