import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx

    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec

    def compute_gradient(h_vec, k_max):
        # Compute gradient w.r.t. the maximum overlap shift
        n = len(h_vec)
        grad = np.zeros(n)
        for i in range(n):
            # For lag k_max, contribution to overlap at position i
            j_plus = (i + k_max) % n
            j_minus = (i - k_max) % n
            # Gradient is (1 - h[j_plus]) - h[j_minus] for positive lag
            grad[i] = (1 - h_vec[j_plus]) - h_vec[j_minus]
        return grad

    def gradient_descent_step(h_vec, max_iter=50):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        alpha = 0.1
        stagnation_count = 0
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Compute all correlation values
            corr = correlate(h_current, 1 - h_current, mode='full')
            # Find maximum lag
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (len(h_current) - 1)
            
            # Compute gradient direction
            grad = compute_gradient(h_current, max_lag)
            
            # Try step sizes
            step_sizes = [alpha, alpha*0.5, alpha*0.1]
            best_new_h = h_current.copy()
            best_new_C5 = best_C5
            
            for step in step_sizes:
                candidate_h = h_current - step * grad
                candidate_h = project_feasible(candidate_h)
                new_C5 = compute_C5(candidate_h)
                if new_C5 < best_new_C5:
                    best_new_C5 = new_C5
                    best_new_h = candidate_h.copy()
            
            # Update if improvement
            if best_new_C5 < best_C5:
                h_current = best_new_h.copy()
                best_C5 = best_new_C5
                stagnation_count = 0
                if best_C5 < 0.3807:
                    break
            else:
                stagnation_count += 1
                if stagnation_count > 20:
                    # Perturb slightly
                    noise = np.random.normal(0, 0.01, n_points)
                    h_current += noise
                    h_current = project_feasible(h_current)
                    stagnation_count = 0
                if stagnation_count % 10 == 0:
                    alpha *= 0.7
                    
        return best_h, best_C5

    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Simulated annealing-like loop
    temperature = 1.0
    while time.time() - start_time < budget_s - 0.1:
        # Try local optimization from current best
        h_opt, C5_opt = gradient_descent_step(h_best)
        if C5_opt < C5_best:
            C5_best = C5_opt
            h_best = h_opt.copy()
        
        # Occasionally try random perturbations
        if np.random.random() < 0.3:
            h_perturbed = h_best + np.random.normal(0, 0.03, n_points)
            h_perturbed = project_feasible(h_perturbed)
            h_opt, C5_opt = gradient_descent_step(h_perturbed)
            if C5_opt < C5_best:
                C5_best = C5_opt
                h_best = h_opt.copy()
                
        # Cool down temperature
        temperature *= 0.995
        
        # Early stopping if good enough
        if C5_best < 0.3807:
            break
    
    # Final projection and evaluation
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
