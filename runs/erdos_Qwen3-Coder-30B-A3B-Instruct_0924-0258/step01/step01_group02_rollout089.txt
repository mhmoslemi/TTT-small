import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def compute_gradient(h_vec):
        # Compute all correlations and find the maximum
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        k_max = np.argmax(corr)
        
        # Analytical gradient for shift k_max
        grad = np.zeros_like(h_vec)
        n = len(h_vec)
        # For each position i, compute contributions to gradient
        for i in range(n):
            # Contribution from h[i] * (1 - h[i+k_max]) 
            if i + k_max < n:
                grad[i] += (1 - h_vec[i + k_max]) * dx
            # Contribution from h[i-k_max] * (1 - h[i])
            if i - k_max >= 0:
                grad[i] -= h_vec[i - k_max] * dx
        return grad
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    
    # Adam optimizer parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.01
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute current gradient
        grad = compute_gradient(h)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible set
        h = project_h(h)
        
        # Compute new C5 and update best
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        # Occasionally add noise to escape local optima
        if iteration % 500 == 0:
            noise = 0.01 * np.random.normal(0, 1, len(h))
            h = h + noise
            h = project_h(h)
            
        iteration += 1
        
        # Early stopping if running out of iterations
        if iteration > 10000:
            break
    
    return (best_h.tolist(), best_c5, n_points)
