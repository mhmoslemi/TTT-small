```python
import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def gradient_descent_step(h, dx, learning_rate=1e-3):
    # Compute gradient of overlap with respect to h
    # Use finite differences for gradient approximation
    n = len(h)
    eps = 1e-6
    grad = np.zeros(n)
    base_c5 = evaluate(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_constraints(h_plus, dx)
        c5_plus = evaluate(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    
    # Update h in direction of negative gradient
    h_new = h - learning_rate * grad
    h_new = project_to_constraints(h_new, dx)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with small resolution
    n_low = 10
    n_high = 100
    n_points = n_low
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
        h = project_to_constraints(h, dx)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Track best ever
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    iter_count = 0
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Gradient descent step
        h_new = gradient_descent_step(h, dx, learning_rate=1e-2)
        c5_new = evaluate(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
        elif c5_new < 0.99 * best_c5:
            h = h_new
        else:
            # Sometimes take small random perturbations to escape local optima
            if np.random.uniform() < 0.1:
                h_new = h + np.random.normal(0, 1e-3, n_points)
                h_new = np.clip(h_new, 0, 1)
                h_new = project_to_constraints(h_new, dx)
                c5_new = evaluate(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new
        
        iter_count += 1
        
        # Occasionally double resolution
        if n_points < n_high and iter_count % 200 == 0:
            if best_c5 < 0.385:
                h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                                  np.linspace(0, 2, n_points), h)
                h = h_long
                n_points = 2 * n_points
                dx = 2.0 / n_points
                h = project_to_constraints(h, dx)
                best_h = h.copy()
                best_c5 = evaluate(h, dx)
    
    # Final refinement
    final_h = best_h.copy()
    final_c5 = evaluate(final_h, dx)
    
    return final_h.tolist(), final_c5, len(final_h)
```
