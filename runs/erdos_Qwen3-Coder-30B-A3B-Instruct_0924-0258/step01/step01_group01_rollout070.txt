import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def compute_gradient(h, dx, critical_lags):
    n = len(h)
    grad = np.zeros(n)
    for k in critical_lags:
        if k < n:
            grad[k] += 1.0
        else:
            # Handle negative lag case
            grad[n - (k - n + 1)] -= 1.0
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_low = 10
    n_high = 120
    n_points = n_low
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    last_c5 = best_c5
    
    # Parameters
    learning_rate = 0.01
    max_iter_per_res = 1000
    max_iter = max_iter_per_res * 20  # Upper bound for safety
    
    iter_count = 0
    last_improvement = 0
    prev_c5 = best_c5
    
    # Main loop
    while iter_count < max_iter and time.time() - start_time < budget_s * 0.95:
        if time.time() - start_time > budget_s:
            break
            
        # Compute cross-correlation
        corr = correlate(h, 1-h, mode='full')
        c5_vals = corr * dx
        max_idx = np.argmax(c5_vals)
        max_c5 = c5_vals[max_idx]
        
        # Identify critical lags contributing to max
        # For safety, we only consider top 10% of the maximum
        critical_lags = []
        sorted_indices = np.argsort(c5_vals)[::-1]
        top_count = max(1, len(c5_vals) // 10)
        critical_lags = sorted_indices[:top_count]
        
        # Less aggressive gradient descent 
        grad = compute_gradient(h, dx, critical_lags)
        
        # Update h
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_constraints(h_new, dx)
        
        # Evaluate new solution
        new_c5 = evaluate(h_new, dx)
        
        # Accept or reject
        if new_c5 < last_c5:
            h = h_new
            last_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                last_improvement = iter_count
        
        # Check for resolution increase if improvement above threshold
        if (iter_count - last_improvement > 200 and 
            n_points < n_high and
            time.time() - start_time < budget_s * 0.9):
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            last_c5 = evaluate(h, dx)
            last_improvement = iter_count
            learning_rate = min(0.05, learning_rate*1.2)  # Increase learning rate
        
        iter_count += 1
    
    # Ensure the final vector meets constraints exactly
    final_h = project_to_constraints(best_h, dx)
    final_c5 = evaluate(final_h, dx)
    
    return final_h.tolist(), final_c5, len(final_h)
