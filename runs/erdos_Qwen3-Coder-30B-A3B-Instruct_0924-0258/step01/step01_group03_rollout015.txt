import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5  # Simple uniform start

    # Initialize h
    h = h0.copy()
    dx = 2.0 / n_points
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iterations = 10000
    patience = 50
    patience_counter = 0
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    last_improvement = 0
    
    # Gradient descent
    for i in range(max_iterations):
        # Break if time is up
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlations for all shifts
        corr = correlate(h, 1 - h, mode="full")
        # Find index of maximum correlation
        k_max = np.argmax(np.abs(corr))
        # Determine the actual shift from correlation index
        actual_shift = k_max - (len(h) - 1)
        
        # Compute gradient components for the maximizing shift
        gradient = np.zeros_like(h)
        # Positive shift case: h[j] contributes to the gradient
        # if it appears in the correlation with 1-h[j-shift]
        # But since we want to reduce the correlation value, 
        # we think of computing how much we should change h[j]  
        # to reduce the contribution to the maximum overlap
        # Since correlation is sum of h[i] * (1 - h[i+shift]), 
        # we compute gradient wrt h[j]: 
        #   grad = (1 - h[j+shift]) if j+shift<n and j+shift>=0
        #          (1 - h[j-shift]) if j-shift >= 0 and j-shift < n
        # But more simply, use: grad_j = (1 - h[j+shift]) when j < n-shift only if shift > 0, 
        # or (1 - h[j-shift]) when shift < 0 and j > shift.
        for j in range(len(h)):
            if actual_shift >= 0:
                # Positive shift: h[j] interacts with h[j+shift]
                if j + actual_shift < len(h):
                    gradient[j] += (1 - h[j + actual_shift])
            else:
                # Negative shift: h[j] interacts with h[j-shift]
                if j - actual_shift < len(h):
                    gradient[j] += (1 - h[j - actual_shift])

        # Apply gradient with learning rate
        h = h - learning_rate * gradient
        
        # Project onto feasible region [0,1]
        h = np.clip(h, 0, 1)
        
        # Maintain the sum constraint exactly by rescaling
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2.0) / current_sum
        else:
            h = np.ones_like(h) * (n_points / 2.0) / n_points
            
        # Evaluate current solution
        current_c5 = evaluate(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
            last_improvement = i
        else:
            patience_counter += 1
            
        # Adaptive learning rate
        if i - last_improvement > 100:
            learning_rate *= 0.9
            last_improvement = i
            
        # Optional adaptive random perturbation
        if patience_counter > patience:
            # Add small random noise to escape local minima
            noise = np.random.uniform(-1e-3, 1e-3, len(h))
            h = h + noise
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            if current_sum > 0:
                h = h * (n_points / 2.0) / current_sum
            patience_counter = 0
            
    return (best_h.tolist(), float(best_c5), n_points)
