import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Scale to target sum
        h = h * target_sum / current_sum
        # Clamp to [0,1] again
        h = np.clip(h, 0, 1)
        # Re-normalize if still outside
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            # Distribute the difference carefully
            diff = target_sum - current_sum
            # Find indices that can be adjusted
            idx = np.argsort(h)
            if diff > 0 and h[idx[0]] < 1.0:
                update = min(diff, 1.0 - h[idx[0]])
                h[idx[0]] += update
                diff -= update
            elif diff < 0 and h[idx[-1]] > 0.0:
                update = max(diff, -h[idx[-1]])
                h[idx[-1]] += update
                diff -= update
            # Distribute remaining diff
            for i in range(len(idx)-1, -1, -1):
                if abs(diff) < 1e-10:
                    break
                j = idx[i]
                if diff > 0 and h[j] < 1.0:
                    update = min(diff, 1.0 - h[j])
                    h[j] += update
                    diff -= update
                elif diff < 0 and h[j] > 0.0:
                    update = max(diff, -h[j])
                    h[j] += update
                    diff -= update
    return h

        corr = correlate(h, 1 - h, mode="full")
        k_max_idx = np.argmax(np.abs(corr))

        gradient = np.zeros_like(h)
        if k_max >= 0:

            for j in range(len(h)):
