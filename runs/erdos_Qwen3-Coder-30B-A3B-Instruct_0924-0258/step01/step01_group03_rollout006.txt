import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region [0,1] and sum=target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to target
    h = h * target_sum / current_sum
    # Clip and renormalize
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Adjust by distributing difference
        diff = target_sum - current_sum
        # Increase smaller values
        idx = np.argsort(h)
        for i in range(len(idx)-1, -1, -1):
            if diff == 0:
                break
            j = idx[i]
            if h[j] < 1.0:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Set up optimization
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize
    h = h0.copy()
    h = project_to_feasible(h, target_sum)
    
    # Parameters
    max_iter = 5000
    learning_rate = 0.01
    patience = 50
    patience_counter = 0
    best_c5 = float('inf')
    best_h = None
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute correlation to identify max shift
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        
        # Compute gradient for the max shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i - max_k) % n_points
            grad[i] = 1 - 2 * h[j]
        
        # Update
        h = h - learning_rate * grad
        
        # Project to feasible region
        h = project_to_feasible(h, target_sum)
        
        # Evaluate
        current_c5 = evaluate(h, dx)
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Adjust learning rate if no improvement
        if patience_counter >= 100:
            learning_rate *= 0.9
            patience_counter = 0
            
        # Perturb if stuck
        if patience_counter >= patience:
            noise = np.random.uniform(-1e-3, 1e-3, n_points)
            h = h + noise
            h = project_to_feasible(h, target_sum)
            patience_counter = 0
    
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        # Fallback
        h_final = project_to_feasible(h0, target_sum)
        final_c5 = evaluate(h_final, dx)
        return (h_final.tolist(), float(final_c5), n_points)
