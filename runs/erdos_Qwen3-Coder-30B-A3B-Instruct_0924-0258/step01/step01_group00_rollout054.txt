import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_autocorrelation(h, dx):
    corr = np.correlate(h, h, mode='full')
    return corr * dx

def compute_min_autocorr(h, dx):
    autocorr = compute_autocorrelation(h, dx)
    min_a = np.min(autocorr)
    return min_a

def smooth_max(x, delta=1e-3):
    return np.log(np.sum(np.exp(x/delta))) * delta

def smooth_min(x, delta=1e-3):
    return -np.log(np.sum(np.exp(-x/delta))) * delta

def objective_with_smooth_max(h, n_points, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr)
    return max_overlap

def gradient_of_objective(h, n_points, dx):
    eps = 1e-6
    grad = np.zeros_like(h)
    base_obj = objective_with_smooth_max(h, n_points, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        obj_plus = objective_with_smooth_max(h_plus, n_points, dx)
        grad[i] = (obj_plus - base_obj) / eps
    return grad

def optimize_at_resolution(h_start, n_points, dx, time_limit):
    start_time = time()
    best_h = h_start.copy()
    best_value = objective_with_smooth_max(h_start, n_points, dx)
    
    # Multiple restarts
    for restart in range(5):
        if time() - start_time > time_limit:
            break
            
        h_current = project_to_feasible(h_start + np.random.normal(0, 0.03, n_points), n_points)
        
        # Try gradient-based optimization
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective_with_smooth_max, h_current, args=(n_points, dx),
                          method='SLSQP', bounds=bounds, constraints=cons, 
                          jac=gradient_of_objective, options={'maxiter': 30, 'ftol': 1e-6})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                val = objective_with_smooth_max(h_opt, n_points, dx)
                if val < best_value:
                    best_h = h_opt.copy()
                    best_value = val
        except:
            pass
        
        # Perturb for next restart
        h_start = project_to_feasible(best_h + np.random.normal(0, 0.05, n_points), n_points)
    
    return best_h, best_value

    start_time = time()
    
    if 'initial_h_values' in globals():

    else:
        n0 = 30
        h_start = np.ones(n0) / 2
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Try different resolutions
    resolutions = [20, 30, 40, 50]
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate h_start if necessary
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 80)
        
        h_opt, c5_opt = optimize_at_resolution(h_start, n_points, dx, time_per_res)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n_points = n_points
            
        h_start = h_opt.copy()
        
    return (list(best_h), best_c5, best_n_points)
