import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        initial_h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial_h satisfies the constraint
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def evaluate_fitness(h_vec):
        return -compute_c5(h_vec)
    
    # Genetic Algorithm parameters
    population_size = 75
    num_generations = 1000
    mutation_rate = 0.05
    elite_size = population_size // 4
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    best_c5 = compute_c5(initial_h)
    best_h = initial_h.copy()
    
    start_time = time.time()
    
    for generation in range(num_generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness for all individuals
        fitness_scores = [evaluate_fitness(h) for h in population]
        
        # Update best solution
        best_idx = np.argmax(fitness_scores)
        if -fitness_scores[best_idx] < best_c5:
            best_c5 = -fitness_scores[best_idx]
            best_h = population[best_idx].copy()
        
        # Selection - tournament selection
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover - uniform crossover
        offspring = []
        for i in range(0, population_size, 2):
            h1 = selected[i]
            h2 = selected[i+1] if i+1 < population_size else selected[0]
            
            # Uniform crossover
            mask = np.random.rand(n_points) < 0.5
            new_h1 = np.where(mask, h1, h2)
            new_h2 = np.where(mask, h2, h1)
            
            # Repair individuals
            new_h1 = project_h(new_h1)
            new_h2 = project_h(new_h2)
            
            offspring.append(new_h1)
            offspring.append(new_h2)
        
        # Mutation
        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                offspring[i] += noise
                offspring[i] = project_h(offspring[i])
        
        # Replacement - keep elite + offspring
        # Sort current population
        sorted_indices = np.argsort(fitness_scores)[::-1]
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Replace worst individuals with offspring
        new_population = elite + offspring[:population_size - elite_size]
        population = new_population
    
    return (best_h.tolist(), best_c5, n_points)
