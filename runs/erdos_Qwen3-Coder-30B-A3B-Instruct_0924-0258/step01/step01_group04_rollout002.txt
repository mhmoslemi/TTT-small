import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    # Adam optimizer setup
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    learning_rate = 0.01
    
    # Main optimization loop
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        if iteration % 100 == 0:
            # Periodic full evaluation
            current_C5 = compute_C5(best_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = best_h.copy()
        
        # Compute correlations to identify maximum k
        h_vec = np.clip(h, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        
        # Identify max k (top shift)
        max_k_idx = np.argmax(overlaps)
        max_k = max_k_idx - (n_points - 1)
        
        # Compute gradient with respect to max_k
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            #_shifted_idx = (i - max_k) % n_points  
            shifted_idx = i - max_k
            if shifted_idx >= 0 and shifted_idx < n_points:
                grad[i] = 1 - h_vec[shifted_idx]
            else:
                # Handle out of bounds by mirroring or ignoring
                grad[i] = 0
                
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project h (clip and rescale)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
            
        # Adaptive learning rate
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            learning_rate *= 1.05  # Increase if improving
        else:
            if iteration % 50 == 0:
                learning_rate *= 0.95  # Decrease if not improving
                
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)
