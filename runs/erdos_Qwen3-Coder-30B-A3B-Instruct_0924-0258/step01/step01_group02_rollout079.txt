import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def compute_gradient(h_vec, k_max):
        # Create gradient vector for shift k_max
        grad = np.zeros_like(h_vec)
        # For correlation at shift k_max, gradient w.r.t. h[i] is:
        # 1 - 2 * h[i] if there's overlap, 0 otherwise
        # We compute for index i, so we look at h[i] * (1 - h[i + k_max])
        # For each i, if (i + k_max) is valid, then:
        # grad[i] += 1 - 2 * h[i] 
        # But this is the term for central index i, which is wrong.
        # Let's be more precise:
        # (h * (1-h))[k_max] = sum_{i=0}^{n-1} h[i] * (1 - h[i+k_max])
        # Gradient w.r.t. h[i] = (1 - h[i+k_max]) if i+k_max < n else 0
        # So that is: 
        for i in range(len(h_vec)):
            j = i + k_max
            if j < len(h_vec):
                grad[i] = 1 - 2 * h_vec[i]
        return grad
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    max_iter = 5000
    stagnation_count = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute current correlation
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        c5_current = corr[k_max] * dx
        
        # Compute analytical gradient for k_max
        grad = compute_gradient(h, k_max)
        
        # Gradient update
        h = h - lr * grad
        h = project_h(h)
        
        # Compute new C5 and update best
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Anneal learning rate
        lr *= 0.999
        
        # Occasionally add noise if stagnating
        if stagnation_count > 100 or iteration % 500 == 0:
            noise = 0.005 * np.random.uniform(-1, 1, len(h))
            h = h + noise
            h = project_h(h)
            
        iteration += 1
        
        # Early termination if no improvement in 1000 iterations
        if stagnation_count > 1000:
            break
    
    return (best_h.tolist(), best_c5, n_points)
