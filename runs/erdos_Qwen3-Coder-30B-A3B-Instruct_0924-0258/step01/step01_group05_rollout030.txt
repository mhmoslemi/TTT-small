import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Genetic algorithm parameters
    pop_size = 50
    tournament_size = 10
    elitism_count = pop_size // 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_init = np.random.rand(n_points)
        h_init = project_feasible(h_init)
        population.append(h_init)
    
    best_C5 = float('inf')
    best_h = None
    stagnation_counter = 0
    
    # Evaluate initial population
    fitness_scores = []
    for h_ind in population:
        C5_val = compute_C5(h_ind)
        fitness_scores.append(C5_val)
        if C5_val < best_C5:
            best_C5 = C5_val
            best_h = h_ind.copy()
    
    # Evolution loop
    generation = 0
    while time.time() - start_time < budget_s - 0.1:
        generation += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness_scores[i] for i in sorted_indices]
        
        if sorted_fitness[0] < best_C5:
            best_C5 = sorted_fitness[0]
            best_h = sorted_population[0].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        if stagnation_counter > patience:
            # Reset worst performers
            for i in range(pop_size // 5):
                rand_h = np.random.rand(n_points)
                population[-(i+1)] = project_feasible(rand_h)
            stagnation_counter = 0
            
        # Selection: Tournament
        new_population = []
        # Elitism: keep best individuals
        new_population.extend(sorted_population[:elitism_count])
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[winner_idx]
            
            # Another parent
            tournament_indices2 = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness2 = [fitness_scores[i] for i in tournament_indices2]
            winner_idx2 = tournament_indices2[np.argmin(tournament_fitness2)]
            parent2 = population[winner_idx2]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Gaussian mutation
                noise = np.random.normal(0, 0.05, n_points)
                child = child + noise
                child = project_feasible(child)
            elif np.random.rand() < mutation_rate:
                # Reset mutation
                reset_indices = np.random.choice(n_points, size=n_points // 20, replace=False)
                child[reset_indices] = np.random.choice([0, 1], len(reset_indices))
                child = project_feasible(child)
            
            new_population.append(child)
        
        # Update population
        population = new_population[:pop_size]
        
        # Re-evaluate fitness
        fitness_scores = [compute_C5(h_ind) for h_ind in population]
    
    # Final check
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
