import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # SDP-based optimization
    def solve_sdp_step(h_init, n):
        h_vec = h_init.copy()
        h_vec = project_h(h_vec)
        
        # Simplified SDP relaxation approach
        # For small grids, we use a direct optimization approach
        def obj_func(h_vec):
            h_vec = np.array(h_vec)
            conv = np.correlate(h_vec, 1 - h_vec, mode='full')
            mid = len(conv) // 2
            overlaps = conv[mid-n+1:mid+n-1] * (2.0/n)
            return -np.max(overlaps)  # Negative because we minimize
        
        # Use scipy minimize with L-BFGS-B for local refinement
        result = optimize.minimize(
            obj_func,
            h_vec,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n)],
            options={'maxiter': 200, 'ftol': 1e-6}
        )
        
        if result.success:
            return result.x
        else:
            return h_vec
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Multi-scale approach
    n_candidates = [20, 30, 45]
    for n in n_candidates:
        if time.time() - start_time > budget_s - 5:
            break
            
        if n > n_points:
            # Upscale
            h_upscaled = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_points), best_h)
        else:
            h_upscaled = best_h[:n].copy()
        
        # Refine with optimization
        refined_h = solve_sdp_step(h_upscaled, n)
        refined_h = project_h(refined_h)
        refined_C5 = compute_C5(refined_h)
        
        if refined_C5 < best_C5:
            best_C5 = refined_C5
            best_h = refined_h.copy()
            n_points = n
    
    # Final local refinement
    if time.time() - start_time < budget_s - 1:
        refined_h = solve_sdp_step(best_h, n_points)
        refined_h = project_h(refined_h)
        refined_C5 = compute_C5(refined_h)
        
        if refined_C5 < best_C5:
            best_C5 = refined_C5
            best_h = refined_h.copy()
    
    return (best_h.tolist(), best_C5, n_points)
