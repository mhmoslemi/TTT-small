import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a good starting point
        n_points = 30
        h_start = np.ones(n_points) * 0.5
        # Add some structure to help with convergence
        h_start[5:10] = 0.8
        h_start[20:25] = 0.2
        
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    # Try various resolutions to balance exploration and performance
    resolutions = [20, 30, 40]
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate to the new resolution
        if len(h_start) != n_points:
            x_orig = np.linspace(0, 2, len(h_start))
            x_new = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_new, x_orig, h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Multiple restarts with different initial perturbations
        best_c5_level = evaluate_erdos_solution(h_start, dx)
        best_h_level = h_start.copy()
        
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Slightly perturb the current solution
            h_perturbed = h_start + np.random.normal(0, 0.03, n_points)
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            
            # Optimize with gradient-based method
            def obj(h):
                return -evaluate_erdos_solution(h, dx)  # Negative for minimization
            
            def grad(h):
                # Using finite differences for gradient
                eps = 1e-6
                grad_val = np.zeros_like(h)
                base_val = obj(h)
                for i in range(n_points):
                    h_p = h.copy()
                    h_p[i] += eps
                    h_p = project_to_feasible(h_p, n_points)
                    grad_val[i] = (obj(h_p) - base_val) / eps
                return grad_val
            
            try:
                # Use SLSQP with gradient
                cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(obj, h_perturbed, method='SLSQP', 
                               bounds=bounds, constraints=cons, jac=grad,
                               options={'maxiter': 20, 'ftol': 1e-6})
                
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5_level:
                        best_c5_level = c5_opt
                        best_h_level = h_opt.copy()
                        
            except Exception as e:
                pass  # Continue if optimization fails
            
            # Alternate strategy: Edge-focused search
            h_alt = h_start.copy()
            # Perturb edges more heavily
            h_alt[0] += np.random.normal(0, 0.05)
            h_alt[-1] += np.random.normal(0, 0.05)
            h_alt = project_to_feasible(h_alt, n_points)
            c5_alt = evaluate_erdos_solution(h_alt, dx)
            if c5_alt < best_c5_level:
                best_c5_level = c5_alt
                best_h_level = h_alt.copy()
            
            # Update the master solution tracking
            if time() - start_time > budget_s * 0.95:
                break
                
        if best_c5_level < best_c5:
            best_c5 = best_c5_level
            best_h = best_h_level.copy()
            best_n_points = n_points
            
        h_start = best_h_level.copy()
    
    return (list(best_h), best_c5, best_n_points)
