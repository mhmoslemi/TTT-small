import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5  # Simple uniform start
    
    # Set dx
    dx = 2.0 / n_points
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 50
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute overlap integral
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for j in range(n_points):
            k = k_max - j
            if k < 0:
                k += 2 * n_points - 1
            if k >= n_points:
                k = k % n_points
            grad[j] = 1 - 2 * h[k]
        
        # Update h
        h = h - learning_rate * grad
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum) > 1e-10:
            h = h * (n_points / 2.0) / h_sum
        
        # Evaluate current solution
        current_c5 = evaluate(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adaptive learning rate
        if patience_counter > 100:
            learning_rate *= 0.9
            patience_counter = 0
            
        # Perturb if stuck
        if patience_counter == patience:
            noise = np.random.uniform(-1e-3, 1e-3, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum) > 1e-10:
                h = h * (n_points / 2.0) / h_sum
            patience_counter = 0
    
    return (best_h.tolist(), float(best_c5), n_points)
