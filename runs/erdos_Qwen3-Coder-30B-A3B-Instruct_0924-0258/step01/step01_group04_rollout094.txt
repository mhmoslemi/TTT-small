import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def smooth_h(h_vec, alpha=0.1):
        # Apply simple smoothing to encourage smoothness
        smoothed = h_vec.copy()
        if len(h_vec) > 1:
            for i in range(1, len(h_vec)-1):
                smoothed[i] = (1-alpha)*h_vec[i] + alpha*(h_vec[i-1] + h_vec[i+1])/2
        return smoothed
    
    # Multi-resolution approach: progressively increase resolution
    resolutions = [5, 10, 20, 40, n_points]
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    for res_idx, current_n in enumerate(resolutions):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize h at this resolution
        if current_n <= len(h):
            h_res = h[:current_n].copy()
        else:
            # Upsample from previous resolution if available
            if res_idx > 0:
                h_prev = best_h
                h_res = np.interp(np.linspace(0, 2, current_n), 
                                  np.linspace(0, 2, len(h_prev)), 
                                  h_prev)
            else:
                h_res = np.ones(current_n) * 0.5
        
        dx_res = 2.0 / current_n
        target_sum_res = current_n / 2.0
        
        # Helper to compute C5 at this resolution
        def compute_C5_res(h_vec):
            h_vec = np.array(h_vec)
            conv = np.correlate(h_vec, 1 - h_vec, mode='full')
            mid = len(conv) // 2
            overlaps = conv[mid-current_n+1:mid+current_n-1] * dx_res
            return np.max(overlaps)
        
        def project_h_res(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * target_sum_res / np.sum(h_vec)
            return h_vec
                
        # Adam-like optimizer
        m = np.zeros_like(h_res)
        v = np.zeros_like(h_res)
        beta1, beta2 = 0.9, 0.999
        epsilon = 1e-8
        lr = 0.01
        
        # Optimization for current resolution
        max_iter = 1000
        iter_count = 0
        best_h_res = h_res.copy()
        best_C5_res = compute_C5_res(best_h_res)
        
        while iter_count < max_iter and time.time() - start_time < budget_s - 1:
            iter_count += 1
            
            # Project
            h_res = project_h_res(h_res)
            
            # Compute overlaps to identify k contributing to max
            conv = np.correlate(h_res, 1 - h_res, mode='full')
            mid = len(conv) // 2
            overlaps = conv[mid-current_n+1:mid+current_n-1] * dx_res
            top_k = np.argsort(overlaps)[-5:][::-1]
            
            # Approximate gradient using finite differences with smoothing
            gradient = np.zeros_like(h_res)
            eps = 1e-5
            
            for i in top_k:
                if i >= len(h_res) or i < 0:
                    continue
                h_plus = h_res.copy()
                h_plus[i] += eps
                h_plus = project_h_res(h_plus)
                C_plus = compute_C5_res(h_plus)
                
                h_minus = h_res.copy()
                h_minus[i] -= eps
                h_minus = project_h_res(h_minus)
                C_minus = compute_C5_res(h_minus)
                
                gradient[i] = (C_plus - C_minus) / (2 * eps)
            
            # Update with Adam
            m = beta1 * m + (1 - beta1) * gradient
            v = beta2 * v + (1 - beta2) * gradient**2
            m_hat = m / (1 - beta1**(iter_count))
            v_hat = v / (1 - beta2**(iter_count))
            h_res = h_res - lr * m_hat / (np.sqrt(v_hat) + epsilon)
            
            # Apply smoothing
            h_res = smooth_h(h_res, alpha=0.05)
            
            # Project again
            h_res = project_h_res(h_res)
            
            # Track best
            current_C5_res = compute_C5_res(h_res)
            if current_C5_res < best_C5_res:
                best_C5_res = current_C5_res
                best_h_res = h_res.copy()
            
            # Early stopping if no improvement
            if iter_count % 500 == 0 and current_C5_res >= best_C5_res:
                break
                
        # Move to next resolution with trajectory
        if res_idx < len(resolutions) - 1:
            # Upsample the best solution from this resolution
            next_n = resolutions[res_idx + 1]
            if next_n <= len(best_h_res):
                h = best_h_res[:next_n].copy()
            else:
                h = np.interp(np.linspace(0, 2, next_n),
                              np.linspace(0, 2, len(best_h_res)),
                              best_h_res)
            
            # Project to full resolution
            h = project_h(h)
            
            # Update best if better
            current_C5 = compute_C5(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
                
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)
