import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = h0.copy().astype(float)
    h = h * (n_points / 2.0) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent parameters
    eta = 0.01
    patience = 0
    last_improvement = 0
    
    iter_count = 0
    while time.time() - start_time < budget_s:
        iter_count += 1
        
        # Compute correlations to find maximum overlap shift
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(np.abs(corr))
        
        # Compute gradient based on the max overlap
        gradient = np.zeros_like(h)
        n = len(h)
        
        # For shift k_max (correlation index):
        # Gradient component for h[j] comes from (1 - h[j + k_max])
        if k_max < n:
            # Positive shift
            for j in range(n):
                k = k_max
                if j + k < n:
                    gradient[j] += (1 - h[j + k])
        else:
            # Negative shift case (consider full correlation)
            k = k_max - (2 * n - 2)
            for j in range(n):
                if j - k >= 0:
                    gradient[j] += (1 - h[j - k])
        
        # Update
        h = h - eta * gradient
        
        # Project to feasible set
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum) > 1e-10:
            h = h * (n_points / 2.0) / h_sum
            
        # Evaluate
        c5 = evaluate(h, dx)
        
        # Track best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iter_count
            patience = 0
        else:
            patience += 1
            if patience > 50:
                # Add small random perturbation
                h += np.random.uniform(-1e-3, 1e-3, n_points)
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if abs(h_sum) > 1e-10:
                    h = h * (n_points / 2.0) / h_sum
                patience = 0
                # Reset learning rate if no improvement for a while
                eta *= 0.9
                
        # Adapt learning rate
        if iter_count - last_improvement > 100:
            eta *= 0.9
            last_improvement = iter_count
            
    return (best_h.tolist(), float(best_c5), n_points)
