import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization with 45 points
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        """Compute C5 using full correlation."""
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        """Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum."""
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def grid_search_level(n_grid, n_candidates, refine=False, prev_best=None):
        """Generate candidates for a given grid resolution."""
        candidates = []
        
        for _ in range(n_candidates):
            if refine and prev_best is not None:
                # Perturb best solution
                h_candidate = prev_best + np.random.normal(0, 0.05, n_grid)
            else:
                # Random sample
                h_candidate = np.random.random(n_grid)
            
            # Project to feasible set
            h_candidate = project_feasible(h_candidate)
            candidates.append(h_candidate)
            
        return candidates
    
    def evaluate_candidates(candidates):
        """Evaluate all candidates and return best."""
        best_h = None
        best_C5 = float('inf')
        
        for h_cand in candidates:
            if time.time() - start_time > budget_s - 0.1:
                break
            C5_cand = compute_C5(h_cand)
            if C5_cand < best_C5:
                best_C5 = C5_cand
                best_h = h_cand.copy()
                
        return best_h, best_C5
    
    # Grid hierarchy levels
    levels = [5, 10, 15, 20, 25, 30, 45]
    
    # Start with current solution
    h_current = h.copy()
    C5_current = compute_C5(h_current)
    best_h = h_current.copy()
    best_C5 = C5_current
    
    # Regular grid search approach
    n_coarse = 50
    n_fine = 20
    
    # Hierarchical search
    for i, n_grid in enumerate(levels):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Determine number of candidates based on level
        if n_grid <= 15:
            n_candidates = n_coarse
        else:
            n_candidates = n_fine
            
        # Only refine at finer levels
        refine = (i > 2) and best_h is not None
        
        # Generate candidates
        candidates = grid_search_level(n_grid, n_candidates, refine, best_h)
        
        # Evaluate candidates at this level
        h_eval, C5_eval = evaluate_candidates(candidates)
        
        # Update if better solution found
        if C5_eval < best_C5:
            best_C5 = C5_eval
            best_h = h_eval.copy()
            
        # Special case: for 45 points, do detailed refinement
        if n_grid == 45:
            # Additional local optimization
            temp_h = best_h.copy()
            for _ in range(1000):
                if time.time() - start_time > budget_s - 0.1:
                    break
                    
                # Perturbation
                pert = np.random.normal(0, 0.01, n_grid)
                candidate = temp_h + pert
                
                # Project
                candidate = project_feasible(candidate)
                
                # Evaluate and accept if better
                C5_candidate = compute_C5(candidate)
                if C5_candidate < best_C5:
                    best_C5 = C5_candidate
                    best_h = candidate.copy()
                    temp_h = candidate.copy()
                
                temp_h = candidate.copy()
    
    # Final project
    final_h = project_feasible(best_h)
    final_C5 = compute_C5(final_h)
    
    return list(final_h), final_C5, n_points
