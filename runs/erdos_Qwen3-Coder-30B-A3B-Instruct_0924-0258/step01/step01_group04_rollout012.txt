import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Wrapper for scipy.optimize
    def objective(h_vec):
        h_vec = project_h(h_vec)
        return compute_C5(h_vec)
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Track the best solution
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Try different optimization approaches
    # 1. Differential Evolution (global search)
    try:
        de_result = differential_evolution(
            objective, 
            bounds, 
            seed=seed,
            maxiter=200,
            popsize=15,
            recombination=0.7,
            atol=1e-6,
            mutation=(0.5, 1.0)
        )
        if not np.isnan(de_result.fun):
            h_candidate = project_h(de_result.x)
            candidate_C5 = compute_C5(h_candidate)
            if candidate_C5 < best_C5:
                best_C5 = candidate_C5
                best_h = h_candidate.copy()
    except:
        pass

    # 2. Local search with Nelder-Mead if time allows
    remaining_time = budget_s - (time.time() - start_time) - 1
    if remaining_time > 1:
        # Simple local refinement starting from best so far
        try:
            # Start from best solution
            h_start = best_h.copy()
            # Refine using Nelder-Mead
            result = minimize(
                objective, 
                h_start, 
                method='Nelder-Mead',
                options={'maxiter': 1000, 'xatol': 1e-6, 'fatol': 1e-6}
            )
            if not result.failed:
                h_candidate = project_h(result.x)
                candidate_C5 = compute_C5(h_candidate)
                if candidate_C5 < best_C5:
                    best_C5 = candidate_C5
                    best_h = h_candidate.copy()
        except:
            pass

    # 3. Additional local search using simple gradient estimation
    # Only if sufficient time left
    remaining_time = budget_s - (time.time() - start_time) - 1
    if remaining_time > 2:
        # Random perturbations with local search
        for _ in range(500):
            if time.time() - start_time > budget_s - 1:
                break
            # Randomly perturb the solution
            h_perturbed = best_h + np.random.normal(0, 0.005, n_points)
            h_perturbed = project_h(h_perturbed)
            new_C5 = compute_C5(h_perturbed)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_perturbed.copy()

    # Final validation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, n_points)
