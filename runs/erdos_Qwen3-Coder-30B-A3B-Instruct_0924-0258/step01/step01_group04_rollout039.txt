import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy import signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def objective(h_vec):
        h_vec = project_h(h_vec)
        return compute_C5(h_vec)
    
    # Constraints for differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try optimization with scipy's differential evolution
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    try:
        # Perform global optimization
        result = differential_evolution(
            objective, 
            bounds, 
            maxiter=500, 
            popsize=15, 
            seed=seed,
            callback=lambda x, convergence: None,
            disp=False,
            strategy='best1bin'
        )
        
        if not result.success:
            # Fallback to local search
            x0 = project_h(result.x)
            result2 = minimize(
                objective,
                x0,
                method='Nelder-Mead',
                options={'maxiter': 500, 'disp': False}
            )
            if result2.success and result2.fun < best_C5:
                best_h = project_h(result2.x)
                best_C5 = result2.fun
        else:
            if result.fun < best_C5:
                best_h = project_h(result.x)
                best_C5 = result.fun
                
    except Exception:
        pass
    
    # Try local refinement with Nelder-Mead starting from the best
    best_h_refined = best_h.copy()
    best_C5_refined = best_C5
    
    try:
        # Local optimization around best solution
        result_local = minimize(
            objective,
            best_h_refined,
            method='Nelder-Mead',
            options={'maxiter': 1000, 'disp': False}
        )
        
        if result_local.success and result_local.fun < best_C5_refined:
            best_C5_refined = result_local.fun
            best_h_refined = project_h(result_local.x)
            
    except Exception:
        pass
        
    # Fallback to tabu search approach if needed
    if time.time() - start_time < budget_s - 10:
        # Start with new initialization and try local search
        h_test = np.random.rand(n_points) * 0.5 + 0.25  # Random values between 0.25 and 0.75
        h_test = project_h(h_test)
        
        try:
            # Local optimization on random initialization
            result_test = minimize(
                objective,
                h_test,
                method='Nelder-Mead',
                options={'maxiter': 500, 'disp': False}
            )
            
            if result_test.success and result_test.fun < best_C5_refined:
                best_C5_refined = result_test.fun
                best_h_refined = project_h(result_test.x)
                
        except Exception:
            pass
    
    # Final check with the best solution found
    final_C5 = compute_C5(best_h_refined)
    if final_C5 < best_C5_refined:
        best_C5_final = final_C5
        best_h_final = best_h_refined.copy()
    else:
        best_C5_final = best_C5_refined
        best_h_final = best_h_refined.copy()
    
    # Use the single best improvement found
    return (best_h_final.tolist(), best_C5_final, n_points)
