import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization: step function
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_gradient(h_vec, max_lag):
        n = len(h_vec)
        grad = np.zeros(n)
        if max_lag >= 0:  # Positive lag
            grad[max_lag:] = -h_vec[:-max_lag]
            grad[:max_lag] = 0
        else:  # Negative lag
            grad[:max_lag] = -h_vec[-max_lag:]
            grad[max_lag:] = 0
        return grad
    
    def compute_gradient_new(h_vec, max_lag):
        n = len(h_vec)
        grad = np.zeros(n)
        # Derivative of sum h[i] * (1 - h[i + k]) w.r.t. h[i]
        # For large k, h[i] affects h[i+k] (if i+k < n)
        # But we want 1 - h[i + k] -- so gradient for h[i] is -1 * delta_{i+k}
        # Normal fit procedure is computing for shifted versions
        # Derive analytical form: for the lag k with max contribution to the sum
        # We compute:
        # For shift k: sum over i of h[i] * (1 - h[i+k]) 
        #                -> gradient of i-th entry = 1 - h[i+k]  
        # For negative lags: h[i] affects (1 - h[i-k])
        # We're already improving max lag case directly. Let's make symmetric derivative.
        # For lag k: contribution to z[i] = h[i] * (1 - h[i+k]) 
        # grad = 1 - h[i+k] * (delta_{i-k} - delta_{i+k}) 
        # Simplify to: for all u in range(n) compute:
        # grad[u] = sum_{i: i+k = u} (1 - h[u]) = (1-h[u]) ...
        # But we implement directly by what the gradient vector does to result
        # We scan h shifts, compute the terms and build gradient.
        # The gradient direction for h[i] to reduce max_lag contribution:
        # When h[i] affects the term h[i] * 1-h[i+k], variance makes that term go down  
        # => grad[i] should influence effect h[i] * (1 - h[i+k])
        # Derivative w.r.t h[i] should reduce that, i.e. - (1 - h[i+k])
        for i in range(n):
            new_grad = 0
            # Check positive lag: i + lag = j -> j - lag = i
            if 0 <= i + max_lag < n:
                # derivative of h[i] * (1 - h[i + max_lag]) w.r.t. h[i]
                new_grad += 0 - (1 - h_vec[i + max_lag])
            # Check negative lag: i - max_lag = j => i = j + max_lag
            if 0 <= i - max_lag < n:
                # derivative of h[i] * (1 - h[i - max_lag]) w.r.t. h[i]
                new_grad += 0 - (1 - h_vec[i - max_lag])
            grad[i] = new_grad
        return grad
    
    def optimize_step(h_current):
        # Step 1: compute overlap function
        corr = correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)  # Convert to actual lag
        
        # Compute gradient
        grad = compute_gradient_new(h_current, max_lag)
        
        # Try several step sizes
        step_sizes = [0.1, 0.05, 0.02, 0.01]
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        
        for alpha in step_sizes:
            h_trial = h_current - alpha * grad
            h_trial = project_feasible(h_trial)
            C5_trial = compute_C5(h_trial)
            if C5_trial < best_C5:
                best_h = h_trial
                best_C5 = C5_trial
                break  # Accept first improving step
        
        return best_h, best_C5
    
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    patience = 0
    alpha = 0.1
    
    while time.time() - start_time < budget_s - 0.1:
        # Optimization loop
        for _ in range(10):
            h_new, C5_new = optimize_step(h_best)
            
            if C5_new < C5_best:
                h_best = h_new
                C5_best = C5_new
                patience = 0
                alpha = 0.1  # reset
            else:
                patience += 1
                if patience > 50:
                    break
                if patience % 10 == 0:
                    alpha *= 0.5
    
    # Final projection and evaluation
    h_best = project_feasible(h_best)
    final_C5 = compute_C5(h_best)
    
    return list(h_best), final_C5, n_points
