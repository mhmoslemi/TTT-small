import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.array(h, dtype=float)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to exact sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Distribute the difference
        diff = target_sum - current_sum
        if diff > 0:
            # Need to increase values
            idx = np.argsort(h)
            for i in range(len(idx)-1, -1, -1):
                if diff <= 0:
                    break
                j = idx[i]
                if h[j] < 1.0:
                    update = min(diff, 1.0 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need to decrease values
            idx = np.argsort(h)
            for i in range(len(idx)):
                if diff >= 0:
                    break
                j = idx[i]
                if h[j] > 0.0:
                    update = max(diff, -h[j])
                    h[j] += update
                    diff -= update
    return h
