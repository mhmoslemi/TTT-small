import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    # Compute the overlap integral using convolution
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def randomized_perturbation(h, perturbation_strength):
    """Apply randomized perturbation to the function"""
    noise = np.random.normal(0, perturbation_strength, len(h))
    return h + noise

def augmented_lagrangian_optimize(h_start, n_points, dx, time_limit):
    """Optimize using augmented Lagrangian approach"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Perturbation settings
    perturbation_strength = 0.05
    max_iterations = 500
    
    for iteration in range(max_iterations):
        if time() - start_time > time_limit:
            break
            
        # Perturb
        h_perturbed = project_to_feasible(randomized_perturbation(best_h, perturbation_strength), n_points)
        
        # Local optimization step
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        # Use SLSQP with bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        
        try:
            res = minimize(obj, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
        except:
            pass
            
        # Adjust perturbation strength
        if iteration % 20 == 0 and iteration > 0:
            perturbation_strength *= 0.95
        
    return best_h, best_c5

    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Sweep through resolutions
    resolutions = [15, 20, 25, 30, 35, 40, 45]
    
    for i, n_points_res in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate if needed
        if len(h_start) != n_points_res:
            x_orig = np.linspace(0, 2, len(h_start))
            x_new = np.linspace(0, 2, n_points_res)
            h_interp = np.interp(x_new, x_orig, h_start)
            h_start = project_to_feasible(h_interp, n_points_res)
        
        dx_res = 2.0 / n_points_res
        time_per_res = (budget_s - (time() - start_time)) / len(resolutions)
        
        # Multiple restarts for better exploration
        best_h_level = h_start.copy()
        best_c5_level = evaluate_erdos_solution(h_start, dx_res)
        
        for _ in range(3):
            h_restart = project_to_feasible(randomized_perturbation(h_start, 0.05), n_points_res)
            h_opt, c5_opt = augmented_lagrangian_optimize(h_restart, n_points_res, dx_res, time_per_res / 3)
            if c5_opt < best_c5_level:
                best_c5_level = c5_opt
                best_h_level = h_opt.copy()
        
        if best_c5_level < best_c5:

            best_c5 = best_c5_level
