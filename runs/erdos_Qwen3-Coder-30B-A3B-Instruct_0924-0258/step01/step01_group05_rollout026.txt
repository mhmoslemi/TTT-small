import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Use full convolution to compute all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx

    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec

    def coordinate_descent(h_vec, max_iter=30):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        no_improve_count = 0
        step_size = 0.05
        
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            improved = False
            # Try moving each point up and down
            for i in range(len(h_current)):
                if time.time() - start_time > budget_s - 0.1:
                    break
                    
                # Try moving up
                h_up = h_current.copy()
                delta = step_size
                if h_up[i] + delta > 1:
                    delta = 1 - h_up[i]
                h_up[i] += delta
                h_up = project_feasible(h_up)
                c5_up = compute_C5(h_up)
                
                # Try moving down
                h_down = h_current.copy()
                delta = step_size
                if h_down[i] - delta < 0:
                    delta = h_down[i]
                h_down[i] -= delta
                h_down = project_feasible(h_down)
                c5_down = compute_C5(h_down)
                
                # Choose the better move
                if c5_up < c5_down and c5_up < best_C5:
                    h_current = h_up
                    best_C5 = c5_up
                    improved = True
                elif c5_down < best_C5:
                    h_current = h_down
                    best_C5 = c5_down
                    improved = True
                    
            if improved:
                best_h = h_current.copy()
                no_improve_count = 0
                if step_size > 0.01:
                    step_size *= 0.95
            else:
                no_improve_count += 1
                if no_improve_count > 10:
                    no_improve_count = 0
                    # Perturb slightly to escape local minimum
                    perturb = np.random.normal(0, 0.02, len(h_current))
                    h_current += perturb
                    h_current = project_feasible(h_current)
                    # Reset step size
                    step_size = 0.05
        return best_h, best_C5

    def swap_optimization(h_vec, max_iter=20):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Try swapping pairs
            for _ in range(5):
                i1, i2 = np.random.choice(len(h_current), 2, replace=False)
                if time.time() - start_time > budget_s - 0.1:
                    break
                    
                # Try the swap
                h_swap = h_current.copy()
                h_swap[i1], h_swap[i2] = h_swap[i2], h_swap[i1]
                h_swap = project_feasible(h_swap)
                c5_swap = compute_C5(h_swap)
                
                if c5_swap < best_C5:
                    h_current = h_swap
                    best_C5 = c5_swap
                    best_h = h_current.copy()
                    
        return best_h, best_C5
    
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Generate diverse initial candidates using Latin Hypercube Sampling
    initial_candidates = []
    for _ in range(5):
        if time.time() - start_time > budget_s - 0.1:
            break
        # Create LHS-like distribution
        h_candidate = np.random.random(n_points) * 0.8 + 0.1  # Keep values reasonable
        # Normalize sum
        h_candidate = project_feasible(h_candidate)
        initial_candidates.append(h_candidate)
    
    # Evaluate these candidates
    for candidate in initial_candidates:
        candidate_C5 = compute_C5(candidate)
        if candidate_C5 < C5_best:
            C5_best = candidate_C5
            h_best = candidate.copy()

    # Main optimization loop
    while time.time() - start_time < budget_s - 0.1:
        # Try local optimizations from current best
        for method_iter in range(10):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Coordinate descent
            h_new, C5_new = coordinate_descent(h_best, max_iter=15)
            if C5_new < C5_best:
                C5_best = C5_new
                h_best = h_new.copy()
                
            # Swap optimization
            h_new, C5_new = swap_optimization(h_best, max_iter=10)
            if C5_new < C5_best:
                C5_best = C5_new
                h_best = h_new.copy()
                
            # Random perturbations
            h_perturbed = h_best + np.random.normal(0, 0.03, n_points)
            h_perturbed = project_feasible(h_perturbed)
            C5_rand = compute_C5(h_perturbed)
            if C5_rand < C5_best:
                C5_best = C5_rand
                h_best = h_perturbed.copy()
    
    # Final validation
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
