import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Normalize to correct sum
    dx = 2.0 / n_points
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Gradient descent parameters
    learning_rate = 0.01
    patience = 50
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s:
        # Compute correlation to find maximizing shift
        corr = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(np.abs(corr))
        
        # Compute gradient (only for the maximizing shift)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max - (n_points - 1)) % n_points
            if 0 <= j < n_points:
                gradient[i] += (1 - h[j])
        
        # Update h with gradient descent
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        
        # Maintain sum constraint
        h = h * (n_points / 2.0) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
            last_improvement = iteration
        else:
            patience_counter += 1
        
        # Adaptive learning rate
        if patience_counter >= patience:
            learning_rate *= 0.9
            patience_counter = 0
            # Add small perturbation to escape local minima
            noise = np.random.uniform(-1e-3, 1e-3, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2.0) / np.sum(h)
        
        iteration += 1
        
        # Break if no significant progress in a long time
        if iteration - last_improvement > 5000 and learning_rate < 1e-6:
            break
    
    # Final evaluation of best solution
    final_c5 = evaluate(best_h, dx)
    
    return (best_h.tolist(), float(final_c5), n_points)
