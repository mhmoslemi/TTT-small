import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

    """Smooth approximation of max function"""
    return 1/beta * np.log(np.sum(np.exp(beta * x)))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 50
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = project_to_constraints(h, dx)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
        
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Augmented Lagrangian with a simple IPS approach 
    # Penalty parameter
    mu = 1.0
    # Multiplier
    lambda_ = 0.0
    # Max iterations
    max_iter = 1000
    # Step size
    alpha = 1e-3
    
    # Track convergence 
    last_c5 = best_c5
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlations
        corr = correlate(h, 1-h, mode='full')
        c5_value = np.max(corr) * dx
        
        if c5_value < best_c5:
            best_h = h.copy()
            best_c5 = c5_value
            
        # Gradient estimation (approximate, using finite differences)
        grad_h = np.zeros_like(h)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_to_constraints(h_plus, dx)
            c5_plus = evaluate(h_plus, dx)
            grad_h[j] = (c5_plus - c5_value) / eps
            
        # Update h
        h_new = h - alpha * grad_h
        # Project
        h_new = project_to_constraints(h_new, dx)
        h = h_new
        
        # Update penalty and multipliers
        constraint_resid = np.sum(h) - n_points/2.0
        lambda_ += mu * constraint_resid
        # Increase penalty if required
        if abs(constraint_resid) > 1e-4:
            mu *= 1.1
            
        if abs(last_c5 - c5_value) < 1e-6:
            break
        last_c5 = c5_value

    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    return best_h.tolist(), final_c5, len(best_h)
