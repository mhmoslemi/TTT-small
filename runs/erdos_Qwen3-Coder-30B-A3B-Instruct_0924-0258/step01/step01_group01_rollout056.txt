import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def smooth_max(x, beta=10.0):
    return np.log(np.sum(np.exp(beta * x))) / beta

def compute_gradient(h, dx):
    n = len(h)
    # Compute the correlation of h with 1-h
    corr = correlate(h, 1-h, mode='full')
    # Find the maximum lag
    max_idx = np.argmax(corr)
    # Get the maximum correlation value
    max_corr = corr[max_idx]
    
    # Since we want the derivative of max_(k) int h(x)(1-h(x+k)) dx
    # This is a tricky part, we use the chain rule and a smooth approximation
    
    # Computing the gradient using a nice FFT trick
    # For now, use finite differences to compute a rough gradient
    
    # Create a grid for computing derivative of correlation
    dx_grid = 1e-6
    grad = np.zeros_like(h)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += dx_grid
        h_plus = project_to_constraints(h_plus, dx)
        corr_plus = correlate(h_plus, 1-h_plus, mode='full')
        max_plus = np.max(corr_plus) * dx
        
        h_minus = h.copy()
        h_minus[i] -= dx_grid
        h_minus = project_to_constraints(h_minus, dx)
        corr_minus = correlate(h_minus, 1-h_minus, mode='full')
        max_minus = np.max(corr_minus) * dx
        
        grad[i] = (max_plus - max_minus) / (2 * dx_grid)
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with a low resolution
    n_points = 50
    dx = 2.0 / n_points
    
    # Initialize with initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) > n_points:
            h = np.interp(np.linspace(0, 2, n_points), 
                         np.linspace(0, 2, len(h)), h)
        else:
            h = np.resize(h, n_points)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Optimization parameters
    lr = 1e-3
    max_iter = 1000
    iter_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient (approximate)
        grad = compute_gradient(h, dx)
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_to_constraints(h_new, dx)
        
        # Evaluate new solution
        c5_new = evaluate(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new
        
        iter_count += 1
        
        # Adjust learning rate
        if i % 500 == 0 and i > 0:
            lr *= 0.5
            
    # Final check with full evaluation
    final_c5 = evaluate(best_h, dx)
    
    return best_h.tolist(), final_c5, len(best_h)
