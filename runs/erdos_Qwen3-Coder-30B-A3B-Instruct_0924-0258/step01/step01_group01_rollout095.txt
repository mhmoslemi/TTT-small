import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def compute_gradient(h, dx, critical_lags):
    # Compute gradients for critical lags
    n = len(h)
    grad = np.zeros(n)
    # For each critical lag k, contribution to gradient of h[i] is -1 if i+k < n, 
    # or 1 if i-k >= 0. More accurately: contributions from corresponding indices
    # For the cross-correlation h * (1-h) at lag k, if we compute it as
    # sum_{i=0}^{n-1} h[i] * (1 - h[i+k]) for k >= 0, then d/dh[i]  = -1 if i+k < n
    # For the forward correlation with mode='full', the result is of length 2*n-1
    # But we already know which lags are most critical
    for k in critical_lags:
        # correlation at lag k (indices from 0 to 2*n-2)
        # The lower index is 0, and the higher is 2*n-2
        # We map this to index of shifted vector: 
        # correlation index 0 corresponds to k=n-1 (h[0](1-h[n-1]) + h[1](1-h[n]) + ...)
        # lag 1 -> k = n-2, lag 2 -> k = n-3, etc.
        # lag n-1 -> k = 0, lag n -> k = -1
        # Going up to 2*n-2-1 = 2*n-3
        # For convolution h , (1-h), we compute index l = k + n - 1, which is from 0 to 2*n-2
        # But lag 0 refers to k = 0, which has 1 h[0](1-h[0]) + h[1](1-h[1]) + ...
        # But now in case of correlation, we're doing cross-correlation h, 1-h
        # So for lag k:
        # h[i] contributes to corr at index i+k, which comes from positions 0 to n-1 for input h
        # Correlation index l = i + k, and l goes from 0 to 2*n-2
        # Correlation at lag k = correlation index k + n - 1
        # Thus for position i, it contributes to the resulting corr(1) * (1-h[i+1]), etc.
        # Let's properly compute it:
        # We have full_convolution between h and (1-h) of length 2*n - 1
        # Index in output is i + j, where i=0..n-1, j=0..n-1, so i+j goes from 0 to 2n-2
        # For correlation h*1-h, the effective lags are -n+1 to n-1, and at lag k = i-j, 
        # the (i-th) element contributes to corr index i-j+n-1 (correlation index)
        # --- Let's work out explicit gradient logic ---
        # At lag k, contribution to corr at position (i+k) is h[i]*(1-h[i+k])  
        # so d/dh[i] = -(1-h[i+k]), but this is only accurate if i+k < n, and possibly
        # d/dh[i+k] = -h[i] (when exist)
        # The gradient we want for h[i] depends on which terms in the correlation 
        # sum involve h[i]. That would be for h[i] * (1-h[i+k]), so the contribution
        # to gradient is -1 + h[i+k] for every valid k where we multiply with h[i]
        # The full gradient at h[i] is sum over valid k
        # for each i, we identify which terms have h[i] in them (and not from other h_j)
        # if h[i] is the i-th element in the forward selection, it multiplies 1-h[i+k]
        # for k = 0 to n-1, and where this is valid for index i+k < n
        # There's an issue in indexing: let's be more explicit by using explicit convolution
        # conv(a,b)[i] = sum_{j=0}^{i} a[j]*b[i-j] when a,b of length n, i < 2*n-1
        # We can also use discreteFourier transform (fft) for faster computation
        # But for gradient we have to work more carefully. 
        # Let's consider k = 0...2n-2 (corresponding to lags -n+1 to n-1) 
        # correlation for h, (1-h) at lag k has a contribution from h[i] to h[i+k] term
        # This is only valid if i+k < n, so the contribution from i to h[i+k] is -(1-h[i+k]) 
        # and contribution to h[i+k] is -h[i]
        # But label from 0 to 2n-2, lag k = l - (n-1), so l = k + n - 1.  
        # So the logic is different: for h[i] to affect everything at lag k,
        # i must be the "former" and h[i+k] the "latter", so l = i + k is the true index.
        # When we compute sum_{i=0}^{n-1} h[i] * (1 - h[i+k]), the contribution to h[j] is 
        # (If i == j, then -1 + h[j+k], but also if i == j+k we get h[j] contribution.
        # Pathetic attempt at gradient details - we'll do it differently via numerical: 
        # Compute using finite differences to estimate gradient directly.
        # Let us define a cleaner gradient estimator
        # Gradient only from lag k: for h[i], it affects cross correlation at indices
        # where i = some_0, i+1, i+2, ..., i+k   or the result index (i + (n-k-1))
        # really, if we compute for fixed lag k the correlation with h*1-h:
        # from comparison:
        # r[k] = sum_{i=0}^{n-1} h[i]*1-h[i+k], 
        # d h[i]  = - (1 - h[i+k])  if i+k < n
        # d h[i+k] =  - h[i] if i < n. 
        # So when we apply this contribution, we must state this as a gradient error.
        # At index k in the correlation array (which has length 2n-1) we are concerned
        # with the property -1 if i+k in [0,n), multiply by 1-h[i+k], 
        # but need to get corresponing d/dr[k] = -1 + h[i+k] only if we look at cross-term, 
        # we pull the derivative inside:  
        # d/dh[i] term at the k-th lag is (1 - h[i+k]) if valid? No we are computing
        # C_k = sum over i h[i] * (1 - h[i+k])
        # dh[i] = - (1-h[i+k]) if i+k < n
        # dh[i+k] = -h[i] if i < n
        # So gradient for current h[i] is derivative of sum = - (1-h[i+k]) for all valid k. 
        # This is incremental, we go through all elements and accumulated gradients.
        # We will resort to finite difference to get smooth estimate of gradient.
        # But, let's not try to formulate it analytically. We use finite difference.
        # Gradient is from directional derivative of C5.
        # We will numerically approximate gradient by finite difference with small epsilon.
        pass
    
    # Use numerical gradient: estimate using finite differences
    # We don't want to compute full_gradient all at once, which is difficult.
    # So we do a slightly less perfect but cleaner approach:
    # First, we only consider the maximal/largest lag contributions -- those that are critical
    # We've already identified critical_lags from(best, dx) 
    # From them, construct effective contribution to the gradient.
    # We'll compute these manually  
    # Let's compute CG by forward differences based on approximation
    # For a function f(h) = max_k corr_k(h), we approximate the gradient.
    eps = 1e-8
    grad = np.zeros_like(h)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_constraints(h_plus, dx)
        f_plus = evaluate(h_plus, dx)
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_constraints(h_minus, dx)
        f_minus = evaluate(h_minus, dx)
        grad[i] = (f_plus - f_minus) / (2 * eps)
    # Normalize by max to make learning rate better:
    norm = np.linalg.norm(grad)
    if norm > 0:
        grad = grad / norm
    return grad
