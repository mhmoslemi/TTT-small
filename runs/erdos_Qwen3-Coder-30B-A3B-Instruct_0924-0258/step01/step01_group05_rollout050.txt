import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Initialize with a simple construction
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def coordinate_descent_step(h_vec, step_sizes=[0.05, 0.1]):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        improved = True
        
        while improved and time.time() - start_time < budget_s - 0.1:
            improved = False
            for i in range(len(h_current)):
                orig_val = h_current[i]
                for step in step_sizes:
                    # Try positive step
                    h_test = h_current.copy()
                    h_test[i] = min(1.0, orig_val + step)
                    h_test = project_feasible(h_test)
                    C5_test = compute_C5(h_test)
                    if C5_test < best_C5:
                        best_C5 = C5_test
                        best_h = h_test.copy()
                        improved = True
                    # Try negative step
                    h_test = h_current.copy()
                    h_test[i] = max(0.0, orig_val - step)
                    h_test = project_feasible(h_test)
                    C5_test = compute_C5(h_test)
                    if C5_test < best_C5:
                        best_C5 = C5_test
                        best_h = h_test.copy()
                        improved = True
                h_current = best_h.copy()
        return best_h, best_C5
    
    def swap_optimization(h_vec, num_swaps=10):
        h_current = h_vec.copy()
        best_C5 = compute_C5(h_current)
        for _ in range(num_swaps):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Pick two random indices
            i, j = np.random.randint(0, len(h_current), 2)
            if i != j:
                # Try swapping values
                h_test = h_current.copy()
                h_test[i], h_test[j] = h_test[j], h_test[i]
                h_test = project_feasible(h_test)
                C5_test = compute_C5(h_test)
                if C5_test < best_C5:
                    best_C5 = C5_test
                    h_current = h_test.copy()
        return h_current, best_C5
    
    def latin_hypercube_sample(n, n_samples):
        samples = np.zeros((n_samples, n))
        for i in range(n):
            # Create Latin hypercube samples
            samples[:, i] = np.random.permutation(np.linspace(0, 1, n_samples))
        return samples
    
    # Generate multiple initial candidates
    n_candidates = 5
    candidates = []
    for _ in range(n_candidates):
        if time.time() - start_time > budget_s - 0.1:
            break
        # Generate a Latin hypercube sample
        sample = latin_hypercube_sample(n_points, 1)[0]
        # Scale to [0, 1] range
        h_sample = sample * 1.0
        h_sample = project_feasible(h_sample)
        candidates.append(h_sample)
    
    if not candidates:
        candidates = [h.copy()]
        
    h_best = candidates[0]
    C5_best = compute_C5(h_best)
    
    # Optimization loop
    iterations = 0
    no_improvement_count = 0
    while time.time() - start_time < budget_s - 0.1:
        if iterations % 20 == 0:
            # Create new random candidate occasionally
            new_sample = latin_hypercube_sample(n_points, 1)[0]
            h_new = new_sample * 1.0
            h_new = project_feasible(h_new)
            C5_new = compute_C5(h_new)
            if C5_new < C5_best:
                h_best = h_new
                C5_best = C5_new

        # Loop through all candidates and optimize
        for i, h_candidate in enumerate(candidates):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Coordinate descent step
            h_opt, C5_opt = coordinate_descent_step(h_candidate)
            if C5_opt < C5_best:
                C5_best = C5_opt
                h_best = h_opt.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            # Try swapping optimization
            h_opt_swap, C5_opt_swap = swap_optimization(h_candidate)
            if C5_opt_swap < C5_best:
                C5_best = C5_opt_swap
                h_best = h_opt_swap.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            # Update the candidate
            if C5_opt < C5_opt_swap:
                candidates[i] = h_opt
            else:
                candidates[i] = h_opt_swap
                
        iterations += 1
        
        # Restart if no improvement
        if no_improvement_count > 50:
            h_restart = np.random.rand(n_points)
            h_restart = project_feasible(h_restart)
            candidates = [h_restart] + [np.random.rand(n_points) for _ in range(n_candidates - 1)]
            candidates = [project_feasible(c) for c in candidates]
            no_improvement_count = 0

    # Final check
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
