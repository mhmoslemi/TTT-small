import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available

        n_points = len(h0)

    # Set dx
    dx = 2.0 / n_points
    
    # Initialize
    h = h0.copy().astype(float)
    h = h * (n_points / 2) / np.sum(h)  # normalize sum to n/2
    h = np.clip(h, 0, 1)  # clip to [0,1]
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent parameters
    eta = 0.01
    patience = 0
    best_c5_past = best_c5
    iter_count = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Evaluate current c5 and find maximizing shift
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(np.abs(corr))
        k_max = max_shift_idx - (len(h) - 1)  # convert to shift value
        
        # Compute gradient for the shift that gives maximum overlap
        gradient = np.zeros_like(h)
        if k_max >= 0:
            for i in range(len(h)):
                j = i + k_max
                if j < len(h):
                    gradient[i] += (1 - h[j])
        else:
            for i in range(len(h)):
                j = i + k_max
                if j >= 0:
                    gradient[i] += (1 - h[j])
        
        # Update h using gradient descent
        h = h - eta * gradient
        
        # Project back to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new solution
        current_c5 = evaluate(h, dx)
        
        # Update best if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Adaptive learning rate
        if patience > 50:
            eta *= 0.9
            patience = 0
            # Add small random perturbation
            noise = np.random.uniform(-1e-3, 1e-3, len(h))
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        iter_count += 1
    
    return (best_h.tolist(), float(best_c5), n_points)
