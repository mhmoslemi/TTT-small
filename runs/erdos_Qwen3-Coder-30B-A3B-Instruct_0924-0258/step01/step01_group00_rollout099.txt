import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def greedy_adjustment(h, dx, target=0.38):
    """Perform greedy adjustment to reduce maximum overlap"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    
    # Determine which shift causes the max overlap
    k = max_idx - (len(h) - 1)
    
    # Compute the adjusted indices where the overlap occurs
    # Find where h[i] * h[i+k] contributes significantly
    indices = []
    for i in range(len(h)):
        if 0 <= i + k < len(h):
            indices.append((i, i + k))
    
    # If the overlap is not that large, try reducing other overlaps
    if max_val < 2 * target:
        # Instead, focus on the most contributing overlaps in the full correlation
        sorted_indices = np.argsort(corr)[-10:]  # Top 10
        indices = []
        for idx in sorted_indices:
            k_shift = idx - (len(h) - 1)
            if 0 <= -k_shift < len(h) or 0 <= k_shift < len(h):
                if -k_shift >= 0:
                    indices.append((0, -k_shift))
                else:
                    indices.append((-k_shift, 0))
    
    # Identify the largest contributions
    contributions = []
    for i, j in indices:
        if i >= 0 and j >= 0:
            contributions.append((h[i] * h[j], i, j))
    
    contributions.sort(reverse=True)
    
    # Take top 10% of contributions and reduce h at those indices
    n_to_reduce = max(1, len(contributions) // 10)
    reduced_indices = set()
    for _, i, j in contributions[:n_to_reduce]:
        reduced_indices.add(i)
        reduced_indices.add(j)
    
    # Reduction step
    reduction_factor = min(0.2, (max_val - target) / len(h))
    
    for i in reduced_indices:
        h[i] -= reduction_factor
        h[i] = max(0, h[i])
    
    return h

    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_n_points = n_points
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 200
    
    while time() - start_time < budget_s * 0.95:
        if no_improvement_count > max_no_improvement:
            break
            
        try:
            h_new = h.copy()
            
            # Perform greedy adjustment
            h_new = greedy_adjustment(h_new, dx)
            
            # Project to feasible
            h_new = project_to_feasible(h_new, n_points)
            
            # Compute new c5
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                no_improvement_count = 0
            else:
                # If no improvement, add small noise and try again
                no_improvement_count += 1
                h_new = h + np.random.normal(0, 0.01, n_points)
                h_new = project_to_feasible(h_new, n_points)
                c5_new = evaluate_erdos_solution(h_new, dx)

                    best_h = h_new.copy()

            h = project_to_feasible(h + np.random.normal(0, 0.01, n_points), n_points)
