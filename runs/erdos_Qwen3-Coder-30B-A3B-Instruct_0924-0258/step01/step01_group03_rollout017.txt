import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.array(h, dtype=float)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * (target_sum / len(h))
    else:
        h = h * target_sum / current_sum
        h = np.clip(h, 0, 1)
        # Re-normalize
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            diff = target_sum - current_sum
            idx = np.argsort(h)
            for i in range(len(idx)-1, -1, -1):
                if diff == 0:
                    break
                j = idx[i]
                if h[j] < 1.0 and diff > 0:
                    update = min(diff, 1.0 - h[j])
                    h[j] += update
                    diff -= update
    return h

def gradient_descent_step(h, dx, eta=0.01, smoothing=1):
    """Perform one gradient descent step."""
    corr = correlate(h, 1 - h, mode="full")
    k_max = np.argmax(np.abs(corr))
    # Determine shift from full correlation index
    n = len(h)
    shift = k_max - (2 * n - 2) // 2
    
    # Compute gradient for the maximizing shift
    grad = np.zeros_like(h)
    for i in range(n):
        j = i + shift
        if 0 <= j < n:
            grad[i] += (1 - h[j])
    
    # Update h
    h_new = h - eta * grad
    h_new = project_h(h_new, n / 2.0)
    return h_new
