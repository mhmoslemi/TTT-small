import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def initialize_h(n_points):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) > n_points:
            h = h[:n_points]
        else:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        h = np.random.uniform(0, 1, n_points)
    return project_to_constraints(h, 2.0/n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # Start with small resolution
    n_points = 10
    dx = 2.0 / n_points
    
    # Try multiple resolutions to find good local minima
    resolutions = [10, 20, 40, 70]
    
    for res in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        n_points = res
        dx = 2.0 / n_points
        h = initialize_h(n_points)
        
        # Use gradient-based approach with constraint handling
        def objective(h_flat):
            h = np.clip(h_flat, 0, 1)
            h = project_to_constraints(h, dx)
            return evaluate(h, dx)
            
        def constraint_func(h_flat):
            h = np.clip(h_flat, 0, 1)
            h = project_to_constraints(h, dx)
            return np.sum(h) - n_points / 2.0
            
        # Use L-BFGS-B for subproblem with constraints
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
                
            h_flat = np.clip(h, 0, 1)
            # Small perturbation to avoid getting stuck
            if np.random.random() < 0.7:
                h_flat += np.random.normal(0, 0.01, n_points) 
                
            h = project_to_constraints(h_flat, dx)
            current_c5 = evaluate(h, dx)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
            # Simple gradient estimation using finite differences
            eps = 1e-4
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_constraints(h_plus, dx)
                c5_plus = evaluate(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_to_constraints(h_minus, dx)
                c5_minus = evaluate(h_minus, dx)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
                
            # Step size
            step_size = 0.1 / (1 + np.sum(grad**2)**0.5)
            h -= step_size * grad
            h = project_to_constraints(h, dx)
            
    return best_h.tolist(), best_c5, len(best_h)
