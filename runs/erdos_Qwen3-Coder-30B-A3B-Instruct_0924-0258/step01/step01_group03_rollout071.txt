import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Normalize initial values
    h = h0.copy()
    h = h * (n_points / 2.0) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent settings
    learning_rate = 0.01
    max_iter = 10000
    patience = 50
    patience_counter = 0
    last_improvement = 0
    
    for iter_num in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute cross-correlation to find max overlap
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        max_shift = max_shift_idx - (len(h) - 1)
        
        # Compute gradient
        gradient = np.zeros_like(h)
        if max_shift >= 0:
            for i in range(len(h) - max_shift):
                gradient[i] += (1 - 2 * h[i + max_shift])
        else:
            for i in range(-max_shift, len(h)):
                gradient[i] += (1 - 2 * h[i + max_shift])
        
        # Update using gradient descent
        h_new = h - learning_rate * gradient
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)
        if abs(sum_h - n_points / 2.0) > 1e-10:
            # Adjust to maintain sum constraint
            diff = (n_points / 2.0) - sum_h
            if diff > 0:
                # Need to increase values
                idx = np.argsort(h_new)
                for i in range(len(idx)-1, -1, -1):
                    if diff <= 0:
                        break
                    j = idx[i]
                    if h_new[j] < 1:
                        increase = min(diff, 1 - h_new[j])
                        h_new[j] += increase
                        diff -= increase
            else:
                # Need to decrease values
                idx = np.argsort(h_new)
                for i in range(len(idx)):
                    if diff >= 0:
                        break
                    j = idx[i]
                    if h_new[j] > 0:
                        decrease = min(-diff, h_new[j])
                        h_new[j] -= decrease
                        diff += decrease
        
        h = h_new
        
        # Evaluate and check improvement
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
            last_improvement = iter_num
        else:
            patience_counter += 1
            if patience_counter >= patience:
                # Reduce learning rate
                learning_rate *= 0.5
                patience_counter = 0
                if learning_rate < 1e-6:
                    break
    
    return (best_h.tolist(), float(best_c5), n_points)
