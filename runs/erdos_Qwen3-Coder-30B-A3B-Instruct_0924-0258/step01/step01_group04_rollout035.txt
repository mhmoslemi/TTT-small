import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Objective function for optimization
    def objective(h_vec):
        h_vec = project_h(h_vec)
        return compute_C5(h_vec)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try multiple approaches
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Use differential evolution as primary optimizer
    try:
        result = differential_evolution(
            objective, 
            bounds, 
            maxiter=100, 
            popsize=15, 
            tol=1e-6,
            seed=seed,
            recombination=0.7,
            mutation=(0.5, 1.0)
        )
        if result.success:
            h_candidate = project_h(result.x)
            candidate_C5 = compute_C5(h_candidate)
            if candidate_C5 < best_C5:
                best_C5 = candidate_C5
                best_h = h_candidate.copy()
    except:
        pass
    
    # Local refinement using Nelder-Mead
    if time.time() - start_time < budget_s - 1:
        try:
            # Random restarts for better exploration
            for _ in range(5):
                if time.time() - start_time > budget_s - 1:
                    break
                h_start = np.random.rand(n_points) * 0.2 + 0.4  # [0.4, 0.6]
                h_start = project_h(h_start)
                
                result = minimize(
                    objective,
                    h_start,
                    method='Nelder-Mead',
                    options={'maxiter': 500, 'disp': False}
                )
                
                if result.success:
                    h_candidate = project_h(result.x)
                    candidate_C5 = compute_C5(h_candidate)
                    if candidate_C5 < best_C5:
                        best_C5 = candidate_C5
                        best_h = h_candidate.copy()
        except:
            pass
    
    # Final check and refinement
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, n_points)
