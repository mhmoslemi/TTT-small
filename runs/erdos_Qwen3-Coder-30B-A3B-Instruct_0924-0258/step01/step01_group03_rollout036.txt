import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the required sum while keeping values in [0,1]."""
    h = np.array(h, dtype=float)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        return np.ones_like(h) * (target_sum / len(h))
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    while abs(diff) > 1e-10:
        idx = np.argsort(h)
        for i in range(len(idx)-1, -1, -1):
            j = idx[i]
            if diff <= 0:
                break
            if h[j] < 1.0:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h
