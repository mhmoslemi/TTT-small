import numpy as np
import time
from scipy.signal import correlate

def compute_C5(h_vec, dx):
    """Compute C5 for given h vector"""
    h_vec = np.array(h_vec)
    corr = correlate(h_vec, 1 - h_vec, mode='full')
    return np.max(corr) * dx

def project_feasible(h_vec, target_sum):
    """Project h onto feasible set: 0 <= h <= 1, sum(h) = target_sum"""
    h_vec = np.clip(h_vec, 0, 1)
    h_sum = np.sum(h_vec)
    if h_sum > 0:
        h_vec = h_vec * target_sum / h_sum
    return h_vec

    h_best = h.copy()
