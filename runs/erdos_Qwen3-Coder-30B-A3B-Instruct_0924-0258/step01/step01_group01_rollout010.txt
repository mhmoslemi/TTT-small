import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def admm_step(h, z, lambda_z, mu, rho, dx, n):
    # Update h: minimize wrt h
    # Lagrangian: corr(h, z) + lambda_z^T (z - (1-h)) + rho/2 ||z - (1-h)||^2 + mu (sum h - n/2)
    # Take derivative wrt h: -corr(1-h, z) + rho (h - (1-z)) + mu = 0
    # So: h = (corr(1-h, z) + rho(1-z) - mu) / rho
    # Simplified: h = (corr(1-h, z) + rho - rho z - mu) / rho
    
    # We'll use a simple gradient descent approach
    h_new = h.copy()
    for _ in range(5):
        # Compute gradients
        corr_h_z = correlate(h_new, z, mode='full')
        corr_h_z = corr_h_z[:n]
        # Simplified gradient assumption, direct optimization
        grad = np.zeros_like(h_new)
        for i in range(n):
            grad[i] = np.sum(corr_h_z[i:] * (1 - h_new[i:])) if i < n else 0  
        
        grad = grad * dx 
        h_new = h_new - 0.01 * grad  # learning rate
        h_new = np.clip(h_new, 0, 1)
    
    # Project to ensure correct sum
    h_new = project_to_constraints(h_new, dx)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 10
    dx = 2.0 / n_points
    max_resolutions = 25  # Limit number of rescaling steps
    current_res = 0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
        h = project_to_constraints(h, dx)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # ADMM parameters
    rho = 1.0
    lambda_z = np.zeros(n_points)
    mu = 0.0
    z = 1 - h
    
    iter_per_res = 100
    max_time_per_res = budget_s / max_resolutions
    
    while current_res < max_resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        res_start_time = time.time()
        
        # ADMM iterations
        for _ in range(iter_per_res):
            if time.time() - start_time > budget_s:
                break
                
            # Update h (using gradient-like approach)
            h_new = h.copy()
            for _ in range(5):
                h_new = admm_step(h_new, z, lambda_z, mu, rho, dx, n_points)
                # In-place projection
                h_new = project_to_constraints(h_new, dx)
            
            # Update z
            z_new = 1 - h_new + lambda_z / rho
            z_new = np.clip(z_new, 0, 1)
            
            # Update multipliers
            lambda_z += rho * (z_new - (1 - h_new))
            mu += 0.1 * (np.sum(h_new) - n_points / 2.0)
            
            h = h_new
            z = z_new
            
            # Check for better solution
            current_c5 = evaluate(h, dx)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
            if time.time() - res_start_time > max_time_per_res * 0.9:
                break
                
        # Try to refine resolution
        if n_points < 200 and best_c5 < 0.3815:
            # Double resolution
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            n_points *= 2
            dx = 2.0 / n_points
            h = h_long
            h = project_to_constraints(h, dx)
            z = 1 - h
            lambda_z = np.zeros(n_points)
            mu = 0.0
            current_res += 1
        else:
            current_res += 1
            
        if time.time() - start_time > budget_s * 0.95:
            break
    
    return best_h.tolist(), best_c5, len(best_h)
