import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint and bounds."""
    h = np.array(h, dtype=float)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    # Ensure final sum is correct
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():

        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5  # Simple uniform start
    
    # Set dx
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize
    h = h0.copy()
    h = project_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent parameters
    lr = 0.01
    decay_factor = 0.9
    patience = 0
    patience_limit = 50
    min_lr = 1e-6
    
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        # Evaluate current C5
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute correlations for all shifts
        corr = correlate(h, 1 - h, mode="full")
        # Find the shift that maximizes overlap
        k_max = np.argmax(np.abs(corr))
        
        # Compute gradient for that shift
        gradient = np.zeros_like(h)
        # Standard shift case
        for i in range(len(h)):
            j = i + k_max - (len(h) - 1)
            if 0 <= j < len(h):
                gradient[i] += (1 - h[j])
        
        # Normalize gradient
        if np.linalg.norm(gradient) > 0:
            gradient = gradient / np.linalg.norm(gradient)
        
        # Update step
        h = h - lr * gradient
        
        # Project to feasible region
        h = project_h(h, target_sum)
        
        # Adapt learning rate
        if iteration % 100 == 0:
            lr = max(lr * decay_factor, min_lr)
        
        # Add noise periodically to escape local minima
        if iteration % patience_limit == 0 and patience > 0:
            h += np.random.normal(0, 1e-3, len(h))
            h = project_h(h, target_sum)
        
        if patience >= patience_limit:
            patience = 0
        patience += 1
    
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    return (best_h.tolist(), float(final_c5), n_points)
