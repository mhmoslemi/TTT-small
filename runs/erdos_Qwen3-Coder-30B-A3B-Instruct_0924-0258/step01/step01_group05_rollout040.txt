import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization with 45 points
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def generate_candidates(n, num_candidates, ref_h=None):
        candidates = []
        for _ in range(num_candidates):
            if ref_h is not None and np.random.random() < 0.7:
                # Perturb reference solution
                h_perturbed = ref_h + np.random.normal(0, 0.05, n)
                h_perturbed = project_feasible(h_perturbed)
            else:
                # Randomly sample
                h_random = np.random.random(n)
                h_random = project_feasible(h_random)
            candidates.append(h_random)
        return candidates
    
    def evaluate_candidates(candidates):
        results = []
        for h in candidates:
            c5 = compute_C5(h)
            results.append((h, c5))
        return results
    
    def refine_candidates(candidates, n):
        refined = []
        sorted_cands = sorted(candidates, key=lambda x: x[1])
        top_20 = sorted_cands[:max(1, len(sorted_cands)//5)]
        
        for h, _ in top_20:
            for _ in range(2):
                h_refined = h + np.random.uniform(-0.1, 0.1, n)
                h_refined = project_feasible(h_refined)
                refined.append(h_refined)
        return refined
    
    # Start with coarse grid
    coarse_sizes = [5, 10, 15]
    fine_sizes = [20, 25, 30]
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Grid search
    for size in coarse_sizes:
        if time.time() - start_time > budget_s - 0.1:
            break
        candidates = generate_candidates(size, 50, best_h if size == coarse_sizes[0] else None)
        results = evaluate_candidates(candidates)
        for h_cand, c5 in results:
            if c5 < best_C5:
                best_C5 = c5
                best_h = h_cand.copy()
        
        # Refine
        refined = refine_candidates(results, size)
        for h_cand in refined:
            if time.time() - start_time > budget_s - 0.1:
                break
            c5 = compute_C5(h_cand)
            if c5 < best_C5:
                best_C5 = c5
                best_h = h_cand.copy()
    
    # Fine grid search
    for size in fine_sizes:
        if time.time() - start_time > budget_s - 0.1:
            break
        candidates = generate_candidates(size, 20, best_h if size == fine_sizes[0] else None)
        results = evaluate_candidates(candidates)
        for h_cand, c5 in results:
            if c5 < best_C5:
                best_C5 = c5
                best_h = h_cand.copy()
    
    # Final local search
    current_h = best_h.copy()
    current_c5 = best_C5
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Random perturbation
        h_perturb = current_h + np.random.normal(0, 0.03, n_points)
        h_perturb = project_feasible(h_perturb)
        
        c5_new = compute_C5(h_perturb)
        if c5_new < current_c5:
            current_h = h_perturb
            current_c5 = c5_new
    
    # Final projection
    final_h = project_feasible(current_h)
    final_c5 = compute_C5(final_h)
    
    return list(final_h), final_c5, n_points
