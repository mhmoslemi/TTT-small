import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize
    h = h0.copy()
    h = h * (n_points / 2.0) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 5000
    patience = 50
    patience_counter = 0
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Find shift that maximizes overlap
        corr = correlate(h, 1 - h, mode="full")
        s_max = np.argmax(corr) - (len(h) - 1)
        
        # Compute gradient: d/dh_j \sum_i h_i (1 - h_{i+s_max})
        # This is 1 - 2 * h_{j-s_max} if j-s_max is valid
        grad = np.zeros_like(h)
        if s_max > 0:
            grad[s_max:] = 1 - 2 * h[:-s_max]
        elif s_max < 0:
            grad[:s_max] = 1 - 2 * h[-s_max:]
        else:
            grad = 1 - 2 * h
            
        # Update with gradient descent
        h_new = h - learning_rate * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        h = h_new
        
        # Evaluate new solution
        current_c5 = evaluate(h, dx)
        
        # Update best if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adaptive learning rate
        if patience_counter > patience:
            learning_rate *= 0.5
            patience_counter = 0
    
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    
    return (best_h.tolist(), float(final_c5), n_points)
