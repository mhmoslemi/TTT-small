import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    # Project h to satisfy sum(h) * dx = 1
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Scale to have the right sum
    h = h * target_sum / current_sum
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust again to ensure correct sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find largest element and increment it
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with low resolution
    n_points = 10
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # We'll try a novel approach: gradient-based optimization
    # On a coarse grid, then refine
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Track best
    current_c5 = evaluate(h, dx)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Try doing gradient-based optimization only on the coarse grid
    
    # Use scipy.optimize with constraints
    def objective(h_flat):
        h = h_flat.copy()
        h = project_to_constraints(h, dx)
        return evaluate(h, dx)
    
    def constraint_sum(h_flat):
        h = h_flat.copy()
        h = project_to_constraints(h, dx)
        return np.sum(h) - n_points / 2.0
    
    # Optimize using SLSQP or L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    h0 = h.copy()
    
    try:
        res = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 500, 'ftol': 1e-8})
        if res.success:
            h_opt = res.x
            h_opt = project_to_constraints(h_opt, dx)
            c5_opt = evaluate(h_opt, dx)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass  # Continue optimization below
    
    # Simulated annealing with adaptive restarts
    T = 0.1
    cooling = 0.95
    iter_per_temp = 500
    iter_count = 0
    
    # Keep refining with greedy improvements
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for _ in range(iter_per_temp):
            if time.time() - start_time > budget_s:
                break
                
            # Perturb with gaussian noise
            h_new = h + np.random.normal(0, T, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_constraints(h_new, dx)
            
            c5_new = evaluate(h_new, dx)
            
            delta = c5_new - current_c5
            
            if delta < 0 or np.random.uniform() < np.exp(-delta / T):
                h = h_new
                current_c5 = c5_new
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    
            iter_count += 1
            
        # Cool down
        T *= cooling
        
        if T < 1e-6:
            # Increment resolution
            if n_points < 50:  # Cap at reasonable resolution
                n_points *= 2
                dx = 2.0 / n_points
                h_long = np.interp(np.linspace(0, 2, n_points), 
                                 np.linspace(0, 2, n_points//2), h)
                h = h_long
                h = project_to_constraints(h, dx)
                current_c5 = evaluate(h, dx)
                T = 0.1
            else:
                break
    
    # If no good solution yet, fallback to simple approach on finest resolution
    if best_h is None or best_c5 == float('inf'):
        n_points = 50
        dx = 2.0 / n_points
        
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = h[:n_points]
        else:
            h = np.random.uniform(0, 1, n_points)
            h = project_to_constraints(h, dx)
        
        for _ in range(1000):
            if time.time() - start_time > budget_s:
                break
            h_new = h + np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_constraints(h_new, dx)
            c5_new = evaluate(h_new, dx)
            
            if c5_new < current_c5:
                h = h_new
                current_c5 = c5_new
                
        best_c5 = current_c5
        best_h = h.copy()
    
    return best_h.tolist(), best_c5, len(best_h)
