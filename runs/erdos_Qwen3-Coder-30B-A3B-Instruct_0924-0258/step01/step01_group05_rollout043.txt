import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def evaluate_candidate(h_vec):
        h_vec = project_feasible(h_vec)
        return compute_C5(h_vec)
    
    def randomized_grid_search(n_points, n_candidates=50, max_iter=5):
        best_h = h.copy()
        best_C5 = compute_C5(best_h)
        
        # Start with a coarse grid
        coarse_n = 15
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
            
            # Sample candidates from random grid
            candidates = []
            for _ in range(n_candidates):
                # Random piecewise constant function
                h_cand = np.random.rand(coarse_n)
                h_cand = project_feasible(h_cand)
                
                # Interpolate to full resolution
                h_full = np.interp(
                    np.linspace(0, 2, n_points),
                    np.linspace(0, 2, coarse_n),
                    h_cand
                )
                h_full = project_feasible(h_full)
                candidates.append(h_full)
            
            # Evaluate each candidate
            for h_cand in candidates:
                C5 = evaluate_candidate(h_cand)
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = h_cand.copy()
            
            # Refine grid
            coarse_n = min(coarse_n * 2, n_points)
            if coarse_n >= n_points:
                break
                
        return best_h, best_C5
    
    def local_improvement(h_start, n_iter=50):
        h_current = project_feasible(h_start.copy())
        best_C5 = compute_C5(h_current)
        best_h = h_current.copy()
        
        for _ in range(n_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Try small perturbations
            h_perturbed = h_current.copy()
            
            # Randomly modify a small number of points
            n_mod = max(1, n_points // 20)
            indices = np.random.choice(n_points, size=n_mod, replace=False)
            for i in indices:
                delta = np.random.normal(0, 0.05)
                h_perturbed[i] += delta
                
            h_perturbed = project_feasible(h_perturbed)
            
            C5_new = compute_C5(h_perturbed)
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h_perturbed.copy()
                h_current = h_perturbed.copy()
                
        return best_h, best_C5
    
    # High-level search
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Grid search phase
    h_grid, C5_grid = randomized_grid_search(n_points, 20, 3)
    if C5_grid < C5_best:
        C5_best = C5_grid
        h_best = h_grid.copy()
        
    # Local improvement from best found
    h_local, C5_local = local_improvement(h_best, 100)
    if C5_local < C5_best:
        C5_best = C5_local
        h_best = h_local.copy()
    
    # Final refinement
    h_final = project_feasible(h_best)
    C5_final = compute_C5(h_final)
    
    if C5_final < C5_best:
        C5_best = C5_final
        h_best = h_final.copy()
        
    return list(h_best), C5_best, n_points
