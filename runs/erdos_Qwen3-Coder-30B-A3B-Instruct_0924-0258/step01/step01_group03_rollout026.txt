import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
        # Ensure it's still feasible after scaling
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5  # Simple uniform start
    h = project_to_feasible(h, n_points / 2.0)
    
    dx = 2.0 / n_points
    
    # Parameters for optimization
    learning_rate = 0.01
    max_iter = 10000
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    patience_counter = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate current state
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Find index of maximum correlation
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        
        # Compute gradient of the objective with respect to h
        # The objective is max_k of the correlation
        # The gradient w.r.t. h[i] is the contribution to the correlation at max_k
        gradient = np.zeros_like(h)
        # For each h[i], the contribution to correlation at shift max_k
        for i in range(len(h)):
            if max_k - i >= 0 and max_k - i < len(h):
                gradient[i] += (1 - h[max_k - i])
            if i - max_k >= 0 and i - max_k < len(h):
                gradient[i] += (1 - h[i - max_k])
        
        # Update h using gradient descent
        h = h - learning_rate * gradient
        
        # Project back to feasible region
        h = project_to_feasible(h, n_points / 2.0)
        
        # Adaptive learning rate
        if patience_counter > 20:
            learning_rate *= 0.95
            
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    
    return (best_h.tolist(), float(final_c5), n_points)
