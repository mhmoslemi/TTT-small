import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_gradient(h_vec, k_max):
        # Compute the gradient for maximum overlap at lag k_max
        n = len(h_vec)
        grad = np.zeros(n)
        if k_max >= 0:
            # Forward correlation
            grad[k_max:] += -h_vec[:-k_max]
            grad[:k_max] += (1 - h_vec[-k_max:])
        else:
            # Backward correlation
            k_abs = -k_max
            grad[:k_abs] += -h_vec[-k_abs:]
            grad[k_abs:] += (1 - h_vec[:k_abs])
        return grad
    
    def optimize_step(h_vec, alpha=0.1):
        # Compute all correlations
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_vec) - 1)
        
        # Compute gradient
        grad = compute_gradient(h_vec, max_lag)
        
        # Try to update
        h_new = h_vec - alpha * grad
        h_new = project_feasible(h_new)
        
        return h_new, max_lag
    
    h_current = h.copy()
    best_h = h_current.copy()
    best_C5 = compute_C5(h_current)
    
    # Main optimization loop
    max_iter = 1000
    iter_count = 0
    no_improvement_count = 0
    alpha = 0.1
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 0.1:
        iter_count += 1
        
        # Try to improve
        h_new, max_lag = optimize_step(h_current, alpha)
        new_C5 = compute_C5(h_new)
        
        if new_C5 < best_C5:
            h_current = h_new.copy()
            best_C5 = new_C5
            best_h = h_current.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 50:
                # Reduce step size
                alpha *= 0.5
                no_improvement_count = 0
                if alpha < 1e-6:
                    break
        
        # Occasionally perturb if stuck
        if no_improvement_count > 20 and np.random.random() < 0.1:
            h_current = project_feasible(h_current + np.random.normal(0, 0.05, n_points))
            no_improvement_count = 0
    
    # Final projection and evaluation
    h_current = project_feasible(h_current)
    final_C5 = compute_C5(h_current)
    
    return list(h_current), final_C5, n_points
