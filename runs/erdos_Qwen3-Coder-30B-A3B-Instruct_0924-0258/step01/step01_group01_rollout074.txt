import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def compute_gradients(h, dx):
    corr = correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    n = len(h)
    # Consider only top contributors to max correlation
    top_contributors = int(max(1, n // 20))  # Top 5%
    critical_lags = np.argsort(corr)[-top_contributors:]
    
    grad = np.zeros_like(h)
    for lag in critical_lags:
        if lag < n:
            # Positive contribution
            grad += 1 - h[lag:]  # h[0] contributes to lag n-1, etc.
        else:
            # Negative contribution  
            lag_neg = lag - n + 1
            grad -= h[:n-lag_neg]  # h[0] contributes to lag n-1, etc.
    
    # Ensure we're minimizing the maximum correlation
    # Scalar normalization
    grad = grad / (np.linalg.norm(grad) + 1e-12)
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start at low resolution
    n_low = 10
    n_high = 70
    n_points = n_low
    dx = 2.0 / n_points
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Parameters
    max_iter_per_res = 1000  # iterations per resolution
    learning_rate = 0.1
    min_lr = 1e-6
    adaptive_lr = True
    
    iter_count = 0
    while True:
        if time.time() - start_time > budget_s * 0.98:
            break
            
        # Inner optimization loop for current resolution
        current_iter = 0
        improvement_count = 0
        
        while current_iter < max_iter_per_res:
            if time.time() - start_time > budget_s:
                break
                
            c5_current = evaluate(h, dx)
            
            # Compute gradients
            grad = compute_gradients(h, dx)
            
            # Update with learning rate
            h_new = h - learning_rate * grad
            
            # Clip to valid range
            h_new = np.clip(h_new, 0, 1)
            
            # Project to satisfy integral constraint
            h_new = project_to_constraints(h_new, dx)
            
            c5_new = evaluate(h_new, dx)
            
            if c5_new < c5_current:
                h = h_new
                if c5_new < best_c5:
                    best_h = h.copy()  
                    best_c5 = c5_new
                improvement_count += 1
            elif improvement_count == 0 and adaptive_lr and learning_rate > min_lr:
                # Reduce learning rate if no improvements
                learning_rate *= 0.8
            elif improvement_count >= 2 and adaptive_lr:
                # Increase learning rate
                learning_rate *= 1.1
                
            current_iter += 1
            iter_count += 1
            
        # Refinement to higher resolution
        if n_points < n_high and iter_count > 100:
            # Interpolate to make h longer
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            learning_rate = 0.1
            improvement_count = 0
        else:
            # No refinement - adjust learning rate
            if not adaptive_lr or learning_rate < min_lr:
                break
                
    return best_h.tolist(), best_c5, len(best_h)
