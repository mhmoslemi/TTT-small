import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_low = 10
    n_high = 50
    n_points = n_low
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Optimized gradient descent approach
    learning_rate = 0.01
    max_iter = 1000
    iter_per_temp = 100
    
    iter_count = 0
    last_c5 = best_c5
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for _ in range(iter_per_temp):
            if time.time() - start_time > budget_s:
                break
                
            # Compute gradient using finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            base_c5 = evaluate(h, dx)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] = min(1.0, h[i] + eps)
                h_plus = project_to_constraints(h_plus, dx)
                c5_plus = evaluate(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] = max(0.0, h[i] - eps)
                h_minus = project_to_constraints(h_minus, dx)
                c5_minus = evaluate(h_minus, dx)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update h
            h_new = h - learning_rate * grad
            h_new = project_to_constraints(h_new, dx)
            
            c5_new = evaluate(h_new, dx)
            
            if c5_new < last_c5:
                h = h_new
                last_c5 = c5_new
                if last_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = last_c5
                    
            iter_count += 1
            
        # Adaptive resolution
        if n_points < n_high and best_c5 < 0.381:
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), best_h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            last_c5 = evaluate(h, dx)
            learning_rate = max(0.001, learning_rate * 0.9)
        
        if time.time() - start_time > budget_s * 0.99:
            break
    
    return best_h.tolist(), best_c5, len(best_h)
