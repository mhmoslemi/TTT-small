import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # GA parameters
    pop_size = 50
    tournament_size = 10
    elitism = int(0.2 * pop_size)
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_init = np.random.rand(n_points)
        h_init = project_feasible(h_init)
        population.append(h_init)
    
    best_C5 = float('inf')
    best_h = None
    patience_counter = 0
    
    # Genetic algorithm main loop
    generation = 0
    while time.time() - start_time < budget_s - 0.1:
        generation += 1
        
        # Evaluate fitness
        fitness = [compute_C5(individual) for individual in population]
        
        # Track best
        current_best_idx = np.argmin(fitness)
        current_best_C5 = fitness[current_best_idx]
        if current_best_C5 < best_C5:
            best_C5 = current_best_C5
            best_h = population[current_best_idx].copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            # Reset some individuals to avoid stagnation
            for i in range(int(pop_size * 0.2)):
                idx = np.random.randint(0, pop_size)
                population[idx] = np.random.rand(n_points)
                population[idx] = project_feasible(population[idx])
            patience_counter = 0
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism: keep best individuals
        new_population = population[:elitism]
        
        # Generate offspring
        while len(new_population) < pop_size:
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Selection
            parents = []
            for _ in range(2):
                tournament = np.random.choice(pop_size, tournament_size, replace=False)
                best_in_tourney = min(tournament, key=lambda i: fitness[i])
                parents.append(population[best_in_tourney])
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                child = np.where(np.random.rand(n_points) < 0.5, parents[0], parents[1])
            else:
                # Copy one parent
                child = parents[0].copy()
                
            # Mutation
            if np.random.rand() < mutation_rate:
                # Gaussian mutation
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = project_feasible(child)
            elif np.random.rand() < 0.3:  # 30% chance of reset mutation
                # Reset mutation
                reset_indices = np.random.choice(n_points, int(0.05 * n_points), replace=False)
                child[reset_indices] = np.random.choice([0, 1], len(reset_indices))
                child = project_feasible(child)
            
            new_population.append(child)
            
        population = new_population[:pop_size]
        
        # Early termination if we've improved significantly
        if generation > 100 and best_C5 < 0.381:
            break
    
    # Final refinement with local search
    if best_h is not None:
        # Try local descent from the best GA solution
        h_current = best_h.copy()
        current_C5 = compute_C5(h_current)
        
        for _ in range(50):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (len(h_current) - 1)
            
            grad = np.zeros_like(h_current)
            if max_lag >= 0:
                grad[max_lag:] = -h_current[:-max_lag]
            else:
                grad[:max_lag] = -h_current[-max_lag:]
                
            alpha = 0.1
            candidate_h = h_current - alpha * grad
            candidate_h = project_feasible(candidate_h)
            new_C5 = compute_C5(candidate_h)
            
            if new_C5 < current_C5:
                h_current = candidate_h.copy()
                current_C5 = new_C5
            else:
                # Try random perturbation
                noise = np.random.normal(0, 0.02, n_points)
                h_current += noise
                h_current = project_feasible(h_current)
    
    final_C5 = compute_C5(h_current)
    
    return list(h_current), final_C5, n_points
