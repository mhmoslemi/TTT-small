import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default initial vector
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def conv_constraint(h_vec, k):
        # Compute correlation at lag k
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return corr[k] * dx
    
    # Binary search on the upper bound
    low = 0.37
    high = compute_c5(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start_time = time.time()
    
    while high - low > 1e-4:
        if time.time() - start_time > budget_s:
            break
            
        mid = (low + high) / 2.0
        
        # Formulate convex feasibility problem
        h_var = cp.Variable(n_points)
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        
        # Add overlap constraints
        for k in range(2 * n_points - 1):
            # This is an approximation of the constraint
            # In reality, computing exact correlations for all k can be O(n^2)
            # We only add a few key constraints to improve bound
            if k < n_points:  # This is a more practical constraint subset
                left_idx = max(0, k)
                right_idx = min(n_points, n_points - k)
                if right_idx > left_idx:
                    corr_sum = cp.sum(
                        cp.multiply(
                            h_var[left_idx:right_idx],
                            1 - h_var[left_idx:right_idx]
                        )
                    )
                    # Adjust indexing for the shifted case
                    shift = k
                    if shift < n_points:
                        max_corr = corr_sum * dx
                    else:
                        max_corr = corr_sum * dx
                    constraints.append(max_corr <= mid)
        
        objective = cp.Minimize(0)  # Dummy objective
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                h_new = project_h(h_new)
                new_c5 = compute_c5(h_new)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                high = mid
            else:
                low = mid
        except:
            low = mid
    
    # Final optimization with gradient-based method if needed
    if time.time() - start_time < budget_s * 0.9:
        # Try a few optimization steps with projected gradient descent
        h_opt = best_h.copy()
        
        def loss_and_grad(h_vec):
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            k_max = np.argmax(corr)
            max_corr = corr[k_max]
            loss = max_corr * dx
            
            # Compute gradient
            grad = np.zeros_like(h_vec)
            eps = 1e-6
            for i in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                grad[i] = (corr_plus[k_max] - max_corr) / eps
            return loss, grad
        
        lr = 0.01
        for i in range(200):
            if time.time() - start_time > budget_s:
                break
            loss, grad = loss_and_grad(h_opt)
            h_opt = h_opt - lr * grad
            h_opt = project_h(h_opt)
            new_loss = compute_c5(h_opt)
            if new_loss < best_c5:
                best_c5 = new_loss
                best_h = h_opt.copy()
            lr *= 0.995
            if i % 50 == 0:
                lr = min(lr, 0.1)
    
    return (best_h.tolist(), best_c5, n_points)
