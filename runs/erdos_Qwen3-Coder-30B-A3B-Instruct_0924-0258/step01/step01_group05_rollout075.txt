import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def generate_initial_candidates(n_candidates, n_points):
        candidates = []
        for _ in range(n_candidates):
            # Latin hypercube sampling in [0,1] for initial diverse points
            h_sample = np.random.random(n_points)
            h_sample = project_feasible(h_sample)
            candidates.append(h_sample.copy())
        return candidates
    
    def coordinate_descent_step(h_current, step_size=0.05):
        h_new = h_current.copy()
        best_C5 = compute_C5(h_current)
        best_h = h_current.copy()
        
        # Try small coordinate-wise changes in both directions
        for i in range(n_points):
            # Try positive step
            new_h = h_new.copy()
            new_h[i] += step_size
            new_h = project_feasible(new_h)
            new_C5 = compute_C5(new_h)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = new_h.copy()
            
            # Try negative step
            new_h = h_new.copy()
            new_h[i] -= step_size
            new_h = project_feasible(new_h)
            new_C5 = compute_C5(new_h)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = new_h.copy()
                
        return best_h, best_C5
    
    def swap_optimization(h_current, n_swaps=10):
        h_new = h_current.copy()
        best_C5 = compute_C5(h_current)
        best_h = h_current.copy()
        
        for _ in range(n_swaps):
            if time.time() - start_time > budget_s - 0.1:
                break
            
            # Randomly sample two indices
            i, j = np.random.choice(n_points, 2, replace=False)
            # Try swapping values
            new_h = h_new.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
            new_h = project_feasible(new_h)
            new_C5 = compute_C5(new_h)
            
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = new_h.copy()
                
        return best_h, best_C5
    
    def simplex_search(h_current, simplex_size=3):
        h_new = h_current.copy()
        best_C5 = compute_C5(h_current)
        best_h = h_current.copy()
        
        # Generate simplex around current point
        simplex = [h_current.copy()]
        for i in range(simplex_size - 1):
            perturb = np.random.normal(0, 0.02, n_points)
            h_perturbed = h_new + perturb
            h_perturbed = project_feasible(h_perturbed)
            simplex.append(h_perturbed.copy())
            
        # Evaluate all simplex points
        for h_point in simplex:
            new_C5 = compute_C5(h_point)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_point.copy()
                
        return best_h, best_C5
    
    # Generate initial candidates
    candidates = generate_initial_candidates(5, n_points)
    best_C5 = compute_C5(candidates[0])
    h_best = candidates[0].copy()
    
    # Find best among initial candidates
    for candidate in candidates[1:]:
        c5 = compute_C5(candidate)
        if c5 < best_C5:
            best_C5 = c5
            h_best = candidate.copy()
    
    # Local optimization loop
    no_improve_count = 0
    step_size = 0.05
    
    while time.time() - start_time < budget_s - 0.1:
        old_best = best_C5
        improvement = False
        
        # Try several optimization strategies
        for _ in range(20):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Strategy 1: coordinate descent
            h_opt1, c5_opt1 = coordinate_descent_step(h_best, step_size)
            if c5_opt1 < best_C5:
                h_best = h_opt1.copy()
                best_C5 = c5_opt1
                improvement = True
                
            # Strategy 2: swap optimization
            h_opt2, c5_opt2 = swap_optimization(h_best, 10)
            if c5_opt2 < best_C5:
                h_best = h_opt2.copy()
                best_C5 = c5_opt2
                improvement = True
                
            # Strategy 3: simplex search
            h_opt3, c5_opt3 = simplex_search(h_best, 3)
            if c5_opt3 < best_C5:
                h_best = h_opt3.copy()
                best_C5 = c5_opt3
                improvement = True
            
            # Occasionally perturb aggressively
            if np.random.random() < 0.3:
                perturbed = h_best + np.random.normal(0, 0.05, n_points)
                h_best = project_feasible(perturbed)
                    
        # Adjust step size if needed
        if not improvement:
            no_improve_count += 1
            if no_improve_count > 10:
                step_size = min(step_size * 1.2, 0.2)  # Increase step size gradually
        else:
            no_improve_count = 0
        
        # Occasionally generate a new candidate and restart search
        if np.random.random() < 0.1 and time.time() - start_time < budget_s - 20:
            new_candidate = np.random.random(n_points)
            new_candidate = project_feasible(new_candidate)
            c5_new = compute_C5(new_candidate)
            if c5_new < best_C5:
                h_best = new_candidate
                best_C5 = c5_new
    
    # Final projection to ensure constraints
    h_best = project_feasible(h_best)
    best_C5 = compute_C5(h_best)
    
    return list(h_best), best_C5, n_points
