import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def gradient_descent_step(h, dx, learning_rate=1e-3):
    n = len(h)
    h = h.copy()
    
    # Compute correlation to find gradient direction
    corr = correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    
    # Create perturbation in direction of negative gradient
    # This is a simple heuristic - in practice, we'd use proper gradients
    # But for simplicity, we try perturbing around the problematic k
    
    # For optimization, we construct a small perturbation
    pert = np.zeros_like(h)
    # Perturb at positions that affect the max correlation
    start_idx = max_idx - n + 1
    if start_idx < 0:
        start_idx = 0
    for i in range(start_idx, min(start_idx + n, n)):
        pert[i] += 1e-4
    
    h_new = h - learning_rate * pert
    h_new = project_to_constraints(h_new, dx)
    return h_new

def optimize_h(h, dx, max_iter=1000):
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    for i in range(max_iter):
        h_new = gradient_descent_step(h, dx)
        c5_new = evaluate(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        if c5_new < h_new.sum() * dx * 0.5:  # Prevent degenerate case
            h = h_new
        else:
            h = h_new
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_low = 10
    n_high = 500
    n_points = n_low
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) > n_points:
            h = h[:n_points]
        else:
            h = np.interp(np.linspace(0, 2, n_points), 
                          np.linspace(0, 2, len(h)), h)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    iter_count = 0
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        last_h = h.copy()
        h, c5 = optimize_h(h, dx, max_iter=100)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        iter_count += 1
        
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Adaptive refinement
        if n_points < n_high and iter_count % 10 == 0:
            improvement = (evaluate(last_h, dx) - c5) / evaluate(last_h, dx)
            if improvement > 0.05:  # 5% improvement
                n_points *= 2
                # Interpolate
                h_long = np.interp(np.linspace(0, 2, n_points), 
                                  np.linspace(0, 2, n_points // 2), h)
                h = h_long
                dx = 2.0 / n_points
                # Project to ensure constraints
                h = project_to_constraints(h, dx)
                last_h = h.copy()
    
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    return best_h.tolist(), final_c5, len(best_h)
