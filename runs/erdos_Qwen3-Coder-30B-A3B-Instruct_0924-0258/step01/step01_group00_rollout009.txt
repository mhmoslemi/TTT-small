import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(30) * 0.5
    
    n_points = len(h_start)
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Resolution levels to try
    resolutions = [20, 30, 40, 50, 60]
    
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n_points
        
        # Interpolate if necessary
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Time allocation for optimization
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Multiple local optimizations
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Slightly perturbed starting point
            h_perturbed = project_to_feasible(
                h_start + np.random.normal(0, 0.02, n_points), n_points)
            
            # Direct minimization approach with bounds
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            
            # Try different optimization methods
            try:
                res = minimize(
                    lambda h: evaluate_erdos_solution(project_to_feasible(h, n_points), dx),
                    h_perturbed,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 20, 'ftol': 1e-6}
                )
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5 = evaluate_erdos_solution(h_opt, dx)
                    if c5 < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5
            except:
                pass
            
            # Add additional random perturbations to escape local minima
            h_perturbed = project_to_feasible(
                h_start + np.random.normal(0, 0.05, n_points), n_points)
            
            try:
                res = minimize(
                    lambda h: evaluate_erdos_solution(project_to_feasible(h, n_points), dx),
                    h_perturbed,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 20, 'ftol': 1e-6}
                )
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5 = evaluate_erdos_solution(h_opt, dx)
                    if c5 < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5
            except:
                pass
            
            # Proceed with next resolution level only after a few restarts
            if time() - start_time > time_per_res:
                break
                
        h_start = best_h.copy()
    
    return (list(best_h), best_c5, len(best_h))
