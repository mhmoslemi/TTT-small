import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def nelder_mead_step_simplex(points, values, alpha=1.0, beta=2.0, gamma=0.5, delta=0.5):
    """Perform one Nelder-Mead step."""
    n = len(points[0])
    # Order points by function values
    indices = np.argsort(values)
    points = [points[i] for i in indices]
    values = [values[i] for i in indices]
    
    # Calculate centroid (excluding worst point)
    centroid = np.mean(points[:-1], axis=0)
    
    # Reflection
    reflected = centroid + alpha * (centroid - points[-1])
    reflected = project_to_feasible(reflected, n/2.0)
    reflected_val = evaluate(reflected, 2.0/n)
    
    if values[0] <= reflected_val < values[-2]:
        points[-1] = reflected
        values[-1] = reflected_val
    elif reflected_val < values[0]:
        # Expansion
        expanded = centroid + beta * (reflected - centroid)
        expanded = project_to_feasible(expanded, n/2.0)
        expanded_val = evaluate(expanded, 2.0/n)
        if expanded_val < reflected_val:
            points[-1] = expanded
            values[-1] = expanded_val
        else:
            points[-1] = reflected
            values[-1] = reflected_val
    else:
        # Contraction
        if reflected_val < values[-1]:
            # Outside contraction
            contracted = centroid + gamma * (reflected - centroid)
        else:
            # Inside contraction
            contracted = centroid + gamma * (points[-1] - centroid)
        
        contracted = project_to_feasible(contracted, n/2.0)
        contracted_val = evaluate(contracted, 2.0/n)
        if contracted_val < values[-1]:
            points[-1] = contracted
            values[-1] = contracted_val
        else:
            # Shrink
            for i in range(1, len(points)):
                points[i] = points[0] + delta * (points[i] - points[0])
                points[i] = project_to_feasible(points[i], n/2.0)
                values[i] = evaluate(points[i], 2.0/n)
    
    return points, values

    simplex = [h0.copy()]
    for _ in range(simplex_size - 1):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points/2.0)
        simplex.append(h)
    
    # Evaluate initial simplex
    values = [evaluate(h, 2.0/n_points) for h in simplex]
    
    best_h = None
    best_c5 = float('inf')
    iteration = 0
    
    while time.time() - start_time < budget_s:
        # Update best
        min_val = min(values)
        if min_val < best_c5:
            best_c5 = min_val
            best_h = simplex[np.argmin(values)].copy()
        
        # Step
        simplex, values = nelder_mead_step_simplex(simplex, values)
        
        # Mutation
        if iteration % 50 == 0:
            for i in range(len(simplex)):
                if np.random.random() < 0.3:
                    h = simplex[i] + np.random.normal(0, 0.01, n_points)
                    simplex[i] = project_to_feasible(h, n_points/2.0)
                    values[i] = evaluate(simplex[i], 2.0/n_points)
        
        iteration += 1

        final_c5 = evaluate(best_h, 2.0/n_points)
