import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # SDP solver for small n_points
    def solve_sdp(h_init, n_points):
        # Create variables
        h = cp.Variable(n_points)
        X = cp.Variable((n_points, n_points), symmetric=True)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            X >> 0,  # X is positive semidefinite
            cp.diag(X) == h,  # Diagonal of X equals h
            0 <= h, h <= 1,  # Bounds on h
            cp.sum(h) == target_sum,  # Sum constraint
        ]
        
        # Overlaps constraint: t <= sum of correlations for all k != 0
        for k in range(1, n_points):
            # Sum of X_{i, i+k} over all valid i
            corr_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_sum += X[i, j]
            constraints.append(t <= corr_sum)
            
        # Objective
        objective = cp.Maximize(t)
        
        # Create and solve problem
        prob = cp.Problem(objective, constraints)
        
        # Solve with MOSEK
        prob.solve(solver=cp.MOSEK, verbose=False, feastol=1e-6, abstol=1e-6, reltol=1e-6, max_iter=100)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_opt = t.value
            return h_opt, c5_opt
        else:
            return None, None
    
    # Helper to get overlaps for gradient estimation
    def get_overlaps(h_vec):
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return overlaps
    
    # Use the hybrid optimization approach
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # This will be the final output
    final_h = best_h.copy()
    final_C5 = best_C5
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        elapsed = time.time() - start_time
                
        # Periodic SDP refinement at smaller resolutions
        if iteration % 10 == 0 and n_points > 10:
            try:
                # Try to solve SDP with smaller resolution
                n_small = min(n_points // 2, 20)
                if n_small >= 8:
                    h_small = np.interp(np.linspace(0, n_points-1, n_small), 
                                      np.arange(n_points), best_h)
                    h_sdp, c5_sdp = solve_sdp(h_small, n_small)
                    if h_sdp is not None:
                        # Interpolate back to full resolution
                        h_interp = np.interp(np.linspace(0, n_points-1, n_points), 
                                           np.arange(n_small), h_sdp)
                        h_interp = project_h(h_interp)
                        c5_new = compute_C5(h_interp)
                        if c5_new < best_C5:
                            best_C5 = c5_new
                            best_h = h_interp.copy()
            except Exception:
                pass
                        
        # Local gradient-based search
        # Sample optimization
        dx = 0.001
        gradient = np.zeros_like(best_h)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += dx
            h_plus = project_h(h_plus)
            C_plus = compute_C5(h_plus)
            
            h_minus = best_h.copy()
            h_minus[i] -= dx
            h_minus = project_h(h_minus)
            C_minus = compute_C5(h_minus)
            
            gradient[i] = (C_plus - C_minus) / (2 * dx)
        
        # Update using simple gradient descent
        step_size = 0.01
        new_h = best_h - step_size * gradient
        new_h = project_h(new_h)
        new_C5 = compute_C5(new_h)
        
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = new_h.copy()
        
        if iteration % 50 == 0:
            final_h = best_h.copy()
            final_C5 = best_C5
    
    # Final validation
    final_C5 = compute_C5(final_h)
    if final_C5 < best_C5:
        final_h = best_h.copy()
        final_C5 = best_C5
    
    return (final_h.tolist(), final_C5, n_points)
