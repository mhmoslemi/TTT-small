import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Initialize
    h = h0.copy()
    dx = 2.0 / n_points
    
    # Normalize initial h
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (n_points / 2.0) / h_sum
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Main optimization loop
    iter_count = 0
    last_improvement = 0
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s:
        iter_count += 1
        
        # Gradient descent step
        # Find shift that maximizes correlation
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        shift = max_shift_idx - (len(h) - 1)
        
        # Compute gradient
        gradient = np.zeros_like(h)
        if shift >= 0:
            # h[i] contributes to correlation with (1-h[i+shift])
            for i in range(len(h) - shift):
                gradient[i] -= (1 - 2 * h[i + shift])
        else:
            # shift < 0
            for i in range(-shift, len(h)):
                gradient[i] -= (1 - 2 * h[i + shift])
        
        # Gradient descent update
        h_new = h - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        
        # Project back to feasible region
        h_sum = np.sum(h_new)
        if h_sum != 0:
            h_new = h_new * (n_points / 2.0) / h_sum
        
        # Evaluate new solution
        c5_new = evaluate(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improvement = iter_count
            # Reset learning rate on improvement
            learning_rate = 0.01
        else:
            # Reduce learning rate if no improvement for a while
            if iter_count - last_improvement > 100:
                learning_rate *= 0.95
                if learning_rate < 1e-6:
                    break
        
        h = h_new
        
        # Occasionally re-solve convex problem as a better restart
        if iter_count % 100 == 0:
            # Convex relaxation step
            h_relaxed = h.copy()
            # Simple projection if needed
            h_sum = np.sum(h_relaxed)
            if h_sum != 0:
                h_relaxed = h_relaxed * (n_points / 2.0) / h_sum
            h = h_relaxed
    
    return (best_h.tolist(), float(best_c5), n_points)
