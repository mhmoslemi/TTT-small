import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    max_iter = 5000
    stagnation_count = 0
    last_improvement = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute current overlap
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        c5_current = corr[k_max] * dx
        
        # Analytical gradient computation
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # Gradient of h[i] contribution to correlation at k_max
            # Contribution comes from h[i] * (1 - h[i + k_max]) 
            # so gradient for h[i] is (1 - 2*h[i]) if i+k_max < n_points
            if i + k_max < n_points:
                grad[i] = (1 - 2 * h[i]) * dx
            # Also contribute from h[i - k_max] * (1 - h[i]) if i >= k_max
            if i >= k_max:
                grad[i] += -h[i - k_max] * dx
                
        # Update with gradient
        h = h - lr * grad
        h = project_h(h)
        
        # Compute new C5 and update best
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = iteration
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Anneal learning rate
        lr *= 0.999
        
        # Occasionally add noise to escape local minima
        if iteration % 500 == 0 and stagnation_count > 100:
            noise_magnitude = 0.02
            noise_indices = np.random.choice(len(h), int(0.05 * len(h)), replace=False)
            h[noise_indices] += np.random.uniform(-noise_magnitude, noise_magnitude, len(noise_indices))
            h = project_h(h)
            stagnation_count = 0
        
        # Early termination if not improving much for a long time
        if iteration - last_improvement > 1000:
            break
            
        iteration += 1
        
        # Early termination for safety
        if iteration > max_iter:
            break
    
    return (best_h.tolist(), best_c5, n_points)
