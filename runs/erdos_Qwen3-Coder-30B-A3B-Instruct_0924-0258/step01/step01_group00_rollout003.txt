import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Try multiple resolutions and starting points
    resolutions = [15, 30, 45, 60, 75]
    
    # Initial starting values from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(30) * 0.5
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Try several different initialization strategies
    init_strategies = [
        lambda n: np.ones(n) * 0.5,  # uniform
        lambda n: np.hstack([np.ones(n//2), np.zeros(n - n//2)]) * 2,  # step
        lambda n: np.ones(n) * 0.25,  # low constant
    ]
    
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if needed
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Try different starting points for this resolution
        for strategy in init_strategies:
            if time() - start_time > budget_s * 0.95:
                break
                
            h_current = strategy(n_points)
            h_current = project_to_feasible(h_current, n_points)
            
            # Local optimization for this resolution
            best_h_res = h_current.copy()
            best_c5_res = evaluate_erdos_solution(h_current, dx)
            
            # Tune with local search
            for _ in range(50):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Small random perturbations
                h_pert = project_to_feasible(h_current + np.random.normal(0, 0.05, n_points), n_points)
                
                # Use simpler optimization (trust-constr to avoid issues with SLSQP)
                try:
                    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
                    bounds = [(0, 1) for _ in range(n_points)]
                    
                    res = minimize(lambda h: evaluate_erdos_solution(h, dx), 
                                  h_pert, 
                                  method='trust-constr',
                                  bounds=bounds, 
                                  constraints=cons,
                                  options={'maxiter': 20})
                    
                    if res.success:
                        h_opt = project_to_feasible(res.x, n_points)
                        c5_opt = evaluate_erdos_solution(h_opt, dx)
                        if c5_opt < best_c5_res:
                            best_c5_res = c5_opt
                            best_h_res = h_opt.copy()
                except:
                    pass
                
                h_current = best_h_res.copy()
            
            # Update global best if improvement
            if best_c5_res < best_c5:
                best_c5 = best_c5_res
                best_h = best_h_res.copy()
                best_n_points = n_points
        
        # Update h_start for next resolution
        h_start = best_h_res.copy()
        
    return (list(best_h), best_c5, best_n_points)
