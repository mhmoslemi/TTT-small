import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    n_particles = 100
    inertia_start = 0.7
    inertia_end = 0.4
    c1 = 2.0
    c2 = 2.0
    max_speed = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Start particles near the initial solution
    for _ in range(n_particles):
        perturbation = np.random.normal(0, 0.05, n_points)
        h_candidate = h + perturbation
        h_candidate = project_h(h_candidate)
        particles.append(h_candidate.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Track personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [compute_C5(p) for p in particles]
    
    # Track global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO iterations
    max_iter = 1000
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia linearly
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_num / max_iter)
        
        # Update each particle
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            
            # Update velocity
            v = inertia * velocities[i] + \
                c1 * r1 * (personal_best[i] - particles[i]) + \
                c2 * r2 * (global_best - particles[i])
            
            # Clamp velocity
            v = np.clip(v, -max_speed, max_speed)
            
            # Update position
            new_pos = particles[i] + v
            new_pos = project_h(new_pos)
            
            # Evaluate new position
            new_score = compute_C5(new_pos)
            
            # Update personal best
            if new_score < personal_best_scores[i]:
                personal_best[i] = new_pos.copy()
                personal_best_scores[i] = new_score
                
                # Update global best if needed
                if new_score < global_best_score:
                    global_best = new_pos.copy()
                    global_best_score = new_score
            
            # Set new position and velocity
            particles[i] = new_pos
            velocities[i] = v
            
    # Final check
    final_score = compute_C5(global_best)
    if final_score < global_best_score:
        global_best_score = final_score
        
    return (global_best.tolist(), global_best_score, n_points)
