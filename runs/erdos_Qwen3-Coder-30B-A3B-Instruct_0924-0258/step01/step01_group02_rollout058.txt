import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        initial_h = np.full(45, 1.0/45)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h to satisfy constraint
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def compute_c5(h_vec):
        """Compute the maximum overlap integral"""
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        """Ensure h is in [0,1] and sums to target_sum"""
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def crossover(parent1, parent2):
        """Uniform crossover between two parents"""
        mask = np.random.rand(len(parent1)) > 0.5
        child = np.where(mask, parent1, parent2)
        return project_h(child)
    
    def mutate(h_vec, mutation_rate=0.05):
        """Gaussian mutation with clipping and projection"""
        noise = np.random.normal(0, 0.05, len(h_vec))
        h_mut = h_vec + noise
        return project_h(h_mut)
    
    # GA parameters
    population_size = 50
    generations = 1000
    tournament_size = 3
    mutation_rate = 0.05
    
    # Initialize population
    population = [initial_h.copy()]
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    # Evaluate initial population
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    no_improvement = 0
    
    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Early termination
        if no_improvement > 1000:
            break
            
        # Selection - tournament selection
        new_population = [population[0]]  # Elitism
        
        while len(new_population) < population_size:
            # Tournament selection
            tourn_indices = np.random.choice(population_size, tournament_size)
            tourn_fitness = [fitness[i] for i in tourn_indices]
            winner_idx = tourn_indices[np.argmin(tourn_fitness)]
            
            # Clone or mutate
            if np.random.random() < 0.8:  # 80% chance of crossover
                parent1 = population[winner_idx]
                parent2 = population[np.random.choice(population_size)]
                child = crossover(parent1, parent2)
            else:
                child = mutate(population[winner_idx], mutation_rate)
                
            new_population.append(child)
            
        population = new_population
        fitness = [compute_c5(h) for h in population]
    
    return (best_h.tolist(), best_c5, n_points)
