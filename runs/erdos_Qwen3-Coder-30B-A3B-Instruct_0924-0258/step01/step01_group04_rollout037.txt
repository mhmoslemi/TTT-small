import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.random.uniform(0.4, 0.6, n_points)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Define the optimization problem
    def objective(h_vec):
        return compute_C5(h_vec)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution with bounds
    result = differential_evolution(
        objective,
        bounds,
        maxiter=100,
        popsize=15,
        recombination=0.7,
       mutation=(0.5, 1.0),
        atol=1e-6,
        tol=1e-6,
        seed=seed,
        callback=lambda x, convergence: None
    )
    
    # Refine with local search
    best_h = project_h(result.x)
    best_C5 = compute_C5(best_h)
    
    # Try to improve with Nelder-Mead from several starting points
    for _ in range(5):
        if time.time() - start_time > budget_s - 5:
            break
        # Random perturbation around current best
        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = project_h(h_perturbed)
        
        # Local optimization with Nelder-Mead
        try:
            res = minimize(
                objective,
                h_perturbed,
                method='Nelder-Mead',
                options={'maxiter': 50, 'xtol': 1e-6, 'ftol': 1e-6}
            )
            if res.success:
                current_h = project_h(res.x)
                current_C5 = compute_C5(current_h)
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = current_h
        except:
            pass
    
    # Final refinement with local search
    for i in range(10):
        if time.time() - start_time > budget_s - 2:
            break
        # Stochastic gradient descent with momentum
        h_vec = best_h.copy()
        velocity = np.zeros_like(h_vec)
        lr = 0.01
        momentum = 0.9
        
        for _ in range(20):
            # Compute gradient approximation
            gradient = np.zeros_like(h_vec)
            eps = 1e-5
            for j in range(n_points):
                h_plus = h_vec.copy()
                h_plus[j] += eps
                h_plus = project_h(h_plus)
                C_plus = compute_C5(h_plus)
                
                h_minus = h_vec.copy()
                h_minus[j] -= eps
                h_minus = project_h(h_minus)
                C_minus = compute_C5(h_minus)
                
                gradient[j] = (C_plus - C_minus) / (2 * eps)
            
            # Update with momentum
            velocity = momentum * velocity - lr * gradient
            h_vec = h_vec + velocity
            h_vec = project_h(h_vec)
        
        current_C5 = compute_C5(h_vec)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_vec
    
    return (best_h.tolist(), best_C5, n_points)
