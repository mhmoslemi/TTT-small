import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def gen_mutation(h, mutation_rate=0.1):
    """Generate mutation by adding noise and polynomial adjustment"""
    n = len(h)
    # Add random noise
    noise = np.random.normal(0, mutation_rate, n)
    h_muted = h + noise
    
    # Apply polynomial fit to smooth
    x = np.linspace(0, 2, n)
    coeffs = np.polyfit(x, h_muted, deg=min(3, n-1))
    h_smooth = np.polyval(coeffs, x)
    
    return project_to_feasible(h_smooth, n)

def solve_convex_relaxation(h, n_points, dx):
    """Solve a convex relaxation for improved h"""
    # Create a quadratic penalty form to approximate max
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    
    # Define a convex optimization to maximize A(k) = integral h(x)h(x+k)dx
    # This becomes a semidefinite program or quadratic program.
    # Instead of full SDP, we will solve a simpler convex proxy
    
    # For this simple proxy, just try to shrink the max correlation
    penalty = np.sum(corr**2) * dx**2
    return penalty + 0.1 * (max_val - 0.3)**2
