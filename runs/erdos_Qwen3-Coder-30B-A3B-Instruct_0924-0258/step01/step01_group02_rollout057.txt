import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # Default construction if none provided
        n_points = 45
        h0 = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h0
    h0 = h0 * target_sum / np.sum(h0)
    
    def compute_c5(h_vec):
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Initialize population with diverse individuals
    population_size = 75
    population = []
    
    # Add the initial solution
    population.append(h0.copy())
    
    # Add random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    # Ensure all individuals are feasible
    for i in range(len(population)):
        population[i] = project_h(population[i])
    
    def evaluate_population(pop):
        fitnesses = []
        for h in pop:
            c5 = compute_c5(h)
            fitnesses.append(-c5)  # Negative for maximization
        return np.array(fitnesses)
    
    # GA parameters
    elite_size = population_size // 4
    generations = 0
    max_generations = 2000
    mutation_rate = 0.05
    crossover_rate = 0.8
    stagnation_count = 0
    best_c5_so_far = float('inf')
    
    # Main GA loop
    start_time = time.time()
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness of current population
        fitnesses = evaluate_population(population)
        
        # Find best individual
        best_idx = np.argmax(fitnesses)
        current_best_c5 = -fitnesses[best_idx]
        
        if current_best_c5 < best_c5_so_far:
            best_c5_so_far = current_best_c5
            best_h = population[best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > 1000:
            break
            
        # Sort by fitness (descending)
        sorted_indices = np.argsort(fitnesses)[::-1]
        sorted_pop = [population[i] for i in sorted_indices]
        
        # Select elite
        elite = sorted_pop[:elite_size]
        
        # Create offspring through crossover and mutation
        offspring = []
        while len(offspring) < population_size - elite_size:
            if np.random.rand() < crossover_rate:
                # Tournament selection
                parent1_idx = sorted_indices[np.random.randint(0, population_size//2)]
                parent2_idx = sorted_indices[np.random.randint(0, population_size//2)]
                
                parent1 = population[parent1_idx]
                parent2 = population[parent2_idx]
                
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
                
                # Mutate
                if np.random.rand() < mutation_rate:
                    noise = np.random.normal(0, 0.05, n_points)
                    child1 += noise
                    child1 = project_h(child1)
                    
                if np.random.rand() < mutation_rate:
                    noise = np.random.normal(0, 0.05, n_points)
                    child2 += noise
                    child2 = project_h(child2)
                
                offspring.extend([child1, child2])
            else:
                # Copy an individual from elite
                idx = np.random.randint(0, elite_size)
                mutated = elite[idx].copy()
                if np.random.rand() < mutation_rate:
                    noise = np.random.normal(0, 0.05, n_points)
                    mutated += noise
                    mutated = project_h(mutated)
                offspring.append(mutated)
        
        # Trim offspring to right size
        offspring = offspring[:population_size - elite_size]
        
        # Create new population
        population = elite + offspring
        
        generations += 1
        
        if generations >= max_generations:
            break
            
    return (best_h.tolist(), best_c5_so_far, n_points)
