    h = h * target_sum / np.sum(h)
    
    def evaluate_c5(h_vec):
        """Computes the maximum overlap integral."""
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        """Projects h to feasible region [0,1] with sum = target_sum."""
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    # Phase 1: Optimize using gradient-based method to get close to a good local minimum
    def objective(h_vec):
        return evaluate_c5(h_vec)

    def constraint_bounds(h_vec):
        return np.concatenate([h_vec, 1 - h_vec])
    
    # Use a simple gradient-based optimizer to find a nearby local minimum
    iteration_limit = 3000
    lr = 0.005
    
    # Run optimization with early stopping
    for iter_count in range(iteration_limit):
        now = time.time()
        if now - start_time > budget_s:
            break
            
        # Compute current gradients numerically
        eps = 1e-5
        grad = np.zeros_like(h)

        for i in range(len(h)):
            h_test = h.copy()

        h = h - lr * grad
        h = project_h(h)
        
        # Check for new best
        current_c5 = evaluate_c5(h)

            best_c5 = current_c5

        lr *= 0.9995

    def is_feasible(h_vec, max_c5):
        """Check if h_vec satisfies C5 <= max_c5 via convex programming."""
        # Simplified approach: use a brute-force but convexified check
        # Create a matrix constraint that lowers the overlap
        # This is to verify the overlap constraint numerically with good precision
        
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr) <= max_c5
    
    # Binary search setup
    low = 0.37  # Known lower bound

    # Perform binary search

            break
            
        mid = (low + high) / 2.0

            high = mid
        else:
            low = mid
        
        if (high - low) < tolerance:
            break
    
    # Final check with the final best solution
    final_c5 = min(best_c5, high)
