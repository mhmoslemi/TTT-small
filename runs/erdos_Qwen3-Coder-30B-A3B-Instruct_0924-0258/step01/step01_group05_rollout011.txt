import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization: step function
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Simulated Annealing parameters
    T0 = 0.1
    T_min = 1e-5
    cooling_rate = np.log(10) / budget_s
    sigma = 0.1
    max_iter = 1000
    
    # Initialize
    h_current = h.copy()
    h_current = project_feasible(h_current)
    C5_current = compute_C5(h_current)
    h_best = h_current.copy()
    C5_best = C5_current
    
    # Temperature
    T = T0
    
    # Acceptance tracking
    accept_count = 0
    total_steps = 0
    
    # Main annealing loop
    while time.time() - start_time < budget_s - 0.1:
        # Random perturbation
        h_new = h_current.copy()
        
        # Perturb
        # Select 30% of indices to perturb
        n_perturb = max(1, int(0.3 * n_points))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        noise = np.random.normal(0, sigma, n_perturb)
        h_new[indices] += noise
        
        # Project to feasible region
        h_new = project_feasible(h_new)
        
        # Compute new objective
        C5_new = compute_C5(h_new)
        
        # Accept or reject
        if C5_new < C5_current:
            h_current = h_new
            C5_current = C5_new
            if C5_new < C5_best:
                h_best = h_new.copy()
                C5_best = C5_new
            accept_count += 1
        else:
            delta = C5_new - C5_current
            if np.random.rand() < np.exp(-delta / T):
                h_current = h_new
                C5_current = C5_new
                accept_count += 1
        
        total_steps += 1
        
        # Update temperature
        T = max(T_min, T0 * np.exp(-cooling_rate * (time.time() - start_time)))
        
        # Adjust sigma based on acceptance rate
        if total_steps % 50 == 0 and total_steps > 0:
            acceptance_rate = accept_count / total_steps
            if acceptance_rate < 0.1:
                sigma = min(0.5, sigma * 1.2)
            elif acceptance_rate > 0.4:
                sigma = max(0.01, sigma * 0.8)
            accept_count = 0
            total_steps = 0

    # Final projection to ensure feasibility
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
