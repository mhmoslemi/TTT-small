import numpy as np
from scipy import optimize
import time
try:
    from cvxpy import *
except ImportError:
    # Fallback if cvxpy not available
    def solve_sdp_fallback(h, n_points):
        return h.copy(), 1.0
else:
    def solve_sdp(h, n_points):
        # CVXPY SDP implementation
        h_var = Variable(n_points)
        X = Variable((n_points, n_points), PSD=True)
        t = Variable()
        
        obj = Maximize(t)
        constraints = [
            X[i,i] == h_var[i] for i in range(n_points)
        ]
        constraints += [0 <= h_var[i] for i in range(n_points)]
        constraints += [h_var[i] <= 1 for i in range(n_points)]
        constraints += [sum(h_var) == n_points / 2]
        constraints += [t <= sum(X[i, (i+k) % n_points] for i in range(n_points)) 
                        for k in range(1, n_points)]
        
        prob = Problem(obj, constraints)
        try:
            prob.solve(solver=MOSEK, verbose=False, max_iter=100)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                return h_var.value, prob.value
        except:
            pass
        # Fallback
        return h, 1.0

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    dx = 2.0
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: simple initialization
        n_points = 20
        h = np.ones(n_points) * 0.5
    
    if n_points > 45:
        n_points = 45
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * (dx / n_points)
        return np.max(overlaps)
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    MIN_POINTS = 20
    MAX_POINTS = 45
    
    current_n = n_points
    
    # Try to refine with SDP for small n
    while time.time() - start_time < budget_s - 5 and current_n <= MAX_POINTS:
        current_n = min(current_n + 5, MAX_POINTS)
        if current_n > MAX_POINTS:
            break
            
        # Try SDP-based approach
        try:
            h_sdp, c5_sdp = solve_sdp(h, current_n)
            if not np.isnan(c5_sdp):
                current_C5 = compute_C5(h_sdp)
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = h_sdp.copy()
        except:
            pass
            
        # Gradient descent refinement
        learning_rate = 0.01
        momentum = 0.8
        velocity = np.zeros_like(h)
        
        iteration = 0
        while time.time() - start_time < budget_s - 5 and iteration < 500:
            iteration += 1
            
            # Compute gradient
            h_vec = np.clip(h, 0, 1)
            h_vec = h_vec * (current_n / 2.0) / np.sum(h_vec)
            
            # Approximate gradient using finite differences
            gradient = np.zeros_like(h_vec)
            eps = 1e-4
            for i in range(current_n):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (current_n / 2.0) / np.sum(h_plus)
                C_plus = compute_C5(h_plus)
                
                h_minus = h_vec.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (current_n / 2.0) / np.sum(h_minus)
                C_minus = compute_C5(h_minus)
                
                gradient[i] = (C_plus - C_minus) / (2 * eps)
            
            # Update with momentum
            velocity = momentum * velocity - learning_rate * gradient
            h = h + velocity
            
            # Project
            h = np.clip(h, 0, 1)
            h = h * (current_n / 2.0) / np.sum(h)
            
            # Evaluate
            current_C5 = compute_C5(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
                
            # Adaptive learning rate
            if iteration % 50 == 0:
                learning_rate *= 0.9
                
    # Final shape correction
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points / 2.0) / np.sum(h_final)
    
    final_C5 = compute_C5(h_final)
    
    return (h_final.tolist(), final_C5, n_points)
