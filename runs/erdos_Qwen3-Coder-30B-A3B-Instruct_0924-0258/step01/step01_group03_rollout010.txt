import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.array(h, dtype=float)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize
    current_sum = np.sum(h)
    if np.abs(current_sum) < 1e-12:
        return np.ones_like(h) * (target_sum / len(h))
    h = h * target_sum / current_sum
    # Clip again in case of over/under flow
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available

        n_points = len(h0)
    else:

        h0 = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial projection
    h0 = project_to_feasible(h0, target_sum)
    
    # Nelder-Mead optimization with constraints
    def objective(h):
        h = project_to_feasible(h, target_sum)
        return evaluate_erdos_solution(h, dx)
    
    # Initialize simplex
    simplex_size = n_points + 2
    simplex = [h0]
    for _ in range(simplex_size - 1):
        h = h0 + np.random.normal(0, 0.01, n_points)
        h = project_to_feasible(h, target_sum)
        simplex.append(h)
    
    simplex = np.array(simplex)
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Nelder-Mead main loop
    iteration = 0
    max_iterations = 10000
    while iteration < max_iterations and (time.time() - start_time) < budget_s - 5:
        # Evaluate simplex
        func_values = []
        for i in range(len(simplex)):
            h = simplex[i]
            f = objective(h)
            func_values.append(f)
        
        func_values = np.array(func_values)
        # Order by function values
        order = np.argsort(func_values)
        simplex = simplex[order]
        func_values = func_values[order]
        
        # Update best
        if func_values[0] < best_c5:
            best_c5 = func_values[0]
            best_h = simplex[0].copy()
        
        # Nelder-Mead operations
        x0 = simplex[0]  # Best
        x1 = simplex[1]  # Second best
        x2 = simplex[-1]  # Worst
        
        # Compute centroid (excluding worst)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        xr = centroid + (centroid - x2) * 1.0
        xr = project_to_feasible(xr, target_sum)
        fr = objective(xr)
        
        if func_values[0] <= fr < func_values[-2]:
            simplex[-1] = xr
        elif fr < func_values[0]:
            # Expansion
            xe = centroid + (xr - centroid) * 2.0
            xe = project_to_feasible(xe, target_sum)
            fe = objective(xe)
            if fe < fr:
                simplex[-1] = xe
            else:
                simplex[-1] = xr
        else:
            # Contraction
            xc = centroid + (x2 - centroid) * 0.5
            xc = project_to_feasible(xc, target_sum)
            fc = objective(xc)
            if fc < func_values[-1]:
                simplex[-1] = xc
            else:
                # Reduction
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + (simplex[i] - simplex[0]) * 0.5
        
        # Mutation
        if iteration % 50 == 0:
            for i in range(len(simplex)):
                if np.random.random() < 0.1:
                    simplex[i] += np.random.normal(0, 0.01, n_points)
                    simplex[i] = project_to_feasible(simplex[i], target_sum)
        
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (best_h.tolist(), float(final_c5), n_points)
