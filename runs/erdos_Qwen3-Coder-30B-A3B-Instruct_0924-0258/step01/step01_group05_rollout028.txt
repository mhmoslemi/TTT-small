import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def coordinate_descent_step(h_vec, step_size=0.05, max_iter=50):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        
        for i in range(n_points):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Try small increases and decreases
            for delta in [-step_size, step_size]:
                h_test = h_current.copy()
                h_test[i] += delta
                h_test = project_feasible(h_test)
                test_C5 = compute_C5(h_test)
                if test_C5 < best_C5:
                    best_C5 = test_C5
                    best_h = h_test.copy()
            h_current = best_h.copy()
        return best_h, best_C5
    
    def swap_optimization(h_vec, max_swaps=10):
        h_current = h_vec.copy()
        best_C5 = compute_C5(h_current)
        best_h = h_current.copy()
        
        for _ in range(max_swaps):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Pick two random indices
            i, j = np.random.randint(0, n_points, 2)
            if i == j:
                continue
            h_test = h_current.copy()
            h_test[i], h_test[j] = h_test[j], h_test[i]
            h_test = project_feasible(h_test)
            test_C5 = compute_C5(h_test)
            if test_C5 < best_C5:
                best_C5 = test_C5
                best_h = h_test.copy()
        return best_h, best_C5
    
    def simplex_search(h_vec, simplex_size=3):
        h_current = h_vec.copy()
        best_C5 = compute_C5(h_current)
        best_h = h_current.copy()
        
        # Create simplex around current point
        simplex = [h_current]
        for _ in range(simplex_size - 1):
            h_perturb = h_current + np.random.normal(0, 0.05, n_points)
            h_perturb = project_feasible(h_perturb)
            simplex.append(h_perturb)
        
        # Evaluate simplex
        for h_s in simplex:
            test_C5 = compute_C5(h_s)
            if test_C5 < best_C5:
                best_C5 = test_C5
                best_h = h_s.copy()
        
        return best_h, best_C5
    
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Refined local optimization loop
    no_improvement_count = 0
    while time.time() - start_time < budget_s - 0.1:
        # Try several techniques
        for technique in ['coord', 'swap', 'simplex']:
            if time.time() - start_time > budget_s - 0.1:
                break
                
            if technique == 'coord':
                h_new, C5_new = coordinate_descent_step(h_best, step_size=0.05)
            elif technique == 'swap':
                h_new, C5_new = swap_optimization(h_best, max_swaps=10)
            elif technique == 'simplex':
                h_new, C5_new = simplex_search(h_best, simplex_size=3)
            
            if C5_new < C5_best:
                C5_best = C5_new
                h_best = h_new.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Stuck? Add jitter
        if no_improvement_count > 100:
            h_best = h_best + np.random.normal(0, 0.02, n_points)
            h_best = project_feasible(h_best)
            no_improvement_count = 0
        
        # Occasionally restart from good solution
        if np.random.random() < 0.01:
            # Generate new random solution
            h_new = np.random.rand(n_points)
            h_new = project_feasible(h_new)
            h_best = h_new.copy()
            C5_best = compute_C5(h_best)
            no_improvement_count = 0
    
    # Final validation
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
