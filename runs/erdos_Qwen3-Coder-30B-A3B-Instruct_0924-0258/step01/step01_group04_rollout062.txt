import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Genetic Algorithm Parameters
    population_size = 50
    num_generations = 1000
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 5
    local_search_freq = 50
    
    # Initialize population
    def create_population():
        if 'initial_h_values' in globals():
            population = [np.array(initial_h_values)]
        else:
            population = []
        for _ in range(population_size - len(population)):
            h_new = np.random.uniform(0.4, 0.6, n_points)
            h_new = project_h(h_new)
            population.append(h_new)
        return population
    
    # Fitness function
    def fitness(h_vec):
        return compute_C5(h_vec)
    
    # Tournament selection
    def tournament_selection(population, fitnesses, tournament_size=3):
        selected_idx = np.random.choice(len(population), tournament_size)
        best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
        return population[best_idx]
    
    # Crossover
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        crossover_point = np.random.randint(1, len(parent1) - 1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    # Mutation
    def mutate(h_vec):
        h_mut = h_vec.copy()
        for i in range(len(h_vec)):
            if np.random.rand() < mutation_rate:
                h_mut[i] = np.clip(h_mut[i] + np.random.normal(0, 0.05), 0, 1)
        return h_mut
    
    # Local optimization with Nelder-Mead using best candidates
    def local_search(population, fitnesses, num_candidates=5):
        best_indices = np.argsort(fitnesses)[:num_candidates]
        for i in best_indices:
            h_i = population[i]
            bounds = [(0,1) for _ in range(n_points)]
            # Ensure sum constraint is met during optimization
            def constraint_sum(x):
                return np.sum(x) - target_sum
                
            cons = {'type': 'eq', 'fun': constraint_sum}
            try:
                result = minimize(fitness, h_i, method='Nelder-Mead', bounds=bounds, constraints=cons, 
                                options={'maxiter': 1000, 'adaptive': True})
                if result.success:
                    better_h = result.x.copy()
                    better_h = project_h(better_h)
                    population[i] = better_h
            except:
                pass

    # Main optimization loop
    population = create_population()
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    try:
        for gen in range(num_generations):
            # Check budget
            if time.time() - start_time > budget_s - 1:
                break
            
            # Evaluate population
            fitnesses = [fitness(h_vec) for h_vec in population]
            
            # Find current best
            current_best_idx = np.argmin(fitnesses)
            if fitnesses[current_best_idx] < best_C5:
                best_C5 = fitnesses[current_best_idx]
                best_h = population[current_best_idx].copy()
            
            # Selection, crossover, mutation
            elite = [population[i] for i in np.argsort(fitnesses)[:elite_size]]
            new_population = elite.copy()
            
            while len(new_population) < population_size:
                parent1 = tournament_selection(population, fitnesses)
                parent2 = tournament_selection(population, fitnesses)
                child1, child2 = crossover(parent1, parent2)
                
                child1 = mutate(child1)
                child2 = mutate(child2)
                
                new_population.extend([project_h(child1), project_h(child2)])
                
            population = new_population[:population_size]
            
            # Periodic local search (Nelder-Mead)
            if gen % local_search_freq == 0:
                fitnesses = [fitness(h_vec) for h_vec in population]
                local_search(population, fitnesses)
                
    except Exception as e:
        pass
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, n_points)
