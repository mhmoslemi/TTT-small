import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    resolutions = [30, 45, 60, 75, 90]
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:

        dx = 2.0 / n_points
        
        # Interpolate initial guess if needed
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points),

            h_start = project_to_feasible(h_interp, n_points)
        
        # Try multiple restarts with different perturbations
        best_h_level = h_start.copy()

        # Use more aggressive search with better initializations
        for restart in range(5):
            if time() - start_time > budget_s * 0.9:

            # Start with different perturbations
            if restart == 0:

                eps = 1e-6

                    h_plus = h.copy()
                    h_plus[i] += eps

                    obj_plus = objective(h_plus)
                    grad[i] = (obj_plus - base_obj) / eps
                return grad

                # Limit iterations for faster convergence
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
