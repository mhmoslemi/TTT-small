import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with the parent construction or a fallback
    if 'initial_h_values' in globals():

    else:
        h = np.ones(45) / 2
        
    n_points = len(h)
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    target = 0.38
    max_iter = 1000
    patience = 50
    patience_counter = 0
    
    # Use progressively smaller adjustments over iterations
    adjustment_schedule = np.logspace(0, -2, max_iter)
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        c5 = evaluate_erdos_solution(h, dx)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            break
            
        # Find indices where maximum overlap occurs
        corr = np.correlate(h, 1 - h, mode='full')
        k_idx = np.argmax(corr)
        max_val = corr[k_idx]
        
        # Determine shift k
        k = k_idx - (n_points - 1)
        
        # Augment h with 0s for easier handling
        h_aug = np.concatenate([np.zeros(n_points - 1), h, np.zeros(n_points - 1)])
        
        # Find contributing indices in domain
        indices = np.arange(n_points)
        h_k = h_aug[n_points - 1 + k: 2 * n_points - 1 + k]
        products = h * h_k
        
        # Sort indices by probability of contributing to overlap
        sorted_idx = np.argsort(products)[::-1]
        
        # Take top 10%
        top_indices = sorted_idx[:max(1, n_points // 10)]
        
        # Adjust h values
        adjust_fact = adjustment_schedule[i] * (c5 - target) / n_points
        if adjust_fact > 0:
            for idx in top_indices:
                h[idx] = max(0, h[idx] - adjust_fact)
        
        # Normalize
        h = project_to_feasible(h, n_points)
        
        # Occasionally perturb to escape local optima
        if i % 20 == 0:
            h = project_to_feasible(h + np.random.normal(0, 0.01, n_points), n_points)
    
    return (list(best_h), best_c5, n_points)
