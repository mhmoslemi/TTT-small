import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization: step function with 1s in first half, 0s in second
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Clip values to [0, 1] and rescale to sum to target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Gradient descent with computation of gradient for max overlap
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    alpha = 0.1
    stagnation_count = 0
    
    iter_count = 0
    while time.time() - start_time < budget_s - 0.1:
        iter_count += 1
        if iter_count > 1000:
            break
            
        # Compute all overlaps and find the maximum overlap shift
        corr = correlate(best_h, 1 - best_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(best_h) - 1)
        
        # Compute gradient for this maximum lag
        grad = np.zeros_like(best_h)
        if max_lag >= 0:  # positive lag
            grad[max_lag:] = -best_h[:-max_lag]
        else:  # negative lag
            grad[:max_lag] = -best_h[-max_lag:]
        
        # Try different step sizes
        step_sizes = [0.1, 0.05, 0.01]
        best_new_h = best_h.copy()
        best_new_C5 = best_C5
        
        for step in step_sizes:
            candidate_h = best_h - step * grad
            candidate_h = project_feasible(candidate_h)
            new_C5 = compute_C5(candidate_h)
            if new_C5 < best_new_C5:
                best_new_C5 = new_C5
                best_new_h = candidate_h.copy()
                
        if best_new_C5 < best_C5:
            best_h = best_new_h.copy()
            best_C5 = best_new_C5
            stagnation_count = 0
            alpha = 0.1  # reset step size on improvement
        else:
            stagnation_count += 1
            if stagnation_count > 50:
                # Add small noise to avoid local minima
                noise = np.random.normal(0, 0.005, n_points)
                best_h = best_h + noise
                best_h = project_feasible(best_h)
                stagnation_count = 0
                alpha *= 0.9  # reduce step size
                
    # Final projection and evaluation
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
