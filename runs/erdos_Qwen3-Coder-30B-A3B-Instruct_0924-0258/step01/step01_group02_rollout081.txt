import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        # Fallback: create a reasonable initial guess
        n_points = 45
        initial_h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial solution is valid
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range and rescale
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # GA parameters
    pop_size = 50
    num_generations = 1000
    mutation_rate = 0.05
    crossover_rate = 0.8
    elite_size = pop_size // 4
    
    # Initialize population
    population = []
    population.append(initial_h.copy())
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    best_c5 = compute_c5(initial_h)
    best_h = initial_h.copy()
    
    start_time = time.time()
    
    for gen in range(num_generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = compute_c5(h)
            fitness_scores.append(-c5)  # Negative because we maximize fitness to minimize C5
            
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            # Tournament of size 3
            indices = np.random.choice(pop_size, 3, replace=False)
            best_idx = indices[np.argmax([fitness_scores[i] for i in indices])]
            selected.append(population[best_idx].copy())
        
        # Create new population
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitness_scores)[-elite_size:]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Select two parents
            p1_idx, p2_idx = np.random.choice(pop_size, 2, replace=False)
            p1, p2 = selected[p1_idx], selected[p2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, p1, p2)
                child2 = np.where(mask, p2, p1)
            else:
                child1, child2 = p1.copy(), p2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.05)
            
            # Repair
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population
        
        # Update best solution
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
