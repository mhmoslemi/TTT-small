import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

    alpha = 0.01
    max_iter = 100
    
    # Start at coarse resolution
    n_points = 10
    dx = 2.0 / n_points

        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        
    h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    iter_count = 0
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # ADMM iterations
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            # Update h
            # This approximates minimizing correlation with constraint
            h_old = h.copy()
            
            # Simple gradient-like update using finite differences
            # to decrease max cross-correlation
            corr = correlate(h, 1-h, mode='full')
            max_k = np.argmax(corr)
            # Apply a small negative gradient step
            h_grad = np.zeros_like(h)
            for i in range(len(h)):
                # Compute gradient approximation
                h_plus = h.copy()
                h_plus[i] += 1e-4
                h_plus = project_to_constraints(h_plus, dx)
                corr_plus = correlate(h_plus, 1-h_plus, mode='full')
                corr_plus = np.max(corr_plus)
                
                h_minus = h.copy()
                h_minus[i] -= 1e-4
                h_minus = project_to_constraints(h_minus, dx)
                corr_minus = correlate(h_minus, 1-h_minus, mode='full')
                corr_minus = np.max(corr_minus)
                
                h_grad[i] = (corr_minus - corr_plus) / (2e-4)
            
            h = h - alpha * h_grad
            h = project_to_constraints(h, dx)
            
            # Check c5
            c5 = evaluate(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                
            iter_count += 1
            
        # Resolution doubling
        if n_points >= 10 and n_points < 512:
            new_n = n_points * 2
            h_interp = np.interp(np.linspace(0, 2, new_n),

            h = h_interp
            n_points = new_n
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            c5 = evaluate(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        else:

    return best_h.tolist(), best_c5, len(best_h)
