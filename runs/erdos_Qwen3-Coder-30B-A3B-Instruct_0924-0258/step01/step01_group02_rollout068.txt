import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    
    # Use a coordinate descent approach
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Coordinate descent step
        best_improvement = 0
        best_h_new = best_h.copy()
        best_c5_new = best_c5
        
        # Try modifying each component
        for i in range(n_points):
            if iteration % (n_points * 2) == i:
                # Small perturbation
                h_test = best_h.copy()
                delta = 0.01 * (np.random.rand() - 0.5)
                h_test[i] += delta
                h_test = project_h(h_test)
                
                c5_test = compute_c5(h_test)
                if c5_test < best_c5_new:
                    best_c5_new = c5_test
                    best_h_new = h_test.copy()
                    best_improvement = 1
        
        if best_improvement > 0:
            best_h = best_h_new
            best_c5 = best_c5_new
        else:
            # Random restart occasionally
            if np.random.rand() < 0.05:
                h_restart = np.random.rand(n_points)
                h_restart = h_restart * target_sum / np.sum(h_restart)
                c5_restart = compute_c5(h_restart)
                if c5_restart < best_c5:
                    best_h = h_restart
                    best_c5 = c5_restart
                    
        iteration += 1
        
    return (best_h.tolist(), best_c5, n_points)
