import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5  # Simple uniform start
    
    # Initialize
    h = h0.copy().astype(float)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h
    current_sum = np.sum(h)
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        idx = np.argsort(h)
        update = min(diff, 1.0 - h[idx[0]])
        h[idx[0]] += update
        diff -= update
        for i in range(len(idx)-1, -1, -1):
            if diff == 0:
                break
            j = idx[i]
            if h[j] < 1.0 and diff > 0:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent with convex relaxation
    # Parameters
    learning_rate = 0.01
    max_iter = 1000
    patience = 50
    patience_counter = 0
    prev_best = best_c5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute current correlation to find maximizing shift
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        shift = max_idx - (len(h) - 1)
        
        # Gradient computation
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if 0 <= i - shift < len(h):
                grad[i] += 1 - 2 * h[i - shift]
        
        # Update using gradient descent
        h -= learning_rate * grad
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h = h * target_sum / current_sum
        
        # Evaluate and update best
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= patience:
                learning_rate *= 0.5
                patience_counter = 0
                if learning_rate < 1e-6:
                    break
    
    return (best_h.tolist(), float(best_c5), n_points)
