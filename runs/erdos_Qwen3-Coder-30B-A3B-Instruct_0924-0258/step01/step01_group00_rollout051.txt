import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()

        n_points = len(h_start)

        h_start = np.ones(n_points) / 2
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = n_points
    
    # Multi-resolution optimization with different strategies
    resolutions = [20, 30, 40, 50]
    
    for res_idx, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if necessary
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Multiple restarts
        for restart in range(3):
            # Random initialization around current best
            if restart == 0:
                h_current = h_start.copy()
            else:
                # Inject more randomness on restarts, but preserve structure
                h_perturbed = h_start + np.random.normal(0, 0.1, n_points)
                h_current = project_to_feasible(h_perturbed, n_points)
            
            # Optimization with time constraint
            local_start = time()
            local_timeout = min(150, (budget_s - (time() - start_time)) / len(resolutions))
            
            # Try different optimization methods
            methods = ['L-BFGS-B', 'Nelder-Mead']
            
            for method in methods:
                if time() - local_start > local_timeout:
                    break
                    
                try:
                    # Objective function
                    def obj(h):
                        # Make sure constraint is satisfied first
                        h = project_to_feasible(h, n_points)
                        corr = np.correlate(h, 1 - h, mode='full')
                        return -np.min(corr)  # Minimize -min_corr is max min_corr
                    
                    def obj_grad(h):
                        h = project_to_feasible(h, n_points)
                        corr = np.correlate(h, 1 - h, mode='full')
                        # Not entirely accurate gradient but provides useful info
                        grad = np.zeros_like(h)
                        min_idx = np.argmin(corr)
                        # Simple approximation based on gradient around min
                        eps = 1e-6
                        for i in range(n_points):
                            h_plus = h.copy()
                            h_plus[i] += eps
                            h_plus = project_to_feasible(h_plus, n_points)
                            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                            grad[i] = (corr_plus[min_idx] - corr[min_idx]) / eps
                        return -grad  # Return negative for minimization

                        # Use L-BFGS for smooth optimization
                        res = minimize(obj, h_current, method='L-BFGS-B',
                                       bounds=bounds, jac=obj_grad,
                                       options={'maxiter': 30, 'ftol': 1e-6})

                    else:  # Nelder-Mead

                                       options={'maxiter': 50, 'ftol': 1e-6})
                        if res.success:
                            h_opt = project_to_feasible(res.x, n_points)
                            c5 = evaluate_erdos_solution(h_opt, dx)

                                best_c5 = c5
                                best_n_points = n_points
