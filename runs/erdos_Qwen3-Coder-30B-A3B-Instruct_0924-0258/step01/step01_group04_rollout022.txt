import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 20
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)

    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # SDP-based optimization
    def solve_sdp(h_init, n_points):
        h = cp.Variable(n_points)
        X = cp.Variable((n_points, n_points), symmetric=True)
        t = cp.Variable()
        
        constraints = [
            X >> 0,  # PSD
            cp.diag(X) == h,  # X_ii = h[i]
            0 <= h, h <= 1,  # bounds on h
            cp.sum(h) == target_sum,  # sum constraint
        ]
        
        # Add overlap constraints
        for k in range(1, n_points):
            # Sum of X[i, (i+k)%n_points] for all i
            overlap_sum = cp.sum([X[i, (i + k) % n_points] for i in range(n_points)])
            constraints.append(t <= overlap_sum)
            
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, max_iter=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h.value
                return project_h(h_sol)
            else:
                return h_init
        except:
            return h_init

    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Try SDP optimization
    if n_points <= 40:  # SDP solver is slow for large n
        # Reducing n_points to allow faster SDP solve
        reduced_n = min(40, n_points)
        h = np.array(initial_h_values[:reduced_n]) if 'initial_h_values' in globals() else np.ones(reduced_n) * 0.5
        # SDP solve
        h_sdp = solve_sdp(h, reduced_n)
        c5_sdp = compute_C5(h_sdp)
        if c5_sdp < best_C5:
            best_h = h_sdp
            best_C5 = c5_sdp
            
    # Refine with gradient-based optimization if time allows
    if time.time() - start_time < budget_s - 5:
        learning_rate = 0.01
        momentum = 0.9
        velocity = np.zeros_like(best_h)
        
        for _ in range(500):
            if time.time() - start_time > budget_s - 2:
                break
                
            h_vec = project_h(best_h)
            conv = np.correlate(h_vec, 1 - h_vec, mode='full')
            mid = len(conv) // 2
            overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
            top_k = np.argsort(overlaps)[-3:][::-1]
            
            gradient = np.zeros_like(h_vec)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus)
                C_plus = compute_C5(h_plus)
                
                h_minus = h_vec.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus)
                C_minus = compute_C5(h_minus)
                
                gradient[i] = (C_plus - C_minus) / (2 * eps)
            
            velocity = momentum * velocity - learning_rate * gradient
            best_h = best_h + velocity
            best_h = project_h(best_h)
            
            current_C5 = compute_C5(best_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)
