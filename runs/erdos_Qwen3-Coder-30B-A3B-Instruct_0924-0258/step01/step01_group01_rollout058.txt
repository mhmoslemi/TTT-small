import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def smooth_max(x, beta=10.0):
    return np.log(np.sum(np.exp(beta * x))) / beta

def compute_correlation_gradient(h, dx):
    n = len(h)
    # Correlation of h with (1-h) using full mode
    corr = correlate(h, 1-h, mode='full')
    # Find max lag index
    max_idx = len(corr) // 2
    max_corr = corr[max_idx]
    # Compute gradients (simplified)
    # Approximate using finite differences for smoothness
    grad = np.zeros_like(h)
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_constraints(h_plus, dx)
        corr_plus = correlate(h_plus, 1-h_plus, mode='full')
        grad[i] = (corr_plus[max_idx] - max_corr) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points

    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Optimization parameters
    max_iter = 1000
    learning_rate = 1e-3
    beta = 10.0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute current correlation
        corr = correlate(h, 1-h, mode='full')
        max_corr_idx = len(corr) // 2
        current_max = corr[max_corr_idx]
        
        # Gradients via finite differences for smooth max
        grad = np.zeros_like(h)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_to_constraints(h_plus, dx)
            corr_plus = correlate(h_plus, 1-h_plus, mode='full')
            grad[j] = (corr_plus[max_corr_idx] - current_max) / eps
            
        # Update step
        h -= learning_rate * grad
        h = project_to_constraints(h, dx)
        
        # Evaluate
        c5 = evaluate(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Adaptive learning rate
        if i % 100 == 0 and i > 0:
