import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_constraints(h, n_points, dx):
    """Evaluate constraints for the optimization"""
    # Check if h is feasible
    h_feasible = np.clip(h, 0, 1)
    h_sum = np.sum(h_feasible)
    constraint_violation = abs(h_sum - n_points / 2)
    return constraint_violation

def augmented_lagrangian(h, n_points, dx, alpha, lambda_bal, mu):
    """Compute augmented Lagrangian for the optimization problem"""
    corr = np.correlate(h, 1 - h, mode='full')
    # Max correlation term
    max_val = np.max(corr)
    # Constraint violation penalty
    constraint_violation = evaluate_constraints(h, n_points, dx)
    # Augmented Lagrangian: minimize -alpha + penalty terms
    penalty = -alpha + 0.5 * mu * (max_val - alpha)**2 + lambda_bal * constraint_violation
    return penalty
