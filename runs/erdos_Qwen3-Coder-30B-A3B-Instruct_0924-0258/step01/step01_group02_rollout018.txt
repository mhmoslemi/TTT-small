import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def convex_feasibility(h_vec, max_overlap):
        # Create a convex formulation to check if h_vec is feasible for 
        # maximum overlap <= max_overlap
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            # We compute correlation at lag k in a simpler way
            # Since we're looking at convolutions, actually just use primal form
            # Correlation of h with 1-h, shifted by k
            valid_range = min(n_points - k, n_points)
            if valid_range > 0:
                c = np.sum(h_vec[:valid_range] * (1 - h_vec[k:k+valid_range]))
                c *= dx
                constraints.append(c <= max_overlap)
        
        # Minimize 0 to check feasibility
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INACCURACY
        except:
            return False
    
    def binary_search_constraint(h_vec, low, high):
        # Perform binary search for optimal C5 bound
        while high - low > 1e-5:
            mid = (low + high) / 2
            if convex_feasibility(h_vec, mid):
                high = mid
            else:
                low = mid
        return high

    # Clone the initial h and project
    h_current = h.copy()
    h_current = project_h(h_current)
    
    start_time = time.time()
    best_c5 = compute_c5(h_current)
    best_h = h_current.copy()
    
    # Phase 1: Perturb and try to improve with gradient descent
    for iteration in range(1000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Get current cost and gradient
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        c5_current = corr[k_max] * dx
    
        # Simple gradient estimation by finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        corr_base = np.correlate(h_current, 1 - h_current, mode='full')
        k_max_base = np.argmax(corr_base)
        base_val = corr_base[k_max_base]
        
        for i in range(n_points):
            h_test = h_current.copy()
            h_test[i] += eps
            h_test = project_h(h_test)
            corr_test = np.correlate(h_test, 1 - h_test, mode='full')
            grad[i] = (corr_test[k_max_base] - base_val) / eps
        
        # Update h using gradient
        lr = 0.01
        h_current -= lr * grad
        h_current = project_h(h_current)
        
        # Update best
        new_c5 = compute_c5(h_current)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_current.copy()
            
        # Occasionally add noise to escape local minima
        if iteration % 100 == 0 and iteration > 0:
            noise = 0.005 * np.random.uniform(-1, 1, n_points)
            h_current += noise
            h_current = project_h(h_current)
    
    # Phase 2: Binary search refinement
    # Use the best h to find tight bound
    low = 0.37
    high = float(best_c5)
    
    # Binary search loop (with max iterations to avoid unlimited loop)
    for _ in range(100):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        mid = (low + high) / 2
        if convex_feasibility(best_h, mid):
            high = mid
        else:
            low = mid
            
        if high - low < 1e-5:
            break
    
    final_c5 = high
    return (best_h.tolist(), final_c5, n_points)
