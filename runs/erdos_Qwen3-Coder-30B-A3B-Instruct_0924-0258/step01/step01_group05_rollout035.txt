import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Simulated Annealing parameters
    T0 = 0.1
    tau = budget_s / np.log(10)
    sigma = 0.1
    min_T = 1e-5
    max_iter = 10000
    
    h_current = h.copy()
    h_current = project_feasible(h_current)
    C5_current = compute_C5(h_current)
    
    h_best = h_current.copy()
    C5_best = C5_current
    
    T = T0
    accept_count = 0
    total_count = 0
    
    # Priority sampling: identify intervals that most contribute to overlap
    def get_priority_intervals(h_temp):
        # Compute the correlation matrix for all valid lag values
        corr = correlate(h_temp, 1 - h_temp, mode='full')
        # Get maximum across all lags
        max_lag_idx = np.argmax(corr)
        # Calculate lag from index
        lag = max_lag_idx - (len(h_temp) - 1)
        # Create array of all overlapping contributions and get top intervals
        overlap_contributions = np.zeros_like(h_temp)
        if lag >= 0:
            # Positive lag: h[i] contributes to (1-h[i+lag])
            contribs = h_temp[:-lag]
            overlap_contributions[lag:] += contribs
        else:
            # Negative lag: h[i] contributes to (1-h[i+lag])
            contribs = h_temp[-lag:]
            overlap_contributions[:lag] += contribs
        
        # Get indices of top 20% intervals by contribution
        n_priority = max(1, len(h_temp) // 5)  # Top 20%
        priority_indices = np.argsort(overlap_contributions)[-n_priority:]
        return priority_indices
    
    iter_count = 0
    while time.time() - start_time < budget_s - 0.1 and iter_count < max_iter:
        iter_count += 1
        
        # Adapt temperature based on progress
        if iter_count % 100 == 0:
            if accept_count / max(1, total_count) < 0.1:
                sigma *= 1.2
            elif accept_count / max(1, total_count) > 0.4:
                sigma *= 0.8
            accept_count = 0
            total_count = 0

        # Construct a new candidate solution
        h_new = h_current.copy()
        priority_indices = get_priority_intervals(h_current)
        
        # Perturb the priority intervals with Gaussian noise
        perturb_indices = np.random.choice(priority_indices, size=max(1, len(priority_indices) // 2), replace=False)
        noise = np.random.normal(0, sigma, len(perturb_indices))
        h_new[perturb_indices] += noise
        
        # Random flips for diversity
        if np.random.rand() < 0.3:
            flip_indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
            for i in flip_indices:
                h_new[i] = 1 - h_new[i]
        
        # Project the new solution into feasible space
        h_new = project_feasible(h_new)
        
        # Evaluate new solution
        C5_new = compute_C5(h_new)
        
        # Accept or reject
        total_count += 1
        if C5_new < C5_current:
            h_current = h_new
            C5_current = C5_new
            accept_count += 1
            
            if C5_new < C5_best:
                h_best = h_new.copy()
                C5_best = C5_new
        else:
            # Accept with probability based on temperature
            delta = C5_new - C5_current
            prob = np.exp(-delta / max(T, min_T))
            if np.random.rand() < prob:
                h_current = h_new
                C5_current = C5_new
                accept_count += 1
        
        # Update temperature
        T = T0 * np.exp(-iter_count / tau)
        
    # Final cleanup to ensure feasibility
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
