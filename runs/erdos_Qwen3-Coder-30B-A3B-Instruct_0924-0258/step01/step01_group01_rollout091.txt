import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with low resolution
    n_low = 10
    n_high = 50
    n_points = n_low
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Adaptive cooling SA
    T = 0.1
    cooling = 0.99
    iter_per_temp = 500
    
    iter_count = 0
    last_improvement = 0
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for _ in range(iter_per_temp):
            if time.time() - start_time > budget_s:
                break
                
            # Adapt perturbation based on current temperature
            temp = T * 0.5
            h_new = h + np.random.normal(0, temp, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_constraints(h_new, dx)
            
            c5_new = evaluate(h_new, dx)
            
            delta = c5_new - best_c5
            if delta < 0 or np.random.uniform() < np.exp(-delta / T):
                h = h_new
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
                    last_improvement = iter_count
                    
            iter_count += 1
            
        T *= cooling
        
        # Adaptive resolution doubling
        if n_points < n_high and iter_count - last_improvement > 300:
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            last_improvement = iter_count
            T = 0.1
            
        if T < 1e-6:
            break
    
    return best_h.tolist(), best_c5, len(best_h)
