import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)

    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # PSO parameters
    n_particles = 100
    max_iter = int(budget_s * 1.5)  # Estimate iterations
    w_start = 0.7
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    max_speed = 0.1

    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for _ in range(n_particles):
        # Start near initial solution with small perturbations
        h_p = h + np.random.normal(0, 0.05, n_points)
        h_p = project_h(h_p)
        particles.append(h_p.copy())
        velocities.append(np.zeros(n_points))
        pbest_positions.append(h_p.copy())
        pbest_values.append(compute_C5(h_p))

    # Global best
    best_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[best_idx].copy()
    gbest_value = pbest_values[best_idx]

    # PSO main loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest_positions[i] - particles[i]) +
                c2 * r2 * (gbest_position - particles[i])
            )
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_speed, max_speed)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_h(particles[i])
            
            # Evaluate fitness
            current_value = compute_C5(particles[i])
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest_values[i] = current_value
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if current_value < gbest_value:
                    gbest_value = current_value
                    gbest_position = particles[i].copy()

    return (gbest_position.tolist(), gbest_value, n_points)
