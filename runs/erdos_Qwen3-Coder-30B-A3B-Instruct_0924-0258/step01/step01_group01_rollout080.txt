import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def smooth_max(x, beta=20.0):
    # Smooth approximation to max
    return np.log(np.sum(np.exp(beta * x))) / beta

def compute_correlation(h, dx):
    # Compute correlation of h with (1-h) 
    corr = correlate(h, 1-h, mode='full')
    return corr * dx

def compute_gradient(h, dx):
    # Compute gradient of smooth max correlation 
    n = len(h)
    corr = compute_correlation(h, dx)
    
    # Gradient of smooth max approximation
    beta = 20.0
    exp_vals = np.exp(beta * corr)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    
    # Approximate gradient with finite differences
    grad = np.zeros(n)
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_constraints(h_plus, dx)
        corr_plus = compute_correlation(h_plus, dx)
        grad[i] = (smooth_max(corr_plus, beta) - smooth_max(corr, beta)) / eps
    
    return grad

def backtracking_armijo(f, grad, h, alpha=1e-3, rho=0.5, c=1e-4):
    """Simple backtracking line search"""
    while True:
        h_new = h - alpha * grad
        h_new = project_to_constraints(h_new, 2.0/len(h))
        if f(h_new, 2.0/len(h)) <= f(h, 2.0/len(h)) + c * alpha * np.dot(grad, grad):
            return alpha
        alpha *= rho

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with lower resolution
    n_points = 50
    dx = 2.0 / n_points
    
    # Initialize with existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), 
                         np.linspace(0, 2, len(h)), h)
    else:
        # Random initial
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    last_c5 = best_c5
    
    # Optimization parameters
    learning_rate = 1e-3
    max_iter = 1000
    iter_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        iter_count += 1
        
        # Compute gradient
        grad = compute_gradient(h, dx)
        
        # Line search
        alpha = backtracking_armijo(evaluate, grad, h, learning_rate)
        
        # Update
        h_new = h - alpha * grad
        h_new = project_to_constraints(h_new, dx)
        
        # Evaluate
        c5_new = evaluate(h_new, dx)
        
        if c5_new < last_c5:
            h = h_new
            last_c5 = c5_new
            if last_c5 < best_c5:
                best_h = h.copy()
                best_c5 = last_c5
                
        # Adapt learning rate
        if iter_count % 500 == 0:
            learning_rate *= 0.5
            
    return best_h.tolist(), best_c5, len(best_h)
