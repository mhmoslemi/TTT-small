import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Adam optimizer state
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    learning_rate = 0.01
    
    # Main optimization loop
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        if iteration % 100 == 0:
            # Periodic full evaluation
            current_C5 = compute_C5(best_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = best_h.copy()
                
        # Compute overlaps to identify k contributing to max
        h_vec = project_h(h)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        # Get the index of the maximum overlap
        max_k_idx = np.argmax(overlaps)
        
        # Compute gradient for that specific k using analytical formula
        gradient = np.zeros_like(h_vec)
        for i in range(n_points):
            # Gradient for h[i] w.r.t. the maximum overlap term
            # For the correlation term at shift k_idx:
            # The contribution is h[i] * (1 - h[i + k_idx]) if positive
            # But we're computing the derivative of sum over all k
            # Simplified: d/dh[i] of sum_k int h(x) (1 - h(x+k)) dx
            # Is directly related to values of h shifted by k_idx.
            gradient[i] = -conv[mid-n_points+1+max_k_idx] * dx if i == 0 else 0
            
        # Actually compute correct gradient for the max overlap k
        gradient = np.zeros_like(h_vec)
        k = max_k_idx - n_points + 1
        for i in range(n_points):
            # For correlation at shift k, d/dh[i] of h[i] * (1-h[i+k]) 
            j = (i + k) % n_points
            gradient[i] = -h_vec[j] * dx
            
        # Update with Adam
        t += 1
        m = beta1 * m + (1 - beta1) * gradient
        v = beta2 * v + (1 - beta2) * gradient**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project
        h = project_h(h)
            
        # Adaptive learning rate
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            learning_rate *= 1.1  # Increase if improving
        else:
            learning_rate *= 0.9  # Decrease if not improving
            
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)
