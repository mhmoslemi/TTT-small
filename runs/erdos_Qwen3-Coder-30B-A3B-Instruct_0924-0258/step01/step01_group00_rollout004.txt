import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def solve_subproblem(h, n_points, dx, time_limit):
    """Solve a subproblem using a targeted approach to improve solution"""
    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Use L-BFGS-B with bounds
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.min(corr) * dx  # Maximize min correlation
    
    def grad(h):
        eps = 1e-6
        grad = np.zeros_like(h)
        base_c = np.correlate(h, 1 - h, mode='full')
        min_idx = np.argmin(base_c)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (corr_plus[min_idx] - base_c[min_idx]) / eps
        return -grad * dx
    
    # Constraints
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    for _ in range(3):  # Multiple restarts
        if time() - start_time > time_limit:
            break
        try:
            # Rand perturb
            h_perturbed = project_to_feasible(h + np.random.normal(0, 0.01, n_points), n_points)
            res = minimize(obj, h_perturbed, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons, jac=grad, 
                           options={'maxiter': 30})
            if res.success:
                h_res = project_to_feasible(res.x, n_points)
                c5 = evaluate_erdos_solution(h_res, dx)
                if c5 < best_c5:
                    best_h = h_res.copy()
                    best_c5 = c5
        except:
            pass
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 30
        h_start = np.ones(n_points) / 2
    
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different resolutions
    resolutions = [15, 20, 25, 30, 35, 40, 45]
    
    for n_points_try in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        
        # Interpolate if needed
        if len(h_start) != n_points_try:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
        
        # Allocate time
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 40)
        
        # Multiple restarts and optimization
        best_h_level = h_start.copy()
        best_c5_level = evaluate_erdos_solution(h_start, dx_try)
        
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            
            h_restart = project_to_feasible(h_start + np.random.normal(0, 0.03, n_points_try), n_points_try)
            h_opt, c5_opt = solve_subproblem(h_restart, n_points_try, dx_try, time_per_res / 5)
            
            if c5_opt < best_c5_level:
                best_c5_level = c5_opt
                best_h_level = h_opt.copy()
        
        if best_c5_level < best_c5:
            best_h = best_h_level.copy()
            best_c5 = best_c5_level
            n_points = n_points_try
            
        h_start = best_h_level.copy()
    
    return (list(best_h), best_c5, n_points)
