import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def pattern_perturb(h, strength=0.1):
    """Perturb the function in a structured way"""
    pert = np.random.normal(0, strength, len(h))
    # Add some periodicity to avoid completely random perturbations
    x = np.linspace(0, 2, len(h))
    periodic = 0.5 * np.sin(2 * np.pi * x) + 0.5 * np.cos(np.pi * x)
    return h + pert + 0.2 * periodic

def augmented_lagrangian_optimization(h_start, n_points, dx, time_limit):
    """Optimize using a simple augmented Lagrangian approach"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Get initial constraints
    def compute_constraints(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr * dx  # This is the overlap integral for all k
    
    # Main optimization loop
    for iteration in range(500):
        if time() - start_time > time_limit:
            break
            
        # Get current overlaps
        overlaps = compute_constraints(best_h)
        
        # Objective to minimize - negative of minimum overlap (we want to maximize min overlap)
        def objective(h):
            overlaps = compute_constraints(h)
            return -np.min(overlaps)
        
        # Constraints: enforce bounds
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        try:
            # Optimize with SLSQP
            res = minimize(objective, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                new_h = project_to_feasible(res.x, n_points)
                new_c5 = evaluate_erdos_solution(new_h, dx)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
        except:
            pass
            
        # Small random perturbation to escape local minima
        perturbed = project_to_feasible(best_h + np.random.normal(0, 0.01, n_points), n_points)
        c5_pert = evaluate_erdos_solution(perturbed, dx)
        if c5_pert < best_c5:
            best_h = perturbed
            best_c5 = c5_pert
            
        # Pattern-based perturbation every few iterations
        if iteration % 5 == 0:
            pattern_h = project_to_feasible(pattern_perturb(best_h, 0.05), n_points)
            c5_pattern = evaluate_erdos_solution(pattern_h, dx)
            if c5_pattern < best_c5:
                best_h = pattern_h
                best_c5 = c5_pattern
    
    return best_h, best_c5

    best_n_points = len(h_start)
    
    # Try different resolutions
    resolutions = [20, 25, 30, 35, 40, 45, 50, 55, 60]
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if necessary
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 80)
        
        # Multiple restarts to prevent local minima
        best_h_level = h_start.copy()
        best_c5_level = evaluate_erdos_solution(h_start, dx)
        
        for restart in range(3):
            h_restart = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            h_opt, c5_opt = augmented_lagrangian_optimization(h_restart, n_points, dx, time_per_res / 3)
            
            if c5_opt < best_c5_level:
                best_c5_level = c5_opt
                best_h_level = h_opt.copy()

            best_h = best_h_level.copy()
            best_c5 = best_c5_level
