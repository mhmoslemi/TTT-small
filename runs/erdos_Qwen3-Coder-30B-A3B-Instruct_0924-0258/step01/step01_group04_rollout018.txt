import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def upsample_h(h_vec, target_len):
        # Linear interpolation
        x_orig = np.linspace(0, 1, len(h_vec))
        x_target = np.linspace(0, 1, target_len)
        return np.interp(x_target, x_orig, h_vec)

    # Multi-resolution scheme
    res_levels = [5, 10, 20, 40, n_points]
    best_h = h.copy()
    best_C5 = compute_overlap(best_h)
    
    # Optimization parameters
    lr = 0.01
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    for i, level_points in enumerate(res_levels):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize at this level
        if level_points <= len(h):
            h_level = h[:level_points].copy()
        else:
            h_level = upsample_h(h, level_points)
        
        # Adam optimizer
        max_iter = 5000
        for t in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_level = project_feasible(h_level)
            current_C5 = compute_overlap(h_level)
            
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_level.copy()
            
            # Compute gradient using finite differences
            gradient = np.zeros_like(h_level)
            eps_val = 1e-4
            # Top 5 shifts contributing to max overlap
            conv = np.correlate(h_level, 1 - h_level, mode='full')
            mid = len(conv) // 2
            overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
            top_k = np.argsort(overlaps)[-5:][::-1]
            
            for j in range(len(h_level)):
                h_plus = h_level.copy()
                h_plus[j] += eps_val
                h_plus = project_feasible(h_plus)
                C_plus = compute_overlap(h_plus)
                
                h_minus = h_level.copy()
                h_minus[j] -= eps_val
                h_minus = project_feasible(h_minus)
                C_minus = compute_overlap(h_minus)
                
                gradient[j] = (C_plus - C_minus) / (2 * eps_val)

            # Adam update
            m = beta1 * m + (1 - beta1) * gradient
            v = beta2 * v + (1 - beta2) * gradient**2
            m_hat = m / (1 - beta1**(t+1))
            v_hat = v / (1 - beta2**(t+1))
            h_level = h_level - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Upscale to next level
        if i < len(res_levels) - 1:
            h = upsample_h(h_level, res_levels[i+1])
    
    # Final evaluation
    final_C5 = compute_overlap(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, n_points)
