import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region maintaining sum constraint."""
    h = np.array(h, dtype=float)
    
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Normalize to maintain sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        # Distribute the difference among elements
        if diff > 0:
            # We need to increase values
            idx = np.argsort(h)
            for i in range(len(idx)-1, -1, -1):
                if diff <= 0:
                    break
                j = idx[i]
                if h[j] < 1.0:
                    update = min(diff, 1.0 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # We need to decrease values
            idx = np.argsort(h)
            for i in range(len(idx)):
                if diff >= 0:
                    break
                j = idx[i]
                if h[j] > 0.0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def init_simplex(h0, n_points, target_sum, size=72):
    """Initialize simplex for Nelder-Mead."""
    simplex = [h0.copy()]
    for _ in range(size - 1):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, target_sum)
        simplex.append(h)
    return simplex

def nelder_mead_step(simplex, values, alpha=1.0, beta=2.0, gamma=0.5, delta=0.5):
    """Perform one Nelder-Mead step."""
    # Sort simplex by values
    idx = np.argsort(values)
    simplex = [simplex[i] for i in idx]
    values = [values[i] for i in idx]
    
    # Centroid of all but worst
    best = simplex[0]
    worst = simplex[-1]
    centroid = np.mean(simplex[:-1], axis=0)
    
    # Reflection
    reflected = centroid + alpha * (centroid - worst)
    reflected = project_to_feasible(reflected, len(reflected) / 2.0)
    
    # Evaluate
    r_val = evaluate(reflected, 2.0 / len(reflected))
    
    if values[0] <= r_val < values[-2]:
        # Replace worst
        simplex[-1] = reflected
        values[-1] = r_val
    elif r_val < values[0]:
        # Expansion
        expanded = centroid + beta * (reflected - centroid)
        expanded = project_to_feasible(expanded, len(expanded) / 2.0)
        e_val = evaluate(expanded, 2.0 / len(expanded))
        if e_val < r_val:
            simplex[-1] = expanded
            values[-1] = e_val
        else:
            simplex[-1] = reflected
            values[-1] = r_val
    else:
        # Contraction
        if r_val < values[-1]:
            # Outside contraction
            contracted = centroid + gamma * (reflected - centroid)
        else:
            # Inside contraction
            contracted = centroid + gamma * (worst - centroid)
        contracted = project_to_feasible(contracted, len(contracted) / 2.0)
        c_val = evaluate(contracted, 2.0 / len(contracted))
        if c_val < values[-1]:
            simplex[-1] = contracted
            values[-1] = c_val
        else:
            # Reduction
            for i in range(1, len(simplex)):
                simplex[i] = simplex[0] + delta * (simplex[i] - simplex[0])
                simplex[i] = project_to_feasible(simplex[i], len(simplex[i]) / 2.0)
            values = [evaluate(h, 2.0 / len(h)) for h in simplex]
    
    return simplex, values

    # Initialize simplex
    simplex = init_simplex(h0, n_points, n_points / 2.0, 72)
    
    # Evaluate initial simplex
    dx = 2.0 / n_points
    values = [evaluate(h, dx) for h in simplex]
    
    best_h = simplex[np.argmin(values)].copy()
    best_c5 = min(values)
    
    iter_count = 0
    while time.time() - start_time < budget_s:
        # Nelder-Mead step
        simplex, values = nelder_mead_step(simplex, values)
        
        # Update best if better
        current_min = min(values)
        if current_min < best_c5:
            best_c5 = current_min
            best_h = simplex[np.argmin(values)].copy()
        
        # Occasionally add some mutation to avoid local minima
        if iter_count % 50 == 0:
            for i in range(len(simplex)):
                if np.random.random() < 0.1:
                    simplex[i] = project_to_feasible(
                        simplex[i] + np.random.normal(0, 0.01, n_points),
                        n_points / 2.0
                    )
                    values[i] = evaluate(simplex[i], dx)
                    if values[i] < best_c5:
                        best_c5 = values[i]
                        best_h = simplex[i].copy()
        
        iter_count += 1
    
    return (best_h.tolist(), float(best_c5), n_points)
