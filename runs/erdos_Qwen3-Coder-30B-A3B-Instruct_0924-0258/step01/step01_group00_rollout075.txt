import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    """Smooth approximation of max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def smooth_objective(h, n_points, dx):
    """Objective with smooth max to approximate the true max"""
    corr = np.correlate(h, 1 - h, mode='full')
    # Minimize max overlap by maximizing min correlation
    # Use smooth max of negative correlations to approximate max of positive overlaps
    return -smooth_max(-corr, alpha=200) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        n0 = 30
        h_start = np.ones(n0) * 0.5
    
    # Parameters
    resolutions = [20, 25, 30, 35, 40, 45] 
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / 5, 100)
        
        # Multiple restarts
        best_h_level = h_start.copy()
        best_c5_level = evaluate_erdos_solution(h_start, dx)
        
        for _ in range(5):
            if time() - start_time > budget_s * 0.98:
                break
                
            h_restart = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            
            def obj(h):
                return smooth_objective(h, n_points, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            
            try:
                res = minimize(obj, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5_level:
                        best_c5_level = c5_opt
                        best_h_level = h_opt.copy()
            except:
                pass
            
            h_start = best_h_level.copy()
            
        if best_c5_level < best_c5:
            best_h = best_h_level.copy()
            best_c5 = best_c5_level
            best_n_points = n_points
            
        h_start = best_h_level.copy()
        
        if time() - start_time > budget_s * 0.95:
            break
    
    # Final refinement
    h_final = project_to_feasible(best_h, best_n_points)
    c5_final = evaluate_erdos_solution(h_final, 2.0 / best_n_points)
    
    return (list(h_final), c5_final, best_n_points)
