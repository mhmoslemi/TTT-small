import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization with step function
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Initialize
    h_current = h.copy()
    h_current = project_feasible(h_current)
    C5_current = compute_C5(h_current)
    C5_best = C5_current
    h_best = h_current.copy()
    
    # Parameters
    T0 = 0.1
    cooling_rate = np.log(10) / budget_s
    sigma = 0.1
    min_T = 1e-5
    max_iterations = 50000
    n_iter = 0
    
    # Main simulated annealing loop
    while time.time() - start_time < budget_s - 0.1 and n_iter < max_iterations:
        n_iter += 1
        
        # Compute current temperature
        elapsed = time.time() - start_time
        T = T0 * np.exp(-elapsed * cooling_rate)
        if T < min_T:
            T = min_T
        
        # Identify intervals contributing most to overlap (top 20%)
        # Using a heuristic to identify "critical" intervals
        # Compute gradient estimate from correlations
        corr = correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Create perturbation
        h_new = h_current.copy()
        
        # Perturb a mix of high and low priority elements
        # Focus on critical intervals that caused the max overlap
        if max_lag >= 0:
            indices = np.arange(max_lag, len(h_current))
        else:
            indices = np.arange(0, len(h_current) + max_lag)
        
        # Randomly perturb 30% of the indices
        indices_to_perturb = np.random.choice(indices, size=max(1, len(indices)//3), replace=False)
        
        # Add noise to specific indices
        noise = np.random.normal(0, sigma, len(indices_to_perturb))
        h_new[indices_to_perturb] += noise
        
        # Apply random flips on a small percentage of elements
        n_flip = max(1, len(h_new) // 20)
        flip_indices = np.random.choice(len(h_new), size=n_flip, replace=False)
        for i in flip_indices:
            h_new[i] = 1.0 - h_new[i]
        
        # Project back to feasible set
        h_new = project_feasible(h_new)
        
        # Compute new objective
        C5_new = compute_C5(h_new)
        
        # Accept or reject
        delta = C5_new - C5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_current = h_new.copy()
            C5_current = C5_new
            if C5_current < C5_best:
                C5_best = C5_current
                h_best = h_current.copy()
        
        # Adapt learning rate based on acceptance rate
        if n_iter % 100 == 0:
            if sigma > 0.001:
                sigma *= 0.99
                sigma = max(0.001, sigma)
    
    # Final projection and check
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
