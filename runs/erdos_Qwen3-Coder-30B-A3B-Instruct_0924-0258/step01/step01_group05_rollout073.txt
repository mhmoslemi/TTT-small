```python
import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_gradient(h_vec, max_lag):
        # Compute gradient for correlation at max_lag
        n = len(h_vec)
        grad = np.zeros(n)
        if max_lag >= 0:
            # h[i] contributes to correlation with lag k when we look at:
            # h[i] * (1 - h[i+k]) for i + k < n
            grad[max_lag:] = -h_vec[:-max_lag]
            # And we also need to consider h[i] * h[i-k] for k > 0, i-k ≥ 0
            # This contributes to lag -k, so when we look at lag k, we have
            # 1-h[i+k] term corresponds to contribution of h[i] to h[i+k] term
            # More careful calculation:
            # For correlation at lag k, when i-th element is aligned with 
            # (i+k)-th element, we have h[i] * (1-h[i+k])
            # So partial of corr w.r.t h[i] is (1-h[i+k]) 
            # and partial of corr w.r.t h[i+k] is -h[i]
            # For max_lag k, we go through each i:
            # Grad[i] += (1 - h[i+k]) 
            # Grad[i+k] -= h[i]  
            # Except this is ambiguous - it depends on which side of the 
            # correlation we're looking at.
            # Better way: 
            # When we compute full correlation, we have:
            # result[k] = sum_i h[i] * (1 - h[i+k]) where k is the lag
            # So for fixed k and fixed i:
            # d/dh[i] (sum_{i'} h[i'] * (1 - h[i'+k])) = (1 - h[i+k])
            # d/dh[i+k] (sum_{i'} h[i'] * (1 - h[i'+k])) = -h[i]
            # Our gradient for max_lag corresponds to the max correlogram 
            # element, we want to minimize it,
            # so the gradient is - d/dh[i] of that specific element
            # which means we compute for that specific k.
            # Correlation at lag k = sum_{i} h[i] * (1 - h[i+k])
            # Gradient w.r.t. h[i] is (1 - h[i+k])
            # But only when (i+k) < n, i.e. k <= n-1-i
            # But since we shifted, i goes from 0 to n-1, 
            # and (i+k) from k to n+k-1
            # The range of i that can use shift k is such that (i+k) < n
            # So i from 0 to n-k-1
            # Grad[i] += (1 - h[i+k]) for i in range(0, n-k)
            # Grad[i+k] -= h[i] for i in range(0, n-k)
            # Simpler to use more direct gradient computation for max_lag:
            # correlation = sum over i of h[i] * (1 - h[i+lag])
            # g[i] = d/dh[i] sum h[i] * (1 - h[i+lag]) = (1 - h[i+lag]) 
            # if i+lag < n (i.e. i < n-lag) else 0
            # g[i+lag] = d/dh[i+lag] sum h[i] * (1 - h[i+lag]) = -h[i]
            # if i+lag < n (i.e. i < n-lag) else 0
            # Only go up to n-lag-1 and construct gradient from:
            # For i = 0 to n-lag-1: 
            #   grad[i] += (1 - h[i+lag])
            #   grad[i+lag] -= h[i]
            # But this does not work in full correlation mode. 
            # Actually it's simpler in the full mode:
            # 1. We compute correlation for ALL values.
            # 2. We clip h to [0,1] and rescale to sum to target_sum.
            # 3. Generate gradient w.r.t. the maximizer of correlation,
            #    recognizing it's a non-linear function of the variables.
            # Let's compute full correlation matrix:
            # For lag k, we sum i h[i] * (1 - h[i-k])  
            # (we shift h forward) 
            #       indices [0 1 2 3 4 5]  size n  lag k=2
            # h is [a b c d e f] -> 
            # h is [a b c d e f]
            # ...     ...     ...
            # corr = sum_{i=0}^{n-1} h[i] (1 - h[(i+k) % n]) 
            # when we do the full convolution with 1-h[::-1], that's equivalent 
            # to correlating forward.
            # correlation_lag_k = sum_i h[i] * (1 - h[(i+k) % n])
            # g[i] = d/dh[i] sum_i h[i] (1 - h[(i+k) % n]) = (1 - h[(i+k) % n])
            # g[i+k] = d/dh[i+k] sum_i h[i] (1 - h[(i+k) % n]) = -h[i]
            # So if k is the lag that maximizes correlation:
            # 1. Direct update as i ranges over [0, n-1]:
            #    g[i] += (1 - h[(i+k) % n])
            #    g[(i+k) % n] -= h[i]
            # But this means the gradient is rank 1... which doesn't make sense.
            # What we really want is: one step gradient, which minimizes 
            # the max of that. The delicate derivative in correlation is: 
            # fix h{}, let the argument in correlation shift:
            # d/d(delta_h[i]) (sum_{j} h[j] (1 - h[j+k]) ) = (1 - h[j+k])
            # Define g = d/dh[i] = (1 - h[(i+k) % n]) for all i
            # but that's not right, it is not a function from i to i across the sliding
            # window. 
            # After ε step of optimization: 
            # Correlation at position k = h^T * (1 - h_shifted_by_k)  
            # where h_shifted_by_k is h with all entries shifted right by k 
            # # Count: i goes 0 to N-1. This means for a fixed k, the sum is:
            # sum_{i=0}^{n-1} h[i] * (1 - h[(i+k) % n])
            # d/dh[i] at this specific lag k: 
            #   = (1 - h[(i+k) % n]) if i is valid for correlation lag k
            #   = 0 otherwise.
            # This is hard to express in a simple gradient formula because 
            # the indices depend on both variables.
            # A different, classical way of computing gradient of one scalar
            # function f(h) which is a sum of terms involving h_i:
            # For f(h) = sum_i f_i(h_i)  
            #   d/dh_i f = d/dh_i sum_i h_i * (1 - h_{i+k})  
            # However, this is NOT a clean sum of h_i terms since what we are
            # minimizing is a max over lags k, not a sum.
            # Let's stick to the formal recipe:
            # Compute the product for a particular lag k_o 
            # which achieves max in correlation.
            # Gradient of this is: derivative with respect to h[i]
            # = (1 - h[(i+k_o) % n] ) if there is no wrapping (index valid)
            # OR is it... 
            # Let's take the convolution definition:
            # We compute:
            #      h*x[idx] = sum_{i} h[i] * x[i-idx] 
            # where x = 1-h in our case.
            # So to compute the conv sum with some shift idx,
            # take h[i] * (1 - h[i - idx]) and sum
            # Differentiate: 
            # d/dh[i] sum_{j} h[j] * (1 - h[j - idx]) = (1 - h[i - idx])
            # This suggests the gradient for computing entry at lag idx is:
            # For all i from 0 to n-1:
            # grad[i] = (1 - h[i - idx]) if (i - idx) >= 0, else 0
            # and 
            # grad[i - idx] -= h[i] if (i - idx) >= 0, else 0  
            # Or simpler: In correlation with lag k,
            # grad[i] += (1 - h[i+k]) for i+k < n,
            # grad[i+k] -= h[i] for i+k < n.
            # Gives matrix of updates i -> (i+k). 
            # Let's use a more ra

            # Got it. Use:
            # grad[i] = 0 for all i.
            # For i = 0 to n-1-k (i.e. those such that i+k < n):  
            #   grad[i] += (1 - h[i+k])
            #   grad[i+k] -= h[i]
            
            for i in range(n - max_lag):
                grad[i] += (1 - h[i + max_lag])
                grad[i + max_lag] -= h[i]
            
        else:
            # negative lag, i.e., shift backward:
            # we are looking at h[i] * (1 - h[i-k]) for k > 0
            # grad[i] += (1 - h[i-k])
            # grad[i-k] -= h[i]
            for i in range(-max_lag, n):
                if i - max_lag >= 0:
                    grad[i] += (1 - h[i - max_lag])
                    grad[i - max_lag] -= h[i]
        
        return grad

    # Main optimization loop
    h_current = h.copy()
    C5_best = compute_C5(h_current)
    iteration = 0
    stagnation_count = 0
    alpha = 0.1
    
    while time.time() - start_time < budget_s - 0.1:
        # Compute all correlations 
        corr = correlate(h_current, 1 - h_current, mode='full')
        # Get max lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Compute gradient for maximum correlation
        grad = compute_gradient(h_current, max_lag)
        
        # Backtracking line search - try different step sizes
        step_sizes = [0.1, 0.05, 0.01, 0.005]
        best_h_candidate = h_current.copy()
        best_C5 = C5_best
        
        for step_size in step_sizes:
            h_candidate = h_current - step_size * grad
            
            # Project to feasible region
            h_candidate = project_feasible(h_candidate)
            
            # Compute new C5
            C5_candidate = compute_C5(h_candidate)
            
            if C5_candidate < best_C5:
                best_C5 = C5_candidate
                best_h_candidate = h_candidate.copy()
        
        # Update current solution if improved
        if best_C5 < C5_best:
            h_current = best_h_candidate.copy()
            C5_best = best_C5
            stagnation_count = 0
            if iteration % 50 == 0 and alpha > 1e-6:
                alpha *= 0.5
        else:
            stagnation_count += 1
            if stagnation_count >= 50:
                # Random perturbation 
                noise = np.random.normal(0, 0.02, n_points)
                h_current = h_current + noise
                h_current = project_feasible(h_current)
                stagnation_count = 0
                alpha = 0.1  # reset step size
        
        iteration += 1
        if iteration > 1000:
            break
            
    # Final projection and recompute
    h_current = project_feasible(h_current)
    C5_final = compute_C5(h_current)
    
    return list(h_current), C5_final, n_points
```
