import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    
    # Final adjustment to ensure sum = target_sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argsort(h)
        i = 0
        while diff != 0 and i < len(idx):
            j = idx[i]
            if h[j] < 1.0 and diff > 0:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
            i += 1
    
    dx = 2.0 / n_points
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Main loop
    iter_count = 0
    last_improvement = 0
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s:
        if iter_count - last_improvement > 50:
            learning_rate *= 0.5
        
        # Compute cross-correlation to find shift k that maximizes overlap
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        # The shift corresponding to the maximum correlation
        shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient approximation
        grad = np.zeros(n_points)
        if shift <= 0:
            # h[i] * (1 - h[i-shift]) terms contribute to gradient
            for i in range(n_points + shift):
                grad[i] += 1 - 2 * h[i + shift]  # derivative of h[i] * (1 - h[i-shift])
        else:
            # h[i] * (1 - h[i-shift]) terms for i >= shift
            for i in range(shift, n_points):
                grad[i] += 1 - 2 * h[i - shift]  # derivative
        
        # Update h
        h_new = h - learning_rate * grad
        
        # Project to feasible region [0,1] and re-normalize
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * target_sum / current_sum
        h_new = np.clip(h_new, 0, 1)
        
        # Ensure exact sum constraint
        diff = target_sum - np.sum(h_new)
        if abs(diff) > 1e-10:
            idx = np.argsort(h_new)
            i = 0
            while diff != 0 and i < len(idx):
                j = idx[i]
                if h_new[j] < 1.0 and diff > 0:
                    update = min(diff, 1.0 - h_new[j])
                    h_new[j] += update
                    diff -= update
                i += 1
                
        h = h_new
        
        # Evaluate and potentially update best solution
        c5 = evaluate(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iter_count
            
        iter_count += 1
        
        # Every 20 iterations, verify convergence by checking improvement
        if iter_count % 20 == 0 and iter_count > last_improvement + 100:
            # Early stopping based on lack of progress
            if iter_count - last_improvement > 200:
                break
    
    return (best_h.tolist(), float(best_c5), n_points)
