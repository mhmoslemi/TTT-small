import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        # Create an initial h with 30 points
        n_points = 30
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / len(h_start)
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = len(h_start)
    
    # Parameters for the greedy improvement
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 50
    target_c5 = 0.38
    adjustment_fraction = 0.1
    perturbation_strength = 0.01
    
    while time() - start_time < budget_s * 0.95:
        # Evaluate current solution
        c5 = evaluate_erdos_solution(best_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()
        
        # If we're close to target and no improvement, break
        if c5 <= target_c5 and no_improvement_count > 10:
            break
            
        # Find the maximum overlap and corresponding shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Adjust the step function
        h_new = best_h.copy()
        # The maximum appears in the middle of the correlation array
        # So find the actual shift (k) that corresponds to the maximum
        shift = max_idx - len(best_h) + 1
        
        # Identify regions where overlap occurs
        # For the maximum overlap, we're looking at h(x) * (1 - h(x + shift))
        # We focus on top contributing regions
        contributions = np.zeros(len(best_h))
        for i in range(len(best_h)):
            # Compute overlap contribution from this position
            j = i + shift
            if 0 <= j < len(best_h):
                contributions[i] = best_h[i] * (1 - best_h[j])
        
        # Select top contributors
        top_indices = np.argsort(contributions)[-int(len(best_h)*0.1):]
        # Decrease h in these regions
        adjustment = (max_overlap - target_c5) * adjustment_fraction / len(top_indices)
        for i in top_indices:
            h_new[i] = max(0, h_new[i] - adjustment)
        
        # Project to feasible region
        h_new = project_to_feasible(h_new, n_points)
        
        # Normalize and evaluate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # If there was an improvement, accept it
        if new_c5 < c5:
            best_h = h_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Perturb if no improvement
            if no_improvement_count > max_no_improvement:
                h_perturbed = best_h + np.random.normal(0, perturbation_strength, len(best_h))
                h_perturbed = project_to_feasible(h_perturbed, n_points)
                new_c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
                if new_c5_perturbed < c5:
                    best_h = h_perturbed
                    no_improvement_count = 0
                
        iterations += 1
        
        # Occasionally increase resolution if there's still room
        if iterations % 10 == 0 and n_points < 100:
            # Try a higher resolution
            higher_res = min(100, n_points + 15)
            if higher_res > n_points:
                dx_higher = 2.0 / higher_res
                h_interp = np.interp(np.linspace(0, 2, higher_res), 
                                     np.linspace(0, 2, n_points), 
                                     best_h)
                # Project and evaluate with new resolution
                h_proj = project_to_feasible(h_interp, higher_res)
                c5_higher = evaluate_erdos_solution(h_proj, dx_higher)
                if c5_higher < best_c5:
                    best_c5 = c5_higher
                    best_h = h_proj
                    best_n_points = higher_res
                    n_points = higher_res
                    dx = dx_higher
    
    return list(best_h), best_c5, best_n_points
