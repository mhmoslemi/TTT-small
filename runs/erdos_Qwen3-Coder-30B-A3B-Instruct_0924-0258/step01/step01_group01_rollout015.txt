import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with lower resolution
    n_low = 10
    n_high = 100
    n_points = n_low
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Track best ever
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Adaptive Simulated Annealing parameters
    T = 0.1
    cooling = 0.99
    iter_per_temp = 500
    
    iter_count = 0
    last_c5 = best_c5
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for _ in range(iter_per_temp):
            if time.time() - start_time > budget_s:
                break
                
            # Perturb with variance proportional to temperature
            noise = np.random.normal(0, T, n_points)
            h_new = h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_constraints(h_new, dx)
            c5_new = evaluate(h_new, dx)
            
            delta = c5_new - last_c5
            if delta < 0 or np.random.uniform() < np.exp(-delta / T):
                h = h_new
                last_c5 = c5_new
                if last_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = last_c5
                    
            iter_count += 1
            
        # Cool down
        T *= cooling
        
        if T < 1e-6:
            # Try to interpolate to next resolution
            if n_points < n_high:
                h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                                  np.linspace(0, 2, n_points), h)
                h = h_long
                n_points = 2 * n_points
                dx = 2.0 / n_points
                h = project_to_constraints(h, dx)
                last_c5 = evaluate(h, dx)
                T = 0.1
            else:
                break
    
    return best_h.tolist(), best_c5, len(best_h)
