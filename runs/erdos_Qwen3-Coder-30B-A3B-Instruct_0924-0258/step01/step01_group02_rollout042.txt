import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    # Binary search parameters
    low = 0.37
    high = compute_c5(h)
    tolerance = 1e-4
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    def is_feasible(c5_bound):
        """Check if there exists a feasible h with C5 <= c5_bound"""
        # Create variables
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == target_sum
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            # Compute correlation for shift k
            # This is a rough approximation, but sufficient for feasibility checking
            corr_sum = 0
            for i in range(n_points - k):
                corr_sum += h_vars[i] * (1 - h_vars[i + k])
            # Approximate as convex constraint (will be tight for large n)
            constraints.append(corr_sum * dx <= c5_bound)
        
        # Objective (minimize something to make feasibility problem well-defined)
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            return prob.status == 'optimal'
        except:
            return False
    
    # Binary search on C5
    while high - low > tolerance and (time.time() - start_time) < budget_s - 2:
        mid = (low + high) / 2
        if is_feasible(mid):
            high = mid
        else:
            low = mid
    
    # Final optimization to improve solution (within remaining time)
    # Start with best solution from binary search
    h_opt = best_h.copy()
    
    # Enhanced local search with multiple restarts
    for restart in range(10):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Perturb solution slightly for restart
        h_local = h_opt.copy() + np.random.normal(0, 0.01, n_points)
        h_local = np.clip(h_local, 0, 1)
        h_local = h_local * target_sum / np.sum(h_local)
        
        # Simple gradient approximation with projected gradient descent
        lr = 0.005
        for _ in range(1000):
            if (time.time() - start_time) > budget_s - 1:
                break
                
            # Compute gradient (approximately)
            eps = 1e-5
            grads = np.zeros(n_points)
            base_c5 = compute_c5(h_local)
            
            for i in range(n_points):
                h_test = h_local.copy()
                h_test[i] += eps
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * target_sum / np.sum(h_test)
                test_c5 = compute_c5(h_test)
                grads[i] = (test_c5 - base_c5) / eps
            
            # Update with gradient and project
            h_local = h_local - lr * grads
            h_local = np.clip(h_local, 0, 1)
            h_local = h_local * target_sum / np.sum(h_local)
            
            new_c5 = compute_c5(h_local)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_local.copy()
    
    return (best_h.tolist(), best_c5, n_points)
