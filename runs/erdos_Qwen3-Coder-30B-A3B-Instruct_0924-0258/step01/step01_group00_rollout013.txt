import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():

    else:
        h_start = np.ones(30) / 2
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Hyperparameters
    resolutions = [15, 20, 25, 30, 35, 40, 45]
    max_iter_per_res = 50
    restarts_per_res = 3
    max_time_per_res = 50
    
    # Run optimization at different resolutions
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if necessary
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), max_time_per_res)
        
        # Multiple restarts
        best_h_level = h_start.copy()
        best_c5_level = evaluate_erdos_solution(h_start, dx)
        
        for _ in range(restarts_per_res):
            if time() - start_time > budget_s:
                break
                
            # Random perturbation
            h_perturbed = project_to_feasible(h_start + np.random.normal(0, 0.03, n_points), n_points)
            
            # Optimization using SLSQP
            def objective(h):
                return -evaluate_erdos_solution(h, dx)
            
            def constraint_eq(h):
                return np.sum(h) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint_eq}
            
            try:
                res = minimize(objective, h_perturbed, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': max_iter_per_res, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5_level:
                        best_c5_level = c5_opt
                        best_h_level = h_opt.copy()
            except:
                pass
        
        # Update if better solution found
        if best_c5_level < best_c5:
            best_h = best_h_level.copy()
            best_c5 = best_c5_level
            best_n_points = n_points
            
        h_start = best_h_level.copy()
        
        # Early stopping if we're close to target
        if best_c5 < 0.3807:
            break
    
    return (list(best_h), best_c5, best_n_points)
