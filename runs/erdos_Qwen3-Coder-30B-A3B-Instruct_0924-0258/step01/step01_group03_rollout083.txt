import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5  # Simple uniform start
    
    # Initialize
    h = h0.copy().astype(float)
    dx = 2.0 / n_points
    max_iter = 10000
    lr = 0.01
    patience = 0
    best_c5 = float('inf')
    best_h = None
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute correlation for all shifts
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        
        # Compute gradient: derivative of max correlation
        grad = np.zeros(n_points)
        for j in range(n_points):
            # Find corresponding index in correlation
            idx = (j + k_max) % (2 * n_points - 1)
            if idx < n_points:
                grad[j] = 1 - 2 * h[idx]
        
        # Update h
        h = h - lr * grad
        
        # Project to feasible region [0,1]
        h = np.clip(h, 0, 1)
        
        # Project to constraint ∫h dx = 1 (i.e. sum(h) = n/2)
        h_sum = np.sum(h)
        if abs(h_sum) > 1e-12:
            h = h * (n_points / 2.0) / h_sum
        else:
            h = np.ones_like(h) * (n_points / 2.0) / n_points
        
        # Evaluate
        c5 = evaluate(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 100:
                lr *= 0.9
                patience = 0
                
        # Occasionally perturb to escape local minima
        if i % 50 == 0 and i > 0:
            h += np.random.normal(0, 1e-4, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum) > 1e-12:
                h = h * (n_points / 2.0) / h_sum
    
    if best_h is not None:
        return (best_h.tolist(), float(best_c5), n_points)
    else:
        # Fallback
        final_c5 = evaluate(h0, dx)
        return (h0.tolist(), float(final_c5), n_points)
