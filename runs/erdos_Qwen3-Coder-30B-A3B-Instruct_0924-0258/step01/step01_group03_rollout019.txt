import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5  # Simple uniform start

    dx = 2.0 / n_points
    
    # Optimization parameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 50
    improvement_threshold = 1e-6
    best_c5 = float('inf')
    best_h = h.copy()
    no_improvement_count = 0
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute correlations
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(np.abs(corr))
        
        # Compute gradient for this correlation
        gradient = np.zeros_like(h)
        # k_max corresponds to shifting h by k_max-n_points+1 to n_points-1
        # we compute the derivative w.r.t to h:
        # for some shift k, contribution to partial C5 / partial h_j
        # is (1 - h_{j+k}) if k is positive and j+k < n, or (1 - h_{j-k}) if k is negative and j-k >= 0
        
        # Direct gradient computation for the maximizing k
        max_shift = k_max - (n_points - 1)
        if max_shift >= 0:
            # positive shift: h[j] contributes to h[j+shift] if j+shift < n
            for j in range(n_points):
                j_new = j + max_shift
                if j_new < n_points:
                    gradient[j] += (1 - h[j_new])
        else:
            # negative shift: h[j] contributes to h[j+shift] if j+shift >= 0
            for j in range(n_points):
                j_new = j + max_shift  # negative shift
                if j_new >= 0:
                    gradient[j] += (1 - h[j_new])
        
        # Normalize gradient
        gradient /= n_points
        
        # Update h
        h = h - learning_rate * gradient
        # Project to [0,1]
        h = np.clip(h, 0, 1)
        # Renormalize to correct sum
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > patience:
                # Add small noise and reproject
                h += np.random.uniform(-1e-3, 1e-3, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                no_improvement_count = 0
                # Reduce learning rate
                learning_rate *= 0.9
                
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    return (best_h.tolist(), float(final_c5), n_points)
