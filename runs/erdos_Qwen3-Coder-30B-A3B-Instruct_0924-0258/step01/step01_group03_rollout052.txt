import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Adjust to meet target sum while staying in [0,1]
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        idxs = np.argsort(h)
        for i in range(len(idxs)-1, -1, -1):
            j = idxs[i]
            if diff <= 0:
                break
            can_add = 1.0 - h[j]
            to_add = min(can_add, diff)
            h[j] += to_add
            diff -= to_add
    else:
        # Need to decrease values
        idxs = np.argsort(h)
        for i in range(len(idxs)):
            j = idxs[i]
            if diff >= 0:
                break
            can_subtract = h[j]
            to_sub = min(can_subtract, -diff)
            h[j] -= to_sub
            diff += to_sub
    return h
