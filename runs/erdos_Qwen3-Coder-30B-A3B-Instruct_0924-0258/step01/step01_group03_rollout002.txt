import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum to target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Normalize to target sum
    h = h * target_sum / current_sum
    # Clamp again
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5  # Simple uniform start
    
    # Set dx
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize
    h = h0.copy()
    h = project_to_feasible(h, target_sum)
    
    # Hyperparameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 100
    patience_counter = 0
    best_c5 = float('inf')
    best_h = None
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Evaluate current
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            # Add small noise to escape local minima
            noise = np.random.normal(0, 0.001, n_points)
            h = h + noise
            h = project_to_feasible(h, target_sum)
            patience_counter = 0
        
        # Compute correlation for all shifts
        corr = correlate(h, 1 - h, mode="full")
        # Find shift that gives maximum correlation
        max_idx = np.argmax(corr)
        # The correlation at max_idx is for shift (max_idx - n_points + 1):
        # shift = max_idx - (n_points - 1) = max_idx - n_points + 1
        
        # Compute gradient w.r.t. h
        # The contribution to corr at shift k is sum over i of h[i] * (1 - h[i+k])
        # The gradient w.r.t. h[i] at shift k is sum over j such that j+k=i of (1-h[j])
        # i.e. gradient[i] = sum over j of 1-h[j] where j+k=i => j = i-k
        # But we need to be careful with indices.
        # For shift k (k elements to the right):
        # Contribution of h[i] to the k-shift is h[i] * (1-h[i+k]) 
        # => gradient w.r.t. h[i] is (1 - h[i+k]) 
        # We'll assume the max shift we want is from the middle to avoid boundary effects
        # Fix: use derivative approximation by finite difference
        # Actually we do want gradient calculation in terms of the window of correlation
        grad = np.zeros(n_points)
        k = max_idx - (n_points - 1)  # Convert from full correlation index to shift
        
        # Evaluating gradient more carefully:
        # When we shift by k (k>0), we correlate h[i] with (1-h[i+k]); 
        # For i in range(0, n_points-k), we have term h[i] * (1-h[i+k])
        # Gradient w.r.t. h[i] is (1-h[i+k]) for i >= 0 and i+k < n_points
        # Gradient w.r.t. h[i+k] is -h[i] for i >= 0 and i+k < n_points
        for i in range(max(0, -k), min(n_points, n_points-k)):
            if i+k < n_points and i+k >= 0:
                if i >= 0 and i < n_points:
                    grad[i] += (1. - h[i+k])
                if i+k >= 0 and i+k < n_points:
                    grad[i+k] -= h[i]
        
        # Update with gradient
        h = h - learning_rate * grad
        h = project_to_feasible(h, target_sum)
        
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        # Fallback
        final_h = project_to_feasible(h0, target_sum)
        final_c5 = evaluate(final_h, dx)
        return (final_h.tolist(), float(final_c5), n_points)
