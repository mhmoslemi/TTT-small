import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():

        n_points = len(h)

    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Ensure exact sum constraint
    current_sum = np.sum(h)
    diff = (n_points / 2) - current_sum
    if abs(diff) > 1e-10:
        # Distribute diff to elements
        idx = np.argsort(h)
        for i in range(len(idx)-1, -1, -1):
            if diff == 0:
                break
            j = idx[i]
            if h[j] < 1.0 and diff > 0:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
        # Renormalize to ensure sum exactly n_points/2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient-based optimization parameters
    learning_rate = 0.01
    max_iter = 1000
    patience = 20
    patience_counter = 0
    prev_best = best_c5
    
    # Main loop
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute correlation to find max shift k
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        # k ranges from -(n-1) to n-1
        k = max_k - (n_points - 1)
        
        # Compute gradient (approximate)
        grad = np.zeros_like(h)
        if k < 0:
            # h[i+k] for i in range(len(h)-|k|) 
            for i in range(len(h) + k):
                # grad[i] += (1 - 2*h[i+k])
                if i + k < len(h) and i + k >= 0:
                    grad[i] += (1 - 2 * h[i + k])
        elif k > 0:
            # h[i-k] for i in range(k, len(h)) 
            for i in range(k, len(h)):
                # grad[i] += (1 - 2*h[i-k])
                grad[i] += (1 - 2 * h[i - k])
        else:
            # k == 0, h[i] terms 
            for i in range(len(h)):
                grad[i] += (1 - 2 * h[i]) * 2  # Actually, that's wrong - let's recompute properly
			
        # Recompute gradient properly (parsed from correlation logic)
        # We want the gradient for the function that maximizes the overlap
        # The overlap for shift k is sum h[i] * (1 - h[i+k]) 
        # For gradient w.r.t h[j]:
        # If h[j] appears in term h[i] * (1 - h[i+k]), i.e., j==i
        # and we also have a term h[i] * (1 - h[i+k]), i.e., j == i+k, so there's 2 terms
        # Each element contributes to two terms (itself and shifted)
        # For shift k=1 (shift right by one), j contributes to terms:
        # - i=j: h[i](1-h[i+1]) => this term's gradient to h[j] is (1-h[j+1]) 
        # - i=j-1 (if j-1>=0): h[j-1](1-h[j]) => this term's gradient is -h[j-1]
        # So grad[j] = (1-h[j+1]) - h[j-1], where r = j+1, l = j-1
        # Consider k>0:   1: right shift by k, so h[j] is in contribution to h[j+k](1-h[j+k+1]), h[j+k][1-h[j+k+1]]
        # So last term: (1-h[j+k]) for h'[j] where j+k is in valid range
        # But only for range where valid
        # Again checking old method
        # Simpler is that h[i] shows up in term (1-h[i+k]) gradient (which is -h'[i])
        # and it appears in term h[i+k] * (1-h[i+k+k]) gradient (which is h[i+k+1] - h[k])
        # No, it's clearer. 
        # Derivative of sum h[i] * (1 - h[i+k]) w.r.t. h[j]: 
        # For j = i: contributes (1 - h[i+k])  
        # For j = i+k: contributes -h[i] 
        # For all other j:    0
        # So we build it:
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j1 = i   # For term h[i] * (1-h[i+k]) -> grad wrt h[j1==i] = (1-h[i+k])

            if 0 <= j1 < len(h):
                grad[j1] += 1 - h[i+k] if 0 <= i+k < len(h) else 1
            if 0 <= j2 < len(h):
                grad[j2] -= h[i] if 0 <= i < len(h) else 0
                
        # Update step
        h = h - learning_rate * grad
        
        # Project back to feasible set: [0,1] and sum = n_points/2
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - n_points / 2) > 1e-10:
            h = h * (n_points / 2) / current_sum
            
        # Evaluation
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            learning_rate *= 0.95
            patience_counter = 0
            
    return (best_h.tolist(), float(best_c5), n_points)
