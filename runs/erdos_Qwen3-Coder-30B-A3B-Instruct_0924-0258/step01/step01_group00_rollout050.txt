import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def optimize_with_perturbation(h_start, n_points, dx, time_limit):
    """Use a hybrid approach combining gradient descent and geometric perturbation"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate a set of perturbations with different magnitudes
    perturbations = [0.01, 0.02, 0.05]
    
    # Use a more structured optimization approach
    for _ in range(1000):
        if time() - start_time > time_limit:
            break
            
        # Use a sampled gradient descent method
        h_current = best_h.copy()
        
        # Try different perturbations
        for pert in perturbations:
            if time() - start_time > time_limit:
                break
                
            h_perturbed = project_to_feasible(h_current + np.random.normal(0, pert, n_points), n_points)
            
            # Evaluate with a simple geometric drift toward better values
            c5_new = evaluate_erdos_solution(h_perturbed, dx)
            if c5_new < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_new
                
                # Double down on this good direction
                h_direction = h_perturbed - h_current
                if np.linalg.norm(h_direction) > 1e-6:
                    h_reinforce = project_to_feasible(h_perturbed + 0.5 * h_direction, n_points)
                    c5_reinforce = evaluate_erdos_solution(h_reinforce, dx)
                    if c5_reinforce < best_c5:
                        best_h = h_reinforce.copy()
                        best_c5 = c5_reinforce
                        
        # Occasionally try a bigger step to avoid local minima
        if np.random.rand() < 0.15:
            idx = np.random.randint(0, n_points)
            scale = np.random.normal(0, 0.1)
            h_perturbed = h_current.copy()
            h_perturbed[idx] = np.clip(h_perturbed[idx] + scale, 0, 1)
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            c5_new = evaluate_erdos_solution(h_perturbed, dx)
            if c5_new < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_new
    
    return best_h, best_c5
