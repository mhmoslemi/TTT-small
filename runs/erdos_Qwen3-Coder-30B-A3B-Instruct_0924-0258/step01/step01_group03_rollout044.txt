import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

    # Gradient-based optimization
    best_h = None
    best_c5 = float('inf')
    learning_rate = 0.01
    stagnation_counter = 0
    last_improvement = 0
    
    # Ensure initial h meets constraints
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h)

    iteration = 0
    while time.time() - start_time < budget_s:
        # Compute cross-correlations
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(np.abs(corr))
        
        # Compute gradient
        gradient = np.zeros(n_points)
        shift = k_max - (len(corr) - 1) // 2  # Convert index to actual shift
        
        for i in range(n_points):
            j = (i + shift) % n_points
            if 0 <= j < n_points:
                gradient[i] += (1 - h[j])
        
        # Normalize gradient
        gradient /= n_points
        
        # Update h
        h = h - learning_rate * gradient
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        
        # Normalize to maintain sum constraint
        h = h * (n_points / 2.0) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate(h, dx)
        
        if current_c5 < best_c5:

            last_improvement = iteration
