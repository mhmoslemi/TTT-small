import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def compute_gradient(h, dx, critical_lag):
    n = len(h)
    # Build gradient based on the critical lag
    # For consistency with correlation result:
    # The correlation of h and (1-h) at lag k has contribution:
    # sum h[i] * (1 - h[i+k]) for valid i
    # Gradient: -h[i+k] if i is in valid range for lag k, for varying i
    # However, it's more direct to use the exact derivative form.
    # Estimate gradient via finite differences
    grad = np.zeros_like(h)
    epsilon = 1e-4
    h_plus = h.copy()
    h_minus = h.copy()
    for i in range(n):
        h_plus[i] = min(h[i] + epsilon, 1.0)
        h_minus[i] = max(h[i] - epsilon, 0.0)
        c_plus = evaluate(h_plus, dx)
        c_minus = evaluate(h_minus, dx)
        grad[i] = (c_plus - c_minus) / (2 * epsilon)
        h_plus[i] = h[i]
        h_minus[i] = h[i]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with lower resolution
    n_low = 10
    n_high = 70
    n_points = n_low
    dx = 2.0 / n_points
    
    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Adaptive gradient descent with line search
    max_iter_per_res = 1000
    learning_rate = 1e-2
    
    iteration = 0
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for _ in range(max_iter_per_res):
            if time.time() - start_time > budget_s:
                break
                
            # Compute correlation to determine critical ping
            corr = correlate(h, 1-h, mode='full')
            # Find top lags
            abs_corr = np.abs(corr)
            idxs = np.argsort(abs_corr)[-n_points//8:]  # Top 1/8 of lags
            critical_lags = [i - (n_points - 1) for i in idxs]  # Convert to actual lag indices

            # Estimate gradient using finite differences
            grad = compute_gradient(h, dx, critical_lags[0] if critical_lags else 0)
            
            # Line search: try multiple step sizes
            best_step_size = 0.0
            best_h_new = h.copy()
            best_c5_new = best_c5
            
            step_sizes = [1e-4, 5e-4, 1e-3, 5e-3, 1e-2]
            for step_size in step_sizes:
                h_new = h - step_size * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = project_to_constraints(h_new, dx)
                c5_new = evaluate(h_new, dx)
                if c5_new < best_c5_new:
                    best_c5_new = c5_new
                    best_h_new = h_new.copy()
                    best_step_size = step_size
                    
            # Accept best step
            if best_step_size > 0:
                h = best_h_new
                if best_c5_new < best_c5:
                    best_c5 = best_c5_new
                    best_h = h.copy()
                    
            iteration += 1
            
        # Adaptive resolution upgrade
        if n_points < n_high and time.time() - start_time < budget_s * 0.7:
            # Interpolate to higher resolution
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            if evaluate(h, dx) < best_c5:
                best_c5 = evaluate(h, dx)
                best_h = h.copy()
                
        # Occasionally re-initialize
        if iteration % (max_iter_per_res * 2) == 0:
            h = project_to_constraints(np.random.uniform(0, 1, n_points), dx)
            
    return best_h.tolist(), best_c5, len(best_h)
