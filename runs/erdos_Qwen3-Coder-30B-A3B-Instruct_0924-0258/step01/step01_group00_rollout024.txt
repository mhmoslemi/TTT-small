import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) / 2
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    target_c5 = 0.3807
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time() - start_time < budget_s * 0.95:
        iter_count += 1
        if no_improvement_count >= max_no_improvement:
            # Add noise to escape local minima
            h = project_to_feasible(h + np.random.normal(0, 0.01, n_points), n_points)
            no_improvement_count = 0
        
        c5 = evaluate_erdos_solution(h, dx)
        
        if c5 <= target_c5:
            break
            
        # Find the maximum overlap and corresponding shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # The shift corresponds to the middle of the full correlation
        shift = max_idx - (len(h) - 1)
        
        # Identify the most influential indices for this shift
        # Indices x where h(x)*h(x+shift) contributes most to overlap
        if shift >= 0:
            # h(x+shift) exists for all x in 0 to len(h)-shift-1
            valid_x = np.arange(len(h) - shift)
            overlap_terms = h[valid_x] * h[valid_x + shift]
        else:
            # shift is negative, h(x+shift) = h(x-shift)
            valid_x = np.arange(-shift, len(h))
            overlap_terms = h[valid_x] * h[valid_x - shift]
        
        # Sort indices by overlap contribution
        sorted_indices = np.argsort(overlap_terms)[::-1]
        top_indices = sorted_indices[:max(1, len(sorted_indices) // 10)]
        
        # Reduce h values at these top indices
        adjustment = (max_val * dx - target_c5) / (len(h) * 2)
        if adjustment <= 0:
            adjustment = 1e-6
        
        h_new = h.copy()
        for idx in top_indices:
            if shift >= 0:
                if idx + shift < len(h):
                    h_new[idx] = max(0, h_new[idx] - adjustment)
                    h_new[idx + shift] = max(0, h_new[idx + shift] - adjustment)
            else:
                if idx - shift < len(h):
                    h_new[idx] = max(0, h_new[idx] - adjustment)
                    h_new[idx - shift] = max(0, h_new[idx - shift] - adjustment)
                    
        # Project to feasible domain
        h_new = project_to_feasible(h_new, n_points)
        
        # Check if this improved the solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < c5:
            h = h_new
            no_improvement_count = 0
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            no_improvement_count += 1
            
        # Occasionally upgrade resolution
        if iter_count % 100 == 0 and len(h) < 100:
            new_n_points = min(len(h) + 15, 100)
            new_h = np.interp(np.linspace(0, 2, new_n_points), 
                             np.linspace(0, 2, len(h)), h)
            h = project_to_feasible(new_h, new_n_points)
            n_points = new_n_points
            dx = 2.0 / n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, len(best_h))
