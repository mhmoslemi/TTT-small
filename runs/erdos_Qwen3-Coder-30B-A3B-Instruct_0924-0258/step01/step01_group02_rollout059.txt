import numpy as np
import time

def evaluate_h(h):
    """Evaluate C5 for given h"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def project_h(h):
    """Project h to feasible region"""
    h = np.clip(h, 0, 1)
    h = h * (len(h) / 2.0) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial h meets constraint
    h0 = h0 * target_sum / np.sum(h0)
    
    # Genetic Algorithm parameters
    pop_size = 50
    n_generations = 1000
    mutation_rate = 0.05
    crossover_rate = 0.8
    elite_size = pop_size // 4
    
    # Initialize population
    population = [h0.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    # Fitness evaluation
    def fitness(h):
        return -evaluate_h(h)
    
    start_time = time.time()
    best_c5 = evaluate_h(h0)
    best_h = h0.copy()
    
    generation = 0
    no_improvement_count = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness
        fitness_scores = [fitness(h) for h in population]
        sorted_indices = np.argsort(fitness_scores)
        
        # Update best solution
        current_best = -fitness_scores[sorted_indices[0]]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[sorted_indices[0]].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early termination
        if no_improvement_count > 1000:
            break
            
        # Selection (tournament)
        new_population = []
        # Elitism
        for i in range(elite_size):
            new_population.append(population[sorted_indices[i]].copy())
            
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            tourn_size = 3
            idx1 = np.random.choice(sorted_indices[:pop_size//2])
            idx2 = np.random.choice(sorted_indices[:pop_size//2])
            if fitness_scores[idx1] > fitness_scores[idx2]:
                parent1 = population[idx1]
            else:
                parent1 = population[idx2]
                
            idx3 = np.random.choice(sorted_indices[:pop_size//2])
            idx4 = np.random.choice(sorted_indices[:pop_size//2])
            if fitness_scores[idx3] > fitness_scores[idx4]:
                parent2 = population[idx3]
            else:
                parent2 = population[idx4]
                
            # Crossover
            if np.random.rand() < crossover_rate:
                mask = np.random.rand(n_points) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)

                child1 = parent1.copy()
                child2 = parent2.copy()
                
            # Mutation
            for child in [child1, child2]:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.normal(0, 0.05)
                child = project_h(child)
                new_population.append(child)
                
        population = new_population[:pop_size]
        generation += 1
    
    return (best_h.tolist(), best_c5, n_points)
