import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def evaluate_candidate(h_vec):
        h_vec = project_feasible(h_vec)
        return compute_C5(h_vec)
    
    def coordinate_descent(h_vec, max_iter=30):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = evaluate_candidate(h_current)
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
            improved = False
            # Try each coordinate with small steps
            for i in range(len(h_current)):
                if time.time() - start_time > budget_s - 0.1:
                    break
                original = h_current[i]
                steps = [0.01, -0.01, 0.02, -0.02]
                for step in steps:
                    h_current[i] = original + step
                    if 0 <= h_current[i] <= 1:
                        new_C5 = evaluate_candidate(h_current)
                        if new_C5 < best_C5:
                            best_C5 = new_C5
                            best_h = h_current.copy()
                            improved = True
                h_current[i] = original  # restore
            if not improved:
                break
        return best_h, best_C5
    
    def simplex_search(h_vec, max_iter=30):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = evaluate_candidate(h_current)
        
        # Create initial simplex around current solution
        simplex = [h_current]
        for i in range(2):
            perturbed = h_current + np.random.normal(0, 0.05, len(h_current))
            simplex.append(project_feasible(perturbed))
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Evaluate simplex members
            simplex_values = [evaluate_candidate(h) for h in simplex]
            
            # Sort by objective value
            indices = np.argsort(simplex_values)
            best_idx = indices[0]
            worst_idx = indices[-1]
            
            # Try reflection
            centroid = np.mean([h for i, h in enumerate(simplex) if i != worst_idx], axis=0)
            reflected = 2 * centroid - simplex[worst_idx]
            reflected = project_feasible(reflected)
            reflected_value = evaluate_candidate(reflected)
            
            if simplex_values[best_idx] <= reflected_value < simplex_values[worst_idx]:
                simplex[worst_idx] = reflected
            elif reflected_value < simplex_values[best_idx]:
                # Expansion
                expanded = 2 * reflected - centroid
                expanded = project_feasible(expanded)
                expanded_value = evaluate_candidate(expanded)
                if expanded_value < reflected_value:
                    simplex[worst_idx] = expanded
                else:
                    simplex[worst_idx] = reflected
            else:
                # Contraction
                if reflected_value < simplex_values[worst_idx]:
                    contracted = 0.5 * (simplex[worst_idx] + reflected)
                else:
                    contracted = 0.5 * (simplex[worst_idx] + simplex[best_idx])
                contracted = project_feasible(contracted)
                contracted_value = evaluate_candidate(contracted)
                if contracted_value < simplex_values[worst_idx]:
                    simplex[worst_idx] = contracted
                else:
                    # Shrink simplex
                    best_point = simplex[best_idx]
                    for i in range(1, len(simplex)):
                        simplex[i] = best_point + 0.5 * (simplex[i] - best_point)
            
            # Check for improvement
            new_values = [evaluate_candidate(h) for h in simplex]
            if np.min(new_values) < best_C5:
                best_C5 = np.min(new_values)
                best_h = simplex[np.argmin(new_values)].copy()
        
        return best_h, best_C5
    
    def swap_optimization(h_vec, attempts=30):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = evaluate_candidate(h_current)
        
        for _ in range(attempts):
            if time.time() - start_time > budget_s - 0.1:
                break
            i, j = np.random.choice(len(h_current), 2, replace=False)
            # Try swapping
            h_test = h_current.copy()
            h_test[i], h_test[j] = h_test[j], h_test[i]
            new_C5 = evaluate_candidate(h_test)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_test.copy()
        return best_h, best_C5
    
    # Generate diverse initial candidates
    candidates = []
    for _ in range(5):
        if time.time() - start_time > budget_s - 0.1:
            break
        # Latin hypercube sampling for diversity
        h_sample = np.random.random(n_points)
        h_sample = project_feasible(h_sample)
        candidates.append(h_sample)
    
    # Add the current best as first candidate
    candidates.insert(0, h.copy())
    
    # Evaluate candidates and select the best initial
    initial_candidates = [(p, evaluate_candidate(p)) for p in candidates]
    initial_candidates.sort(key=lambda x: x[1])
    h_best, C5_best = initial_candidates[0]
    
    # Optimization loop
    no_improvement_count = 0
    restart_count = 0
    
    while time.time() - start_time < budget_s - 0.1:
        # Part 1: Coordinate descent
        h_coord, C5_coord = coordinate_descent(h_best, 10)
        if C5_coord < C5_best:
            h_best, C5_best = h_coord, C5_coord
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Part 2: Simplex search
        h_simplex, C5_simplex = simplex_search(h_best, 10)
        if C5_simplex < C5_best:
            h_best, C5_best = h_simplex, C5_simplex
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Part 3: Swap optimization
        h_swap, C5_swap = swap_optimization(h_best, 20)
        if C5_swap < C5_best:
            h_best, C5_best = h_swap, C5_swap
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Restart if no improvement for a while
        if no_improvement_count > 50 and restart_count < 5:
            no_improvement_count = 0
            restart_count += 1
            # Generate new candidate
            h_new = np.random.random(n_points)
            h_new = project_feasible(h_new)
            h_best, C5_best = h_new, evaluate_candidate(h_new)
    
    # Final cleanup
    h_best = project_feasible(h_best)
    C5_final = compute_C5(h_best)
    
    return list(h_best), C5_final, n_points
