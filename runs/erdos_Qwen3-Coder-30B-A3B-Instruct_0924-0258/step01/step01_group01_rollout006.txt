import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def smooth_max(x, beta=10):
    return np.log(np.sum(np.exp(beta * x))) / beta

def compute_correlation_gradient(h, dx):
    n = len(h)
    # Compute correlation using FFT for efficiency
    h_fft = fft(h)
    one_minus_h_fft = fft(1 - h)
    corr_fft = h_fft * np.conj(one_minus_h_fft)
    corr = ifft(corr_fft).real
    # Shift to match correlate behavior
    corr = np.roll(corr, n-1)
    return corr * dx

def compute_objective_and_gradient(h, dx, beta=10):
    corr = compute_correlation_gradient(h, dx)
    # Compute smooth max approximation
    max_corr = smooth_max(corr, beta)
    # Gradient of smooth max
    exp_vals = np.exp(beta * corr)
    sum_exp = np.sum(exp_vals)
    grad = beta * exp_vals / sum_exp
    # Convolve back with h to get gradient in time domain
    h_fft = fft(h)
    one_minus_h_fft = fft(1-h)
    grad_fft = grad * np.conj(one_minus_h_fft)
    grad_corr = ifft(grad_fft).real
    grad_corr = np.roll(grad_corr, n-1)
    return max_corr, grad_corr

def backtracking_armijo(h, grad, alpha_max=1.0, c1=1e-4, beta=0.5):
    alpha = alpha_max
    obj_prev = evaluate(h, dx)
    while True:
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_constraints(h_new, dx)
        obj_new = evaluate(h_new, dx)
        if obj_new <= obj_prev - c1 * alpha * np.dot(grad, grad):
            return alpha
        alpha *= beta
        if alpha < 1e-10:
            return alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with a reasonably sized resolution
    n_points = 64
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) > n_points:
            h = h[:n_points]
        else:
            h = np.interp(np.linspace(0, 2, n_points), 
                         np.linspace(0, 2, len(h)), h)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
        
    # Track best ever
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Optimization parameters
    beta = 10.0
    learning_rate = 1e-3
    max_iter = 1000
    iter_count = 0
    
    # Main optimization loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.98:
            break
            
        # Compute current objective and gradient
        obj_val, grad = compute_objective_and_gradient(h, dx, beta)
        
        # Perform line search to get good step size
        alpha = backtracking_armijo(h, grad, learning_rate)
        
        # Take step
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_constraints(h_new, dx)
        
        # Evaluate new value
        new_obj_val = evaluate(h_new, dx)
        
        if new_obj_val < obj_val:
            h = h_new
            obj_val = new_obj_val
            if obj_val < best_c5:
                best_h = h.copy()
                best_c5 = obj_val
                
        # Adaptive learning rate
        if (iter_count + 1) % 200 == 0:
            learning_rate *= 0.8
            
        # Adaptive smoothness parameter
        if (iter_count + 1) % 500 == 0:
            beta *= 2.0
            
    return best_h.tolist(), best_c5, len(best_h)
