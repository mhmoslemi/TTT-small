import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def gradient_step(h, dx, step_size=0.01):
    # Analytical gradient approximation
    n = len(h)
    grad = np.zeros(n)
    # Approximate gradient as if we have the full correlation
    # This uses the fact that the derivative is roughly (1-2h)*dx
    # but we're being smart about it with a proper correlation
    h1 = np.concatenate([h, np.zeros(n-1)])
    h2 = np.concatenate([np.zeros(n-1), 1-h])
    corr = correlate(h1, h2, mode='full')
    
    # We want to be more accurate here
    # Going to compute a sharper gradient using finite differences:
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += 0.001
        h_minus[i] -= 0.001
        h_plus = project_to_constraints(h_plus, dx)
        h_minus = project_to_constraints(h_minus, dx)
        
        c_plus = evaluate(h_plus, dx)
        c_minus = evaluate(h_minus, dx)
        grad[i] = (c_plus - c_minus) / 0.002
    
    h_new = h - step_size * grad
    return project_to_constraints(h_new, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial or small size
    n_points = 10
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals() and len(initial_h_values) >= 10:
        h = np.array(initial_h_values[:n_points])
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    last_c5 = best_c5
    
    # Parameters
    max_iter = 50000
    iter_count = 0
    time_limit = budget_s * 0.95
    
    # Optimization loop
    while iter_count < max_iter and time.time() - start_time < time_limit:
        # Try gradient step first
        h_new = gradient_step(h, dx, step_size=0.01)
        c5_new = evaluate(h_new, dx)
        
        # Accept improvement or with probability
        delta = c5_new - last_c5
        if delta < 0 or np.random.uniform() < np.exp(-delta * 5):
            h = h_new
            last_c5 = c5_new
            if last_c5 < best_c5:
                best_h = h.copy()
                best_c5 = last_c5
        
        # Occasionally also do a small random perturbation
        if np.random.rand() < 0.1:
            # Random perturbation
            h_pert = h + np.random.normal(0, 0.01, n_points)
            h_pert = project_to_constraints(h_pert, dx)
            c5_pert = evaluate(h_pert, dx)
            if c5_pert < last_c5:
                h = h_pert
                last_c5 = c5_pert
        
        iter_count += 1
        
        # Also occasionally interpolate
        if iter_count % 1000 == 0 and n_points < 70:
            # Make it longer
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            last_c5 = evaluate(h, dx)
    
    return best_h.tolist(), best_c5, len(best_h)
