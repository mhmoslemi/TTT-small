import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def setup_optimization_problem(h_start, n_points, dx):
    """Set up optimization problem for a given resolution"""
    # Initial feasible projection
    h_feasible = project_to_feasible(h_start, n_points)
    
    # Set up bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    return h_feasible, bounds, cons

def local_optimization_step(h_start, bounds, cons, dx, max_iter=30):
    """Perform local optimization using SLSQP"""
    try:
        res = minimize(
            lambda h: evaluate_erdos_solution(h, dx),
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-6}
        )
        if res.success:
            return project_to_feasible(res.x, len(h_start))
    except:
        pass
    return h_start

def global_search_step(h_start, bounds, cons, dx, n_points, iteration=0):
    """Perform global search step with adaptive perturbation"""
    # Create a more diverse perturbation
    entropy = np.std(h_start) if np.std(h_start) > 0.01 else 0.1
    perturbation = np.random.normal(0, 0.05 * (1 + iteration * 0.05), n_points) 
    
    # Occasionally introduce larger perturbations for exploration
    if iteration % 5 == 0:
        perturbation += np.random.uniform(-0.1, 0.1, n_points)
    
    h_perturbed = project_to_feasible(h_start + perturbation, n_points)
    
    # Refine with local optimization
    h_opt = local_optimization_step(h_perturbed, bounds, cons, dx)
    
    return h_opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    if time() > time() + budget_s:
        return [], float('inf'), 0
    
    start_time = time()
    
    # Initial construction

    else:

    # Resolution levels to try
    resolutions = [15, 20, 25, 30, 35, 40, 45]
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Optimizing across multiple resolutions
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if necessary
        if len(h_start) != n_points:
            h_interp = np.interp(
                np.linspace(0, 2, n_points), 
                np.linspace(0, 2, len(h_start)), 
                h_start
            )
            h_start = project_to_feasible(h_interp, n_points)
        
        # Create problem setup
        h_feasible, bounds, cons = setup_optimization_problem(h_start, n_points, dx)
        
        # Multiple restarts with increasing iteration count to improve exploration
        n_restarts = 2 + i // 2
        best_h_level = h_feasible.copy()
        best_c5_level = evaluate_erdos_solution(h_feasible, dx)
        
        for restart in range(n_restarts):
            if time() - start_time > budget_s * 0.98:
                break
                
            # Start with a valid point from previous level or random
            h_restart = project_to_feasible(
                h_start + np.random.normal(0, 0.02, n_points), 
                n_points
            )
            
            # Improve the solution with local and global search
            h_opt = h_restart.copy()
            for iter_step in range(15):
                h_opt = global_search_step(h_opt, bounds, cons, dx, n_points, iter_step)
                if time() - start_time > budget_s * 0.98:

            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < best_c5_level:

        # If improved, update global best
        if best_c5_level < best_c5:
            best_h = best_h_level.copy()
            best_c5 = best_c5_level
            best_n_points = n_points
        
        h_start = best_h_level.copy()
