import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available

        n_points = len(h0)

    # Initialize
    h = h0.copy().astype(float)
    h = h * (n_points / 2.0) / np.sum(h)  # Normalize to sum = n_points/2
    h = np.clip(h, 0, 1)  # Ensure in [0,1]
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent parameters
    eta = 0.01  # Learning rate
    patience = 0  # For learning rate decay
    last_improvement = 0  # For adaptive perturbations
    smoothing_window = 5  # For gradient smoothing
    
    # Store recent gradients for smoothing
    grad_history = []
    
    iteration = 0
    while time.time() - start_time < budget_s:
        # Compute correlation to find max overlap shift
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(np.abs(corr))
        
        # Compute gradient based on k_max shift
        grad = np.zeros_like(h)
        n = len(h)
        # Gradient contribution to each point from its contribution to max shift
        for i in range(n):
            # For shift k_max:
            # if k_max is positive, j = i + k_max
            # if k_max is negative, j = i + k_max (wrap around if need be)
            if k_max < n:
                j = (i + k_max) % n
                grad[i] += (1 - h[j])  # sign convention is already in k_max
            else:
                j = (i - (2*n - k_max)) % n
                grad[i] += (1 - h[j])
        
        # Store gradient history
        grad_history.append(grad)
        if len(grad_history) > smoothing_window:
            grad_history.pop(0)
            
        # Average gradients
        mean_grad = np.mean(grad_history, axis=0)
        
        # Update step with gradient descent
        h = h - eta * mean_grad
        
        # Project to feasible region: [0, 1] and sum constraint
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - n_points / 2.0) > 1e-10:
            # Distribute the difference
            diff = n_points / 2.0 - current_sum
            # Just scale the entire vector
            if abs(current_sum) > 1e-10:
                h = h * (n_points/2.0) / current_sum
            else:
                h.fill(0.5)
            # Clamp again to ensure range
            h = np.clip(h, 0, 1)
        
        # Check for improvement
        c5 = evaluate(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
            last_improvement = iteration

            patience += 1
            if patience >= 50:

                pert = np.random.uniform(-1e-3, 1e-3, n_points)

                patience = 0
                # Reset gradient history
                grad_history = []
                # Decay learning rate
                eta *= 0.9
        
        # Adaptive learning rate decay
        if iteration - last_improvement > 100:
            eta *= 0.9
            
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    return (best_h.tolist(), float(final_c5), n_points)
