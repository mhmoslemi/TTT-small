import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust to maintain sum
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        idx = np.argsort(h)
        for i in range(len(idx)-1, -1, -1):
            if diff <= 0:
                break
            j = idx[i]
            if h[j] < 1:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
    else:
        # Need to decrease values
        idx = np.argsort(h)
        for i in range(len(idx)):
            if diff >= 0:
                break
            j = idx[i]
            if h[j] > 0:
                update = min(-diff, h[j])
                h[j] -= update
                diff += update
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Normalize initial h
    target_sum = n_points / 2.0
    h0 = project_to_feasible(h0, target_sum)
    dx = 2.0 / n_points
    
    best_h = h0.copy()
    best_c5 = evaluate(h0, dx)
    
    # Parameters for gradient descent
    learning_rate = 0.01
    max_iterations = 5000
    patience = 50
    patience_counter = 0
    prev_best_c5 = best_c5
    
    for i in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute cross-correlation to find maximum overlap index
        corr = correlate(h0, 1 - h0, mode="full")
        max_idx = np.argmax(corr)
        # The shift for max overlap
        shift = max_idx - (len(h0) - 1)
        
        # Compute gradient for the maximizer shift
        gradient = np.zeros_like(h0)
        for j in range(len(h0)):
            if 0 <= j + shift < len(h0):
                gradient[j] += (1 - 2 * h0[j + shift])
        
        # Update using gradient descent
        h0 = h0 - learning_rate * gradient
        
        # Project to feasible region
        h0 = project_to_feasible(h0, target_sum)
        
        # Evaluate
        current_c5 = evaluate(h0, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h0.copy()
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Adapt learning rate
        if patience_counter >= patience:
            learning_rate *= 0.5
            patience_counter = 0
            
    return (best_h.tolist(), float(best_c5), n_points)
