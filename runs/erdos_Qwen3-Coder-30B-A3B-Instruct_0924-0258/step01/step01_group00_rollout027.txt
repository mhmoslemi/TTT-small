import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        # Create a structured initial solution
        n0 = 30
        h_start = np.ones(n0) * 0.5
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Resolution levels: finer grids for better precision
    resolutions = [20, 30, 40, 50]
    time_per_res = (budget_s * 0.95) / len(resolutions)
    
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate starting point if necessary
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Define objective function for this resolution
        def objective(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        # Define bounds for optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use differential evolution for global search
        try:
            # Run with a fixed number of iterations for time control
            de_res = differential_evolution(
                objective,
                bounds,
                maxiter=int(300 * (time_per_res / 100)),  # scale iterations with time
                popsize=10,
                seed=seed+i,
                disp=False
            )
            
            if de_res.success:
                h_opt = project_to_feasible(de_res.x, n_points)
                c5 = evaluate_erdos_solution(h_opt, dx)
                if c5 < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5
                    best_n_points = n_points
        except:
            pass
        
        # Also try local refinement on best solution found so far
        try:
            # Local refinement with SLSQP
            h_refine = best_h.copy()
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                objective,
                h_refine,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-6}
            )
            
            if res.success:
                h_refined = project_to_feasible(res.x, n_points)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
                    best_n_points = n_points
        except:
            pass
        
        # Update the starting point for the next resolution
        h_start = project_to_feasible(best_h, n_points)
    
    return (list(best_h), best_c5, best_n_points)
