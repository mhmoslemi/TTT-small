import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Initialize
    h = h0.copy().astype(float)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize to correct sum
    current_sum = np.sum(h)
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Optimization parameters
    max_iter = 10000
    learning_rate = 0.01
    lr_decay = 0.5
    lr_patience = 50
    patience_counter = 0
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.99:
            break
            
        # Compute cross-correlation to find maximum overlap
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_shift = max_idx - len(h) + 1  # Convert to shift index
        
        # Compute gradient approximation
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # Gradient contribution from shift = max_shift
            j = i + max_shift
            if 0 <= j < len(h):
                grad[i] += (1 - 2 * h[j])
        
        # Update using gradient descent
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum constraint
        current_sum = np.sum(h_new)
        h_new = h_new * target_sum / current_sum
        
        h = h_new
        
        # Evaluate and update best solution
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= lr_patience:
            learning_rate *= lr_decay
            patience_counter = 0
    
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    return (best_h.tolist(), float(final_c5), n_points)
