import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def perturb_pattern(h, strength=0.1):
    """Perturb the solution with a patterned noise"""
    n_points = len(h)
    # Use a sinusoidal pattern for diversity
    pattern = np.sin(np.linspace(0, 4*np.pi, n_points)) * strength
    h_perturbed = h + pattern
    return project_to_feasible(h_perturbed, n_points)
