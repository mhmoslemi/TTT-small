import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Adjust carefully to meet sum constraint
        diff = target_sum - current_sum
        idx = np.argsort(h)
        # Increase values from the smallest
        for i in range(len(idx)-1, -1, -1):
            if diff == 0:
                break
            j = idx[i]
            if h[j] < 1.0 and diff > 0:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
        # Decrease values from the largest if necessary
        if diff < 0:
            for i in range(len(idx)):
                if diff == 0:
                    break
                j = idx[i]
                if h[j] > 0.0 and diff < 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    return h

    # Precompute max overlap for reference
    def compute_max_overlap(h):
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = compute_max_overlap(best_h)
    
    # Gradient descent based optimization
    max_iters = 3000
    learning_rate = 0.02
    patience = 200
    patience_counter = 0
    last_improvement = 0
    
    for i in range(max_iters):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute full correlation
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        max_correlation = corr[k_max]
        # We want to minimize: max_k (1 - h(x) * h(x+k))
        # Which is equivalent to minimizing: max_k (h(x) * h(x+k))  
        # So we want to maximize max_k (h(x) * h(x+k))
        # Or minimize -max_k (h(x) * h(x+k))        
        # But in terms of overlap integral: C5 = max_k int h(x)(1-h(x+k)) dx
        # This equals max_k (int h(x) dx - int h(x)h(x+k) dx) 
        # = max_k (1 - int h(x)h(x+k) dx) 
        # So we want to MAXIMIZE the sum h(x)h(x+k) for the worst k
        # So we minimize (1 - sum(h(x)h(x+k))) for the worst k
        # That is, we want to MAXIMIZE sum(h(x)h(x+k)) for the worst k
        
        # Therefore, gradient is w.r.t. h(x) the gradient of sum(h(x)h(x+k))
        # Which is h(x+k) 
        # But we only compute the gradient for the shift k that maximizes sum(h(x)h(x+k))
        grad = np.zeros_like(h)
        h_shifted = np.roll(h, -k_max) if k_max > 0 else h.copy()
        grad = h_shifted
        
        # Gradient descent step
        h_new = h - learning_rate * grad
        
        # Project to feasible region
        h_new = project_to_feasible(h_new, target_sum)
        
        h = h_new
        
        # Evaluate
        current_c5 = compute_max_overlap(h)

        if patience_counter > patience:
            # Small noise to escape local minima
            h = h + np.random.normal(0, 0.001, n_points)
            h = project_to_feasible(h, target_sum)
            patience_counter = 0
    
    return (best_h.tolist(), float(best_c5), n_points)
