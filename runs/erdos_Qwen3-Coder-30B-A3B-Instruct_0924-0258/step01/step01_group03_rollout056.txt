import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy constraints: 0≤h≤1 and sum(h)=target_sum."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5  # Simple uniform start
    
    # Initialize
    h = h0.copy()
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Project to feasible region
    h = project_to_feasible(h, target_sum)
    
    # Parameters
    learning_rate = 0.01
    min_lr = 1e-6
    patience = 50
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s:
        # --- Compute correlation ---
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        
        # Compute gradient for this shift
        gradient = np.zeros_like(h)
        for i in range(n_points):
            j = (i - k_max) % n_points
            gradient[i] = 1 - 2 * h[j]
        
        # --- Update step ---
        h = h - learning_rate * gradient
        
        # --- Project to feasible region ---
        h = project_to_feasible(h, target_sum)
        
        # --- Evaluate ---
        c5 = evaluate(h, dx)
        
        # --- Update best ---
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
        
        # --- Adaptive learning rate ---
        if patience_counter >= patience and learning_rate > min_lr:
            learning_rate *= 0.9
            patience_counter = 0
        
        # --- Random perturbation if stuck ---
        if patience_counter >= patience:
            noise = np.random.uniform(-1e-3, 1e-3, n_points)
            h = h + noise
            h = project_to_feasible(h, target_sum)
            patience_counter = 0
            
        iteration += 1
        
        # Safety check for NaNs
        if np.isnan(h).any():
            h = project_to_feasible(np.random.rand(n_points), target_sum)
    
    return (best_h.tolist(), float(best_c5), n_points)
