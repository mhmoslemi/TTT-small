import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with potential start point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:10]
    else:
        h = np.random.uniform(0, 1, 10)
        h = project_to_constraints(h, 0.2)
    
    best_h = h.copy()
    best_c5 = evaluate(h, 0.2)
    
    # Multi-resolution approach with hybrid PSO and gradient descent
    resolutions = [10, 20, 40, 70]
    time_per_res = budget_s / len(resolutions)
    
    for res_idx, n_points in enumerate(resolutions):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        if n_points > len(h):
            h_long = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, len(h)), h)
            h = h_long
        else:
            h = h[:n_points]
        
        h = project_to_constraints(h, dx)
        best_in_res = h.copy()
        best_res_c5 = evaluate(h, dx)
        
        # PSO parameters
        n_particles = 100
        max_iter = 2000
        inertia_w = 0.7
        c1, c2 = 1.5, 1.5
        
        # Initialize particles
        particles = [h.copy() for _ in range(n_particles)]
        velocities = [np.random.uniform(-0.05, 0.05, n_points) for _ in range(n_particles)]
        pbest = [p.copy() for p in particles]
        pbest_eval = [evaluate(p, dx) for p in particles]
        gbest_idx = np.argmin(pbest_eval)
        gbest = pbest[gbest_idx].copy()
        gbest_eval = pbest_eval[gbest_idx]
        
        # Gradient descent parameters
        lr = 0.01
        max_grad_iter = 500
        grad_tolerance = 1e-6
        
        iter_count = 0
        res_start = time.time()
        while iter_count < max_iter and time.time() - res_start < time_per_res * 0.95:
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(), np.random.rand()
                velocities[i] = (inertia_w * velocities[i] + 
                                c1 * r1 * (pbest[i] - particles[i]) +
                                c2 * r2 * (gbest - particles[i]))
                velocities[i] = np.clip(velocities[i], -0.05, 0.05)
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = project_to_constraints(particles[i], dx)
                
                # Evaluate and update personal best
                eval_val = evaluate(particles[i], dx)
                if eval_val < pbest_eval[i]:
                    pbest[i] = particles[i].copy()
                    pbest_eval[i] = eval_val
                    if eval_val < gbest_eval:
                        gbest = particles[i].copy()
                        gbest_eval = eval_val
                        
            iter_count += 1
            
            # Periodical gradient descent refinement
            if iter_count % 50 == 0:
                # Apply gradient descent for few steps
                h_refine = gbest.copy()
                for _ in range(max_grad_iter):
                    if time.time() - start_time > budget_s:
                        break
                    grad = np.zeros(n_points)
                    # Gradient of each h_i w.r.t. max_k integral:
                    # Each take from correlation
                    corr_back = correlate(h_refine, 1-h_refine, mode='full')
                    max_k = np.argmax(corr_back)
                    # For this optimization only approximate gradient
                    # This is a simplified version assuming key k is fixed;
                    # actual gradient would depend on correlation structure
                    # Use finite differences
                    eps = 1e-5
                    h_plus = h_refine.copy()
                    h_minus = h_refine.copy()
                    h_plus = np.clip(h_plus + eps, 0, 1)
                    h_plus = project_to_constraints(h_plus, dx)
                    h_minus = np.clip(h_minus - eps, 0, 1)
                    h_minus = project_to_constraints(h_minus, dx)

                    # These are crude estimates
                    c5_plus = evaluate(h_plus, dx)
                    c5_minus = evaluate(h_minus, dx)
                    for j in range(n_points):
                        grad[j] = (c5_plus - c5_minus) * 0.5 / eps

                    h_refine -= lr * grad
                    h_refine = np.clip(h_refine, 0, 1)
                    h_refine = project_to_constraints(h_refine, dx)

                    # Check for convergence
                    if np.allclose(h_refine, gbest, atol=grad_tolerance):
                        break
                        
                gbest = h_refine
                gbest_eval = evaluate(gbest, dx)
                
        h = gbest.copy()
        if gbest_eval < best_res_c5:
            best_in_res = gbest.copy()
            best_res_c5 = gbest_eval
        
        if best_res_c5 < best_c5:
            best_h = best_in_res.copy()
            best_c5 = best_res_c5
    
    return best_h.tolist(), best_c5, len(best_h)
