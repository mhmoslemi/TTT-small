import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region maintaining sum constraint."""
    h = np.array(h, dtype=float)
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute adjustment
    diff = target_sum - current_sum
    if diff > 0:  # Need to increase
        idx = np.argsort(h)
        for i in range(len(idx)-1, -1, -1):
            if diff <= 0:
                break
            j = idx[i]
            if h[j] < 1.0:
                increase = min(diff, 1.0 - h[j])
                h[j] += increase
                diff -= increase
    else:  # Need to decrease
        idx = np.argsort(h)
        for i in range(len(idx)):
            if diff >= 0:
                break
            j = idx[i]
            if h[j] > 0.0:
                decrease = min(-diff, h[j])
                h[j] -= decrease
                diff += decrease
    
    return h

def nelder_mead_step_simplex(func, points, target_sum, dx, alpha=1.0, beta=2.0, gamma=0.5, delta=0.5):
    """Perform one step of Nelder-Mead optimization."""
    # Sort points by function value (ascending)
    f_vals = [func(p, dx) for p in points]
    idxs = np.argsort(f_vals)
    points = [points[i] for i in idxs]
    f_vals = [f_vals[i] for i in idxs]
    
    # Simplex vertices
    x0 = points[0]
    x1 = points[1] if len(points) > 1 else x0.copy()
    xn = points[-1]
    
    # Compute centroid (excluding worst point)
    centroid = np.mean(points[:-1], axis=0)
    
    # Reflection
    x_r = centroid + alpha * (centroid - xn)
    x_r = project_to_feasible(x_r, target_sum)
    f_r = func(x_r, dx)
    
    if f_vals[0] <= f_r < f_vals[-2]:
        # Accept reflection
        points[-1] = x_r
    elif f_r < f_vals[0]:
        # Expansion
        x_e = centroid + beta * (x_r - centroid)
        x_e = project_to_feasible(x_e, target_sum)
        f_e = func(x_e, dx)
        if f_e < f_r:
            points[-1] = x_e
        else:
            points[-1] = x_r
    else:
        # Contraction
        if f_r < f_vals[-1]:
            x_c = centroid + gamma * (x_r - centroid)
            x_c = project_to_feasible(x_c, target_sum)
            f_c = func(x_c, dx)
            if f_c < f_r:
                points[-1] = x_c
            else:
                # Shrink
                for i in range(1, len(points)):
                    points[i] = x0 + delta * (points[i] - x0)
                    points[i] = project_to_feasible(points[i], target_sum)
        else:
            # Outside contraction
            x_c = centroid + gamma * (xn - centroid)
            x_c = project_to_feasible(x_c, target_sum)
            f_c = func(x_c, dx)
            if f_c < f_vals[-1]:
                points[-1] = x_c
            else:
                # Shrink
                for i in range(1, len(points)):
                    points[i] = x0 + delta * (points[i] - x0)
                    points[i] = project_to_feasible(points[i], target_sum)
    
    return points

        h = project_to_feasible(h, target_sum)

    best_h = None
    best_c5 = float('inf')
    iteration = 0

        # Evaluate the simplex
        f_vals = [evaluate(p, dx) for p in simplex]
        min_idx = np.argmin(f_vals)

            best_c5 = f_vals[min_idx]

        simplex = nelder_mead_step_simplex(evaluate, simplex, target_sum, dx)

        if iteration % mutation_freq == 0:
            for i in range(len(simplex)):

                    simplex[i] = project_to_feasible(h, target_sum)

            h0 = best_h.copy()
