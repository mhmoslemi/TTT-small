import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_gradient(h_vec, k_max):
        # Compute gradient for correlation at lag k_max
        n = len(h_vec)
        grad = np.zeros(n)
        # For lag k_max, h[i] contributes to (1-h[i+k_max]) in the correlation
        # So the contribution to gradient at position j is:
        # If j + k_max < n, then gradient[j] += -h[j + k_max]
        # If j - k_max >= 0, then gradient[j] += -h[j - k_max]
        if k_max >= 0:
            # shift is positive (h[i] correlated with h[i+k_max])
            # gradient[j] += -h[j+k_max] if j+k_max < n
            for i in range(n):
                if i + k_max < n:
                    grad[i] -= h_vec[i + k_max]
        else:
            # shift is negative (h[i] correlated with h[i-k_max])
            # gradient[j] += -h[j-k_max] if j-k_max >= 0
            for i in range(n):
                if i - k_max >= 0:
                    grad[i] -= h_vec[i - k_max]
        return grad
    
    def optimize_step(h_vec, alpha=0.1, max_iter=100):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        
        no_improvement_count = 0
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Compute all correlations
            corr = correlate(h_current, 1 - h_current, mode='full')
            # Find maximum lag
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (len(h_current) - 1)
            
            # Compute gradient for the maximum lag
            grad = compute_gradient(h_current, max_lag)
            
            # Try multiple step sizes
            for step_size in [alpha, alpha/2, alpha/4]:
                candidate_h = h_current - step_size * grad
                candidate_h = project_feasible(candidate_h)
                new_C5 = compute_C5(candidate_h)
                if new_C5 < best_C5:
                    h_current = candidate_h
                    best_C5 = new_C5
                    no_improvement_count = 0
                    break
            else:
                no_improvement_count += 1
            
            # Early termination if no improvement for 50 iterations
            if no_improvement_count >= 50:
                break
                
        return h_current, best_C5
    
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Meta-optimization loop
    alpha = 0.1
    for iteration in range(1000):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Small random perturbation
        h_perturbed = h_best + np.random.normal(0, 0.01, n_points)
        h_perturbed = project_feasible(h_perturbed)
        
        # Optimize from this perturbed point
        h_opt, C5_opt = optimize_step(h_perturbed, alpha, 50)
        if C5_opt < C5_best:
            C5_best = C5_opt
            h_best = h_opt.copy()
            alpha = min(0.1, alpha * 1.05)  # Increase alpha if successful
        else:
            alpha = max(0.001, alpha * 0.95)  # Decrease alpha if stagnant
            
    # Final projection for safety
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
