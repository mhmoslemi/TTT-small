import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create one
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def project_h(h_vec):
        """Project h to feasible region: [0,1] and sum = target_sum"""
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def compute_C5(h_vec):
        """Compute the maximum overlap integral"""
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    # Objective function for optimization
    def objective(h_vec):
        return compute_C5(h_vec)
    
    # Constraints for the optimization
    def sum_constraint(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Bounds for the variables [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    best_h = project_h(h)
    best_C5 = compute_C5(best_h)
    
    # Nelder-Mead optimization around the best solution
    def local_optimize(h_start):
        try:
            result = minimize(objective, h_start, method='Nelder-Mead', 
                             options={'maxiter': 1000, 'fatol': 1e-6})
            return result.x
        except:
            return h_start
    
    # Record best solution found
    best_so_far_h = best_h.copy()
    best_so_far_C5 = best_C5
    
    # Use differential evolution for global search
    de_result = differential_evolution(objective, bounds, maxiter=100, popsize=15,
                                       recombination=0.7, seed=seed,
                                       callback=lambda x, convergence: None)
    # If it found better, update
    if compute_C5(de_result.x) < best_C5:
        best_h = de_result.x
        best_C5 = compute_C5(best_h)
        
    # Refine with additional local search
    local_refine_h = project_h(best_h)
    for _ in range(1000):
        if time.time() - start_time > budget_s - 2:
            break
        local_refine_h = local_optimize(local_refine_h)
        local_refine_h = project_h(local_refine_h)
        current_C5 = compute_C5(local_refine_h)
        if current_C5 < best_so_far_C5:
            best_so_far_C5 = current_C5
            best_so_far_h = local_refine_h.copy()
    
    # Final refinement
    final_h = local_optimize(best_so_far_h)
    final_h = project_h(final_h)
    final_C5 = compute_C5(final_h)
    
    if final_C5 < best_so_far_C5:
        return (final_h.tolist(), final_C5, n_points)
    else:
        return (best_so_far_h.tolist(), best_so_far_C5, n_points)
