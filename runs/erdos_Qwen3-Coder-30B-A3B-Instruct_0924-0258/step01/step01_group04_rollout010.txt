import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Initialize population for GA
    population_size = 50
    population = []
    for _ in range(population_size):
        h_trial = np.random.uniform(0.4, 0.6, n_points)
        h_trial = project_h(h_trial)
        population.append(h_trial)
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # GA parameters
    elite_size = population_size // 5  # top 20%
    crossover_rate = 0.8
    mutation_rate = 0.1
    
    # Main GA loop
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate current population
        fitness = []
        for individual in population:
            fit = compute_C5(individual)
            fitness.append(fit)
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_C5:
            best_C5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early termination
        if no_improvement_count > max_no_improvement:
            break
            
        # Select elite
        elite = population[:elite_size]
        
        # Local search on top individuals
        if generation % 50 == 0:
            for i in range(min(5, len(population))):
                individual = population[i]
                # Nelder-Mead local search
                res = minimize(
                    lambda x: compute_C5(project_h(x)),
                    individual,
                    method='Nelder-Mead',
                    options={'maxiter': 100, 'adaptive': True}
                )
                if res.success:
                    new_individual = project_h(res.x)
                    new_fitness = compute_C5(new_individual)
                    if new_fitness < fitness[i]:
                        population[i] = new_individual
                        fitness[i] = new_fitness
                        
        # Create new population
        new_population = elite.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitness, 3)
            if np.random.rand() < crossover_rate:
                parent2 = tournament_selection(population, fitness, 3)
                child1, child2 = crossover(parent1, parent2)
            else:
                child1 = parent1.copy()
                child2 = parent1.copy()
                
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        population = new_population[:population_size]
        
    # Final check with local search on best
    res = minimize(
        lambda x: compute_C5(project_h(x)),
        best_h,
        method='Nelder-Mead',
        options={'maxiter': 100, 'adaptive': True}
    )
    if res.success:
        final_h = project_h(res.x)
        final_C5 = compute_C5(final_h)
        if final_C5 < best_C5:
            best_C5 = final_C5
            best_h = final_h.copy()
    
    return (best_h.tolist(), best_C5, n_points)

def tournament_selection(population, fitness, tournament_size):
    """Select individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_index].copy()

def crossover(parent1, parent2):
    """Uniform crossover."""
    child1 = parent1.copy()
    child2 = parent2.copy()
    for i in range(len(parent1)):
        if np.random.rand() < 0.5:
            child1[i], child2[i] = child2[i], child1[i]
    return child1, child2

def mutate(individual, mutation_rate):
    """Gaussian mutation with adaptive variance."""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.rand() < mutation_rate:
            # Mutate with Gaussian perturbations
            mutated[i] += np.random.normal(0, 0.05)
    return mutated
