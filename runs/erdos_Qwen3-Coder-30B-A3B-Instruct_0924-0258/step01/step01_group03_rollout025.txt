import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Linearly adjust values
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial values
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Initialize
    dx = 2.0 / n_points
    h = h0.copy()
    h = project_to_feasible(h, n_points / 2.0)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 100
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Track patience
    no_improvement = 0
    prev_c5 = float('inf')
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Evaluate current solution
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > patience:
            # Reset with a small perturbation
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = project_to_feasible(h, n_points / 2.0)
            no_improvement = 0
            continue
            
        # Find the shift causing max overlap
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        
        # Gradient: contribution to overlap at shift max_k
        grad = np.zeros_like(h)
        for idx in range(len(h)):
            if idx + max_k < len(h):
                grad[idx] += (1 - h[idx + max_k])
            if idx - max_k >= 0:
                grad[idx] += (1 - h[idx - max_k])
        
        # Update using gradient
        h = h - learning_rate * grad
        h = project_to_feasible(h, n_points / 2.0)
        
        # Occasionally reduce learning rate if stagnating
        if i % 100 == 0 and i > 0:
            if abs(current_c5 - prev_c5) < 1e-8:
                learning_rate *= 0.9
            prev_c5 = current_c5
            
    # Final validation
    final_c5 = evaluate(best_h, dx)
    return (best_h.tolist(), float(final_c5), n_points)
