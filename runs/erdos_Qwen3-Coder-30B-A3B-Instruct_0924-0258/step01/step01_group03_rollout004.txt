import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project h to meet sum constraint while staying in [0,1]."""
    h = np.array(h, dtype=float)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale and clip
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    # Fine adjustment
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        idxs = np.argsort(h)
        if diff > 0:
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1.0:
                    update = min(diff, 1.0 - h[j])
                    h[j] += update
                    diff -= update
        else:
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0.0:
                    update = max(diff, -h[j])
                    h[j] += update
                    diff -= update
    return h

def compute_gradient(h, k_max, n_points):
    """Compute gradient for the shift k that maximizes overlap."""
    # h is [0,1] vector of length n_points
    grad = np.zeros(n_points)
    # The gradient w.r.t to h[i] for the max overlap at shift k_max
    # is the contribution to the correlation from that h[i] (1 - h[i+k_max])
    for i in range(n_points):
        if i + k_max < n_points:
            grad[i] = 1 - h[i + k_max]
    return grad

    iteration = 0

        k_max = np.argmax(corr)

        # Compute gradient
