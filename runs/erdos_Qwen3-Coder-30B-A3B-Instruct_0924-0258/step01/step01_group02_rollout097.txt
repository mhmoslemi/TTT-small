import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    temp = 10 * best_c5
    cooling_rate = 0.95
    min_temp = 1e-6
    iterations_per_temp = 500
    
    iteration = 0
    while temp > min_temp:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        for _ in range(iterations_per_temp):
            # Create a perturbed version of h
            h_new = h.copy()
            
            # Choose a random index to perturb
            idx = np.random.randint(0, n_points)
            
            # Perturb the value
            delta = np.random.normal(0, 0.05)
            h_new[idx] += delta
            
            # Project back to feasible set
            h_new = project_h(h_new)
            
            # Compute new C5
            new_c5 = compute_c5(h_new)
            
            # Accept or reject the move
            if new_c5 < best_c5:
                h = h_new
                best_c5 = new_c5
                best_h = h_new.copy()
            else:
                # Accept with probability based on temperature
                if np.random.rand() < np.exp(-(new_c5 - best_c5) / temp):
                    h = h_new
            
            iteration += 1
            
        # Cool down the temperature
        temp *= cooling_rate
    
    return (best_h.tolist(), best_c5, n_points)
