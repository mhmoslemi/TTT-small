import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization - simplest 0,1 step function
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Convolution method - `full` mode includes all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Ensure 0 <= h[i] <= 1 and sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def step_descent(h_vec, max_iter=20):
        h_current = h_vec.copy()
        h_current = project_feasible(h_current)
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        no_improvement_count = 0
        step_size = 0.05
        
        for iter_ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Coordinate descent: try modifying each component
            improved = False
            for i in range(len(h_current)):
                if time.time() - start_time > budget_s - 0.1:
                    break
                    
                original = h_current[i]
                candidates = [original + step_size, original - step_size]
                candidates = [c for c in candidates if 0 <= c <= 1]
                
                best_candidate = original
                best_c5 = best_C5
                
                for c in candidates:
                    h_candidate = h_current.copy()
                    h_candidate[i] = c
                    h_candidate = project_feasible(h_candidate)
                    c5_candidate = compute_C5(h_candidate)
                    if c5_candidate < best_c5:
                        best_c5 = c5_candidate
                        best_candidate = c
                        
                if best_candidate != original:
                    h_current[i] = best_candidate
                    h_current = project_feasible(h_current)
                    improved = True
                    if best_c5 < best_C5:
                        best_C5 = best_c5
                        best_h = h_current.copy()
                        
            # Reset no improvement counter
            if improved:
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            # If stuck for too long, increase step size or exit
            if no_improvement_count > 10:
                step_size = min(step_size * 1.5, 0.5)
                no_improvement_count = 0
                
        return best_h, best_C5
    
    def swap_optimization(h_vec, num_swaps=10):
        h_current = h_vec.copy()
        h_current = project_feasible(h_current)
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        
        for _ in range(num_swaps):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Select two random indices
            i, j = np.random.choice(len(h_current), 2, replace=False)
            # Try swapping
            h_swap = h_current.copy()
            h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
            h_swap = project_feasible(h_swap)
            c5_swap = compute_C5(h_swap)
            if c5_swap < best_C5:
                best_C5 = c5_swap
                best_h = h_swap.copy()
                
        return best_h, best_C5
    
    def simplex_search(h_vec, num_vertices=3, max_iter=10):
        h_current = h_vec.copy()
        h_current = project_feasible(h_current)
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        
        # Create initial simplex
        vertices = [h_current]
        for _ in range(num_vertices - 1):
            perturbed = h_current + np.random.normal(0, 0.05, len(h_current))
            vertices.append(project_feasible(perturbed))
            
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Evaluate all vertices
            c5_vals = [compute_C5(v) for v in vertices]
            
            # Sort by C5
            indices = np.argsort(c5_vals)
            best_idx = indices[0]
            worst_idx = indices[-1]
            
            # Compute centroid excluding worst
            centroid = np.mean([vertices[i] for i in indices[:-1]], axis=0)
            centroid = project_feasible(centroid)
            
            # Reflect worst vertex
            reflected = 2 * centroid - vertices[worst_idx]
            reflected = project_feasible(reflected)
            c5_reflected = compute_C5(reflected)
            
            if c5_vals[best_idx] <= c5_reflected < c5_vals[worst_idx]:
                # Accept reflection
                vertices[worst_idx] = reflected
                if c5_reflected < best_C5:
                    best_C5 = c5_reflected
                    best_h = reflected.copy()
            elif c5_reflected < c5_vals[best_idx]:
                # Expand
                expanded = 2 * reflected - centroid
                expanded = project_feasible(expanded)
                c5_expanded = compute_C5(expanded)
                if c5_expanded < c5_reflected:
                    vertices[worst_idx] = expanded
                    if c5_expanded < best_C5:
                        best_C5 = c5_expanded
                        best_h = expanded.copy()
                else:
                    vertices[worst_idx] = reflected
                    if c5_reflected < best_C5:
                        best_C5 = c5_reflected
                        best_h = reflected.copy()
            
        return best_h, best_C5
    
    # Generate diverse initial candidates using Latin hypercube sampling
    initial_candidates = []
    num_initial = 5
    
    for _ in range(num_initial):
        # Generate uniform samples in [0,1]^n and scale smartly
        h_init = np.random.random(n_points)
        h_init = h_init / np.sum(h_init) * target_sum
        h_init = project_feasible(h_init)
        initial_candidates.append(h_init)
    
    # Evaluate and pick best
    best_C5 = float('inf')
    h_best = h.copy()
    
    for h_candidate in initial_candidates:
        c5_candidate = compute_C5(h_candidate)
        if c5_candidate < best_C5:
            best_C5 = c5_candidate
            h_best = h_candidate.copy()
    
    # Main optimization loop
    restart_counter = 0
    max_restarts = 10
    
    while time.time() - start_time < budget_s - 0.1:
        # Try several local optimization techniques
        
        # 1. Coordinate descent
        h_opt1, c5_opt1 = step_descent(h_best, max_iter=20)
        
        # 2. Swap optimization
        h_opt2, c5_opt2 = swap_optimization(h_best, num_swaps=10)
        
        # 3. Simplex search
        h_opt3, c5_opt3 = simplex_search(h_best, num_vertices=3, max_iter=10)
        
        # Choose best
        results = [(h_opt1, c5_opt1), (h_opt2, c5_opt2), (h_opt3, c5_opt3)]
        best_result = min(results, key=lambda x: x[1])
        
        # Update best
        if best_result[1] < best_C5:
            best_C5 = best_result[1]
            h_best = best_result[0].copy()
            restart_counter = 0
        else:
            restart_counter += 1
            
        # Occasionally restart with a new candidate
        if restart_counter > 30 or restart_counter > max_restarts:
            # Generate new candidate via simple perturbation
            h_perturbed = h_best + np.random.normal(0, 0.02, len(h_best))
            h_perturbed = project_feasible(h_perturbed)
            c5_perturbed = compute_C5(h_perturbed)
            if c5_perturbed < best_C5:
                best_C5 = c5_perturbed
                h_best = h_perturbed.copy()
            restart_counter = 0
            
    h_best = project_feasible(h_best)
    final_C5 = compute_C5(h_best)
    
    return list(h_best), final_C5, n_points
