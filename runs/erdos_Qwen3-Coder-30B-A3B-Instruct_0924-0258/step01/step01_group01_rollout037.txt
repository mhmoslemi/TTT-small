import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    # Ensure sum(h) * dx = 1 and h is in [0,1]
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    
    # Recalculate and adjust
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    
    return h

    dx = 2.0 / n_points
    best_h = h.copy()

        h_new = np.zeros_like(h)

            # We simplify to gradient descent
            grad = np.zeros_like(h)
            for j in range(n_points):
