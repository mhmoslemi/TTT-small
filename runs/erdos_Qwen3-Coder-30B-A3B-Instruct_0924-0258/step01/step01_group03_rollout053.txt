import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum = target_sum and values in [0,1]"""
    h = np.array(h, dtype=float)
    # First, clip to [0,1]
    h = np.clip(h, 0, 1)
    # Then normalize to target sum
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-10:
        # Avoid division by zero
        return np.ones_like(h) * (target_sum / len(h))
    h = h * target_sum / current_sum
    # Clamp again to [0,1]
    h = np.clip(h, 0, 1)
    return h
