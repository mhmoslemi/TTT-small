import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Configuration
    n_points = 45
    population_size = 50
    max_generations = 1000
    mutation_rate = 0.05
    elite_size = population_size // 4
    tournament_size = 3
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h * (n_points / 2.0) / np.sum(h)
        population.append(h.copy())
    
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2.0) / np.sum(h)
        population.append(h.copy())
    
    def compute_c5(h_vec):
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2.0) / np.sum(h_vec)
        return h_vec
    
    def tournament_selection(pop, fitnesses, k):
        selected_indices = np.random.choice(len(pop), size=k, replace=False)
        best_idx = selected_indices[np.argmax(fitnesses[selected_indices])]
        return pop[best_idx]
    
    def crossover(parent1, parent2):
        # Uniform crossover
        mask = np.random.rand(n_points) > 0.5
        child = np.where(mask, parent1, parent2)
        return child
    
    def mutate(h_vec):
        # Add Gaussian noise
        noise = np.random.normal(0, mutation_rate, n_points)
        h_mut = h_vec + noise
        return project_h(h_mut)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    
    generation = 0
    stagnation_counter = 0
    
    while time.time() - start_time < budget_s and stagnation_counter < 1000:
        # Evaluate fitness
        fitnesses = np.array([-compute_c5(h) for h in population])
        
        # Track best
        current_best_idx = np.argmax(fitnesses)
        current_best_c5 = -fitnesses[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)[::-1]
        population = [population[i] for i in sorted_indices]
        fitnesses = fitnesses[sorted_indices]
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
        generation += 1
    
    return (best_h.tolist(), best_c5, n_points)
