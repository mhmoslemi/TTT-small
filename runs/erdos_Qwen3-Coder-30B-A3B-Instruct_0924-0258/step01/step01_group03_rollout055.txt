import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy constraints: h in [0,1] and sum(h) = target_sum."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to target sum
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        # Avoid division by zero
        h = np.ones_like(h) * (target_sum / len(h))
    else:
        h = h * target_sum / current_sum
    # Clip again after scaling
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5  # Simple uniform start
    
    # Set dx
    dx = 2.0 / n_points
    
    # Initialize
    h = h0.copy().astype(float)
    h = project_to_feasible(h, n_points / 2.0)
    
    # Parameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 50
    patience_counter = 0
    best_c5 = float('inf')
    best_h = None
    
    # For gradient computation
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode="full")
        # Find the shift that maximizes overlap
        k_max = np.argmax(corr)
        # The overlap value for this shift
        max_overlap = corr[k_max]
        
        # Compute gradient with respect to h
        gradient = np.zeros_like(h)
        # This corresponds to shift k_max in the correlation
        # The gradient w.r.t. h[j] is (1 - 2 * h[j - k_max])
        for j in range(n_points):
            k = k_max - j
            # Wrap index
            idx = k % n_points
            gradient[j] = 1 - 2 * h[idx]
        
        # Update h
        h = h - learning_rate * gradient
        
        # Project to feasible set
        h = project_to_feasible(h, n_points / 2.0)
        
        # Compute C5 for this h
        c5 = evaluate(h, dx)
        
        # Update best if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Adaptive learning rate
        if patience_counter > 100:
            learning_rate *= 0.9
            patience_counter = 0
            
        # Add perturbation if stuck
        if patience_counter >= patience:
            noise = np.random.uniform(-1e-3, 1e-3, n_points)
            h = h + noise
            h = project_to_feasible(h, n_points / 2.0)
            patience_counter = 0
    
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        # Fallback
        final_h = project_to_feasible(h0, n_points / 2.0)
        final_c5 = evaluate(final_h, dx)
        return (final_h.tolist(), float(final_c5), n_points)
