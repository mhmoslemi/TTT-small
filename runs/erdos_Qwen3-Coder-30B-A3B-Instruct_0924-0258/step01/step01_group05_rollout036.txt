import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec

    def perturb_h(h_vec, priority_indices, sigma):
        h_new = h_vec.copy()
        # Perturb priority indices
        noise = np.random.normal(0, sigma, len(priority_indices))
        for i, idx in enumerate(priority_indices):
            h_new[idx] += noise[i]
        return h_new

    # Initial setup
    h_current = project_feasible(h)
    C5_current = compute_C5(h_current)
    h_best = h_current.copy()
    C5_best = C5_current
    
    # Parameters for simulated annealing
    T0 = 0.1
    tau = budget_s / np.log(10)
    T_min = 1e-5
    sigma = 0.1
    max_acceptance_rate = 0.4
    min_acceptance_rate = 0.1
    
    step_size = 0.05
    step_count = 0
    accept_count = 0
    total_steps = 0
    
    # Priority sampling: focus on intervals that contribute most to overlap
    def get_priority_indices():
        corr = correlate(h_current, 1 - h_current, mode='full')
        # Only consider the central peak to identify those that form overlap
        mid = len(corr) // 2
        window_size = min(10, len(corr) // 4)
        start_idx = max(0, mid - window_size)
        end_idx = min(len(corr), mid + window_size)
        peaks = np.argsort(corr[start_idx:end_idx])[-20:][::-1] + start_idx
        # Convert to actual indices in h vector
        # Simplified approach: top 20 percent by value in h
        sort_indices = np.argsort(h_current)[-max(1, n_points//5):][::-1]
        return sort_indices[:max(1, n_points//5)]

    # Main annealing loop
    while time.time() - start_time < budget_s - 0.1:
        total_steps += 1
        
        # Compute current temperature
        t = time.time() - start_time
        T = max(T_min, T0 * np.exp(-t / tau))
        
        # Get indices to focus on
        priority_indices = get_priority_indices()
        
        # Make perturbation
        h_new = perturb_h(h_current, priority_indices, sigma)
        
        # Project to feasible set
        h_new = project_feasible(h_new)
        
        # Compute new objective
        C5_new = compute_C5(h_new)
        
        # Accept or reject
        if C5_new < C5_current:
            accept = True
        else:
            # Metropolis acceptance criterion
            delta = C5_new - C5_current
            accept = np.random.rand() < np.exp(-delta / T)
        
        if accept:
            h_current = h_new
            C5_current = C5_new
            accept_count += 1
            
            # Update best solution
            if C5_current < C5_best:
                h_best = h_current.copy()
                C5_best = C5_current
        
        # Adjust step size based on acceptance rate
        if total_steps % 20 == 0 and total_steps > 0:
            acceptance_rate = accept_count / 20.0
            if acceptance_rate < min_acceptance_rate:
                sigma *= 1.2
            elif acceptance_rate > max_acceptance_rate:
                sigma /= 1.2
            accept_count = 0
        
        # Occasionally add randomness to prevent getting stuck
        if total_steps % 100 == 0:
            h_current = project_feasible(h_current + np.random.normal(0, 0.03, n_points))
            C5_current = compute_C5(h_current)

    # Final projection and evaluation
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
