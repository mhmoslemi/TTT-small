import numpy as np
from scipy import optimize
import time

def compute_overlap(h_vec):
    """Compute the maximum overlap integral C5 for given h."""
    h_vec = np.array(h_vec)
    dx = 2.0 / len(h_vec)
    conv = np.correlate(h_vec, 1 - h_vec, mode='full')
    mid = len(conv) // 2
    overlaps = conv[mid-len(h_vec)+1:mid+len(h_vec)-1] * dx
    return np.max(overlaps)

def project_feasible(h_vec):
    """Project h_vec to feasible region: [0,1] and sum = n_points/2."""
    h_vec = np.clip(h_vec, 0, 1)
    target_sum = len(h_vec) / 2.0
    current_sum = np.sum(h_vec)
    if current_sum > 0:
        h_vec = h_vec * target_sum / current_sum
    return h_vec

def upsample_h(h_vec, target_len):
    """Upsample h using linear interpolation."""
    if len(h_vec) >= target_len:
        return h_vec[:target_len]
    x_orig = np.linspace(0, 1, len(h_vec))
    x_new = np.linspace(0, 1, target_len)
    h_new = np.interp(x_new, x_orig, h_vec)
    return h_new
