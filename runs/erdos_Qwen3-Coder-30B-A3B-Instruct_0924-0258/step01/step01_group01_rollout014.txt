import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 10
    max_points = 512
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        
    h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # ADMM parameters
    rho = 1.0
    max_iter = 100
    alpha = 0.01
    
    iter_count = 0
    iterations_at_current_resolution = 0
    previous_best = best_c5
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break

        # ADMM iterations
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            # Update h by solving subproblem
            h_new = h.copy()
            h_old = h.copy()
            
            for _ in range(20):
                corr_z = correlate(h_new, 1-h_new, mode='full')
                grad = np.zeros_like(h_new)
                for i in range(len(h_new)):
                    for k in range(len(corr_z)):
                        if 0 <= i+k-len(h_new)+1 < len(h_new):
                            grad[i] += (1-h_new[i+k-len(h_new)+1]) * (corr_z[k] > np.max(corr_z)) * dx
                
                h_new = h_new - alpha * grad
                h_new = project_to_constraints(h_new, dx)
                if np.allclose(h_new, h_old, atol=1e-6):
                    break
                h_old = h_new.copy()
            
            # Update z = 1 - h
            z_new = 1 - h_new
            
            # Simple improvement step using gradient descent
            corr_h = correlate(h_new, 1-h_new, mode='full')
            c5_new = np.max(corr_h) * dx
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                
            h = h_new.copy()
            iter_count += 1
            iterations_at_current_resolution += 1
            
        # Check if we should increase resolution
        if n_points < max_points and iterations_at_current_resolution > 10:
            improvement = (previous_best - best_c5) / previous_best
            if improvement > 0.01:  # 1% improvement
                n_points *= 2
                dx = 2.0 / n_points
                h_long = np.interp(np.linspace(0, 2, n_points), 
                                 np.linspace(0, 2, n_points//2), h)
                h = h_long
                h = project_to_constraints(h, dx)
                previous_best = best_c5
                iterations_at_current_resolution = 0
        
        if n_points >= max_points:
            break
    
    return best_h.tolist(), best_c5, len(best_h)
