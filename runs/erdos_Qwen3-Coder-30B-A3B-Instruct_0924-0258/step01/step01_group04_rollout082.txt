import numpy as np
from scipy import signal
import time

def compute_C5(h_vec, dx):
    """Compute the maximum overlap integral for given step function h."""
    h_vec = np.array(h_vec)
    # Compute cross-correlation of h with (1-h)
    conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
    # Extract the relevant portion where k ranges from -1 to 1
    mid = len(conv) // 2
    overlaps = conv[mid - len(h_vec) + 1: mid + len(h_vec) - 1] * dx
    return np.max(overlaps)

def project_feasible(h_vec, target_sum):
    """Project h onto feasible set: [0,1] and sum = target_sum."""
    h_vec = np.clip(h_vec, 0, 1)
    current_sum = np.sum(h_vec)
    if current_sum > 0:
        h_vec = h_vec * target_sum / current_sum
    return h_vec

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from parent if available, else create simple initial guess
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # PSO parameters
    n_particles = 100
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    max_speed = 0.1
    inertia_weight = w_start
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        # Start near initial solution with small random perturbations
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = h + perturbation
        h_perturbed = project_feasible(h_perturbed, target_sum)
        particles.append(h_perturbed.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        pbest.append(h_perturbed.copy())
        pbest_vals.append(compute_C5(h_perturbed, dx))
    
    # Initialize global best
    gb_index = np.argmin(pbest_vals)
    gbest = pbest[gb_index].copy()
    gbest_val = pbest_vals[gb_index]
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight (linearly decreasing)
        inertia_weight = w_start - (w_start - w_end) * (iteration / max_iter)
        
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            vel_update = (
                inertia_weight * velocities[i] +
                c1 * r1 * (pbest[i] - particles[i]) +
                c2 * r2 * (gbest - particles[i])
            )
            
            # Clamp velocity
            vel_update = np.clip(vel_update, -max_speed, max_speed)
            velocities[i] = vel_update
            
            # Update position
            particles[i] += velocities[i]
            
            # Project into feasible region
            particles[i] = project_feasible(particles[i], target_sum)
            
            # Evaluate new position
            new_val = compute_C5(particles[i], dx)
            
            # Update personal best
            if new_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = new_val
                
                # Update global best if needed
                if new_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = new_val
        
        # Periodic check for improvement
        if iteration % 50 == 0:
            current_val = compute_C5(gbest, dx)
            if current_val < gbest_val:
                gbest_val = current_val
    
    final_C5 = compute_C5(gbest, dx)
    if final_C5 < gbest_val:
        gbest_val = final_C5
    
    return (gbest.tolist(), gbest_val, n_points)
