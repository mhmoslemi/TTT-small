import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Use differential evolution with constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define the objective function for scipy
    def objective(h_vec):
        h_vec = project_h(h_vec)
        return compute_C5(h_vec)
    
    # Initial best
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Run differential evolution
    try:
        result = differential_evolution(
            objective,
            bounds,
            maxiter=500,
            popsize=15,
            seed=seed,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.7,
            callback=lambda x, convergence: None,
            disp=False
        )
        
        if result.success:
            h_opt = project_h(result.x)
            c5_opt = compute_C5(h_opt)
            if c5_opt < best_C5:
                best_C5 = c5_opt
                best_h = h_opt
        
    except Exception:
        pass
    
    # Local refinement with Nelder-Mead
    def refine_h(h_initial):
        result = minimize(
            objective,
            h_initial,
            method='Nelder-Mead',
            options={'maxiter': 500, 'disp': False}
        )
        if result.success:
            return project_h(result.x)
        return h_initial
        
    refined_h = refine_h(best_h)
    refined_C5 = compute_C5(refined_h)
    
    if refined_C5 < best_C5:
        best_C5 = refined_C5
        best_h = refined_h
    
    # Continue with a hybrid approach: local search on promising solutions
    start_time = time.time()
    while time.time() - start_time < budget_s - 5:
        # Mutation-based local search
        h_mut = best_h.copy()
        idx = np.random.randint(0, n_points)
        h_mut[idx] += np.random.normal(0, 0.02)
        
        h_new = project_h(h_mut)
        c5_new = compute_C5(h_new)
        
        if c5_new < best_C5:
            best_C5 = c5_new
            best_h = h_new
    
    return (best_h.tolist(), best_C5, n_points)
