import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Initialize
    h_current = h.copy()
    h_current = project_feasible(h_current)
    C5_current = compute_C5(h_current)
    h_best = h_current.copy()
    C5_best = C5_current
    
    # Simulated annealing parameters
    T0 = 0.1
    tau = budget_s / np.log(10)
    T_min = 1e-5
    sigma = 0.1
    acceptance_count = 0
    accepted_count = 0
    
    # Cooling schedule helpers
    last_improvement = 0
    adaptive_sigma = sigma
    
    iteration = 0
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        # Compute current temperature
        elapsed = time.time() - start_time
        T = max(T_min, T0 * np.exp(-elapsed / tau))
        
        # Sample priority intervals based on current solution
        # Higher values = more likely to be selected for perturbation
        importance = np.abs(h_current - 0.5)  # Center toward 0.5 is more important
        # Don't sample too many at once
        priority_indices = np.argsort(importance)[-max(1, n_points//5):]
        
        # Create new candidate by perturbing priority intervals
        h_new = h_current.copy()
        
        # Add Gaussian noise to priority intervals
        for i in priority_indices:
            if np.random.random() < 0.7:  # 70% chance to perturb
                h_new[i] += np.random.normal(0, adaptive_sigma)
        
        # Also randomly flip some low priority intervals (5%)
        low_priority = np.setdiff1d(np.arange(n_points), priority_indices)
        for i in np.random.choice(low_priority, size=max(1, len(low_priority)//20), replace=False):
            h_new[i] = 1 - h_new[i]
        
        # Project onto feasible region
        h_new = project_feasible(h_new)
        
        # Compute new objective
        C5_new = compute_C5(h_new)
        
        # Accept or reject
        if C5_new < C5_current:
            h_current = h_new
            C5_current = C5_new
            if C5_new < C5_best:
                h_best = h_new.copy()
                C5_best = C5_new
            last_improvement = iteration
            accepted_count += 1
        else:
            # Accept with probability
            delta = C5_new - C5_current
            if np.random.random() < np.exp(-delta / T):
                h_current = h_new
                C5_current = C5_new
                accepted_count += 1
                last_improvement = iteration
        
        # Adaptively change perturbation strength based on acceptance rate
        acceptance_count += 1
        if acceptance_count % 20 == 0 and acceptance_count > 0:
            acceptance_rate = accepted_count / acceptance_count
            if acceptance_rate < 0.1:
                adaptive_sigma = min(adaptive_sigma * 1.2, 0.5)
            elif acceptance_rate > 0.4:
                adaptive_sigma = max(adaptive_sigma * 0.8, 0.01)
            acceptance_count = 0
            accepted_count = 0
        
        # Check if stuck for too long
        if iteration - last_improvement > 500 and T > T_min * 10:
            # restart with a random solution
            h_current = np.random.rand(n_points)
            h_current = project_feasible(h_current)
            C5_current = compute_C5(h_current)
            last_improvement = iteration

    # Final check
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
