import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction

    else:

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def solve_feasibility(h_guess, lambda_val):
        # Use convex optimization to check feasibility
        h = cp.Variable(n_points)
        constraints = [
            cp.matmul(cp.eye(n_points), h) == target_sum,  # Sum constraint
            h >= 0, h <= 1  # Box constraints
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            # Compute sum h[i] * (1 - h[i+k]) for valid indices
            # This is a bit tricky due to shifts, but we use the full correlation
            # Lambda should upper bound the max of these
            if k == 0:
                continue
            corr_sum = 0
            for i in range(n_points - k):
                corr_sum += h[i] * (1 - h[i + k])
            # Constraint: correlation <= lambda * dx
            constraints.append(corr_sum <= lambda_val * dx)
        
        # Make a slight relaxation to prevent infeasibility
        problem = cp.Problem(cp.Minimize(0), constraints)
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            return h.value is not None
        except:
            return False

    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    # Binary search for the minimal C5
    low = 0.38
    high = best_c5
    tolerance = 1e-4
    
    while high - low > tolerance:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        mid = (low + high) / 2.0
        # This is a simplified approach: we actually just use a modified
        # projected gradient approach on subproblems
        
        # Okay, let's go with the more direct numeric optimization of h directly
        # But let's update it with a gradient-free optimization
    
        # Let's try an approach closer to convex relaxation:
        # Fixing lambda and solving a convex feasibility subproblem
        # Use current h and update via gradient ascent of the negative of max correlation
        # but only when improvements have been made
        
        # For simplicity, use a gradient-based approach but with higher precision
        def objective(h_vec):
            # Convert to sparse matrix form of correlations to compute value faster
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            corr *= dx
            return -np.max(corr)
        
        def gradient(h_vec):
            # Approximate gradient
            eps = 1e-5
            grad = np.zeros_like(h_vec)
            base_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            base_val = np.max(base_corr) * dx
            k_max = np.argmax(base_corr)
            
            for i in range(len(h_vec)):
                h_test = h_vec.copy()
                h_test[i] += eps
                h_test = project_h(h_test)
                corr_test = np.correlate(h_test, 1 - h_test, mode='full')
                grad[i] = (corr_test[k_max] - base_val) / eps
                
            return grad
        
        # Simple gradient descent step
        lr = 0.01
        for _ in range(30):
            h = h - lr * gradient(h)
            h = project_h(h)
            
        # Reevaluate after optimization
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        # Binary search adjustment
        if new_c5 <= mid:
            high = mid
        else:
            low = mid
    
    # Final projection check
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    # Return best solution
    return (best_h.tolist(), final_c5, n_points)
