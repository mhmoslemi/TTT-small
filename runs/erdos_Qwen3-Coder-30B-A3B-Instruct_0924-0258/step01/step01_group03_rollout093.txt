import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Ensure the sum constraint
    h = h * (n_points / 2.0) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Parameters
    learning_rate = 0.01
    adaptation_interval = 100
    no_improvement_patience = 50
    smoothing_window = 5
    
    iteration = 0
    last_improvement = 0
    gradient_history = []
    
    while time.time() - start_time < budget_s:
        # Compute correlation
        corr = correlate(h, 1 - h, mode="full")
        # Find k that maximizes overlap
        k_max = np.argmax(np.abs(corr))
        # Carefully handle the exact zero case
        if corr[k_max] == 0:
            k_max = np.argmax(corr)  # prefer positive values if 0
        
        # Build gradient
        gradient = np.zeros(n_points)
        if k_max < n_points:
            # Positive shift
            for j in range(n_points):
                if j + k_max < n_points:
                    gradient[j] += (1 - h[j + k_max])
        else:
            # Negative shift equivalent (cross-correlation mirroring)
            shift = k_max - (2 * n_points - 2)
            for j in range(n_points):
                if j - shift >= 0:
                    gradient[j] += (1 - h[j - shift])
        
        # Apply gradient
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        
        # Normalize to sum constraint
        h_sum = np.sum(h)
        h = h * (n_points / 2.0) / h_sum
        
        # Check if we should adapt or perturb
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
        elif iteration - last_improvement > no_improvement_patience:
            # Add noise to escape local minima
            noise = np.random.uniform(-1e-3, 1e-3, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2.0) / np.sum(h)
            last_improvement = iteration
        
        # Adjust learning rate if needed
        if iteration % adaptation_interval == 0 and iteration > 0:
            learning_rate *= 0.9
        
        iteration += 1
        
        # Stop early if we're good
        if best_c5 < 0.3807:
            break
    
    return (best_h.tolist(), float(best_c5), n_points)
