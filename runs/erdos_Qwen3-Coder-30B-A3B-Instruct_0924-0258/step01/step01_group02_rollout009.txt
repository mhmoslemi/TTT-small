import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h_init = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h_init satisfies the constraint
    h_init = h_init * target_sum / np.sum(h_init)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def fitness(h_vec):
        return -compute_c5(h_vec)
    
    def mutate(h_vec, rate=0.05):
        # Add noise and re-project
        noise = np.random.normal(0, rate, len(h_vec))
        h_new = h_vec + noise
        return project_h(h_new)
    
    def crossover(h1, h2):
        # Uniform crossover
        mask = np.random.random(len(h1)) > 0.5
        h_new = np.where(mask, h1, h2)
        return project_h(h_new)
    
    def tournament_selection(population, fitnesses, k=3):
        indices = np.random.choice(len(population), size=k, replace=False)
        best_idx = indices[np.argmax([fitnesses[i] for i in indices])]
        return population[best_idx]
    
    # Initialize population
    population_size = 50
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = np.random.random(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = [fitness(h) for h in population]
    
    # Track best solution
    best_idx = np.argmax(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = -fitnesses[best_idx]
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)[::-1]
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Check for improvement
        current_best = -fitnesses[0]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            break
            
        # Select parents
        new_population = [population[0]]  # Elitism - keep best
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        # Evaluate new population
        population = new_population
        fitnesses = [fitness(h) for h in population]
        
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
