import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Initialize best solution
    h_current = h.copy()
    h_current = project_feasible(h_current)
    C5_current = compute_C5(h_current)
    h_best = h_current.copy()
    C5_best = C5_current
    
    # Parameters
    T0 = 0.1
    tau = budget_s / np.log(10)
    sigma = 0.1
    T_min = 1e-5
    max_iter = 1000
    accept_count = 0
    total_iter = 0
    
    # Main loop
    while time.time() - start_time < budget_s - 0.1 and total_iter < max_iter:
        total_iter += 1
        
        # Compute current C5 for reference
        C5_current = compute_C5(h_current)
        
        # Sample priority intervals
        # Get indices contributing most to correlation
        corr = correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Create perturbed version
        h_new = h_current.copy()
        
        # Perturb high priority intervals - use a larger portion
        n_priority = max(1, n_points // 5)  # Top 20%
        # Sample indices of high-h value intervals
        h_sorted_idx = np.argsort(h_current)[::-1]
        priority_indices = h_sorted_idx[:n_priority]
        
        for idx in priority_indices:
            h_new[idx] += np.random.normal(0, sigma)
        
        # Add some random flips to low priority areas
        low_priority_indices = h_sorted_idx[n_priority:]
        n_flips = max(1, len(low_priority_indices) // 20)  # 5% flips
        flip_indices = np.random.choice(low_priority_indices, size=n_flips, replace=False)
        for idx in flip_indices:
            h_new[idx] = 1 - h_new[idx]  # Flip binary values
        
        # Project to feasible region
        h_new = project_feasible(h_new)
        
        # Compute new C5
        C5_new = compute_C5(h_new)
        
        # Annealing acceptance
        T = max(T_min, T0 * np.exp(-total_iter / tau))
        delta = C5_new - C5_current
        
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_current = h_new.copy()
            if delta < 0:
                accept_count += 1
        else:
            accept_count = max(0, accept_count - 1)
        
        # Update best solution
        if C5_new < C5_best:
            h_best = h_current.copy()
            C5_best = C5_new
        
        # Adaptive perturbation size based on acceptance rate
        if total_iter % 10 == 0:
            accept_rate = accept_count / 10.0 if total_iter > 10 else 0.0
            if accept_rate < 0.1:
                sigma = min(0.5, sigma * 1.2)
            elif accept_rate > 0.4:
                sigma = max(0.01, sigma * 0.8)
            accept_count = 0
    
    # Final projection and evaluation
    h_final = project_feasible(h_best)
    final_C5 = compute_C5(h_final)
    
    return list(h_final), final_C5, n_points
