import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def gradient_descent_step(h, dx, learning_rate=0.01):
    # Analytical gradient for max correlation
    n = len(h)
    # Gradient w.r.t. each h[i] is the sum of contributions from all overlaps
    # More precisely, for correlation h * (1-h), 
    # dC5/dh[i] = (1 - 2*h[i]) * dx
    grad = (1 - 2 * h) * dx
    h_new = h - learning_rate * grad
    return project_to_constraints(h_new, dx)

    np.random.seed(seed)
    start_time = time.time()
    resolution_schedule = [10, 20, 40, 70]
    best_h = None
    best_c5 = float('inf')
    
    for n_points in resolution_schedule:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points

            h = h[:n_points]
        else:
            # Start with a better structured initialization
            h = np.random.uniform(0.1, 0.9, n_points)
            h = project_to_constraints(h, dx)
            
        # Particle swarm optimization (simplified)
        n_particles = 100
        particles = []
        velocities = []
        particle_best = []
        particle_best_c5 = []
        global_best = None
        global_best_c5 = float('inf')
        
        for _ in range(n_particles):
            if 'initial_h_values' in globals():
                h_ref = np.array(initial_h_values)
                h_new = h_ref[:n_points]
                h_new += np.random.normal(0, 0.01, n_points)
                h_new = project_to_constraints(h_new, dx)
            else:
                h_new = np.random.uniform(0.1, 0.9, n_points)
                h_new = project_to_constraints(h_new, dx)
            particles.append(h_new.copy())
            velocities.append(np.random.uniform(-0.05, 0.05, n_points))
            c5 = evaluate(h_new, dx)
            particle_best.append(h_new.copy())
            particle_best_c5.append(c5)
            if c5 < global_best_c5:
                global_best = h_new.copy()
                global_best_c5 = c5
                
        # PSO main loop
        max_iter = 500
        inertia = 0.7
        c1, c2 = 1.5, 1.5
        for it in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            for p in range(n_particles):
                r1, r2 = np.random.random(2), np.random.random(2)
                velocities[p] = (inertia * velocities[p] + 
                                c1 * r1 * (particle_best[p] - particles[p]) + 
                                c2 * r2 * (global_best - particles[p]))
                # Clip velocity
                velocities[p] = np.clip(velocities[p], -0.05, 0.05)
                particles[p] += velocities[p]
                particles[p] = project_to_constraints(particles[p], dx)
                c5 = evaluate(particles[p], dx)
                if c5 < particle_best_c5[p]:
                    particle_best[p] = particles[p].copy()
                    particle_best_c5[p] = c5

                        global_best = particles[p].copy()

        h = global_best.copy()
        h = project_to_constraints(h, dx)
        c5 = global_best_c5

                best_c5 = c5

            best_c5 = c5
    
    return best_h.tolist(), best_c5, len(best_h)
