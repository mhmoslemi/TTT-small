import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def smooth_max(x, beta=10.0):
    return np.log(np.sum(np.exp(beta * x))) / beta

    n = len(h)
    h_rev = h[::-1]
    corr_full = correlate(h, 1-h, mode='full')
    # Use full correlation to get all lags from -n+1 to n-1
    # We want max of this over all k in [-n+1, n-1]
    # Let's compute gradients w.r.t h
    # For correlation at lag k (k from -n+1 to n-1)
    # This is conv(h, 1-h)
    # Gradient of corr(h, 1-h) w.r.t. h_i:
    # = sum_{j=0}^{n-1} d(h_i) / d(h_j) * (1-h_{i+k-j}) if i+k-j in [0,n-1]
    # Which is complex to compute directly
    
    # Efficiently using FFT-based approach:
    # We'll compute gradients by finite differences
    return corr_full, None  # We'll use numeric approach

    # Ensure initial satisfies constraint
    h = project_to_constraints(h, 2.0 / n_points)
    
    dx = 2.0 / n_points
    last_c5 = evaluate(h, dx)
    best_c5 = min(best_c5, last_c5)
    best_h = h.copy()
    
    # Gradient-based optimization with penalty
    def obj_and_grad(h_vec):
        h = h_vec.copy()
        h = project_to_constraints(h, dx)
        corr_full = correlate(h, 1-h, mode='full')
        c5 = np.max(corr_full) * dx
        # Approximate gradient via finite difference perturbation
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_pert = h.copy()
            h_pert[i] += eps
            h_pert = project_to_constraints(h_pert, dx)
            c5_pert = np.max(correlate(h_pert, 1-h_pert, mode='full')) * dx
            grad[i] = (c5_pert - c5) / eps
        return c5, grad
    
    max_iter = 2000
    iter_count = 0
    learning_rate = 1e-3
    
    # Optimization using scipy minimize
    def fun(h_flat):
        h = project_to_constraints(h_flat, dx)
        return evaluate(h, dx)

        h = project_to_constraints(h_flat, dx)
        # Compute numeric gradient

            h_plus = project_to_constraints(h_plus, dx)
