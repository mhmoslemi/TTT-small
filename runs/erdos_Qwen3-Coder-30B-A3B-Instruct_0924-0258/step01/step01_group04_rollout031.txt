import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    n_particles = 100
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    max_speed = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    # Start particles near the initial solution
    for i in range(n_particles):
        # Slightly perturb the initial solution
        perturbation = np.random.normal(0, 0.05, n_points)
        h_init = project_h(h + perturbation)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
        pbest_positions.append(h_init.copy())
        pbest_values.append(compute_C5(h_init))
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx]
    gbest_value = pbest_values[gbest_idx]
    
    # PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            vel_cognitive = c1 * r1 * (pbest_positions[i] - particles[i])
            vel_social = c2 * r2 * (gbest_position - particles[i])
            velocities[i] = w * velocities[i] + vel_cognitive + vel_social
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_speed, max_speed)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project into feasible region
            particles[i] = project_h(particles[i])
            
            # Evaluate
            current_value = compute_C5(particles[i])
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest_values[i] = current_value
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if current_value < gbest_value:
                    gbest_value = current_value
                    gbest_position = particles[i].copy()
    
    # Final check
    final_C5 = compute_C5(gbest_position)
    if final_C5 < gbest_value:
        gbest_value = final_C5
        gbest_position = gbest_position.copy()
        
    return (gbest_position.tolist(), gbest_value, n_points)
