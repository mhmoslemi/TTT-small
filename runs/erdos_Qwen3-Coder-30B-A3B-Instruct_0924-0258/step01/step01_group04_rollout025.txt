import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Use Adam optimizer
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    learning_rate = 0.01
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        h_vec = project_h(h)
        
        # Find maximum overlap index
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        
        # Update using Adam with exact gradient for max k
        max_k_idx = np.argmax(overlaps)
        max_k = max_k_idx - (n_points - 1)
        
        # Compute exact gradient for this k
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            j = (i - max_k) % n_points
            grad[i] = (1 - h_vec[j])
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration))
        v_hat = v / (1 - beta2**(iteration))
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project
        h = project_h(h)
        
        # Periodic evaluation
        if iteration % 100 == 0:
            current_C5 = compute_C5(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)
