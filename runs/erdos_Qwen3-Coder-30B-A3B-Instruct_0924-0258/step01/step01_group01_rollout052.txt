import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def gradient_step(h, dx, step_size=0.01):
    # Analytical gradient: dC5/dhi = (1 - 2*hi) * dx
    n = len(h)
    grad = (1 - 2 * h) * dx
    h_new = h - step_size * grad
    return project_to_constraints(h_new, dx)

def particle_swarm_optimize(h, dx, n_particles=100, max_iter=1000, 
                           inertia=0.7, c1=2.0, c2=2.0, vmax=0.05):
    n = len(h)
    # Initialize particles
    particles = [h + np.random.uniform(-0.1, 0.1, n) for _ in range(n_particles)]
    velocities = [np.random.uniform(-vmax, vmax, n) for _ in range(n_particles)]
    personal_best = [p.copy() for p in particles]
    personal_best_vals = [evaluate(p, dx) for p in particles]
    
    # Global best
    best_idx = np.argmin(personal_best_vals)
    global_best = personal_best[best_idx].copy()
    global_best_val = personal_best_vals[best_idx]
    
    for _ in range(max_iter):
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -vmax, vmax)
            # Update position
            particles[i] += velocities[i]
            # Project to constraints and bounds
            particles[i] = project_to_constraints(particles[i], dx)
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Evaluate
            current_val = evaluate(particles[i], dx)
            if current_val < personal_best_vals[i]:
                personal_best[i] = particles[i].copy()
                personal_best_vals[i] = current_val
                
            if current_val < global_best_val:
                global_best = particles[i].copy()
                global_best_val = current_val
                
    return global_best, global_best_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution schedule
    resolutions = [10, 20, 40, 70]
    best_h = None
    best_c5 = float('inf')
    
    # Dict to track current resolution
    current_res = 0
    
    for n_points in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Initialize
        if 'initial_h_values' in globals() and current_res == 0:
            h = np.array(initial_h_values)
            h = h[:n_points]
        else:
            h = np.random.uniform(0, 1, n_points)
            h = project_to_constraints(h, dx)
            
        # Particle swarm optimization
        h_opt, c5_opt = particle_swarm_optimize(h, dx)
        h = h_opt
        c5 = c5_opt
        
        # Gradient descent refinement
        prev_c5 = c5 + 1
        iter_count = 0
        while abs(c5 - prev_c5) > 1e-6 and iter_count < 1000 and time.time() - start_time < budget_s * 0.95:
            prev_c5 = c5
            h = gradient_step(h, dx, step_size=0.005)
            c5 = evaluate(h, dx)
            iter_count += 1
            
        # Update best solution
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        current_res += 1
        
    # Final validation
    if best_h is not None:
        final_c5 = evaluate(best_h, 2.0/len(best_h))
        return best_h.tolist(), final_c5, len(best_h)
    else:
        # Return default if failed
        return [0.5] * 70, 0.5, 70
