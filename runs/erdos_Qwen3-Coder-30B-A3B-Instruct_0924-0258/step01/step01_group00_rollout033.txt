import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), 
                         np.linspace(0, 2, len(h)), h)
    else:
        # Create a structured pattern
        h = np.ones(n_points) * 0.5
        # Add some structure to help optimization
        mid = n_points // 2
        h[mid-2:mid+2] = 0.8
    return h

def perturb_h(h, strength=0.1):
    h_pert = h + np.random.normal(0, strength, len(h))
    return project_to_feasible(h_pert, len(h))

def augmented_lagrangian_objective(h, constraints, lambda_vals, mu, n_points, dx):
    # Objective: minimize -min_corr (minimize negative of minimum correlation)
    corr = np.correlate(h, 1 - h, mode='full')
    min_corr = np.min(corr)
    obj = -min_corr
    
    # Augmented Lagrangian terms
    for i, (constraint_func, value) in enumerate(constraints):
        violation = constraint_func(h) - value
        obj += lambda_vals[i] * violation + 0.5 * mu * violation**2
    
    return obj

def optimize_at_resolution(h_start, n_points, dx, time_limit):
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    # Formulate constraints: A(k) >= alpha for all k
    # We'll use a discrete set of shifts for manageable computation
    max_shifts = n_points
    
    def compute_correlations(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr
    
    # Loop to refine the solution
    max_iter = 500
    lambda_vals = np.zeros(1)  # Only one constraint (min overlap)
    mu = 1.0
    tol = 1e-6
    patience = 0
    last_best = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > time_limit:
            break
            
        # Optimization step
        def obj(h):
            corr = compute_correlations(h)
            min_corr = np.min(corr)
            # Use augmented Lagrangian style
            return -min_corr
        
        # Simple gradient approximation - use finite differences
        def grad(h):
            eps = 1e-6
            grad_vals = np.zeros_like(h)
            base_obj = obj(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                obj_plus = obj(h_plus)
                grad_vals[i] = (obj_plus - base_obj) / eps
            return grad_vals
        
        # Constraints 
        def constraint_func(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.min(corr)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try multiple random restarts to avoid local optima
        best_iter_h = h_start.copy()
        best_iter_c5 = evaluate_erdos_solution(h_start, dx)
        
        for _ in range(3):
            h_restart = perturb_h(h_start, 0.05)
            try:
                res = minimize(obj, h_restart, method='L-BFGS-B', 
                               bounds=bounds, constraints=cons, 
                               jac=grad, options={'maxiter': 20, 'ftol': 1e-6})
                if res.success:
                    h_res = project_to_feasible(res.x, n_points)
                    c5 = evaluate_erdos_solution(h_res, dx)
                    if c5 < best_iter_c5:
                        best_iter_h = h_res.copy()
                        best_iter_c5 = c5
            except:
                pass
    
        # Update with the best result from this iteration
        if best_iter_c5 < best_c5:
            best_h = best_iter_h.copy()
            best_c5 = best_iter_c5
            patience = 0
        else:
            patience += 1
        
        # Early stopping if no progress after several iterations
        if patience > 10:
            break
            
        # Update starting point for next iteration
        h_start = best_iter_h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Test several resolutions
    resolutions = [30, 40, 50, 60]
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation for each resolution
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        h_opt, c5_opt = optimize_at_resolution(h_start, n_points, dx, time_per_res)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n_points = n_points
            
    return (list(best_h), best_c5, best_n_points)
