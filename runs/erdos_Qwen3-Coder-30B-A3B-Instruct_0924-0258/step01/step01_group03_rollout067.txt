import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5  # Simple uniform start
    
    # Set dx
    dx = 2.0 / n_points
    
    # Initialize parameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 50
    patience_counter = 0
    best_h = None
    best_c5 = float('inf')
    
    # Gradient-based optimization
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Evaluate current overlap
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute cross-correlation to find the shift with max overlap
        # Return indices and values
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(np.abs(corr))
        k_max = max_idx - (len(h) - 1)
        
        # Compute gradient of C5 with respect to h
        gradient = np.zeros_like(h)
        for j in range(len(h)):
            if k_max >= 0:
                neighbor_idx = j + k_max
                if neighbor_idx < len(h):
                    gradient[j] += (1 - h[neighbor_idx])
            else:
                neighbor_idx = j + k_max
                if neighbor_idx >= 0:
                    gradient[j] += (1 - h[neighbor_idx])
        
        # Apply gradient descent step
        h = h - learning_rate * gradient
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        
        # Normalize to maintain sum constraint
        current_sum = np.sum(h)
        if abs(current_sum) > 1e-10:
            h = h * (n_points / 2) / current_sum
        
        # Adaptive learning rate
        patience_counter += 1
        if patience_counter >= patience:
            learning_rate *= 0.9
            patience_counter = 0
        
        # Add small random perturbation if no improvement for a while
        if patience_counter == 1:
            perturbation = np.random.uniform(-1e-3, 1e-3, len(h))
            h += perturbation
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            if abs(current_sum) > 1e-10:
                h = h * (n_points / 2) / current_sum
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        final_h = h.copy()
        final_c5 = evaluate(final_h, dx)
        return (final_h.tolist(), float(final_c5), n_points)
