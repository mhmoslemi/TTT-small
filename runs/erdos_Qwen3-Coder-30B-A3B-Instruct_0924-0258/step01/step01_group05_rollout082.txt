import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Genetic Algorithm Parameters
    pop_size = 50
    tournament_size = 10
    elite_size = pop_size // 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_init = np.random.rand(n_points)
        h_init = project_feasible(h_init)
        population.append(h_init)
    
    best_C5 = float('inf')
    best_h = None
    no_improvement_count = 0
    
    def evaluate_population(pop):
        fitness = []
        for h in pop:
            c5 = compute_C5(h)
            fitness.append(c5)
        return np.array(fitness)
    
    def tournament_selection(fitness, tournament_size):
        indices = np.random.choice(len(fitness), tournament_size, replace=False)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        return best_idx
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        # Uniform crossover
        mask = np.random.rand(n_points) > 0.5
        child1 = np.where(mask, parent1, parent2)
        child2 = np.where(mask, parent2, parent1)
        
        # Apply feasibility projection
        child1 = project_feasible(child1)
        child2 = project_feasible(child2)
        
        return child1, child2
    
    def mutate(h_vec):
        # Gaussian mutation
        mask = np.random.rand(n_points) < mutation_rate
        noise = np.random.normal(0, 0.05, n_points)
        h_mut = h_vec + noise * mask
        h_mut = project_feasible(h_mut)
        return h_mut
    
    # Main GA loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate fitness
        fitness = evaluate_population(population)
        
        # Track best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_C5:
            best_C5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Check for early stopping
        if no_improvement_count >= patience:
            # Try to restart with a better solution
            # Find a diverse set of good individuals
            sorted_indices = np.argsort(fitness)
            top_indices = sorted_indices[:elite_size]
            # Sample from top individuals to diversify
            if len(top_indices) > 1:
                i1, i2 = np.random.choice(top_indices, 2, replace=False)
                h1 = population[i1]
                h2 = population[i2]
                child1, child2 = crossover(h1, h2)
                # Replace the worst with children
                worst_idx = np.argmax(fitness)
                population[worst_idx] = child1
                if np.random.rand() > 0.5:
                    population[worst_idx] = child2
        else:
            # Selection, crossover, mutation
            new_population = []
            
            # Elitism
            sorted_indices = np.argsort(fitness)
            for i in range(elite_size):
                new_population.append(population[sorted_indices[i]].copy())
            
            # Create rest of population
            while len(new_population) < pop_size:
                parent1_idx = tournament_selection(fitness, tournament_size)
                parent2_idx = tournament_selection(fitness, tournament_size)
                
                child1, child2 = crossover(population[parent1_idx], 
                                          population[parent2_idx])
                
                child1 = mutate(child1)
                child2 = mutate(child2)
                
                new_population.extend([child1, child2])
            
            # Trim to exact population size
            population = new_population[:pop_size]
    
    # Final best solution check
    if best_h is None:
        # If we didn't find a good solution, return the best individual from last pop
        fitness = evaluate_population(population)
        min_fitness_idx = np.argmin(fitness)
        best_h = population[min_fitness_idx]
        best_C5 = fitness[min_fitness_idx]
    
    return list(best_h), best_C5, n_points
