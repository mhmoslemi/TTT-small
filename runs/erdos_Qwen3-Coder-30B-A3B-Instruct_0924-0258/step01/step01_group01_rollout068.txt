import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def update_step(h, dx, learning_rate=1e-3, max_iter=50):
    # Gradient-based refinement to reduce maximum cross-correlation
    n = len(h)
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    for _ in range(max_iter):
        # Compute current correlation
        corr = correlate(h, 1-h, mode='full')
        # Find the maximum correlation (this is the C5 bound)
        max_idx = np.argmax(corr)
        # Find which lag this corresponds to
        k = max_idx - (n - 1)  # lag from -n+1 to n-1
        # Find indices in the original arrays that contribute to this correlation
        # We'll modify the function values in a way that directly impacts this
        # Specifically we could use the gradient of the correlation w.r.t. h values,
        # But for this simple derivative-based approach, we'll try to adjust values
        # that contribute to this specific lag k
        # Adjust left part: shift k positions forward
        # We need to be a bit more subtle here, as C5 is over all k, but we focus
        # only on reducing the current max
        
        # Simple gradient step: decrease h where it contributes most to the current max
        # We approximate the gradient direction by looking at the correlation's behavior
        # in the time domain and adjust in a simplified way
        # Note: per iteration, this is approximating how much to change each h[i]
        
        # Make a copy for computation
        h_new = h.copy()
        # For a 1D signal h, the derivative of the cross-correlation term h(x) * (1-h(x+k))
        # w.r.t. h[i] is 1 - h[i+k] if i+k is valid, else 0. We modify h[i] to decrease this,
        # but importantly: we focus on directions that reduce the maximum C5, not all terms
        
        # Simple strategy: increase h[i] in positions contributing most to max overlap
        # This is not precise but effective in terms of decreasing overlap
        
        # Adjusting h from current direction
        for i in range(n):
            if i == 0:
                # No left neighbor to pull from, will adjust with right neighbor
                # Just increase toward constraints if possible
                if h_new[i] < 1:
                    h_new[i] += learning_rate * 0.1
            elif i == n - 1:
                # No right neighbor
                if h_new[i] > 0:
                    h_new[i] -= learning_rate * 0.1
            else:
                # Adjust towards positive gradient
                if k > 0:
                    # For positive lag, increase h[i] if h[i-k] is present and high
                    # Simplified approach: ratchet h[i] up in this direction
                    if h_new[i] < 1:
                        h_new[i] += learning_rate * 0.05
                    if h_new[i-k] > 0 and i-k >= 0:
                        h_new[i-k] -= learning_rate * 0.05
                else:
                    # Negative lag
                    if h_new[i] < 1:
                        h_new[i] += learning_rate * 0.05
                    if i-k < n:
                        h_new[i-k] -= learning_rate * 0.05

        # Ensure all constraints are preserved
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_constraints(h_new, dx)
        
        c5_new = evaluate(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            h = h_new.copy()

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with relatively low resolution to be fast and explore diverse solutions
    n_low = 10
    n_high = 100
    n_points = n_low
    dx = 2.0 / n_points
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    iter_count = 0
    same_best_count = 0
    max_same = 20
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Gradient descent refinement on current resolution
        h, c5_new = update_step(h, dx, learning_rate=1e-2, max_iter=100)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            same_best_count = 0
        else:
            same_best_count += 1
            
        # Periodically reevaluate with more fidelity
        if iter_count % 10 == 0:
            h = project_to_constraints(h, dx)
            c5_current = evaluate(h, dx)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()
                
        # Double resolution if improvements are seen or time allows
        if same_best_count >= max_same and n_points < n_high:
            # Interpolate to higher resolution
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points *= 2
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            same_best_count = 0
            
        iter_count += 1
        
        if time.time() - start_time > budget_s:
            break
            
    return best_h.tolist(), best_c5, len(best_h)
