import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Initialize
    h_current = h.copy()
    h_current = project_feasible(h_current)
    C5_current = compute_C5(h_current)
    h_best = h_current.copy()
    C5_best = C5_current
    
    # Simulated Annealing parameters
    T0 = 0.1
    tau = budget_s / np.log(10)
    T_min = 1e-5
    sigma = 0.1
    accept_count = 0
    accept_target = 20  # Target acceptance rate percentage
    step_size = 0.5
    
    iteration = 0
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        T = max(T_min, T0 * np.exp(-iteration / tau))
        
        # Identify high priority intervals (top 20% by current values)
        sorted_indices = np.argsort(h_current)[::-1]
        num_priority = max(1, n_points // 5)
        high_priority = sorted_indices[:num_priority]
        
        # Create proposal
        h_proposal = h_current.copy()
        
        # Perturb high priority intervals with Gaussian noise
        noise = np.random.normal(0, sigma, n_points)
        h_proposal += noise
        h_proposal[high_priority] += np.random.normal(0, sigma * 2, num_priority)
        
        # Occasionally flip some low-priority intervals
        low_priority = sorted_indices[num_priority:]
        if len(low_priority) > 0 and np.random.random() < 0.1:
            flip_indices = np.random.choice(low_priority, size=max(1, len(low_priority) // 20))
            h_proposal[flip_indices] = 1 - h_proposal[flip_indices]
        
        # Project to feasible set
        h_proposal = project_feasible(h_proposal)
        C5_proposal = compute_C5(h_proposal)
        
        # Acceptance criterion
        delta = C5_proposal - C5_current
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h_current = h_proposal
            C5_current = C5_proposal
            accept_count += 1
            if C5_current < C5_best:
                h_best = h_current.copy()
                C5_best = C5_current
        else:
            accept_count = max(0, accept_count - 1)
        
        # Adapt step size based on acceptance rate
        if iteration % 100 == 0 and iteration > 0:
            acceptance_rate = accept_count / 100.0
            if acceptance_rate < 0.1:
                sigma *= 1.2
            elif acceptance_rate > 0.4:
                sigma *= 0.8
            accept_count = 0
            
        # Occasionally random restart
        if iteration % 1000 == 0 and iteration > 0:
            h_current = np.random.rand(n_points)
            h_current = project_feasible(h_current)
            C5_current = compute_C5(h_current)
            
    # Final feasible check and evaluation
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
