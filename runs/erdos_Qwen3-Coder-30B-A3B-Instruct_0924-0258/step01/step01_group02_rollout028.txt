import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def compute_analytical_gradient(h_vec, k_max):
        # Gradient w.r.t. h for the k_max correlation term
        # For correlation at shift k_max: (h * (1-h))[k_max] = sum h[i] * (1-h[i+k_max])
        # Gradient d/dh[i] = (1 - 2*h[i]) if (i+k_max) is within bounds, else 0
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            if i + k_max < len(h_vec):
                grad[i] = 1.0 - 2.0 * h_vec[i]
        return grad
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    max_iter = 5000
    stagnation_count = 0
    stagnation_threshold = 1000
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute current overlap
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        c5_current = corr[k_max] * dx
        
        # Compute analytical gradient
        grad = compute_analytical_gradient(h, k_max)
        
        # Gradient update
        h = h - lr * grad
        h = project_h(h)
        
        # Compute new C5 and update best
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Anneal learning rate
        lr *= 0.999
        
        # Occasionally add noise to escape local minima
        if iteration % 500 == 0 and iteration > 0:
            noise_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            noise_magnitude = 0.01
            h[noise_indices] += np.random.uniform(-noise_magnitude, noise_magnitude, len(noise_indices))
            h = project_h(h)
            
        # Early termination if no improvement for too long
        if stagnation_count > stagnation_threshold:
            break
            
        iteration += 1
        
        # Occasionally print progress
        if iteration % 1000 == 0:
            print(f"Iteration {iteration}, C5: {new_c5:.6f}, LR: {lr:.6f}")
    
    return (best_h.tolist(), best_c5, n_points)
