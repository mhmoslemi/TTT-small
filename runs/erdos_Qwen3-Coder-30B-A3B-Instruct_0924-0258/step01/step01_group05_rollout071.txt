import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization with n=45 points
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_gradient(h_vec, k_max):
        # Compute gradient for the maximum overlap case
        n = len(h_vec)
        grad = np.zeros(n)
        if k_max >= 0:  # Positive lag
            # Gradient contribution from h[i] affecting (1-h[i+k_max])
            grad[k_max:] = -h_vec[:-k_max]
            # Gradient contribution from (1-h[i]) affecting h[i+k_max]
            grad[:k_max] = -(1 - h_vec[k_max:])
        else:  # Negative lag
            # Gradient contribution from h[i] affecting (1-h[i+k_max])
            grad[:k_max] = -h_vec[-k_max:]
            # Gradient contribution from (1-h[i]) affecting h[i+k_max]
            grad[k_max:] = -(1 - h_vec[:k_max])
        return grad
    
    # Initialize
    h_current = project_feasible(h)
    C5_current = compute_C5(h_current)
    best_h = h_current.copy()
    best_C5 = C5_current
    
    alpha = 0.1
    stagnation_count = 0
    max_stagnation = 50
    
    iteration = 0
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        # Compute all overlap values for current h
        corr = correlate(h_current, 1 - h_current, mode='full')
        # Find maximum lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Compute gradient for this maximum overlap
        grad = compute_gradient(h_current, max_lag)
        
        # Try different step sizes
        alpha_candidates = [0.1, 0.05, 0.01]
        best_alpha = 0.0
        best_new_h = h_current.copy()
        best_new_C5 = C5_current
        
        # Try different alpha values
        for alpha_test in alpha_candidates:
            candidate_h = h_current - alpha_test * grad
            candidate_h = project_feasible(candidate_h)
            new_C5 = compute_C5(candidate_h)
            if new_C5 < best_new_C5:
                best_new_C5 = new_C5
                best_new_h = candidate_h.copy()
                best_alpha = alpha_test
                
        if best_new_C5 < C5_current:
            h_current = best_new_h.copy()
            C5_current = best_new_C5
            if C5_current < best_C5:
                best_C5 = C5_current
                best_h = h_current.copy()
            stagnation_count = 0
            alpha = 0.1  # Reset step size on improvement
        else:
            stagnation_count += 1
            if stagnation_count >= max_stagnation:
                # Add small random perturbation
                noise = np.random.normal(0, 0.01, n_points)
                h_current += noise
                h_current = project_feasible(h_current)
                stagnation_count = 0
                # Decrease alpha for more careful search
                alpha *= 0.5
                
        # Occasionally try random restarts from feasible points
        if iteration % 100 == 0 and time.time() - start_time < budget_s - 1.0:
            # Initialize with new random feasible point
            h_new = np.random.rand(n_points)
            h_new = project_feasible(h_new)
            C5_new = compute_C5(h_new)
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h_new.copy()
                h_current = h_new.copy()
                C5_current = C5_new

    return list(best_h), best_C5, n_points
