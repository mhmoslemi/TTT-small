import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Initialize
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent parameters
    learning_rate = 0.01
    patience = 0
    max_patience = 50
    min_improvement = 1e-6
    gradient_smoothing = 5
    smooth_grad = None
    
    iter_count = 0
    while time.time() - start_time < budget_s:
        # Compute cross-correlations
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(np.abs(corr))
        
        # Compute gradient w.r.t. h
        grad = np.zeros_like(h)
        if k_max < len(corr):
            # This is a bit tricky: we need to map the full correlation index to shift
            shift = k_max - (len(h) - 1)
            
            # Apply gradient contribution from this shift
            for i in range(len(h)):
                j = i + shift
                if 0 <= j < len(h):
                    grad[i] += (1 - h[j])
        
        # Smooth gradient if needed
        if smooth_grad is None:
            smooth_grad = grad.copy()
        else:
            smooth_grad = (smooth_grad * (gradient_smoothing - 1) + grad) / gradient_smoothing
        
        # Update h
        h = h - learning_rate * smooth_grad
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum) > 1e-10:
            h = h * (n_points / 2.0) / current_sum
        
        # Evaluate and track best solution
        c5 = evaluate(h, dx)
        if c5 < best_c5 - min_improvement:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
            learning_rate = max(1e-6, learning_rate * 0.99)
        else:
            patience += 1
        
        # Adaptive perturbation
        if patience > max_patience:
            h += np.random.uniform(-1e-3, 1e-3, len(h))
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            if abs(current_sum) > 1e-10:
                h = h * (n_points / 2.0) / current_sum
            patience = 0
            learning_rate = min(0.1, learning_rate * 1.1)
        
        iter_count += 1
        
        # Check if we're stuck
        if iter_count > 10000 and best_c5 < 0.381:
            break
    
    return (best_h.tolist(), float(best_c5), n_points)
