import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) / 2
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    best_n_points = n_points
    
    iteration = 0
    no_improvement_count = 0
    min_improvement = 1e-6
    
    while time() - start_time < budget_s * 0.98:
        iteration += 1
        
        # Compute current overlap and find worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx] * dx
        k_shift = max_idx - (n_points - 1)  # Convert to shift value
        
        # Identify contributing regions
        indices = np.arange(n_points)
        # For each x, we look at h[x] * h[x+k_shift]
        # But we must be careful about domain
        source_indices = indices[:-abs(k_shift)] if k_shift < 0 else indices[abs(k_shift):]
        target_indices = indices[abs(k_shift):] if k_shift < 0 else indices[:-abs(k_shift)]
        
        # Calculate the product h[x]*h[x+k_shift] for contributing indices
        products = h[source_indices] * h[target_indices]
        top_indices = np.argsort(products)[-max(1, n_points // 10):]
        
        # Create a copy and reduce h in top contributing regions
        h_new = h.copy()
        adjustment = max(0.001, (max_val - 0.38) / n_points)
        for idx in top_indices:
            if k_shift < 0:
                source, target = idx, idx - k_shift
            else:
                source, target = idx + k_shift, idx
            if 0 <= source < n_points and 0 <= target < n_points:
                h_new[source] = max(0, h_new[source] - adjustment)
                h_new[target] = max(0, h_new[target] - adjustment)
        
        # Normalize and project
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate the new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better or sometimes randomly
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Random perturbation if no improvement
            if no_improvement_count > 50 or np.random.rand() < 0.05:
                perturbation = np.random.normal(0, 0.01, n_points)
                h_perturbed = h + perturbation
                h_perturbed = project_to_feasible(h_perturbed, n_points)
                c5_perturb = evaluate_erdos_solution(h_perturbed, dx)
                if c5_perturb < best_c5:
                    best_h = h_perturbed.copy()
                    best_c5 = c5_perturb
                h = h_perturbed
            else:
                h = h_new
        
        # Try aggressive resolution steps if time is available
        if iteration % 20 == 0 and iteration < 400:
            n_points_new = min(75, n_points + 15)
            if n_points_new != n_points:
                h_interp = np.interp(np.linspace(0, 2, n_points_new), 
                                    np.linspace(0, 2, n_points), 
                                    h)
                h = project_to_feasible(h_interp, n_points_new)
                n_points = n_points_new
                dx = 2.0 / n_points
                best_c5 = evaluate_erdos_solution(best_h, dx)
                # Update best_h if needed
                if n_points != len(best_h):
                    best_h_interp = np.interp(np.linspace(0, 2, n_points), 
                                             np.linspace(0, 2, len(best_h)), 
                                             best_h)
                    best_h = project_to_feasible(best_h_interp, n_points)
    
    return (list(best_h), best_c5, n_points)
