import numpy as np
from scipy import optimize
import time
import cvxpy as cvx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Try SDP approach first to get a better starting solution
    try:
        # Use a smaller resolution first (n=20) to find good structure
        n_small = min(20, n_points)
        
        # Create variables for SDP approach
        h_small = cvx.Variable(n_small)
        X_small = cvx.Variable((n_small, n_small), symmetric=True)
        t = cvx.Variable()
        
        # Constraints for SDP
        constraints = [
            X_small >> 0,  # Positive semidefinite
            cvx.diag(X_small) == h_small,  # Diagonal elements equal h
            0 <= h_small, h_small <= 1,  # Bounds on h
            cvx.sum(h_small) == n_small / 2.0,  # Sum constraint
        ]
        
        # Add overlap constraints
        for k in range(1, n_small):
            # Sum over shifts
            sum_overlap = cvx.sum(
                cvx.multiply(
                    h_small, 
                    cvx.roll(h_small, -k)
                )
            )
            constraints.append(t <= sum_overlap)
            
        # Objective
        objective = cvx.Maximize(t)
        
        # Solve
        prob = cvx.Problem(objective, constraints)
        prob.solve(solver=cvx.MOSEK, verbose=False, max_iters=100)
        
        if prob.status in [cvx.OPTIMAL, cvx.OPTIMAL_INACCURATE]:
            h_small_sol = h_small.value
            # Interpolate back to original resolution
            h_interpolated = np.interp(
                np.linspace(0, 1, n_points),
                np.linspace(0, 1, n_small),
                h_small_sol
            )
            h = project_h(h_interpolated)
        else:
            # Revert to existing h
            h = project_h(h)
    except:
        # Fallback to original approach
        h = project_h(h)
    
    # Refine with gradient-based optimization
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Optimizer settings
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        if iteration % 100 == 0:
            # Periodic full evaluation
            current_C5 = compute_C5(best_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = best_h.copy()
                
        # Compute overlaps to identify k contributing to max
        h_vec = project_h(h)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        # Top 5 contribution indices
        top_k = np.argsort(overlaps)[-5:][::-1]
        
        # Approximate gradient using finite differences
        gradient = np.zeros_like(h_vec)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            C_plus = compute_C5(h_plus)
            
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            C_minus = compute_C5(h_minus)
            
            gradient[i] = (C_plus - C_minus) / (2 * eps)
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * gradient
        h = h + velocity
        
        # Project
        h = project_h(h)
            
        # Adaptive learning rate
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            learning_rate *= 1.1  # Increase if improving
        else:
            learning_rate *= 0.9  # Decrease if not improving
            
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)
