import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def generate_candidates(n_points, n_candidates, base_h=None):
        candidates = []
        for _ in range(n_candidates):
            if base_h is not None:
                # Perturb existing solution
                h_new = base_h + np.random.normal(0, 0.05, n_points)
            else:
                # Random construction
                h_new = np.random.rand(n_points)
            h_new = project_feasible(h_new)
            candidates.append(h_new)
        return candidates
    
    # Grid resolution levels
    levels = [5, 10, 15, 20, 25, 30, 45]
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Search from coarse to fine
    for i, n in enumerate(levels):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # For first few levels, we do grid search
        if i < 3:  # Coarse levels
            n_candidates = 50
        else:  # Fine levels
            n_candidates = 20
            
        candidates = generate_candidates(n, n_candidates, 
                                       best_h if i >= 1 else None)
        
        # Project candidates to desired resolution
        projected_candidates = []
        for c in candidates:
            if len(c) != n:
                # Interpolate
                x_old = np.linspace(0, 2, len(c))
                x_new = np.linspace(0, 2, n)
                c_new = np.interp(x_new, x_old, c)
            else:
                c_new = c
            c_new = project_feasible(c_new)
            projected_candidates.append(c_new)
        
        # Evaluate all
        for c in projected_candidates:
            if time.time() - start_time > budget_s - 0.1:
                break
            C5 = compute_C5(c)
            if C5 < best_C5:
                best_C5 = C5
                best_h = c.copy()
    
    # Local refinement
    h_current = best_h.copy()
    for _ in range(50):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Random perturb
        h_perturbed = h_current + np.random.normal(0, 0.02, n_points)
        h_perturbed = project_feasible(h_perturbed)
        
        # Evaluate
        C5_perturbed = compute_C5(h_perturbed)
        if C5_perturbed < best_C5:
            best_C5 = C5_perturbed
            best_h = h_perturbed.copy()
            h_current = h_perturbed.copy()
    
    # Final check
    final_h = project_feasible(best_h)
    final_C5 = compute_C5(final_h)
    
    return list(final_h), final_C5, n_points
