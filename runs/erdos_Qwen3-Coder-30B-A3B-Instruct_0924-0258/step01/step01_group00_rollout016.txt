import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def generate_initial_h(n_points):
    if 'initial_h_values' in globals():
        return np.array(initial_h_values)
    else:
        # Start with a piecewise constant construction
        h = np.ones(n_points) * 0.5
        # Modify to sum to n_points / 2
        h = project_to_feasible(h, n_points)
        return h

def convex_optimize_step(h_start, n_points, dx, time_limit):
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Formulate convex relaxation: approximate max with a smooth function
    def smooth_maximum(terms, penalty_factor=1.0):
        # QP approximation of max function
        return cp.sum_squares(terms) + penalty_factor * cp.max(terms)

    # Sub-sampled optimization
    subsample = max(1, n_points // 20)
    indices = np.arange(0, n_points, subsample)
    
    # So we don't process too many indices
    indices = indices[:min(20, len(indices))]
    
    for _ in range(50):  # iterations
        if time() - start_time > time_limit:
            break
            
        # Create a strong initial solution
        h_test = project_to_feasible(h_start + np.random.normal(0, 0.02, n_points), n_points)
        
        # If we have a good enough solution, skip expensive optimization
        c5_test = evaluate_erdos_solution(h_test, dx)
        if c5_test < best_c5 * 0.9999:
            best_h = h_test.copy()
            best_c5 = c5_test
        
        # Simple gradient step with decay
        step_size = 0.01 * (1.0 - (time() - start_time) / time_limit)
        if step_size > 1e-6:
            h_test = project_to_feasible(h_start - step_size * np.random.normal(0, 1, n_points), n_points)
            c5_test = evaluate_erdos_solution(h_test, dx)
            if c5_test < best_c5:
                best_h = h_test.copy()
                best_c5 = c5_test
                
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Start coarse resolution
    n_points = 30
    h_start = generate_initial_h(n_points)
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try up to 4 resolutions with increasing detail
    resolutions = [30, 45, 60, 75]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to new resolution
        h_interp = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, len(h_start)), h_start)
        h_start = project_to_feasible(h_interp, res)
        dx = 2.0 / res
        
        # Optimize with localized gradient steps
        time_per_res = (budget_s - (time() - start_time)) / len(resolutions)
        h_opt, c5_opt = convex_optimize_step(h_start, res, dx, time_per_res)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n_points = res
            
        h_start = h_opt.copy()
    
    return (list(best_h), best_c5, len(best_h))
