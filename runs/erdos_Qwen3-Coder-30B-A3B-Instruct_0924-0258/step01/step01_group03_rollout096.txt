import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    # Ensure final sum is exact
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Normalize to exact sum
    h = project_to_feasible(h, n_points / 2)
    dx = 2.0 / n_points
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Main optimization loop
    iteration = 0
    max_iter = 10000
    patience = 0
    best_improvement = 0.0001
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        # Gradient-based local refinement
        # Compute correlations to find optimal shift
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        max_shift = max_shift_idx - (len(h) - 1)
        
        # Compute gradient direction
        grad = np.zeros_like(h)
        if max_shift < 0:
            # h[i] contributes to 1-h[i-max_shift]
            for i in range(len(h)):
                if 0 <= i - max_shift < len(h):
                    grad[i] += 1 - 2 * h[i - max_shift]
        elif max_shift > 0:
            # h[i-max_shift] contributes to 1-h[i]
            for i in range(len(h)):
                if 0 <= i + max_shift < len(h):
                    grad[i] += 1 - 2 * h[i + max_shift]
        else:
            # max_shift == 0, gradients do not depend on shift
            for i in range(len(h)):
                grad[i] = 1 - 2 * h[i]
        
        # Update with learning rate
        eta = 0.01 * (0.99 ** (iteration // 100))
        h_new = h - eta * grad
        
        # Project to feasible set
        h_new = project_to_feasible(h_new, n_points / 2)
        
        # Evaluate new solution and update best
        c5_new = evaluate(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1

        h = h_new
        iteration += 1
        
        # Early stopping if no improvement for too long
        if patience > 500 and iteration > 1000:
            break
    
    return (best_h.tolist(), float(best_c5), n_points)
