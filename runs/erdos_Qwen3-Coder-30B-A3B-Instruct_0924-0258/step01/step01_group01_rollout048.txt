import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def compute_gradient(h, dx, critical_lags):
    n = len(h)
    grad = np.zeros(n)
    # For each critical lag, add contribution to gradient
    for k in critical_lags:
        # Cross correlation contribution to gradient at each position
        # To avoid repeated computations, we compute these in a vectorized way
        # Use corr(h, 1-h) and its shifts to get gradient terms
        # The gradient is approximately -sign(h_i) * (1-h_{i+k}) for kth lag
        # But actually, since we're doing min over max, we only consider
        # those terms where the correlation is maximal
        if k < n:  # valid shift
            # Add contribution from h[i] * (1 - h[i+k]) 
            # We use reverse indexing - contribution of each h[i] 
            # to the cross-correlation at k where h[i+k] is involved
            # if we write correlation as conv(h, 1-h), then:
            # corr_k = sum_{i=0}^{n-1} h[i] * (1 - h[i+k]) 
            # so gradient of this with respect to h[i] is (1 - h[i+k]) if i+k < n
            if k >= 0:
                for i in range(n-k):
                    grad[i] -= (1 - h[i+k])
            else:  # k < 0 means we are looking at h[i+k] compared to h[i]  
                for i in range(-k, n):
                    grad[i] -= (1 - h[i+k])
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_low = 10
    n_high = 70
    n_points = n_low
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    last_c5 = best_c5
    
    # Adaptive step sizes
    step_size = 0.01
    min_step = 1e-5
    learning_rate_decay = 0.98
    
    last_improvement = 0
    improvement_threshold = 50
    max_iter_per_res = 1000
    iter_count = 0
    res_count = 0
    
    while True:
        if time.time() - start_time > budget_s:
            break
            
        # Reset for new resolution
        iter_count = 0
        gamma = 0.9
        iteration_count = 0
        
        while iter_count < max_iter_per_res:
            if time.time() - start_time > budget_s:
                break
                
            # Compute C5 and correlation for gradient selection
            corr = correlate(h, 1-h, mode='full')
            corr_values = corr * dx
            max_corr_idx = np.argmax(corr_values)
            
            # Identify critical lags corresponding to high contribution
            # Let's histogram the position to identify main contributors
            # We consider maybe 5% of the largest correlations as critical
            abs_corr = np.abs(corr_values)
            threshold_idx = np.argsort(abs_corr)[-max(1, len(abs_corr) // 20)]
            threshold = abs_corr[threshold_idx]
            
            # Select lags with strong correlation
            critical_lags = []
            for i, val in enumerate(corr_values):
                if abs(val) >= 0.9 * threshold:
                    # Convert index back to lag (from full correlation)
                    # length is 2*n_points -1 for full mode
                    lag = i - (n_points - 1)
                    critical_lags.append(lag)
                    
            # Compute gradient (at fixed points)
            grad = np.zeros(n_points)
            # This is a bit more approximate, but we can use FFT to speed up 
            # Final gradient computation
            # Iterate over correlation lags and contribute to gradient
            for lag in critical_lags[:10]:  # limit to first 10
                # Compute h_conv_1_minus_h gradient contribution
                # When computing cross-correlation diff: (h * (1-h)) with respect to h[i]
                # The derivative of sum h[i] (1-h[i+k]) over h[i] is (1-h[i+k])
                # if we shift accordingly
                if lag >= 0:
                    for i in range(n_points-lag):
                        grad[i] -= (1 - h[i+lag])
                else:
                    for i in range(-lag, n_points):
                        grad[i] -= (1 - h[i+lag])
                        
            # Normalize gradient for stability
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 0:
                grad = grad / grad_norm
            
            # Simpler: use the standard steepest descent approach
            # Since we're dealing with non-smooth functions, simple updates work well
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_constraints(h_new, dx)
            
            # Evaluate new solution
            c5_new = evaluate(h_new, dx)
            
            # Accept if better or with small probability if worse (annealing)
            if c5_new < last_c5 or np.random.rand() < 0.1:
                h = h_new
                last_c5 = c5_new
                if last_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = last_c5
                    last_improvement = 0
                else:
                    last_improvement += 1
                    
                # Adjust step size based on learning rate decay
                step_size *= learning_rate_decay
                if step_size < min_step:
                    step_size = min_step
            else:
                # Reset step size if didn't change
                step_size = min_step * 2
                
            iter_count += 1
            
        # Scale up resolution if improvement observed
        if res_count < 3 and last_improvement < improvement_threshold:
            if n_points < n_high:
                h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                                  np.linspace(0, 2, n_points), h)
                h = h_long
                n_points = 2 * n_points
                dx = 2.0 / n_points
                h = project_to_constraints(h, dx)
                last_c5 = evaluate(h, dx)
                step_size = 0.01
                last_improvement = 0
            else:
                break
        else:
            if last_improvement >= improvement_threshold and n_points > n_low:
                h = np.interp(np.linspace(0, 2, n_points//2), 
                             np.linspace(0, 2, n_points), h)
                n_points //= 2
                dx = 2.0 / n_points
                h = project_to_constraints(h, dx)
                last_c5 = evaluate(h, dx)
                step_size = 0.01
                last_improvement = 0
            else:
                break
        res_count += 1
        
    return best_h.tolist(), best_c5, len(best_h)
