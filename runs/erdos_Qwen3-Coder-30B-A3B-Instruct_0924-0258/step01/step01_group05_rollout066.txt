import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Grid-based search approach
    best_h, best_C5 = h.copy(), compute_C5(h)
    
    # Grid levels from coarse to fine
    grid_levels = [5, 10, 15, 20, 25, 30, 45]
    grid_candidates = {}
    
    # Generate coarse candidates
    for n in [5, 10, 15]:
        candidates = []
        tries = 1000 if n < 15 else 500
        for _ in range(tries):
            if time.time() - start_time > budget_s - 0.1:
                break
            h_c = np.random.rand(n)
            h_c = project_feasible(h_c)
            candidates.append(h_c)
        grid_candidates[n] = candidates
    
    # Process each level from coarse to fine
    for level in grid_levels:
        if time.time() - start_time > budget_s - 0.1:
            break
            
        if level not in grid_candidates:
            # Generate candidates for this level
            candidates = []
            tries = 200 if level < 30 else 100
            for _ in range(tries):
                if time.time() - start_time > budget_s - 0.1:
                    break
                # Interpolation from coarser grid or random
                if level <= 15:
                    h_coarse = np.random.choice(grid_candidates[15], replace=False)
                    if len(h_coarse) == level:
                        h_c = h_coarse.copy()
                    else:
                        # Interpolate
                        h_c = np.interp(np.linspace(0, len(h_coarse)-1, level), 
                                      np.arange(len(h_coarse)), h_coarse)
                else:
                    h_c = np.random.rand(level)
                h_c = project_feasible(h_c)
                candidates.append(h_c)
            grid_candidates[level] = candidates
            
        # Evaluate all candidates at this level
        for h_cand in grid_candidates[level]:
            h_cand = project_feasible(h_cand)
            c5_val = compute_C5(h_cand)
            if c5_val < best_C5:
                best_C5 = c5_val
                best_h = h_cand.copy()
                
        # Refine best candidates
        if len(grid_candidates[level]) > 0:
            top_candidates = sorted(grid_candidates[level], 
                                  key=lambda h_c: compute_C5(h_c))[:max(5, len(grid_candidates[level])//10)]
            refined = []
            for h_cand in top_candidates:
                if time.time() - start_time > budget_s - 0.1:
                    break
                # Add small perturbations to refine
                for _ in range(5):
                    if time.time() - start_time > budget_s - 0.1:
                        break
                    noise = np.random.normal(0, 0.05, level)
                    h_perturbed = h_cand + noise
                    h_perturbed = project_feasible(h_perturbed)
                    refined.append(h_perturbed)
                
            # Add these back to the candidates for next level if applicable
            grid_candidates[level] = refined
    
    # Final optimization - local search on best
    h_opt = best_h.copy()
    C5_opt = best_C5
    
    # Gradient descent with random restarts
    for _ in range(1000):  # Limited iterations to stay within budget
        if time.time() - start_time > budget_s - 0.1:
            break
            
        C5_current = compute_C5(h_opt)
        # Compute correlations
        corr = correlate(h_opt, 1 - h_opt, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_opt) - 1)
        
        # Gradient direction
        grad = np.zeros_like(h_opt)
        if max_lag >= 0:
            grad[max_lag:] = -h_opt[:-max_lag]
        else:
            grad[:max_lag] = -h_opt[-max_lag:]
            
        # Try large and small steps
        for step_size in [0.05, 0.1, 0.2]:
            h_trial = h_opt - step_size * grad
            h_trial = project_feasible(h_trial)
            C5_trial = compute_C5(h_trial)
            if C5_trial < C5_opt:
                h_opt = h_trial
                C5_opt = C5_trial
                break
                
        # Random perturbation
        h_pert = h_opt + np.random.normal(0, 0.02, len(h_opt))
        h_pert = project_feasible(h_pert)
        C5_pert = compute_C5(h_pert)
        if C5_pert < C5_opt:
            h_opt = h_pert
            C5_opt = C5_pert
            
        if C5_opt < best_C5:
            best_C5 = C5_opt
            best_h = h_opt.copy()
    
    h_best = project_feasible(best_h)
    C5_best = compute_C5(h_best)
    
    # Final evaluation might show better result, so return it
    return list(h_best), C5_best, n_points
