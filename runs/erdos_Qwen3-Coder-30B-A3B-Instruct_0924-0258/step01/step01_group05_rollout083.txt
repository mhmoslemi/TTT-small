import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # GA parameters
    pop_size = 50
    tournament_size = 10
    elitism = int(pop_size * 0.2)
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    max_gens = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_temp = np.random.rand(n_points)
        h_temp = project_feasible(h_temp)
        population.append(h_temp)
    
    best_C5 = float('inf')
    best_h = None
    no_improvement = 0
    
    # Evaluation of population
    def evaluate_population(pop):
        fitness = []
        for h in pop:
            c5 = compute_C5(h)
            fitness.append(c5)
        return np.array(fitness)
    
    # Tournament selection
    def tournament_selection(pop, fitness, k=tournament_size):
        selected = []
        for _ in range(len(pop)):
            idx = np.random.choice(len(pop), k)
            best_idx = idx[np.argmin([fitness[i] for i in idx])]
            selected.append(pop[best_idx].copy())
        return selected
    
    # Crossover
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        # Uniform crossover
        mask = np.random.rand(n_points) < 0.5
        child1 = np.where(mask, parent1, parent2)
        child2 = np.where(mask, parent2, parent1)
        
        # Project back to feasible region
        child1 = project_feasible(child1)
        child2 = project_feasible(child2)
        
        return child1, child2
    
    # Mutation
    def mutate(h_vec):
        h_mut = h_vec.copy()
        
        # Gaussian mutation
        mask = np.random.rand(n_points) < mutation_rate
        noise = np.random.normal(0, 0.05, n_points)
        h_mut[mask] += noise[mask]
        
        # Project back to feasible region
        h_mut = project_feasible(h_mut)
        return h_mut
    
    # Main GA loop
    for gen in range(max_gens):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate population
        fitness = evaluate_population(population)
        
        # Keep track of best
        current_best_idx = np.argmin(fitness)
        current_best_C5 = fitness[current_best_idx]
        if current_best_C5 < best_C5:
            best_C5 = current_best_C5
            best_h = population[current_best_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Early stopping
        if no_improvement > patience:
            break
        
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Create new population
        new_pop = []
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elitism]
        for i in elite_indices:
            new_pop.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_pop) < pop_size:
            p1, p2 = np.random.choice(len(selected), 2, replace=False)
            child1, child2 = crossover(selected[p1], selected[p2])
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_pop.append(child1)
            if len(new_pop) < pop_size:
                new_pop.append(child2)
        
        population = new_pop[:pop_size]
    
    # Final check
    if best_h is None:
        # If we never found a good solution, use the last generation
        fitness = evaluate_population(population)
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_C5 = fitness[best_idx]
    
    best_h = project_feasible(best_h)
    best_C5 = compute_C5(best_h)
    
    return list(best_h), best_C5, n_points
