import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    """Evaluate the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.array(h, dtype=float)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
        # Ensure we don't exceed bounds after scaling
        h = np.clip(h, 0, 1)
    else:
        # If current sum is 0, make it uniform
        h = np.ones_like(h) * (target_sum / len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():

        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5

    # dx for the integral
    dx = 2.0 / n_points
    
    # Define objective for scipy optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)

    # Initialize best solution
    best_h = project_to_feasible(h0, n_points / 2)
    best_c5 = objective(best_h)
    
    # Nelder-Mead parameters
    simplex_size = n_points + 2
    max_iter = 5000
    iteration = 0
    elapsed = 0

    # Generate initial simplex
    simplex = [best_h.copy()]
    for i in range(simplex_size - 1):
        perturbed = best_h + np.random.normal(0, 0.05, n_points)
        simplex.append(project_to_feasible(perturbed, n_points / 2))

    # Nelder-Mead iteration
    while elapsed < budget_s and iteration < max_iter:
        # Evaluate all points in simplex
        values = [objective(s) for s in simplex]
        # Order by value (ascending)
        order = np.argsort(values)
        simplex = [simplex[i] for i in order]
        values = [values[i] for i in order]
        
        # Update best
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
        
        # Centroid of best points (excluding worst)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        alpha = 1.0
        reflected = centroid + alpha * (centroid - simplex[-1])
        reflected = project_to_feasible(reflected, n_points / 2)
        reflected_val = objective(reflected)
        
        if values[0] <= reflected_val < values[-2]:
            simplex[-1] = reflected
        elif reflected_val < values[0]:
            # Expansion
            beta = 2.0
            expanded = centroid + beta * (reflected - centroid)
            expanded = project_to_feasible(expanded, n_points / 2)
            expanded_val = objective(expanded)
            if expanded_val < reflected_val:
                simplex[-1] = expanded
            else:
                simplex[-1] = reflected
        else:
            # Contraction
            gamma = 0.5
            if reflected_val < values[-1]:
                contracted = centroid + gamma * (reflected - centroid)
            else:
                contracted = centroid + gamma * (simplex[-1] - centroid)
            contracted = project_to_feasible(contracted, n_points / 2)
            contracted_val = objective(contracted)
            if contracted_val < values[-1]:
                simplex[-1] = contracted
            else:
                # Reduction: move all but best to midpoint of best and itself
                for i in range(1, simplex_size):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    simplex[i] = project_to_feasible(simplex[i], n_points / 2)

        # Every 50 iterations, add some randomness
        if iteration % 50 == 0 and iteration > 0:
            for i in range(len(simplex)):
                if np.random.random() < 0.3:
                    simplex[i] = project_to_feasible(
                        simplex[i] + np.random.normal(0, 0.02, n_points),
                        n_points / 2
                    )
        
        iteration += 1
        elapsed = time.time() - start_time

    # Final evaluation
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
