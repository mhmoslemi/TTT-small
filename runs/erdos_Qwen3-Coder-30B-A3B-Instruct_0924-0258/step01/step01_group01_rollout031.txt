import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def compute_smooth_max(h, dx, beta=10.0):
    corr = correlate(h, 1-h, mode='full')
    corr *= dx
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * corr)))

def compute_gradient(h, dx, beta=10.0):
    n = len(h)
    corr = correlate(h, 1-h, mode='full')
    corr *= dx
    exp_corr = np.exp(beta * corr)
    sum_exp = np.sum(exp_corr)
    if sum_exp == 0:
        return np.zeros_like(h)
    weights = exp_corr / sum_exp
    # Gradient of smooth max with respect to h
    # Need the gradient of each corr term with respect to h
    g = np.zeros(n)
    # For simplicity, use the FFT-based correlation gradient approach
    # Precompute forward differences or use finite differences
    # Or in frequency domain: FFT-based approach
    # This is a bit complex. Just do finite difference for now
    eps = 1e-6
    g = np.zeros_like(h)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        c_plus = compute_smooth_max(h_plus, dx, beta)
        c_minus = compute_smooth_max(h_minus, dx, beta)
        g[i] = (c_plus - c_minus) / (2.0 * eps)
    return g  

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 50
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
        
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Hyperparameters
    max_iter = 2000
    beta = 10.0
    learning_rate = 1e-3
    penalty = 1.0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute objective and gradient
        c5_val = compute_smooth_max(h, dx, beta)
        grad = compute_gradient(h, dx, beta)
        
        # Line search
        alpha = learning_rate
        h_new = h - alpha * grad
        h_new = project_to_constraints(h_new, dx)
        
        new_c5 = evaluate(h_new, dx)
        
        # Armijo backtracking line search
        while new_c5 >= c5_val and alpha > 1e-8:
            alpha *= 0.5
            h_new = h - alpha * grad
            h_new = project_to_constraints(h_new, dx)
            new_c5 = evaluate(h_new, dx)
            
        h = h_new
        c5 = new_c5
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Increase beta if needed to get closer to true max
        if iteration % 200 == 0 and beta < 200:
            beta *= 2
            
        if iteration % 100 == 0:
            if learning_rate > 1e-6:
                learning_rate *= 0.95
                
    # Refine on best solution
    refine_steps = 50
    for _ in range(refine_steps):
        if time.time() - start_time > budget_s:
            break
        c5_val = evaluate(best_h, dx)
        grad = compute_gradient(best_h, dx, beta=50)
        alpha = 1e-4
        h_new = best_h - alpha * grad
        h_new = project_to_constraints(h_new, dx)
        new_c5 = evaluate(h_new, dx)
        if new_c5 < c5_val:
            best_h = h_new
            best_c5 = new_c5
    
    return best_h.tolist(), best_c5, len(best_h)
