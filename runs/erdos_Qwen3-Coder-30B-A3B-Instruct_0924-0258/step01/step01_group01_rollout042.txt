import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with low resolution
    n_points = 10
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
        h = project_to_constraints(h, dx)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    iter_count = 0
    # Allow more iterations at higher resolutions
    max_iter_per_res = 1000
    
    # Optimization parameters
    T = 0.1
    cooling = 0.995
    iter_per_temp = 100
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for iter_num in range(max_iter_per_res):
            if time.time() - start_time > budget_s:
                break
            iter_count += 1
            
            # Perturb h slightly
            noise = np.random.normal(0, T, n_points)
            h_new = h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_constraints(h_new, dx)
            
            c5_new = evaluate(h_new, dx)
            
            delta = c5_new - best_c5
            if delta < 0 or np.random.uniform() < np.exp(-delta / T):
                h = h_new
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
                    
        # Adaptive cooling
        T *= cooling
        
        # Refine resolution
        if n_points < 100 and best_c5 < 0.381:
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points *= 2
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            T = 0.1
            best_c5 = evaluate(h, dx)
            best_h = h.copy()
            max_iter_per_res = 1000  # Reset iteration count
            
        if T < 1e-6 and n_points >= 100:
            break
            
    return best_h.tolist(), best_c5, len(best_h)
