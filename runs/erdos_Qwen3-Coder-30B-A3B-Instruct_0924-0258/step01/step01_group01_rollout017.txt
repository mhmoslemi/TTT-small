import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    # Project h to satisfy sum(h) * dx = 1
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Scale to have the right sum
    h = h * target_sum / current_sum
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust again to ensure correct sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find largest element and increment it
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

    np.random.seed(seed)
    start_time = time.time()

    n_low = 10
    n_high = 100
    n_points = n_low

        h = h[:n_points]

        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Track best ever
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    iter_count = 0
    max_iter_per_temp = 300
    
    # Optimize in batches, not all at once
    # Adaptive resolution
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Local optimization loop
        for _ in range(max_iter_per_temp):
            if time.time() - start_time > budget_s:
                break
                
            # Perturb with gradient-like moves + random noise
            h_new = h.copy()
            # Add noise for exploration
            noise = np.random.normal(0, 0.02, n_points)
            h_new = h_new + noise / 2
            h_new = project_to_constraints(h_new, dx)
            
            # Random small adjustments
            for i in range(n_points):
                if np.random.random() < 0.1:
                    # Small perturbation
                    h_new[i] += np.random.normal(0, 0.01)
            h_new = project_to_constraints(h_new, dx)
            
            c5_new = evaluate(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                
            if c5_new < best_c5 or np.random.random() < 0.7:
                h = h_new
            iter_count += 1
            
        # Try to interpolate to higher resolution
        if n_points < n_high and best_c5 < 0.4:
            old_h = h.copy()
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            best_c5 = evaluate(h, dx)
            if best_c5 < best_c5:
                best_h = h.copy()
            # Reset iteration count or cooling here
        else:
            if time.time() - start_time > budget_s * 0.9:
                break
    
    return best_h.tolist(), best_c5, len(best_h)
