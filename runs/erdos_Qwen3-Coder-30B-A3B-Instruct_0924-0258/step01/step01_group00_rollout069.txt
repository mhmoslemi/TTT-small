import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def genetic_mutation(h, mutation_rate=0.1):
    """Apply genetic mutation to h with some probability"""
    # 10% chance to mutate
    if np.random.random() < mutation_rate:
        # Select random point to modify
        idx = np.random.randint(0, len(h))
        # Slight change via normal distribution
        h[idx] += np.random.normal(0, 0.05)
        # Clamp
        h[idx] = np.clip(h[idx], 0, 1)
    return h

def create_initial_h(n_points):
    """Create a structured initial h"""
    if 'initial_h_values' in globals():
        return np.array(initial_h_values)
    # Start with an initial guess that has some structure
    h = np.ones(n_points) * 0.5
    # Add a bit of structure
    for i in range(n_points):
        h[i] = 0.5 + 0.3 * np.sin(2 * np.pi * i / n_points)
    return project_to_feasible(h, n_points)

def convex_relaxation_step(h, n_points, dx):
    """Use convex relaxation to improve h"""
    try:
        # Create the correlation matrix form
        corr_vals = np.correlate(h, 1-h, mode='full')
        # Sum of squares term to penalize max correlation
        objective_value = np.sum(corr_vals**2) * dx**2
        return objective_value
    except:
        # Fallback to a simpler heuristic
        return np.sum((h - 0.5)**2) * dx  # penalty for deviating from 0.5

def local_optimization_step(h, n_points, dx, maxiter=50):
    """Use local optimization step to refine h"""
    def obj(h):
        return convex_relaxation_step(h, n_points, dx)
    
    # Bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    # Ensure h is feasible before optimization
    h = project_to_feasible(h, n_points)
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, 
                       constraints=cons, options={'maxiter': maxiter})
        if res.success:
            h = project_to_feasible(res.x, n_points)
    except:
        pass
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Manage resolution hierarchy - start coarse and go fine
    res_levels = [15, 30, 45, 60]
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    current_h = create_initial_h(res_levels[0])
    current_n = res_levels[0]
    dx = 2.0 / current_n
    
    for level, n_points in enumerate(res_levels):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate or start from scratch
        if n_points != current_n:
            if n_points > current_n:
                # Interpolate
                x_old = np.linspace(0, 2, current_n)
                x_new = np.linspace(0, 2, n_points)
                current_h = np.interp(x_new, x_old, current_h)
            else:
                # Downsample
                step = current_n // n_points
                current_h = current_h[::step]
                current_h = project_to_feasible(current_h, n_points)
            current_n = n_points
            dx = 2.0 / current_n
        
        # Evolution loop
        current_c5 = evaluate_erdos_solution(current_h, dx)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            best_n_points = current_n
            
        # Run evolution with mutations + local search
        # 100 iterations of evolution + local optimization
        for iteration in range(100):
            if time() - start_time > budget_s * 0.99:
                break
                
            # Mutate
            h_mut = genetic_mutation(current_h.copy())
            # Local optimization
            h_opt = local_optimization_step(h_mut, current_n, dx, maxiter=20)
            # Evaluate
            c5 = evaluate_erdos_solution(h_opt, dx)
            if c5 < current_c5:
                current_h = h_opt
                current_c5 = c5
                # Update best if needed
                if c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = c5
                    best_n_points = current_n
        
        # Prepare for next level
        current_h = local_optimization_step(current_h, current_n, dx)
    
    # Final safeguard
    if best_h is None:
        best_h = create_initial_h(30)
        best_c5 = evaluate_erdos_solution(best_h, 2.0 / 30)
        best_n_points = 30
        
    return (list(best_h), best_c5, best_n_points)
