import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from parent or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization using a smooth step function
        n_points = 45
        h = np.zeros(n_points)
        mid = n_points // 2
        # Create a piecewise constant approximation of a step function
        h[:mid] = 1.0
        # Add some randomness to break symmetry
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Annealing parameters
    T0 = 0.1
    tau = budget_s / np.log(10) if budget_s > 1 else 100
    sigma = 0.1
    T_min = 1e-5
    
    def adaptive_sigma(acceptance_rate):
        if acceptance_rate < 0.1:
            return min(sigma * 1.2, 0.5)
        elif acceptance_rate > 0.4:
            return max(sigma * 0.8, 0.01)
        return sigma
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Track acceptance statistics
    acceptance_count = 0
    total_proposals = 0
    
    # Initialize temperature
    T = T0
    
    while time.time() - start_time < budget_s - 0.1:
        # Sample a random number for decision making
        r = np.random.random()
        
        # Perturb h
        h_new = h.copy()
        
        # Focus on top 20% contributing intervals if possible
        # Compute correlation to find high overlap regions
        corr = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        # Sample some intervals that contribute to this peak
        high_indices = np.random.choice(n_points, size=max(1, n_points // 5), replace=False)
        
        # Perturb selected indices
        for i in high_indices:
            h_new[i] += np.random.normal(0, sigma)
            
        # Optional: Add some flips for diversity
        if np.random.random() < 0.3:
            flip_indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
            for i in flip_indices:
                h_new[i] = 1.0 - h_new[i]
                
        # Project
        h_new = project_feasible(h_new)
        
        # Compute new objective
        C5_new = compute_C5(h_new)
        
        # Accept or reject
        if C5_new < best_C5 or (T > T_min and np.random.random() < np.exp(-(C5_new - best_C5) / T)):
            h = h_new.copy()
            if C5_new < best_C5:
                best_h = h_new.copy()
                best_C5 = C5_new
        
        # Update temperature and parameters
        t = time.time() - start_time
        T = T0 * np.exp(-t / tau)
        
        # Update sigma adaptively every few iterations
        if total_proposals % 20 == 0 and total_proposals > 0:
            acceptance_rate = acceptance_count / total_proposals if total_proposals > 0 else 0
            sigma = adaptive_sigma(acceptance_rate)
        
        total_proposals += 1
        if C5_new < best_C5 or (T > T_min and np.random.random() < np.exp(-(C5_new - best_C5) / T)):
            acceptance_count += 1
    
    # Final feasibility check
    best_h = project_feasible(best_h)
    best_C5 = compute_C5(best_h)
    
    return list(best_h), best_C5, n_points
