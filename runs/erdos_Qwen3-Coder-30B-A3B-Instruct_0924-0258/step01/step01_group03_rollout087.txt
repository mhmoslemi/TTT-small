import numpy as np

def evaluate_erdos_solution(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum constraint and bounds."""
    h = np.array(h, dtype=float)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Distribute difference
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        # Increase smaller values first
        idx = np.argsort(h)
        for i in range(len(idx)-1, -1, -1):
            if diff <= 0:
                break
            j = idx[i]
            if h[j] < 1.0:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
    else:
        # Need to decrease values
        # Decrease larger values first
        idx = np.argsort(h)
        for i in range(len(idx)):
            if diff >= 0:
                break
            j = idx[i]
            if h[j] > 0.0:
                update = min(-diff, h[j])
                h[j] -= update
                diff += update
    return h

def setup_simplex(h0, n_points):
    """Create initial simplex around h0."""
    simplex = [h0.copy()]
    for i in range(n_points):
        h = h0.copy()
        # Small random perturbation
        pert = np.random.normal(0, 0.05, n_points)
        h += pert
        h = project_to_feasible(h, n_points / 2)
        simplex.append(h)
    return simplex

    xworst = simplex[-1]

    centroid = np.mean(simplex[:-1], axis=0)

    freflected = evaluate_erdos_solution(xreflected, 2.0 / len(xbest))
    
    if fvals[0] <= freflected < fvals[-2]:
        # Accept reflection
        simplex[-1] = xreflected
        fvals[-1] = freflected
    elif freflected < fvals[0]:
        # Expansion
        xexpanded = centroid + beta * (xreflected - centroid)
        xexpanded = project_to_feasible(xexpanded, len(xbest) / 2)
        fexpanded = evaluate_erdos_solution(xexpanded, 2.0 / len(xbest))
        if fexpanded < freflected:
            simplex[-1] = xexpanded
            fvals[-1] = fexpanded
        else:
            simplex[-1] = xreflected
            fvals[-1] = freflected
    else:
        # Contraction
        if freflected < fvals[-1]:
            xcontracted = centroid + gamma * (xreflected - centroid)
        else:
            xcontracted = centroid + gamma * (xworst - centroid)
        xcontracted = project_to_feasible(xcontracted, len(xbest) / 2)
        fcontracted = evaluate_erdos_solution(xcontracted, 2.0 / len(xbest))
        if fcontracted < fvals[-1]:
            simplex[-1] = xcontracted
            fvals[-1] = fcontracted
        else:
            # Reduction
            for i in range(1, len(simplex)):
                simplex[i] = simplex[0] + delta * (simplex[i] - simplex[0])
                simplex[i] = project_to_feasible(simplex[i], len(xbest) / 2)
            # Recalculate function values
            fvals = [evaluate_erdos_solution(h, 2.0 / len(xbest)) for h in simplex]
    
    return simplex, fvals
