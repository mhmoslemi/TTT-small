import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Multi-resolution sweep approach
    resolution_levels = [5, 10, 20, 40, n_points]
    current_h = h.copy()
    best_h = current_h.copy()
    best_C5 = compute_C5(best_h)
    
    params = {
        'learning_rate': 0.01,
        'beta1': 0.9,
        'beta2': 0.999,
        'epsilon': 1e-8,
        'max_iter': 5000
    }
    
    # Adam optimizer variables
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    
    for level_idx, level_n in enumerate(resolution_levels):
        if time.time() - start_time > budget_s - 2:
            break
            
        if level_n == n_points:
            h_to_opt = current_h.copy()
        else:
            # Upsample current solution
            h_to_opt = np.interp(np.linspace(0, 1, level_n), np.linspace(0, 1, len(current_h)), current_h)
            h_to_opt = project_h(h_to_opt)
        
        max_iter = params['max_iter'] // (2**(len(resolution_levels) - 1 - level_idx))
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Compute overlaps to identify k contributing to max
            h_vec = project_h(h_to_opt)
            conv = np.correlate(h_vec, 1 - h_vec, mode='full')
            mid = len(conv) // 2
            overlaps = conv[mid-level_n+1:mid+level_n-1] * (2.0 / level_n)
            top_k = np.argsort(overlaps)[-5:][::-1]
            
            # Estimate gradient based on dominant shifts
            gradient = np.zeros_like(h_vec)
            eps = 1e-4
            for k in top_k[:3]:  # Just top 3 for efficiency
                # Approximate using finite differences along shift k
                weighted_grad = np.zeros_like(h_vec)
                for j in range(len(h_vec)):
                    h_plus = h_vec.copy()
                    h_plus[j] += eps
                    h_plus = project_h(h_plus)
                    conv_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    mid_plus = len(conv_plus) // 2
                    overlaps_plus = conv_plus[mid_plus-level_n+1:mid_plus+level_n-1] * (2.0 / level_n)
                    C_plus = overlaps_plus[k]
                    
                    h_minus = h_vec.copy()
                    h_minus[j] -= eps
                    h_minus = project_h(h_minus)
                    conv_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                    mid_minus = len(conv_minus) // 2
                    overlaps_minus = conv_minus[mid_minus-level_n+1:mid_minus+level_n-1] * (2.0 / level_n)
                    C_minus = overlaps_minus[k]
                    
                    weighted_grad[j] += (C_plus - C_minus) / (2 * eps)
                    
            # Adam update
            t += 1
            m = params['beta1'] * m + (1 - params['beta1']) * weighted_grad
            v = params['beta2'] * v + (1 - params['beta2']) * weighted_grad**2
            m_hat = m / (1 - params['beta1']**t)
            v_hat = v / (1 - params['beta2']**t)
            h_update = params['learning_rate'] * m_hat / (np.sqrt(v_hat) + params['epsilon'])
            
            h_to_opt = h_to_opt - h_update
            h_to_opt = project_h(h_to_opt)
            
            current_C5 = compute_C5(h_to_opt)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_to_opt.copy()
                
        # Use this as starting point for next level
        current_h = h_to_opt.copy()
    
    # Final check with more precise evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)
