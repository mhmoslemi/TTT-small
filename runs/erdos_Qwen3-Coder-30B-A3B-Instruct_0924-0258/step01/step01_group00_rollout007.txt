import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def optimize_with_augmented_lagrangian(h_start, n_points, dx, time_limit):
    """Optimize using augmented Lagrangian method"""
    start_time = time()
    
    # Initialize parameters
    h = h_start.copy()
    h = project_to_feasible(h, n_points)
    
    # Precompute correlation indices for efficiency
    full_correlation_length = 2 * n_points - 1
    correlation_indices = np.arange(full_correlation_length)
    
    # Set up optimization parameters
    max_iter = 500
    alpha = 0.0  # Initial guess for minimum overlap
    penalty = 1.0
    tolerance = 1e-6
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Augmented Lagrangian optimization loop
    for iteration in range(max_iter):
        if time() - start_time > time_limit:
            break
            
        # Compute current overlaps for all shifts
        corr = np.correlate(h, 1 - h, mode='full') * dx
        
        # Find current minimum overlap
        current_min = np.min(corr)
        
        # If we improved, update best solution
        if current_min < best_c5:
            best_h = h.copy()
            best_c5 = current_min
            
        # If converged, stop
        if abs(current_min - alpha) < tolerance:
            break
            
        # Compute new alpha (target for minimum overlap)
        alpha = current_min
        
        # Define the objective with penalty term
        def augmented_lagrangian(h_vals):
            h_tmp = project_to_feasible(h_vals, n_points)
            corr_tmp = np.correlate(h_tmp, 1 - h_tmp, mode='full') * dx
            min_val = np.min(corr_tmp)
            # Penalty for violating alpha constraint
            penalty_term = 0.5 * penalty * max(0, alpha - min_val)**2
            return -min_val + penalty_term  # Minimize negative of min overlap
        
        # Define gradient (numerically approximated)
        def grad_aug_lagrangian(h_vals):
            eps = 1e-6
            grad = np.zeros_like(h_vals)
            base_val = augmented_lagrangian(h_vals)
            for i in range(len(h_vals)):
                h_plus = h_vals.copy()
                h_plus[i] += eps
                grad[i] = (augmented_lagrangian(h_plus) - base_val) / eps
            return grad
        
        # Perform local optimization
        try:
            res = minimize(augmented_lagrangian, h, method='L-BFGS-B', 
                           bounds=[(0, 1) for _ in range(n_points)],
                           jac=grad_aug_lagrangian,
                           options={'maxiter': 20})
            if res.success:
                h = project_to_feasible(res.x, n_points)
            else:
                # Fallback: perturb and try again
                h = project_to_feasible(h + np.random.normal(0, 0.02, n_points), n_points)
        except:
            # Fallback to random perturbation
            h = project_to_feasible(h + np.random.normal(0, 0.02, n_points), n_points)
    
    return best_h, best_c5

    best_n_points = len(h_start)
    
    # Try different resolutions
    resolutions = [30, 40, 50, 60]
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate or adjust if needed
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points),

        remaining_time = budget_s - (time() - start_time)

        for _ in range(3):

                best_h = h_opt.copy()
                best_c5 = c5_opt
