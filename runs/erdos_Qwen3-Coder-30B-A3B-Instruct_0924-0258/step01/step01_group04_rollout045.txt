import numpy as np
from scipy import optimize
import time
try:
    from cvxpy import *
except ImportError:
    # Fall back to a simple implementation if CVXPY not available
    pass

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Read initial construction if available
    try:
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
    except NameError:
        # Fallback to a simple starting point
        n_points = 45
        h_initial = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Define helper functions
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Simple gradient descent approach
    best_h = h_initial.copy()
    best_C5 = compute_C5(best_h)
    
    # Optimization parameters
    momentum = 0.9
    velocity = np.zeros_like(h_initial)
    learning_rate = 0.01
    eps = 1e-4
    
    iterations = 0
    while time.time() - start_time < budget_s - 1:
        iterations += 1
        h_vec = project_h(best_h)
        
        # Compute approximate gradient
        gradient = np.zeros_like(h_vec)
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            C_plus = compute_C5(h_plus)
            
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            C_minus = compute_C5(h_minus)
            
            gradient[i] = (C_plus - C_minus) / (2 * eps)
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * gradient
        h_new = h_vec + velocity
        
        # Project and evaluate
        h_new = project_h(h_new)
        new_C5 = compute_C5(h_new)
        
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
            learning_rate *= 1.1
        else:
            learning_rate *= 0.9
            
        # Periodically evaluate
        if iterations % 100 == 0:
            current_C5 = compute_C5(best_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = best_h.copy()
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)
