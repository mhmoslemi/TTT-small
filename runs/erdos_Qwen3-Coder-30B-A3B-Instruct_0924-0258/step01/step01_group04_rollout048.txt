import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 20
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper function to evaluate C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    # Helper function to solve SDP
    def solve_sdp(h_initial, n):
        # Initialize variables
        h = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()
        
        # Objective
        objective = cp.Maximize(t)
        
        # Constraints
        constraints = [
            X >> 0,  # X is positive semidefinite
            cp.diag(X) == h,  # Diagonal elements of X should equal h
            0 <= h, h <= 1,  # Bounds on h
            cp.sum(h) == target_sum,  # Sum constraint
        ]
        
        # Add overlap constraints
        for k in range(1, n):
            sum_term = 0
            for i in range(n):
                j = (i + k) % n
                sum_term += X[i, j]
            constraints.append(t <= sum_term * dx)
        
        # Form and solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, max_iter=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_solution = h.value
                return h_solution
        except:
            pass
        return h_initial
    
    # Initial solution
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Try to improve with SDP
    for n in [20, 40, 45]:
        if time.time() - start_time > budget_s - 10:
            break
        if n_points < n:
            # Upscale using interpolation
            h_upscaled = np.interp(np.linspace(0, 2, n), 
                                 np.linspace(0, 2, n_points), 
                                 np.concatenate([h, h[:1]]))
        else:
            h_upscaled = h[:n]
        
        # Solve SDP
        h_sdp = solve_sdp(h_upscaled, n)
        
        if h_sdp is not None:
            h_projected = np.clip(h_sdp, 0, 1)
            h_projected = h_projected * target_sum / np.sum(h_projected)
            
            current_C5 = compute_C5(h_projected)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_projected.copy()
    
    # Final optimization with local search
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
        
    def compute_gradient(h_vec, eps=1e-4):
        gradient = np.zeros_like(h_vec)
        h_base = project_h(h_vec)
        base_C5 = compute_C5(h_base)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            C_plus = compute_C5(h_plus)
            
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            C_minus = compute_C5(h_minus)
            
            gradient[i] = (C_plus - C_minus) / (2 * eps)
        return gradient
    
    # Gradient descent
    h_opt = best_h.copy()
    learning_rate = 0.1
    velocity = np.zeros_like(h_opt)
    momentum = 0.9
    
    while time.time() - start_time < budget_s - 1:
        gradient = compute_gradient(h_opt)
        velocity = momentum * velocity - learning_rate * gradient
        h_opt = h_opt + velocity
        h_opt = project_h(h_opt)
        
        current_C5 = compute_C5(h_opt)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_opt.copy()
            
        # Adapt learning rate
        if learning_rate > 1e-5:
            learning_rate *= 0.995
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        
    return (best_h.tolist(), best_C5, len(best_h))
