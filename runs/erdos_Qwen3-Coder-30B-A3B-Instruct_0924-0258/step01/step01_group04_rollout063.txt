import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:

        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.random.uniform(0.4, 0.6, n_points)
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    # Optimization with differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(x):
        # Project to feasible domain
        x = project_h(x)
        return compute_C5(x)
    
    # Start with the initial solution to reduce search space
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Run differential evolution for random sampling
    try:
        result = differential_evolution(
            objective, 
            bounds, 
            maxiter=50, 
            popsize=15,
            tol=1e-6,
            recombination=0.7,
            atol=1e-6,
            seed=seed,
            callback=lambda x, convergence: None
        )
        if result.success:
            candidate_h = project_h(result.x)
            candidate_C5 = compute_C5(candidate_h)
            if candidate_C5 < best_C5:
                best_h = candidate_h
                best_C5 = candidate_C5
    except:
        pass
        
    # Local refinement with Nelder-Mead
    def refine_objective(x):
        x = project_h(x)
        return compute_C5(x)
    
    # Run Nelder-Mead starting from current best
    try:
        result_nm = minimize(refine_objective, best_h, method='Nelder-Mead', 
                           options={'maxiter': 500, 'disp': False})
        if not result_nm.failed:
            candidate_h = project_h(result_nm.x)
            candidate_C5 = compute_C5(candidate_h)
            if candidate_C5 < best_C5:
                best_h = candidate_h
                best_C5 = candidate_C5
    except:
        pass

    # Final check with more thorough local search
    # Simple gradient-based refinement (only for last few seconds)
    iteration = 0
    while time.time() - start_time < budget_s - 2:
        iteration += 1
        
        # Simple gradient via finite differences
        eps = 1e-4
        gradient = np.zeros(n_points)
        h_current = project_h(best_h)
        base_eval = compute_C5(h_current)
        
        for i in range(n_points):
            # Compute gradient at point i
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            C_plus = compute_C5(h_plus)
            
            h_minus = h_current.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            C_minus = compute_C5(h_minus)
            
            gradient[i] = (C_plus - C_minus) / (2 * eps)
        
        # Step with gradient descent
        step_size = 0.01 / (iteration**0.5 + 1)
        best_h = best_h - step_size * gradient
        best_h = project_h(best_h)
        
        # Update best solution
        current_C5 = compute_C5(best_h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            
    return (best_h.tolist(), best_C5, n_points)
