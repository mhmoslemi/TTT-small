import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def smooth_max(x, beta=10.0):
    # Smooth approximation to max
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))

def compute_correlation_gradient(h, dx):
    # Compute gradient of correlation with respect to h
    n = len(h)
    # corr(h, 1-h) for all lags
    # Use FFT for efficient computation
    h1 = np.concatenate([h, np.zeros(n-1)])
    h2 = np.concatenate([1-h, np.zeros(n-1)])
    corr_fft = np.fft.fft(h1) * np.conj(np.fft.fft(h2))
    corr = np.fft.ifft(corr_fft).real
    # Only positive lags
    cc = corr[n-1:]
    return cc

def run(seed=42, budget_s=600, **kwargs):

        else:
            h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
    
    h = project_to_constraints(h, dx)
    
    # Augmented Lagrangian parameters
    lambda_ = 0.0
    mu = 1.0

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:

        # Evaluate current point
        c5 = evaluate(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Compute constraint residual
        constraint_val = np.sum(h) - n_points / 2.0
        
        # Try direct gradient descent with projected updates
        # Compute correlation at each shift
        corr_full = correlate(h, 1-h, mode='full')
        # Find the maximum correlation position
        max_idx = np.argmax(corr_full)
        # Compute gradient w.r.t h at max position
        # Approximate gradient using finite differences for numerical stability
        eps = 1e-4
        grad_h = np.zeros_like(h)
        h_plus = h.copy()
        h_minus = h.copy()
        
        for i in range(n_points):
            # Perturb h[i]
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Project
            h_plus = project_to_constraints(h_plus, dx)
            h_minus = project_to_constraints(h_minus, dx)

            c5_plus = evaluate(h_plus, dx)

            # Reset
            h_plus[i] = h[i]

        # Gradient descent update
        if np.any(np.isnan(grad_h)):
