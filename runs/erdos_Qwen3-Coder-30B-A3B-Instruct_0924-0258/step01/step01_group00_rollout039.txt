import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def robust_objective(h, n_points, dx):
    """Compute a smooth robust objective with focus on minimizing max."""
    h = project_to_feasible(h, n_points)
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    # Smooth penalty for deviation from target
    penalty = 0.01 * np.sum(np.maximum(corr - max_val + 0.01, 0)**2)
    return max_val * dx + penalty

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Start from initial construction or uniform distribution
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 30
        h_start = np.ones(n_points) * 0.5

    # Use a multiresolution search
    resolutions = [20, 30, 40, 50]
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # If different resolution, interpolate
        if len(h_start) != n_points:
            x_orig = np.linspace(0, 2, len(h_start))
            x_new = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_new, x_orig, h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Multiple restarts
        best_h_level = h_start.copy()
        best_c5_level = evaluate_erdos_solution(h_start, dx)
        
        # Try several random restarts
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Random perturbation around current best solution
            h_perturbed = project_to_feasible(
                best_h_level + np.random.normal(0, 0.05, n_points), n_points
            )
            
            # Optimization using L-BFGS-B
            def obj(h):
                return robust_objective(h, n_points, dx)
            
            # Bounds and constraints
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            
            try:
                res = minimize(obj, h_perturbed, method='L-BFGS-B',
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5_level:
                        best_c5_level = c5_opt
                        best_h_level = h_opt.copy()
            except:
                pass
        
        # Update global best if improved
        if best_c5_level < best_c5:
            best_c5 = best_c5_level
            best_h = best_h_level.copy()
            best_n_points = n_points
        
        h_start = best_h_level.copy()
    
    return (list(best_h), best_c5, best_n_points)
