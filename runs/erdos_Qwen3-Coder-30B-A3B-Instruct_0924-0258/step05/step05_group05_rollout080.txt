import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a piecewise linear function that ramps up and down
        h_start = np.linspace(0, 1, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    population_size = 30
    mutation_factor = 0.8
    crossover_rate = 0.8
    max_generations = 500
    
    # Initialize population
    population = []
    fitness = []
    
    # Add the initial solution and perturbed versions
    for _ in range(population_size):
        h = h_start + np.random.normal(0, 0.03, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Evolution loop
    stagnation_counter = 0
    current_best = min(fitness)
    
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        if sorted_fit[0] < best_c5:
            best_h = sorted_pop[0].copy()
            best_c5 = sorted_fit[0]
            if best_c5 < 0.3807:
                break
        
        if sorted_fit[0] < current_best:
            current_best = sorted_fit[0]
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Create new population
        new_pop = [sorted_pop[0], sorted_pop[1]]  # Elitism
        
        # Differential evolution
        for i in range(2, population_size):
            # Select three distinct individuals
            a, b, c = np.random.choice(len(sorted_pop), 3, replace=False)
            trial = sorted_pop[a] + mutation_factor * (sorted_pop[b] - sorted_pop[c])
            
            # Apply crossover
            crossover_mask = np.random.random(n_points) < crossover_rate
            if not np.any(crossover_mask):
                crossover_mask[np.random.randint(0, n_points)] = True
            
            trial = np.where(crossover_mask, trial, sorted_pop[i])
            
            # Repair and evaluate
            trial = np.clip(trial, 0, 1)
            if np.sum(trial) == 0:
                trial = np.ones_like(trial) * 0.5
            else:
                trial = trial * (n_points/2) / np.sum(trial)
                
            new_pop.append(trial)
            fitness.append(evaluate_erdos_solution(trial, dx))
        
        # Update population
        population = new_pop
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Local refinement of the best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 300, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
