import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary algorithm parameters
    pop_size = 50
    max_generations = 1000
    mutation_rate = 0.15
    elite_size = 5
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Add initial solution if it exists and is feasible
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = h_init * (n_points/2) / np.sum(h_init)
        if np.all(h_init >= 0) and np.all(h_init <= 1):
            population[0] = h_init
    
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate fitness for all individuals
        fitness_scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness_scores.append(c5)
        
        # Sort population by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitness[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = sorted_fitness[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Check for timeout or stagnation
        if no_improvement_count > max_no_improvement:
            # Restart with better individuals from last generation
            population = sorted_population[:elite_size]
            while len(population) < pop_size:
                h = sorted_population[np.random.randint(0, elite_size)].copy()
                h += np.random.normal(0, 0.02, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                population.append(h)
            no_improvement_count = 0
        
        # Selection: tournament selection
        def tournament_selection(tournament_size=3):
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [sorted_fitness[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]
            return sorted_population[winner_index]
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_selection()
            parent2 = tournament_selection()
            
            # Crossover: single point
            crossover_point = np.random.randint(1, n_points-1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            for child in [child1, child2]:
                for i in range(n_points):
                    if random.random() < mutation_rate:
                        child[i] += np.random.normal(0, 0.05)
                # Ensure feasibility
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                new_population.append(child)
                if len(new_population) >= pop_size:
                    break
        
        population = new_population[:pop_size]
        generation += 1
    
    # Final optimization with local search if time permits
    if time() - start_time < budget_s * 0.9:
        try:
            # Local refinement around best solution with different approaches
            h_refined = best_h.copy()
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            
            # Additional local improvements using different optimizer approaches
            improved = True
            attempts = 0
            while improved and attempts < 10 and time() - start_time < budget_s * 0.9:
                improved = False
                attempts += 1
                
                # SLSQP optimization on a perturbed version
                h_pert = h_refined + np.random.normal(0, 0.01, n_points)
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points/2) / np.sum(h_pert)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                try:
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    bounds = [(0, 1) for _ in range(n_points)]
                    res_opt = minimize(
                        lambda h: evaluate_erdos_solution(h, dx),
                        h_pert,
                        method='SLSQP',
                        bounds=bounds,
                        constraints=cons_sum,
                        options={'maxiter': 100, 'ftol': 1e-6}
                    )
                    if res_opt.success:
                        h_new = res_opt.x
                        c5_new = evaluate_erdos_solution(h_new, dx)
                        if c5_new < c5_refined:
                            h_refined = h_new
                            c5_refined = c5_new
                            improved = True
                except:
                    pass
            
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
