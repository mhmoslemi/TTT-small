import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Initialization
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.4
        h_start[1] = 0.6
    
    # Ensure constraints
    dx = 2.0 / n_points
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Main optimization loop
    max_iter = 1000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Reference to current h for optimization
    current_h = best_h.copy()
    
    # Add a refined perturbation approach
    def perturb_h(h, strength=0.03):
        h_pert = h + np.random.normal(0, strength, len(h))
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        return h_pert
    
    # Constraint function for sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Optimization step
    def optimize_step(h_init, maxiter=50):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx),
                h_init,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-8}
            )
            if res.success:
                return res.x
        except Exception:
            pass
        return h_init
    
    # Trust region optimization or fallback
    def trust_region_optimize(h_init, maxiter=100):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx),
                h_init,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': maxiter}
            )
            if res.success:
                return res.x
        except Exception:
            pass
        return h_init
    
    # Refinement loop
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1
        
        # Try to reduce overlap at each step
        h_optimized = optimize_step(current_h.copy(), maxiter=30)
        
        # Ensure constraints and normalization
        h_optimized = np.clip(h_optimized, 0, 1)
        h_optimized = h_optimized * (n_points/2) / np.sum(h_optimized)
        c5_opt = evaluate_erdos_solution(h_optimized, dx)
        
        # Check if we have an improvement
        if c5_opt < best_c5 - 1e-8:
            best_h = h_optimized.copy()
            best_c5 = c5_opt
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        if no_improvement_count > 20:

        else:
            current_h = h_optimized

        if iter_count % 50 == 0:
            current_h = trust_region_optimize(current_h, maxiter=20)

    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        res = minimize(
            lambda h: evaluate_erdos_solution(h, dx),

            bounds=bounds,
            options={'maxiter': 50}

            final_h = np.clip(final_h, 0, 1)

            final_c5 = evaluate_erdos_solution(final_h, dx)

        pass
    
    # Ensure final constraints are respected
    final_h = best_h.copy()

    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
