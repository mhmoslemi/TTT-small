import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm with multiresolution approach
    max_generations = 1000
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def create_individual():
        h = np.random.rand(n_points)
        return normalize(h)
    
    if 'initial_h_values' in globals():
        population = [h_start.copy()]
        for _ in range(population_size - 1):
            population.append(create_individual())
    else:
        population = [create_individual() for _ in range(population_size)]
    
    generation = 0
    last_improvement = 0
    stagnation_count = 0
    
    while generation < max_generations and time() - start_time < budget_s * 0.9:
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            last_improvement = generation
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Check for stagnation
        if stagnation_count > 100:
            # Random restart
            for i in range(population_size):
                if np.random.rand() < 0.3:
                    population[i] = create_individual()
            stagnation_count = 0
        
        # Selection (tournament)
        new_population = []
        for _ in range(population_size):
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            new_population.append(winner[0].copy())
        
        # Crossover and mutation
        for i in range(0, population_size, 2):
            if random.random() < crossover_rate and i + 1 < population_size:
                # Single-point crossover
                point = random.randint(1, n_points - 1)
                parent1, parent2 = new_population[i], new_population[i+1]
                child1 = np.concatenate([parent1[:point], parent2[point:]])
                child2 = np.concatenate([parent2[:point], parent1[point:]])
                
                # Repair
                child1 = normalize(child1)
                child2 = normalize(child2)
                new_population[i] = child1
                new_population[i+1] = child2
        
        # Mutation
        for i in range(population_size):
            if random.random() < mutation_rate:
                individual = new_population[i]
                indices = random.sample(range(n_points), max(1, int(mutation_rate * n_points)))
                for idx in indices:
                    individual[idx] += np.random.normal(0, 0.02)
                individual = np.clip(individual, 0, 1)
                new_population[i] = normalize(individual)
        
        population = new_population
        generation += 1
        
        # Adapt resolution if needed and enough time left
        if 0.8 * budget_s < time() - start_time and best_c5 > 0.3808 and n_points == 35:
            n_points = 40
            dx = 2.0 / n_points
            # Interpolate best solution to new resolution
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_points)
            best_h = np.interp(x_new, x_old, best_h)
            best_h = normalize(best_h)
            # Restart with smaller population and different initialization
            population_size = 20
            population = [best_h.copy()]
            for _ in range(population_size - 1):
                population.append(create_individual())
            best_c5 = evaluate_erdos_solution(best_h, dx)
            last_improvement = generation
            stagnation_count = 0
    
    # Final local refinement of best solution with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Coerce final solution to constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
