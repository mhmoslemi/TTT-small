import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    max_iter = 5000
    mu = 1e-3
    tol = 1e-6
    admm_iter = 0
    
    # Initialize ADMM variables
    h = h_start.copy()
    h_prev = h.copy()
    z = np.zeros(2 * n_points - 1)
    lambda_vec = np.zeros(2 * n_points - 1)
    
    # Precompute 1-h for efficiency
    one_minus_h = 1 - h
    
    # Main ADMM loop
    while admm_iter < max_iter and (time() - start_time) < max_time * 0.95:
        admm_iter += 1
        
        # Update h with solution to subproblem
        # minimize ||h - h_prev||^2 + mu * ||z - (h * (1-h))||^2 
        # s.t. 0 <= h <= 1, sum(h) = n_points / 2
        
        # Form system A^T A h = A^T (A h_prev + mu * (z - (h_prev * (1-h_prev)) + lambda_vec/mu)
        # This is a quadratic problem with fixed constraints
        
        def obj(h_vec):
            # Constraint violation for sum
            sum_viol = (np.sum(h_vec) - n_points / 2.0) ** 2
            # Overlap violations
            overlap_terms = 0
            for k in range(2 * n_points - 1):
                overlap_val = 0
                for i in range(n_points):
                    j = i + k - (n_points - 1)
                    if 0 <= j < n_points:
                        overlap_val += h_vec[i] * (1 - h_vec[j])
                overlap_terms += (overlap_val - z[k]) ** 2
            
            return mu * overlap_terms + sum_viol
        
        # Warm start for local refinement
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
            
        sums_cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            # Use latest h as initialization
            res = minimize(obj, h, method='SLSQP', bounds=bounds, 
                           constraints=sums_cons, options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                h_new = res.x.copy()
                # Project to [0,1] and normalize
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h_prev = h.copy()
                h = h_new
            else:
                # Fallback if SLSQP fails
                h = h_prev.copy()
        except:
            h = h_prev.copy()
            
        # Update z and lambda
        # z = prox_{1/mu}(h * (1-h) + lambda/mu)
        # Since we do not want to make new variables in the overrelaxation here,
        # we relax degree of freedom in constraints.
        # This is a simplified interface. The actual update for ADMM:
        # z^new = prox_{1/mu}(rho * (h - h_prev) + lambda)
        # But to simplify and overcome local minima,
        # we anneal mu and make cheap to evaluate local improvements
            
        # Simple update of z. We can do better here, but for now stick to standard.
        h_extended = np.concatenate([h, np.zeros(n_points - 1)])
        one_minus_h_extended = 1 - h_extended
        # Efficient convolution using FFT:
        corr = np.correlate(h_extended[:n_points], one_minus_h_extended[:n_points], mode='full')
        z = corr
        
        # Simple update of lambda
        lambda_vec += mu * (z - corr)
        
        # Compute current estimate of C5
        c5_current = evaluate_erdos_solution(h, dx)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # Early stopping
        if best_c5 < 0.3807:
            break
            
    # Final local refinement
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
