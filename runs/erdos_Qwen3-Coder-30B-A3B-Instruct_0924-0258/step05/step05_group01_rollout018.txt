import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary tabu search approach
    # Convert to binary
    def to_binary(h):
        # Round to nearest binary and adjust to meet constraint
        h_binary = (h > 0.5).astype(int)
        current_sum = np.sum(h_binary)
        target_sum = n_points // 2
        
        if current_sum > target_sum:
            # Need to flip some 1s to 0s
            ones_indices = np.where(h_binary == 1)[0]
            flip_indices = np.random.choice(ones_indices, current_sum - target_sum, replace=False)
            h_binary[flip_indices] = 0
        elif current_sum < target_sum:
            # Need to flip some 0s to 1s
            zeros_indices = np.where(h_binary == 0)[0]
            flip_indices = np.random.choice(zeros_indices, target_sum - current_sum, replace=False)
            h_binary[flip_indices] = 1
            
        return h_binary
    
    def generate_neighbor(h, tabu_list):
        # Try to flip one 1 to 0 and one 0 to 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            # Can't generate neighbor, return copy
            return h.copy()
            
        # Pick a random 1 to flip to 0 and a random 0 to flip to 1
        i1 = np.random.choice(ones)
        i2 = np.random.choice(zeros)
        
        # Check tabu
        move = tuple(sorted([i1, i2]))
        if move in tabu_list:
            # Try to find a move that is not tabu
            for _ in range(10):  # Try up to 10 times
                i1 = np.random.choice(ones)
                i2 = np.random.choice(zeros)
                move = tuple(sorted([i1, i2]))
                if move not in tabu_list:
                    break
            else:
                # If no non-tabu move found, just return current
                return h.copy()
        
        h_new = h.copy()
        h_new[i1] = 0
        h_new[i2] = 1
        return h_new
    
    # Initialize
    h_current = to_binary(best_h)
    
    tabu_list = set()
    tabu_tenure = 10
    no_improvement_count = 0
    max_no_improvement = 100
    shaking_count = 0
    shaking_interval = 50
    
    while time() - start_time < budget_s * 0.95:
        h_new = generate_neighbor(h_current, tabu_list)
        
        # Add old move to tabu list
        old_indices = np.where((h_current != h_new) & (h_current == 1))[0]
        new_indices = np.where((h_current != h_new) & (h_current == 0))[0]
        if len(old_indices) > 0 and len(new_indices) > 0:
            new_move = tuple(sorted([old_indices[0], new_indices[0]]))
            tabu_list.add(new_move)
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop()
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Check aspiration
        aspirational = c5_new < best_c5
        
        # Update best if improved
        if c5_new < best_c5 or aspirational:
            if c5_new < best_c5:
                # Clear no_improvement count on actual improvement
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            best_c5 = c5_new
            best_h = h_new.copy()
            h_current = h_new.copy()
        else:
            no_improvement_count += 1
            h_current = h_new
        
        # Shaking
        if no_improvement_count >= shaking_interval:
            # Flip a percentage of bits
            frac = 0.05
            n_flip = max(1, int(n_points * frac))
            indices = np.random.choice(n_points, n_flip, replace=False)
            for idx in indices:
                h_current[idx] = 1 - h_current[idx]
                
            no_improvement_count = 0
            shaking_count += 1
            # Re-add some elements to tabu if we have too many
            if len(tabu_list) > tabu_tenure:
                tabu_list.clear()
        
        # Break if no improvement for too long
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final refinement
    try:
        h_final_binary = to_binary(best_h)
        c5_final = evaluate_erdos_solution(h_final_binary, dx)
        if c5_final < best_c5:
            best_h = h_final_binary
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
