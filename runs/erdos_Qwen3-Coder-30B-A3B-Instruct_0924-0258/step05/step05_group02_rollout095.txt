import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_best = h_start.copy()
    h_best = np.clip(h_best, 0, 1)
    h_best = h_best * (n_points/2) / np.sum(h_best)
    c5_best = evaluate_erdos_solution(h_best, dx)
    
    # Coarse optimization
    n_coarse = 20
    h_coarse = np.interp(np.linspace(0, 1, n_coarse), 
                         np.linspace(0, 1, n_points), 
                         h_best)
    h_coarse = h_coarse * (n_coarse/2) / np.sum(h_coarse)
    
    def coarse_objective(h_coarse):
        h_interp = np.interp(np.linspace(0, 1, n_points), 
                             np.linspace(0, 1, n_coarse), 
                             h_coarse)
        return evaluate_erdos_solution(h_interp, dx)
    
    def coarse_constraint_sum(h):
        return np.sum(h) - n_coarse / 2.0
    
    bounds_coarse = [(0, 1) for _ in range(n_coarse)]
    res_coarse = minimize(
        lambda h: coarse_objective(h), 
        h_coarse, 
        method='SLSQP', 
        bounds=bounds_coarse, 
        constraints={'type': 'eq', 'fun': coarse_constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    
    if res_coarse.success:
        h_upsampled = np.interp(np.linspace(0, 1, n_points), 
                                np.linspace(0, 1, n_coarse), 
                                res_coarse.x)
        h_upsampled = np.clip(h_upsampled, 0, 1)
        h_upsampled = h_upsampled * (n_points/2) / np.sum(h_upsampled)
        c5_upsampled = evaluate_erdos_solution(h_upsampled, dx)
        if c5_upsampled < c5_best:
            h_best = h_upsampled
            c5_best = c5_upsampled
    
    # Refinement loop
    max_iter = 200
    iter_count = 0
    no_improvement_count = 0
    epsilon = 1e-6
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        h_current = h_best.copy()
        c5_current = c5_best
        
        # Get all cross-correlations
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        max_overlap = max_corr * dx
        
        # Add constraint for current maximum
        k = max_k - (n_points - 1)
        if k < 0:
            corr_sum = np.sum(h_current[:n_points+k] * (1 - h_current[-k:]))
        else:
            corr_sum = np.sum(h_current[k:] * (1 - h_current[:n_points-k]))
            
        # Refine with new constraint
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        def constraint_max_overlap(h):
            if k < 0:
                corr_val = np.sum(h[:n_points+k] * (1 - h[-k:]))
            else:
                corr_val = np.sum(h[k:] * (1 - h[:n_points-k]))
            return max_overlap - corr_val * dx - epsilon
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': constraint_max_overlap}
        ]
        
        h_refined = h_current.copy()
        # Try local optimization
        try:
            res = minimize(
                objective, 
                h_refined, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-7}
            )
            if res.success:
                h_test = res.x
                # Ensure normalized
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points/2) / np.sum(h_test)
                c5_test = evaluate_erdos_solution(h_test, dx)
                if c5_test < c5_current:
                    h_best = h_test
                    c5_best = c5_test
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                h_test = h_current.copy()
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points/2) / np.sum(h_test)
                c5_test = evaluate_erdos_solution(h_test, dx)
                if c5_test < c5_current:
                    h_best = h_test
                    c5_best = c5_test
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
        except Exception:
            h_test = h_current.copy()
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points/2) / np.sum(h_test)
            c5_test = evaluate_erdos_solution(h_test, dx)
            if c5_test < c5_current:
                h_best = h_test
                c5_best = c5_test
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
        iter_count += 1
        
        # Random perturbation if no improvement
        if no_improvement_count > 50:
            h_perturbed = h_current + np.random.normal(0, 0.02, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < c5_best:
                h_best = h_perturbed
                c5_best = c5_perturbed
            no_improvement_count = 0
            
    # Final check and normalization
    h_final = np.clip(h_best, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
