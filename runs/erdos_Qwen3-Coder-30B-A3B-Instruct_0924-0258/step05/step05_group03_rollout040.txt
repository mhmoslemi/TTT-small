import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Fallback: create a reasonable initial vector
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # CCP: Convex-Concave Procedure
    dx = 2.0 / n_points
    max_iter = 1000
    tol = 1e-6
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find worst shift k* - sample potential shifts
        shifts = np.linspace(0, 2, 100)
        max_corr_val = -np.inf
        best_shift = 0
        
        for shift in shifts:
            # Compute correlation for shift
            shift_idx = int(shift / dx)
            if shift_idx >= n_points:
                shift_idx = n_points - 1
                
            # Shift h by shift_idx positions and compute overlap
            if shift_idx == 0:
                corr_val = np.sum(h * (1 - h)) * dx
            else:
                h_shifted = np.concatenate([h[shift_idx:], h[:shift_idx]])
                corr_val = np.sum(h * (1 - h_shifted)) * dx
                
            if corr_val > max_corr_val:
                max_corr_val = corr_val
                best_shift = shift
            
        # Solve convex subproblem for fixed shift
        # Minimize sum(h_new[i] * (1 - h_prev[i + k*]) * dx) subject to constraints
        # This is a linear problem in h_new
        # Use cvxpy for numerics stability
        h_prev = h.copy()
        
        # Reconstruct the convolution efficiently
        shift_idx = int(best_shift / dx)
        if shift_idx >= n_points:
            shift_idx = n_points - 1
        
        h_shifted = np.concatenate([h_prev[shift_idx:], h_prev[:shift_idx]])
        
        # Create and solve convex program
        h_new = cp.Variable(n_points)
        
        # Objective: minimize sum(h_new * (1 - h_shifted))
        objective = cp.Minimize(cp.sum(cp.multiply(h_new, 1 - h_shifted)))
        
        # Constraints
        constraints = [
            h_new >= 0,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2
        ]
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]:
                h_update = h_new.value
                h_update = np.clip(h_update, 0, 1)
                # Project onto sum constraint
                h_update = h_update * (n_points / 2) / np.sum(h_update)
                h = h_update
                current_c5 = evaluate_erdos_solution(h, dx)
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    
                if abs(best_c5 - current_c5) < tol:
                    break
                    
        except:
            # Fallback to direct update if optimization fails
            h = h_prev.copy()
            
        # Debugging check
        # if iteration % 50 == 0:
        #     print(f"Iteration {iteration}: C5 = {best_c5}")
            
        if best_c5 < 0.3807:
            break
    
    # Final polishing with SLSQP
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
