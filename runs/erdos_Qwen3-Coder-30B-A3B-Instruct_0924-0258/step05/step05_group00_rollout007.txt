import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    pop_size = 50
    max_generations = 1000
    tournament_size = 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    mutation_strength = 0.02
    
    # Population
    def initialize_population(size, n):
        population = []
        if 'initial_h_values' in globals():
            # Use initial construction as one parent
            h_initial = np.array(initial_h_values)
            h_initial = h_initial * (n/2) / np.sum(h_initial)
            population.append(h_initial.copy())
            size -= 1
            
        for _ in range(size):
            h = np.random.random(n)
            h = h * (n/2) / np.sum(h)
            population.append(h)
        return population
    
    def tournament_selection(population, fitnesses, k):
        best_idx = random.sample(range(len(population)), k)
        best_fit = min([fitnesses[i] for i in best_idx])
        return population[fitnesses.index(best_fit)]
    
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        n = len(parent1)
        point = random.randint(1, n-1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(individual, strength):
        n = len(individual)
        mask = np.random.random(n) < mutation_rate
        noise = np.random.normal(0, strength, n)
        individual = individual + noise * mask
        individual = np.clip(individual, 0, 1)
        individual = individual * (n/2) / np.sum(individual)
        return individual
    
    def ga_step(population, n_points, dx, max_iter):
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        
        new_population = []
        
        for _ in range(pop_size // 2):
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_strength)
            child2 = mutate(child2, mutation_strength)
            new_population.append(child1)
            new_population.append(child2)
        
        # Keep best individuals from old generation
        sorted_indices = np.argsort(fitnesses)
        for i in range(5):  # Keep top 5
            new_population[i] = population[sorted_indices[i]]
        
        return new_population
    
    # GA optimization loop
    population = initialize_population(pop_size, n_points)
    generation = 0
    
    while time() - start_time < budget_s * 0.8 and generation < max_generations:
        population = ga_step(population, n_points, dx, max_generations - generation)
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        current_best = min(fitnesses)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[fitnesses.index(current_best)].copy()
        generation += 1
    
    # Adaptive resolution improvement
    if time() - start_time < budget_s * 0.9 and best_c5 > 0.3808:
        n_points = 40
        dx = 2.0 / n_points
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, len(best_h)), 
                           best_h)
        best_h = h_interp * (n_points/2) / np.sum(h_interp)
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
        # Refine with local search
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                best_h = res.x
                best_c5 = objective(best_h)
        except Exception:
            pass
    
    # Final refinement
    try:
        h_final = np.clip(best_h, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
