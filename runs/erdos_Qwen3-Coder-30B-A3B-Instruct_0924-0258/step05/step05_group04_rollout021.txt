import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import cvxpy as cp
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.full(n_points, 0.5)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Build overlap matrix A_k for each shift k
    A_full = np.zeros((2 * n_points - 1, n_points))
    for k in range(2 * n_points - 1):
        shift = k - (n_points - 1)
        A_full[k, max(0, -shift):min(n_points, n_points-shift)] = 1
    
    # Precompute A_k for each shift
    A_list = []
    for k in range(2 * n_points - 1):
        # Build shift matrix A_k 
        A_k = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = i + k - (n_points - 1)
            if 0 <= j < n_points:
                A_k[i, j] = 1
        A_list.append(A_k)
    
    # Lagrangian relaxation approach
    max_iter = 500
    rho = 1e3
    lambda_multipliers = np.full(2 * n_points - 1, 0.1)  # Initialize multipliers
    mu = 1.0  # Penalty for sum constraint
    
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Solve convex subproblem
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(h_var))  # dummy objective to minimize first
        
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # Add validity constraints from overlap constraints
        for k in range(2 * n_points - 1):
            # Lagrange dual: add penalty term for overlap constraint
            # Actually just updating the constraints through dual method var, so we use a simplified proxy
            # and the projection step instead. 

        # Use quadratic programming solver (although not exact) to solve constrained problem
        # We use a direct approach with CVXPY - approach modulates the constraints via slack
        try:
            # Projected gradient approach or CVXPY constrained quadratic prog
            # However, for now, we extract a more direct approach
            
            # Build matrix representation for all shifts:
            # B matrix where row i represents the overlap integral for shift i
            
            # Quadratic programming to minimize sum(h_i^2) subject to:
            # A_k * h * (1 - h) < some threshold and sum(h) = n/2
            
            # For simplicity, start using an iterative algorithm without convex reformulation for now
            # Just use scipy optimization
                
            def solve_subproblem(h_init):
                def obj(h):
                    # Approximate non-convex function with simple heuristic for now
                    return evaluate_erdos_solution(h, dx)
                
                def con_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons = {'type': 'eq', 'fun': con_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(obj, h_init, method='SLSQP', bounds=bounds, 
                                   constraints=cons, options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        return res.x, obj(res.x)
                except:
                    pass
                return h_init, obj(h_init)
            
            h_opt, c5_opt = solve_subproblem(best_h)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            
            # Refine if improved solution found
            if best_c5 < 0.3807:

        # Simple random perturbation for escaping local minima:
        if iteration % 50 == 0:

                best_c5 = c5_noisy
                best_h = h_noisy.copy()
                
    # Final SLSQP refinement
    remaining_time = max_time - (time() - start_time)

            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP',

                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
