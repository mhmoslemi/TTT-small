import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Consts for Lagrangian-formulation
    max_iter = 500
    rho = 1e3
    tol = 1e-4
    
    # Variables
    h = cp.Variable(n_points)
    h_prev = h_start.copy()
    
    # Problem constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2.0]
    
    # Define matrix for shifts
    A = np.zeros((2*n_points-1, n_points))
    for i in range(2*n_points-1):
        shift = i - n_points + 1
        if shift >= 0:
            A[i, shift:] = 1
        else:
            A[i, :shift] = 1
    
    # Lagrangian solver
    for iter_num in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Try different starting points to escape local optima
        if iter_num % 10 == 0:
            h_init = h_prev.copy()
            h_init += np.random.normal(0, 0.01, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            h.value = h_init
        else:
            h.value = h_prev.copy()
        
        # Incremental constraint handling with Lagrangian
        # Simple approach: express overlap as matrix form and solve:
        # minimize    h^T * (I - h)
        # subject to  0 <= h <= 1, sum(h) = n/2
        
        # Use CVXPY for convex subproblem solve
        try:
            obj = cp.Minimize(cp.sum(cp.multiply(h, 1 - h)))  # Should be done right via full overlap contracts
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.OSQP, verbose=False, max_iter=1000)
            
            if prob.status == cp.OPTIMAL:
                h_new = h.value
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    
                # Update for next run
                h_prev = h_new
                
            # Early stop if target hit
            if best_c5 < 0.3807:
                break
                
        except:
            pass
    
    # Final local refinement using scipy minimize
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
