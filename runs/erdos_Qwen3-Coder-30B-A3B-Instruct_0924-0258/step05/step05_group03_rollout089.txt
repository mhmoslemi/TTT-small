import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:

    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    # Parameters
    n_samples = 50  # Number of shifts to sample
    max_iter = 500
    patience = 50
    stagnation_count = 0
    
    # Gradient-based optimization with restarts
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample shifts
        shifts = np.random.choice(n_points, size=n_samples, replace=False)
        
        # Find shift with maximum overlap
        max_overlap = -1
        best_shift = 0
        for k in shifts:
            # Compute overlap for shift k
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlap *= (2.0 / n_points)
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = k
        
        # Compute gradient for this shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_shift) % n_points
            grad[i] += (1 - h[j]) * (2.0 / n_points)
            grad[j] -= h[i] * (2.0 / n_points)
        
        # Adaptive line search for step size
        alpha = 1.0
        step_size = 0.1
        while alpha > 1e-8:
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_c5 = evaluate_erdos_solution(h_new, 2.0 / n_points)
            if new_c5 < best_c5 or alpha < 1e-6:
                h = h_new
                break
            alpha *= 0.5
        
        c5 = evaluate_erdos_solution(h, 2.0 / n_points)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Restart with perturbation if stuck
        if stagnation_count > patience:
            h = np.clip(initial_h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0
            
        if best_c5 < 0.3807:
            break
    
    # Final refinement with optimization
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
            
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
