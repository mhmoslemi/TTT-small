import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a triangular-like initial function
        x = np.linspace(0, 2, 35)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start / np.sum(h_start) * 17.5  # scale to sum to 17.5 for n=35
        n_points = 35
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach with simulated annealing
    scales = [35, 50, 75, 100]
    
    for res in scales:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to current resolution if needed
        if res > n_points:
            h_current = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), best_h)
            n_points = res
            dx = 2.0 / n_points
        else:
            h_current = best_h
            n_points = res
            dx = 2.0 / n_points
        
        # Simulated Annealing parameters
        T = 0.1  # Initial temperature
        alpha = 0.99  # Cooling rate
        max_iter = 2000  # Iterations per temperature
        T_min = 1e-6
        
        h_temp = h_current.copy()
        c5_temp = evaluate_erdos_solution(h_temp, dx)
        
        # SA loop
        for iter_count in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturbation: randomly change some values
            h_new = h_temp.copy()
            # Flip about 10% of values
            flip_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            for i in flip_indices:
                # Alternate between using random value and local adjustment
                if np.random.random() < 0.5:
                    # Random small perturbation
                    pert = np.random.normal(0, 0.05)
                    h_new[i] += pert
                else:
                    # Adjustment towards center of mass
                    h_new[i] = 1 - h_new[i]
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate new
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept with probability
            if c5_new < c5_temp or np.random.random() < math.exp(-(c5_new - c5_temp) / T):
                h_temp = h_new
                c5_temp = c5_new
            
            # Cool temperature
            T = max(T_min, alpha * T)
            
            # Update best
            if c5_temp < best_c5:
                best_h = h_temp.copy()
                best_c5 = c5_temp
                
    return (list(best_h), best_c5, n_points)
