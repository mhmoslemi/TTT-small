import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    max_iter = 5000
    mu = 1e-3  # penalty parameter
    tolerance = 1e-6
    rho = 1.0  # ADMM rho parameter
    
    # Initialize variables
    h = h_start.copy()
    h_prev = h.copy()
    z = np.zeros(2 * n_points - 1)  # auxiliary variable for overlap constraints
    lambda_ = np.zeros(2 * n_points - 1)  # dual variables
    
    # Precompute index mappings
    # The cross-correlation indices to match the formulation
    
    # ADMM Iterations
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Update h by solving convex subproblem
        # minimize ||h - h_prev||^2 + mu*||z - (h*(1-h))||^2
        # s.t. 0 <= h <= 1, sum(h) = 1
        
        # Form the effective constraint matrix for sum(h) = 1
        # Since it's linear, we can handle with convex optimization
        
        # Approximate the dual problem with SLSQP
        def constrained_obj(h):
            # objective includes penalty term for overlap constraints
            # and L2 regularizer for stability
            return evaluate_erdos_solution(h, dx) + 0.5 * mu * np.sum((h - h_prev)**2)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(constrained_obj, h, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_new = res.x
                # Ensure normalization
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            else:
                h = h_prev.copy()
        except:
            h = h_prev.copy()
        
        # Update z = prox_{1/mu}(h*(1-h) + lambda/mu)

        h1_h = np.correlate(h, 1-h, mode='full')
        z_new = h1_h + lambda_ / mu
        z_new = np.clip(z_new, 0, 1)  # Project to feasible set (approximate)
        z = z_new
        
        # Update dual variable
        lambda_ = lambda_ + mu * (h1_h - z)
        
        # Update h_prev
        h_prev = h.copy()
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Convergence check (every 50 iterations)
        if iteration % 50 == 0:
            # Simple convergence estimate via change in C5
            if iteration > 0 and abs(current_c5 - best_c5) < tolerance * 10:

        if best_c5 < 0.3807:

            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0

        bounds = [(0, 1) for _ in range(n_points)]

            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
