import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    num_particles = 40
    max_iter = 500
    inertia_weight = 0.7
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.2
    warm_restart_after = 200
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    for _ in range(num_particles):
        if 'initial_h_values' in globals():
            h = h_start.copy()
            # Add small noise to break symmetry
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
        else:
            h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_fitness.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_fitness)
    gbest = pbest[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    # Iteration counter
    iteration = 0
    stagnation_count = 0
    
    while time() - start_time < max_time * 0.95 and iteration < max_iter:
        iteration += 1
        
        if gbest_fitness < best_c5:
            best_c5 = gbest_fitness
            best_h = gbest.copy()
            
        # Check stagnation
        if iteration > 1 and gbest_fitness >= np.min(pbest_fitness) * 0.9999:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        # Warm restart
        if stagnation_count > warm_restart_after:
            velocities = [np.zeros(n_points) for _ in range(num_particles)]
            for i in range(num_particles // 2):
                particles[i] = np.random.random(n_points)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = evaluate_erdos_solution(particles[i], dx)
            gbest_idx = np.argmin(pbest_fitness)
            gbest = pbest[gbest_idx].copy()
            gbest_fitness = pbest_fitness[gbest_idx]
            stagnation_count = 0
        
        # Update particles
        for i in range(num_particles):
            r1, r2 = np.random.random(2)
            vel = velocities[i]
            pos = particles[i]
            personal_best = pbest[i]
            
            # Update velocity
            vel = (inertia_weight * vel + 
                   c1 * r1 * (personal_best - pos) + 
                   c2 * r2 * (gbest - pos))
            
            # Clip velocity
            vel = np.clip(vel, -max_velocity, max_velocity)
            
            # Update position
            pos = pos + vel
            
            # Enforce constraints
            pos = np.clip(pos, 0, 1)
            sum_h = np.sum(pos)
            if sum_h > 1e-10:
                pos = pos * (n_points / 2) / sum_h
            
            # Update particle
            particles[i] = pos
            
            # Compute fitness
            fitness = evaluate_erdos_solution(pos, dx)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = pos.copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest = pos.copy()
                    gbest_fitness = fitness
    
    # Local refinement of best solution
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
