import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        # Convert to binary: round to nearest 0 or 1, adjust sum to n/2
        h_binary = np.round(h_start).astype(int)
        current_sum = np.sum(h_binary)
        target_sum = n_points // 2
        
        if current_sum != target_sum:
            # Adjust by flipping bits
            if current_sum > target_sum:
                # Need to flip some 1s to 0s
                ones_indices = np.where(h_binary == 1)[0]
                for i in range(current_sum - target_sum):
                    idx = random.choice(ones_indices)
                    h_binary[idx] = 0
                    ones_indices = np.delete(ones_indices, np.where(ones_indices == idx)[0])
            else:
                # Need to flip some 0s to 1s
                zeros_indices = np.where(h_binary == 0)[0]
                for i in range(target_sum - current_sum):
                    idx = random.choice(zeros_indices)
                    h_binary[idx] = 1
                    zeros_indices = np.delete(zeros_indices, np.where(zeros_indices == idx)[0])
    else:
        n_points = 35
        h_binary = np.zeros(n_points, dtype=int)
        # Choose n/2 random positions to be 1
        ones_indices = random.sample(range(n_points), n_points // 2)
        h_binary[ones_indices] = 1
    
    dx = 2.0 / n_points
    best_h = h_binary.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Tabu search parameters
    max_no_improve = 100
    max_tabu_tenure = 10
    tabu_list = []
    no_improve_count = 0
    max_shake_iter = 50
    
    # Generate initial neighbors and find best
    def get_neighbors(h):
        neighbors = []
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in ones:
            for j in zeros:
                # Create neighbor by flipping one 1 to 0 and one 0 to 1
                h_neighbor = h.copy()
                h_neighbor[i] = 0
                h_neighbor[j] = 1
                neighbors.append((h_neighbor, (i, j)))
        return neighbors
    
    # Main Tabu Search Loop
    iter_count = 0
    while time() - start_time < budget_s * 0.95:
        iter_count += 1
        
        neighbors = get_neighbors(best_h)
        if not neighbors:
            break
            
        # Evaluate all neighbors and find the best valid one
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        best_move = None
        
        for h_n, move in neighbors:
            # Check if move is tabu
            is_tabu = move in tabu_list or (move[1], move[0]) in tabu_list
            
            if not is_tabu or evaluate_erdos_solution(h_n, dx) < best_c5 * 0.95:
                c5 = evaluate_erdos_solution(h_n, dx)
                if c5 < best_neighbor_c5:
                    best_neighbor_c5 = c5
                    best_neighbor = h_n
                    best_move = move
        
        if best_neighbor is None:
            # If no good neighbor, perform shaking
            if no_improve_count > max_shake_iter:
                h_shake = best_h.copy()
                num_flip = max(1, n_points // 20)
                flip_indices = random.sample(range(n_points), num_flip)
                for i in flip_indices:
                    h_shake[i] = 1 - h_shake[i]
                best_neighbor = h_shake
                best_neighbor_c5 = evaluate_erdos_solution(best_neighbor, dx)
                no_improve_count = 0
            else:
                no_improve_count += 1
                continue
                
        # Update best solution if improved
        if best_neighbor_c5 < best_c5:
            best_h = best_neighbor
            best_c5 = best_neighbor_c5
            no_improve_count = 0
            tabu_list = []
        else:
            no_improve_count += 1
            
        # Update tabu list (only keep last max_tabu_tenure moves)
        if best_move:
            tabu_list.append(best_move)
            if len(tabu_list) > max_tabu_tenure:
                tabu_list.pop(0)
        
        # Check for termination
        if no_improve_count >= max_no_improve:
            break
    
    # Final validation and cleanup
    if not np.all((best_h == 0) | (best_h == 1)):
        print("Warning: Non-binary solution found")
        # Convert to binary if needed
        best_h = np.round(best_h).astype(int)
        
    actual_sum = np.sum(best_h)
    target_sum = n_points // 2
    
    if actual_sum != target_sum:
        # Adjust to make sum correct
        if actual_sum > target_sum:
            ones_indices = np.where(best_h == 1)[0]
            for _ in range(actual_sum - target_sum):
                idx = random.choice(ones_indices)
                best_h[idx] = 0
                ones_indices = np.where(best_h == 1)[0]
        else:
            zeros_indices = np.where(best_h == 0)[0]
            for _ in range(target_sum - actual_sum):
                idx = random.choice(zeros_indices)
                best_h[idx] = 1
                zeros_indices = np.where(best_h == 0)[0]
                
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
