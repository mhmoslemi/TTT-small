import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create symmetric triangular function
        mid = n_points // 2
        h_start = np.minimum(np.arange(n_points), 2*mid - np.arange(n_points))
        h_start = h_start / (mid)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    max_iter = 200
    patience = 100
    improvement_threshold = 1e-6
    
    # Initialize
    h_current = h_start.copy()
    patience_counter = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Use convex optimization (approximately) to get a better point
        # We'll do a local optimization with constraints
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # SLSQP minimize
        try:
            res = minimize(objective, h_current, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_candidate = res.x
                c5_candidate = objective(h_candidate)
                
                # Check if we improve
                if c5_candidate < best_c5 - improvement_threshold:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                    last_improvement = iteration
                    patience_counter = 0
                else:
                    patience_counter += 1
        
        except:
            pass
            
        # Regular gradient descent step (L-BFGS-B) to fine-tune solution
        def obj_and_grad(h):
            # Gradient using finite differences
            eps = 1e-6
            base_val = objective(h)
            
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                
                grad[i] = (objective(h_plus) - base_val) / eps
            
            return base_val, grad
        
        try:
            # L-BFGS-B 
            res = minimize(objective, h_current, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_new = res.x
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                c5_new = objective(h_new)
                
                if c5_new < best_c5 - improvement_threshold:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    last_improvement = iteration
                    patience_counter = 0
                else:
                    patience_counter += 1
                    
        except:
            pass
            
        # Simple local search perturbation for escaping local minimum
        if patience_counter > 50 or iteration % 10 == 0:
            h_perturbed = h_current + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            c5_perturbed = objective(h_perturbed)
            
            if c5_perturbed < best_c5 - improvement_threshold:
                best_h = h_perturbed.copy()
                best_c5 = c5_perturbed
                last_improvement = iteration
                patience_counter = 0
            elif c5_perturbed < evaluate_erdos_solution(h_current, dx) - improvement_threshold:
                h_current = h_perturbed
                
        # Update h_current 
        h_current = best_h.copy()
        
        # Early exit if patience exceeded
        if patience_counter > patience or (iteration - last_improvement) > patience:
            break
    
    # Final check with L-BFGS if we have time
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        try:
            res = minimize(evaluate_erdos_solution, best_h, args=(dx,), method='L-BFGS-B',
                          bounds=[(0, 1) for _ in range(n_points)],
                          options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                h_final = np.clip(h_final, 0, 1)
                h_final = h_final * (n_points/2) / np.sum(h_final)
                c5_final = evaluate_erdos_solution(h_final, dx)
                
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
