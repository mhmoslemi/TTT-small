import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize Lagrange multipliers
    lambda_init = 0.1 * np.ones(2 * n_points - 1)
    mu_init = 1.0
    rho = 1e3
    
    # Convex optimization with Lagrangian relaxation
    max_iter = 500
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Build the constraint matrix A_k for each shift
        A_k_list = []
        for k in range(1 - n_points, n_points):
            A_k = np.zeros((n_points, n_points))
            if k >= 0:
                A_k[:-k, k:] = 1
            else:
                A_k[-k:, :k] = 1
            A_k_list.append(A_k)
        
        # Create convex optimization problem
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(h_var, 1 - h_var)))
        
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add Lagrangian penalty to objective
        lagrangian_expr = 0
        # For each overlap constraint, compute A_k * h * (1-h)
        for i, A_k in enumerate(A_k_list):
            overlap = cp.sum(cp.multiply(h_var, A_k @ (1 - h_var)))
            lagrangian_expr += lambda_init[i] * (overlap - best_c5 * dx)
        objective = cp.Minimize(cp.sum(cp.multiply(h_var, 1 - h_var)) + lagrangian_expr)
        
        # Solve convex problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.OSQP, warm_start=True, max_iter=1000)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_opt = h_var.value
            else:
                # Fallback to projected gradient if optimization fails
                h_opt = best_h
        except:
            h_opt = best_h
        
        # Update Lagrange multipliers
        new_lambda = lambda_init.copy()
        new_mu = mu_init
        
        for i, A_k in enumerate(A_k_list):
            overlap_val = np.dot(h_opt, A_k @ (1 - h_opt))
            new_lambda[i] = max(0, lambda_init[i] + rho * (overlap_val - best_c5 * dx))
            
        new_mu = max(0, mu_init + rho * (np.sum(h_opt) - n_points / 2))
        
        lambda_init = new_lambda
        mu_init = new_mu
        
        # Project h to feasible set
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        
        # Compute new max overlap
        c5_new = evaluate_erdos_solution(h_opt, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_opt.copy()
        
        # Adjust penalty weight based on constraint violations
        if np.abs(np.sum(h_opt) - n_points / 2) > 1e-4:
            rho *= 2
        else:
            rho *= 0.5 if rho > 1e-3 else 0.9
            
        # Early stopping
        if best_c5 < 0.3807:
            break
            
    # Final local refinement using SLSQP
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
