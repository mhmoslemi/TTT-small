import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a symmetric triangular function for initialization
        mid = n_points // 2
        h_start = np.minimum(np.arange(n_points), 2*mid - np.arange(n_points))
        h_start = h_start / (mid)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Local optimization using L-BFGS-B
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Optimization parameters
    max_iter = 500
    patience = 50
    stagnation_counter = 0
    current_best = best_c5
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Prepare optimization
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Optimization step
        try:
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9})
            if res.success:
                h_opt = res.x
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
                    if best_c5 < 0.3807:
                        break
        except Exception:
            pass
        
        # Update with gradient descent and retry if needed
        h_new = best_h.copy()
        for i in range(n_points):
            h_test = h_new.copy()
            h_test[i] += 0.01
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            if evaluate_erdos_solution(h_test, dx) < evaluate_erdos_solution(h_new, dx):
                h_new = h_test
        
        # Apply the update
        best_h = h_new
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
        # Check for stagnation
        if best_c5 < current_best:
            current_best = best_c5
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        
        if stagnation_counter > patience:
            # Add some random perturbation to escape local minima
            noise = np.random.normal(0, 0.02, n_points)
            best_h = np.clip(best_h + noise, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            current_best = evaluate_erdos_solution(best_h, dx)
            stagnation_counter = 0
    
    # Final local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
