import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    def initialize_population(size, n):
        if 'initial_h_values' in globals():
            pop = [h_start.copy()]
            pop.extend([np.random.rand(n) for _ in range(size-1)])
        else:
            pop = [np.random.rand(n) for _ in range(size)]
        return pop
    
    def normalize(h, target_sum):
        return h * target_sum / np.sum(h)
    
    def tournament_selection(pop, fitnesses, k):
        indices = np.random.choice(len(pop), k)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        return pop[best_idx]
    
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = random.randint(1, len(parent1) - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(h, rate):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < rate:
                h[i] += np.random.normal(0, 0.02)
        return np.clip(h, 0, 1)
    
    # Main GA loop
    population = initialize_population(population_size, n_points)
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = []
        for ind in population:
            norm_ind = normalize(ind, n_points / 2)
            fitness = evaluate_erdos_solution(norm_ind, dx)
            fitnesses.append(fitness)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitnesses[min_fitness_idx]
        
        # Selection, crossover, and mutation
        new_population = []
        elites = [population[i] for i in np.argsort(fitnesses)[:elite_size]]
        new_population.extend(elites)
        
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
    
    # Multi-resolution refinement
    if time() - start_time < budget_s * 0.90:
        # Increase resolution if needed
        if best_c5 > 0.3808:
            n_points = 40
            dx = 2.0 / n_points
            # Interpolate best solution to higher resolution
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, len(best_h)), 
                               best_h)
            best_h = h_interp
            best_h = normalize(best_h, n_points / 2)
            best_c5 = evaluate_erdos_solution(best_h, dx)
            
            # Further local optimization
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = normalize(h_restart, n_points / 2)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 200, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final refinement
    try:
        h_final = normalize(best_h, n_points / 2)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
