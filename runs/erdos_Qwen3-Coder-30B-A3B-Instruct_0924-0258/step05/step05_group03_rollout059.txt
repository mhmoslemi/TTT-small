import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the given construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
        
    # Precompute dx
    dx = 2.0 / n_points
    
    # Sample shifts to add constraints
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts)
    
    # Initialize solution and bounds
    h = initial_h.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Convex solver parameters
    max_iter = 50
    iter_count = 0
    stagnation = 0
    
    while time() - start_time < budget_s * 0.9 and iter_count < max_iter:
        # Evaluate overlaps for all shifts
        overlaps = []
        for shift in shifts:
            # Compute convolution with offset
            shift_idx = int(shift / 2 * n_points)
            if shift_idx >= n_points:
                shift_idx = n_points - 1
            
            # For simplicity, compute overlap manually
            # Compute integral: int h(x)(1 - h(x+k)) dx
            overlap = 0.0
            for i in range(n_points):
                j = i + shift_idx
                if j < n_points:
                    overlap += h[i] * (1 - h[j]) * dx
                else:
                    break
            overlaps.append(overlap)
        
        # Add constraint from the maximum overlap
        max_overlap = max(overlaps)
        max_idx = np.argmax(overlaps)
        
        # For next iteration, we try to minimize this constraint
        # But this is challenging to represent in a convex manner directly
        # Instead, we will do local optimization with the current best
        
        # Local optimization: SLSQP from current solution
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Small optimization step
        try:
            res = minimize(obj, h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-8})
            if res.success and res.fun < best_c5:
                h = res.x
                best_c5 = res.fun
                best_h = h.copy()
                stagnation = 0
            else:
                stagnation += 1
        except Exception:
            stagnation += 1
            
        # Ensure constraint satisfaction
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Early termination if target achieved
        if best_c5 < 0.3807:
            break
            
        iter_count += 1
        
        # Occasionally tighten the solution using more direct optimization
        if iter_count % 10 == 0:
            # Emphasis on current best solution
            try:
                # Run again with smaller step if improvement
                res = minimize(obj, h, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 50, 'ftol': 1e-9})
                if res.success and res.fun < best_c5:
                    h = res.x
                    best_c5 = res.fun
                    best_h = h.copy()
            except:
                pass

    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)

            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP',

    best_h = np.clip(best_h, 0, 1)

    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
