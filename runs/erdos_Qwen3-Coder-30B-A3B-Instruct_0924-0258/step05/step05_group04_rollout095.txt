import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Precompute correlation matrices
    def make_correlation_matrix(n, k):
        # Create matrix A_k where A_k[i,j] = 1 if |i-j| = k, 0 otherwise
        A = np.zeros((n, n))
        for i in range(n):
            j = i + k
            if j < n:
                A[i,j] = 1
        return A
    
    # Generate matrix for all shifts
    A_matrices = []
    for k in range(1, n_points):
        A_k = make_correlation_matrix(n_points, k)
        A_matrices.append(A_k)
    
    # Lagrangian-based approach
    max_iter = 500
    rho = 1e3
    lambda_vals = np.ones(2*n_points - 1) * 0.1  # For all overlap constraints
    mu = 1.0  # For integral constraint
    
    h = h_start.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Compute current overlaps
        overlaps = []
        for A_k in A_matrices:
            overlap = np.sum(h @ A_k @ (1 - h))
            overlaps.append(overlap)
            
        current_max_overlap = max(overlaps) * dx
        
        # Set up convex problem (strictly upper bound via Lagrangian)
        h_var = cp.Variable(n_points)
        
        # Constraint on integral
        integral_constraint = cp.sum(h_var) == n_points / 2
        
        # Constraint on bounds
        bound_constraints = [h_var >= 0, h_var <= 1]
        
        # Objective: (sum of overlap terms with multipliers) + penalty term
        # But more accurately minimize sum(1-h) * A_k * h + penalize violations
        objective = cp.Minimize(
            cp.sum(cp.multiply(lambda_vals[:len(overlaps)], 
                              [cp.sum(h_var.T @ A_matrices[i] @ (1 - h_var)) 
                               for i in range(len(overlaps))]))
            + mu * (cp.sum(h_var) - n_points / 2)**2
        )
        
        # Construct and solve problem
        prob = cp.Problem(objective, bound_constraints + [integral_constraint])
        
        try:
            prob.solve(solver=cp.OSQP, verbose=False, max_iter=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
            else:
                h_new = h.copy()
        except:
            h_new = h.copy()
            
        # Update Lagrange multipliers (active set)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new overlaps
        new_overlaps = []
        for i, A_k in enumerate(A_matrices):
            overlap = np.sum(h_new @ A_k @ (1 - h_new))
            new_overlaps.append(overlap)
            
        new_max_overlap = max(new_overlaps) * dx
        
        # Update lambda values
        for i in range(len(lambda_vals)):
            if i < len(new_overlaps):
                delta = new_overlaps[i] - current_max_overlap * dx
                lambda_vals[i] = max(0, lambda_vals[i] + rho * delta)
            else:
                lambda_vals[i] = max(0, lambda_vals[i] - rho * 0.1)
                
        # Update mu
        integral_error = np.sum(h_new) - n_points / 2
        mu = max(0, mu + rho * integral_error)
        
        # Adjust rho based on constraint violations
        if abs(integral_error) > 1e-4:
            rho *= 2
        else:
            rho *= 0.5
            
        h = h_new
        
        # Update best solution
        if new_max_overlap < best_c5:
            best_c5 = new_max_overlap
            best_h = h.copy()
        
        # Early stopping
        if best_c5 < 0.3807:
            break
    
    # Final local refinement
    remaining_time = max_time - (time() - start_time)
    
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def con_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons = {'type': 'eq', 'fun': con_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=cons, 
                         options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                if evaluate_erdos_solution(h_final, dx) < best_c5:
                    best_h = h_final
                    best_c5 = evaluate_erdos_solution(h_final, dx)
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
