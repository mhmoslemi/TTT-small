import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a focused optimization approach based on discrete choices and refined local search
    def neighbor_perturb(h, scale=0.05):
        h_new = h + np.random.normal(0, scale, len(h))
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        return h_new
    
    # Gradient-guided local search with adaptive learning rate
    def adaptive_local_search(start_h):
        h_current = start_h.copy()
        current_c5 = evaluate_erdos_solution(h_current, dx)
        best_h_local = h_current.copy()
        best_c5_local = current_c5
        
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        # Gradient estimation
        def gradient_approx(h, eps=1e-4):
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                c_plus = evaluate_erdos_solution(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = h_minus * (n_points/2) / np.sum(h_minus)
                c_minus = evaluate_erdos_solution(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            return grad

            if new_c5 < current_c5:
                h_current = h_new
                current_c5 = new_c5
                if new_c5 < best_c5_local:
                    best_h_local = h_current.copy()
                    best_c5_local = new_c5
                patience = 0
                learning_rate *= 1.05
            else:
                patience += 1
                learning_rate *= 0.9
                if patience > max_patience:
                    break
        
        return best_h_local, best_c5_local

    max_restarts = 100
    restart_count = 0

        h_perturbed = neighbor_perturb(best_h, 0.05)

                best_c5 = c5_refined

                restart_count += 1

        if restart_count > 10:

                    best_h = h_refined
                    best_c5 = c5_refined
            except Exception:
                pass

            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
        
    # Ensure that it meets sufficient constraints
    final_h = best_h.copy()

    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
