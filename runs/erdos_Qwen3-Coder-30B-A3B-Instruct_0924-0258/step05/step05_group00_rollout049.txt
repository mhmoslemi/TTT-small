import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution alternating optimization
    resolutions = [35, 40, 50, 60]
    last_improvement = 0
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolation to new resolution
        if res > n_points:
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            h_interp = np.interp(x_new, x_old, h_start)
        else:
            h_interp = h_start[:res].copy()
            if res < n_points:
                h_interp = np.pad(h_interp, (0, n_points - res), 'constant', constant_values=0)
            else:
                h_interp = h_interp[:n_points]
        
        n_points = len(h_interp)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_interp * (n_points/2) / np.sum(h_interp)
        
        # Convex relaxation and local refinement
        max_iter = 500
        iter_count = 0
        
        while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
            # Solve convex relaxation
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # consider only a subset of correlations for efficiency
                corrs = []
                for k in range(1, n_points - 1, max(1, n_points//10)):
                    if k < n_points // 2:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                    corrs.append(corr_sum)
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, abstol=1e-6, reltol=1e-6, feastol=1e-6)
                
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    
                    # Local refinement with SLSQP
                    def objective(h):
                        return evaluate_erdos_solution(h, dx)
                    
                    def constraint_sum(h):
                        return np.sum(h) - n_points / 2.0
                    
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    bounds = [(0, 1) for _ in range(n_points)]
                    res_slsqp = minimize(objective, h_feasible, method='SLSQP',
                                         bounds=bounds, constraints=cons_sum,
                                         options={'maxiter': 50, 'ftol': 1e-8})
                    
                    if res_slsqp.success:
                        h_refined = res_slsqp.x
                        c5_refined = objective(h_refined)
                        
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                            last_improvement = iter_count
                        else:
                            # Random perturbation to escape local minimum
                            h_pert = h_feasible + np.random.normal(0, 0.005, n_points)
                            h_pert = np.clip(h_pert, 0, 1)
                            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
                            c5_pert = objective(h_pert)
                            if c5_pert < best_c5:
                                best_h = h_pert
                                best_c5 = c5_pert
                                last_improvement = iter_count
                        
                else:
                    # Just perturb from current best_h
                    h_pert = best_h + np.random.normal(0, 0.005, n_points)
                    h_pert = np.clip(h_pert, 0, 1)
                    h_pert = h_pert * (n_points/2) / np.sum(h_pert)
                    c5_pert = objective(h_pert)
                    if c5_pert < best_c5:
                        best_h = h_pert
                        best_c5 = c5_pert
                        last_improvement = iter_count
                        
            except Exception as e:
                # Fallback to simple perturbation
                h_pert = best_h + np.random.normal(0, 0.005, n_points)
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points/2) / np.sum(h_pert)
                c5_pert = objective(h_pert)
                if c5_pert < best_c5:
                    best_h = h_pert
                    best_c5 = c5_pert
                    last_improvement = iter_count
            
            iter_count += 1
            
            if iter_count - last_improvement > 50:
                break

    return (list(best_h), best_c5, n_points)
