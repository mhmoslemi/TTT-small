import numpy as np
from scipy.signal import correlate
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def construct_block_h(n_points, M, on_blocks):
    """Construct h with M blocks, exactly on_blocks of them turned on."""
    block_size = n_points // M
    h = np.zeros(n_points)
    
    # Select which blocks to turn on
    on_block_indices = set(random.sample(range(M), min(on_blocks, M)))
    
    for i in range(M):
        start_idx = i * block_size
        end_idx = start_idx + block_size
        if i in on_block_indices:
            h[start_idx:end_idx] = 1.0
    
    # Ensure integral constraint
    current_sum = np.sum(h)
    if current_sum != n_points / 2:
        # Adjust to match exactly
        diff = int(n_points / 2 - current_sum)
        if diff > 0:
            # Turn on some blocks
            for i in range(M):
                if i not in on_block_indices and diff > 0:
                    start_idx = i * block_size
                    end_idx = start_idx + block_size
                    h[start_idx:end_idx] = 1.0
                    diff -= 1
        elif diff < 0:
            # Turn off some blocks
            off_block_indices = [i for i in range(M) if i in on_block_indices]
            for i in off_block_indices[:abs(diff)]:
                start_idx = i * block_size
                end_idx = start_idx + block_size
                h[start_idx:end_idx] = 0.0
    
    return h

    max_iter = 2000
