import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    num_particles = 40
    max_iter = 500
    inertia_weight = 0.7
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.2
    warm_restart_after = 200
    stagnation_count = 0
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    for i in range(num_particles):
        if i < len(h_start):
            h = h_start.copy()
        else:
            h = np.random.random(n_points)
            h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_fitness.append(evaluate_erdos_solution(h, dx))
    
    # Find global best
    best_idx = np.argmin(pbest_fitness)
    gbest = pbest[best_idx].copy()
    gbest_fitness = pbest_fitness[best_idx]
    
    if gbest_fitness < best_c5:
        best_c5 = gbest_fitness
        best_h = gbest.copy()
        
    # PSO iterations
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Check for stagnation
        if iteration > 0 and pbest_fitness[best_idx] >= gbest_fitness * 0.9999:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        # Warm restart
        if stagnation_count > warm_restart_after:
            velocities = [np.zeros(n_points) for _ in range(num_particles)]
            # Perturb half the particles
            for i in range(num_particles // 2):
                if i < len(particles):
                    idx = random.randint(0, num_particles - 1)
                    particles[idx] = np.random.random(n_points)
                    particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                    pbest[idx] = particles[idx].copy()
                    pbest_fitness[idx] = evaluate_erdos_solution(particles[idx], dx)
            stagnation_count = 0
            # Update gbest
            best_idx = np.argmin(pbest_fitness)
            gbest = pbest[best_idx].copy()
            gbest_fitness = pbest_fitness[best_idx]
            if gbest_fitness < best_c5:
                best_c5 = gbest_fitness
                best_h = gbest.copy()
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h > 1e-10:
                particles[i] = particles[i] * (n_points / 2) / sum_h
            else:
                particles[i] = np.ones(n_points) * (n_points / 2)
                
            # Evaluate fitness
            current_fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = current_fitness
                
                # Update global best
                if current_fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = current_fitness
                    if gbest_fitness < best_c5:
                        best_c5 = gbest_fitness
                        best_h = gbest.copy()
                        
        # Early stopping
        if best_c5 < 0.3807:
            break
    
    # Local refinement of the best solution using SLSQP
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_init = best_h.copy()
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 300, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
