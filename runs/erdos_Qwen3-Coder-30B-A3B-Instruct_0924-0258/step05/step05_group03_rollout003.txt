import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)

    max_iterations = 50
    max_shifts = 100  # Number of shifts to sample for constraints
    
    # Precompute shifts
    shifts = np.linspace(0, 2, max_shifts, endpoint=False)
    
    # Build the cutting plane approach
    h = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints for normalization and bounds
    constraints = [
        cp.sum(h) == n_points / 2,
        h >= 0,
        h <= 1,
        C5 >= 0
    ]
    
    # Store best solution
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    # Cutting plane loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate overlap with current h for all shifts
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_shift_idx = np.argmax(corr)

        # Add constraint from the maximum overlap shift
        offset = max_shift_idx - (n_points - 1)

        constraint_expr = 0

        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)

                new_c5 = C5.value

                    if new_c5 < best_c5:

        except Exception as e:
            # Fall back to current solution
            pass
        
        # Remove the added constraint for next iteration to avoid a fixed issue
