import numpy as np
from scipy.optimize import minimize
from time import time
import hashlib

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_max_size = max(5, int(0.2 * n_points))
    max_stagnation = 100
    mutation_scale = 0.04
    max_iter = 1000
    refinement_interval = 10
    
    # Hash function for tabu list
    def hash_h(h):
        # Sort values for consistent hashing
        sorted_h = tuple(sorted(h))
        return hashlib.md5(str(sorted_h).encode()).hexdigest()
    
    # Generate new neighbor
    def generate_neighbor(h, scale):
        h_new = h.copy()
        # Randomly select ~30% of elements to perturb
        n_perturb = max(1, int(0.3 * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, scale)
        # Clip to bounds [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Rescale to meet constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        return h_new
    
    # Main loop
    stagnation_count = 0
    improvement_count = 0
    current_best = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate new candidate
        h_new = generate_neighbor(best_h, mutation_scale)
        hash_new = hash_h(h_new)
        
        # Check tabu list
        if hash_new in tabu_list:
            # Accept with low probability to escape local optima
            if np.random.random() > 0.7:
                pass
            else:
                continue
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update best if improved
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            improvement_count += 1
            stagnation_count = 0
            current_best = c5_new
            
            # Local refinement every few improvements
            if improvement_count % refinement_interval == 0:
                try:
                    def objective(h):
                        return evaluate_erdos_solution(h, dx)
                    
                    def constraint_sum(h):
                        return np.sum(h) - n_points / 2.0
                    
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    bounds = [(0, 1) for _ in range(n_points)]
                    
                    res = minimize(objective, best_h, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-8})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
            
            # Early termination if target reached
            if best_c5 < 0.3807:
                break
        else:
            stagnation_count += 1
            
        # Update tabu list 
        tabu_list.append(hash_new)
        if len(tabu_list) > tabu_max_size:
            tabu_list.pop(0)
        
        # Adapt mutation scale
        if stagnation_count > max_stagnation:
            mutation_scale = min(0.5, mutation_scale * 1.2)
            stagnation_count = 0
        else:
            mutation_scale = max(0.005, mutation_scale * 0.99)
    
    # Final optimization of best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
