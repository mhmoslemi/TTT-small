import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Smoothing parameter for smooth maximum
    tau = 1.0
    max_iter = 3000
    tol = 1e-7
    step_size = 1.0
    
    # Gradient-based optimization using L-BFGS-B
    def smooth_max_overlap(h, tau):
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        # Avoid numerical overflow by subtracting max
        corr = corr - np.max(corr)
        smoothed = np.log(np.sum(np.exp(tau * corr))) / tau
        return smoothed * dx
    
    def gradient_smooth_max_overlap(h, tau):
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        # Avoid numerical overflow
        corr = corr - np.max(corr)
        exp_corr = np.exp(tau * corr)
        sum_exp = np.sum(exp_corr)
        # Gradient computation for each point
        grad = np.zeros_like(h)
        n = len(h)
        # Compute gradient components
        # derivative of log(sum(exp(tau * corr))) wrt h[i]
        # Use chain rule: (1/sum_exp) * sum_exp * tau * d(corr)/d(h[i])
        for i in range(n):
            # Compute partial derivative of correlation with respect to h[i]
            # For each shift k, we compute the derivative of h[i] * (1 - h[i+k])
            # d/dh[i] (h[i] * (1 - h[i+k])) = (1 - h[i+k]) if i+k < n, 0 otherwise
            # and d/dh[i] (h[i] * (1 - h[i-k])) = (1 - h[i-k]) if i-k >= 0, 0 otherwise
            # All terms in full correlation correspond to different shifts
            # We compute gradient for each i, sum over all correlation coefficients
            # which are d/dh[i] h[i] * (1 - h[i+k]) = 1 - h[i+k] if there is a shift k
            # from i to i+k and 0 otherwise. For i+k out of range, we have 0
            # But since we compute full correlation, the indices are: (i, i+k)
            # with -n+1 <= k <= n-1, the shift k=0 means correlation with h[i] itself
            # So we consider this sum of (1-h[i+k]) weights.
            deriv = 0.0
            for k in range(-n+1, n):
                if 0 <= i+k < n:
                    deriv += (1 - h[i+k]) * exp_corr[k + n - 1]  # k+n-1 is the index in corr array
            grad[i] = (tau / sum_exp) * deriv * dx
        return grad
    
    def obj_and_grad(h):
        # Normalize the constraint
        h = h * (n_points / 2) / np.sum(h)
        value = smooth_max_overlap(h, tau)
        gradient = gradient_smooth_max_overlap(h, tau)
        return value, gradient
    
    # Optimization loop
    prev_cost = float('inf')
    stagnation = 0
    iteration = 0
    
    while time() - start_time < max_time * 0.95 and iteration < max_iter:
        iteration += 1
        
        # Get current constraint-satisfied values
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        
        try:
            # Use L-BFGS-B to update h, with bounds
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                lambda x: smooth_max_overlap(x, tau),
                h_current,
                method='L-BFGS-B',
                jac=lambda x: gradient_smooth_max_overlap(x, tau),
                bounds=bounds,
                options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-6},
                callback=lambda x: None
            )
            if res.success:
                h_current = res.x.copy()
            else:
                # Try with less strict objective
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points / 2) / np.sum(h_current)
            # Enforce normalization after update
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            h_current = np.clip(h_current, 0, 1)
            current_c5 = evaluate_erdos_solution(h_current, dx)
            
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
                stagnation = 0
            else:
                stagnation += 1
                
            # Adaptive adjustments
            if iteration % 100 == 0:
                tau *= 0.99
            
            # Adaptive convergence check
            if abs(current_c5 - prev_cost) < tol:
                stagnation += 1
            prev_cost = current_c5
            
            # Early termination condition
            if stagnation > 100 or current_c5 < 0.3807:
                break
                
        except Exception as e:
            # fallback to previous value if error occurs
            h_current = best_h.copy()
            continue
    
    # Final refinement with stricter tolerance
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_init = best_h.copy()
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        try:
            # Use SLSQP for final refinement with stricter tolerance
            def objective(h):
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_init, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 200, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                h_final = np.clip(h_final, 0, 1)
                h_final = h_final * (n_points / 2) / np.sum(h_final)
                final_c5 = objective(h_final)
                if final_c5 < best_c5:
                    best_h = h_final
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
