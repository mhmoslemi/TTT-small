import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid initial optimization
    n_low = 20
    h_coarse = np.ones(n_low) * 0.5
    dx_low = 2.0 / n_low
    
    try:
        h_relaxed = cp.Variable(n_low)
        constraints = [
            cp.sum(h_relaxed) == n_low / 2.0,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        # Add a subset of constraints to manage complexity
        for k in range(1, min(10, n_low)):  
            if k < n_low // 2:
                corr_sum = cp.sum(cp.multiply(h_relaxed[:n_low-k], 1 - h_relaxed[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_low-k]))
            constraints.append(corr_sum <= best_c5 * dx_low + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            h_coarse = h_relaxed.value
    except:
        pass
    
    # Upsample to high resolution
    h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, n_low), 
                            h_coarse)
    h_upsampled = np.clip(h_upsampled, 0, 1)
    h_upsampled = h_upsampled * (n_points/2) / np.sum(h_upsampled)
    h_start = h_upsampled
    
    # Fine grid optimization with adaptive mutation
    max_iter = 1000
    patience = 0
    mutation_patience = 0
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def gradient_descent_step(h, dx, epsilon=1e-6):
        # Compute approximate gradient using finite differences
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            val_plus = evaluate_erdos_solution(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            val_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (val_plus - val_minus) / (2 * epsilon)
        return grad
    
    def mutate_h(h, strength=0.05):
        # Mutation step to escape local optima
        mutated = h.copy()
        mask = np.random.rand(len(h)) < 0.1  # 10% chance to mutate each element
        mutation = np.random.normal(0, strength, len(h))
        mutated[mask] += mutation[mask]
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points/2) / np.sum(mutated)
        return mutated
    
    # Initial refinement
    h_current = h_start.copy()
    current_c5 = evaluate_erdos_solution(h_current, dx)
    if current_c5 < best_c5:
        best_h = h_current.copy()
        best_c5 = current_c5
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Gradient descent
        lr = 0.01
        try:
            grad = gradient_descent_step(h_current, dx)
            h_new = h_current - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            if new_c5 < best_c5:
                h_current = h_new
                best_c5 = new_c5
                patience = 0
                mutation_patience = 0
            else:
                patience += 1
                mutation_patience += 1
                
                # Apply mutation if stuck
                if mutation_patience > 50:
                    h_current = mutate_h(h_current, strength=0.05)
                    mutation_patience = 0
                    new_c5 = evaluate_erdos_solution(h_current, dx)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        
        except Exception as e:
            # Fall back to mutation if gradient computation fails
            h_current = mutate_h(h_current, strength=0.05)
            new_c5 = evaluate_erdos_solution(h_current, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
            patience += 1
                
        # Early stopping
        if patience > 100:
            break
            
        # Adaptive learning rate
        if patience > 50:
            lr *= 0.9
    
    # Finalize
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
