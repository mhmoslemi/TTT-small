import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Construct correlation matrices for each shift 
    def build_correlation_matrices(n):
        A = []
        for k in range(2 * n - 1):
            # Create correlation matrix for shift k
            A_k = np.zeros((n, n))
            if k < n:
                A_k[:n-k, k:] = 1
            else:
                A_k[k-n+1:, :n-(k-n+1)] = 1
            A.append(A_k)
        return A

    # Initialize multipliers and parameters
    max_iter = 500
    rho = 1e3
    lambda_init = 0.1
    
    # Build correlation matrices
    A = build_correlation_matrices(n_points)
    
    # Main loop
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Solve convex subproblem
        h = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(h, 1 - h)))
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Lagrangian relaxation approach
        # Using simpler approach: project and solve with SLSQP for better local refinement
        def obj(h_vals):
            h_vec = np.array(h_vals)
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return np.max(corr) * dx
            
        def con_sum(h_vals):
            return np.sum(h_vals) - n_points / 2
            
        cons = {'type': 'eq', 'fun': con_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Regular SLSQP optimization on the current solution
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = obj(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass
            
        # Local refinement with tighter tolerance
        if iteration % 10 == 0 and time() - start_time < max_time * 0.9:
            try:
                res = minimize(obj, best_h, method='SLSQP', bounds=bounds, 
                               constraints=cons, options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    h_new = res.x
                    c5_new = obj(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
            except:
                pass

        # Early stopping
        if best_c5 < 0.3807:
            break

    # Final refinement
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(evaluate_erdos_solution, best_h, 
                           args=(dx,), method='SLSQP', bounds=bounds, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass

    return (list(best_h), best_c5, n_points)
