import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a triangular function as starting point
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start / np.sum(h_start) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach with Simulated Annealing
    resolutions = [35, 50, 75]
    current_res = 0
    
    while time() - start_time < budget_s * 0.95 and current_res < len(resolutions):
        res = resolutions[current_res]
        if res > n_points:
            # Interpolate to higher resolution
            h_new = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), best_h)
            n_points = res
            dx = 2.0 / n_points
        else:
            h_new = best_h
            n_points = res
            dx = 2.0 / n_points
            
        # Simulated Annealing at current resolution
        h_current = h_new.copy()
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        # SA parameters
        T = 0.1
        alpha = 0.99
        max_iter = 3000
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate neighbor solution
            h_new = h_current.copy()
            # Perturb 10% of elements
            n_perturb = max(1, n_points // 10)
            indices = np.random.choice(n_points, n_perturb, replace=False)
            for i in indices:
                # Add small random change
                h_new[i] += np.random.normal(0, 0.03)
            
            # Ensure bounds and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Acceptance probability
            if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / T):
                h_current = h_new
                c5_current = c5_new
                
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
            
            # Cool down
            T = alpha * T
        
        current_res += 1
    
    # Final validation and refinement
    if time() - start_time < budget_s * 0.1:
        # Simple local refinement at finest resolution
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
