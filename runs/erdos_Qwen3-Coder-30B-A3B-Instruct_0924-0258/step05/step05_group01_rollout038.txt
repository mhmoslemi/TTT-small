import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 50
    max_gens = 1000
    mutation_rate = 0.15
    crossover_rate = 0.8
    tournament_size = 3
    min_res = 35
    max_res = 50
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def ensure_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def tournament_selection(pop, fitnesses):
        selected_idx = random.randint(0, len(pop)-1)
        for _ in range(tournament_size - 1):
            idx = random.randint(0, len(pop)-1)
            if fitnesses[idx] < fitnesses[selected_idx]:
                selected_idx = idx
        return pop[selected_idx]
    
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy()
        cut = random.randint(1, n_points-1)
        child = np.concatenate([parent1[:cut], parent2[cut:]])
        return child
    
    def mutate(individual):
        child = individual.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                child[i] += np.random.normal(0, 0.05)
        return ensure_feasible(child)
    
    def evaluate_population(pop):
        return [evaluate_erdos_solution(h, dx) for h in pop]
    
    # Main EA loop
    pop = [create_individual() for _ in range(pop_size)]
    current_res = min_res
    gen = 0
    
    while time() - start_time < budget_s * 0.95 and gen < max_gens:
        fitnesses = evaluate_population(pop)
        
        # Track best
        min_fitness = min(fitnesses)
        best_idx = fitnesses.index(min_fitness)
        if min_fitness < best_c5:
            best_h = pop[best_idx].copy()
            best_c5 = min_fitness
        
        # Selection, crossover, mutation
        new_pop = []
        for _ in range(pop_size):
            parent1 = tournament_selection(pop, fitnesses)
            parent2 = tournament_selection(pop, fitnesses)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_pop.append(child)
        
        pop = new_pop
        
        # Resolution increase
        if gen > 0 and gen % 100 == 0 and current_res < max_res:
            current_res = min(current_res + 5, max_res)
            dx = 2.0 / current_res
            
            # Interpolate population
            new_pop = []
            for h in pop:
                h_interp = np.interp(np.linspace(0, 2, current_res), np.linspace(0, 2, n_points), h)
                h_interp = ensure_feasible(h_interp)
                new_pop.append(h_interp)
            
            pop = new_pop
            n_points = current_res
        
        gen += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
