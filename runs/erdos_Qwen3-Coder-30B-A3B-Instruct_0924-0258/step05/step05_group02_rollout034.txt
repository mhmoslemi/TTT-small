from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    mutation_rate = 0.1
    crossover_alpha_range = (0.4, 0.6)
    
    # Initialize population
    population = []
    population_fitness = []

        h = h * (n_points/2) / np.sum(h)

        population.append(h)
        fitness = evaluate_erdos_solution(h, dx)
        population_fitness.append(fitness)

    patience = 0
    best_fitness_history = [min(population_fitness)]
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Sort population by fitness
        sorted_indices = np.argsort(population_fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [population_fitness[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitness[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = sorted_fitness[0]
            patience = 0

            for i in range(pop_size // 2, pop_size):

        elite = sorted_population[:elitism]

        new_population = elite.copy()
        new_fitness = elite_fitness.copy()

            parent1 = elite[np.random.randint(0, len(elite))]

            alpha = np.random.uniform(*crossover_alpha_range)

        best_fitness_history.append(min(population_fitness))

    final_fitness = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(final_fitness)
    best_h = population[best_idx]
    best_c5 = final_fitness[best_idx]
    
    # Ensure constraint satisfaction
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
