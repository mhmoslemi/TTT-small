import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create triangular shape as starting point
        x = np.linspace(0, 2, 35)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start / np.sum(h_start) * 17.5  # Scale to sum to 17.5 (35/2)
        n_points = 35
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach with Simulated Annealing
    scales = [35, 50, 75, 100]
    current_scale_idx = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale_idx < len(scales):
        scale = scales[current_scale_idx]
        if scale > n_points:
            # Interpolate to higher resolution
            x_fine = np.linspace(0, 2, scale)
            x_coarse = np.linspace(0, 2, n_points)
            h_new = np.interp(x_fine, x_coarse, best_h)
            n_points = scale
            dx = 2.0 / n_points
        else:
            h_new = best_h
            n_points = scale
            dx = 2.0 / n_points

        # Simulated Annealing
        T = 0.1
        alpha = 0.99
        min_T = 1e-6
        max_iter = 2000
        
        h_current = h_new.copy()
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        iter_count = 0
        while time() - start_time < budget_s * 0.95 and iter_count < max_iter and T > min_T:
            # Generate neighbor by randomly perturbing
            h_neighbor = h_current.copy()
            idx_flip = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            for i in idx_flip:
                h_neighbor[i] = 1 - h_neighbor[i]  # Flip between 0 and 1
            
            # Enforce constraints
            h_neighbor = np.clip(h_neighbor, 0, 1)
            h_neighbor = h_neighbor * (n_points / 2) / np.sum(h_neighbor)
            
            c5_neighbor = evaluate_erdos_solution(h_neighbor, dx)
            
            # Accept or reject
            if c5_neighbor < c5_current:
                h_current = h_neighbor
                c5_current = c5_neighbor
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
            else:
                # Simulated Annealing acceptance probability
                delta = c5_neighbor - c5_current
                if np.random.rand() < np.exp(-delta / T):
                    h_current = h_neighbor
                    c5_current = c5_neighbor
            
            T = alpha * T
            iter_count += 1
            
        current_scale_idx += 1
    
    # Final polishing with local optimization
    if time() - start_time < budget_s * 0.1:
        try:
            # Simple L-BFGS-like refinement
            from scipy.optimize import minimize
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
            res = minimize(lambda x: evaluate_erdos_solution(x, dx), 
                          best_h, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 100})
            if res.success:
                h_final = res.x
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
