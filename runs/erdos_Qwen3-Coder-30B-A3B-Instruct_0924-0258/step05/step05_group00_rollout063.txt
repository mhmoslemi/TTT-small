import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization approach
    resolutions = [35, 40, 50, 60]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Local search with random restarts
        max_restarts = 20
        restart_count = 0
        
        while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
            # Random perturbation
            h_restart = h_current + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            # SLSQP local optimization
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                h_guess = h_restart.copy()
                res = minimize(objective, h_guess, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-8})

                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        restart_count = 0
                else:
                    restart_count += 1
            except Exception as e:
                restart_count += 1
        
        h_start = best_h.copy()
    
    # For final refinement, use higher resolution if we have resources
    if time() - start_time < budget_s * 0.95:
        try:
            n_points = 70
            dx = 2.0 / n_points
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, len(best_h)), 
                               best_h)
            h_current = h_interp
            h_current = h_current * (n_points/2) / np.sum(h_current)
            
            # Optimization with trust-constr
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_current, method='trust-constr', 
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'gtol': 1e-8, 'xtol': 1e-8})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
