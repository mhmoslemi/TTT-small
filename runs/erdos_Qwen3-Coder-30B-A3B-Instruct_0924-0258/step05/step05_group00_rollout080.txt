import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm Parameters
    pop_size = 50
    num_generations = 1000
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    stagnation_limit = 50
    
    # Initialize population
    def initialize_population(size, n_points):
        if 'initial_h_values' in globals():
            population = [h_start.copy()]
            size -= 1
        else:
            population = []
        for _ in range(size):
            h = np.random.random(n_points)
            h = h * (n_points/2) / np.sum(h)
            population.append(h)
        return population
    
    population = initialize_population(pop_size, n_points)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fitness.append(evaluate_erdos_solution(h, dx))
    
    best_idx = np.argmin(fitness)
    if fitness[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    # GA Loop
    stagnation_count = 0
    for gen in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0])
        
        # Crossover and Mutation
        new_pop = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % pop_size] if i+1 < pop_size else selected[0]
            
            if random.random() < crossover_rate:
                # Single-point crossover
                point = random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:point], parent2[point:]])
                child2 = np.concatenate([parent2[:point], parent1[point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if random.random() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.02)
                if random.random() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.02)
            
            # Clamp and rescale
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_pop.extend([child1, child2])
        
        new_pop = new_pop[:pop_size]
        population = new_pop
        
        # Evaluate new population and find best
        new_fitness = []
        for h in population:
            new_fitness.append(evaluate_erdos_solution(h, dx))
        
        best_idx = np.argmin(new_fitness)
        if new_fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = new_fitness[best_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > stagnation_limit:
            break
    
    # Resolution refinement
    if time() - start_time < budget_s * 0.9 and best_c5 > 0.3808:
        n_points = 40
        dx = 2.0 / n_points
        # Interpolate best solution
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, len(best_h)), 
                           best_h)
        h_current = h_interp.copy()
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Local search with SLSQP
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_current, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    
    # Final refinement with SLSQP
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
