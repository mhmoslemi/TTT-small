import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Local search with pattern-based perturbations
    max_restarts = 100
    restart_count = 0
    
    def perturb_pattern(h, pattern_std=0.02):
        h_pert = h.copy()
        # Apply Gaussian noise with pattern-aware scaling
        noise = np.random.normal(0, pattern_std, n_points)
        h_pert += noise
        h_pert = np.clip(h_pert, 0, 1)
        # Re-scale to meet constraint
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        return h_pert
    
    # Synthesize a better pattern from dominant harmonics
    def synthetic_pattern(h):
        # Extract dominant frequencies via FFT
        fft_h = np.fft.fft(h)
        # Keep 1st and 2nd frequencies, zero others
        synth = np.zeros_like(fft_h)
        synth[0] = fft_h[0]  # DC component
        synth[1] = fft_h[1]  # First harmonic
        synth[-1] = fft_h[-1]  # Nyquist component if odd
        if len(fft_h) > 2:
            synth[2] = fft_h[2]
            synth[-2] = fft_h[-2]
        # Inverse FFT
        h_synth = np.fft.ifft(synth).real
        return np.clip(h_synth, 0, 1)
    
    while time() - start_time < budget_s * 0.95:
        # Perturb and attempt refinement
        h_perturbed = perturb_pattern(best_h)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Local optimization
        try:
            def obj(h_pert):
                return evaluate_erdos_solution(h_pert, dx)
            
            cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                obj,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-7}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
                else:
                    restart_count += 1
            else:
                restart_count += 1
        except Exception:
            restart_count += 1
        
        # Apply synth pattern once in a while to escape local minima
        if restart_count % 15 == 0:
            try:
                h_synthetic = synthetic_pattern(best_h)
                h_synthetic = h_synthetic * (n_points/2) / np.sum(h_synthetic)
                c5_synthetic = evaluate_erdos_solution(h_synthetic, dx)
                if c5_synthetic < best_c5:
                    best_h = h_synthetic
                    best_c5 = c5_synthetic
                    restart_count = 0
            except Exception:
                pass
        
        # Global restart
        if restart_count > max_restarts:
            h_restart = np.random.rand(n_points)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            h_restart = np.clip(h_restart, 0, 1)
            restart_count = 0
            
            # Update best if restart is better
            c5_restart = evaluate_erdos_solution(h_restart, dx)
            if c5_restart < best_c5:
                best_h = h_restart
                best_c5 = c5_restart
                
        if time() - start_time > budget_s * 0.9:
            break
    
    # Final refinement of best solution
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
