import numpy as np
from scipy.optimize import minimize
from time import time
import hashlib

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def hash_h(h):
    # Create a hashable representation of the solution
    return hashlib.md5(np.sort(h).tobytes()).hexdigest()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    max_iter = 1000
    tabu_tenure = min(50, max(5, n_points // 4))
    tabu_list = []
    max_stagnation = 100
    stagnation_counter = 0
    mutation_scale = 0.04
    
    # Initialize current solution
    current_h = h_start.copy()
    current_c5 = best_c5
    
    def perturb_solution(h, scale):
        # Create a perturbation
        h_new = h.copy()
        # Select 30% of indices to perturb
        indices = np.random.choice(n_points, size=int(0.3 * n_points), replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, scale)
        # Clip values
        h_new = np.clip(h_new, 0, 1)
        # Rescale to correct sum
        h_new = h_new * (n_points/2) / np.sum(h_new)
        return h_new
    
    # Main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate new candidate
        candidate_h = perturb_solution(current_h, mutation_scale)
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Check tabu list
        candidate_key = hash_h(candidate_h)
        is_tabu = candidate_key in tabu_list
        
        # Accept if better or not tabu and we have a good solution
        if candidate_c5 < current_c5 or not is_tabu:
            current_h = candidate_h
            current_c5 = candidate_c5
            
            # Update best if improved
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                if best_c5 < 0.3807:
                    break
                stagnation_counter = 0
            else:
                stagnation_counter += 1
                
            # Add to tabu list
            tabu_list.append(candidate_key)
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
                
        else:
            stagnation_counter += 1
        
        # Adjust mutation scale if stagnating
        if stagnation_counter > max_stagnation:
            mutation_scale = min(0.2, mutation_scale * 1.2)
            stagnation_counter = 0
        
        # Occasionally do local refinement
        if iteration % 10 == 0 and candidate_c5 < best_c5 * 1.01:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, current_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    refined_h = res.x
                    refined_c5 = objective(refined_h)
                    if refined_c5 < current_c5:
                        current_h = refined_h
                        current_c5 = refined_c5
                        if refined_c5 < best_c5:
                            best_h = refined_h.copy()
                            best_c5 = refined_c5
            except:
                pass
    
    # Final refinement with better control
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
