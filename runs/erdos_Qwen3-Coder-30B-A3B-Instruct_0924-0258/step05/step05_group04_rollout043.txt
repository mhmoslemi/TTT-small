import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Sequential Convex Programming approach
    max_iter = 500
    mu = 1e-3
    C5_nih = 0.4  # Initial guess for C5
    tol = 1e-6 # tolerance for convergence
    batch_size = 10 # processing constraints in batches
    
    # Variables
    h = Variable(n_points)
    C5 = Variable()
    
    # Objective: minimize C5
    objective = Minimize(C5)
    
    # Constraints for each shift
    constraints = []
    
    # Add the constraint that sum(h) = n_points/2
    constraints.append(sum(h) == n_points / 2)
    
    # Add bounds on h
    for i in range(n_points):
        constraints.append(h[i] >= 0)
        constraints.append(h[i] <= 1)
    
    # Add QCQP constraints
    for k in range(n_points):
        # Compute the shift
        shifted = np.concatenate([h_start[n_points-k:], h_start[:n_points-k]])
        # Add the constraint: sum(h_i * (1 - h_{i+k})) <= C5 * dx
        # Using linearization around current iterate
        left_side = sum(h[i] * (1 - shifted[i]) for i in range(n_points))
        constraints.append(left_side <= C5 * dx)
        
    # Add augmented Lagrangian terms as penalties
    # For now, focus on the basic QP solver approach
    
    # Main SCP loop
    for iteration in range(max_iter):
        if time.time() - start_time > max_time * 0.95:
            break
            
        # Evaluate current overlaps to determine if constraints are tight
        current_c5 = evaluate_erdos_solution(h_start, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_start.copy()
        
        # Stop early if target achieved
        if best_c5 < 0.3807:
            break
            
        # At each iteration, solve the convex approximation
        # Evaluate overlap integrals
        # Define constraints using current h_start
        
        # Create constraints for the batch
        batch_constraints = []
        temp_h = np.copy(h_start)
        
        # Linearize about current iterate
        for k in range(0, n_points, batch_size):
            end_idx = min(k + batch_size, n_points)
            for i in range(k, end_idx):
                # We're adding constraint sum(h_j * (1 - h_{j+i})) <= C5 * dx
                # This is non-convex; we linearize around current point
                # The linearization of h_i * (1 - h_{i+k}) around current points
                shifted = np.zeros(n_points)
                shifted[i:] = h_start[:-i]
                shifted[:i] = h_start[-i:]
                left_side = sum(h[j] * (1 - shifted[j]) for j in range(n_points))
                batch_constraints.append(left_side <= C5 * dx)
                
        # Combine with basic constraints
        full_constraints = constraints + batch_constraints
        
        # Formulate and solve convex QP
        try:
            prob = Problem(objective, full_constraints)
            prob.solve(solver=OSQP, verbose=False, max_iter=5000)
            
            # Check if it solved
            if prob.status in ["optimal", "optimal_inaccurate"]:
                # Get solution
                h_new = np.array([var.value for var in h]).flatten()
                C5_val = C5.value
                h_start = h_new
                # Update upper bound on C5
                if C5_val is not None:
                    C5_nih = max(C5_val, C5_nih)
        
        except:
            pass
            
        # Reform after 10 iters
        if iteration % 10 == 0:
            # Some heuristic update of mu or search strategy
            mu = min(1e3, mu * 1.1)
            
    # Final local refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                       constraints=cons_sum, options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
