import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        # Start with a coarse grid and upsample
        coarse_n = 20
        h_start = np.ones(coarse_n) * 0.5
        n_points = 35
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Ensure initial sum constraint
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    
    # Refinement loop
    max_iter = 100
    iter_count = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Get current solution
        current_h = best_h.copy()
        current_c5 = best_c5
        
        # Perform local optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Perturb slightly
        h_perturbed = current_h + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        try:
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if res.success:
                h_new = res.x
                c5_new = evaluate_erdos_solution(h_new, dx)

                    best_h = h_new

                    no_improvement_count = 0

        except Exception:
            no_improvement_count += 1
            
        # Check for convergence
        if no_improvement_count > 50:
            # Add small perturbation
            h_perturbed = best_h + np.random.normal(0, 0.05, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            
            try:
                res = minimize(
                    lambda h: evaluate_erdos_solution(h, dx), 
                    h_perturbed,

                    c5_new = evaluate_erdos_solution(h_new, dx)

    # Final evaluation
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
