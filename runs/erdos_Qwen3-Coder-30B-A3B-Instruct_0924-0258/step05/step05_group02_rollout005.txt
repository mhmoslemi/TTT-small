import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for genetic algorithm
    pop_size = 50
    elite_size = int(0.3 * pop_size)
    mutation_rate = 0.1
    max_generations = 1000
    
    # Population initialization
    population = []
    population.append(best_h.copy())
    
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    def compute_fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    gen = 0
    stagnation_count = 0
    stagnation_threshold = 50
    
    while time() - start_time < budget_s * 0.98 and gen < max_generations:
        # Evaluate fitness of population
        fitness = [compute_fitness(ind) for ind in population]
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        if stagnation_count > stagnation_threshold:
            # Random restart
            population = [population[0].copy()]
            for _ in range(pop_size - 1):
                h = np.random.rand(n_points)
                h = h * (n_points/2) / np.sum(h)
                h = np.clip(h, 0, 1)
                population.append(h)
            stagnation_count = 0
        
        # Create new population
        new_population = population[:elite_size]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = population[np.random.randint(0, elite_size)]
            parent2 = population[np.random.randint(0, elite_size)]
            
            # Crossover (uniform with sparsity bias)
            alpha = np.random.uniform(0.4, 0.6)
            child = alpha * parent1 + (1 - alpha) * parent2
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child[i] += np.random.normal(0, 0.02)
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        gen += 1
        
        # Periodic gradient refinement
        if gen % 10 == 0 and time() - start_time < budget_s * 0.95:
            # Refine top 10 candidates with local optimization
            for i in range(min(10, len(population))):
                h_ref = population[i].copy()
                
                # Local optimization using scipy minimize
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
                
                try:
                    res = minimize(obj, h_ref, method='SLSQP', bounds=bounds, constraints=cons, 
                                 options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        refined_h = res.x
                        refined_c5 = evaluate_erdos_solution(refined_h, dx)
                        if refined_c5 < population[i].copy():
                            population[i] = refined_h
                except:
                    pass
    
    # Final refinement of best solution
    final_h = best_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
    
    try:
        res = minimize(lambda h: evaluate_erdos_solution(h, dx), 
                       final_h, 
                       method='SLSQP', 
                       bounds=bounds, 
                       constraints=cons,
                       options={'maxiter': 100, 'ftol': 1e-7})
        if res.success:
            final_h = res.x
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_c5 = final_c5
    except:
        pass
    
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
