import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Ensure initial solution is valid
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    # Optimization parameters
    max_iterations = 500
    stagnation_limit = 50
    shift_samples = 50
    learning_rate = 0.01
    
    stagnation_count = 0
    last_best = best_c5
    
    # Main optimization loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample shifts to find problematic one
        shifts = np.random.choice(n_points, shift_samples, replace=False)
        max_overlap = -1
        best_shift = 0
        
        for k in shifts:
            # Compute the overlap for shift k
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlap *= 2.0 / n_points
            
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = k
        
        # Compute gradient for the worst shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_shift) % n_points
            grad[i] += (1 - h[j]) * 2.0 / n_points
            grad[j] -= h[i] * 2.0 / n_points
        
        # Line search for appropriate step size
        alpha = learning_rate
        while alpha > 1e-10:
            h_candidate = h - alpha * grad
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            c5_candidate = evaluate_erdos_solution(h_candidate, 2.0 / n_points)
            if c5_candidate < best_c5:
                h = h_candidate.copy()
                best_c5 = c5_candidate
                stagnation_count = 0
                last_best = best_c5
                break
            alpha *= 0.5
        
        # Early termination if target reached
        if best_c5 < 0.3807:
            break
        
        # Check for stagnation
        if best_c5 == last_best:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        # Restart with small perturbation if stagnating
        if stagnation_count > stagnation_limit:
            h = h + np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0
        
        # Update best solution
        if best_c5 < evaluate_erdos_solution(best_h, 2.0 / n_points):
            best_h = h.copy()
    
    # Final refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
