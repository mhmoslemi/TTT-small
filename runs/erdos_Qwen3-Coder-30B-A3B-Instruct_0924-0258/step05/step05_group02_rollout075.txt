import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid optimization (20 points)
    n_low = 20
    dx_low = 2.0 / n_low
    h_low = np.ones(n_low) * 0.5
    h_low = h_low * (n_low/2) / np.sum(h_low)
    
    # Convex relaxation on coarse grid
    try:
        h_relaxed = cp.Variable(n_low)
        constraints = [
            cp.sum(h_relaxed) == n_low / 2.0,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        
        # Add some overlap constraints
        for k in range(1, min(10, n_low)):  
            if k < n_low // 2:
                corr_sum = cp.sum(cp.multiply(h_relaxed[:n_low-k], 1 - h_relaxed[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_low-k]))
            constraints.append(corr_sum <= best_c5 * dx_low + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            h_low = h_relaxed.value
            h_low = np.clip(h_low, 0, 1)
    except Exception:
        pass
    
    # Interpolate to high resolution
    h_interp = np.interp(np.linspace(0, 2, n_points), 
                        np.linspace(0, 2, n_low), h_low)
    h_interp = h_interp * (n_points/2) / np.sum(h_interp)
    h_interp = np.clip(h_interp, 0, 1)
    
    # Refine
    h_refined = h_interp.copy()
    
    # Use a more aggressive local optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        # First local refinement with SLSQP
        res = minimize(
            objective,
            h_refined,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if res.success:
            h_refined = res.x
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Multi-restart local search
    max_restarts = 100
    restart_count = 0
    
    while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Perturb
        epsilon = 0.1 * (1 + 0.1 * np.random.rand())
        h_perturbed = best_h + epsilon * np.random.randn(n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Local refinement
        try:
            res = minimize(
                objective,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
                else:
                    restart_count += 1
            else:
                restart_count += 1
        except Exception:
            restart_count += 1
    
    # Final validation and normalization
    h_final = np.clip(best_h.copy(), 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
