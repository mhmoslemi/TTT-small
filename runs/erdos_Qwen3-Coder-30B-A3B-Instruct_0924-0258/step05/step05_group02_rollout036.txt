import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def refine_solution(h_guess):
        h_refined = h_guess.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        return h_refined
    
    # Refinement using Nelder-Mead
    max_iterations = 200
    iteration = 0
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Step 1: Local optimization with SLSQP
        try:
            cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                best_h.copy(), 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            if res.success:
                h_refined = refine_solution(res.x)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    iteration = 0  # Reset iteration count on success
                else:
                    iteration += 1
            else:
                iteration += 1
        except Exception:
            iteration += 1
        
        # Step 2: Random perturbations
        if iteration > 10:
            h_perturbed = best_h.copy()
            perturbation = np.random.normal(0, 0.02, n_points)
            h_perturbed += perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
                iteration = 0
            else:
                iteration += 1
                
        # Step 3: Periodic restart
        if iteration > 30:
            # Try a periodic construction
            h_periodic = np.zeros(n_points)
            for i in range(n_points):
                h_periodic[i] = 1.0 if i % 3 == 0 else 0.0
            h_periodic = h_periodic * (n_points/2) / np.sum(h_periodic)
            c5_periodic = evaluate_erdos_solution(h_periodic, dx)
            if c5_periodic < best_c5:
                best_h = h_periodic
                best_c5 = c5_periodic
                iteration = 0
            else:
                iteration += 1
                
        # Step 4: Global restart with random small positive values
        if iteration > 50:
            h_random = np.random.rand(n_points)
            h_random = h_random * (n_points/2) / np.sum(h_random)
            h_random = np.clip(h_random, 0, 1)
            c5_random = evaluate_erdos_solution(h_random, dx)
            if c5_random < best_c5:
                best_h = h_random
                best_c5 = c5_random
                iteration = 0
            else:
                iteration += 1
    
    # Final normalization
    h_final = refine_solution(best_h)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
