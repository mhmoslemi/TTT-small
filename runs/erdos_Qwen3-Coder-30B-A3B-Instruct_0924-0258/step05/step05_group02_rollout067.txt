import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid relaxation for initial warm-up
    n_low = 20
    h_low = np.ones(n_low) * 0.5
    h_low = h_low * (n_low/2) / np.sum(h_low)
    
    # Define constraints for the coarse grid
    try:
        h_low_var = cp.Variable(n_low)
        constraints = [
            cp.sum(h_low_var) == n_low / 2.0,
            h_low_var >= 0,
            h_low_var <= 1
        ]
        
        # Add constraints for a subset of k values
        for k in range(1, min(10, n_low)):
            if k < n_low // 2:
                corr_sum = cp.sum(cp.multiply(h_low_var[:n_low-k], 1 - h_low_var[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_low_var[k:], 1 - h_low_var[:n_low-k]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_low_var)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            h_low = h_low_var.value
    except Exception:
        pass
    
    # Upsample coarse to fine grid
    h_upsampled = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, n_low), h_low)
    h_upsampled = h_upsampled * (n_points/2) / np.sum(h_upsampled)
    
    # Refine with gradient descent
    def grad_c5(h, dx):
        # Approximate gradient using finite differences for the max correlation
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        
        # Gradient for the k_max case only
        for i in range(len(h)):
            if i + k_max < len(h):
                grad[i] += (1 - h[i + k_max])
            if i - k_max >= 0:
                grad[i] += h[i - k_max]
        return grad * dx
    
    h_refined = h_upsampled.copy()
    lr = 0.01
    patience = 0
    mutation_strength = 0.05
    
    while time() - start_time < budget_s * 0.95:
        # Mini-batch gradient descent
        c5_current = evaluate_erdos_solution(h_refined, dx)
        grad = grad_c5(h_refined, dx)
        h_new = h_refined - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < c5_current:
            h_refined = h_new
            best_c5 = min(best_c5, c5_new)
            patience = 0
        else:
            patience += 1
            
        # Mutation to escape local minima
        if patience > 50:
            mask = np.random.random(n_points) < 0.1
            h_refined += mutation_strength * np.random.normal(0, 1, n_points) * mask
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            patience = 0
            lr = max(0.001, lr * 0.95)
            
        if patience > 100:
            break
    
    # Final validation
    h_final = h_refined.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
