import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use coarse grid refinement first
    n_coarse = min(20, n_points)
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.ones(n_coarse) * 0.5
    
    # Try running a convex relaxation on the coarse grid
    try:
        h_relaxed = cp.Variable(n_coarse)
        constraints = [
            cp.sum(h_relaxed) == n_coarse / 2.0,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        
        # Add constraints for a subset of shifts to avoid too many
        max_shift = min(10, n_coarse-1)
        for k in range(1, max_shift):  
            corr_sum = cp.sum(cp.multiply(h_relaxed[:n_coarse-k], 1 - h_relaxed[k:]))
            constraints.append(corr_sum <= best_c5 * dx_coarse + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            h_coarse = h_relaxed.value
            h_coarse = np.clip(h_coarse, 0, 1)
            # Interpolate to full resolution
            h_interpolated = np.interp(
                np.linspace(0, 2, n_points),
                np.linspace(0, 2, n_coarse),
                h_coarse
            )
            h_interpolated = h_interpolated * (n_points/2) / np.sum(h_interpolated)
            
            h_refined = h_interpolated
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    # Fine-grained local search
    max_restarts = 30
    restart_count = 0
    early_stop = 0
    
    while time() - start_time < budget_s * 0.95:
        # Gradient-based local optimization
        h_perturbed = best_h.copy()
        h_perturbed += np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        try:
            cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-7}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
                    early_stop = 0
                else:
                    restart_count += 1
                    early_stop += 1
            else:
                restart_count += 1
                early_stop += 1
        except Exception:
            restart_count += 1
            early_stop += 1
        
        # Adaptive restart with perturbation
        if restart_count > max_restarts:
            # Try more aggressive perturbation
            h_perturbed = np.random.rand(n_points)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            restart_count = 0
            
        # Check timeout
        if time() - start_time > budget_s * 0.9:
            break
    
    # Final refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    # Enforce constraints explicitly for final result
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
