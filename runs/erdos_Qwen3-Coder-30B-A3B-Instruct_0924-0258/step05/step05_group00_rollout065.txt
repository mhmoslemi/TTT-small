import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        n_points = len(h_current)
    else:
        n_points = 35
        h_current = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Multi-resolution optimization approach
    resolutions = [35, 40, 50, 60]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_current)
            h_current = h_interp
        else:
            h_current = h_current[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Cutting-plane approach with local refinement
        max_iterations = 100
        for iteration in range(max_iterations):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute all overlaps
            corr = np.correlate(h_current, 1-h_current, mode='full')
            k_max = np.argmax(corr)
            max_overlap = corr[k_max] * dx
            
            # If we're improving, proceed to optimization
            if max_overlap < best_c5 * 0.999:
                best_c5 = max_overlap
                best_h = h_current.copy()
            
            # Create convex model
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add constraints for all known worst overlaps
            # We add a constraint for the current worst shift only
            k = k_max
            if k < n_points:
                overlap_expr = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                overlap_expr = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            
            constraints.append(overlap_expr <= best_c5 * dx + 1e-8)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
                if prob.status == cp.OPTIMAL:
                    h_new = h_opt.value
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    if evaluate_erdos_solution(h_new, dx) < best_c5:
                        h_current = h_new.copy()
                        best_c5 = evaluate_erdos_solution(h_new, dx)
                        best_h = h_new.copy()
            except:
                pass
                
            # Local refinement with SLSQP
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                    
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
                
                res = minimize(objective, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-8})
                
                if res.success and res.fun < best_c5:
                    h_current = res.x
                    best_c5 = res.fun
                    best_h = res.x.copy()
                elif res.success:
                    h_current = res.x
            except:
                pass
                
            # Random restart if no improvement
            if iteration % 10 == 0 and iteration > 0:
                h_restart = np.clip(h_current + np.random.normal(0, 0.02, n_points), 0, 1)
                h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                h_current = h_restart.copy()
    
    # Final refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
