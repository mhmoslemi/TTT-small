import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    elitism = int(0.3 * pop_size)
    mutation_rate = 0.1
    max_generations = 1000
    
    # Initialize population
    population = []
    population.append(h_start.copy())
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    generation = 0
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append((c5, h))
        
        # Sort by fitness (lower C5 better)
        fitness.sort(key=lambda x: x[0])
        
        # Keep elite
        elites = [h for _, h in fitness[:elitism]]
        
        # Create new population
        new_population = elites[:]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = random.choice(elites)
            parent2 = random.choice(elites)
            
            # Crossover
            alpha = random.uniform(0.4, 0.6)
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if random.random() < mutation_rate:
                noise = np.random.normal(0, 0.02, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            
            # Rescale to satisfy constraint
            child = child * (n_points/2) / np.sum(child)
            new_population.append(child)
        
        population = new_population
        
        # Check if we found a better solution
        current_best = min(fitness, key=lambda x: x[0])
        if current_best[0] < best_c5:
            best_h = current_best[1].copy()
            best_c5 = current_best[0]
            
        generation += 1
    
    # Gradient descent refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    for i in range(min(10, len(population))):  # refine top candidates
        if time() - start_time > budget_s * 0.95:
            break
            
        h_start = population[i]
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            res = minimize(objective, h_start, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-8})
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Final normalization to ensure constraint
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
