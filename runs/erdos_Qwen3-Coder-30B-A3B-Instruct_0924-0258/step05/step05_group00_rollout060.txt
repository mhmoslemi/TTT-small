import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution grid search
    resolutions = [35, 40, 50, 60]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        h_current = np.clip(h_current, 0, 1)
        
        # Local SLSQP optimization with multiple restarts
        max_restarts = 20
        for restart in range(max_restarts):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Random perturbation
            h_restart = h_current + np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            # SLSQP local optimization
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 300, 'ftol': 1e-10})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
                
        h_start = best_h.copy()
    
    # Use QCQP for final refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add key overlap constraints to speed up search
        for k in range(1, n_points//2):
            shifted = cp.hstack([h_opt[k:], h_opt[:k]])
            corr_sum = cp.sum(cp.multiply(h_opt, 1 - shifted))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-8, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Refinement with gradient-based optimization
    try:
        h_final = np.clip(best_h, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
