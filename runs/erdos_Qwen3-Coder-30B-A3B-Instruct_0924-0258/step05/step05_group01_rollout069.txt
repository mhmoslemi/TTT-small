import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        # Convert to binary and adjust
        h_binary = (h_start > 0.5).astype(int)
        current_sum = np.sum(h_binary)
        target_sum = n_points // 2
        if current_sum > target_sum:
            # Need to flip some 1s to 0s
            indices_ones = np.where(h_binary == 1)[0]
            flip_indices = np.random.choice(indices_ones, size=current_sum - target_sum, replace=False)
            h_binary[flip_indices] = 0
        elif current_sum < target_sum:
            # Need to flip some 0s to 1s
            indices_zeros = np.where(h_binary == 0)[0]
            flip_indices = np.random.choice(indices_zeros, size=target_sum - current_sum, replace=False)
            h_binary[flip_indices] = 1
        h_start = h_binary.astype(float)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1
        np.random.shuffle(h_start)
    
    # Initialize
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_tenure = 10
    tabu_list = []
    max_no_improve = 100
    no_improve_count = 0
    shake_frequency = 50
    shake_ratio = 0.05
    
    # Set up the binary optimization
    def generate_neighbor(current_h, tabu_list):
        # Generate a neighbor by flipping one 1 to 0 and one 0 to 1
        ones_idx = np.where(current_h == 1)[0]
        zeros_idx = np.where(current_h == 0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            # If all are 1s or 0s, do a random flip
            flip_idx_0 = np.random.choice(n_points)
            flip_idx_1 = np.random.choice(n_points)
            neighbor = current_h.copy()
            neighbor[flip_idx_0] = 1 - neighbor[flip_idx_0]
            neighbor[flip_idx_1] = 1 - neighbor[flip_idx_1]
            return neighbor, (flip_idx_0, flip_idx_1)
        
        # Pick a 1 to flip to 0 and a 0 to flip to 1
        idx_1 = np.random.choice(ones_idx)
        idx_0 = np.random.choice(zeros_idx)
        
        # Avoid using tabu
        move = (idx_1, idx_0)
        if move in tabu_list or (idx_0, idx_1) in tabu_list:
            # Try a different pair if this is tabu
            # If we can't find a non-tabu pair, we will proceed anyway (will be accepted due to aspiration)
            pass
        
        neighbor = current_h.copy()
        neighbor[idx_1] = 0
        neighbor[idx_0] = 1
        return neighbor, move
    
    def is_tabu(move, tabu_list):
        return move in tabu_list or (move[1], move[0]) in tabu_list
    
    # Main tabu search loop
    iteration = 0
    while time() - start_time < budget_s * 0.9:
        iteration += 1
        
        # Generate neighbor
        neighbor_h, move = generate_neighbor(best_h, tabu_list)
        
        # Evaluate neighbor
        neighbor_c5 = evaluate_erdos_solution(neighbor_h, dx)
        
        # Manage tabu list
        if len(tabu_list) >= tabu_tenure:
            tabu_list.pop(0)
        tabu_list.append(move)
        
        # Check aspiration criterion
        if neighbor_c5 < best_c5:
            # Accept the improvement
            best_h = neighbor_h
            best_c5 = neighbor_c5
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Shake if no improvement for a while
        if no_improve_count >= shake_frequency:
            # Do a random restart
            # Flip 5% of bits
            num_flip = max(1, int(n_points * shake_ratio))
            flip_indices = np.random.choice(n_points, size=num_flip, replace=False)
            shake_h = best_h.copy()
            shake_h[flip_indices] = 1 - shake_h[flip_indices]
            # Ensure sum constraint via adjustment
            current_sum = np.sum(shake_h)
            target_sum = n_points / 2
            if current_sum > target_sum:
                ones_idx = np.where(shake_h == 1)[0]
                to_flip = int(current_sum - target_sum)
                flip_indices = np.random.choice(ones_idx, size=to_flip, replace=False)
                shake_h[flip_indices] = 0
            elif current_sum < target_sum:
                zeros_idx = np.where(shake_h == 0)[0]
                to_flip = int(target_sum - current_sum)
                flip_indices = np.random.choice(zeros_idx, size=to_flip, replace=False)
                shake_h[flip_indices] = 1
            # Update best if this is better
            shake_c5 = evaluate_erdos_solution(shake_h, dx)
            if shake_c5 < best_c5:
                best_h = shake_h
                best_c5 = shake_c5
            no_improve_count = 0
            
        # Early termination if no improvement
        if no_improve_count >= max_no_improve:
            break
    
    # Final validation
    final_sum = np.sum(best_h)
    if abs(final_sum - n_points / 2) > 1e-6:
        # Adjust to satisfy constraint
        current_sum = np.sum(best_h)
        target_sum = n_points / 2
        
        if current_sum > target_sum:
            # Flip some 1s to 0s
            ones_idx = np.where(best_h == 1)[0]
            num_to_flip = int(current_sum - target_sum)
            flip_idx = np.random.choice(ones_idx, size=num_to_flip, replace=False)
            best_h[flip_idx] = 0
        elif current_sum < target_sum:
            # Flip some 0s to 1s
            zeros_idx = np.where(best_h == 0)[0]
            num_to_flip = int(target_sum - current_sum)
            flip_idx = np.random.choice(zeros_idx, size=num_to_flip, replace=False)
            best_h[flip_idx] = 1
    
    # Final C5 computation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
