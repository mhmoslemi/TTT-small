import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary algorithm parameters
    pop_size = 50
    max_gens = 1000
    mutation_rate = 0.15
    elite_size = 5
    tournament_size = 3
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Add initial solution if it exists
    if 'initial_h_values' in globals():
        population[0] = h_start.copy()
    
    # Evolutionary loop
    gen = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time() - start_time < budget_s * 0.9 and gen < max_gens:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Track best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Check for reset
        if no_improvement_count > max_no_improvement:
            # Reset with better initial
            if 'initial_h_values' in globals():
                population[0] = h_start.copy()
            else:
                h_reset = np.ones(n_points) * 0.5
                h_reset = h_reset * (n_points/2) / np.sum(h_reset)
                population[0] = h_reset
            no_improvement_count = 0
            
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Create new population
        new_pop = []
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        for i in elite_indices:
            new_pop.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_pop) < pop_size:
            # Select two parents
            p1, p2 = np.random.choice(pop_size, 2, replace=False)
            h1, h2 = selected[p1], selected[p2]
            
            # Crossover (single point)
            crossover_point = np.random.randint(1, n_points-1)
            child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
            child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
            
            # Mutation
            for h in [child1, child2]:
                if np.random.random() < mutation_rate:
                    mut_idx = np.random.randint(0, n_points)
                    h[mut_idx] += np.random.normal(0, 0.05)
                    h[mut_idx] = np.clip(h[mut_idx], 0, 1)
                
                # Rescale to meet constraint
                h = h * (n_points/2) / np.sum(h)
                new_pop.append(h.copy())
                if len(new_pop) >= pop_size:
                    break
        
        population = new_pop[:pop_size]
        gen += 1
    
    # Final refinement with local optimization
    if time() - start_time < budget_s * 0.1:
        h_refined = best_h.copy()
        try:
            # Use scipy to refine the best solution
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Local optimization with multiple restarts
            for _ in range(10):
                # Random perturbation
                h_restart = h_refined + np.random.normal(0, 0.03, n_points)
                h_restart = np.clip(h_restart, 0, 1)
                h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                
                res_opt = minimize(objective, h_restart, method='SLSQP', 
                                bounds=bounds, constraints=cons_sum, 
                                options={'maxiter': 100, 'ftol': 1e-6})
                
                if res_opt.success:
                    h_test = res_opt.x
                    c5_test = objective(h_test)
                    if c5_test < best_c5:
                        best_c5 = c5_test
                        best_h = h_test.copy()
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
