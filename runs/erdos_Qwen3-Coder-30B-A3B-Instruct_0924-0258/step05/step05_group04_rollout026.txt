import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    max_iter = 5000
    mu = 1e-3
    tol = 1e-6
    residual_check_freq = 50
    
    # Initialize variables
    h = h_start.copy()
    h_prev = h.copy()
    z = np.zeros(2 * n_points - 1)
    lambda_ = np.zeros(2 * n_points - 1)
    
    def compute_c5_estimate(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def solve_subproblem(h_prev, z_prev, lambda_prev, mu_val):
        # Create variables
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2.0
        ]
        
        # Objective (quadratic penalty)
        # ||h - h_prev||^2 + mu * ||h o (1-h) - z||^2
        # where o represents convolution
        corr = np.correlate(h_var.value, 1 - h_var.value, mode='full') if h_var.value is not None else np.zeros(2*n_points-1)
        obj = cp.norm2(h_var - h_prev) ** 2 + mu_val * cp.norm2(corr - z_prev - lambda_prev) ** 2
        prob = cp.Problem(cp.Minimize(obj), constraints)
        prob.solve(solver=cp.SCS, verbose=False, max_iters=1000, eps=1e-6)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.SOLVER_ERROR]:
            return h_var.value
        else:
            return h_prev
    
    # ADMM iterations
    for iteration in range(max_iter):
        if time() - start_time >= max_time * 0.95:
            break
            
        # Update h using convex optimization
        h_new = solve_subproblem(h, z, lambda_, mu)
        if h_new is None:
            h_new = h.copy()
        
        # Update z (proximal operator)
        conv_h = np.correlate(h_new, 1 - h_new, mode='full') 
        z_new = z + lambda_ + mu * (conv_h - z)
        
        # Projection of z to [0, inf)^N
        z_new = np.maximum(z_new, 0)
        
        # Update lambda
        lambda_new = lambda_ + mu * (conv_h - z_new)
        
        # Update variables
        h = h_new
        z = z_new
        lambda_ = lambda_new
        
        # Check convergence
        if iteration % residual_check_freq == 0:
            # Compute residual
            conv_h = np.correlate(h, 1 - h, mode='full')
            residual = np.linalg.norm(conv_h - z)
            
            if residual < tol:
                break
                
            # Compute current C5
            current_c5 = compute_c5_estimate(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Early stopping
            if best_c5 < 0.3807:
                break
    
    # Final local refinement
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
