import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse-to-fine grid approach
    n_low = 20
    dx_low = 2.0 / n_low
    
    # Initialize coarse grid to a simple construction
    h_low = np.ones(n_low) * 0.5
    h_low = h_low * (n_low / 2) / np.sum(h_low)
    
    # Convex relaxation
    try:
        h_relaxed = cp.Variable(n_low)
        constraints = [
            cp.sum(h_relaxed) == n_low / 2.0,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        
        # Add a subset of constraints to manage complexity
        k_subset = min(10, n_low // 2)
        for k in range(1, k_subset):  
            corr_sum = cp.sum(cp.multiply(h_relaxed[:n_low-k], 1 - h_relaxed[k:]))
            constraints.append(corr_sum <= best_c5 * dx_low + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_relaxed.value
            h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                                   np.linspace(0, 2, n_low), h_feasible)
            h_upsampled = np.clip(h_upsampled, 0, 1)
            h_upsampled = h_upsampled * (n_points / 2) / np.sum(h_upsampled)
            h_init = h_upsampled
        else:
            h_init = np.ones(n_points) * 0.5
            h_init = h_init * (n_points / 2) / np.sum(h_init)
    except Exception:
        h_init = np.ones(n_points) * 0.5
        h_init = h_init * (n_points / 2) / np.sum(h_init)

    # Further refinement with local optimization
    h_current = h_init.copy()
    
    # Helper for computing gradient approximations
    def compute_gradient(h, dx):
        # Simple finite difference approximation for gradient
        n = len(h)
        grad = np.zeros(n)
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            up = evaluate_erdos_solution(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= 1e-5
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            down = evaluate_erdos_solution(h_minus, dx)
            grad[i] = (up - down) / (2e-5)
        return grad

    # Optimization loop
    max_iter = 10000
    iter_count = 0
    patience = 0
    mutation_rate = 0.1
    mutation_strength = 0.05
    current_lr = 0.01
    
    while iter_count < max_iter and time() - start_time < budget_s * 0.95:
        # Compute current value and gradient
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > 100:
            break
            
        # Gradient-based update step
        try:
            grad = compute_gradient(h_current, dx)
            h_new = h_current - current_lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        except Exception:
            h_new = h_current.copy()
        
        # Occasionally mutate with probability mutation_rate
        if np.random.random() < mutation_rate:
            # Add noise to some components
            mask = np.random.random(n_points) < mutation_rate
            noise = np.random.normal(0, mutation_strength, n_points)
            h_new = h_current + noise * mask
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
        h_current = h_new
        
        # Adjust learning rate adaptively
        if current_lr > 1e-6 and iter_count % 500 == 0:
            current_lr *= 0.95
            
        iter_count += 1
    
    # Final validation
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
