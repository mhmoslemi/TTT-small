import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with provided construction or default
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    h = initial_h.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    # Parameters
    max_iter = 500
    stagnation_limit = 50
    shift_samples = 50
    learning_rate = 0.01
    
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample shifts
        shift_indices = random.sample(range(n_points), min(shift_samples, n_points))
        
        # Evaluate overlaps for sampled shifts
        max_overlap = -1
        best_shift = 0
        for k in shift_indices:
            # Compute overlap for shift k
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlap *= 2.0 / n_points
            
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = k
        
        # Compute gradient for the worst case
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_shift) % n_points
            grad[i] += (1 - h[j]) * (2.0 / n_points)
            grad[j] -= h[i] * (2.0 / n_points)
        
        # Line search for step size
        alpha = learning_rate
        new_h = h - alpha * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        new_c5 = evaluate_erdos_solution(new_h, 2.0 / n_points)
        
        # Backtracking line search
        while new_c5 >= best_c5 and alpha > 1e-10:
            alpha *= 0.5
            new_h = h - alpha * grad
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_c5 = evaluate_erdos_solution(new_h, 2.0 / n_points)
            
        # Update solution
        h = new_h
        current_c5 = new_c5

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Restart if stalled
        if stagnation_count > stagnation_limit:
            h = best_h + np.random.normal(0, 0.001, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0
            
        # Early termination
        if best_c5 < 0.3807:
            break
    
    # Final refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
