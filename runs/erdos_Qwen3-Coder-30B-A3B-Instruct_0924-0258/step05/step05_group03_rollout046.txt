import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Initialize parameters
    max_iterations = 100
    stagnation_limit = 50
    shift_samples = 100
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    # Iterative improvement
    iteration = 0
    stagnation_count = 0
    last_best = best_c5
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Step 1: Find optimal shift k for current h
        best_k = 0
        max_overlap = -1
        
        # Sample shifts
        shifts = np.linspace(0, 2, shift_samples)
        for k in shifts:
            # Compute overlap
            if k == 0:
                overlap = np.sum(h * (1 - h)) * (2.0 / n_points)
            else:
                # Circular correlation
                h_shifted = np.roll(h, int(k * n_points / 2))
                overlap = np.sum(h * (1 - h_shifted)) * (2.0 / n_points)
                
            if overlap > max_overlap:
                max_overlap = overlap
                best_k = k
        
        # Step 2: Optimize h to reduce overlap at k*
        # We'll use a convex relaxation approach for a better update step
        try:
            # Create variable for h
            h_var = cp.Variable(n_points)
            
            # Objective: maximize the overlap at k*
            # This is equivalent to minimizing -overlap
            # But for our case, we want to minimize max_k overlap
            
            # Simplified: since we know k*, just enforce current h
            # So we treat this as gradient-based improvement
            
            # Try to use local quadratic model or SLSQP for refinement
            def obj_func(h_test):
                # Compute overlap at best_k
                if best_k == 0:
                    return np.sum(h_test * (1 - h_test)) * (2.0 / n_points)
                else:
                    h_shifted = np.roll(h_test, int(best_k * n_points / 2))
                    return np.sum(h_test * (1 - h_shifted)) * (2.0 / n_points)
            
            def constraint_sum(h_test):
                return np.sum(h_test) - n_points / 2.0
            
            # Local optimization
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Run local optimization with SLSQP
            res = minimize(obj_func, h, method='SLSQP',
                          bounds=bounds, constraints=cons_sum,
                          options={'maxiter': 50, 'ftol': 1e-8})
            
            if res.success:
                projected_h = np.clip(res.x, 0, 1)
                sum_h = np.sum(projected_h)
                if sum_h > 0:
                    projected_h = projected_h * (n_points / 2) / sum_h
                h = projected_h.copy()
                
        except:
            pass
            
        # Check improvement
        current_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            stagnation_count = 0
            last_best = best_c5
        else:
            stagnation_count += 1
            
        # Apply random perturbation to escape local minima if stagnating
        if stagnation_count > stagnation_limit // 2:
            # Add small random noise to h
            noise = np.random.normal(0, 1e-4, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * (n_points / 2) / sum_h
                
        iteration += 1
        if best_c5 < 0.3807:
            break
            
    # Final refinement
    try:
        def obj_func(h_test):
            return evaluate_erdos_solution(h_test, 2.0 / n_points)
        def constraint_sum(h_test):
            return np.sum(h_test) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj_func, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons_sum, 
                      options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj_func(best_h)
    except:
        pass

    return (list(best_h), best_c5, n_points)
