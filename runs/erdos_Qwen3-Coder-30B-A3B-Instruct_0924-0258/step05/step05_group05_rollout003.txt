import numpy as np
from scipy.optimize import minimize
from time import time
import hashlib

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    max_iter = 1000
    max_stagnation = 100
    mutation_scale = 0.04
    tabu_list_size = max(1, int(0.2 * n_points))
    tabu_list = []
    stagnation_counter = 0
    
    # Current solution
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Initialize with base solution if needed
    if current_c5 < best_c5:
        best_h = current_h.copy()
        best_c5 = current_c5
    
    def hash_solution(h):
        # Create a hashable representation
        sorted_h = tuple(sorted(h))
        return hashlib.md5(str(sorted_h).encode()).hexdigest()
    
    def perturb_solution(h, scale):
        # Create perturbed version
        h_new = h.copy()
        # Pick indices to mutate
        n_mutate = max(1, int(0.3 * n_points))
        indices = np.random.choice(n_points, size=n_mutate, replace=False)
        
        for i in indices:
            h_new[i] += np.random.normal(0, scale)
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        return h_new
    
    # Main search loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb current solution
        candidate_h = perturb_solution(current_h, mutation_scale)
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Hash to check tabu list
        candidate_key = hash_solution(candidate_h)
        
        # Accept if better, or if not tabu and acceptable
        is_better = candidate_c5 < current_c5
        not_tabu = candidate_key not in tabu_list
        
        if is_better or not_tabu:
            current_h = candidate_h
            current_c5 = candidate_c5
            
            # Update best
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
                # Local refinement
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, best_h, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-8})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
            
            # Update tabu list
            if len(tabu_list) >= tabu_list_size:
                tabu_list.pop(0)
            tabu_list.append(candidate_key)
            
            # Reset stagnation
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Increase mutation if stagnated
        if stagnation_counter > max_stagnation:
            mutation_scale = min(0.5, mutation_scale * 1.2)
            stagnation_counter = 0
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
