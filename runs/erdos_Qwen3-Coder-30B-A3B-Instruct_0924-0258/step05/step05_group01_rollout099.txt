```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
from scipy.spatial.distance import cdist

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1, sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initial population of candidates
    n_initial = 20
    initial_candidates = []
    
    # Start with best_h
    initial_candidates.append(best_h.copy())
    
    # Add random candidates
    for _ in range(n_initial - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        initial_candidates.append(h)
    
    # Evaluate initial candidates
    X = []
    y = []
    for h in initial_candidates:
        c5 = evaluate_erdos_solution(h, dx)
        X.append(h)
        y.append(c5)
        
    X = np.array(X)
    y = np.array(y)
    
    # Train surrogate model
    kernel = C(1.0, (1e-3, 1e3)) * RBF(1.0, (1e-2, 1e2))
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-6, 
                                  normalize_y=True, n_restarts_optimizer=5)
    gp.fit(X, y)
    
    n_evaluations = n_initial
    n_resolutions = 0
    max_res = 100
    
    while time() - start_time < budget_s * 0.95 and n_points < max_res:
        # Generate new candidates using GP
        n_candidates = 50
        candidates = []
        
        # Sample from GP predictive distribution
        h_sample = np.random.rand(n_candidates, n_points)
        h_sample = np.clip(h_sample, 0, 1)
        for i in range(n_candidates):
            h_sample[i] = project_to_feasible(h_sample[i], n_points)
        
        # Predict using GP
        try:
            mu, sigma = gp.predict(h_sample, return_std=True)
            # Acquisition function: Expected Improvement
            best_y = np.min(y)
            improvement = best_y - mu
            # EI with small exploration term
            ei = improvement * (improvement > 0) * (sigma > 0)
            idx = np.argsort(ei)[-10:]  # Top 10 based on EI
            candidates = h_sample[idx]
        except:
            # Fall back to random samples if GP fails
            candidates = []
            for _ in range(10):
                h = np.random.rand(n_points)
                h = project_to_feasible(h, n_points)
                candidates.append(h)
            candidates = np.array(candidates)
        
        # Evaluate candidates
        for h in candidates:
            h = project_to_feasible(h, n_points)
            c5 = evaluate_erdos_solution(h, dx)
            n_evaluations += 1
            
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                
            # Add to GP training data
            X = np.vstack([X, h])
            y = np.append(y, c5)
        
        # Retrain GP
        try:
            gp.fit(X, y)
        except:
            pass
        
        # Adaptively increase resolution
        if n_evaluations % 50 == 0 and n_points < max_res:
            n_points += 5
            if n_points > max_res:
                n_points = max_res
            dx = 2.0 / n_points
            # Interpolate best solution to new resolution
            if len(best_h) == n_points - 5:
                # Interpolate best_h
                x_old = np.linspace(0, 2, len(best_h))
                x_new = np.linspace(0, 2, n_points)
                best_h = np.interp(x_new, x_old, best_h)
            # Ensure correct sum
            best_h = project_to_feasible(best_h, n_points)
            
            # Update bounds on the surrogate model if needed
            # ... other maintenance steps for GP (see above)
            
            # Evaluate again after resolution change
            current_c5 = evaluate_erdos_solution(best_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
    
    # Final refinement on best solution
    if n_points > 100:
        # Use local optimization on best solution
        h_current = best_h.copy()
        h_current = project_to_feasible(h_current, n_points)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, h_current, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 100})
            if res.success:
                h_refined = res.x
                h_refined = project_to_feasible(h_refined, n_points)
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass

    return (list(best_h), best_c5, n_points)
```
