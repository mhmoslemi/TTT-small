import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # SCP parameters
    max_iter = 500
    mu = 1e-3
    C5_nih = 0.4
    batch_size = 10
    
    # Setup variables
    h_var = cp.Variable(n_points)
    C5_var = cp.Variable()
    
    # Constraint setup: for each shift k, we have overlap constraint
    # We'll collect constraints in batches for efficiency
    
    # Main SCP loop
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Evaluate current h
        current_c5 = evaluate_erdos_solution(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
            
        if best_c5 < 0.3807:
            break
            
        # Build quadratic approximation of the constraints
        h_current = best_h.copy()
        constraints = []
        
        # Ascend in k, set max k as 2*n_points to cover full shift space
        for k in range(2*n_points):
            # Compute shift for correlation: h shift by k positions backwards
            shifted = np.roll(h_current, k)
            # The overlap integral is h(x)*(1-h(x+k)) so we compute:
            # sum(h_current * (1 - shifted)) * dx <= C5_nih
            overlap_val = np.sum(h_current * (1 - shifted)) * dx
            if overlap_val > 0.99 * C5_nih:
                # Linearize constraint around h_current
                # Constraint: sum(h_current * (1 - shifted)) * dx <= C5_nih
                # Linearization gives:
                # sum((h_current - h_current) * (1 - shifted) + h_var * (1 - shifted)) * dx <= C5_nih
                # sum(h_var * (1 - shifted)) * dx <= C5_nih - sum(h_current * (1 - shifted)) * dx
                
                linearized_term = np.sum((1 - shifted) * dx)
                linearized_const = C5_nih - overlap_val
                linear_constraint = cp.sum(cp.multiply(h_var, 1 - shifted)) * dx <= linearized_const
                constraints.append(linear_constraint)
        
        if not constraints:
            # If no strong constraints, just make a single overlap constraint
            for k in range(1, n_points):
                shifted = np.roll(h_current, k)
                overlap_val = np.sum(h_current * (1 - shifted)) * dx
                linearized_term = np.sum((1 - shifted) * dx)
                linearized_const = C5_nih - overlap_val
                linear_constraint = cp.sum(cp.multiply(h_var, 1 - shifted)) * dx <= linearized_const
                constraints.append(linear_constraint)
        
        # Add constraint that sum(h) = 1
        sum_constraint = cp.sum(h_var) == n_points / 2.0
        constraints.append(sum_constraint)
        
        # Add box constraints
        for i in range(n_points):
            constraints.append(h_var[i] >= 0)
            constraints.append(h_var[i] <= 1)
        
        # Add objective
        objective = cp.Minimize(C5_var)
        
        # Solve the QCQP
        try:
            # Create the problem with single constraint - this is a less robust simplification
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.OSQP, max_iter=1000, eps_abs=1e-6, eps_rel=1e-6)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                if new_h is not None:
                    # Apply bounds
                    new_h = np.clip(new_h, 0, 1)
                    # Normalize
                    new_h = new_h * (n_points / 2) / np.sum(new_h)
                    best_h = new_h.copy()
        except:
            pass
    
    # Final refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                       constraints=cons_sum, options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
