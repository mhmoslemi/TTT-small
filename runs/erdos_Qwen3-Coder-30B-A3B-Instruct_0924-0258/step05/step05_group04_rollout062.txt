import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    num_particles = 40
    max_iter = 500
    inertia_weight = 0.7
    c1, c2 = 2.0, 2.0
    max_velocity = 0.2
    warm_restart_after = 200
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    for _ in range(num_particles):
        if 'initial_h_values' in globals():
            h = h_start.copy()
            # Add small random noise
            noise = np.random.normal(0, 0.02, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.random.random(n_points)
            h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_fitness.append(evaluate_erdos_solution(h, dx))
    
    # Find global best
    best_idx = np.argmin(pbest_fitness)
    gbest = pbest[best_idx].copy()
    gbest_fitness = pbest_fitness[best_idx]
    
    if gbest_fitness < best_c5:
        best_c5 = gbest_fitness
        best_h = gbest.copy()
    
    iter_without_improvement = 0
    
    # PSO iterations
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        if iter_without_improvement > warm_restart_after:
            # Warm restart
            for i in range(num_particles):
                velocities[i] = np.zeros(n_points)
                noise = np.random.normal(0, 0.05, n_points)
                particles[i] += noise
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            iter_without_improvement = 0
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            for j in range(n_points):
                velocities[i][j] = (inertia_weight * velocities[i][j] + 
                                  c1 * r1 * (pbest[i][j] - particles[i][j]) + 
                                  c2 * r2 * (gbest[j] - particles[i][j]))
                
                # Clip velocity
                velocities[i][j] = np.clip(velocities[i][j], -max_velocity, max_velocity)
            
            # Update position
            for j in range(n_points):
                particles[i][j] += velocities[i][j]
                
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h > 1e-10:
                particles[i] = particles[i] * (n_points / 2) / sum_h
            else:
                particles[i] = np.ones(n_points) * (n_points / 2)
            
            # Evaluate new fitness
            fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
            
            # Update global best
            if fitness < gbest_fitness:
                gbest = particles[i].copy()
                gbest_fitness = fitness
                iter_without_improvement = 0
                if gbest_fitness < best_c5:
                    best_c5 = gbest_fitness
                    best_h = gbest.copy()
            else:
                iter_without_improvement += 1
                
        # Early stopping
        if best_c5 < 0.3807:
            break
    
    # Post-processing: local refinement of top 5 particles
    top_indices = np.argsort(pbest_fitness)[:5]
    for idx in top_indices:
        h_init = pbest[idx].copy()
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        def con(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': con}
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(obj, h_init, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_new = res.x
                f_new = obj(h_new)
                if f_new < best_c5:
                    best_h = h_new
                    best_c5 = f_new
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
