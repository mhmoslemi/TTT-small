import numpy as np
from scipy.signal import correlate
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use block-based construction
    # Try to find a good structure based on sqrt(n_points)
    M = int(np.sqrt(n_points))
    if M < 2:
        M = 2
    block_size = n_points // M
    if block_size * M != n_points:
        block_size = n_points // M
        M = n_points // block_size
    
    # Ensure we're not too aggressive with M
    if M > n_points:
        M = n_points
        block_size = 1
    
    # Initialize with a structured solution
    h = np.zeros(n_points)
    
    # For better structure, try to get about half the points on
    target_on = n_points // 2
    on_blocks = target_on // block_size
    
    # Pad or truncate to make it valid
    if on_blocks * block_size > target_on:
        on_blocks = target_on // block_size
    elif on_blocks * block_size < target_on:
        # Add one more block but adjust size
        on_blocks = target_on // block_size + 1
        # Recalculate block size
        block_size = n_points // on_blocks
        if block_size * on_blocks != n_points:
            block_size = n_points // on_blocks
            on_blocks = n_points // block_size
    
    if on_blocks > M:
        on_blocks = M
    
    # Build basic block structure
    # Start with uniform pattern
    for i in range(on_blocks):
        start_idx = i * block_size
        end_idx = start_idx + block_size
        if end_idx <= n_points:
            h[start_idx:end_idx] = 1.0
    
    # Now adjust to exact sum
    current_sum = np.sum(h)
    diff = current_sum - target_on
    if diff > 0:
        # Need to turn off diff unit blocks
        indices = np.where(h == 1)[0]
        to_turn_off = indices[:diff]
        for idx in to_turn_off:
            h[idx] = 0.0
    elif diff < 0:
        # Need to turn on -diff unit blocks
        indices = np.where(h == 0)[0]
        to_turn_on = indices[:abs(diff)]
        for idx in to_turn_on:
            h[idx] = 1.0
    
    # Alternative: introduce specific bias to see improvement
    # Adjust to match tight integral constraint exactly
    h = h * (n_points / 2) / np.sum(h)
    
    # Refinement step: randomize some blocks
    num_blocks_to_change = 1
    for _ in range(num_blocks_to_change):
        i = random.randint(0, M - 1)
        # Toggle the entire block
        start_idx = i * block_size
        end_idx = start_idx + block_size
        if end_idx <= n_points:
            h[start_idx:end_idx] = 1.0 - h[start_idx:end_idx]
    
    current_best = evaluate_erdos_solution(h, dx)
    if current_best < best_c5:
        best_h = h.copy()
        best_c5 = current_best

    # Optimization loop
    max_iter = 300
    stagnation = 0
    max_stagnation = 50
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create perturbed version
        h_new = h.copy()
        
        # Just try to flip one block
        i = random.randint(0, M - 1)
        start_idx = i * block_size
        end_idx = start_idx + block_size
        if end_idx <= n_points:
            h_new[start_idx:end_idx] = 1.0 - h_new[start_idx:end_idx]
        
        # Renormalize
        current_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / current_sum
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < current_best:
            h = h_new
            current_best = c5_new
            stagnation = 0
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
                if best_c5 < 0.3807:
                    break
        else:
            stagnation += 1
        
        # Escalate if stuck
        if stagnation > max_stagnation:
            sg = stagnation // 10  # Scalable steps
            for _ in range(1 + sg):
                h_new = h.copy()
                # Good chance to flip more blocks
                tries = 1 + min(3, M // 4)
                for _ in range(tries):
                    i = random.randint(0, M - 1)
                    start_idx = i * block_size
                    end_idx = start_idx + block_size
                    if end_idx <= n_points:
                        h_new[start_idx:end_idx] = 1.0 - h_new[start_idx:end_idx]
                # Renormalize
                current_sum = np.sum(h_new)
                h_new = h_new * (n_points / 2) / current_sum
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                if c5_new < current_best:
                    h = h_new
                    current_best = c5_new
                    stagnation = 0
                    if c5_new < best_c5:
                        best_h = h.copy()
                        best_c5 = c5_new
                        if best_c5 < 0.3807:
                            break

    return (list(best_h), best_c5, n_points)
