import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

            # Interpolate back to full resolution

            h_full = np.clip(h_full, 0, 1)
            return evaluate_erdos_solution(h_full, dx)

            return np.sum(h) - n_coarse / 2.0
        
        bounds = [(0, 1) for _ in range(n_coarse)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum_coarse}
        res = minimize(objective_coarse, h_coarse, method='SLSQP',
