```python
import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
import warnings

warnings.filterwarnings('ignore')

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with parent construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize GP surrogate model
    kernel = C(1.0, (1e-3, 1e3)) * RBF(1.0, (1e-2, 1e2))
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=5,
                                  normalize_y=True, alpha=1e-10)

    # Generate initial population
    initial_candidates = []
    initial_values = []
    
    # Add parent if available
    if 'initial_h_values' in globals():
        h = h_start.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        initial_candidates.append(h)
        initial_values.append(evaluate_erdos_solution(h, dx))
    
    # Add random candidates
    n_initial = 15
    for _ in range(n_initial):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        initial_candidates.append(h)
        initial_values.append(evaluate_erdos_solution(h, dx))
    
    # Train GP on initial data
    X_train = np.array(initial_candidates)
    y_train = np.array(initial_values)
    try:
        gp.fit(X_train, y_train)
    except:
        # fallback if GP training fails
        pass
    
    # Main optimization loop
    max_iter = 500
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Generate candidate samples using GP
        n_samples = 50
        candidates = []
        
        # Try to sample from the GP
        try:
            # Generate random samples in feasible space
            for _ in range(n_samples):
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                candidates.append(h)
            
            # Try to use GP for better sampling
            if iter_count > 0:
                X_pred = np.random.rand(20, n_points)
                for i in range(20):
                    h = np.random.rand(n_points)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points/2) / np.sum(h)
                    X_pred[i] = h
                
                try:
                    y_pred = gp.predict(X_pred)
                    # Sort by predicted values 
                    idxs = np.argsort(y_pred)
                    for i in range(min(10, len(idxs))):
                        candidates.append(X_pred[idxs[i]])
                except:
                    pass
            
        except:
            pass
        
        # Evaluate candidates
        for h in candidates:
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            c5 = evaluate_erdos_solution(h, dx)
            
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                
        # Update surrogate model
        try:
            X_train = np.array(initial_candidates + candidates)
            y_train = np.array(initial_values + [evaluate_erdos_solution(h, dx) for h in candidates])
            gp.fit(X_train, y_train)
        except:
            pass
            
        # Adaptive resolution adaptation
        if iter_count % 20 == 0 and n_points < 100:
            n_points += 5
            dx = 2.0 / n_points
            
            # Interpolate previous best to new resolution
            try:
                x_old = np.linspace(0, 2, len(best_h))
                x_new = np.linspace(0, 2, n_points)
                best_h = np.interp(x_new, x_old, best_h)
                
                # Re-scale to sum constraint
                best_h = np.clip(best_h, 0, 1)
                best_h = best_h * (n_points/2) / np.sum(best_h)
                best_c5 = evaluate_erdos_solution(best_h, dx)
            except:
                pass

        iter_count += 1

    # Final verification and refinements
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass

    return (list(best_h), best_c5, n_points)
```
