import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    max_iter = 5000
    mu = 1e-3
    rho = 1.0
    tol = 1e-6
    max_stagnation = 100
    
    # Initialize variables
    h = h_start.copy()
    h_prev = h.copy()
    z = np.zeros(2 * n_points - 1)
    lambda_vec = np.zeros(2 * n_points - 1)
    
    # Compute the indices for the correlation
    corr_indices = np.arange(2 * n_points - 1)
    
    # Main ADMM loop
    for iteration in range(max_iter):
        # Check time limit
        if time() - start_time > max_time * 0.95:
            break
            
        # Update h
        # Solve: minimize 0.5 * ||h - h_prev + (1/mu) * lambda||^2 + 0.5 * rho * ||h - z||^2
        # subject to 0 <= h <= 1 and sum(h) = n_points/2
        
        # Formulate and solve the constrained optimization subproblem using SLSQP
        def obj(h):
            return 0.5 * np.sum((h - h_prev + lambda_vec / mu)**2) + 0.5 * rho * np.sum((h - z)**2)
        
        def con_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons = {'type': 'eq', 'fun': con_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_new = res.x
                # Ensure bounds
                h_new = np.clip(h_new, 0, 1)
                # Renormalize to satisfy sum(h) = n_points/2
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
        except:
            pass
            
        # Update z using the correlation
        h_pad = np.pad(h, (0, n_points-1), 'constant')
        h_rev = np.flip(h)
        # Perform convolution (overlap)
        correlation = np.convolve(h_pad, h_rev, mode='valid')
        one_minus_h = 1 - h
        correlation_2 = np.convolve(h_pad, one_minus_h, mode='valid')
        overlaps = correlation_2
        
        # Update z to be close to overlaps but not exceeding current C5 bound
        # Let's try to keep the best C5 estimate
        current_c5 = evaluate_erdos_solution(h, dx)
        target = np.minimum(overlaps, current_c5 / dx)
        z = target
        
        # Update dual variable lambda
        lambda_vec += mu * (h - z)
        
        # Update h_prev
        h_prev = h.copy()
        
        # Check for convergence
        if iteration % 50 == 0:
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Check stagnation
            if iteration > 0 and abs(current_c5 - best_c5) < tol:
                if iteration > max_stagnation:
                    break
            else:
                pass  # No stagnation is okay
    
    # Final refinement with SLSQP
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
