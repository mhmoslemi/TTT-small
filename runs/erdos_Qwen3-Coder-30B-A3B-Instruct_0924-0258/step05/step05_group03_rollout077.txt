import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Initialize h with constraints satisfied
    h = initial_h.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    # Sampling shifts for overlap computation
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    
    # Use CVXPY for convex relaxation with cutting planes
    max_iter = 50
    prev_c5 = float('inf')
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    # Optimization parameters
    tol = 1e-5
    iterations = 0
    
    while time() - start_time < budget_s * 0.95 and iterations < max_iter:
        # Evaluate overlaps for all shifts
        overlaps = []
        for shift in shifts:
            # Compute overlap shift
            shift_int = int(shift * n_points / 2)
            if shift_int == 0:
                overlap = np.sum(h * (1 - h)) * (2.0 / n_points)
            else:
                # Use correlation-like computation
                h_shifted = np.roll(h, shift_int, axis=0)
                overlap = np.sum(h * (1 - h_shifted)) * (2.0 / n_points)
            overlaps.append(overlap)
        
        # Find maximum overlap
        max_overlap = np.max(overlaps)
        
        # Update best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if abs(prev_c5 - max_overlap) < tol:
            break
        prev_c5 = max_overlap
        
        # Apply cutting plane method with CVXPY
        # Variable definition
        x = cp.Variable(n_points)
        
        # Define constraints
        constraints = []
        
        # Integral constraint
        constraints.append(cp.sum(x) == n_points / 2)
        
        # Box constraints
        constraints.extend([x[i] >= 0 for i in range(n_points)])
        constraints.extend([x[i] <= 1 for i in range(n_points)])
        
        # Add overlap constraint
        max_overlap_expr = 0
        for i, shift in enumerate(shifts):
            shift_int = int(shift * n_points / 2)
            if shift_int == 0:
                overlap = cp.sum(cp.multiply(x, 1 - x))
            else:
                x_shifted = cp.hstack([x[shift_int:], x[:shift_int]])
                overlap = cp.sum(cp.multiply(x, 1 - x_shifted))
            max_overlap_expr = cp.maximum(max_overlap_expr, overlap)
        
        constraints.append(max_overlap_expr <= max_overlap)
        
        # Objective - maximize overlap (but in our case we want to minimize max overlap)
        # We add the constraint directly in the model
        
        # Problem formulation
        prob = cp.Problem(cp.Minimize(max_overlap_expr), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                x_val = x.value
                h = np.array(x_val).reshape(-1)
                # Re-normalize
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fall back to a local method if CVXPY fails
                pass
        except:
            # Continue with local optimization
            pass
            
        iterations += 1
        
        # Early termination
        if best_c5 < 0.3807:
            break
        
    # Final local refinement using scipy
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
