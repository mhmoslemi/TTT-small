import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the given construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Helper to get max overlap index
    def get_max_k(h):
        corr = np.correlate(h, 1-h, mode='full')
        return np.argmax(corr)
    
    # Simple local search
    max_iter = 100
    iter_since_improvement = 0
    max_no_improvement = 100
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    while time() - start_time < budget_s * 0.95 and iter_since_improvement < max_no_improvement:
        # Perturb the current solution
        h_perturbed = h_current + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Optimize with constraints
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-7}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    h_current = h_refined
                    best_c5 = c5_refined
                    iter_since_improvement = 0
                else:
                    iter_since_improvement += 1
            else:
                iter_since_improvement += 1
        except Exception:
            iter_since_improvement += 1
            
    # Final refinement to ensure constraints
    h_final = h_current.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
