import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    best_c5 = evaluate_erdos_solution(h_current, dx)
    best_h = h_current.copy()
    
    # Smooth approximation parameters
    tau = 1.0
    max_iter = 3000
    tolerance = 1e-7
    step_size = 1.0
    
    # L-BFGS-B optimization
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
    
    # Optimization loop
    for i in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Smooth approximation of max overlap
        def smooth_max_overlap(h):
            corr = np.correlate(h, 1 - h, mode='full')
            # Use smooth maximum
            return np.log(np.sum(np.exp(tau * corr))) / tau
        
        # Gradient of smooth maximum
        def smooth_max_gradient(h):
            corr = np.correlate(h, 1 - h, mode='full')
            exp_corr = np.exp(tau * corr)
            exp_sum = np.sum(exp_corr)
            # Gradient computation
            grad = np.zeros_like(h)
            for k in range(len(corr)):
                # Correlation gradient for h[i] due to lag k
                if k < len(h):
                    grad[k] += (1 - h[k]) * exp_corr[k]
                if k + len(h) < len(corr):
                    grad[k + len(h) - len(corr)] += (-h[k + len(h) - len(corr)]) * exp_corr[k]
            return (tau * grad) / exp_sum
        
        # Compute gradient numerically for stability
        def grad_func(h):
            eps = 1e-5
            grad = np.zeros_like(h)
            for j in range(len(h)):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                grad[j] = (smooth_max_overlap(h_plus) - smooth_max_overlap(h_minus)) / (2 * eps)
            return grad
        
        # Use gradient descent
        grad = grad_func(h_current)
        h_new = h_current - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Possibly update using optimize.minimize if better
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        try:
            res = minimize(obj, h_new, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 20, 'ftol': 1e-8})
            if res.success:
                h_test = res.x
                c5_test = evaluate_erdos_solution(h_test, dx)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_test.copy()
        except:
            pass
        
        # Simplified direct update for practicality
        current_c5 = evaluate_erdos_solution(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        
        h_current = h_new
        
        # Adaptive learning rate
        if i % 100 == 0 and i > 0:
            tau *= 0.99
        if i % 50 == 0:
            step_size = max(0.01, step_size * 0.95)
    
    # Final local refinement using SLSQP
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:

            return evaluate_erdos_solution(h, dx)
        def con_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': con_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = obj(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    # Ensure final solution meets constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
