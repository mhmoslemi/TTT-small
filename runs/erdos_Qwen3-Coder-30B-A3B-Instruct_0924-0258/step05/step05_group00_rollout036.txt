import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization with adaptive refinement
    resolutions = [35, 40, 50, 60]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate or truncate to target resolution
        h_current = np.interp(np.linspace(0, 2, res), 
                            np.linspace(0, 2, n_points), 
                            h_start)
        n_points = res
        dx = 2.0 / n_points
        
        # Normalize to match constraint
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Strengthen constraints by using fewer shifts but at optimized step
        max_iter = 10
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Local optimization using SLSQP
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                # Use different initial perturbations
                h_test = h_current + np.random.normal(0, 0.01, n_points)
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points/2) / np.sum(h_test)
                
                res = minimize(objective, h_test, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 300, 'ftol': 1e-9})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                else:
                    # Try another random start
                    h_test = np.random.rand(n_points) * 0.5
                    h_test = h_test * (n_points/2) / np.sum(h_test)
                    res = minimize(objective, h_test, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 200, 'ftol': 1e-8})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
            except Exception:
                pass
                
            # Use a more structured update
            h_current = best_h.copy()
        
        # Have a final QCQP pass at the highest resolution 
        if res == resolutions[-1] and time() - start_time < budget_s * 0.95:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]

                # Simplify overlap constraints
                overlap_indices = list(range(1, min(n_points//2 + 1, 20)))
                full_corr = np.correlate(best_h, 1 - best_h, mode='full')
                max_corr = np.max(full_corr)
                
                for k in overlap_indices:
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)

                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass

    # Final refinement and validation
    try:
        h_final = np.clip(best_h, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass

    return (list(best_h), best_c5, n_points)
