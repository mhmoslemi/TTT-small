import numpy as np
from time import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Block-based approach parameters
    M = int(np.sqrt(n_points))  # Number of blocks
    block_size = n_points // M if n_points % M == 0 else n_points // (M + 1)
    
    # Initialize h with block structure
    h = np.zeros(n_points)
    on_blocks = n_points // (2 * block_size)  # Number of blocks to turn on
    
    # Create block arrangement
    for i in range(on_blocks):
        start_idx = i * block_size
        end_idx = min(start_idx + block_size, n_points)
        h[start_idx:end_idx] = 1
    
    # Adjust sum to exactly n_points/2
    current_sum = np.sum(h)
    if current_sum > n_points / 2:
        # Turn off some points from the end
        diff = int(current_sum - n_points / 2)
        idxs = np.where(h == 1)[0][-diff:]
        for i in idxs:
            h[i] = 0
    elif current_sum < n_points / 2:
        # Turn on some points from the beginning
        diff = int(n_points / 2 - current_sum)
        idxs = np.where(h == 0)[0][:diff]
        for i in idxs:
            h[i] = 1
    
    # Ensure exact sum
    h = h * (n_points/2) / np.sum(h)
    
    # Block optimization loop
    max_iter = 200
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 < 0.3807:
            break
            
        # Quick local search
        for i in range(n_points):
            if h[i] == 1:
                h[i] = 0
                c5_new = evaluate_erdos_solution(h, dx)
                if c5_new < c5:
                    c5 = c5_new
                else:
                    h[i] = 1
            elif h[i] == 0:
                h[i] = 1
                c5_new = evaluate_erdos_solution(h, dx)
                if c5_new < c5:
                    c5 = c5_new
                else:
                    h[i] = 0
                    
        # Try block-based perturbations
        # Create new arrangement by shifting a block
        h_new = h.copy()
        try:
            # Find a 1-block and turn it off
            ones = np.where(h_new == 1)[0]
            if len(ones) > 0:
                # Turn off a random 1-block
                block_to_off = np.random.choice(ones)
                start_block = (block_to_off // block_size) * block_size
                end_block = min(start_block + block_size, n_points)
                for i in range(start_block, end_block):
                    h_new[i] = 0
                    
                # Turn on a 0-block in a different position
                zeros = np.where(h_new == 0)[0]
                if len(zeros) > 0:
                    block_to_on = np.random.choice(zeros)
                    start_block = (block_to_on // block_size) * block_size
                    end_block = min(start_block + block_size, n_points)
                    for i in range(start_block, end_block):
                        h_new[i] = 1
                        
                # Prevent overflow
                if np.sum(h_new) > n_points + 1e-3:
                    h_new = h_new * (n_points/2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < c5:
                    c5 = c5_new 
                    h = h_new.copy()
        except:
            pass
    
    # Final check of best solution
    c5_best = evaluate_erdos_solution(best_h, dx)
    if c5_best < best_c5:
        best_c5 = c5_best
    
    # Ensure final solution meets constraints
    final_h = best_h.copy()
    if np.sum(final_h) < n_points/2 - 1e-3 or np.sum(final_h) > n_points/2 + 1e-3:
        final_h = final_h * (n_points/2) / np.sum(final_h)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
