import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize initial solution to satisfy constraint
    dx = 2.0 / n_points
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # CCP optimization
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Set up iteration parameters
    max_iter = 1000
    tol = 1e-6
    
    # For each iteration, solve convex subproblem in h
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample shifts to find the most problematic one
        shifts = np.linspace(0, 2, 100)
        max_shift = 0
        max_overlap = -1
        
        for shift in shifts:
            k = int(shift / dx)
            if k >= 0 and k < n_points:
                # Compute overlap for this shift
                overlap = np.sum(h * np.roll(1-h, k)) * dx
                if overlap > max_overlap:
                    max_overlap = overlap
                    max_shift = k
        
        # Now solve convex subproblem: minimize sum(h_new * (1 - h_shifted))
        # This uses a linear approximation around current h
        def objective(h_new):
            # Compute overlaps for the worst shift
            h_shifted = np.roll(h, max_shift) if max_shift != 0 else h.copy()
            return np.sum(h_new * (1 - h_shifted)) * dx

        def constraint_sum(h_new):
            return np.sum(h_new) - n_points / 2.0

        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Solve with SLSQP
        try:
            res = minimize(objective, h.copy(), method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_new = res.x
                new_c5 = evaluate_erdos_solution(h_new, dx)
                
                # Accept the new solution if it improves
                if new_c5 < best_c5:
                    h = h_new.copy()
                    best_h = h.copy()
                    best_c5 = new_c5
                    
                    # Early stopping
                    if best_c5 < 0.3807:
                        break
        except:
            pass
            
        # Check for convergence
        if i > 0 and abs(best_c5 - evaluate_erdos_solution(h, dx)) < tol:
            break

    # Final polishing using direct optimization
    try:
        def obj(h_new):
            return evaluate_erdos_solution(h_new, dx)
        def constraint_sum(h_new):
            return np.sum(h_new) - n_points / 2.0

        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    # Final verification
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
