import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize initial solution
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    h = initial_h.copy()
    
    # Initialize tracking variables
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    # Algorithm parameters
    max_iter = 1000
    stagnation_limit = 50
    shift_samples = 100
    step_size = 0.01
    
    # Main optimization loop with CCP-style approach
    stagnation_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Estimate current worst-case shift by sampling
        k_candidates = np.linspace(0, 2, shift_samples)
        max_overlap = -1
        best_k = 0
        
        for k in k_candidates:
            # Compute overlap for this shift
            overlap = 0
            k_int = int(k * n_points / 2)
            k_frac = (k * n_points / 2) - k_int
            
            # Linear interpolation for fractional shifts
            for i in range(n_points):
                j1 = (i + k_int) % n_points
                j2 = (i + k_int + 1) % n_points
                overlap += h[i] * (1 - (h[j1] * (1 - k_frac) + h[j2] * k_frac))
                
            overlap *= 2.0 / n_points
            if overlap > max_overlap:
                max_overlap = overlap
                best_k = k
                
        # Optimize h to minimize this overlap
        # This is a simplified linear approximation approach
        # Use gradient descent with projection
        old_c5 = best_c5
        
        # Gradient approximation
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            overlap_plus = 0
            k_int = int(best_k * n_points / 2)
            k_frac = (best_k * n_points / 2) - k_int
            
            for j in range(n_points):
                k1 = (j + k_int) % n_points
                k2 = (j + k_int + 1) % n_points
                overlap_plus += h_plus[j] * (1 - (h_plus[k1] * (1 - k_frac) + h_plus[k2] * k_frac))
            overlap_plus *= 2.0 / n_points
            
            grad[i] = (overlap_plus - max_overlap) / eps
            
        # Update step
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Recompute C5
        c5 = evaluate_erdos_solution(h, 2.0 / n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Perturb if stuck
        if stagnation_count > stagnation_limit:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0
            
        if best_c5 < 0.3807:
            break
    
    # Final SLSQP refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
