import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a smooth ramp
        h_start = np.linspace(0, 1, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    population_size = 30
    mutation_factor = 0.8
    crossover_prob = 0.8
    max_generations = 500
    
    # Initialize population
    population = []
    fitness = []
    
    # Add initial solution
    population.append(h_start)
    fitness.append(best_c5)
    
    # Add perturbed versions
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Evolution loop
    stagnation_counter = 0
    current_best = min(fitness)
    
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        # Update best
        if sorted_fit[0] < best_c5:
            best_h = sorted_pop[0].copy()
            best_c5 = sorted_fit[0]
            if best_c5 < 0.3807:
                break
        
        # Check stagnation
        if sorted_fit[0] < current_best:
            current_best = sorted_fit[0]
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        if stagnation_counter > 20:
            # Reset with new diversified population
            population = []
            fitness = []
            population.append(sorted_pop[0])
            fitness.append(sorted_fit[0])
            for _ in range(population_size - 1):
                # Randomly create new individuals
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points/2) / np.sum(h)
                population.append(h)
                fitness.append(evaluate_erdos_solution(h, dx))
            stagnation_counter = 0
            continue
            
        # Create new population using differential evolution
        new_pop = [sorted_pop[0], sorted_pop[1]]  # Elitism
        
        for i in range(2, population_size):
            # Select three different individuals
            candidates = list(range(len(sorted_pop)))
            candidates.remove(i)
            a, b, c = random.sample(candidates, 3)
            
            # Mutation
            mutant = sorted_pop[a] + mutation_factor * (sorted_pop[b] - sorted_pop[c])
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            crosspoints = np.random.rand(n_points) < crossover_prob
            if not np.any(crosspoints):
                crosspoints[np.random.randint(0, n_points)] = True
            
            trial = np.where(crosspoints, mutant, sorted_pop[i])
            
            # Repair and rescale
            trial = np.clip(trial, 0, 1)
            if np.sum(trial) == 0:
                trial = np.ones_like(trial) * 0.5
            else:
                trial = trial * (n_points/2) / np.sum(trial)
            
            new_pop.append(trial)
        
        # Evaluate new population
        fitness = [evaluate_erdos_solution(h, dx) for h in new_pop]
        population = new_pop
    
    # Final refinement using local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Try several restarts from best candidates
            for _ in range(5):
                h_init = best_h + np.random.normal(0, 0.01, n_points)
                h_init = np.clip(h_init, 0, 1)
                h_init = h_init * (n_points/2) / np.sum(h_init)
                
                res = minimize(objective, h_init, method='L-BFGS-B',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 100, 'ftol': 1e-9})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
