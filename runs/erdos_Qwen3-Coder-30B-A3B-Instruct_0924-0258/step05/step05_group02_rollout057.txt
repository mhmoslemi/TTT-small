import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    
    # Ensure sum constraint
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    pop_size = 50
    num_generations = 1000
    elite_size = pop_size // 3
    mutation_rate = 0.1
    
    # Population initialization
    population = []
    population.append(best_h.copy())
    
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    # Main GA loop
    generation = 0
    while time() - start_time < budget_s * 0.95 and generation < num_generations:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection
        elite = population[:elite_size]
        
        # Crossover and mutation
        new_population = elite[:]
        
        while len(new_population) < pop_size:
            # Select two parents
            parents = random.sample(elite, 2)
            h1, h2 = parents[0], parents[1]
            
            # Crossover with sparsity bias
            alpha = random.uniform(0.4, 0.6)
            child = alpha * h1 + (1 - alpha) * h2
            
            # Mutation
            if random.random() < mutation_rate:
                noise = np.random.normal(0, 0.02, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            
            # Normalize to satisfy sum constraint
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        generation += 1
    
    # Final refinement with local search using LBFGS-B
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Refine best solution found so far
    best_refined = best_h.copy()
    try:
        res = minimize(
            objective,
            best_refined,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100}
        )
        if res.success:
            h_refined = res.x
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Ensure final constraint satisfaction
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
