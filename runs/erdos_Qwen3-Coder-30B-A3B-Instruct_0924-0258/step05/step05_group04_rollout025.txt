import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    mu = 1e-3

    rho = 1.0
    max_rho = 1000.0

    h = h_start.copy()
    h_prev = h.copy()
    lambda_ = np.zeros(2 * n_points - 1)

    C5_est = best_c5

        current_overlaps = np.correlate(h, 1 - h, mode='full')
        
        # Update dual variable
        lambda_ = lambda_ + rho * (current_overlaps - C5_est * dx)
        
        # New constraints vector
        b = (lambda_ / rho) + C5_est * dx
        
        # Solve convex subproblem using scipy minimize
        def obj(h_new):
            # Cost using proximal term
            prox = 0.5 * np.sum((h_new - h_prev)**2)
            # Penalty on overlap constraints
            overlaps_new = np.correlate(h_new, 1 - h_new, mode='full')
            penalty = 0.5 * mu * np.sum((overlaps_new - b)**2)
            return prox + penalty
            
        def con_sum(h_new):
            return np.sum(h_new) - n_points / 2.0
            
        # Optimization bounds
