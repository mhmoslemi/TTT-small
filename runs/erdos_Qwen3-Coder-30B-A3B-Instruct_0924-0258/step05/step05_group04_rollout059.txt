import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Smooth maximum approximation parameters
    tau = 1.0
    tol = 1e-7
    max_iter = 3000
    
    # Gradient computation helper
    def compute_overlap_and_gradient(h):
        # Compute all cross-correlations for different lags
        h1 = h
        h2 = 1 - h
        corr = np.correlate(h1, h2, mode='full')
        max_k = len(corr) // 2
        
        # The maximum correlation
        max_corr_val = np.max(corr)
        
        # Now compute gradient w.r.t h
        # For smooth approximation of max: log(sum(exp(tau * corr))) / tau
        
        # exp(tau * corr) values for each lag
        exp_tau_corr = np.exp(tau * corr)
        sum_exp = np.sum(exp_tau_corr)
        
        # Gradient of smooth max: (1/tau) * (sum of d/dh(exp(tau * corr)) / sum_exp)
        # d/dh(exp(tau * corr)) = exp(tau * corr) * tau * d/dh(corr)
                
        # Approximate gradient using finite differences for d/dh(corr)
        grad = np.zeros_like(h1)
        eps = 1e-6
        for i in range(len(h1)):
            h_plus = h1.copy()
            h_plus[i] += eps
            corr_plus = np.correlate(h_plus, h2, mode='full')
            corr_minus = np.correlate(h1, h2, mode='full')
            grad[i] = (corr_plus - corr_minus) / eps * tau
        grad = grad / sum_exp * exp_tau_corr / sum_exp
        return max_corr_val, grad
    
    # Optimizer steps
    stagnant = 0
    prev_best = best_c5
    
    # L-BFGS-B optimization
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
    
    for i in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Compute approximate gradient
        try:
            c5_current, grad = compute_overlap_and_gradient(h_current)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
                stagnant = 0
            else:
                stagnant += 1
        except Exception as e:
            pass
            
        # Compute the gradient using numerical differentiation
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        # Adjust gradient smoothing
        if i % 100 == 0 and i > 0:
            tau *= 0.99
            
        # Update using L-BFGS-B
        try:
            res = minimize(obj, h_current, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 10, 'ftol': 1e-9})
            if res.success:
                h_current = res.x
            else:
                # Revert to previous if failed
                pass
        except:
            pass
            
        # Maintain sum constraint
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        h_current = np.clip(h_current, 0, 1)
        
        # Check for stagnation and early exit
        if stagnant > 100:
            if abs(prev_best - best_c5) < 1e-8:
                break
            prev_best = best_c5
            stagnant = 0
            
        # Adaptively change tau if no significant progress
        if i > 0 and i % 50 == 0:
            if abs(prev_best - best_c5) < 1e-6:
                tau *= 1.1
            else:
                prev_best = best_c5
            
    # Final refinement with SLSQP on best solution
    if time() - start_time < max_time * 0.9:
        try:
            h_init = best_h.copy()
            res = minimize(evaluate_erdos_solution, h_init, args=(dx,), method='SLSQP', 
                           bounds=bounds, options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
