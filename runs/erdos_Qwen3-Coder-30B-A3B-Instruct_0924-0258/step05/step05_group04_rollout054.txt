import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM Parameters
    max_iter = 5000
    mu = 1e-3
    tol = 1e-6
    rho = 1.0
    
    # Initialize variables
    h = h_start.copy()
    h_prev = h.copy()
    z = np.zeros(2 * n_points - 1)
    lambda_vec = np.zeros(2 * n_points - 1)
    
    # Compute initial correlation
    corr = np.correlate(h, 1 - h, mode='full')
    
    # Prioritize speed and use efficient ADMM approach
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Update h by solving: argmin_h ||h - y||^2 + mu * ||A h - z + lambda||^2
        # subject to 0 <= h <= 1 and sum(h) = n_points / 2
        
        # For simplicity, solve a projected quadratic subproblem
        def obj(h):
            # Quadratic penalty for deviation from (1-h) correlation
            H_shifted = np.correlate(h, 1 - h, mode='full')
            # Approximate objective: minimize deviation from correlation threshold
            return np.sum((H_shifted - z + lambda_vec)**2) * mu
            
        def obj_with_constraints(h):
            # Regularized objective with hard constraints
            H_shifted = np.correlate(h, 1 - h, mode='full')
            reg = np.sum((H_shifted - z + lambda_vec)**2) * mu
            # Penalty for boundary violations and sum constraint
            penalty = 0
            penalty += 1e6 * max(0, np.sum(h) - n_points / 2)**2
            penalty += 1e6 * max(0, np.sum(h) - n_points / 2)**2
            for i in range(n_points):
                if h[i] < 0:
                    penalty += 1e6 * (0 - h[i])**2
                if h[i] > 1:
                    penalty += 1e6 * (h[i] - 1)**2
            return reg + penalty
        
        # Solve projected subproblem using L-BFGS-B
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
        
        try:
            res = minimize(obj_with_constraints, h.copy(), method='L-BFGS-B', 
                           bounds=bounds, constraints=cons, options={'maxiter': 50})
            if res.success:
                h_new = res.x
                # Ensure normalization (sum = n_points / 2)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h_prev = h.copy()
                h = h_new
            else:
                # fallback to previous iteration
                pass
        except:
            # fallback to previous iteration
            pass
            
        # Update auxiliary variables z and lambda
        corr = np.correlate(h, 1 - h, mode='full')
        z_new = z + rho * (corr - z + lambda_vec)
        z = z_new
        lambda_vec = lambda_vec + rho * (corr - z)

        # Check convergence criteria
        norm_residual = np.linalg.norm(corr - z)
        if norm_residual < tol:
            break
            
        # Check if we've improved the solution
        c5_current = evaluate_erdos_solution(h, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # Early stopping
        if best_c5 < 0.3807:
            break

    # Final refinement with SLSQP
    h_final = best_h.copy()
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    def con_sum(h):
        return np.sum(h) - n_points / 2.0
    cons = {'type': 'eq', 'fun': con_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, h_final, method='SLSQP', bounds=bounds, 
                       constraints=cons, options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            h_final = res.x
            c5_final = obj(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    # Final validity check
    if np.sum(best_h) < n_points * 0.499 or np.sum(best_h) > n_points * 0.501:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
