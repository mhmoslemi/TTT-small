import numpy as np
from cvxpy import *
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 35
        h_init = np.full(n_points, 0.5)
    
    # Scale to unit integral
    dx = 2.0 / n_points
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Sampling shifts
    shift_samples = np.linspace(0, 2, 100)
    
    # Print starting C5
    c5_current = evaluate_erdos_solution(h_init, dx)
    
    # Convex relaxation approach
    h_vars = Variable(n_points)
    c5_var = Variable()
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        sum(h_vars) == n_points / 2,
        c5_var >= 0
    ]
    
    # Cutting plane method
    iteration = 0
    max_iter = 50
    stagnation_count = 0
    last_c5 = c5_current
    
    h_best = h_init.copy()
    c5_best = c5_current
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iter:
        # Compute overlaps for current h
        overlaps = []
        for k in shift_samples:
            k_idx = int(k / dx)
            if k_idx < n_points:
                overlap = np.sum(h_vars.value * (1 - np.roll(h_vars.value, k_idx)))
                overlaps.append(overlap)
            else:
                # Handle boundary
                k_idx %= n_points
                overlap = np.sum(h_vars.value * (1 - np.roll(h_vars.value, k_idx)))
                overlaps.append(overlap)
        
        # Add constraint from maximum overlap
        max_overlap_idx = np.argmax(overlaps)
        k_opt = shift_samples[max_overlap_idx]
        k_idx = int(k_opt / dx)
        if k_idx < n_points:
            overlap_expr = h_vars @ (1 - np.roll(h_vars, k_idx))
        else:
            k_idx %= n_points
            overlap_expr = h_vars @ (1 - np.roll(h_vars, k_idx))
        constraints.append(overlap_expr <= c5_var)
        
        # Solve current convex relaxation
        objective = Minimize(c5_var)
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_current = h_vars.value
                c5_current = c5_var.value
                
                # Ensure constraints are satisfied
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points / 2) / np.sum(h_current)
                
                # Update best solution
                if c5_current < c5_best:
                    c5_best = c5_current
                    h_best = h_current.copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                
                # Terminate early if target hit
                if c5_best < 0.3807:
                    break
            else:
                break
        except:
            break
        iteration += 1

        # Early termination on stagnation
        if stagnation_count > 50:
            break
    
    # Final local refinement using SLSQP
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, h_best, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            h_best = res.x
            c5_best = obj(h_best)
    except:
        pass
    
    return (list(h_best), c5_best, n_points)
