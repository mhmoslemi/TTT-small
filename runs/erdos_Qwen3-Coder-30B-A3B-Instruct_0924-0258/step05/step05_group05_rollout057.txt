import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a smooth starting function
        h_start = np.linspace(0, 1, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    pop_size = 30
    mutation_factor = 0.8
    crossover_rate = 0.8
    max_gens = 500
    tv_weight = 0.01
    
    # Initialize population
    population = []
    fitness = []
    
    # Create initial population with perturbed versions of the best
    for i in range(pop_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Evolution loop
    stagnation_counter = 0
    current_best = min(fitness)
    
    for gen in range(max_gens):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        if sorted_fit[0] < best_c5:
            best_h = sorted_pop[0].copy()
            best_c5 = sorted_fit[0]
            if best_c5 < 0.3807:
                break
                
        # Check stagnation
        if sorted_fit[0] < current_best:
            current_best = sorted_fit[0]
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Create new population with DE
        new_pop = [sorted_pop[0], sorted_pop[1]]  # Elitism
        
        for i in range(2, pop_size):
            # Select three distinct parents
            idxs = random.sample(range(len(sorted_pop)), 3)
            a, b, c = sorted_pop[idxs[0]], sorted_pop[idxs[1]], sorted_pop[idxs[2]]
            
            # Mutation
            mutant = a + mutation_factor * (b - c)
            # Clip to valid range
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            trial = np.copy(sorted_pop[i])
            for j in range(n_points):
                if random.random() < crossover_rate:
                    trial[j] = mutant[j]
            
            # Repair and enforce sum constraint
            trial = trial * (n_points/2) / np.sum(trial)
            
            new_pop.append(trial)
        
        # Local refinement every 10 generations
        if gen % 10 == 0 and len(new_pop) >= 5:
            for i in range(min(5, len(new_pop))):
                h_init = new_pop[i]
                
                def objective(h):
                    val = evaluate_erdos_solution(h, dx)
                    # Add total variation penalty for smoothness
                    tv = np.sum(np.abs(np.diff(h))) * tv_weight
                    return val + tv
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_init, method='L-BFGS-B', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100})
                    if res.success:
                        h_refined = res.x
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < fitness[sorted_indices[i]]:
                            new_pop[i] = h_refined
                except:
                    pass
        
        # Evaluate new population
        fitness = [evaluate_erdos_solution(h, dx) for h in new_pop]
        population = new_pop
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
