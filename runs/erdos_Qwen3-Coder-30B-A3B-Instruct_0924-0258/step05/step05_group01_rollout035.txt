import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 50
    max_gen = 1000
    mutation_rate = 0.15
    elite_size = 5
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = best_h.copy()
        else:
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evolutionary loop
    gen = 0
    no_improvement = 0
    max_no_improvement = 50
    
    while time() - start_time < budget_s * 0.95 and gen < max_gen:
        # Evaluate fitness
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > max_no_improvement:
            # Reset with better starting point
            h_reset = np.random.uniform(0, 1, n_points)
            h_reset = h_reset * (n_points/2) / np.sum(h_reset)
            population = [h_reset] + population[1:]
            no_improvement = 0
        
        # Selection (tournament)
        new_pop = population[:elite_size]  # Keep elites
        
        while len(new_pop) < pop_size:
            # Tournament selection
            tourn_size = 3
            selected_indices = np.random.choice(pop_size, tourn_size)
            selected_fitness = [fitness[i] for i in selected_indices]
            winner_idx = selected_indices[np.argmin(selected_fitness)]
            
            # Crossover
            if np.random.random() < 0.8:
                # Select second parent
                selected_indices = np.random.choice(pop_size, tourn_size)
                selected_fitness = [fitness[i] for i in selected_indices]
                winner_idx2 = selected_indices[np.argmin(selected_fitness)]
                
                # One-point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([population[winner_idx][:crossover_point],
                                      population[winner_idx2][crossover_point:]])
            else:
                child = population[winner_idx].copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                mutation_indices = np.random.choice(n_points, 
                                                   int(n_points * mutation_rate), 
                                                   replace=False)
                for idx in mutation_indices:
                    child[idx] += np.random.normal(0, 0.05)
            
            # Ensure feasibility
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_pop.append(child)
        
        population = new_pop
        gen += 1
        
        # Multi-resolution
        if gen % 100 == 0 and n_points < 50:
            if n_points == 35:
                n_points = 40
            elif n_points == 40:
                n_points = 50
            
            dx = 2.0 / n_points
            # Interpolate current best to new resolution
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_points)
            best_h = np.interp(x_new, x_old, best_h)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
            
            # Reset population with new resolution
            population = []
            for i in range(pop_size):
                if i == 0:
                    h = best_h.copy()
                else:
                    h = np.random.uniform(0, 1, n_points)
                    h = h * (n_points/2) / np.sum(h)
                population.append(h)
    
    # Final validation at best solution
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
