import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Sequential Convex Programming approach
    max_iter = 500
    mu = 1e-3
    c5_nih = 0.4  # initial C5 estimate
    
    # For iterative tightening
    converged = False
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        current_c5 = evaluate_erdos_solution(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
            
        if best_c5 < 0.3807 or best_c5 < 0.38085:
            break
            
        # Calculate overlaps for all shifts
        h_shifted = np.zeros((n_points, n_points))
        for k in range(n_points):
            h_shifted[:, k] = np.roll(best_h, k)
            
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            overlaps[k] = np.sum(best_h * (1 - h_shifted[:, k])) * dx
            
        # Find maximum overlap index
        max_k = np.argmax(overlaps)
        max_overlap_val = overlaps[max_k]
        
        # If we are within tolerance or doing well, continue
        if abs(max_overlap_val - best_c5) < 1e-6:
            break
        
        # Build QP based on current h estimate
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(c5_nih)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2.0,
            h_var >= 0,
            h_var <= 1,
        ]
        
        # Add the constraint that the maximum overlap is bounded by c5_nih
        # Linearization around current best_h:
        # sum(h * (1 - h_k)) <= c5_nih
        # Using the shifted version and linear approximation:
        # sum(h * (1 - h_k)) ≈ sum(h_current * (1 - h_k)) + sum((h - h_current) * -h_k)
        # For each k, we build that constraint:
        for k in range(n_points):
            h_k_current = np.roll(best_h, k)
            linear_term = np.sum(best_h * (1 - h_k_current))
            linearized_constraint = cp.sum(cp.multiply(h_var, (1 - h_k_current))) <= c5_nih
            constraints.append(linearized_constraint)
        
        # Solve the QP
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.OSQP, warm_start=True, verbose=False, 
                       eps_abs=1e-6, eps_rel=1e-6, max_iter=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    best_h = h_new.copy()
                    c5_nih = c5_nih * 0.99  # Slight tightening
                    continue
        except:
            pass
            
        # Fallback: local refinement with SLSQP
        try:
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            def con(h):
                return np.sum(h) - n_points / 2.0
                
            cons = {'type': 'eq', 'fun': con}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 30, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = obj(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
        
        # Update the guess slightly
        c5_nih = max(best_c5, 0.9 * c5_nih)
    
    # Final SLSQP refinement with tight tolerance
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 20:
        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
