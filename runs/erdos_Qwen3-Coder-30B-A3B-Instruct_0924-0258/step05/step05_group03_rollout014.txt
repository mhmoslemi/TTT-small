import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Ensure constraints are met
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Gradient-based optimization with multi-start
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    # Parameters
    max_iters = 500
    patience = 50
    stagnation_count = 0
    n_samples = 50  # Number of shifts to sample for gradient estimation
    
    for iteration in range(max_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample shifts
        shifts = np.random.choice(n_points, n_samples, replace=False)
        
        # Find the most problematic shift
        max_overlap = -1
        best_shift = 0
        for k in shifts:
            # Compute overlap manually for k
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += best_h[i] * (1 - best_h[j])
            overlap *= 2.0 / n_points
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = k
                
        # Compute gradient for this shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_shift) % n_points
            grad[i] += 1 - best_h[j]
            grad[j] -= best_h[i]
        grad *= 2.0 / n_points
        
        # Line search for step size
        alpha = 1.0
        step_size = 0.01
        while alpha > 1e-10:
            new_h = best_h - alpha * grad
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_c5 = evaluate_erdos_solution(new_h, 2.0 / n_points)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
                stagnation_count = 0
                break
            alpha *= 0.5
        
        # If no improvement, restart with small noise
        if alpha <= 1e-10 or stagnation_count > patience:
            perturbation = np.random.normal(0, 0.01, n_points)
            best_h = best_h + perturbation
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Early termination
        if best_c5 < 0.3807:
            break
    
    # Final local refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
