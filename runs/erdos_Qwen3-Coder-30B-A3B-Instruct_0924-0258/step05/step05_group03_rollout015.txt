import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
        
    # CCP approach for optimization
    h = initial_h.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    # Iteration parameters
    max_iters = 500
    tolerance = 1e-6
    
    # Generate shift values to sample
    shift_vals = np.linspace(0, 2, 100)
    
    for iteration in range(max_iters):
        # Early termination if target achieved
        if best_c5 < 0.3807:
            break
            
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find worst shift k
        max_corr = -np.inf
        worst_k = 0
        
        for k in shift_vals:
            # Compute correlation for shift k
            # h(x) * (1 - h(x + k))
            # This is a discrete convolution problem
            # We can use direct computation for this
            corr = 0.0
            dx = 2.0 / n_points
            for i in range(n_points):
                j = i + int(k * n_points / 2)
                if 0 <= j < n_points:
                    corr += h[i] * (1 - h[j]) * dx
                else:
                    # Handle boundary conditions
                    pass
                    
            if corr > max_corr:
                max_corr = corr
                worst_k = k
                
        # Now solve convex problem for optimal h given the worst shift
        # We want to minimize sum h[i] * (1 - h[i + worst_k])
        # subject to 0 <= h[i] <= 1 and sum(h) = n_points / 2
        # since we are optimizing linearly in h, with linear constraints
        # we can solve this efficiently by finding the support set
        # and solving for the optimal values
        
        # Alternative approach using scipy to solve the constrained problem
        # We directly try to solve the dual problem or use Lagrange multipliers
        
        # Simple approach: solve with SLSQP using worst_k to set vector
        def obj(h_vec):
            dx = 2.0 / n_points
            return np.sum(h_vec * (1 - np.roll(h_vec, int(worst_k * n_points / 2)))) * dx
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, h.copy(), method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_new = res.x
                # Ensure the solution is valid
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                new_c5 = evaluate_erdos_solution(h_new, 2.0 / n_points)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                    h = h_new.copy()
            else:
                # fallback to projected gradient method
                pass
        except:
            pass
            
        # Check for convergence
        if iteration % 20 == 0:
            # Periodic check for improvement
            current_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
            if abs(current_c5 - best_c5) < tolerance:
                break
                
    # Final polishing with scipy minimize
    try:
        def obj(h_vec):
            return evaluate_erdos_solution(h_vec, 2.0 / n_points)
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h.copy(), method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    except:
        pass
        
    return (list(best_h), best_c5, n_points)
