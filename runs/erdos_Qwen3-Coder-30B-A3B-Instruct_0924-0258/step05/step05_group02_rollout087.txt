import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Initialize with the best known values or simple uniform
    dx = 2.0 / n_points
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based local search
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    max_iters = 500
    iter_count = 0
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iters:
        iter_count += 1
        
        # Local optimization step
        h_perturbed = h_best.copy()
        h_perturbed += np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                objective, 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-7}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < c5_best:
                    h_best = h_refined
                    c5_best = c5_refined
                    last_improvement = 0
                else:
                    last_improvement += 1
            else:
                last_improvement += 1
        except Exception:
            last_improvement += 1
        
        # Perturb and try to escape local minima
        if last_improvement > 5:
            h_perturbed = h_best.copy()
            noise = np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed + noise, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < c5_best:
                h_best = h_perturbed
                c5_best = c5_perturbed
                last_improvement = 0
        
        # Occasionally restart with a new random solution
        if np.random.rand() < 0.05 or last_improvement > 20:
            h_restart = np.random.rand(n_points)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            h_restart = np.clip(h_restart, 0, 1)
            c5_restart = evaluate_erdos_solution(h_restart, dx)
            if c5_restart < c5_best:
                h_best = h_restart
                c5_best = c5_restart
                last_improvement = 0
                
        if time() - start_time > budget_s * 0.98:
            break
    
    # Final adjustment
    h_final = h_best.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    if c5_final < c5_best:
        h_best = h_final
        c5_best = c5_final
    
    return (list(h_best), c5_best, n_points)
