import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a symmetric triangular initialization
        mid = n_points // 2
        h_start = np.minimum(np.arange(n_points), 2*mid - np.arange(n_points))
        h_start = h_start / (mid)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Log for monitoring
    best_history = [best_c5]
    iter_count = 0
    
    # Main optimization loop
    while time() - start_time < budget_s * 0.95:
        iter_count += 1
        current_time = time()
        if current_time - start_time > budget_s * 0.95:
            break

        # First try local optimization with L-BFGS-B
        def objective(h):
            return evaluate_erdos_solution(h, dx)

        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0

        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]

        try:
            res = minimize(objective, best_h, method='L-BFGS-B',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass

        # Next try randomized perturbation with feasible normalization
        h_perturbed = best_h + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
        if c5_perturbed < best_c5:
            best_h = h_perturbed
            best_c5 = c5_perturbed

        # Store best to history
        best_history.append(best_c5)
        if len(best_history) > 50:
            best_history.pop(0)
        
        if best_c5 < 0.3807:
            break

        # Adaptive strategy
        if iter_count % 10 == 0:
            # If stuck, increase perturbations
            if len(best_history) >= 10 and all(abs(x - best_history[0]) < 1e-6 for x in best_history[-10:]):
                # Add larger random perturbations
                for i in range(n_points):
                    if np.random.rand() < 0.3:
                        best_h[i] += np.random.normal(0, 0.03)
                best_h = np.clip(best_h, 0, 1)
                best_h = best_h * (n_points/2) / np.sum(best_h)
                best_c5 = evaluate_erdos_solution(best_h, dx)

    # Final check and refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)

            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0

            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, best_h, method='L-BFGS-B',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass

    return (list(best_h), best_c5, n_points)
