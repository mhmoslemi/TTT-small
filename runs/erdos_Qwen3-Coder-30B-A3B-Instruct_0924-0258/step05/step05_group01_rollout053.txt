import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create an initial triangular-like function
        x = np.linspace(0, 2, 35)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start / np.sum(h_start) * 17.5  # scale to sum to 17.5 (n_points/2)
        n_points = 35
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Scales to iterate through
    scales = [35, 50, 75, 100]
    
    for scale in scales:
        if time() - start_time > budget_s * 0.95:
            break
            
        if scale > n_points:
            # Interpolate up to higher resolution
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            h_new = np.interp(x_new, x_orig, best_h)
            n_points = scale
        else:
            h_new = best_h
        
        dx = 2.0 / n_points
        
        # Try multiple local optimizations
        for restart in range(10):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Randomly perturb
            h_perturbed = h_new + np.random.normal(0, 0.02, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            
            # Optimize using L-BFGS-B
            try:
                bounds = [(0, 1) for _ in range(n_points)]
                result = minimize(
                    lambda h: evaluate_erdos_solution(h, dx),
                    h_perturbed,
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 150, 'ftol': 1e-9}
                )
                
                if result.success:
                    h_candidate = result.x
                    c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
            except Exception:
                continue
                
        # Store current best
        h_new = best_h.copy()
    
    # Final polishing at highest resolution
    if time() - start_time < budget_s * 0.1:
        try:
            # Do one final L-BFGS-B optimization at highest resolution
            bounds = [(0, 1) for _ in range(n_points)]
            result = minimize(
                lambda h: evaluate_erdos_solution(h, dx),
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 200, 'ftol': 1e-10}
            )
            
            if result.success:
                h_final = result.x
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
