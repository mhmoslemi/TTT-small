import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a symmetric triangular function as initial guess
        mid = n_points // 2
        h_start = np.minimum(np.arange(n_points), 2*mid - np.arange(n_points))
        h_start = h_start / (mid)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization
    max_iterations = 100
    tolerance = 1e-8
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Local optimization step using L-BFGS-B
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': tolerance})
            if res.success:
                h_refined = res.x
                # Enforce constraints
                h_refined = np.clip(h_refined, 0, 1)
                if np.sum(h_refined) != 0:
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
                    if best_c5 < 0.3807:
                        break
        except:
            pass
        
        # Random perturbation to escape local minima
        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        if np.sum(h_perturbed) != 0:
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
        if c5_perturbed < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_perturbed
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': tolerance})
            if res.success:
                h_final = res.x
                h_final = np.clip(h_final, 0, 1)
                if np.sum(h_final) != 0:
                    h_final = h_final * (n_points / 2) / np.sum(h_final)
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
