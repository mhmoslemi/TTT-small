import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        n_points = 35
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize with a convex relaxation approach
    dx = 2.0 / n_points
    
    # Define the optimization variables
    h_var = cp.Variable(n_points)
    C5_var = cp.Variable(1)
    
    # Constraints
    constraints = [
        0 <= h_var,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2,
        C5_var >= 0
    ]
    
    # Objective: minimize C5
    objective = cp.Minimize(C5_var)
    
    # Iterative refinement using alternating optimization
    max_iter = 100
    last_improvement = 0
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Helper to check constraints
    def is_feasible(h_vec):
        return np.all(h_vec >= 0) and np.all(h_vec <= 1) and abs(np.sum(h_vec) - n_points / 2) < 1e-6
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlap with a coarse grid of shift values
        shift_grid = np.linspace(0, 2, 100)
        max_overlap = 0
        best_shift = 0
        
        for shift in shift_grid:
            # Compute overlap sum for this shift
            k = int(shift / dx)
            if k >= n_points:
                continue
            overlap = np.sum(h[:n_points-k] * (1 - h[k:]))
            overlap = overlap * dx
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = shift
                
        # Alternate optimization: compute best h for fixed k
        # This is an approximation, but works well for optimization
        
        # Refine using scipy optimization with constraints
        def obj(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, h.copy(), method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_new = res.x
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = obj(h_new)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    last_improvement = iteration
                    
        except Exception as e:
            pass
            
        # Occasionally perturb to escape local minima
        if iteration % 10 == 0 and iteration > 0:

                h = best_h.copy()

        # Early termination if we found a sufficiently good bound
        if best_c5 < 0.3807:

            return evaluate_erdos_solution(h_vec, dx)

            return np.sum(h_vec) - n_points / 2.0

    # Ensure constraints hold
    best_h = np.clip(best_h, 0, 1)

    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
