import numpy as np
from cvxpy import *
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 35
        h0 = np.ones(n_points) * 0.5
    
    # Normalize initial solution
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Set up optimization
    h = Variable(n_points)
    C5 = Variable()
    
    # Define constraints
    constraints = [h >= 0, h <= 1, sum(h) == n_points / 2, C5 >= 0]
    
    # Parameters
    max_iterations = 50
    shift_samples = 100
    shifts = np.linspace(0, 2, shift_samples, endpoint=False)
    
    # Initialize best solution
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Cutting plane approach
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps with current h
        overlaps = []
        for k in shifts:
            k_int = int(k * n_points / 2)
            if k_int == 0:
                corr = np.correlate(h0, 1 - h0, mode='full')
            else:
                # Use moving window for correlation
                corr = np.correlate(h0, 1 - h0, mode='full')
            overlap_val = np.max(corr) * 2.0 / n_points
            overlaps.append(overlap_val)
        
        # Find maximum overlap
        max_overlap_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_overlap_idx]
        
        # Add constraint to model
        constraints.append(C5 >= max_overlap)
        
        # Formulate and solve problem
        try:
            prob = Problem(Minimize(C5), constraints)
            prob.solve(solver=ECOS, verbose=False)
            
            if prob.status == 'optimal':
                current_c5 = prob.value
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.value.copy()
                
                # Check for early termination
                if current_c5 < 0.3807:
                    break
            else:
                # Try standard approach with SLSQP
                def obj(x):
                    return evaluate_erdos_solution(x, 2.0 / n_points)
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
                
                res = minimize(obj, h0, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h0 = res.x
                else:
                    h0 = best_h
        except:
            # Fallback to local optimization
            def obj(x):
                return evaluate_erdos_solution(x, 2.0 / n_points)
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
            
            res = minimize(obj, h0, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                h0 = res.x
            else:
                h0 = best_h
        
        # Update h0 for next iteration
        h0 = best_h.copy()
    
    # Final local refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
        
        res = minimize(obj, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    # Ensure final bounds
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
