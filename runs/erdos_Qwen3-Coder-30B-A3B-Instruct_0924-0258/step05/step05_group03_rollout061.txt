import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with available initial construction or create default
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize to correct sum
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    h = initial_h.copy()
    
    # Parameters
    max_iter = 500
    stagnation_limit = 50
    shift_samples = 50
    learning_rate = 0.01
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample shifts to approximate real shifts
        shifts = np.random.choice(n_points, size=shift_samples, replace=False)
        
        # Evaluate overlap for each shift and find maximum
        max_overlap = -1
        best_shift = 0
        for k in shifts:
            # Compute overlap for shift k
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlap *= 2.0 / n_points
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = k
        
        # Compute gradient for this shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_shift) % n_points
            grad[i] += 1 - h[j]
            grad[j] -= h[i]
        grad *= 2.0 / n_points
        
        # Line search to find appropriate step size
        alpha = learning_rate
        old_c5 = best_c5
        while alpha > 1e-10:
            new_h = h - alpha * grad
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_c5 = evaluate_erdos_solution(new_h, 2.0 / n_points)
            if new_c5 < old_c5:
                h = new_h
                best_c5 = new_c5
                break
            alpha *= 0.5
            
        # Update best solution
        if best_c5 < evaluate_erdos_solution(best_h, 2.0 / n_points):
            best_h = h.copy()
            best_c5 = best_c5
            
        # Check for stagnation
        if abs(best_c5 - old_c5) < 1e-8:
            stagnation_count += 1
            if stagnation_count > stagnation_limit:
                # Perturb the solution
                h += np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                stagnation_count = 0
        else:
            stagnation_count = 0
            
        # Early termination
        if best_c5 < 0.3807:
            break
    
    # Final optimization
    def obj(x):
        return evaluate_erdos_solution(x, 2.0 / n_points)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            final_h = res.x
            final_c5 = obj(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
        
    return (list(best_h), best_c5, n_points)
