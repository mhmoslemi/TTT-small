import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():

        n_points = len(h)

    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    # Gradient-based optimization
    max_iterations = 500
    stagnation_limit = 50
    stagnation_count = 0
    
    # Sample shifts for gradient computation
    shift_samples = min(50, n_points)
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample shifts
        shifts = random.sample(range(n_points), shift_samples)
        
        # Evaluate overlaps for sampled shifts
        max_overlap = -1
        best_shift = 0
        for k in shifts:
            # Calculate overlap for shift k
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlap *= 2.0 / n_points
            
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = k
        
        # Compute gradient for the worst shift
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_shift) % n_points
            gradient[i] += 1 - h[j]
            gradient[j] -= h[i]
        gradient *= 2.0 / n_points
        
        # Line search for step size
        alpha = 1.0
        min_alpha = 1e-10
        max_alpha = 1.0
        step_size = 0.5
        
        while alpha >= min_alpha:
            # Try step
            h_new = h - alpha * gradient
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, 2.0 / n_points)
            
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                stagnation_count = 0
                break
            else:
                alpha *= step_size
                
        # Stagnation check
        if best_c5 < best_c5:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        # Restart if stagnated
        if stagnation_count > stagnation_limit:
            h = np.ones(n_points) * 0.5
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0
            
        # Early termination
        if best_c5 < 0.3807:
            break
            
    # Final refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, h, method='SLSQP', bounds=bounds, 
                       constraints=cons_sum, options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            h = res.x
            best_c5 = obj(h)
    except:
        pass
    
    return (list(h), best_c5, n_points)
