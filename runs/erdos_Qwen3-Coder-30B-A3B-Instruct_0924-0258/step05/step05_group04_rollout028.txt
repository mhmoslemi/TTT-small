import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    max_iter = 5000
    mu = 1e-3
    tol = 1e-6
    rho = 1.0
    
    # Initialize variables
    h = h_start.copy()
    h_prev = h.copy()
    z = np.zeros(2 * n_points - 1)
    lambda_ = np.zeros(2 * n_points - 1)
    
    # Precompute shifted versions
    ones = np.ones(n_points)
    
    # ADMM iterations
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Update h using proximal operator
        # minimize ||h - h_prev||^2 + mu * ||z - h * (1-h) + lambda||^2
        # subject to 0 <= h <= 1, sum(h) = n_points / 2
        
        # Construct matrix and vector for quadratic optimization
        # We approximate with a quadratic form that linearizes constraints
        def obj(h_vec):
            # Quadratic penalty version of the problem
            overlap = np.correlate(h_vec, 1 - h_vec, mode='full')
            obj_val = np.sum((overlap - z + lambda_) ** 2) * mu * 0.5
            # Add constraint penalty for sum(h) = n_points/2, using soft constraint
            obj_val += 1e3 * (np.sum(h_vec) - n_points / 2) ** 2
            # Add bound penalties for h in [0,1]
            obj_val += 1e3 * (np.sum(np.maximum(0, -h_vec)) + 
                              np.sum(np.maximum(0, h_vec - 1)))
            return obj_val
            
        # Optimization with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        def con(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': con}
        
        try:
            res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                # Ensure sum constraint more precisely
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            else:
                h = h_prev
        except:
            h = h_prev
            
        # Update z
        overlap = np.correlate(h, 1 - h, mode='full')
        z = overlap + lambda_ / mu
        
        # Update lambda
        lambda_ += mu * (overlap - z)
        
        # Check convergence
        if iteration % 50 == 0:
            residual = np.linalg.norm(overlap - z)
            if residual < tol:
                break
        
        # Update h_prev
        h_prev = h.copy()
        
        # Update best solution
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Early stopping
        if best_c5 < 0.3807:
            break
    
    # Final local refinement
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 20:
        h_init = best_h.copy()
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 300, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
