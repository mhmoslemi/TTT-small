import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Start with coarse grid optimization
        n_points = 20
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to correct sum
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    h_start = np.clip(h_start, 0, 1)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid optimization if needed
    if n_points < 35:
        # Upsample and refine
        h_upsampled = np.interp(np.linspace(0, 2, 35), np.linspace(0, 2, n_points), h_start)
        h_upsampled = h_upsampled * 17.5 / np.sum(h_upsampled)
        h_upsampled = np.clip(h_upsampled, 0, 1)
        best_h = h_upsampled
        best_c5 = evaluate_erdos_solution(h_upsampled, 2.0/35)
        n_points = 35
    
    # Iterative refinement
    max_iter = 100
    iter_count = 0
    no_improvement_count = 0
    constraint_epsilon = 1e-6
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Reoptimize with current best
        h_current = best_h.copy()
        
        def objective(h):
            return evaluate_erdos_solution(h, 2.0/n_points)
            
        def constraint_sum(h):
            return np.sum(h) - 17.5
            
        def constraint_bounds(h):
            return np.concatenate([h, 1 - h])
            
        # Create bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try local optimization
        try:
            res = minimize(
                objective,
                h_current,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, 2.0/n_points)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
        except Exception:
            no_improvement_count += 1
            
        # Perturb solution if no improvement
        if no_improvement_count > 50:
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * 17.5 / np.sum(h_perturbed)
            c5_perturbed = evaluate_erdos_solution(h_perturbed, 2.0/n_points)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
            no_improvement_count = 0
            
        iter_count += 1
        
        # Early stop condition
        if no_improvement_count > 100:
            break
    
    # Final check and refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * 17.5 / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, 2.0/n_points)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
