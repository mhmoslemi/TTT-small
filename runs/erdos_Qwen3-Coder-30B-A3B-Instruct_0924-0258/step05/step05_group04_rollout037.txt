import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    num_particles = 40
    max_iter = 500
    inertia_weight = 0.7
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.2
    warm_restart_after = 200
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    for i in range(num_particles):
        if i < len(h_start):
            h = h_start.copy()
        else:
            h = np.random.random(n_points)
            h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_fitness.append(evaluate_erdos_solution(h, dx))
    
    # Find global best
    best_idx = np.argmin(pbest_fitness)
    gbest = pbest[best_idx].copy()
    gbest_fitness = pbest_fitness[best_idx]
    
    stagnation_count = 0
    
    # PSO iterations
    for iter_num in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        if stagnation_count >= warm_restart_after:
            # Warm restart: reset velocities and perturb half the particles
            for i in range(num_particles):
                velocities[i] = np.zeros(n_points)
                if i < num_particles // 2:
                    # Perturb
                    particles[i] += np.random.normal(0, 0.01, n_points)
                    # Enforce constraints
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                    
            stagnation_count = 0
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h > 1e-10:
                particles[i] = particles[i] * (n_points / 2) / sum_h
            else:
                particles[i] = np.ones(n_points) * (n_points / 2)
            
            # Evaluate fitness
            fit = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if fit < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fit
                
                # Update global best
                if fit < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = fit
                    stagnation_count = 0
            else:
                stagnation_count += 1
    
    # Local refinement of top particles
    for i in range(min(5, num_particles)):
        h_refined = pbest[i].copy()
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        def con(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': con}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, h_refined, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_new = res.x
                f_new = obj(h_new)
                if f_new < pbest_fitness[i]:
                    pbest[i] = h_new
                    pbest_fitness[i] = f_new
                    if f_new < gbest_fitness:
                        gbest = h_new
                        gbest_fitness = f_new
        except:
            pass
    
    if gbest_fitness < best_c5:
        best_h = gbest
        best_c5 = gbest_fitness

    return (list(best_h), best_c5, n_points)
