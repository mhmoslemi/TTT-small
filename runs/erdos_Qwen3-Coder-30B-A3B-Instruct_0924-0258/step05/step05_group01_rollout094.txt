import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Convert to binary if needed, ensuring sum = n/2
    if np.allclose(h_start, np.round(h_start)):
        h_binary = np.round(h_start).astype(int)
    else:
        h_binary = np.zeros(n_points, dtype=int)
        # Set exactly half to 1
        ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h_binary[ones_indices] = 1
    
    dx = 2.0 / n_points
    best_h = h_binary.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Tabu search parameters
    tabu_tenure = 10
    tabu_list = []
    max_no_improvement = 100
    no_improvement_count = 0
    shaking_frequency = 50
    
    # Neighborhood function: flip one 1 to 0 and one 0 to 1
    def get_neighbors(h):
        neighbors = []
        ones_indices = np.where(h == 1)[0]
        zeros_indices = np.where(h == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            return neighbors
            
        for i in ones_indices:
            for j in zeros_indices:
                # Create neighbor by flipping bit i and j
                neighbor = h.copy()
                neighbor[i] = 0
                neighbor[j] = 1
                neighbors.append(neighbor)
        return neighbors
    
    # Ensure the best solution has correct sum
    while np.sum(best_h) != n_points // 2:
        if np.sum(best_h) > n_points // 2:
            # Too many 1s - turn some to 0
            ones_indices = np.where(best_h == 1)[0]
            to_flip = len(ones_indices) - n_points // 2
            flip_indices = np.random.choice(ones_indices, size=to_flip, replace=False)
            for idx in flip_indices:
                best_h[idx] = 0
        else:
            # Too few 1s - turn some to 1
            zeros_indices = np.where(best_h == 0)[0]
            to_flip = n_points // 2 - np.sum(best_h)
            flip_indices = np.random.choice(zeros_indices, size=int(to_flip), replace=False)
            for idx in flip_indices:
                best_h[idx] = 1
    
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    iteration = 0
    while time() - start_time < budget_s * 0.95:
        # Generate neighbors
        neighbors = get_neighbors(best_h)
        
        if not neighbors:
            break
        
        # Evaluate neighbors
        neighbor_scores = []
        for neighbor in neighbors:
            score = evaluate_erdos_solution(neighbor, dx)
            neighbor_scores.append(score)
        
        # Find best neighbor that's not tabu or is aspiration
        best_idx = np.argmin(neighbor_scores)
        best_neighbor = neighbors[best_idx]
        best_score = neighbor_scores[best_idx]
        
        # Check if neighbor is in tabu list
        move = tuple(sorted([np.where(best_h == 1)[0][0], np.where(best_h == 0)[0][0]]))
        is_tabu = move in tabu_list
        
        # Aspiration: allow if it improves significantly
        aspiration = best_score < best_c5
        
        # Accept move if it's not tabu or is aspiration
        if not is_tabu or aspiration:
            # Move to best neighbor
            best_h = best_neighbor.copy()
            best_c5 = best_score
            
            # Update tabu list
            tabu_list.append(move)
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
            
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Shaking mechanism
        if no_improvement_count >= shaking_frequency:
            # Random restart: flip a few bits
            num_flips = max(1, n_points // 20)
            indices_to_flip = np.random.choice(n_points, size=num_flips, replace=False)
            for idx in indices_to_flip:
                best_h[idx] = 1 - best_h[idx]
            
            # Ensure sum constraint
            ones_count = np.sum(best_h)
            if ones_count > n_points // 2:
                # Too many ones
                to_flip = int(ones_count - n_points // 2)
                ones_positions = np.where(best_h == 1)[0]
                flip_positions = np.random.choice(ones_positions, size=to_flip, replace=False)
                for i in flip_positions:
                    best_h[i] = 0
            elif ones_count < n_points // 2:
                # Not enough ones
                to_flip = int(n_points // 2 - ones_count)
                zeros_positions = np.where(best_h == 0)[0]
                flip_positions = np.random.choice(zeros_positions, size=to_flip, replace=False)
                for i in flip_positions:
                    best_h[i] = 1
            
            no_improvement_count = 0

        # Early termination
        if no_improvement_count >= max_no_improvement:
            break
            
        iteration += 1
    
    # Final validation
    if np.sum(best_h) != n_points // 2:
        # Reset to valid state
        ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
        best_h = np.zeros(n_points, dtype=int)
        best_h[ones_indices] = 1
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
