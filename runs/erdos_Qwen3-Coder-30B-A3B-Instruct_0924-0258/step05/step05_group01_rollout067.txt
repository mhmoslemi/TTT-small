import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize binary solution
    h_binary = np.round(best_h).astype(int)
    # Adjust to satisfy sum constraint
    current_sum = np.sum(h_binary)
    target_sum = n_points // 2
    if current_sum < target_sum:
        # Need to flip some 0s to 1s
        zero_indices = np.where(h_binary == 0)[0]
        flip_count = target_sum - current_sum
        flip_indices = np.random.choice(zero_indices, flip_count, replace=False)
        h_binary[flip_indices] = 1
    elif current_sum > target_sum:
        # Need to flip some 1s to 0s
        one_indices = np.where(h_binary == 1)[0]
        flip_count = current_sum - target_sum
        flip_indices = np.random.choice(one_indices, flip_count, replace=False)
        h_binary[flip_indices] = 0
    
    current_c5 = evaluate_erdos_solution(h_binary, dx)
    if current_c5 < best_c5:
        best_h = h_binary.astype(float)
        best_c5 = current_c5
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 10
    max_no_improve = 100
    no_improve_count = 0
    shaking_freq = 50
    max_iterations = 10000
    
    # Get initial indices
    ones_indices = np.where(h_binary == 1)[0]
    zeros_indices = np.where(h_binary == 0)[0]
    
    iteration = 0
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Generate neighbors by flipping one 1 and one 0
        neighbor_h = h_binary.copy()
        if len(ones_indices) > 0 and len(zeros_indices) > 0:
            # Select a 1 and a 0 to flip
            flip_1_idx = np.random.choice(ones_indices)
            flip_0_idx = np.random.choice(zeros_indices)
            
            neighbor_h[flip_1_idx] = 0
            neighbor_h[flip_0_idx] = 1
            
            # Compute overlap
            neighbor_c5 = evaluate_erdos_solution(neighbor_h, dx)
            
            # Check if neighbor is better or is an aspiration
            is_aspiration = False
            if neighbor_c5 < best_c5:
                is_aspiration = True
            
            # Check if move is tabu
            is_tabu = False
            if len(tabu_list) > 0:
                for tabu_move in tabu_list:
                    if tabu_move[0] == flip_1_idx and tabu_move[1] == flip_0_idx:
                        is_tabu = True
                        break
            
            # Accept move if it's better, or if aspiration, or if not tabu
            if neighbor_c5 < best_c5 or is_aspiration or not is_tabu:
                h_binary = neighbor_h.copy()
                current_c5 = neighbor_c5
                
                if current_c5 < best_c5:
                    best_h = h_binary.astype(float)
                    best_c5 = current_c5
                    no_improve_count = 0
                else:
                    no_improve_count += 1
            else:
                no_improve_count += 1
            
            # Update tabu list
            tabu_list.append((flip_1_idx, flip_0_idx))
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
        else:
            # Fallback: random restart
            restart_indices = np.where(h_binary == 1)[0]
            if len(restart_indices) > 0:
                idx = np.random.choice(restart_indices)
                h_binary[idx] = 0
                # Find a 0 to transform to 1
                zeros = np.where(h_binary == 0)[0]
                if len(zeros) > 0:
                    new_idx = np.random.choice(zeros)
                    h_binary[new_idx] = 1
                else:
                    # To avoid underflow, leave as is
                    pass
            else:
                # Ensure proper balance
                h_binary = np.zeros(n_points, dtype=int)
                h_binary[:n_points//2] = 1
                np.random.shuffle(h_binary)
                
        # Shake if no improvement for too long
        if no_improve_count >= shaking_freq:
            # Flip 5% of bits
            num_flip = max(1, int(0.05 * n_points))
            flip_indices = np.random.choice(n_points, num_flip, replace=False)
            h_binary[flip_indices] = 1 - h_binary[flip_indices]
            # Ensure sum constraint
            current_sum = np.sum(h_binary)
            if current_sum != n_points // 2:
                # Adjust by flipping some more bits
                diff = int(current_sum - n_points // 2)
                if diff < 0:
                    # Need more 1s
                    zero_indices = np.where(h_binary == 0)[0]
                    random_zero = np.random.choice(zero_indices, -diff, replace=False)
                    h_binary[random_zero] = 1
                else:
                    # Need more 0s
                    one_indices = np.where(h_binary == 1)[0]
                    random_one = np.random.choice(one_indices, diff, replace=False)
                    h_binary[random_one] = 0
                    
            no_improve_count = 0
            iteration -= 1  # Don't count the shake as an iteration
            
        # Re-compute indices
        ones_indices = np.where(h_binary == 1)[0]
        zeros_indices = np.where(h_binary == 0)[0]
        
        iteration += 1
    
    # Final validation
    if np.sum(best_h) > n_points / 2 + 1e-6 or np.sum(best_h) < n_points / 2 - 1e-6:
        # Re-adjust to exact sum
        best_h = np.round(best_h).astype(int).astype(float)
        current_sum = np.sum(best_h)
        target_sum = n_points // 2
        if current_sum < target_sum:
            zero_indices = np.where(best_h == 0)[0]
            flip_count = target_sum - current_sum
            flip_indices = np.random.choice(zero_indices, flip_count, replace=False)
            best_h[flip_indices] = 1
        elif current_sum > target_sum:
            one_indices = np.where(best_h == 1)[0]
            flip_count = current_sum - target_sum
            flip_indices = np.random.choice(one_indices, flip_count, replace=False)
            best_h[flip_indices] = 0
            
    # Final computation for return
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), float(final_c5), int(n_points))
