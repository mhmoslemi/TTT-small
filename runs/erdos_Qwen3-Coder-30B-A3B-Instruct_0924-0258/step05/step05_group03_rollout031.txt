import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    shifts = np.linspace(0, 2, 100)
    
    # Initialize optimization variables
    h_var = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2, C5 >= 0]
    
    # Convex optimization loop
    last_c5 = float('inf')
    iteration = 0
    max_iterations = 50
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Evaluate current overlaps for all shifts
        overlaps = []
        max_overlap = 0
        max_shift_idx = 0
        
        # We only evaluate first 10 shifts to save time during convex iterations
        for i, shift in enumerate(shifts[:10]):
            shift_idx = int(shift / dx)
            if shift_idx < n_points and shift_idx >= 0:
                # Manual computation of the overlap for this shift
                overlap = 0
                for j in range(n_points):
                    k = j + shift_idx
                    if k < n_points:
                        overlap += h[j] * (1 - h[k])
                overlaps.append(overlap)
                if overlap > max_overlap:
                    max_overlap = overlap
                    max_shift_idx = i
        
        # Add constraint from worst overlap so far
        # This is a simplified version of the cutting-plane approach
        if max_overlap > 0:
            # Don't add duplicate constraints
            if iteration == 0 or max_overlap < last_c5 - 1e-8:
                constraints.append(C5 >= max_overlap)
                last_c5 = max_overlap
        
        # Solve convex problem
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                c5_new = C5.value
                if c5_new is not None and not np.isnan(c5_new):
                    h = np.clip(h_new, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    if c5_new < last_c5:
                        last_c5 = c5_new
                else:
                    break
            else:
                break
        except:
            break
        
        iteration += 1
        
        # Early termination if target achieved
        if last_c5 < 0.3807:
            break
    
    # Final local refinement with SLSQP
    def obj(x):
        return evaluate_erdos_solution(x, dx)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, 
                       constraints=cons_sum, options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            h = res.x
            last_c5 = obj(h)
    except:
        pass
    
    # Final validation
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = evaluate_erdos_solution(h, dx)
    
    return (list(h), final_c5, n_points)
