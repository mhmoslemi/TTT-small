import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with parent solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid optimization
    n_low = 20
    dx_low = 2.0 / n_low
    h_low = np.ones(n_low) * 0.5
    
    # Relaxation with fewer constraints for coarse grid
    try:
        h_relaxed = cp.Variable(n_low)
        constraints = [
            cp.sum(h_relaxed) == n_low / 2.0,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        
        # Use subset of constraints
        for k in range(1, min(8, n_low)):  
            if k < n_low // 2:
                corr_sum = cp.sum(cp.multiply(h_relaxed[:n_low-k], 1 - h_relaxed[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_low-k]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            h_low = h_relaxed.value
            h_low = np.clip(h_low, 0, 1)
            h_low = h_low * (n_low / 2) / np.sum(h_low)
    except Exception:
        pass

    # Upsample and refine
    h_interp = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_low), h_low)
    h_interp = np.clip(h_interp, 0, 1)
    h_interp = h_interp * (n_points / 2) / np.sum(h_interp)
    
    # Gradient descent refinement with mutation
    h_current = h_interp.copy()
    patience = 0
    best_h_fine = h_current.copy()
    best_c5_fine = evaluate_erdos_solution(h_current, dx)
    mu = 0.01
    mutation_prob = 0.1
    mutation_strength = 0.05
    
    while time() - start_time < budget_s * 0.9:
        # Evaluate current
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        # Simple gradient approximation
        eps = 1e-5
        grad = np.zeros_like(h_current)
        for i in range(len(h_current)):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (c5_plus - c5_current) / eps

        # Update step
        h_new = h_current - mu * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Mutation
        if np.random.rand() < mutation_prob:
            mut_mask = np.random.rand(len(h_new)) < mutation_prob
            mutation = np.random.normal(0, mutation_strength, len(h_new))
            h_new = h_new + mut_mask * mutation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
        # Accept new point
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5_fine:
            best_h_fine = h_new.copy()
            best_c5_fine = c5_new
            patience = 0
        else:
            patience += 1
            
        # If no improvement, restart with new random point
        if patience > 50:
            h_current = np.random.rand(n_points)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            h_current = np.clip(h_current, 0, 1)
            patience = 0
        else:
            h_current = h_new

    # Final evaluation
    h_final = best_h_fine.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)

    return (list(h_final), final_c5, n_points)
