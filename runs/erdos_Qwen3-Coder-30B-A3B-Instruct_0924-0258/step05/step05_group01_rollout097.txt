import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize with a better seed if possible
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points/2) / np.sum(h_init)
        c5_init = evaluate_erdos_solution(h_init, dx)
        if c5_init < best_c5:
            best_h = h_init
            best_c5 = c5_init

    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Local optimization with multiple restarts
    max_restarts = 30
    restart_count = 0
    
    while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Generate a new random candidate with better distribution
        h_candidate = np.random.uniform(0.1, 0.9, n_points)
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        
        # Refine with SLSQP
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res_opt = minimize(objective, h_candidate, method='SLSQP', 
                            bounds=bounds, constraints=cons_sum, 
                            options={'maxiter': 200, 'ftol': 1e-8})
            
            if res_opt.success:
                h_refined = res_opt.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
                else:
                    restart_count += 1
            else:
                restart_count += 1
        except Exception:
            restart_count += 1
            
        # Add some diversification with shrinkage
        if restart_count % 5 == 0 and restart_count > 0:
            h_candidate = 0.8 * best_h + 0.2 * np.random.uniform(0, 1, n_points)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
            try:
                res_opt = minimize(objective, h_candidate, method='SLSQP', 
                                bounds=bounds, constraints=cons_sum, 
                                options={'maxiter': 150, 'ftol': 1e-8})
                if res_opt.success:
                    h_refined = res_opt.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        restart_count = 0
            except:
                pass
    
    # Final refinement
    try:
        h_final = np.clip(best_h, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
