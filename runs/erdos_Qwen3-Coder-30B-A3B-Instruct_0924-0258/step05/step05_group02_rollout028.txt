import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution scheme
    n_low = 20  # coarse grid
    dx_low = 2.0 / n_low
    
    def upsample(h_coarse, n_high):
        h_high = np.interp(np.linspace(0, 2, n_high), np.linspace(0, 2, n_low), h_coarse)
        h_high = np.clip(h_high, 0, 1)
        h_high = h_high * (n_high/2) / np.sum(h_high)
        return h_high
    
    def coarse_optimization():
        # Create coarse problem
        h_low = cp.Variable(n_low)
        # Relaxation: add only first few constraints to keep it manageable
        constraints = [
            cp.sum(h_low) == n_low / 2.0,
            h_low >= 0,
            h_low <= 1
        ]
        
        # Add some key overlap constraints
        for k in range(1, min(8, n_low)):
            if k < n_low // 2:
                corr_sum = cp.sum(cp.multiply(h_low[:n_low-k], 1 - h_low[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_low[k:], 1 - h_low[:n_low-k]))
            constraints.append(corr_sum <= 0.4 * dx_low + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_low)), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-7)
            if prob.status == cp.OPTIMAL:
                h_coarse = h_low.value
                return h_coarse
        except:
            pass
        # Fallback
        return np.ones(n_low) * 0.5
    
    # Step 1: Coarse optimization
    h_coarse = coarse_optimization()
    h_refined = upsample(h_coarse, n_points)
    
    # Step 2: Fine-grain local refinement
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Iteratively improve
    max_iterations = 500
    early_stop = 0
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Local optimization
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_refined, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-7}
            )
            if res.success:
                h_new = res.x
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    early_stop = 0
                else:
                    early_stop += 1
            else:
                early_stop += 1
        except:
            early_stop += 1
        
        # Restart heuristic
        if early_stop > 20:
            h_new = np.random.rand(n_points)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            early_stop = 0
            
        # Update refinement
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
    
    # Final validation and cleanup
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
