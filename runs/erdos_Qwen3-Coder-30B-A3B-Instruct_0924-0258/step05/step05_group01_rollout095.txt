from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    candidates = [best_h.copy()]
    for _ in range(19):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        candidates.append(h)
    
    # Evaluate initial candidates
    candidate_values = [evaluate_erdos_solution(h, dx) for h in candidates]
    best_idx = np.argmin(candidate_values)
    best_h = candidates[best_idx].copy()
    best_c5 = candidate_values[best_idx]
    
    # Optimization
    iteration = 0
    while time() - start_time < budget_s * 0.95:
        # Generate new candidate by perturbing best solution
        h_new = best_h + np.random.normal(0, 0.03, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

        # Occasionally increase resolution
        if iteration % 20 == 0 and n_points < 100:

            # Interpolate
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_points)
            best_h = np.interp(x_new, x_old, best_h)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Final refinement using local optimization
    if time() - start_time < budget_s * 0.05:

            h_refine = best_h.copy()

            for _ in range(5):

                h_restart = np.random.rand(n_points) * 0.5
                h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)

                    return np.sum(h) - n_points / 2.0

                                bounds=bounds, constraints=cons_sum,
