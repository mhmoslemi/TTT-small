import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Triangular function as more promising initial guess
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization
    scale_levels = [35, 50, 75, 100]
    current_scale = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale < len(scale_levels):
        scale = scale_levels[current_scale]
        
        # Interpolate if necessary
        if scale > n_points:
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            h_interp = np.interp(x_new, x_old, best_h)
            n_points = scale
            dx = 2.0 / n_points
            h_current = h_interp
        else:
            h_current = best_h
            n_points = scale
            dx = 2.0 / n_points
            
        # Local optimization with gradient-based method
        max_iter = 100
        lr = 0.01
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Evaluate current function value
            c5_current = evaluate_erdos_solution(h_current, dx)
            
            # Compute gradient with finite differences
            grad = np.zeros_like(h_current)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                
                grad[i] = (c5_plus - c5_current) / eps
            
            # Update step
            h_new = h_current - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Accept update if it improves the objective
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < c5_current:
                h_current = h_new
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            else:
                lr *= 0.95  # Reduce learning rate if no improvement
                
        current_scale += 1

    # Final refinement at highest resolution
    if time() - start_time < budget_s * 0.1:
        try:
            # Simple local search around best solution
            for _ in range(20):
                if time() - start_time > budget_s * 0.99:
                    break
                h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
                c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
                if c5_perturbed < best_c5:
                    best_h = h_perturbed
                    best_c5 = c5_perturbed
        except:
            pass

    return (list(best_h), best_c5, n_points)
