import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use pre-existing initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Coarse initial grid for optimization
        n_points_coarse = 20
        h_start = np.random.rand(n_points_coarse)
        h_start = h_start * (n_points_coarse / 2) / np.sum(h_start)
        h_start = np.clip(h_start, 0, 1)
        n_points = n_points_coarse
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid optimization
    n_coarse = 20
    dx_coarse = 2.0 / n_coarse
    
    # Try to solve coarse version
    if n_points_coarse < n_points:
        try:
            h_coarse = cp.Variable(n_coarse)
            constraints = [
                cp.sum(h_coarse) == n_coarse / 2.0,
                h_coarse >= 0,
                h_coarse <= 1
            ]
            
            # Add constraints on few shifts to manage complexity
            for k in range(1, min(8, n_coarse)):  
                corr_sum = cp.sum(cp.multiply(h_coarse[:n_coarse-k], 1 - h_coarse[k:]))
                constraints.append(corr_sum <= best_c5 * dx_coarse + 1e-6)
            
            prob = cp.Problem(cp.Minimize(cp.sum(h_coarse)), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-7)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_coarse.value
                # Interpolate to target size
                h_interpolated = np.interp(np.linspace(0, 2, n_points), 
                                        np.linspace(0, 2, n_coarse), h_feasible)
                h_interpolated = h_interpolated * (n_points/2) / np.sum(h_interpolated)
                h_interpolated = np.clip(h_interpolated, 0, 1)
                c5_interpolated = evaluate_erdos_solution(h_interpolated, dx)
                if c5_interpolated < best_c5:
                    best_h = h_interpolated
                    best_c5 = c5_interpolated
        except Exception:
            pass
    
    # Local refinement - SLSQP
    max_restarts = 30
    restart_count = 0
    no_improvement = 0
    
    while time() - start_time < budget_s * 0.95:
        # Use gradient-based optimization on current best
        try:
            h_refined = best_h.copy()
            
            # Add small noise for diversity
            noise = np.random.normal(0, 0.02, n_points)
            h_refined = h_refined + noise
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            
            # Optimization
            cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_refined, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            
            if res.success:
                h_new = res.x
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    no_improvement = 0
                else:
                    no_improvement += 1
            else:
                no_improvement += 1
        except Exception:
            no_improvement += 1
            
        # Restart strategy
        if no_improvement > 20 or restart_count > max_restarts:
            h_restart = np.random.rand(n_points)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            h_restart = np.clip(h_restart, 0, 1)
            restart_count = 0
            no_improvement = 0
            
            # Quickly verify that restart is good
            c5_restart = evaluate_erdos_solution(h_restart, dx)
            if c5_restart < best_c5:
                best_h = h_restart
                best_c5 = c5_restart
        
        restart_count += 1
        
        if time() - start_time > budget_s * 0.9:
            break
    
    # Final refinement
    try:
        h_final = np.clip(best_h, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    except Exception:
        pass
    
    # Ensure that the sum constraint is correct
    h_optimal = best_h.copy()
    h_optimal = h_optimal * (n_points/2) / np.sum(h_optimal)    
    h_optimal = np.clip(h_optimal, 0, 1)
    final_c5 = evaluate_erdos_solution(h_optimal, dx)
    
    return (list(h_optimal), final_c5, n_points)
