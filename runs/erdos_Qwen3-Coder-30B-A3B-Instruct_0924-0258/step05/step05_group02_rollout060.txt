import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the parent solution or a default
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try to improve the solution using a hybrid approach: 
    # 1. Local refinement via optimization
    # 2. Introduce randomness in search space
    # 3. Use CVXPY for projections in high dimensional space
    
    max_iter = 500
    iter_count = 0
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Perturbation step
        h_perturbed = best_h.copy()
        noise = np.random.normal(0, 0.01, n_points)
        h_perturbed += noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Project to feasible set 
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Focus on the optimization step
        cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-7}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    last_improvement = 0
                else:
                    last_improvement += 1
            else:
                last_improvement += 1
        except Exception:
            last_improvement += 1
        
        # Occasionally step with a random sample to escape local minima
        if iter_count % 20 == 0 and last_improvement > 5:
            h_random = np.random.rand(n_points)
            h_random = h_random * (n_points/2) / np.sum(h_random)
            h_random = np.clip(h_random, 0, 1)
            c5_random = evaluate_erdos_solution(h_random, dx)
            if c5_random < best_c5:
                best_h = h_random
                best_c5 = c5_random
                last_improvement = 0
            else:
                last_improvement += 1
        
        iter_count += 1
        if last_improvement > 20:
            break
            
    # Final correction step
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
