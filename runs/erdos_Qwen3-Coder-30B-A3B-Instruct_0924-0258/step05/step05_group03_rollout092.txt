import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize initial solution to satisfy sum constraint
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # CCP parameters
    max_iter = 1000
    eps = 1e-6
    dx = 2.0 / n_points
    
    h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Main CCP loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find worst shift k* (evaluate over a coarse grid)
        shifts = np.linspace(0, 2, 100)
        max_corr = -np.inf
        best_shift = 0
        
        for k in shifts:
            # Compute shift index
            k_idx = int(k / 2 * n_points)
            if k_idx >= n_points:
                k_idx = n_points - 1
                
            # Evaluate correlation at shift k
            corr = np.sum(h * np.roll(1 - h, k_idx)) * dx
            if corr > max_corr:
                max_corr = corr
                best_shift = k                
        
        # Optimization at fixed k
        # We will solve the problem: minimize sum h[i](1 - h[i + k*])
        # subject to sum(h) = n_points/2, h in [0,1]
        
        # Compute the shifted version of 1-h
        k_idx = int(best_shift / 2 * n_points)
        if k_idx >= n_points:
            k_idx = n_points - 1
        shifted_one_minus_h = np.roll(1 - h, k_idx)
        
        # Solve convex optimization: minimize sum h[i] * shifted_one_minus_h[i]
        # subject to sum(h) = n_points / 2, 0 <= h[i] <= 1
        h_new = h.copy()  # Start from current point
        prob = cp.Problem(
            cp.Minimize(shifted_one_minus_h @ h),
            [cp.sum(h) == n_points / 2,
             h >= 0,
             h <= 1]
        )
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h.value
        except:
            pass
            
        # Apply the constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C₅
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Update h
        h = h_new
        
        # Check termination
        if abs(best_c5 - evaluate_erdos_solution(h, dx)) < eps:
            break
            
        # Early stopping if target achieved
        if best_c5 < 0.3807:
            break
    
    # Final polishing with SLSQP
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
