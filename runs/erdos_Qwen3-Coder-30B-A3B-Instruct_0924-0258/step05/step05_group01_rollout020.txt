import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        h_start = project_feasible(h_start, n_points)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize GP surrogate
    kernel = C(1.0, (1e-3, 1e3)) * RBF(1.0, (1e-2, 1e2))
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=5, alpha=1e-6)
    
    # Generate initial population
    initial_candidates = []
    initial_values = []
    
    # Include parent if available
    if 'initial_h_values' in globals():
        initial_candidates.append(h_start)
        initial_values.append(best_c5)
    
    # Add random samples
    for _ in range(15):
        h_rand = np.random.rand(n_points)
        h_rand = project_feasible(h_rand, n_points)
        c5_rand = evaluate_erdos_solution(h_rand, dx)
        initial_candidates.append(h_rand)
        initial_values.append(c5_rand)
        if c5_rand < best_c5:
            best_c5 = c5_rand
            best_h = h_rand.copy()
    
    # Train initial GP model
    X_train = np.array(initial_candidates)
    y_train = np.array(initial_values)
    try:
        gp.fit(X_train, y_train)
    except:
        # Fallback to simple approach if GP fails
        pass
    
    # Optimization loop
    iteration = 0
    max_iterations = 500
    evaluated_count = len(initial_candidates)
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        try:
            # Generate candidates using GP
            candidate_count = 50
            candidates = []
            
            # Sample from GP predictions
            for _ in range(candidate_count):
                # Random sample
                h_sample = np.random.rand(n_points)
                h_sample = project_feasible(h_sample, n_points)
                candidates.append(h_sample)
            
            # Try to add some informed samples
            if len(initial_candidates) > 0:
                # Perturb best solution
                h_perturbed = best_h + np.random.normal(0, 0.03, n_points)
                h_perturbed = project_feasible(h_perturbed, n_points)
                candidates.append(h_perturbed)
                
            # Evaluate all candidates
            candidate_evals = []
            for h in candidates:
                c5 = evaluate_erdos_solution(h, dx)
                candidate_evals.append((h, c5))
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    
            # Update dataset
            new_candidates = [h for h, c5 in candidate_evals]
            new_values = [c5 for h, c5 in candidate_evals]
            
            # Retrain GP on recent data
            if len(new_candidates) > 0:
                X_recent = np.array(new_candidates)
                y_recent = np.array(new_values)
                try:
                    if len(X_train) > 0:
                        X_new = np.vstack([X_train, X_recent])
                        y_new = np.hstack([y_train, y_recent])
                    else:
                        X_new = X_recent
                        y_new = y_recent
                        
                    gp.fit(X_new, y_new)
                    X_train = X_new
                    y_train = y_new
                except:
                    pass
                
            evaluated_count += len(candidates)
            iteration += 1
            
        except Exception as e:
            # If error occurs, continue with basic optimization
            iteration += 1
            pass
            
        # Adaptive resolution scaling
        if iteration % 20 == 0 and n_points < 100:
            n_points += 5
            dx = 2.0 / n_points
            # Interpolate best solution to new resolution
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, len(best_h)), best_h)
            best_h = project_feasible(h_interp, n_points)
            best_c5 = evaluate_erdos_solution(best_h, dx)
            
    return (list(best_h), best_c5, n_points)
