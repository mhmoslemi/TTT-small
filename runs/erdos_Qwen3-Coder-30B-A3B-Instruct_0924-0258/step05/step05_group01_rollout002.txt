import numpy as np
from scipy.signal import correlate
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h_start = np.clip(1 - np.abs(x - 1), 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Scale levels
    scales = [35, 50, 75, 100]
    current_scale_idx = 0
    current_h = best_h.copy()
    current_n = n_points
    
    # Gradient-based refinement
    while time() - start_time < budget_s * 0.95 and current_scale_idx < len(scales):
        target_n = scales[current_scale_idx]
        if target_n > current_n:
            # Interpolate to higher resolution
            x_old = np.linspace(0, 2, current_n)
            x_new = np.linspace(0, 2, target_n)
            current_h = np.interp(x_new, x_old, current_h)
            current_n = target_n
        elif target_n < current_n:
            # Downsample to lower resolution
            step = current_n // target_n
            current_h = current_h[::step]
            current_n = target_n
            
        dx = 2.0 / current_n
        
        # Gradient-based optimization with momentum
        h_current = current_h.copy()
        velocity = np.zeros_like(h_current)
        alpha = 0.02
        beta = 0.9
        epsilon = 1e-8
        
        # Up to 100 iterations per scale
        for iter in range(100):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute correlation to find max overlap
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Compute gradient w.r.t. h for this shift
            # Shift index: k = max_idx - (current_n - 1)
            shift = max_idx - (current_n - 1)
            
            # Approximate gradient
            grad = np.zeros_like(h_current)
            for i in range(current_n):
                if i + shift >= 0 and i + shift < current_n:
                    grad[i] = 1 - h_current[i + shift]
            
            # Update with momentum
            velocity = beta * velocity + (1 - beta) * grad
            h_new = h_current - alpha * velocity
            
            # Project onto feasible set
            h_new = np.clip(h_new, 0, 1)
            # Rescale to satisfy sum constraint
            h_new = h_new * (current_n / 2) / np.sum(h_new)
            
            h_current = h_new
            current_c5 = max_overlap
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                
        current_scale_idx += 1
        
    # Final refinement
    if time() - start_time < budget_s * 0.1:
        try:
            x = np.linspace(0, 2, current_n)
            h_final = best_h.copy()
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (current_n / 2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except:
            pass
            
    return (list(best_h), best_c5, current_n)
