import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    max_gens = 1000
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    stagnation_limit = 50
    min_c5_target = 0.3807
    
    # Initialize population
    def create_individual():
        ind = np.random.rand(n_points)
        ind = ind * (n_points/2) / np.sum(ind)
        return np.clip(ind, 0, 1)
    
    def create_population(size):
        return [create_individual() for _ in range(size)]
    
    def tournament_selection(pop, fitnesses, k):
        selected = random.sample(list(zip(pop, fitnesses)), k)
        return min(selected, key=lambda x: x[1])[0]
    
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        crossover_point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        # Normalize
        child1 = child1 * (n_points/2) / np.sum(child1)
        child2 = child2 * (n_points/2) / np.sum(child2)
        return np.clip(child1, 0, 1), np.clip(child2, 0, 1)
        
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                mutated[i] += np.random.normal(0, 0.02)
        mutated = mutated * (n_points/2) / np.sum(mutated)
        return np.clip(mutated, 0, 1)
    
    # Main loop
    population = create_population(pop_size)
    gen = 0
    stagnation_count = 0
    best_gen_c5 = float('inf')
    
    # Optimization loop
    while time() - start_time < budget_s * 0.9:
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(ind, dx) for ind in population]
        current_best = min(fitnesses)
        
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(fitnesses)].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        if current_best < best_gen_c5:
            best_gen_c5 = current_best
        
        if best_c5 <= min_c5_target or stagnation_count > stagnation_limit:
            break
            
        # Create next generation
        new_population = []
        # Elitism: Keep best individual
        best_idx = np.argmin(fitnesses)
        new_population.append(population[best_idx].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
        gen += 1
        
        # Adaptive resolution
        if (time() - start_time) > budget_s * 0.8 and n_points == 35 and best_c5 > 0.3808:
            n_points = 40
            dx = 2.0 / n_points
            pop_size = 20
            h_best_resized = np.interp(np.linspace(0, 2, n_points),
                                     np.linspace(0, 2, len(best_h)), best_h)
            best_h = h_best_resized
            best_c5 = evaluate_erdos_solution(best_h, dx)
            population = create_population(pop_size)
            population[0] = best_h.copy()
            best_gen_c5 = float('inf')
            stagnation_count = 0

    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 300, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
