import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Ensure the initial solution satisfies constraints
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points/2.0) > 1e-6:
        best_h = best_h * (n_points/2.0) / sum_h
    best_h = np.clip(best_h, 0, 1)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    # Prefer fewer points - try n=20 first for faster optimization
    n_small = 20
    dx_small = 2.0 / n_small
    if n_points >= n_small:
        # Initial coarse grid solution
        h_small = np.ones(n_small) * 0.5
        h_small = h_small * (n_small/2.0) / np.sum(h_small)
        
        # Max iterations for coarse solution
        iter_coarse = 50
        for _ in range(iter_coarse):
            if time() - start_time > budget_s * 0.3:
                break
            try:
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_small)]
                res = minimize(
                    lambda h: evaluate_erdos_solution(h, dx_small), 
                    h_small, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=cons_sum,
                    options={'maxiter': 50, 'ftol': 1e-7}
                )
                if res.success:
                    h_small = res.x
                    c5_small = evaluate_erdos_solution(h_small, dx_small)
                    if c5_small < best_c5:
                        best_h = np.interp(np.linspace(0, 1, n_points),

                        best_c5 = c5_small
                        if c5_small < 0.3807:
                            break
            except Exception:
                pass

            h_small += np.random.normal(0, 0.02, n_small)
            h_small = np.clip(h_small, 0, 1)
            h_small = h_small * (n_small/2.0) / np.sum(h_small)
    
    # Now optimize full resolution
    iter_count = 0
    max_iter = 1000
    patience = 0
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        h_perturbed = best_h.copy()

        noise = np.random.normal(0, 0.02, n_points)
        h_perturbed += noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2.0) / np.sum(h_perturbed)

            cons_sum = {'type': 'eq', 'fun': constraint_sum}

            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)

                    best_h = h_refined
                    best_c5 = c5_refined
                    patience = 0
                    last_improvement = iter_count
                else:
                    patience += 1
            else:

        except Exception:
            patience += 1

            h_restart = np.random.rand(n_points)

            h_restart = np.clip(h_restart, 0, 1)

            best_c5 = evaluate_erdos_solution(h_restart, dx)

            last_improvement = iter_count

        if best_c5 < 0.3807:

    # Final normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2.0) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
