import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.linspace(0, 1, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale gradient descent approach
    scales = [35, 50, 75, 100]
    current_scale_idx = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale_idx < len(scales):
        scale = scales[current_scale_idx]
        if scale > n_points:
            # Interpolate to higher resolution
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            h_interp = np.interp(x_new, x_old, best_h)
            h_current = h_interp
            n_points = scale
        else:
            h_current = best_h
            n_points = scale
            
        dx = 2.0 / n_points
        
        # Gradient-based refinement for this scale
        max_iter = 100
        lr = 0.01
        prev_c5 = float('inf')
        
        for iter_count in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute C5 and identify the k that maximizes overlap
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            k = max_idx - (n_points - 1)
            
            # Compute gradient w.r.t. h for the max overlap
            gradient = np.zeros(n_points)
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    gradient[i] = 1 - 2 * h_current[j]
            
            # Update h
            h_new = h_current - lr * gradient
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Check improvement
            new_c5 = evaluate_erdos_solution(h_new, dx)
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                
            # Adaptive learning rate
            if new_c5 >= prev_c5:
                lr *= 0.8
            prev_c5 = new_c5
            
            h_current = h_new.copy()
            
        current_scale_idx += 1
    
    return (list(best_h), best_c5, n_points)
