import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    # Ensure h is properly normalized
    h = h * (n_points / 2) / np.sum(h)
    
    # Initial evaluation
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization loop
    iteration = 0
    max_iterations = 500
    patience = 0
    best_c5 = current_c5
    best_h = h.copy()
    
    # Parameters for local refinement
    local_refine_freq = 10
    perturb_freq = 20
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Local refinement with SLSQP on current h
        if iteration % local_refine_freq == 0:
            def obj(x):
                return evaluate_erdos_solution(x, dx)
            
            def constraint_sum(x):
                return np.sum(x) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(obj, h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    h_new = res.x
                    c5_new = obj(h_new)
                    if c5_new < current_c5:
                        h = h_new
                        current_c5 = c5_new
                        if current_c5 < best_c5:
                            best_c5 = current_c5
                            best_h = h.copy()
                            patience = 0
                        else:
                            patience += 1
                    else:
                        patience += 1
                else:
                    patience += 1
            except:
                patience += 1
        
        # Heavy perturbations to escape local minima
        if iteration % perturb_freq == 0 and iteration > 0:
            h = h + np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Small random perturbation
        if np.random.random() < 0.1:
            h = h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Early termination if target achieved
        if best_c5 < 0.3807:
            break
            
        iteration += 1
    
    # Final refinement with SLSQP
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-10})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
