import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    pop_size = 50
    elitism = int(0.3 * pop_size)
    mutation_rate = 0.1
    max_generations = 1000
    
    # Initialize population
    population = []
    population.append(best_h.copy())
    
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    # GA main loop
    generation = 0
    stagnation_counter = 0
    max_stagnation = 50
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        
        if stagnation_counter > max_stagnation:
            break
            
        # Selection - keep top elitism
        elites = population[:elitism]
        
        # Create new population
        new_population = elites.copy()
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection for parents
            parent1 = elites[np.random.randint(0, len(elites))]
            parent2 = elites[np.random.randint(0, len(elites))]
            
            # Uniform crossover with sparsity bias
            alpha = np.random.uniform(0.4, 0.6)
            child = alpha * parent1 + (1 - alpha) * parent2
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child[i] += np.random.normal(0, 0.02)
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        # Refine top candidates with gradient descent
        for i in range(min(10, len(new_population))):
            h_candidate = new_population[i]
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h_candidate, method='SLSQP', 
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < fitness[i]:
                        new_population[i] = h_refined.copy()
            except:
                pass
                
        population = new_population
        generation += 1
    
    # Final refinement with optimization
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            h_final = res.x
            best_c5 = evaluate_erdos_solution(h_final, dx)
    except:
        pass
    
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    
    return (list(h_final), best_c5, n_points)
