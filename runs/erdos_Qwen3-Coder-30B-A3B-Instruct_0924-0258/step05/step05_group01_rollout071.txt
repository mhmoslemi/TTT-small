import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
import warnings

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate random initial candidates
    m = 20  # number of initial candidates
    initial_candidates = []
    for _ in range(m):
        h_candidate = np.random.rand(n_points)
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        h_candidate = np.clip(h_candidate, 0, 1)
        initial_candidates.append(h_candidate)
    
    # Add best_h if it's different from the initial
    if not np.allclose(h_start, best_h):
        initial_candidates.append(best_h.copy())
    
    # Evaluate initial candidates
    initial_evals = []
    for h in initial_candidates:
        c5 = evaluate_erdos_solution(h, dx)
        initial_evals.append((h, c5))
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Prepare training data for GP
    X_train = np.array([h for h, _ in initial_evals])
    y_train = np.array([c5 for _, c5 in initial_evals])
    
    # Gaussian Process model
    kernel = C(1.0, (1e-3, 1e3)) * RBF(1.0, (1e-2, 1e2))
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=5, alpha=1e-6)
    gp.fit(X_train, y_train)
    
    # Main optimization loop
    max_iter = 100
    iter_count = 0

    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Generate new candidates using GP
        n_candidates = 50
        candidates = []
        
        # Sample around current best
        for _ in range(n_candidates):
            # Small random perturbations to best_h
            h_new = best_h + np.random.normal(0, 0.02, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            candidates.append(h_new)
            
        # Evaluate candidates
        c5_values = []
        for h in candidates:
            try:
                c5 = evaluate_erdos_solution(h, dx)
                c5_values.append(c5)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
            except Exception:
                c5_values.append(float('inf'))
        
        # Update GP with new points
        if len(c5_values) > 0:
            X_new = np.array(candidates)
            y_new = np.array(c5_values)
            X_train = np.vstack([X_train, X_new])
            y_train = np.hstack([y_train, y_new])
            
            try:
                gp.fit(X_train, y_train)
            except Exception:
                pass
        
        iter_count += 1
        
        # Occasionally increase resolution
        if iter_count % 10 == 0 and n_points < 100:
            n_points += 5
            dx = 2.0 / n_points
            # Project best_h to the new resolution
            best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            # Reset GP to avoid getting trapped in local minima
            X_train = np.array([best_h])
            y_train = np.array([best_c5])
            gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=5, alpha=1e-6)
            gp.fit(X_train, y_train)
    
    # Final refinement
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    if final_c5 < best_c5:
        best_h = h_final
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
