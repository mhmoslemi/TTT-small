import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Use parent solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    max_iter = 5000
    mu = 1e-3
    rho = 1.0
    tol = 1e-6
    admm_iter = 0
    
    # Initialize
    h = h_start.copy()
    h_prev = h.copy()
    z = np.zeros(2 * n_points - 1)
    lambda_ = np.zeros(2 * n_points - 1)
    
    # Optimization variable for ADMM step
    h_var = cp.Variable(n_points)
    C5_var = cp.Variable()
    
    # ADMM main loop
    while admm_iter < max_iter and (time() - start_time) < max_time * 0.95:
        admm_iter += 1
        
        # Compute shifted (1-h) vector - a bit tricky due to convolution structure
        # but we compute it in a block for optimization
        h_shifted = np.zeros(2 * n_points - 1)
        for i in range(n_points):
            for j in range(n_points):
                k = i + j
                if k < 2 * n_points - 1:
                    h_shifted[k] += (1 - h[j]) * h[i]
        
        # Update h using quadratic penalty approach based on last iteration
        # Use constrained minimization
        def obj(h):
            return (np.sum((h - h_prev)**2) + 
                    mu * np.sum((h_shifted[:2*n_points-1] - z)**2))
            
        def con_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons = {'type': 'eq', 'fun': con_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_e = res.x
                feasible = True
                if not (np.all(h_e >= 0) and np.all(h_e <= 1)):
                    feasible = False
                if abs(np.sum(h_e) - n_points/2) > 1e-4:
                    feasible = False
                
                if feasible:
                    h = h_e
        except:
            pass
        
        # Update z and lambda (ADMM steps)
        # Conflict with updating h already via optimizing.
        # Let's use a subgradient approach for basic relaxation to update z
        # Instead just enforce the constraint z_k = max_k h_i(1-h_{i+k}) for k in 0:2n-2
        # Then we can simply compute overlaps as a direct computation to derive C5
        
        # Now compute the overlap integral directly for final evaluation
        new_c5 = evaluate_erdos_solution(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        # Early termination
        if best_c5 < 0.3807:
            break
            
        # Periodic SLSQP refinement
        if admm_iter % 50 == 0:
            h_init = h.copy()
            def obj_full(h_c):
                return evaluate_erdos_solution(h_c, dx)
                
            def con_sum_full(h_c):
                return np.sum(h_c) - n_points / 2.0
                
            cons_full = {'type': 'eq', 'fun': con_sum_full}
            bounds_full = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(obj_full, h_init, method='SLSQP', 
                               bounds=bounds_full, constraints=cons_full,
                               options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = obj_full(h_refined)
                    if c5_refined < best_c5:
                        best_c5 = c5_refined
                        best_h = h_refined
            except:
                pass
    
    # Final refinement with tight tolerance
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        try:
            h_init = best_h.copy()
            
            def obj_final(h):
                return evaluate_erdos_solution(h, dx)
                
            def con_sum_final(h):
                return np.sum(h) - n_points / 2.0
                
            cons_final = {'type': 'eq', 'fun': con_sum_final}
            bounds_final = [(0, 1) for _ in range(n_points)]
            
            res = minimize(obj_final, h_init, method='SLSQP',
                           bounds=bounds_final, constraints=cons_final,
                           options={'maxiter': 200, 'ftol': 1e-10})
            
            if res.success:
                h_final = res.x
                c5_final = obj_final(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
