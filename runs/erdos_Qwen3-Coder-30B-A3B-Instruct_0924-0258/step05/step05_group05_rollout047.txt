import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try SDP relaxation approach
    if n_points <= 50:  # Only use SDP for small enough problems
        try:
            # Set up SDP relaxation
            h_sdp = cp.Variable(n_points)
            X = cp.Variable((n_points, n_points), symmetric=True)
            
            # Constraints for relaxation
            constraints = [
                cp.trace(X) == n_points / 2,  # trace constraint
                X >> 0,  # positive semidefinite
                h_sdp >= 0,
                h_sdp <= 1,
                cp.sum(h_sdp) == n_points / 2,
                X >= 0,
                X <= 1
            ]
            
            # Add constraints that X[i,j] = h_sdp[i] * h_sdp[j] 
            for i in range(n_points):
                for j in range(n_points):
                    if i != j:
                        constraints.append(X[i,j] <= h_sdp[i])
                        constraints.append(X[i,j] <= h_sdp[j])
                        constraints.append(X[i,j] >= h_sdp[i] + h_sdp[j] - 1)
            
            # Objective - maximize the overlap integral
            objective = cp.Minimize(cp.sum(X @ (np.eye(n_points) - np.eye(n_points))))
            prob = cp.Problem(objective, constraints)
            
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == cp.OPTIMAL:
                # Extract solution from SDP
                h_sdp_solution = h_sdp.value
                if h_sdp_solution is not None:
                    h_sdp_solution = np.clip(h_sdp_solution, 0, 1)
                    h_sdp_solution = h_sdp_solution * (n_points/2) / np.sum(h_sdp_solution)
                    c5_sdp = evaluate_erdos_solution(h_sdp_solution, dx)
                    if c5_sdp < best_c5:
                        best_h = h_sdp_solution
                        best_c5 = c5_sdp
        except Exception:
            pass
    
    # Local refinement using trust-constr
    if time() - start_time > budget_s * 0.9:
        return (list(best_h), best_c5, n_points)
        
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try multiple random warmups
    for _ in range(5):
        if time() - start_time > budget_s * 0.95:
            break
        h_warmup = best_h + np.random.normal(0, 0.02, n_points)
        h_warmup = np.clip(h_warmup, 0, 1)
        h_warmup = h_warmup * (n_points/2) / np.sum(h_warmup)
        
        try:
            # Use trust-constr method, which is robust for this problem
            res = minimize(
                objective, 
                h_warmup, 
                method='trust-constr',
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 500, 'gtol': 1e-8, 'ftol': 1e-9}
            )
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Final local refinement using L-BFGS-B
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 50:
        try:
            res = minimize(
                objective, 
                best_h, 
                method='L-BFGS-B',
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 200, 'ftol': 1e-9}
            )
            if res.success:
                h_final = np.clip(res.x, 0, 1)
                h_final = h_final * (n_points/2) / np.sum(h_final)
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
        
    return (list(best_h), best_c5, n_points)
