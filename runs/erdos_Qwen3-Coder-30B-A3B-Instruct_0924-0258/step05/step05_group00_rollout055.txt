import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    population_size = 50
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def normalize_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Genetic Algorithm
    def create_individual():
        h = np.random.rand(n_points)
        return normalize_h(h)
    
    def evaluate_individual(h):
        return evaluate_erdos_solution(h, dx)
    
    # Selection: tournament
    def tournament_selection(pop, k=5):
        selected = []
        for _ in range(len(pop)):
            tournament = random.sample(pop, min(k, len(pop)))
            winner = min(tournament, key=evaluate_individual)
            selected.append(winner.copy())
        return selected
    
    # Crossover: single point with repair
    def crossover(parent1, parent2):
        if random.random() > 0.8:
            return parent1.copy(), parent2.copy()
        point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return normalize_h(child1), normalize_h(child2)
    
    # Mutation: Gaussian noise
    def mutate(h, mutation_rate=0.1, sigma=0.02):
        h = h.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                h[i] += np.random.normal(0, sigma)
        return normalize_h(h)
    
    # Main GA loop
    generation = 0
    stagnation_count = 0
    max_stagnation = 20
    
    while time() - start_time < budget_s * 0.8:
        generation += 1
        if generation % 100 == 0:
            if time() - start_time > budget_s * 0.95:
                break 
            stagnation_count = 0
        
        # Create population if needed
        if generation == 1:
            population = [create_individual() for _ in range(population_size)]
        else:
            # Selection
            selected = tournament_selection(population)
            # Crossover and mutation
            offspring = []
            while len(offspring) < population_size:
                parent1, parent2 = random.sample(selected, 2)
                child1, child2 = crossover(parent1, parent2)
                child1 = mutate(child1)
                child2 = mutate(child2)
                offspring.extend([child1, child2])
            population = offspring[:population_size]
        
        # Evaluate population
        for individual in population:
            c5 = evaluate_individual(individual)
            if c5 < best_c5:
                best_c5 = c5
                best_h = individual.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
        if stagnation_count > max_stagnation:
            break
    
    # Resolution upgrade if needed and time allows
    if time() - start_time < budget_s * 0.9 and best_c5 > 0.3808:
        n_points = 40
        dx = 2.0 / n_points
        # Interpolate best solution
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, len(best_h)), 
                           best_h)
        best_h = normalize_h(h_interp)
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
        # Run GA again at higher resolution
        population_size = 20
        population = [create_individual() for _ in range(population_size)]
        max_stagnation = 10
        stagnation_count = 0
        
        for gen in range(300):
            if time() - start_time > budget_s * 0.95:
                break
            
            selected = tournament_selection(population)
            offspring = []
            while len(offspring) < population_size:
                parent1, parent2 = random.sample(selected, 2)
                child1, child2 = crossover(parent1, parent2)
                child1 = mutate(child1)
                child2 = mutate(child2)
                offspring.extend([child1, child2])
            population = offspring[:population_size]
            
            for individual in population:
                c5 = evaluate_individual(individual)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = individual.copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            if stagnation_count > max_stagnation:
                break
    
    # Final local refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    try:
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
