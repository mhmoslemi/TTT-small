import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Set up SCP parameters
    max_iter = 500
    mu = 1e-3
    C5_init = 0.4
    
    # Precompute shift indices for overlap computations
    shift_indices = []
    for k in range(n_points):
        shift_indices.append(np.arange(n_points) - k)
    
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Step 1: Evaluate overlaps
        current_c5 = evaluate_erdos_solution(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        
        # Early stopping if target hit
        if best_c5 < 0.3807:
            break
            
        # Step 2: Formulate convex QP approximation 
        # Variables: h, C5
        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Objective: minimize C5
        objective = cp.Minimize(C5_var)
        
        # Constraints:
        # 1. h in [0,1] 
        constraints = [h_var >= 0, h_var <= 1]
        
        # 2. sum(h) = n_points / 2  
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # 3. Overlap constraints for each shift shift_indices[k]
        # sum(h[i] * (1 - h[(i+k)%n])) * dx <= C5
        # We compute all overlaps
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr = np.max(corr) * dx
        for k in range(n_points):
            tmp = cp.sum(cp.multiply(h_var, 1 - cp.roll(h_var, k))) * dx
            constraints.append(tmp <= C5_var)
            
        # Add penalty terms for constraint violations
        # Linearized augmented Lagrangian objective
        # Use previous iterate to compute Lagrangian
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=OSQP, verbose=False, max_iter=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                new_c5 = C5_var.value
                
                # Update only if the new solution improves
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
                    
                # Update penalty parameter based on constraint satisfaction
                if best_c5 < C5_init:
                    C5_init = best_c5  
                    
            else:
                # Fallback to previous solution
                pass
                
        except Exception as e:
            # Fallback to previous solution if optimization fails
            pass
    
    # Final refinement with SLSQP to improve tightness
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        def con(h):
            return np.sum(h) - n_points / 2.0
            
        cons = {'type': 'eq', 'fun': con}
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = obj(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
