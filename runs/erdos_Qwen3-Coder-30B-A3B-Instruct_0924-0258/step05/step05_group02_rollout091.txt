import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse-to-fine approach
    n_low = 20  # Lower resolution grid
    n_high = n_points
    
    # Optimization phases
    max_iter = 5000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    learning_rate = 0.01
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points/2) / np.sum(h_vec)
        return h_vec
    
    def project_h_low(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_low/2) / np.sum(h_vec)
        return h_vec
    
    def upsampling(h_low):
        h_high = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, n_low), h_low)
        h_high = h_high * (n_points/2) / np.sum(h_high)
        return h_high
    
    # Try coarse grid first
    if time() - start_time < budget_s * 0.2:
        try:
            h_low = cp.Variable(n_low)
            constraints = [
                cp.sum(h_low) == n_low / 2.0,
                h_low >= 0,
                h_low <= 1
            ]
            # Add constraints for first few shifts
            for k in range(1, min(10, n_low)): 
                if k < n_low // 2:
                    corr_sum = cp.sum(cp.multiply(h_low[:n_low-k], 1 - h_low[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_low[k:], 1 - h_low[:n_low-k]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(cp.sum(h_low)), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-7)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_low.value
                h_upsampled = upsampling(h_feasible)
                c5_feasible = evaluate_erdos_solution(h_upsampled, dx)
                if c5_feasible < best_c5:
                    best_h = h_upsampled
                    best_c5 = c5_feasible
        except Exception:
            pass
    
    # Fine-grained optimization
    h_current = best_h.copy()
    
    def gradient_descent_step(h, lr):
        # Approximate gradient using finite differences
        epsilon = 1e-5
        grad = np.zeros_like(h)
        base_val = evaluate_erdos_solution(h, dx)
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = project_h(h_plus)
            val_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (val_plus - base_val) / epsilon
        
        h_new = h - lr * grad
        return project_h(h_new)
    
    while time() - start_time < budget_s and iter_count < max_iter:
        # Do a descent step
        h_new = gradient_descent_step(h_current, learning_rate)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            no_improvement_count = 0
            h_current = h_new.copy()
            # Decrease learning rate if improving
            learning_rate = max(0.001, learning_rate * 0.99)
        else:
            no_improvement_count += 1
            # Apply mutation with some probability if no improvement
            if no_improvement_count > 20 and np.random.rand() < 0.1:
                h_mutated = h_current + 0.05 * np.random.normal(0, 1, len(h_current))
                h_mutated = project_h(h_mutated)
                c5_mutated = evaluate_erdos_solution(h_mutated, dx)
                if c5_mutated < best_c5:
                    best_h = h_mutated
                    best_c5 = c5_mutated
                    no_improvement_count = 0
                    h_current = h_mutated.copy()
            else:
                h_current = h_new.copy()
        
        if no_improvement_count > max_no_improvement:
            break
            
        iter_count += 1
    
    # Final check
    final_h = project_h(best_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
