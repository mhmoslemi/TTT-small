import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # SLSQP local optimization wrapper
    def optimize_local(h_init, max_iter=100):
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def con_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': con_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(objective, h_init, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-8})
            if res.success:
                return res.x, objective(res.x)
        except:
            pass
        return h_init, objective(h_init)
    
    # Lagrangian relaxation approach
    max_iter = 500
    rho = 1e3
    lambda_k = np.ones(2 * n_points - 1) * 0.1
    mu = 1.0
    tolerance = 1e-4
    
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Formulate convex subproblem using CVXPY
        h = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(lambda_k, 
            cp.multiply(h, 1 - h) @ np.ones(2*n_points-1)) - 
            cp.sum(lambda_k) * best_c5 * dx) + mu * (cp.sum(h) - n_points / 2)**2)
        
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.OSQP, max_iter=1000, eps_abs=1e-6, eps_rel=1e-6)
            
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_opt = h.value
                c5_current = evaluate_erdos_solution(h_opt, dx)
                
                if c5_current < best_c5:
                    best_c5 = c5_current
                    best_h = h_opt.copy()
                
                # Update lambda and mu based on constraints
                corr = np.correlate(h_opt, 1-h_opt, mode='full')
                for k in range(len(lambda_k)):
                    lambda_k[k] = max(0, lambda_k[k] + rho * (corr[k] * dx - best_c5 * dx))
                
                mu = max(0, mu + rho * (np.sum(h_opt) - n_points / 2))
                
                # Adaptive rho
                if abs(np.sum(h_opt) - n_points / 2) > tolerance:
                    rho *= 2
                else:
                    rho *= 0.5
                    
                # Project h_opt back to [0,1] and normalize
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                
        except:
            h_opt = best_h.copy()
            pass
        
        # Local refinement with SLSQP periodically
        if iteration % 10 == 0:
            h_refined, c5_refined = optimize_local(h_opt, 50)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined
        
        if best_c5 < 0.3807:
            break
    
    # Final local refinement
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        final_h, final_c5 = optimize_local(best_h, 100)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
