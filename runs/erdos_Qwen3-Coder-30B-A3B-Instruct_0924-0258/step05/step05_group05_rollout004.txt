```python
import numpy as np
from scipy.optimize import minimize
from time import time
import hashlib

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_max_size = max(1, int(0.2 * n_points))
    max_stagnation = 100
    mutation_scale = 0.04
    mutation_budget = max(1, int(0.3 * n_points))
    max_iter = 1000
    refinement_interval = 10
    
    # Initialize
    current_h = h_start.copy()
    current_c5 = best_c5
    stagnation_counter = 0
    refinement_count = 0
    
    # Generate hash for tabu purposes
    def hash_state(h):
        # Create a hashable representation
        sorted_h = tuple(sorted(h))
        return hashlib.md5(str(sorted_h).encode()).hexdigest()

    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate new candidate by perturbing some elements
        candidate_h = current_h.copy()
        indices = np.random.choice(n_points, size=mutation_budget, replace=False)
        for i in indices:
            candidate_h[i] += np.random.normal(0, mutation_scale)
        
        # Clip to bounds
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Scale to meet constraint
        candidate_h = candidate_h * (n_points/2) / np.sum(candidate_h)
        
        # Evaluate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Tabu check - use a simple hash-based check
        candidate_key = hash_state(candidate_h)
        is_tabu = candidate_key in tabu_list
        
        # Accept if better or not tabu
        if candidate_c5 < current_c5 or not is_tabu:
            current_h = candidate_h
            current_c5 = candidate_c5
            
            # Update best if improved
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                if best_c5 < 0.3807:
                    break
                stagnation_counter = 0
            else:
                stagnation_counter += 1
            
            # Update tabu list
            tabu_list.append(candidate_key)
            if len(tabu_list) > tabu_max_size:
                tabu_list.pop(0)
            
            # Local refinement if needed
            if stagnation_counter % refinement_interval == 0 and stagnation_counter > 0:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, current_h, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 50, 'ftol': 1e-8})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < current_c5:
                            current_h = h_refined
                            current_c5 = c5_refined
                            if c5_refined < best_c5:
                                best_h = current_h.copy()
                                best_c5 = c5_refined
                            stagnation_counter = 0
                except:
                    pass
        
        # Update stagnation counter
        else:
            stagnation_counter += 1
        
        # Adjust mutation rate if stagnating
        if stagnation_counter > max_stagnation:
            mutation_scale *= 2.0
            stagnation_counter = 0
        
        # Reset mutation scale if it's too large
        if mutation_scale > 0.2:
            mutation_scale = 0.04
            
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
```
