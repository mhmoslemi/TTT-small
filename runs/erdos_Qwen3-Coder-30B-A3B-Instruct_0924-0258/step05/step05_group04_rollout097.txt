import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize Lagrange multipliers and parameters
    lambda_k = np.ones(2 * n_points - 1) * 0.1
    mu = 1.0
    rho = 1e3
    max_iter = 500
    
    # Initialize A matrix (correlation kernel)
    A = np.zeros((2 * n_points - 1, n_points))
    for k in range(1 - n_points, n_points):
        shift = max(0, -k)
        end = min(n_points, n_points - k)
        if k < 0:
            A[k + n_points - 1, shift:end] = 1
        elif k == 0:
            A[n_points - 1, :] = 1
        else:
            A[k + n_points - 1, shift:end] = 1
    
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Create CVXOPT problem
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum_squares(h_var))  # dummy objective for minimization
        
        # Constraints
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Lagrangian relaxation constraint approximation
        # We only solve the continuous relaxation for initial steps
        try:
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.OSQP, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                # Update better solutions
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    
        except:
            pass
            
        # Minor local search around current best
        if iteration % 10 == 0 and iteration > 0:
            def obj_func(h):
                return evaluate_erdos_solution(h, dx)
            def con_sum(h):
                return np.sum(h) - n_points / 2.0
                
            cons = {'type': 'eq', 'fun': con_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(obj_func, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-7})
                if res.success:
                    h_local = res.x
                    c5_local = obj_func(h_local)
                    if c5_local < best_c5:
                        best_c5 = c5_local
                        best_h = h_local
            except:
                pass
        
        # Early stopping
        if best_c5 < 0.3807:
            break
            
    # Final refinement with scipy
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_refine = best_h.copy()
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        def con(h):
            return np.sum(h) - n_points / 2.0
            
        cons = {'type': 'eq', 'fun': con}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, h_refine, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = obj(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
