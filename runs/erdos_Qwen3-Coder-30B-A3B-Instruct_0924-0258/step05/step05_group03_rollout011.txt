import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Ensure the initial solution is feasible
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Best solution tracking
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    # Local search settings
    max_iterations = 500
    stagnation_limit = 50
    sample_shifts = 50
    
    # Track stagnation
    stagnation_count = 0
    last_best = best_c5
    
    # Main optimization loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample a set of integer shifts to evaluate
        shifts = np.random.choice(n_points, sample_shifts, replace=False)
        
        # Compute overlap for each sampled shift
        max_overlap = -1
        best_shift = 0
        for k in shifts:
            # Compute correlation for shift k
            corr_k = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_k += best_h[i] * (1 - best_h[j])
            corr_k *= (2.0 / n_points)
            if corr_k > max_overlap:
                max_overlap = corr_k
                best_shift = k
        
        # Compute gradient for this specific shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_shift) % n_points
            grad[i] += (1 - best_h[j]) * (2.0 / n_points)
            grad[j] -= best_h[i] * (2.0 / n_points)
        
        # Line search for step size
        alpha = 1.0
        step_size = 0.1
        while alpha > 1e-8:
            h_proposed = best_h - alpha * grad
            # Project to feasible region
            h_proposed = np.clip(h_proposed, 0, 1)
            h_proposed = h_proposed * (n_points / 2) / np.sum(h_proposed)
            
            proposed_c5 = evaluate_erdos_solution(h_proposed, 2.0 / n_points)
            if proposed_c5 < best_c5:
                best_h = h_proposed.copy()
                best_c5 = proposed_c5
                stagnation_count = 0
                break
            alpha *= step_size
            
        # Check stagnation
        if best_c5 < last_best:
            last_best = best_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Restart if stuck
        if stagnation_count > stagnation_limit:
            # Add small random perturbation
            perturbation = np.random.normal(0, 0.01, n_points)
            best_h = best_h + perturbation
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            stagnation_count = 0
            
        # Early termination
        if best_c5 < 0.3807:
            break
    
    # Final refinement with optimization
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
