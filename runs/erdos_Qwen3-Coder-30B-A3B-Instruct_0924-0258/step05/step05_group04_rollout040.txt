import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize parameters
    max_iter = 500
    mu = 1e-3
    C5_nih = 0.4
    early_stop_threshold = 0.38
    
    # Create a function that builds constraints for current h
    def build_constraints(h_vec):
        # Compute overlaps for all shifts
        h_padded = np.concatenate([h_vec, h_vec[:n_points-1]])
        overlaps = []
        for k in range(n_points):
            shift = h_padded[k:k+n_points]
            overlap = np.sum(h_vec * (1 - shift)) * dx
            overlaps.append(overlap)
        return np.array(overlaps)
    
    # Sequential Convex Programming loop
    for iter_num in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Evaluate current solution
        current_overlaps = build_constraints(best_h)
        max_overlap = np.max(current_overlaps)
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = best_h.copy()
        
        if best_c5 < early_stop_threshold:
            break
            
        # Build QCQP with augmented Lagrangian
        # Variables
        h_vars = cp.Variable(n_points)
        
        # Objective: minimize C5
        C5_var = cp.Variable()
        
        # Constraints:
        # 1. Overlap constraints
        constraints = []
        # Cycle through all shifts
        for k in range(n_points):
            # Create shifted version of h_vars
            h_shifted = cp.hstack([h_vars[k:], h_vars[:k]])
            # Add shifted constraint
            overlap = cp.sum(cp.multiply(h_vars, 1 - h_shifted)) * dx
            constraints.append(overlap <= C5_var)
            
        # 2. Sum constraint
        constraints.append(cp.sum(h_vars) == n_points / 2.0)
        
        # 3. Bound constraints
        constraints.extend([h_vars >= 0, h_vars <= 1])
        
        # Objective function: minimize C5
        objective = cp.Minimize(C5_var)
        
        # Create and solve QP
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.OSQP, verbose=False, 
                      eps_abs=1e-6, eps_rel=1e-6, max_iter=1000)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                C5_new = C5_var.value
                if C5_new < best_c5:
                    best_c5 = C5_new
                    best_h = h_new.copy()
            else:
                # Fallback to local optimization
                pass
        except:
            pass
            
        # Update mu and C5 based on previous performance
        if max_overlap > C5_nih:
            mu *= 2.0
        C5_nih = max_overlap
        
    # Final refinement with local solver
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def con_sum(h):
        return np.sum(h) - n_points / 2.0
    
    h_init = best_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    try:
        res = minimize(objective, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 1000, 'ftol': 1e-9})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
