import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid optimization for initial solution
    n_coarse = 20
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.ones(n_coarse) * 0.5
    h_coarse = h_coarse * (n_coarse/2) / np.sum(h_coarse)
    
    # Convex relaxation approach on coarse grid
    try:
        h_relaxed = cp.Variable(n_coarse)
        constraints = [
            cp.sum(h_relaxed) == n_coarse / 2.0,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        
        # Add a reasonable subset of overlap constraints
        num_constraint_samples = min(5, n_coarse)
        for i in range(1, num_constraint_samples):
            if i < n_coarse // 2:
                corr_sum = cp.sum(cp.multiply(h_relaxed[:n_coarse-i], 1 - h_relaxed[i:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_relaxed[i:], 1 - h_relaxed[:n_coarse-i]))
            constraints.append(corr_sum <= best_c5 * dx_coarse + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            h_coarse = h_relaxed.value
    except Exception:
        pass
    
    # Interpolate to full resolution
    h_interp = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
    h_interp = h_interp * (n_points/2) / np.sum(h_interp)
    h_interp = np.clip(h_interp, 0, 1)
    
    h_refined = h_interp.copy()
    old_c5 = best_c5
    
    # Local refinement around interpolated solution
    restart_count = 0
    max_restarts = 30
    early_stop = 0
    
    while time() - start_time < budget_s * 0.95:
        try:
            # SLSQP local refinement
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
            
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_refined, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-7}
            )
            
            if res.success:
                h_new = res.x
                new_c5 = evaluate_erdos_solution(h_new, dx)
                if new_c5 < best_c5:
                    best_h = h_new
                    best_c5 = new_c5
                    early_stop = 0
                else:
                    early_stop += 1
            else:
                early_stop += 1
        except Exception:
            early_stop += 1

        # Restart if no progress
        if early_stop > 10:
            early_stop = 0
            restart_count += 1
            h_refined = np.random.rand(n_points)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            h_refined = np.clip(h_refined, 0, 1)
            
            if restart_count > max_restarts:
                break
    
    # Final evaluation
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
