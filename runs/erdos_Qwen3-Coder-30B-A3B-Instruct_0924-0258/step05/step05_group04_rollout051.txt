import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    max_iter = 5000
    mu = 1e-3
    tol = 1e-6
    rho = 1.0
    
    # Initialize variables
    h = h_start.copy()
    h_prev = h.copy()
    z = np.zeros(2 * n_points - 1)
    lambda_vec = np.zeros(2 * n_points - 1)
    
    # ADMM iterations
    for it in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Update h
        # Solve: minimize ||h - (z - lambda/mu)||^2 + mu*||conv(h, 1-h) - z||^2
        # subject to 0 <= h <= 1 and sum(h) = n_points/2
        
        # Create a wrapped objective function for SLSQP
        def obj(h):
            # Compute convolution
            conv_h = np.correlate(h, 1 - h, mode='full')
            # Add regularization
            reg_term = 0.5 * mu * np.sum((conv_h - z)**2)
            return reg_term
        
        def con(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con}
        
        # Use the last h as starting point
        try:
            res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_new = res.x
                h = h_new
            else:
                # fallback to simple projection
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Update z
        conv_h = np.correlate(h, 1 - h, mode='full')
        z_new = conv_h + lambda_vec / mu
        z = z_new
        
        # Update lambda
        lambda_vec += mu * (conv_h - z)
        
        # Check for convergence
        if it % 50 == 0:
            c5_test = evaluate_erdos_solution(h, dx)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h.copy()
            
            if best_c5 < 0.3807:
                break

    # Final local refinement with SLSQP
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass

    return (list(best_h), best_c5, n_points)
