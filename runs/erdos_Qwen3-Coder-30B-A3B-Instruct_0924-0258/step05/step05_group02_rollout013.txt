import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different local search strategies with warm starts
    strategies = [
        'SLSQP',
        'L-BFGS-B',
        'TNC'
    ]
    
    max_restarts = 50
    restart_count = 0
    early_stop = 0
    
    while time() - start_time < budget_s * 0.95:
        # Try several random restarts with different strategies
        h_perturbed = best_h.copy()
        h_perturbed += np.random.normal(0, 0.03, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        strategy = strategies[restart_count % len(strategies)]
        
        try:
            cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_perturbed, 
                method=strategy, 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 200, 'ftol': 1e-8}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
                    early_stop = 0
                else:
                    restart_count += 1
                    early_stop += 1
            else:
                restart_count += 1
                early_stop += 1
        except Exception:
            restart_count += 1
            early_stop += 1
        
        # Periodic relaxation
        if early_stop > 15:
            early_stop = 0
            # Try a simple periodic heuristic
            try:
                # Create periodic structure
                h_periodic = np.zeros(n_points)
                # Use alternating pattern
                pattern_length = min(8, n_points // 2)
                pattern = [1.0] * (pattern_length // 2) + [0.0] * (pattern_length - pattern_length // 2)
                pattern = pattern * (n_points // pattern_length + 1)
                pattern = pattern[:n_points]
                h_periodic = np.array(pattern)
                
                # Normalize and refine
                h_periodic = h_periodic * (n_points/2) / np.sum(h_periodic)
                h_periodic = np.clip(h_periodic, 0, 1)
                c5_periodic = evaluate_erdos_solution(h_periodic, dx)
                
                if c5_periodic < best_c5:
                    best_h = h_periodic
                    best_c5 = c5_periodic
            except Exception:
                pass
        
        # Random restart with different heuristic
        if restart_count > max_restarts:
            h_restart = np.random.rand(n_points)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            h_restart = np.clip(h_restart, 0, 1)
            restart_count = 0
        
        # Check for time limit
        if time() - start_time > budget_s * 0.9:
            break
    
    # Final refinement and normalization
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    except Exception:
        pass
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
