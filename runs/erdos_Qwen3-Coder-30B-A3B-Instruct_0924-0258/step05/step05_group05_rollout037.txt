import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create initial symmetric triangular function
        mid = n_points // 2
        h_start = np.minimum(np.arange(n_points), 2*mid - np.arange(n_points))
        h_start = h_start / (mid)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent optimization
    max_iter = 200
    tolerance = 1e-8
    ls_iter = 20
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            # Use L-BFGS-B with constraints
            res = minimize(objective, best_h, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons_sum,
                          options={'maxiter': 100, 'ftol': tolerance})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    if best_c5 < 0.3807:
                        break
            else:
                # If optimization failed, try manual adjustment
                h_refined = best_h.copy()
                # Simple gradient descent approach
                for _ in range(50):
                    # Compute gradient using finite differences
                    eps = 1e-6
                    grad = np.zeros_like(h_refined)
                    base_val = objective(h_refined)
                    
                    for i in range(n_points):
                        h_plus = h_refined.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                        grad[i] = (objective(h_plus) - base_val) / eps
                    
                    # Update with step size from Armijo rule
                    step_size = 1.0
                    for _ in range(ls_iter):
                        h_new = h_refined - step_size * grad
                        h_new = np.clip(h_new, 0, 1)
                        h_new = h_new * (n_points/2) / np.sum(h_new)
                        new_val = objective(h_new)
                        if new_val < base_val:
                            h_refined = h_new
                            break
                        step_size *= 0.5
                
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    if best_c5 < 0.3807:
                        break
                        
        except Exception as e:
            # Fallback to simple perturbation if optimization fails
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            c5_perturbed = objective(h_perturbed)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
        
        # Occasionally check for better solution
        if iteration % 5 == 0:
            current_c5 = evaluate_erdos_solution(best_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5

    # Final verification
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
