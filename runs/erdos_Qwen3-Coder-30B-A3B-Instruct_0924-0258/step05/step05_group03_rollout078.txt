import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from provided construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
        
    # Target goal: C5 <= 0.3807
    target_c5 = 0.3807
    
    # Solver settings
    max_iterations = 50
    shift_samples = 100
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    # Generate shift samples for overlap computation
    shift_vals = np.linspace(0, 2, shift_samples, endpoint=False)
    
    # Cutting-plane approach with CVXPY
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current overlaps for all shifts
        overlaps = []
        for k in shift_vals:
            # Compute overlap for shift k
            k_idx = int(k * n_points / 2)
            if k_idx == 0:
                overlap = np.sum(best_h * (1 - best_h)) * (2.0 / n_points)
            else:
                h_shifted = np.concatenate([best_h[k_idx:], best_h[:k_idx]])
                overlap = np.sum(best_h * (1 - h_shifted)) * (2.0 / n_points)
            overlaps.append(overlap)
            
        max_overlap_idx = np.argmax(overlaps)
        max_overlap_val = overlaps[max_overlap_idx]
        max_shift = shift_vals[max_overlap_idx]
        max_shift_idx = int(max_shift * n_points / 2)
        
        # Add cutting plane constraint using current maximum
        if max_overlap_val < best_c5:
            best_c5 = max_overlap_val
            if best_c5 < target_c5:
                break
                
        # Reformulate the optimization using CVXPY
        h_var = cp.Variable(n_points)
        h_constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Approximate
        if max_shift_idx == 0:
            overlap_expr = cp.sum(cp.multiply(h_var, 1 - h_var))
        else:
            # Periodic shift
            h_shifted = cp.hstack([h_var[max_shift_idx:], h_var[:max_shift_idx]])
            overlap_expr = cp.sum(cp.multiply(h_var, 1 - h_shifted))
            
        objective = cp.Minimize(overlap_expr)
        prob = cp.Problem(objective, h_constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                if new_h is not None:
                    new_h = np.clip(new_h, 0, 1)
                    new_h = new_h * (n_points / 2) / np.sum(new_h)
                    new_c5 = evaluate_erdos_solution(new_h, 2.0 / n_points)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = new_h
                        if best_c5 < target_c5:
                            break
        except:
            pass

    # Final refinement using scipy
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    # Final validation
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    return (list(best_h), best_c5, n_points)
