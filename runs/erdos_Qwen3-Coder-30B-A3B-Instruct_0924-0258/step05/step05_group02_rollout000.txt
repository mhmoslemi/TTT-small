import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # First, check if we have an initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid setup for convex relaxation
    n_coarse = 20
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.ones(n_coarse) * 0.5
    # Ensure sum = 1
    h_coarse = h_coarse * n_coarse / np.sum(h_coarse)
    
    # Use convex relaxation for coarse grid
    try:
        h_relaxed = cp.Variable(n_coarse)
        constraints = [
            cp.sum(h_relaxed) == n_coarse / 2.0,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        
        # Add more constraints: around 10 shifts (not all)
        for k in range(1, min(10, n_coarse)):
            corr_sum = cp.sum(cp.multiply(h_relaxed[:n_coarse-k], 1 - h_relaxed[k:]))
            constraints.append(corr_sum <= best_c5 * dx_coarse + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        
        if prob.status == cp.OPTIMAL:
            h_coarse = h_relaxed.value
            h_coarse = np.clip(h_coarse, 0, 1)
            # Normalize to exact sum
            h_coarse = h_coarse * n_coarse / np.sum(h_coarse)
    except Exception:
        pass
    
    # Interpolate to target grid
    h_interp = np.interp(np.linspace(0, 2, n_points), 
                         np.linspace(0, 2, n_coarse), h_coarse)
    h_interp = np.clip(h_interp, 0, 1)
    h_interp = h_interp * n_points / np.sum(h_interp)
    
    # Refine with local optimization
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Initial guess
    h_guess = h_interp.copy()
    h_guess = np.clip(h_guess, 0, 1)
    h_guess = h_guess * n_points / np.sum(h_guess)
    
    max_restarts = 30
    restart_count = 0
    iter_since_improvement = 0
    
    while time() - start_time < budget_s * 0.95:
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx),
                h_guess,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
                    iter_since_improvement = 0
                else:
                    restart_count += 1
                    iter_since_improvement += 1
            else:
                restart_count += 1
                iter_since_improvement += 1
        except Exception as e:
            restart_count += 1
            iter_since_improvement += 1
        
        # Adaptive restart logic
        if iter_since_improvement > 20 or restart_count > max_restarts:
            # Random perturbation
            h_perturbed = best_h + np.random.normal(0, 0.03, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * n_points / np.sum(h_perturbed)
            h_guess = h_perturbed.copy()
            restart_count = 0
            iter_since_improvement = 0
        
        if time() - start_time > budget_s * 0.9:
            break
    
    # Final check and normalization
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * n_points / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
