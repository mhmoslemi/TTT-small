import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid optimization
    n_low = 20
    dx_low = 2.0 / n_low
    
    # Initialize low-resolution grid
    h_low = np.ones(n_low) * 0.5
    
    # Convex relaxation with subset of constraints
    h_relaxed = cp.Variable(n_low)
    constraints = [
        cp.sum(h_relaxed) == n_low / 2.0,
        h_relaxed >= 0,
        h_relaxed <= 1
    ]
    
    # Constraint subset: just some shifts to manage complexity
    max_shifts = min(5, n_low)
    for k in range(1, max_shifts):
        corr_sum = cp.sum(cp.multiply(h_relaxed[:n_low-k], 1 - h_relaxed[k:]))
        constraints.append(corr_sum <= best_c5 * dx_low + 1e-6)
    
    prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
    try:
        prob.solve(solver=cp.ECOS, eps=1e-7, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_relaxed.value
            h_low = h_feasible
    except Exception:
        pass

    # Upsample to high resolution
    h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, n_low), h_low)
    h_upsampled = np.clip(h_upsampled, 0, 1)
    h_upsampled = h_upsampled * (n_points/2) / np.sum(h_upsampled)
    
    # Refinement with gradient descent and mutation
    h_refined = h_upsampled.copy()
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Gradients support
    def gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        # Simple finite difference gradient estimation for a few key points
        grad = np.zeros_like(h)
        epsilon = 1e-6
        h_plus = h.copy()
        h_minus = h.copy()
        for i in range(len(h)):
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * epsilon)
            h_plus[i] = h_minus[i] = h[i]
        return grad
    
    best_h = h_refined.copy()
    best_c5 = objective(best_h)
    
    # Adaptive local refinement with mutation
    iteration = 0
    patience = 0
    max_patience = 100
    learning_rate = 0.01
    mutation_rate = 0.1
    mutation_strength = 0.05
    max_iters = 1000
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iters:
        h_current = best_h.copy()
        
        # Gradient step
        grad = gradient(h_current)
        h_new = h_current - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)

        # Check if improvement
        c5_new = objective(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience = 0
        else:
            patience += 1

        # Mutation scheduling
        if patience > 50:
            # Apply mutation
            mutation_mask = np.random.random(n_points) < mutation_rate
            mutation = np.random.normal(0, mutation_strength, n_points)
            h_mutation = h_current + mutation_mask * mutation
            h_mutation = np.clip(h_mutation, 0, 1)
            h_mutation = h_mutation * (n_points/2) / np.sum(h_mutation)
            c5_mutation = objective(h_mutation)
            
            # Accept mutation if better
            if c5_mutation < best_c5:
                best_h = h_mutation.copy()
                best_c5 = c5_mutation
                patience = 0
                
        iteration += 1
        if patience >= max_patience:
            break

    # Final polishing with constrained optimization
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(
            lambda h: evaluate_erdos_solution(h, dx),
            h_final,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        if res.success:
            h_refined_final = res.x
            c5_refined_final = evaluate_erdos_solution(h_refined_final, dx)
            if c5_refined_final < best_c5:
                best_h = h_refined_final
                best_c5 = c5_refined_final
    except Exception:
        pass
    
    # Final check and normalization
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
