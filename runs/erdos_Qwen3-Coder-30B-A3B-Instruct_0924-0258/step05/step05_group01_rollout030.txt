import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Initialize with triangular function
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start / np.sum(h_start) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution scaling
    scales = [35, 50, 75, 100]
    current_scale_idx = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale_idx < len(scales):
        current_scale = scales[current_scale_idx]
        
        # Interpolate if needed
        if current_scale > n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, current_scale)
            h_interp = np.interp(x_new, x_orig, best_h)
            h_current = h_interp
            n_points = current_scale
        else:
            h_current = best_h
            n_points = current_scale
            
        dx = 2.0 / n_points
        
        # Simulated Annealing
        temp = 0.1
        alpha = 0.99
        max_iter = int(0.8 * budget_s * 1000 / n_points)  # Reduce based on scale
        
        h_curr = h_current.copy()
        c5_curr = evaluate_erdos_solution(h_curr, dx)
        
        for _ in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Create neighbor by small perturbations
            h_new = h_curr.copy()
            # Flip some values
            for i in range(n_points):
                if random.random() < 0.1:
                    # Small random change
                    delta = random.gauss(0, 0.05)
                    h_new[i] += delta
                    # Ensure within bounds
                    h_new[i] = max(0, min(1, h_new[i]))
            
            # Rescale to maintain sum constraint
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Metropolis criterion
            if c5_new < c5_curr or random.random() < np.exp(-(c5_new - c5_curr) / temp):
                h_curr = h_new
                c5_curr = c5_new
                
                if c5_curr < best_c5:
                    best_h = h_curr.copy()
                    best_c5 = c5_curr
            
            temp *= alpha

        current_scale_idx += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (list(best_h), best_c5, n_points)
