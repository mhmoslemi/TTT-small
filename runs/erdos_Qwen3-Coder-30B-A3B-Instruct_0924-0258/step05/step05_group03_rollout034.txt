import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 35
        h_init = np.ones(n_points) * 0.5
    
    # Normalize initial solution
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Shift sampling
    shift_samples = np.linspace(0, 2, 100)
    dx = 2.0 / n_points
    
    # Convex relaxation with cutting planes
    h_vars = cp.Variable(n_points)
    c5_var = cp.Variable()
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points / 2,
        c5_var >= 0
    ]
    
    # Add initial constraints
    for k in shift_samples:
        shift_idx = int(k / dx)
        if shift_idx < n_points:
            overlap_expr = cp.sum(cp.multiply(h_vars, cp.roll(1 - h_vars, shift_idx))) * dx
            constraints.append(overlap_expr <= c5_var)
    
    # Objective
    objective = cp.Minimize(c5_var)
    prob = cp.Problem(objective, constraints)
    
    # Iterative refinement
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    iteration = 0
    max_iterations = 50
    stagnation_count = 0
    last_best = best_c5
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_vars.value
                c5_new = c5_var.value
                
                # Validate solution
                if np.all(h_new >= 0) and np.all(h_new <= 1):
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        stagnation_count = 0
                        last_best = best_c5
                    else:
                        stagnation_count += 1
                else:
                    break
                    
                # Add constraint from most violated shift
                corr = np.correlate(h_new, 1 - h_new, mode='full')
                max_shift_idx = np.argmax(corr)
                max_corr = corr[max_shift_idx] * dx
                
                # Add constraint from the worst shift
                k = (max_shift_idx - (n_points - 1)) * dx
                shift_idx = int(k / dx)
                if shift_idx >= 0 and shift_idx < n_points:
                    overlap_expr = cp.sum(cp.multiply(h_vars, cp.roll(1 - h_vars, shift_idx))) * dx
                    constraints.append(overlap_expr <= c5_var)
                
                # Re-solve with new constraint
                prob = cp.Problem(objective, constraints)
                
                if best_c5 < 0.3807:
                    break
                    
            else:
                break
                
        except Exception as e:
            break
            
        iteration += 1
        
        # Early termination
        if best_c5 < 0.3807:
            break
    
    # Final polishing with SLSQP
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
