import numpy as np
from scipy.optimize import minimize
from time import time
from scipy.spatial.distance import cdist
import warnings
warnings.filterwarnings('ignore')

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Global optimization with restarts
    max_restarts = 50
    restart_count = 0
    
    while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Random initialization respecting bounds
        h_restart = np.random.uniform(0, 1, n_points)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        # SLSQP local optimization
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res_opt = minimize(objective, h_restart, method='SLSQP', 
                            bounds=bounds, constraints=cons_sum, 
                            options={'maxiter': 300, 'ftol': 1e-9, 'gtol': 1e-9})
            if res_opt.success:
                h_refined = res_opt.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0  # Reset counter on improvement
            else:
                restart_count += 1
        except Exception:
            restart_count += 1
    
    # Additional refinement with simpler method
    if time() - start_time < budget_s * 0.9:
        # Try another approach: gradient-based with projections
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    # Ensure final constraint
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
