import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    # Evolutionary algorithm with constraint handling
    pop_size = 50
    max_generations = 1000
    mutation_rate = 0.15
    crossover_rate = 0.8
    tournament_size = 3

    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
    
    # Add initial solution if not already in population
    if not any(np.allclose(h, h_start) for h in population):
        population[0] = h_start.copy()

    gen = 0
    stagnation_count = 0
    best_history = [best_c5]

    while time() - start_time < budget_s * 0.9 and gen < max_generations:
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Stop if no improvement for long
        if stagnation_count > 50:
            # Respawn with better initial solution
            h_best = best_h.copy()
            population = []
            for _ in range(pop_size):
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                population.append(h.copy())
            population[0] = h_best.copy()
            stagnation_count = 0
        
        # Selection with tournament
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % pop_size] if i+1 < pop_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # One-point crossover
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                    child2[j] += np.random.normal(0, 0.05)
            
            # Enforce constraints
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population
        gen += 1

    # Final optimization at highest resolution with local search
    if time() - start_time < budget_s * 0.1:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)

        # Refine with local optimization
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res_opt = minimize(objective, h_final, method='SLSQP', 
                                bounds=bounds, constraints=cons_sum, 
                                options={'maxiter': 100, 'ftol': 1e-8})
            if res_opt.success:
                h_refined = res_opt.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass

    return (list(best_h), best_c5, n_points)
