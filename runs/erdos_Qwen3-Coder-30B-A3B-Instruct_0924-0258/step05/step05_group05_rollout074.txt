import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # SDP relaxation approach
    def sdp_relaxation(h_init):
        # SDP approach for solving convex relaxation
        X = cp.Variable((n_points, n_points), symmetric=True)
        h = cp.Variable(n_points)
        
        # Build constraint matrix
        constraints = [
            cp.trace(X) == (n_points / 2) ** 2 * dx,
            X >> 0,  # Positive semidefinite
            h >= 0,
            h <= 1,
        ]
        
        # Add trace constraints
        for i in range(n_points):
            constraints.append(X[i, i] == h[i])
        
        # Add cross product constraints
        for i in range(n_points):
            for j in range(n_points):
                if i != j:
                    constraints.append(X[i, j] <= min(h[i], h[j]))
        
        # Objective: Maximize the overlap
        # This is a simplified implementation for demonstration
        # In practice, a more sophisticated approach would be needed
        
        try:
            prob = cp.Problem(cp.Minimize(cp.trace(X)), constraints)
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h.value
        except:
            pass
        return h_init.copy()
    
    # Algorithm: Hybrid SDP + Local optimization
    # First, try SDP relaxation
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 50:
        try:
            h_sdp = sdp_relaxation(h_start)
            if h_sdp is not None:
                c5_sdp = evaluate_erdos_solution(h_sdp, dx)
                if c5_sdp < best_c5:
                    best_h = h_sdp
                    best_c5 = c5_sdp
        except:
            pass
    
    # Then local optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Start from best solution found so far
    h_current = best_h.copy()
    
    # Better initial for local search
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Try multiple restarts from different points
    for _ in range(5):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Local refinement with L-BFGS-B
        try:
            res = minimize(objective, h_current, method='L-BFGS-B', 
                           bounds=bounds, 
                           options={'maxiter': 500, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                # Ensure constraints are met
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = objective(h_refined)
                
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
                    
                if best_c5 < 0.3807:
                    break
        except:
            pass
    
        # Random restart
        h_current = np.random.uniform(0, 1, n_points)
        h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Final check
    if best_c5 > 0.3807:
        # Try one more optimization from the best point
        try:
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, 
                           options={'maxiter': 1000, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                h_final = np.clip(h_final, 0, 1)
                h_final = h_final * (n_points/2) / np.sum(h_final)
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
