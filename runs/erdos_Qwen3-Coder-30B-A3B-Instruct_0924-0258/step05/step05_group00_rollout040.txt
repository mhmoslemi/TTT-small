import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization with cutting plane approach
    resolutions = [35, 40, 50, 60]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Initialize cutting plane approach
        max_iters = 100
        for _ in range(max_iters):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Get current max overlap
            c5_current = evaluate_erdos_solution(h_current, dx)
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
            
            # Find the shift producing maximum overlap
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            k_max = np.argmax(corr)
            max_overlap = corr[k_max] * dx
            
            # Add constraint: overlap at k_max should not exceed best_c5
            # This is a linear constraint in h - see below for details
            
            try:
                # Refine using SLSQP
                h_refined = h_current.copy()
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(objective, h_refined, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-8})
                
                if res.success:
                    h_candidate = res.x
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
                    h_current = h_candidate
                else:
                    break
            except Exception:
                # Fall back to random perturbation
                h_perturbed = h_current + np.random.normal(0, 0.01, n_points)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
                h_current = h_perturbed
                
        h_start = best_h.copy()
    
    # Final polishing with local search
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        
        if res.success:
            h_opt = res.x
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
