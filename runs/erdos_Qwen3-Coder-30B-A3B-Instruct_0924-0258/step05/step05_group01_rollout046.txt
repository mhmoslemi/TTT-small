```python
import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
import warnings
warnings.filterwarnings('ignore')

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate initial candidates
    initial_candidates = []
    initial_values = []
    
    # Add parent if available
    if 'initial_h_values' in globals():
        h_feasible = project_to_feasible(h_start, n_points)
        initial_candidates.append(h_feasible)
        initial_values.append(evaluate_erdos_solution(h_feasible, dx))
    
    # Add random candidates
    n_random = 15 if len(initial_candidates) == 0 else 19
    for _ in range(n_random):
        h_rand = np.random.rand(n_points)
        h_feasible = project_to_feasible(h_rand, n_points)
        initial_candidates.append(h_feasible)
        initial_values.append(evaluate_erdos_solution(h_feasible, dx))
    
    # Train GP surrogate model
    X_train = np.array(initial_candidates)
    y_train = np.array(initial_values)
    
    # Kernel: constant kernel + RBF kernel with automatic relevance determination
    kernel = C(1.0, (1e-3, 1e3)) * RBF(1.0, (1e-2, 1e2))
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=5, 
                                  normalize_y=True, alpha=1e-10)
    gp.fit(X_train, y_train)
    
    max_iter = 600
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Generate candidate points
        n_candidates = 50
        candidate_points = np.random.rand(n_candidates, n_points)
        
        # Project candidates to feasible region
        candidates_projected = []
        for i in range(n_candidates):
            h_projected = project_to_feasible(candidate_points[i], n_points)
            candidates_projected.append(h_projected)
        
        candidates_projected = np.array(candidates_projected)
        
        # Make predictions with GP
        try:
            predictions, stds = gp.predict(candidates_projected, return_std=True)
            
            # Score based on prediction + exploitation exploration balance
            # Lower predictions are preferred, but also consider uncertainty
            scores = predictions - 0.3 * stds  # exploitation 70%, exploration 30%
            best_indices = np.argsort(scores)[:10]  # Top 10 candidates
            
            # Evaluate top candidates
            for idx in best_indices:
                h_candidate = candidates_projected[idx]
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    
        except:
            pass
            
        # Update model with new evaluations (scaled up)
        # Only do it occasionally to save time
        if iter_count % 5 == 0 and len(initial_candidates) < 50:
            # Add best point to initial training set
            h_best = project_to_feasible(best_h, n_points)
            c5_best = evaluate_erdos_solution(h_best, dx)
            
            initial_candidates.append(h_best)
            initial_values.append(c5_best)
            
            # Retrain GP
            X_train = np.array(initial_candidates)
            y_train = np.array(initial_values)
            gp.fit(X_train, y_train)
        
        iter_count += 1
        
        # Occasionally increase resolution
        if iter_count % 25 == 0 and n_points < 100:
            new_res = min(n_points + 5, 100)
            if new_res > n_points:
                # Interpolate to higher resolution
                x_old = np.linspace(0, 2, n_points)
                x_new = np.linspace(0, 2, new_res)
                h_new_interpolated = np.interp(x_new, x_old, best_h)
                best_h = h_new_interpolated
                n_points = new_res
                dx = 2.0 / n_points
                # Re-evaluate with new resolution
                best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
```
