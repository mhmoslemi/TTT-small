import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
import warnings
warnings.filterwarnings('ignore')

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize candidates
    candidates = []
    values = []
    
    # Add parent solution if available
    if 'initial_h_values' in globals():
        h_feasible = project_to_feasible(h_start, n_points)
        c5_val = evaluate_erdos_solution(h_feasible, dx)
        candidates.append(h_feasible.copy())
        values.append(c5_val)
    
    # Generate random initial candidates
    n_init = 20
    for _ in range(n_init):
        h_rand = np.random.rand(n_points)
        h_feasible = project_to_feasible(h_rand, n_points)
        c5_val = evaluate_erdos_solution(h_feasible, dx)
        candidates.append(h_feasible.copy())
        values.append(c5_val)
    
    best_idx = np.argmin(values)
    best_c5 = values[best_idx]
    best_h = candidates[best_idx].copy()
    
    # Gaussian Process for surrogate modeling
    kernel = C(1.0, (1e-3, 1e3)) * RBF(1.0, (1e-2, 1e2))
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-6, normalize_y=True, 
                                  n_restarts_optimizer=5, random_state=seed)
    
    # Training data
    X_train = np.array(candidates)
    y_train = np.array(values)
    
    # Fit initial GP model
    try:
        gp.fit(X_train, y_train)
    except:
        pass
    
    # Optimization loop
    max_iter = 100
    iter_count = 0
    eval_count = len(candidates)
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1
        
        # Generate candidate suggestions using GP
        n_suggest = 30
        suggest_h = []
        
        for _ in range(n_suggest):
            # Random sample in [0,1]^n_points
            h_suggest = np.random.rand(n_points)
            h_suggest = project_to_feasible(h_suggest, n_points)
            suggest_h.append(h_suggest)
        
        suggest_h = np.array(suggest_h)
        
        # Predict with GP if trained
        try:
            # Predict mean and std
            predictions, stds = gp.predict(suggest_h, return_std=True)
            
            # Select candidates with lowest predicted C5 and highest uncertainty
            scores = predictions + 0.3 * stds  # Exploration/exploitation tradeoff
            best_indices = np.argsort(scores)[:10]
            selected_candidates = suggest_h[best_indices]
        except:
            # Fallback to random if GP fails
            selected_candidates = suggest_h[np.argsort([evaluate_erdos_solution(h, dx) for h in suggest_h])[:10]]
        
        # Evaluate selected candidates
        for h_new in selected_candidates:
            try:
                h_feasible = project_to_feasible(h_new, n_points)
                c5_new = evaluate_erdos_solution(h_feasible, dx)
                eval_count += 1
                
                if c5_new < best_c5:
                    best_h = h_feasible.copy()
                    best_c5 = c5_new
            except:
                continue
        
        # Retrain GP model
        try:
            X_train = np.array(candidates + selected_candidates.tolist())
            y_train = np.array(values + [evaluate_erdos_solution(h, dx) for h in selected_candidates])
            gp.fit(X_train, y_train)
        except:
            pass
    
    # Final refinement with local optimization
    if time() - start_time < budget_s * 0.1:
        try:
            # Try local optimization around best solution
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res_refine = minimize(obj, best_h, method='SLSQP', 
                                bounds=bounds, constraints=cons_sum, 
                                options={'maxiter': 100, 'ftol': 1e-8})
            
            if res_refine.success:
                h_refined = project_to_feasible(res_refine.x, n_points)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
