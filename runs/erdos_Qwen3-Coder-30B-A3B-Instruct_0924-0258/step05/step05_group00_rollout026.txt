import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ASD optimization with adaptive step size
    max_iter = 500
    step_size = 0.1
    decay_rate = 0.995
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation
        noise = np.random.normal(0, step_size, n_points)
        h_perturbed = best_h + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Project to correct integral
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_perturbed, dx)
        
        if c5_new < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_new
            
        # Adapt step size
        step_size *= decay_rate
    
    # QCP refinement
    try:
        h_qcp = cp.Variable(n_points)
        constraints = [
            cp.sum(h_qcp) == n_points / 2.0,
            h_qcp >= 0,
            h_qcp <= 1
        ]
        
        # Add overlap constraints for multiple critical lags
        lags = np.linspace(0, 2, min(20, n_points), endpoint=False)
        for k in lags[1:]:  # skip k=0
            idx = int(k / 2 * n_points)
            if idx < n_points:
                # Use a simplified approach for QCP constraints
                corr_sum = cp.sum(cp.multiply(h_qcp[:n_points-idx], 1 - h_qcp[idx:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_qcp.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible.copy()
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final check and normalization
    try:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        final_c5 = evaluate_erdos_solution(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
