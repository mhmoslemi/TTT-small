import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Use initial construction if available

    # CCP approach with alternating optimization
    max_iter = 1000
    tol = 1e-6
    last_c5 = float('inf')
    
    # Precompute shifts for correlation
    shifts = np.linspace(0, 2, 100)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlations for all shifts to find worst-case
        corr_values = []
        for shift in shifts:
            # Calculate correlation at this shift
            k = int(shift / (2.0 / n_points))
            if k >= n_points or k < 0:
                corr_val = 0
            else:
                # Manual correlation for this shift (sliding window)
                corr_val = 0
                for i in range(n_points):
                    j = i + k
                    if j < n_points:
                        corr_val += h[i] * (1 - h[j])
                corr_val *= (2.0 / n_points)
            corr_values.append(corr_val)
        
        # Find worst-case shift (highest correlation)
        worst_shift_idx = np.argmax(corr_values)
        worst_shift = shifts[worst_shift_idx]
        
        # Solve convex subproblem: minimize sum(h[i](1-h[i+k])) over all valid k
        # We'll do this by fixing the worst shift and optimizing h
        # Let's compute correlation as a function of h, taking historical h
        corr_matrix = np.zeros((n_points, n_points))
        # In the full correlation, for each shift k:
        # corr[k] = sum over i of h[i]*(1-h[i+k]) where i+k < n
        for k in range(n_points):
            for i in range(n_points - k):
                corr_matrix[i, k] += h[i] * (1 - h[i + k])

        # This is not directly convex but we solve via CCP
        # Get current C5 by evaluating full correlation
        current_c5 = evaluate_erdos_solution(h, 2.0 / n_points)

        # maximize np.dot(h, v), subject to constraints
        # Instead, we iterate over all shifts, determine the worst one,

        # So we look at: minimize sum_{i,k} h[i]*h[i+k] over all shifts k
        # This is quadratic in h, hard to solve efficiently.
        # So we apply an alternating approach:
        # Try to estimate a good shift k* that gives the current maximum overlap,
        # then optimize h for minimizing h[i]*h[i+k*]
        
        # Simplified: use a coarse grid of shifts  
        shift_indices = np.linspace(0, n_points, 20, dtype=int)
        best_k = 0
        best_sum = float('inf')

            if k < n_points:
                # Compute overlap for shift k
                overlap_sum = 0
                for i in range(n_points - k):
                    overlap_sum += h[i] * h[i + k]

                    best_sum = overlap_sum

        def objective(h_new):

    return (list(h), current_c5, n_points)
