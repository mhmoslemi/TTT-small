import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Early exit if already good enough
    if best_c5 < 0.3807:
        return (list(best_h), best_c5, n_points)
    
    # Try SDP-based approach first
    try:
        # SDP relaxation approach
        # We'll solve a simpler relaxation: maximize sum of all correlations
        # This can serve as good initialization
        def sdp_initialization():
            # Even simpler: initialize with the known good construction
            # or use smooth step function with local fluctuations
            h = h_start.copy()
            # Add some structured variation
            t = np.linspace(0, 2, n_points)
            # Smooth oscillatory pattern
            h = 0.5 + 0.3 * np.sin(2 * np.pi * t)
            h = np.clip(h, 0, 1)
            # Project to unit integral
            h = h * (n_points/2) / np.sum(h)
            
            # Local refinement using SLSQP
            def objective(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            def constraint_sum(h_vec):
                return np.sum(h_vec) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h = res.x
            return h
            
        h_sdp = sdp_initialization()
        c5_sdp = evaluate_erdos_solution(h_sdp, dx)
        if c5_sdp < best_c5:

            best_c5 = c5_sdp

        pass

            return evaluate_erdos_solution(h, dx)

            return np.sum(h) - n_points / 2.0

                try:

                        if c5_refined < best_local_c5:

                            best_local_c5 = c5_refined

                    pass
                    
            if best_local_c5 < best_c5:

                best_c5 = best_local_c5
                
        except:
            pass
    
    # One final projection and evaluation
    try:
        # Ensure final solution satisfies constraints
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        final_c5 = evaluate_erdos_solution(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
