import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for GA
    pop_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    def initialize_population():
        if 'initial_h_values' in globals():
            population = [h_start.copy()]
            for _ in range(pop_size - 1):
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points/2) / np.sum(h)
                population.append(h)
        else:
            population = []
            for _ in range(pop_size):
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points/2) / np.sum(h)
                population.append(h)
        return population
    
    # Tournament selection
    def tournament_selection(population, fitnesses):
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_index].copy())
        return selected
    
    # Single point crossover
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        crossover_point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    # Mutate individual
    def mutate(individual):
        mutated = individual.copy()
        indices_to_mutate = np.random.choice(n_points, int(mutation_rate * n_points), replace=False)
        for i in indices_to_mutate:
            mutated[i] += random.gauss(0, 0.02)
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points/2) / np.sum(mutated)
        return mutated
    
    # Main GA loop
    population = initialize_population()
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(ind, dx) for ind in population]
        
        # Track best
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitnesses[best_idx]
        
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Create new population
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            parent1, parent2 = random.sample(selected, 2)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
    
    # High resolution refinement
    if time() - start_time < budget_s * 0.8:
        n_points = 40
        dx = 2.0 / n_points
        # Interpolate the best solution to higher resolution
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, len(best_h)), 
                           best_h)
        best_h = h_interp
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
        # Local optimization on refined solution
        max_restarts = 10
        for _ in range(max_restarts):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
                
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                              bounds=bounds, constraints=cons_sum, 
                              options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final local refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
