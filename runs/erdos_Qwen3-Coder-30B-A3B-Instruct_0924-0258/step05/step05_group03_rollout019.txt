import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:

        initial_h = np.ones(n_points) * 0.5
    
    # Setting up the optimization framework
    dx = 2.0 / n_points
    max_iter = 1000
    tol = 1e-6
    max_stagnation = 50
    
    # Start with the initial solution
    h = initial_h.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    stagnation_count = 0
    last_best = best_c5
    
    # CCP-like alternating optimization
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find worst shift k* using sampling
        # Instead of full correlation, sample potential k values
        k_samples = np.linspace(0, 2, 100)
        max_correlation = -1
        best_k = 0
        
        for k in k_samples:
            # Because we're dealing with circular correlation,
            # we need to pad and compute properly
            k_index = int(k / dx)
            if k_index >= n_points:
                k_index = n_points - 1
            if k_index < 0:
                k_index = 0

            # Manually compute correlation with shift k
            correlation = 0
            for i in range(n_points):
                j = (i + k_index) % n_points
                correlation += h[i] * (1 - h[j])
            correlation *= dx
            
            if correlation > max_correlation:
                max_correlation = correlation
                best_k = k
                
        # Formulate the convex subproblem:
        # We optimize over h to minimize sum of h[i] * (1 - h[i+k*])
        
        # Create a linearized objective function for this k
        def objective(h):
            k_index = int(best_k / dx)
            k_index = min(k_index, n_points-1)
            k_index = max(0, k_index)
            
            # Compute linearized correlation term
            obj_val = 0
            for i in range(n_points):
                j = (i + k_index) % n_points
                obj_val += h[i] * (1 - h[j])
            return obj_val * dx
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        # Solve this convex subproblem using SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, h.copy(), method='SLSQP',
                            bounds=bounds, constraints=cons, options={'ftol': 1e-8, 'maxiter': 50})
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                current_c5 = evaluate_erdos_solution(h, dx)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                # Fallback to inertia
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        if stagnation_count > max_stagnation:
            break
            
        if abs(last_best - best_c5) < tol:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        last_best = best_c5
        
        if best_c5 < 0.3807:
            break
    
    # Final polishing with full optimization
    try:

            return evaluate_erdos_solution(h, dx)

            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': sum_constraint}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try to get a better local solution
        res = minimize(obj, best_h, method='SLSQP',

        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass

    return (list(best_h), best_c5, n_points)
