import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for SCP approach
    max_iter = 500
    mu = 1e-3
    c5_bound = 0.4
    tolerance = 1e-6
    
    # Precompute shift indices for overlap calculation
    shift_indices = []
    for k in range(n_points):
        shift_indices.append(np.arange(n_points) - k)
    shift_indices = np.array(shift_indices) % n_points
    
    # Create the full overlap matrix for reference
    def compute_overlaps(h):
        overlaps = []
        for i in range(n_points):
            shifted = h[shift_indices[i]]
            overlap = np.sum(h * (1 - shifted))
            overlaps.append(overlap)
        return np.array(overlaps)
    
    # Run SCP
    for it in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Update C5 bound
        current_overlaps = compute_overlaps(best_h)
        current_max = np.max(current_overlaps) * dx
        c5_bound = min(c5_bound, current_max)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = best_h.copy()
            
        # Early stopping
        if best_c5 < 0.3807:
            break
        
        # Build the QP approximation
        # Vectors and matrices for QP
        h_var = cp.Variable(n_points)
        c5_var = cp.Variable(1)
        
        # Objective: minimize C5
        objective = cp.Minimize(c5_var)
        
        # Constraints
        constraints = []
        
        # Integral constraint
        constraints.append(cp.sum(h_var) == n_points / 2.0)
        
        # Bound constraints
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # Overlap constraints (linearized)
        for i in range(n_points):
            # For each shift k, we want sum(h[j] * (1 - h[(j-k)%n])) <= C5
            # Current linear approximation: sum(h[j] * (1 - h_k[j])) <= C5
            # For a given h_k, this becomes: h @ (1 - h_k) <= C5
            h_k = best_h[shift_indices[i]]
            # Linearize around current h
            lin_term = 1 - h_k
            linearized_constraint = cp.sum(cp.multiply(h_var, lin_term)) <= c5_var[0]
            constraints.append(linearized_constraint)
        
        # Solve the QP
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.OSQP, warm_start=True, max_iter=1000)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                new_h = h_var.value
                # Project in [0,1]
                new_h = np.clip(new_h, 0, 1)
                # Re-normalize
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                # Update
                best_h = new_h
            else:
                # Break if problem cannot be solved
                break
        except:
            break
    
    # Final local refinement of best solution using SLSQP
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
