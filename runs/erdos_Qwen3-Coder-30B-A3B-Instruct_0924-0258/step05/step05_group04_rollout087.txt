import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    num_particles = 40
    max_iter = 500
    inertia_weight = 0.7
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.2
    warm_restart_after = 200
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    for i in range(num_particles):
        if i < len(h_start):
            h = h_start.copy()
        else:
            h = np.random.random(n_points)
            h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_fitness.append(evaluate_erdos_solution(h, dx))
    
    # Global best initialization
    gbest_idx = np.argmin(pbest_fitness)
    gbest = pbest[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    # PSO iterations
    stagnation_count = 0
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        if stagnation_count >= warm_restart_after:
            # Warm restart
            for i in range(num_particles):
                velocities[i] = np.zeros(n_points)
                # Perturb some particles
                if random.random() < 0.5:
                    particles[i] = np.random.random(n_points)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                    pbest[i] = particles[i].copy()
                    pbest_fitness[i] = evaluate_erdos_solution(particles[i], dx)
            stagnation_count = 0
            gbest_idx = np.argmin(pbest_fitness)
            gbest = pbest[gbest_idx].copy()
            gbest_fitness = pbest_fitness[gbest_idx]
        
        # Update particles
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = c1 * r1 * (pbest[i] - particles[i])
            social = c2 * r2 * (gbest - particles[i])
            velocities[i] = inertia_weight * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h > 1e-10:
                particles[i] = particles[i] * (n_points / 2) / sum_h
            else:
                particles[i] = np.ones(n_points) * (n_points / 2)
            
            # Evaluate fitness
            fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = fitness
                    stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Early stopping
        if gbest_fitness < 0.3807:
            break
    
    # Final local refinement using SLSQP
    if time() - start_time < max_time - 30:
        h_init = gbest.copy()
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < gbest_fitness:
                    gbest = h_refined
                    gbest_fitness = c5_refined
        except:
            pass
    
    return (list(gbest), gbest_fitness, n_points)
