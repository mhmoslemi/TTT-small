import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization
    resolutions = [35, 40, 50, 60]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Problem formulation
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add overlap constraints
        # For efficiency, only include significant constraints
        max_k = n_points // 2
        for k in range(1, max_k):
            # Only compute a few key shifts
            if k < 5 or k > max_k - 5 or k % 3 == 0:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-8)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            # Try multiple solvers
            solved = False
            for solver in [cp.SCS, cp.ECOS, cp.GLPK]:
                if solved or time() - start_time > budget_s * 0.95:
                    break
                try:
                    prob.solve(solver=solver, eps=1e-6, verbose=False)
                    if prob.status == cp.OPTIMAL:
                        h_feasible = h_opt.value
                        h_feasible = np.clip(h_feasible, 0, 1)
                        h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
                        c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                        if c5_feasible < best_c5:
                            best_h = h_feasible
                            best_c5 = c5_feasible
                        solved = True
                        break
                except:
                    pass
            
            if not solved:
                # Use simple local search if SDP fails
                h_trial = best_h.copy()
                h_trial = np.clip(h_trial, 0, 1)
                for _ in range(100):
                    if time() - start_time > budget_s * 0.95:
                        break
                    # Gradient-free local search
                    h_perturbed = h_trial + np.random.normal(0, 0.02, n_points)
                    h_perturbed = np.clip(h_perturbed, 0, 1)
                    h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
                    c5_trial = evaluate_erdos_solution(h_perturbed, dx)
                    if c5_trial < best_c5:
                        best_h = h_perturbed
                        best_c5 = c5_trial
                        h_trial = h_perturbed.copy()
                    else:
                        h_trial = h_trial * 0.99 + h_perturbed * 0.01  # Small step
                    
        except:
            pass
            
        # Local optimization with SLSQP
        if time() - start_time < budget_s * 0.95:
            # Start from current best and do careful local minimization
            h_restart = best_h.copy()
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass

        h_start = best_h.copy()
    
    # Final polishing with gradient-based optimization
    if time() - start_time < budget_s * 0.90:
        try:
            n_points = len(best_h)
            dx = 2.0 / n_points
            h_restart = best_h.copy()
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_restart, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    # Final validation
    try:
        final_h = np.clip(best_h, 0, 1)
        final_h = final_h * (len(final_h)/2) / np.sum(final_h)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, len(best_h))
