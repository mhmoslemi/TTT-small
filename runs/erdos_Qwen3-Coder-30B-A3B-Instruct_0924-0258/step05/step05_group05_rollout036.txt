import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        # Initial construction based on symmetric triangular function
        mid = n_points // 2
        h_start = np.minimum(np.arange(n_points), 2*mid - np.arange(n_points))
        h_start = h_start / (mid)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization parameters
    max_iter = 200
    tolerance = 1e-8
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Main iteration loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Save current best
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        
        # Gradient-based refinement with constraints
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        # Try to improve with L-BFGS-B
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, current_h, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons_sum,
                          options={'ftol': tolerance, 'gtol': tolerance, 'maxiter': 100})
            if res.success:
                new_h = res.x
                new_c5 = objective(new_h)

            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            c5_pert = evaluate_erdos_solution(h_pert, dx)
            if c5_pert < current_c5:
                current_h = h_pert
                current_c5 = c5_pert

                return np.sum(h) - n_points / 2.0

        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
