import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 50
    max_generations = 1000
    mutation_rate = 0.15
    crossover_rate = 0.8
    tournament_size = 3
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # EA loop
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time() - start_time < budget_s * 0.9 and generation < max_generations:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Track best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Reset if no improvement for too long
        if no_improvement_count > max_no_improvement:
            # Reset with better initial solution
            if 'initial_h_values' in globals():
                population = []
                for _ in range(pop_size):
                    h = initial_h_values + np.random.normal(0, 0.05, n_points)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points/2) / np.sum(h)
                    population.append(h)
            else:
                population = []
                for _ in range(pop_size):
                    h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points/2) / np.sum(h)
                    population.append(h)
            no_improvement_count = 0
        
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < pop_size else selected[0]
            
            if random.random() < crossover_rate:
                # One-point crossover
                crossover_point = random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if random.random() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.03)
                if random.random() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.03)
            
            # Ensure constraints
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
        generation += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
