import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Use initial construction if available

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization with careful nesting
    resolutions = [35, 40, 50, 60]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Stronger local optimization using Nelder-Mead with restarts
        restarts = 0
        max_restarts = 10
        while time() - start_time < budget_s * 0.95 and restarts < max_restarts:
            try:
                # Random perturbation for restart
                h_restart = h_current + np.random.normal(0, 0.02, n_points)
                h_restart = np.clip(h_restart, 0, 1)
                h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                # Use Nelder-Mead for robustness with limited evaluations
                res = minimize(objective, h_restart, method='Nelder-Mead',
                               options={'maxiter': 300, 'ftol': 1e-8, 'xatol': 1e-8})
                
                if res.success:
                    h_refined = res.x.copy()
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = c5_refined
                        restarts = 0
                    else:
                        restarts += 1
                else:
                    restarts += 1
                    
            except Exception:
                restarts += 1
        
        h_start = best_h.copy()
    
    # Final QCQP with fewer constraints for computational efficiency
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sampleed subset of constraints to reduce complexity
        # Consider only those k values where overlap is significant
        sample_k = np.unique(np.concatenate([
            np.arange(1, int(n_points/4)),
            np.arange(int(n_points/4), int(n_points/2), 3),
            np.arange(int(n_points/2), n_points-1, 5)
        ]))
        
        for k in sample_k:
            if k <= n_points//2:
                # Left shift case
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                # Right shift case (circular)
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        # Add one additional global constraint directly from the best solution
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr = np.max(corr_full)
        constraints.append(cp.sum(cp.multiply(h_opt, 1 - h_opt)) <= max_corr * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iter=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final refinement if time remains
    try:
        h_final = np.clip(best_h, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
