import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create symmetric triangular shape
        mid = n_points // 2
        h_start = np.minimum(np.arange(n_points), 2*mid - np.arange(n_points))
        h_start = h_start / (mid)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based refinement with constraints
    max_iter = 1000
    stagnation_limit = 50
    stagnation_count = 0
    last_improvement = 0
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # L-BFGS-B optimization with constraints
        try:
            res = minimize(objective, current_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-9})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                
                # Apply constraints to ensure feasibility
                h_refined = np.clip(h_refined, 0, 1)
                if np.sum(h_refined) != 0:
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = objective(h_refined)
                
                if c5_refined < current_c5:
                    current_h = h_refined
                    current_c5 = c5_refined
                    stagnation_count = 0
                    last_improvement = i
                    
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
                        
                    if best_c5 < 0.3807:
                        break
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
        except:
            stagnation_count += 1
            
        # Random perturbations to escape local minima
        if stagnation_count > 20 and i - last_improvement > 30:
            h_perturbed = current_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            if np.sum(h_perturbed) != 0:
                h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            c5_perturbed = objective(h_perturbed)
            if c5_perturbed < current_c5:
                current_h = h_perturbed
                current_c5 = c5_perturbed
                stagnation_count = 0
                
        # Occasionally restart with a different initialization
        if i % 100 == 0 and i > 0:
            # Create a new one
            mid = n_points // 2
            h_restart = np.minimum(np.arange(n_points), 2*mid - np.arange(n_points))
            h_restart = h_restart / (mid)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
            current_h = h_restart
            current_c5 = evaluate_erdos_solution(current_h, dx)
    
    # Final polishing
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                h_final = np.clip(h_final, 0, 1)
                if np.sum(h_final) != 0:
                    h_final = h_final * (n_points / 2) / np.sum(h_final)
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
