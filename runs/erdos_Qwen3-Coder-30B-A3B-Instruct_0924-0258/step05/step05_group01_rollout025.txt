import numpy as np
from scipy.signal import correlate
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create triangular-type initial guess
        x = np.linspace(0, 2, 35)
        h_start = np.maximum(0, 1 - np.abs(x - 1) / 0.5)
        h_start = h_start / np.sum(h_start) * 17.5  # Scale to sum = 17.5
        n_points = 35
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Scale levels for refinement
    scale_levels = [35, 50, 75, 100]
    current_scale = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale < len(scale_levels):
        current_res = scale_levels[current_scale]
        
        # Interpolate to current resolution if needed
        if current_res > n_points:
            x_original = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, current_res)
            h_interp = np.interp(x_new, x_original, best_h)
            h_current = h_interp
        else:
            h_current = best_h.copy()
        
        n_points = current_res
        dx = 2.0 / n_points
        
        # Gradient-based optimization with multiple restarts
        max_restarts = 30
        for restart in range(max_restarts):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Random slight perturbation
            noise = np.random.normal(0, 0.01, n_points)
            h_perturbed = h_current + noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            
            # Gradient descent step
            learning_rate = 0.02
            for step in range(100):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Compute flat correlation
                corr = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
                max_idx = np.argmax(corr)
                max_corr = corr[max_idx]
                
                # Compute gradient w.r.t. h (simplified approach)
                # We'll use finite differences for the gradient at the max overlap point
                h_grad = np.zeros_like(h_perturbed)
                
                # Finite difference for change of shift
                shift = max_idx - (n_points - 1)
                if shift >= 0:
                    h_grad[shift:] += 1
                else:
                    h_grad[:shift] -= 1
                    
                # Apply gradient
                h_new = h_perturbed - learning_rate * h_grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                
                # Update if improvement
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                h_perturbed = h_new.copy()
                
        # Keep current best solution
        if current_scale == 0 or best_c5 < evaluate_erdos_solution(best_h, dx):
            h_current = best_h.copy()
        else:
            h_current = best_h.copy()
            
        current_scale += 1
    
    # Final validation of solution
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
