import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach with coarse grid initialization
    n_low = 20
    
    # Build coarse solution using convex relaxation
    try:
        h_low = cp.Variable(n_low)
        constraints = [
            cp.sum(h_low) == n_low / 2.0,
            h_low >= 0,
            h_low <= 1
        ]
        
        # Add constraints on a subset of shifts for tractability
        for k in range(1, min(15, n_low//2)):
            if k < n_low // 2:
                corr_sum = cp.sum(cp.multiply(h_low[:n_low-k], 1 - h_low[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_low[k:], 1 - h_low[:n_low-k]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_low)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            h_coarse = h_low.value
        else:
            h_coarse = np.random.rand(n_low)
            h_coarse = h_coarse * (n_low/2) / np.sum(h_coarse)
    except Exception:
        h_coarse = np.random.rand(n_low)
        h_coarse = h_coarse * (n_low/2) / np.sum(h_coarse)
    
    # Interpolate to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, n_low), h_coarse)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_points/2) / np.sum(h_fine)
    
    # Initial fine grid solution
    best_h = h_fine
    best_c5 = evaluate_erdos_solution(h_fine, dx)
    
    # Refinement loop with restarts
    iters_no_improvement = 0
    max_restart_iters = 50
    
    while time() - start_time < budget_s * 0.95:
        # Local optimization using SLSQP
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-7}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    iters_no_improvement = 0
                else:
                    iters_no_improvement += 1
            else:
                iters_no_improvement += 1
        except Exception:
            iters_no_improvement += 1
        
        # Restart with a perturbed solution if no improvement
        if iters_no_improvement > max_restart_iters:
            h_perturbed = best_h + 0.05 * np.random.randn(n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            best_h = h_perturbed
            best_c5 = evaluate_erdos_solution(h_perturbed, dx)
            iters_no_improvement = 0
        
        # Conservative early exit
        if time() - start_time > budget_s * 0.99:
            break
    
    # Final constraint normalization
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
