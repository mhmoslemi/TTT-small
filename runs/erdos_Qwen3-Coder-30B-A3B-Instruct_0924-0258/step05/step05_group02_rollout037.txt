import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Enhanced local optimization
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def local_refine(h_initial, max_iter=50):
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_initial, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': max_iter, 'ftol': 1e-8}
            )
            if res.success:
                return res.x
        except Exception:
            pass
        return h_initial

    # Enhanced global search with multiple strategies
    restart_count = 0
    max_restarts = 30
    iteration = 0
    
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        # Strategy 1: Random perturbation with refinement
        h_perturbed = best_h.copy()
        h_perturbed += np.random.normal(0, 0.015, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        h_refined = local_refine(h_perturbed, max_iter=100)
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
            restart_count = 0
        else:
            restart_count += 1
            
        # Strategy 2: Global restart with better construction
        if restart_count > max_restarts:
            # Try to construct a better pattern
            h_new = np.zeros(n_points)
            # Place a pulse in center with smooth edges
            center = n_points // 2
            width = n_points // 4
            for i in range(max(0, center-width//2), min(n_points, center+width//2)):
                value = 1.0
                if i < center - width//4:
                    value = 0.5
                elif i > center + width//4:
                    value = 0.5
                h_new[i] = value
            
            # Normalize
            h_new = h_new * (n_points/2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            
            h_new = local_refine(h_new, max_iter=50)
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                restart_count = 0

        # Strategy 3: Convex relaxation for constraint satisfaction
        if iteration % 10 == 0:
            try:
                h_relaxed = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_relaxed) == n_points / 2.0,
                    h_relaxed >= 0,
                    h_relaxed <= 1
                ]
                
                # Add some overlap constraints to approximate max overlap
                for k in range(1, min(15, n_points//2)):  
                    corr_sum = cp.sum(cp.multiply(h_relaxed[:n_points-k], 1 - h_relaxed[k:]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-5)
                
                prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_relaxed.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass

        if time() - start_time > budget_s * 0.9:
            break

    # Final refinement
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
