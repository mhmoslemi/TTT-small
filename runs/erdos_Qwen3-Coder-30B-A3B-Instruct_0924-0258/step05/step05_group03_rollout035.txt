import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Ensure feasibility
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Start with the best known solution
    h = initial_h.copy()
    current_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = current_c5
    
    # Parameters
    max_iter = 500
    stagnation_limit = 50
    sample_shifts = 50
    
    # For gradient computation
    dx = 2.0 / n_points
    
    iteration = 0
    stagnation_count = 0
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iter:
        # Sample shifts to evaluate
        shifts = np.random.choice(n_points, sample_shifts, replace=False)
        
        max_overlap = -1
        best_shift = 0
        
        # Find shift with maximum overlap
        for k in shifts:
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlap *= dx
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = k
        
        # Compute gradient for the maximal shift
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_shift) % n_points
            gradient[i] += (1 - h[j]) * dx
            gradient[j] -= h[i] * dx
            
        # Gradient descent step
        step_size = 0.01
        h_new = h - step_size * gradient
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check improvement
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            stagnation_count = 0
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            stagnation_count += 1
            
        # Restart if stagnation
        if stagnation_count > stagnation_limit:
            h = initial_h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0
            
        # Early termination
        if best_c5 < 0.3807:
            break
            
        iteration += 1
    
    # Final refinement using scipy
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
