import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with existing construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for GA
    pop_size = 50
    elitism = 15
    mutation_rate = 0.1
    max_generations = 1000
    generation = 0
    stagnation_count = 0
    max_stagnation = 50
    
    # Initialize population
    population = []
    population.append(best_h.copy())
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    # GA main loop
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            fitness.append(evaluate_erdos_solution(h, dx))
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if fitness[0] < best_c5:
            best_h = sorted_pop[0].copy()
            best_c5 = fitness[0]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > max_stagnation:
            break
            
        # Select elite
        elite = sorted_pop[:elitism]
        
        # Create new population
        new_pop = elite.copy()
        
        # Crossover and mutation
        while len(new_pop) < pop_size:
            # Parent selection (tournament)
            parents = []
            for _ in range(2):
                tournament_size = 3
                tournament_indices = np.random.choice(len(elite), tournament_size, replace=False)
                tournament_fitness = [fitness[i] for i in tournament_indices]
                winner_idx = tournament_indices[np.argmin(tournament_fitness)]
                parents.append(elite[winner_idx])
            
            # Crossover (uniform with bias towards sparsity)
            alpha = np.random.uniform(0.4, 0.6)
            child = alpha * parents[0] + (1 - alpha) * parents[1]
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.normal(0, 0.03)
            
            # Enforce bounds and normalization constraint
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_pop.append(child)
        
        population = new_pop
        generation += 1
    
    # Final gradient refinement of best solution
    h_refined = best_h.copy()
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    try:
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            objective,
            h_refined,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if res.success:
            h_refined = res.x
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final cleanup
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), c5_final, n_points)
