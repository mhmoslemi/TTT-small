import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach
    n_coarse = min(20, n_points)
    dx_coarse = 2.0 / n_coarse
    
    def coarse_optimization():
        # Create coarse grid optimization
        h_low = cp.Variable(n_coarse)
        constraints = [
            cp.sum(h_low) == n_coarse / 2.0,
            h_low >= 0,
            h_low <= 1
        ]
        
        try:
            # Add a small subset of overlap constraints to make it manageable
            for k in range(1, min(8, n_coarse)):
                if k < n_coarse // 2:
                    corr_sum = cp.sum(cp.multiply(h_low[:n_coarse-k], 1 - h_low[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_low[k:], 1 - h_low[:n_coarse-k]))
                constraints.append(corr_sum <= best_c5 * dx_coarse + 1e-6)
            
            prob = cp.Problem(cp.Minimize(cp.sum(h_low)), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-7)
            
            if prob.status == cp.OPTIMAL:
                return h_low.value
        except Exception:
            pass
        return None
    
    # Run coarse optimization
    h_coarse = coarse_optimization()
    if h_coarse is not None:
        # Interpolate to full resolution
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, n_coarse), h_coarse)
        h_interp = h_interp * (n_points/2) / np.sum(h_interp)
        h_interp = np.clip(h_interp, 0, 1)
        c5_interp = evaluate_erdos_solution(h_interp, dx)
        if c5_interp < best_c5:
            best_h = h_interp
            best_c5 = c5_interp
    
    # Improve with local refinement
    max_restarts = 100
    iter_count = 0
    restart_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < 10000:
        iter_count += 1
        
        # Refine with SLSQP
        try:
            # Use a subset of the solution as initial guess
            h_refine = best_h.copy()
            
            # Add small noise or use interpolated version as starting point
            if iter_count % 10 == 0:
                # Perturb more significantly every 10 iterations
                h_perturb = h_refine + np.random.normal(0, 0.05, n_points)
                h_perturb = np.clip(h_perturb, 0, 1)
                h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
                h_refine = h_perturb
            
            cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_refine, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 20, 'ftol': 1e-6}
            )
            
            if res.success:
                h_new = res.x
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    restart_count = 0
                else:
                    restart_count += 1
            else:
                restart_count += 1
        except Exception:
            restart_count += 1

        # Periodic restart with random search
        if restart_count > 10 and time() - start_time < budget_s * 0.9:
            h_restart = np.random.rand(n_points)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            h_restart = np.clip(h_restart, 0, 1)
            c5_restart = evaluate_erdos_solution(h_restart, dx)
            if c5_restart < best_c5:
                best_h = h_restart
                best_c5 = c5_restart
            restart_count = 0
            iter_count = 0  # Restart iteration counter
            
        # Early stop condition
        if restart_count > max_restarts:
            break
    
    # Final validation
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
