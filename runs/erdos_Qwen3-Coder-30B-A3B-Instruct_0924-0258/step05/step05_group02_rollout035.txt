import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different search strategies
    def local_search(h, max_iter=50):
        current_h = h.copy()
        current_c5 = evaluate_erdos_solution(current_h, dx)
        best_local_h = current_h.copy()
        best_local_c5 = current_c5
        
        for i in range(max_iter):
            # Perturb
            h_pert = current_h + np.random.normal(0, 0.02, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            
            c5_pert = evaluate_erdos_solution(h_pert, dx)
            
            if c5_pert < current_c5:
                current_h = h_pert
                current_c5 = c5_pert
                if c5_pert < best_local_c5:
                    best_local_h = h_pert
                    best_local_c5 = c5_pert
            
            if time() - start_time > budget_s * 0.95:
                break
                
        return best_local_h, best_local_c5
    
    # Try optimization with different initializations
    def optimize_with_strategy(initial_h, max_iter=50):
        h_init = initial_h.copy()
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points/2) / np.sum(h_init)
        
        # Simple local search
        h_opt, c5_opt = local_search(h_init, max_iter)
        
        # Try CvxPy if feasible
        try:
            h_cvx = cp.Variable(n_points)
            constraints = [
                cp.sum(h_cvx) == n_points / 2.0,
                h_cvx >= 0,
                h_cvx <= 1
            ]
            
            # Simplified constraint - focus on largest few correlations
            for k in range(1, min(5, n_points)):  
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_cvx[:n_points-k], 1 - h_cvx[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_cvx[k:], 1 - h_cvx[:n_points-k]))
                constraints.append(corr_sum <= c5_opt * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(cp.sum(h_cvx)), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
            
            if prob.status == cp.OPTIMAL:
                h_feas = h_cvx.value
                c5_feas = evaluate_erdos_solution(h_feas, dx)
                if c5_feas < c5_opt:
                    h_opt, c5_opt = h_feas, c5_feas
        except Exception:
            pass
            
        return h_opt, c5_opt
    
    # Multiple restarts with different strategies
    max_restarts = 100
    restart_count = 0
    
    while time() - start_time < budget_s * 0.95:
        # Strategy 1: Random restart with local search
        h_random = np.random.rand(n_points) * 0.5 + 0.25
        h_random = h_random * (n_points/2) / np.sum(h_random)
        h_opt1, c5_opt1 = optimize_with_strategy(h_random, max_iter=20)
        
        # Strategy 2: Modified initial construction
        h_mod1 = best_h + np.random.normal(0, 0.01, n_points)
        h_mod1 = np.clip(h_mod1, 0, 1)
        h_mod1 = h_mod1 * (n_points/2) / np.sum(h_mod1)
        h_opt2, c5_opt2 = optimize_with_strategy(h_mod1, max_iter=20)
        
        # Strategy 3: Extend more towards peaked structure
        h_peak = np.zeros(n_points)
        h_peak[5:10] = 1.0
        h_peak = h_peak * (n_points/2) / np.sum(h_peak)
        h_opt3, c5_opt3 = optimize_with_strategy(h_peak, max_iter=20)
        
        # Keep best solution found
        best_of_three = min(c5_opt1, c5_opt2, c5_opt3)
        if best_of_three < best_c5:
            if best_of_three == c5_opt1:
                best_h = h_opt1
            elif best_of_three == c5_opt2:
                best_h = h_opt2
            else:
                best_h = h_opt3
            best_c5 = best_of_three
            restart_count = 0
        else:
            restart_count += 1
            
        # Stop after too many failed attempts
        if restart_count > max_restarts:
            break
    
    # Final validation
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
