import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a symmetric triangular initial guess
        mid = n_points // 2
        h_start = np.minimum(np.arange(n_points), 2*mid - np.arange(n_points))
        h_start = h_start / (mid)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # For local refinement - use L-BFGS-B
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    # Optimization parameters
    max_iter = 1000
    patience = 50
    patience_counter = 0
    best_iter = 0

    # Initialize
    h_current = h_start.copy()
    c5_current = best_c5
    last_c5 = c5_current + 1.0

    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break

        # Local optimization using L-BFGS-B with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}

        try:
            res = minimize(objective, h_current, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < c5_current:
                    h_current = h_refined
                    c5_current = c5_refined
        except:
            pass

        # Ensure feasibility
        h_current = np.clip(h_current, 0, 1)
        if np.sum(h_current) > 0:
            h_current = h_current * (n_points / 2) / np.sum(h_current)
        
        # Update best solution
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            best_iter = iteration
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Early termination if no improvement
        if patience_counter > patience:
            break

    # Final refinement step with more detailed optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        try:
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=[(0,1) for _ in range(n_points)], 
                           constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass

    return (list(best_h), best_c5, n_points)
