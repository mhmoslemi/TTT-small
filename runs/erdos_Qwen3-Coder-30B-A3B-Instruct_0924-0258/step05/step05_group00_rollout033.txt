import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_current)
    else:

        h_current = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # GA parameters
    pop_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 5
    tournament_size = 5
    
    # Initial population
    population = []
    for _ in range(pop_size):
        h = np.random.random(n_points)
        h = repair_h(h)
        population.append(h)
    
    # If initial_h_values is provided, use it as first individual
    if 'initial_h_values' in globals():
        population[0] = h_current.copy()
    
    # Evolution loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        sorted_indices = np.argsort(fitness)
        
        # Track best
        if fitness[sorted_indices[0]] < best_c5:
            best_h = population[sorted_indices[0]].copy()
            best_c5 = fitness[sorted_indices[0]]
        
        # Selection (tournament)
        new_population = []
        # Elite
        for i in range(elite_size):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            tourn_indices = np.random.choice(sorted_indices[:pop_size//2], 
                                           tournament_size, replace=False)
            parent1 = population[tourn_indices[np.argmin([fitness[i] for i in tourn_indices])]]
            
            tourn_indices = np.random.choice(sorted_indices[:pop_size//2], 
                                           tournament_size, replace=False)
            parent2 = population[tourn_indices[np.argmin([fitness[i] for i in tourn_indices])]]
            
            # Crossover
            if np.random.random() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                for i in range(n_points):
                    if np.random.random() < mutation_rate:
                        child[i] += np.random.normal(0, 0.02)
                child = repair_h(child)
                
                if len(new_population) < pop_size:
                    new_population.append(child)
                if len(new_population) < pop_size:
                    new_population.append(child)

    # Resolution upgrade if needed and time allows
    if best_c5 > 0.3808 and time() - start_time < budget_s * 0.8:

        dx = 2.0 / n_points
        h_interp = np.interp(np.linspace(0, 2, n_points),

        best_h = repair_h(h_interp)
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
        # Refine with local optimization
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)

                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP',

                best_h = res.x
                best_c5 = objective(best_h)
        except Exception:
            pass
    
    # Final refinement
    try:
        best_h = repair_h(best_h)
        best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
