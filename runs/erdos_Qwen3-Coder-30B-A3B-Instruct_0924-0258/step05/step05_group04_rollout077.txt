import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    max_iter = 5000
    mu = 1e-3
    tol = 1e-6
    rho = 1.0
    
    # Initialize variables
    h = h_start.copy()
    h_prev = h.copy()
    z = np.zeros(2 * n_points - 1)
    lambda_vec = np.zeros(2 * n_points - 1)
    
    # Compute initial overlaps for monitoring
    def compute_overlap(h):
        return evaluate_erdos_solution(h, dx)
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    remaining_time = max_time * 0.95
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Update h using proximal gradient descent on augmented Lagrangian
        # We solve: min h (1/2)||h - h_prev||^2 + mu * sum(alpha_k * (h * (1 - h_shifted) - z_k + lambda_k/mu)^2)
        # subject to h in [0,1] and sum(h) = n_points/2
        
        def augmented_lagrangian(h):
            # Create shifted versions of (1-h)
            shifted_ones_minus_h = np.concatenate([np.ones(n_points-1), 1-h, np.ones(n_points-1)])
            conv_result = np.convolve(h, shifted_ones_minus_h, mode='valid')
            diff = conv_result - z + lambda_vec / mu
            penalty = 0.5 * mu * np.sum(diff**2)
            
            # Add projection penalty
            penalty += 1e10 * np.maximum(0, np.sum(h) - n_points/2)**2
            penalty += 1e10 * np.maximum(0, n_points/2 - np.sum(h))**2
            
            return penalty
        
        def gradient_aug_lagrangian(h):
            shifted_ones_minus_h = np.concatenate([np.ones(n_points-1), 1-h, np.ones(n_points-1)])
            conv_result = np.convolve(h, shifted_ones_minus_h, mode='valid')
            diff = conv_result - z + lambda_vec / mu
            
            # Compute derivative
            grad = np.zeros_like(h)
            for k in range(n_points):
                contrib = np.zeros(2*n_points - 1)
                contrib[k: k+n_points] = (1-h)  # derivative of h[i] term on conv result
                contrib[2*n_points-2-k: 2*n_points-2-k+n_points] = -h  # derivative in reverse
                # Convolve with diff
                conv_diff = np.convolve(diff, contrib, mode='valid')
                grad[k] = conv_diff[k] * mu + (h[k] - h_prev[k]) - 2 * 1e10 * np.maximum(0, np.sum(h) - n_points/2) + 2 * 1e10 * np.maximum(0, n_points/2 - np.sum(h))
            return grad
        
        # Simple solve with L-BFGS-B within box constraints
        bounds = [(0,1) for _ in range(n_points)]
        try:
            # Simple quasi-Newton update without explicit gradient computation
            res = minimize(
                lambda x: augmented_lagrangian(x),
                h,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 100}
            )
            if res.success:
                # Project solution properly
                h_new = project_h(res.x)
                # Update h if improvement
                if compute_overlap(h_new) < compute_overlap(h):
                    h = h_new
                else:
                    h = project_h(h)
        except Exception as e:
            # fallback: just project
            h = project_h(h)
        
        # Update z and lambda (ADMM steps)
        shifted_ones_minus_h = np.concatenate([np.ones(n_points-1), 1-h, np.ones(n_points-1)]) 
        conv_h = np.convolve(h, shifted_ones_minus_h, mode='valid') 
        z_new = z + lambda_vec/mu + conv_h - (1/mu) * (conv_h - z)
        
        # Project z to be within bounds (for numerical reasons, but not required)
        z = z_new
        
        # Update lambda
        lambda_vec = lambda_vec + mu * (conv_h - z)
        
        # Update h_prev
        h_prev = h.copy()
        
        # Update best solution if improved
        c5_current = compute_overlap(h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
        
        # Early stopping
        if best_c5 < 0.3807:
            break
            
        # Check convergence every 50 steps
        if iteration % 50 == 0:
            # Check if overlap is decreasing enough
            if iteration > 0 and abs(compute_overlap(h) - compute_overlap(h_prev)) < tol:
                break
    
    # Final refinement with full SLSQP to reduce numerical errors
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        try:
            h_refined = best_h.copy()
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            def con_sum(h):
                return np.sum(h) - n_points / 2.0
                
            cons = {'type': 'eq', 'fun': con_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(obj, h_refined, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                c5_final = obj(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass

    return (list(best_h), best_c5, n_points)
