import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Initialize with a triangular function
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start / np.sum(h_start) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T0 = 0.1
    alpha = 0.99
    max_iter = int(1e5)
    
    # Multi-resolution levels
    scales = [35, 50, 75, 100]
    
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Process scales from coarse to fine
    for scale in scales:
        if scale > n_points:
            # Interpolate current solution to higher resolution
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            current_h = np.interp(x_new, x_old, current_h)
            n_points = scale
            dx = 2.0 / n_points
        
        # Target resolution: number of samples we want
        target_res = n_points
        T = T0
        iteration = 0
        
        # Annealing loop
        while iteration < max_iter and time() - start_time < budget_s * 0.95:
            # Perturb current solution
            h_new = current_h.copy()
            
            # Flip some values randomly
            flip_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            for idx in flip_indices:
                # Perturb a bit, within bounds
                h_new[idx] += np.random.normal(0, 0.05)
            
            # Enforce bounds
            h_new = np.clip(h_new, 0, 1)
            # Enforce sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Calculate new C5
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            if c5_new < current_c5:
                current_h = h_new
                current_c5 = c5_new
            else:
                # Accept with probability based on temperature
                delta = c5_new - current_c5
                if np.random.rand() < math.exp(-delta / T):
                    current_h = h_new
                    current_c5 = c5_new
            
            # Update temperature
            T *= alpha
            iteration += 1
            
            # Update best so far
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        if time() - start_time > budget_s * 0.95:
            break
    
    # Final refinement step
    if time() - start_time < budget_s * 0.95:
        # Fit a smooth function to the best solution and re-optimize
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    return (list(best_h), best_c5, n_points)
