import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization with SDP
    resolutions = [35, 40, 50, 60]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Integer programming relaxation approach
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add constraints on overlaps (only a few key lags for computational efficiency)
            key_lags = list(range(1, min(10, n_points//2)))
            for k in key_lags:
                # For the most effective overlap constraints
                corr_cross = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_cross <= best_c5 * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                if np.all(h_feasible >= 0) and np.all(h_feasible <= 1):
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
                        
        except Exception:
            pass
        
        # Local search to improve result
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            # Try different optimization methods with different starting points
            restarts = 20
            for _ in range(restarts):
                if time() - start_time > budget_s * 0.95:
                    break
                h_perturbed = h_current + np.random.normal(0, 0.02, n_points)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': constraint_sum}
                
                res = minimize(objective, h_perturbed, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        
        except Exception:
            pass
            
        h_start = best_h.copy()
    
    # Final SLSQP refinement on the best solution found
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
