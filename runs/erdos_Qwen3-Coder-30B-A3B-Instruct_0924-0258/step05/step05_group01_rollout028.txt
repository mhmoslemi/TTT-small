import numpy as np
from scipy.signal import correlate
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:

        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Scale levels to try
    scales = [35, 50, 75, 100]
    current_scale_idx = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale_idx < len(scales):
        scale = scales[current_scale_idx]
        
        # Resample to target scale if needed
        if scale != n_points:
            h_resampled = np.interp(np.linspace(0, 2, scale), np.linspace(0, 2, n_points), best_h)
            n_points = scale
            dx = 2.0 / n_points
        else:
            h_resampled = best_h.copy()
        
        # Gradient-based optimization at this scale
        h_opt = h_resampled.copy()
        learning_rate = 0.01
        max_iter = 100
        last_improvement = 0
        
        for iter_num in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            c5_current = evaluate_erdos_solution(h_opt, dx)
            
            # Find the shift k that maximizes correlation
            corr = correlate(h_opt, 1 - h_opt, mode='full')
            k_max = np.argmax(corr)
            # Get the actual offset in range [0, 2*dx)
            shift_offset = k_max - (len(h_opt) - 1)
            actual_shift = shift_offset * dx
            
            # Compute gradient w.r. to h at this shift
            grad = np.zeros_like(h_opt)
            # Approximate gradient through finite differences for stochastic gradient
            eps = 1e-5
            h_plus = h_opt + eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            # Finite difference estimation of gradient
            grad = (c5_plus - c5_current) / eps
            
            # Apply gradient step
            h_next = h_opt - learning_rate * grad
            h_next = np.clip(h_next, 0, 1)
            h_next = h_next * (n_points/2) / np.sum(h_next)
            
            # Update
            h_opt = h_next
            
            # Check for improvement
            c5_new = evaluate_erdos_solution(h_opt, dx)
            if c5_new < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_new
                last_improvement = 0
                learning_rate = min(0.1, learning_rate * 1.05)  # Increase learning rate if improving
            else:
                last_improvement += 1
                if last_improvement > 10:
                    learning_rate *= 0.9  # Decrease learning rate if stuck
                if learning_rate < 1e-6:
                    break

        current_scale_idx += 1
    
    # Final verification at highest resolution
    if time() - start_time < budget_s * 0.1:
        # Try a few more local refinements
        h_final = best_h.copy()
        for _ in range(10):
            if time() - start_time > budget_s * 0.99:
                break
            # Do small random perturbations
            h_perturb = h_final + np.random.normal(0, 0.01, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
            c5_perturb = evaluate_erdos_solution(h_perturb, dx)
            if c5_perturb < best_c5:
                best_h = h_perturb.copy()
                best_c5 = c5_perturb
    
    return (list(best_h), best_c5, n_points)
