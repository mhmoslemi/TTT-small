import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def normalize_h(h):
        return h * (n_points/2) / np.sum(h)
    
    def mutate(h, noise_std=0.02):
        h_new = h.copy()
        indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        h_new[indices] += np.random.normal(0, noise_std, len(indices))
        h_new = np.clip(h_new, 0, 1)
        return normalize_h(h_new)
    
    def crossover(h1, h2):
        point = np.random.randint(1, n_points-1)
        h_new = np.concatenate([h1[:point], h2[point:]])
        return normalize_h(h_new)
    
    def tournament_selection(population, fitnesses, k=5):
        idxs = np.random.choice(len(population), size=k, replace=False)
        best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
        return population[best_idx]
    
    # GA parameters
    pop_size = 50
    max_gens = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    
    population = [normalize_h(np.random.rand(n_points)) for _ in range(pop_size)]
    if 'initial_h_values' in globals():
        population[0] = normalize_h(np.array(initial_h_values))
    
    for gen in range(max_gens):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best solution found
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_h = population[min_idx].copy()
            best_c5 = fitnesses[min_idx]
        
        # Create new population
        new_population = []
        for _ in range(pop_size // 2):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            if np.random.rand() < crossover_rate:
                child1 = crossover(parent1, parent2)
                child2 = crossover(parent2, parent1)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            child1 = mutate(child1) if np.random.rand() < mutation_rate else child1
            child2 = mutate(child2) if np.random.rand() < mutation_rate else child2
            
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
    
    # Resolution increase if needed
    if n_points == 35 and best_c5 > 0.3808 and time() - start_time < budget_s * 0.9:
        n_points = 40
        dx = 2.0 / n_points
        best_h = np.interp(np.linspace(0, 2, n_points), 
                          np.linspace(0, 2, len(best_h)), 
                          best_h)
        best_h = normalize_h(best_h)
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
        # Run GA at higher resolution
        pop_size = 20
        max_gens = 300
        population = [normalize_h(np.random.rand(n_points)) for _ in range(pop_size)]
        if 'initial_h_values' in globals():
            population[0] = normalize_h(np.array(initial_h_values))
        
        for gen in range(max_gens):
            if time() - start_time > budget_s * 0.95:
                break
                
            fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
            
            min_idx = np.argmin(fitnesses)
            if fitnesses[min_idx] < best_c5:
                best_h = population[min_idx].copy()
                best_c5 = fitnesses[min_idx]
            
            new_population = []
            for _ in range(pop_size // 2):
                parent1 = tournament_selection(population, fitnesses)
                parent2 = tournament_selection(population, fitnesses)
                
                if np.random.rand() < crossover_rate:
                    child1 = crossover(parent1, parent2)
                    child2 = crossover(parent2, parent1)
                else:
                    child1, child2 = parent1.copy(), parent2.copy()
                
                child1 = mutate(child1, noise_std=0.01) if np.random.rand() < mutation_rate else child1
                child2 = mutate(child2, noise_std=0.01) if np.random.rand() < mutation_rate else child2
                
                new_population.extend([child1, child2])
            
            population = new_population[:pop_size]
        
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Final local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 300, 'ftol': 1e-9})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
