import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Triangular construction
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution simulated annealing
    resolutions = [35, 50, 75, 100]
    current_res = 0
    
    while time() - start_time < budget_s * 0.95 and current_res < len(resolutions):
        res = resolutions[current_res]
        if res > n_points:
            # Interpolate to higher resolution
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            h_new = np.interp(x_new, x_old, best_h)
            n_points = res
            dx = 2.0 / n_points
        else:
            h_new = best_h
            n_points = res
            dx = 2.0 / n_points
            
        # Simulated Annealing optimization
        h_current = h_new.copy()
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        T = 0.1  # Initial temperature
        alpha = 0.99  # Cooling rate
        max_iter = int(800 * (res / 35))  # Adjust based on resolution
        
        for _ in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate neighbor by perturbing a fraction of points
            h_neighbor = h_current.copy()
            num_perturb = max(1, int(0.1 * n_points))
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for i in indices:
                # Perturb by small random amount
                delta = np.random.normal(0, 0.05)
                h_neighbor[i] += delta
            # Clip and rescale
            h_neighbor = np.clip(h_neighbor, 0, 1)
            h_neighbor = h_neighbor * (n_points/2) / np.sum(h_neighbor)
            
            c5_neighbor = evaluate_erdos_solution(h_neighbor, dx)
            
            # Accept or reject
            if c5_neighbor < c5_current:
                h_current = h_neighbor
                c5_current = c5_neighbor
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
            else:
                # Accept with probability based on temperature
                if np.random.rand() < math.exp(-(c5_neighbor - c5_current) / T):
                    h_current = h_neighbor
                    c5_current = c5_neighbor
            
            T = alpha * T  # Cool down
            
        current_res += 1
    
    # Final refinement with local optimization
    if time() - start_time < budget_s * 0.1:
        from scipy.optimize import minimize
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0

        # Local optimization with random restarts
        max_restarts = 10
        for _ in range(max_restarts):
            if time() - start_time > budget_s:
                break
            # Random perturbation of best solution
            h_restart = best_h + np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res_opt = minimize(objective, h_restart, method='SLSQP', 
                                bounds=bounds, constraints=cons_sum, 
                                options={'maxiter': 100, 'ftol': 1e-8})
                if res_opt.success:
                    h_refined = res_opt.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                continue

    return (list(best_h), best_c5, n_points)
