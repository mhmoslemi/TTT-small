import numpy as np
from scipy.signal import correlate
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    # Initialize
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Block-based optimization
    block_size = int(np.sqrt(n_points))
    if block_size < 1:
        block_size = 1
    M = n_points // block_size
    
    # Initialize with block structure
    h = np.zeros(n_points)
    on_blocks = (n_points / 2) // block_size
    if on_blocks * block_size > n_points / 2:
        on_blocks = on_blocks - 1
        
    # Set random blocks to 1
    block_indices = list(range(M))
    random.shuffle(block_indices)
    for i in range(int(on_blocks)):
        start_idx = block_indices[i] * block_size
        end_idx = start_idx + block_size
        h[start_idx:end_idx] = 1.0
    
    # Ensure sum constraint
    actual_sum = np.sum(h)
    if actual_sum != n_points / 2:
        diff = int((n_points / 2) - actual_sum)
        if diff > 0:  # Need to add more ones
            # Find blocks that can be turned on
            for i in range(M):
                if h[i * block_size] == 0:
                    h[i * block_size:(i + 1) * block_size] = 1.0
                    diff -= block_size
                    if diff <= 0:
                        break
        elif diff < 0:  # Need to remove some ones
            # Find blocks that can be turned off
            block_ones = [i for i in range(M) if h[i * block_size] == 1]
            random.shuffle(block_ones)
            count = 0
            for i in block_ones:
                if count < abs(diff):
                    h[i * block_size:(i + 1) * block_size] = 0.0
                    count += block_size
                else:
                    break
    
    # Start optimization
    iteration = 0
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        h_current = h.copy()
        
        # Evaluate current solution
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            if best_c5 < 0.3807:
                break
        
        # Small random perturbation on block configuration
        if random.random() < 0.8:
            # Try to slightly change number of on blocks
            try:
                if np.sum(h_current) > n_points / 2:
                    # Need to turn off fewer blocks - find one that's on and turn off
                    on_blocks = np.where(h_current == 1)[0]
                    if len(on_blocks) > 0:
                        indices_to_turn_off = random.sample(list(on_blocks), min(len(on_blocks), 2))
                        for idx in indices_to_turn_off:
                            h_current[idx] = 0.0
                else:
                    # Turn on more blocks
                    off_blocks = np.where(h_current == 0)[0]
                    if len(off_blocks) > 0:
                        indices_to_turn_on = random.sample(list(off_blocks), min(len(off_blocks), 2))
                        for idx in indices_to_turn_on:
                            h_current[idx] = 1.0
                
                if np.sum(h_current) == n_points / 2:
                    h = h_current
                else:
                    # Reconstruct proper arrangement
                    h = np.zeros(n_points)
                    actual_on_blocks = int(n_points / 2) // block_size
                    for i in range(actual_on_blocks):
                        start = i * block_size
                        end = start + block_size
                        h[start:end] = 1.0
            except:
                # Fall back to frustrated randomization
                h = np.zeros(n_points)
                new_on_blocks = int(n_points / 2) // block_size
                block_indices = list(range(M))
                random.shuffle(block_indices)
                for i in range(new_on_blocks):
                    start_idx = block_indices[i] * block_size
                    end_idx = start_idx + block_size
                    h[start_idx:end_idx] = 1.0
        else:
            # Random perturbation of existing result
            h_perturbed = h.copy()
            # randomly toggle some values
            for _ in range(2):
                pos = random.randint(0, n_points - 1)
                h_perturbed[pos] = 1.0 - h_perturbed[pos]
            # Fix summation
            current_sum = np.sum(h_perturbed)
            diff = current_sum - n_points / 2
            if abs(diff) > 0.01:
                # Adjust values to meet constraint
                indices = np.where(h_perturbed == 1)[0]
                if len(indices) > 0 and diff > 0:
                    to_turn_off = min(int(np.ceil(diff)), len(indices))
                    for _ in range(to_turn_off):
                        pos = random.choice(indices)
                        h_perturbed[pos] = 0.0
                        indices = np.where(h_perturbed == 1)[0]
            h = h_perturbed
    
    return (list(best_h), best_c5, n_points)
