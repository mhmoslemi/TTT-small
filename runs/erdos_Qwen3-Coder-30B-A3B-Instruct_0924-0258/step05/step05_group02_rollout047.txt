import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialization
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a simple construction that's known to work
        h_start = np.ones(n_points) * 0.5
        h_start[10:25] = 1.0  # More concentrated region
        h_start = np.clip(h_start, 0, 1)
        
    dx = 2.0 / n_points
    
    # Normalize to sum to 17.5
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Constraint for sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Constraint bounds
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    convergence_count = 0
    max_convergence = 50
    
    while time() - start_time < budget_s * 0.95:
        # Local optimization step
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Sample random perturbations to escape local minima
        h_trial = best_h.copy()
        noise = np.random.normal(0, 0.03, n_points)
        h_trial += noise
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        
        try:
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_trial, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-7}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    convergence_count = 0
                else:
                    convergence_count += 1
            else:
                convergence_count += 1
        except Exception as e:
            convergence_count += 1

        # Add extra diversity by perturbing
        if convergence_count > max_convergence:
            h_perturbed = best_h.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h_perturbed += noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            best_h = h_perturbed
            best_c5 = evaluate_erdos_solution(best_h, dx)
            convergence_count = 0

        # Check for early stopping
        if time() - start_time > budget_s * 0.9:
            break
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
