import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Sample shifts
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    
    # Use CVXPY for convex relaxation approach
    h_vars = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points / 2,
        C5 >= 0
    ]
    
    # Add overlap constraints for sampled shifts
    for k in shifts:
        # Compute indices for shift k
        if k == 0:
            corr_expr = h_vars @ (1 - h_vars)
        else:
            # Correlation-like expression
            k_indices = np.arange(int(np.floor(k * n_points / 2)), 
                                min(int(np.ceil((k + 1) * n_points / 2)), n_points))
            corr_expr = 0
            for i in range(n_points):
                if i + int(k * n_points / 2) < n_points:
                    corr_expr += h_vars[i] * (1 - h_vars[i + int(k * n_points / 2)])
                else:
                    break
        constraints.append(corr_expr <= C5)
    
    # Optimization problem
    objective = cp.Minimize(C5)
    prob = cp.Problem(objective, constraints)
    
    # Iterative refinement using cutting planes
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    max_iter = 50
    iter_count = 0
    improvement_threshold = 1e-6
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.9:
        iter_count += 1
        if iter_count > max_iter or iter_count - last_improvement > 50:
            break
            
        # Try to solve convex problem
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_solution = h_vars.value
                c5_solution = C5.value
                
                # Check if better
                if c5_solution < best_c5 - improvement_threshold:
                    best_h = h_solution.copy()
                    best_c5 = c5_solution
                    last_improvement = iter_count
                    
        except:
            pass
            
        # Add more shift constraints based on current best
        shifts_extended = []
        for i in range(n_points):
            j = (i + int(0.5 * n_points / 2)) % n_points
            k = (j - i) * 2.0 / n_points
            if k >= 0 and k < 2:
                shifts_extended.append(k)
        
        shifts_extended = sorted(set(shifts_extended))
        
        # Local refinement step with SLSQP
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_new = res.x
                c5_new = obj(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    last_improvement = iter_count
        except:
            pass
            
        # Early exit
        if best_c5 < 0.3807:
            break
    
    # Final polish
    def obj_final(x):
        return evaluate_erdos_solution(x, 2.0 / n_points)
    
    def constraint_sum_final(x):
        return np.sum(x) - n_points / 2.0
    
    cons_sum_final = {'type': 'eq', 'fun': constraint_sum_final}
    bounds_final = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj_final, best_h, method='SLSQP', 
                       bounds=bounds_final, constraints=cons_sum_final, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj_final(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
