import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary algorithm parameters
    pop_size = 50
    max_generations = 1000
    mutation_rate = 0.15
    crossover_rate = 0.8
    
    # Population initialization
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Add the initial solution to population
    population[0] = best_h.copy()
    
    gen = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time() - start_time < budget_s * 0.95 and gen < max_generations:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Selection: tournament
        new_population = []
        for _ in range(pop_size):
            # Tournament selection
            idx1, idx2 = np.random.choice(pop_size, 2, replace=False)
            if fitness[idx1] < fitness[idx2]:
                parent1 = population[idx1]
            else:
                parent1 = population[idx2]
                
            idx1, idx2 = np.random.choice(pop_size, 2, replace=False)
            if fitness[idx1] < fitness[idx2]:
                parent2 = population[idx1]
            else:
                parent2 = population[idx2]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Perturb some elements
                num_mutate = max(1, int(mutation_rate * n_points))
                indices = np.random.choice(n_points, num_mutate, replace=False)
                for i in indices:
                    child[i] += np.random.normal(0, 0.05)
                child = np.clip(child, 0, 1)
            
            # Rescale to maintain constraint
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        
        # Check for best solution
        current_fitness = [evaluate_erdos_solution(h, dx) for h in population]
        min_c5 = min(current_fitness)
        if min_c5 < best_c5:
            best_c5 = min_c5
            best_h = population[current_fitness.index(min_c5)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                # Reset with better solution
                best_h = h_start.copy()
                best_c5 = evaluate_erdos_solution(best_h, dx)
                no_improvement_count = 0
    
    # Final refinement
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
