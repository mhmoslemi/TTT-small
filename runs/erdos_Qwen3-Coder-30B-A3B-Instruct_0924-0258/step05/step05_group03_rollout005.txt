import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize initial solution
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Sample shifts for inequality constraints
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    dx = 2.0 / n_points
    
    # Convex relaxation approach with cutting planes
    h_vars = cp.Variable(n_points)
    c5_var = cp.Variable()
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points / 2,
        c5_var >= 0
    ]
    
    # Add initial constraints from shifts
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Cutting plane loop
    iterations = 0
    max_iterations = 50
    stagnation = 0
    last_best = best_c5
    
    while time() - start_time < budget_s * 0.95 and iterations < max_iterations:
        # Compute overlaps for all shifts
        overlaps = []
        for shift in shifts:
            # Compute overlap for this shift
            corr = np.correlate(best_h, 1 - np.roll(best_h, int(shift / dx)), mode='full')
            # Find the maximum overlap at this shift
            max_overlap = np.max(corr) * dx
            overlaps.append(max_overlap)
        
        max_overlap_idx = np.argmax(overlaps)
        max_shift = shifts[max_overlap_idx]
        
        # Add constraint for this shift
        # Starting from the full correlation (1d) to be more precise
        # Calculate the corresponding correlation at this shift
        rolled_h = np.roll(best_h, int(max_shift / dx))
        correlation = np.correlate(best_h, 1 - rolled_h, mode='full')
        max_correlation = np.max(correlation) * dx
        # Add constraint
        objective = c5_var
        # The constraint is that max_correlation <= c5_var
        constraints.append(c5_var >= max_correlation)
        
        # Solve the convex problem
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                new_c5 = c5_var.value
                new_h = h_vars.value
                # Check if we improved
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
                    stagnation = 0
                    last_best = best_c5
                else:
                    stagnation += 1
                # Remove the last constraint, as we already used it
                constraints.pop()
                # Add the actual constraint from this iteration
                constraints.append(c5_var >= max_correlation)
                if best_c5 < 0.3807:
                    break
            else:
                break
        except:
            break
        
        iterations += 1
        if stagnation > 20:
            break
        
    # Local refinement using SLSQP
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
            
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
