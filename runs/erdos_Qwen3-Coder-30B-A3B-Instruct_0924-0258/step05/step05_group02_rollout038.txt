import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a hybrid algorithm: alternating between local search and global refinement
    max_restarts = 30
    restart_count = 0
    early_stop = 0
    
    def refine_with_optimization(h_init):
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(
                objective, 
                h_init, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-7}
            )
            if res.success:
                return res.x
        except Exception:
            pass
        return h_init
    
    def discrete_search():
        # Try structured patterns
        candidates = []
        
        # Pattern 1: Two spikes
        for i in range(10, n_points-10):
            h = np.zeros(n_points)
            h[i] = 1.0
            h[(i + n_points//2) % n_points] = 1.0
            h = h * (n_points/2) / np.sum(h)
            candidates.append(h)
        
        # Pattern 2: Simple binary pattern
        h = np.zeros(n_points)
        for i in range(0, n_points, 2):
            h[i] = 1.0
        h = h * (n_points/2) / np.sum(h)
        candidates.append(h)
        
        # Pattern 3: Gaussian-like shape
        x = np.linspace(0, 2, n_points)
        mu = 1.0
        sigma = 0.2
        h = np.exp(-0.5 * ((x - mu) / sigma)**2)
        h = h * (n_points/2) / np.sum(h)
        candidates.append(h)
        
        for h_candidate in candidates:
            c5_val = evaluate_erdos_solution(h_candidate, dx)
            if c5_val < best_c5:
                return h_candidate, c5_val
        
        return best_h, best_c5
    
    while time() - start_time < budget_s * 0.95:
        # Local refinement
        h_refined = refine_with_optimization(best_h.copy())
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
            restart_count = 0
            early_stop = 0
        else:
            restart_count += 1
            early_stop += 1
        
        # Periodic random search
        if restart_count > 15:
            # Generate random feasible solutions
            h_random = np.random.rand(n_points)
            h_random = h_random * (n_points/2) / np.sum(h_random)
            h_random = np.clip(h_random, 0, 1)
            h_random = h_random * (n_points/2) / np.sum(h_random)
            
            c5_random = evaluate_erdos_solution(h_random, dx)
            if c5_random < best_c5:
                best_h = h_random
                best_c5 = c5_random
                restart_count = 0
                early_stop = 0
        
        # Structural search
        if early_stop > 10:
            h_struct, c5_struct = discrete_search()
            if c5_struct < best_c5:
                best_h = h_struct
                best_c5 = c5_struct
            early_stop = 0
            
        # Global restart
        if restart_count > max_restarts:
            h_restart = np.random.rand(n_points)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            h_restart = np.clip(h_restart, 0, 1)
            restart_count = 0
            
            # Try a more structured restart if we're running low on time
            if time() - start_time > budget_s * 0.8:
                h_struct, c5_struct = discrete_search()
                if c5_struct < best_c5:
                    best_h = h_struct
                    best_c5 = c5_struct
        
        if time() - start_time > budget_s * 0.95:
            break
    
    # Final validation and normalization
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
