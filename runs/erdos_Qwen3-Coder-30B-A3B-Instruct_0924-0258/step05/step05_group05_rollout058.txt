import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create smooth initial sequence
        h_start = np.linspace(0, 1, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    population_size = 30
    mutation_factor = 0.8
    crossover_rate = 0.8
    max_generations = 500
    
    # Initialize population
    population = [h_start.copy()]
    fitness = [best_c5]
    
    # Add perturbed versions
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.03, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    stagnation_count = 0
    last_improvement = 0
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        if sorted_fit[0] < best_c5:
            best_h = sorted_pop[0].copy()
            best_c5 = sorted_fit[0]
            last_improvement = generation
            if best_c5 < 0.3807:
                break
        
        # Check for stagnation
        if generation - last_improvement > 50:
            stagnation_count += 1
            # Increase mutation factor
            mutation_factor = min(1.5, mutation_factor + 0.1)
        else:
            stagnation_count = 0
        
        # Create new population
        new_pop = []
        k = len(sorted_pop) // 2
        
        # Elitism
        for i in range(k):
            new_pop.append(sorted_pop[i])
        
        # Differential evolution
        while len(new_pop) < population_size:
            # Select three different individuals
            a, b, c = np.random.choice(k, 3, replace=False)
            x1, x2, x3 = sorted_pop[a], sorted_pop[b], sorted_pop[c]
            
            # Mutation
            mutant = x1 + mutation_factor * (x2 - x3)
            
            # Crossover
            trial = np.where(np.random.random(n_points) < crossover_rate, mutant, sorted_pop[0])
            
            # Repair
            trial = np.clip(trial, 0, 1)
            if np.sum(trial) > 0:
                trial = trial * (n_points/2) / np.sum(trial)
            else:
                trial = np.ones(n_points) * 0.5
                
            new_pop.append(trial)
        
        # Evaluate new population
        population = new_pop
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Final local refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
