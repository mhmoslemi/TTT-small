import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm with Local Refinement
    pop_size = 50
    generations = 1000
    elitism = pop_size // 3
    mutation_rate = 0.1
    max_iter = int(budget_s * 0.9 / 0.1) if budget_s < 100 else 1000
    
    # Population initialization
    population = []
    population.append(best_h.copy())
    
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    def mutate(h):
        h_mut = h.copy()
        for i in range(n_points):
            if np.random.random() < mutation_rate:
                h_mut[i] += np.random.normal(0, 0.02)
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * (n_points/2) / np.sum(h_mut)
        return h_mut
    
    def crossover(h1, h2):
        alpha = np.random.uniform(0.4, 0.6)
        h_child = alpha * h1 + (1 - alpha) * h2
        h_child = h_child * (n_points/2) / np.sum(h_child)
        h_child = np.clip(h_child, 0, 1)
        return h_child
    
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    iter_count = 0
    
    while iter_count < max_iter and time() - start_time < budget_s * 0.95:
        # Evaluate fitness
        fitness_scores = [fitness(h) for h in population]
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores.sort()
        
        # Track best
        if fitness_scores[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness_scores[0]
        
        # Selection: elite
        elite = population[:elitism]
        
        # Create new population
        new_pop = elite[:]
        
        # Crossover + Mutation
        while len(new_pop) < pop_size:
            # Select two parents from elite
            idx1, idx2 = np.random.choice(elitism, 2, replace=False)
            h_child = crossover(elite[idx1], elite[idx2])
            h_child = mutate(h_child)
            new_pop.append(h_child)
        
        # Local refinement for top candidates
        for i in range(min(10, len(new_pop))):
            h_refined = new_pop[i].copy()
            try:
                # Create constraint
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(
                    lambda h: evaluate_erdos_solution(h, dx),
                    h_refined,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons_sum,
                    options={'maxiter': 50, 'ftol': 1e-8}
                )
                if res.success:
                    h_revised = res.x
                    c5_revised = evaluate_erdos_solution(h_revised, dx)
                    if c5_revised < best_c5:
                        best_h = h_revised
                        best_c5 = c5_revised
                        if iter_count % 50 == 0:
                            print(f"Improved C5: {best_c5}")
            except:
                pass
        
        population = new_pop
        iter_count += 1
    
    # Final validation
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    best_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), best_c5, n_points)
