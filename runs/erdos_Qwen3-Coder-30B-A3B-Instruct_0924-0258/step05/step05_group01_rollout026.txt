import numpy as np
from scipy.signal import convolve
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a triangular function
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale approach with gradient updates
    scales = [35, 50, 75]
    current_scale_idx = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale_idx < len(scales):
        scale = scales[current_scale_idx]
        if scale > n_points:
            # Upsample to higher resolution
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            h_current = np.interp(x_new, x_old, best_h)
            n_points = scale
        else:
            h_current = best_h.copy()
            n_points = scale
        
        dx = 2.0 / n_points
        
        # Gradient-based optimization with restarts
        max_iter = 100
        learning_rate = 0.01
        patience = 10
        patience_counter = 0
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute C5 and corresponding max overlap index
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            max_val = corr[max_idx]
            
            # Gradient estimate -- discrete approximation of derivative
            # The maximum overlap is at shift k = max_idx - (n_points - 1)
            k_shift = max_idx - (n_points - 1)
            
            # Compute gradient with respect to h for this specific k
            # Set up shifted version of 1-h to compute gradient approximation
            shifted_ones_minus_h = np.zeros_like(h_current)
            if k_shift >= 0:
                shifted_ones_minus_h[k_shift:] = 1 - h_current[:-k_shift]
            else:
                shifted_ones_minus_h[:k_shift] = 1 - h_current[-k_shift:]
            
            # Approximate gradient of correlation w.r.t. h
            grad_h = np.zeros_like(h_current)
            # For simplicity, use analytical result or finite difference of the same term
            if k_shift >= 0:
                grad_h = shifted_ones_minus_h[k_shift:] 
                # Apply finite difference for core of gradient
                grad_h = np.concatenate([grad_h, np.zeros(k_shift)])
            else:
                grad_h = shifted_ones_minus_h[:k_shift]
                grad_h = np.concatenate([np.zeros(-k_shift), grad_h])
            
            # Forward difference
            h_new = h_current - learning_rate * grad_h
            # Project to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                patience_counter = 0
                learning_rate = min(learning_rate * 1.1, 0.1)
            else:
                patience_counter += 1
                if patience_counter > patience:
                    learning_rate *= 0.5
                    patience_counter = 0
                # Random restart
                h_restart = h_current + np.random.normal(0, 0.01, n_points)
                h_restart = np.clip(h_restart, 0, 1)
                h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                c5_restart = evaluate_erdos_solution(h_restart, dx)
                if c5_restart < best_c5:
                    best_h = h_restart.copy()
                    best_c5 = c5_restart
                    patience_counter = 0
            
            h_current = h_new.copy()
            
        current_scale_idx += 1
    
    return (list(best_h), best_c5, n_points)
