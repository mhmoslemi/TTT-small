import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for EA
    pop_size = 50
    max_gens = 1000
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 3
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # If initial_h_values is available, include it in population
    if 'initial_h_values' in globals():
        population[0] = np.array(initial_h_values)
    
    gen = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95 and gen < max_gens:
        # Evaluate population
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Find best in current population
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_h = population[min_fit_idx].copy()
            best_c5 = fitness[min_fit_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Selection and reproduction
        new_population = []
        
        # Elitism - keep best individual
        new_population.append(population[min_fit_idx].copy())
        
        while len(new_population) < pop_size:
            # Tournament selection
            idx1 = np.random.choice(pop_size, tournament_size)
            idx2 = np.random.choice(pop_size, tournament_size)
            parent1 = population[np.argmin([fitness[i] for i in idx1])]
            parent2 = population[np.argmin([fitness[i] for i in idx2])]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.normal(0, 0.05)
                # Ensure feasibility
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                new_population.append(child)
                if len(new_population) >= pop_size:
                    break
        
        population = new_population[:pop_size]
        gen += 1
        
        # Multi-resolution strategy
        if gen % 100 == 0 and gen > 0 and n_points < 50:
            # Interpolate and refines to higher resolution
            n_points = min(50, n_points + 5)
            dx = 2.0 / n_points
            new_population = []
            for old_h in population:
                h_new = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(old_h)), old_h)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                new_population.append(h_new)
            population = new_population
        
        # Reset if no improvement for a while
        if no_improvement_count > 50:
            population = []
            for _ in range(pop_size):
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points/2) / np.sum(h)
                population.append(h)
            no_improvement_count = 0
            
            # Add parent if available
            if 'initial_h_values' in globals():
                population[0] = np.array(initial_h_values)
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
