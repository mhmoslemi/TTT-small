import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach
    # Start with a coarse resolution for global search
    n_coarse = min(20, n_points)
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.ones(n_coarse) * 0.5
    
    # Try solving with convex relaxation on coarse grid
    try:
        h_relaxed = cp.Variable(n_coarse)
        constraints = [
            cp.sum(h_relaxed) == n_coarse / 2.0,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        
        # Add constraints for first few shifts
        for k in range(1, min(10, n_coarse)):  
            if k < n_coarse // 2:
                corr_sum = cp.sum(cp.multiply(h_relaxed[:n_coarse-k], 1 - h_relaxed[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_coarse-k]))
            constraints.append(corr_sum <= best_c5 * dx_coarse + 1e-6)

        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            h_coarse = h_relaxed.value
    except Exception:
        pass

    # Interpolate to target resolution
    h_interpolated = np.interp(
        np.linspace(0, 2, n_points),
        np.linspace(0, 2, n_coarse),
        h_coarse
    )
    h_interpolated = np.clip(h_interpolated, 0, 1)
    h_interpolated = h_interpolated * (n_points/2) / np.sum(h_interpolated)
    
    # Refine with local optimization
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        res = minimize(
            lambda h: evaluate_erdos_solution(h, dx), 
            h_interpolated, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons_sum,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if res.success:
            h_final = res.x
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    # Further fine-tune on target resolution
    max_restarts = 50
    restart_count = 0
    early_stop = 0
    
    while time() - start_time < budget_s * 0.9 and restart_count < max_restarts:
        if restart_count % 5 == 0 and restart_count > 0:
            # Perturb solution
            perturbation = np.random.normal(0, 0.05, n_points)
            h_perturbed = best_h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        else:
            h_perturbed = best_h.copy()
        
        try:
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
                    early_stop = 0
                else:
                    restart_count += 1
                    early_stop += 1
            else:
                restart_count += 1
                early_stop += 1
        except Exception:
            restart_count += 1
            early_stop += 1

        if early_stop > 10:
            # Restart with random solution
            h_restart = np.random.rand(n_points)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            h_restart = np.clip(h_restart, 0, 1)
            h_final = h_restart
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
            early_stop = 0

    # Final check and normalization
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
