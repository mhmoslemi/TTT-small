import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Start with a finer resolution
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid resolution for convex relaxation
    n_low = 20
    dx_low = 2.0 / n_low
    
    # Initialize low-resolution step function
    h_low_init = np.ones(n_low) * 0.5
    h_low_init = h_low_init * (n_low / 2) / np.sum(h_low_init)
    
    # Try convex relaxation on coarse grid
    try:
        h_low = cp.Variable(n_low)
        constraints = [
            cp.sum(h_low) == n_low / 2.0,
            h_low >= 0,
            h_low <= 1
        ]
        
        # Add some overlap constraints for stability
        max_shift = min(10, n_low)
        for k in range(1, max_shift):
            if k < n_low // 2:
                corr_sum = cp.sum(cp.multiply(h_low[:n_low-k], 1 - h_low[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_low[k:], 1 - h_low[:n_low-k]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_low)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_low.value
            h_feasible = np.clip(h_feasible, 0, 1)
            h_feasible = h_feasible * (n_low / 2) / np.sum(h_feasible)
            
            # Interpolate to higher resolution
            if n_points > n_low:
                h_high = np.interp(np.linspace(0, 2, n_points), 
                                   np.linspace(0, 2, n_low), h_feasible)
                h_high = np.clip(h_high, 0, 1)
                h_high = h_high * (n_points/2) / np.sum(h_high)
            else:
                h_high = h_feasible[:n_points]
                h_high = h_high * (n_points/2) / np.sum(h_high)
                
            c5_interp = evaluate_erdos_solution(h_high, dx)
            if c5_interp < best_c5:
                best_h = h_high
                best_c5 = c5_interp
    except Exception:
        pass
    
    # Local refinement with SLSQP
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def obj_func(h):
        return evaluate_erdos_solution(h, dx)
    
    restart_count = 0
    max_restarts = 20
    penalty = 0.0
    
    while time() - start_time < budget_s * 0.9:
        h_perturbed = best_h.copy()
        if restart_count > 0:
            # Add noise for perturbation
            noise = np.random.normal(0, 0.05, n_points)
            h_perturbed += noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                obj_func,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-7}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
                else:
                    restart_count += 1
            else:
                restart_count += 1
        except Exception:
            restart_count += 1
        
        # Restart if stuck
        if restart_count > max_restarts:
            # Create new random starting point
            h_rand = np.random.rand(n_points)
            h_rand = h_rand * (n_points/2) / np.sum(h_rand)
            h_rand = np.clip(h_rand, 0, 1)
            restart_count = 0
            if time() - start_time > budget_s * 0.9:
                break

    # Final validation and adjust to exact constraint
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
