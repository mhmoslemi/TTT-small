import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Use initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Smooth approximation parameters
    tau = 1.0
    max_iter = 3000
    tol = 1e-7
    lbfgs_history = 10
    
    # Precomputed cross-correlations indices
    def compute_cross_correlations(h, tau):
        n = len(h)
        # compute (h * (1 - h_{.,k})) for all lags k
        result = np.zeros(2*n - 1)
        for i in range(n):
            for j in range(n):
                k = i + n - 1 - j  # lag k in correlation
                result[k] += h[i] * (1 - h[j])
        
        # Smooth max approximation
        smooth_max = np.log(np.sum(np.exp(tau * result))) / tau
        return smooth_max, result
    
    # Compute gradient of smooth approximation
    def compute_gradient(h, result, tau):
        n = len(h)
        grad = np.zeros(n)
        exp_term = np.exp(tau * result)
        Z = np.sum(exp_term)
        
        for i in range(n):
            d_i = 0.0
            for j in range(n):
                k = i + n - 1 - j
                d_i += (1 - h[j]) * exp_term[k]
            grad[i] = d_i / (Z * tau)
        return grad
    
    def objective_and_gradient(h, tau):
        # Normalize h
        h = h * (n_points / 2) / np.sum(h)
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Compute smooth approximation
        smooth_max, cross_corr = compute_cross_correlations(h, tau)
        # Gradient
        grad = compute_gradient(h, cross_corr, tau)
        # Adjust gradient to satisfy sum constraint
        grad_sum = np.sum(grad)
        grad = grad - grad_sum / n_points
        return smooth_max, grad

    # L-BFGS optimization
    h_opt = h_start.copy()
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
    
    # Main optimization loop
    last_improvement = 0
    stagnation_count = 0
    best_iter = 0
    
    for i in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Compute objective and gradient
        obj_val, grad = objective_and_gradient(h_opt, tau)
        
        # Apply constraint
        grad_sum = np.sum(grad)
        grad = grad - grad_sum / n_points
        
        # Update step
        step_size = 1.0
        
        # Simple line search
        h_new = h_opt - step_size * grad
        
        # Ensure bounds, project and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate this new point
        new_obj = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if new_obj < obj_val:
            h_opt = h_new.copy()
            c5_current = new_obj
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_opt.copy()
                last_improvement = i
                best_iter = i
        else:
            # Another try with modified h if no progress
            for _ in range(3):
                h_trial = h_opt - 0.1 * grad
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                c5_trial = evaluate_erdos_solution(h_trial, dx)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h_opt = h_trial.copy()
                    last_improvement = i
                    break
        
        # Update tau schedule
        if i % 100 == 0 and i > 0:
            tau *= 0.99
            
        # Early stopping if improvement is stale
        if i - last_improvement > 100:
            tau *= 1.1  # Increase smoothing for more exploration
            
    # Final refinement with exact function
    final_h = best_h.copy()
    try:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        def con(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': con}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, final_h, method='SLSQP', bounds=bounds, 
                       constraints=cons, options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            final_h = res.x
            final_c5 = obj(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
