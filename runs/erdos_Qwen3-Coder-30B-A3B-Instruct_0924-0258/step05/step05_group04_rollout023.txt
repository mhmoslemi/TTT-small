import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Lagrangian relaxation with iterative convex optimization
    max_iter = 500
    rho = 1e3
    lambda_init = 0.1
    mu_init = 1.0
    
    # Initialize Lagrange multipliers
    lambda_k = np.full(2 * n_points - 1, lambda_init)
    mu = mu_init
    
    # Precompute correlation matrices for all shifts
    A_matrices = []
    for k in range(1 - n_points, n_points):  # From -n+1 to n-1
        A = np.zeros((n_points, n_points))
        if k >= 0:
            A[:-k, k:] = 1  # shift to the right
        else:
            A[-k:, :k] = 1  # shift to the left
        A_matrices.append(A)
    
    # Compute reference correlations
    ref_corr = np.correlate(h_start, 1 - h_start, mode='full')
    ref_max = np.max(ref_corr) * dx
    
    for iter_num in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Convex subproblem using cvxpy to minimize approximated Lagrangian
        # L = C5 + sum(lambda_k * (h^T * A_k * (1-h) - C5*dx)) + mu * (sum(h) - n/2)
        # Easier to solve as quadratic program:
        # minimize (1/2)*h^T*Q*h + c^T*h
        # subject to: A_eq*h = b_eq
        #             lb <= h <= ub
        
        # Prepare quadratic problem elements
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(h_var))  # Dummy quadratic term
        
        # Build constraints
        constraints = [h_var >= 0, h_var <= 1]
        constraints.append(cp.sum(h_var) == n_points / 2.0)
        
        # These are not directly used in convex solver, but for update expressions
        h_np = np.array(h_start)
        
        # Perform convex optimization for this iteration
        # Using subproblem to compute approximate h
        try:
            # Initialize a simple problem
            problem = cp.Problem(objective, constraints)
            problem.solve(solver=OSQP, max_iter=1000, eps_abs=1e-6, eps_rel=1e-6)
            
            if problem.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Evaluate new function value
                current_c5 = evaluate_erdos_solution(h_new, dx)
                
                # Update solution if better
                if current_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = current_c5
                
                # Update h
                h_start = h_new.copy()
                
        except:
            # if convex solve fails, keep previous solution 
            pass
            
    # Final local refinement using SLSQP
    h_final = best_h.copy()
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Use local minimization to try to improve the current solution
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            h_final = res.x
            final_c5 = objective(h_final)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = h_final
    except:
        pass
    
    # Ensure all bounds and constraints are satisfied
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    assert np.isclose(np.sum(best_h), n_points / 2)
    
    return (list(best_h), best_c5, n_points)
