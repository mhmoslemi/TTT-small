import numpy as np
from scipy.optimize import minimize
from time import time
import hashlib

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def hash_solution(h):
    # Create a hashable representation of the solution
    sorted_h = tuple(sorted(h))
    return hashlib.md5(str(sorted_h).encode()).hexdigest()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_tenure = max(1, int(0.2 * n_points))
    tabu_list = []
    max_stagnation = 100
    stagnation_counter = 0
    mutation_scale = 0.04
    mutation_budget = max(1, int(0.3 * n_points))
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    max_iter = 1000
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor solution
        new_h = current_h.copy()
        indices = np.random.choice(n_points, mutation_budget, replace=False)
        for i in indices:
            new_h[i] += np.random.normal(0, mutation_scale)
        
        # Repair solution
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2.0) / np.sum(new_h)
        
        # Check tabu list
        new_hash = hash_solution(new_h)
        if new_hash in tabu_list:
            # Accept with low probability
            if np.random.random() > 0.1:
                continue
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Update best solution
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            stagnation_counter = 0
            if best_c5 < 0.3807:
                break
        else:
            stagnation_counter += 1
            
        # Update current solution
        current_h = new_h
        current_c5 = new_c5
        
        # Update tabu list
        tabu_list.append(new_hash)
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
            
        # Adjust mutation scale if stagnated
        if stagnation_counter > max_stagnation:
            mutation_scale = min(0.5, mutation_scale * 1.2)
            stagnation_counter = 0
            
        # Periodic local optimization
        if iteration % 10 == 0 and iteration > 0:
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(objective, current_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-9})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < current_c5:
                        current_h = h_refined
                        current_c5 = c5_refined
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                            stagnation_counter = 0
            except:
                pass
    
    # Final optimization if time permits
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
