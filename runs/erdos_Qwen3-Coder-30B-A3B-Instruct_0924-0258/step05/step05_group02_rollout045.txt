import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Ensure initial solution meets constraints
    sum_h = np.sum(best_h)
    best_h = best_h * (n_points / 2) / sum_h
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Refinement loop
    max_iter = 1000
    iter_count = 0
    no_improve_count = 0
    last_improve_time = time()
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all overlaps for current h
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        max_overlap = corr[max_k] * dx
        
        if max_overlap < best_c5 - 1e-8:
            best_c5 = max_overlap
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Add constraint for the maximum overlap shift
        k_max = max_k - (n_points - 1)
        if k_max < 0:
            k_max = -k_max

        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0

            # Check the constraint that triggers the current max overlap
            corr_current = np.correlate(h, 1 - h, mode='full')
            return best_c5 * dx - np.max(corr_current) * dx
            
        bounds = [(0, 1) for _ in range(n_points)]

                c5_new = evaluate_erdos_solution(h_new, dx)

            else:
