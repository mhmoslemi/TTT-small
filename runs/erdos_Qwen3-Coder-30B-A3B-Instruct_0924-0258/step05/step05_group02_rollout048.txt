import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial setup
    n_points = 35
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = h_current * (n_points/2) / np.sum(h_current)
    else:
        # Start with a symmetric step function
        h_current = np.ones(n_points) * 0.5
        
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Optimization parameters
    max_iter = 500
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    eps = 1e-6
    
    # Main optimization loop
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            # Perturb the solution
            noise = np.random.normal(0, 0.02, n_points)
            h_current = h_current + noise
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
            no_improvement_count = 0
            continue
            
        # Local optimization step
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        # Constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            # Use SLSQP local optimizer
            res = minimize(
                objective,
                h_current,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if res.success:
                h_current = res.x
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points/2) / np.sum(h_current)
            else:
                # Fallback to random perturbation
                noise = np.random.normal(0, 0.02, n_points)
                h_current = h_current + noise
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points/2) / np.sum(h_current)
                
        except Exception:
            # Fallback to random perturbation
            noise = np.random.normal(0, 0.02, n_points)
            h_current = h_current + noise
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
            
        iter_count += 1
        
        # Occasionally re-evaluate to ensure quality
        if iter_count % 20 == 0:
            test_c5 = evaluate_erdos_solution(h_current, dx)
            if test_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = test_c5
                
    # Final polishing
    try:
        # Try optimizing from best solution
        def final_objective(h):
            return evaluate_erdos_solution(h, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        res = minimize(
            final_objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            final_h = res.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points/2) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
        
    # Verify constraints
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
