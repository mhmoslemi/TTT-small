import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Sequential Convex Programming approach
    max_iter = 500
    mu = 1e-3
    outer_iter = 0
    
    # Precompute indices for overlap constraints
    all_shifts = list(range(n_points))
    
    while time() - start_time < max_time * 0.95 and outer_iter < max_iter:
        outer_iter += 1
        
        # Compute current overlaps
        overlaps = []
        for k in all_shifts:
            # Compute h(x) * (1 - h(x+k))
            if k == 0:
                overlap = h_start * (1 - h_start)
            else:
                h_shifted = np.roll(h_start, k)
                overlap = h_start * (1 - h_shifted)
            overlaps.append(np.sum(overlap) * dx)
        
        # Find maximum overlap
        max_overlap = max(overlaps)
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h_start.copy()
            
        if best_c5 < 0.3807:
            break
            
        # Formulate QCQP using CVXPY
        h_var = cp.Variable(n_points)
        
        # Objective: minimize max_overlap
        C5_var = cp.Variable()
        
        # Constraints: 
        # For each shift k: sum(h * (1 - h_shifted)) <= C5 
        constraints = []
        
        for k in all_shifts:
            if k == 0:
                overlap_expr = h_var * (1 - h_var)
            else:
                h_shifted = cp.Variable(n_points)
                # Set up the shifted constraint manually
                # The shifted version of h_var at index k is h_var[(k+i)%n_points]
                overlap_expr = cp.multiply(h_var, 1 - cp.roll(h_var, k))
            constraints.append(cp.sum(overlap_expr) <= C5_var)
        
        # Feasibility constraint: sum(h) = 1
        constraints.append(cp.sum(h_var) == n_points / 2.0)
        
        # Bounds
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # Add penalty term for constraint violations (augmented Lagrangian)
        # Just using simple penalization via quadratic penalty term for now
        
        # Objective function
        objective = cp.Minimize(C5_var)
        
        # Formulate and solve the problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.OSQP, verbose=False, max_iter=1000, eps_abs=1e-6, eps_rel=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                # Ensure values remain in [0,1]
                h_new = np.clip(h_new, 0, 1)
                # Renormalize to satisfy sum condition
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h_start = h_new
                mu *= 1.2  # Increase penalty parameter
            else:
                # If does not converge, fall back to gradient descent with constraints
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                def con(h):
                    return np.sum(h) - n_points / 2.0
                
                cons = {'type': 'eq', 'fun': con}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(obj, h_start, method='SLSQP', bounds=bounds, 
                               constraints=cons, options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_start = res.x
                else:
                    # Fallback to simpler optimization
                    h_start = np.clip(h_start + np.random.normal(0, 0.01, n_points), 0, 1)
                    h_start = h_start * (n_points / 2) / np.sum(h_start)
        except:
            h_start = np.clip(h_start + np.random.normal(0, 0.01, n_points), 0, 1)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Final refinement
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
