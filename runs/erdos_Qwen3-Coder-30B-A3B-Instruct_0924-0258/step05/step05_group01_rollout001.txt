import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Triangular initial guess
        x = np.linspace(0, 2, 35)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start * (35/2) / np.sum(h_start)
        n_points = 35
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Scale levels for multiresolution optimization
    scale_levels = [35, 50, 75, 100]
    current_scale_idx = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale_idx < len(scale_levels):
        scale = scale_levels[current_scale_idx]
        
        # Interpolate to target scale if needed
        if scale > n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            h_interp = np.interp(x_new, x_orig, best_h)
            n_points = scale
            dx = 2.0 / n_points
            h_current = h_interp
        else:
            h_current = best_h.copy()
            n_points = scale
            dx = 2.0 / n_points
        
        # Optimizer settings
        max_iter = 100
        learning_rate = 0.01
        patience = 10
        best_iter = 0
        best_loss = float('inf')
        
        for it in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute overlap and associated shift
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Compute gradient of overlap w.r.t h
            # The derivative is computed as the convolution of (1-h) with delta_k
            # where k is the optimal shift (max_idx - (n-1))
            shift_k = max_idx - (n_points - 1)  
            # Gradient approximation
            grad = np.zeros_like(h_current)
            if shift_k >= 0:
                grad[shift_k:] += (1 - h_current[:-shift_k]) if shift_k > 0 else (1 - h_current)
            else:
                grad[:shift_k] += (1 - h_current[-shift_k:]) if shift_k < 0 else (1 - h_current)
            
            # Gradient descent step
            h_new = h_current - learning_rate * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            loss_new = max_overlap
            if loss_new < best_loss:
                best_loss = loss_new
                best_iter = 0
                h_current = h_new.copy()
                if loss_new < best_c5:
                    best_h = h_current.copy()
                    best_c5 = loss_new
            else:
                best_iter += 1
                if best_iter >= patience:
                    break
                    
        current_scale_idx += 1
    
    return (list(best_h), best_c5, n_points)
