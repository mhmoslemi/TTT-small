import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize initial solution
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Parameters
    max_iter = 1000
    min_improvement = 1e-8
    stagnation_limit = 50
    shift_samples = 100
    
    # Initialize
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    # Search variables
    last_improvement = 0
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find worst shift k
        best_shift = 0
        max_overlap = -1
        
        # Approximate shift space
        shifts = np.linspace(0, 2, shift_samples)
        for k in shifts:
            # Circular convolution approximation with proper indexing
            corr_val = 0.0
            dx = 2.0 / n_points
            for i in range(n_points):
                j = (i + int(k * n_points / 2)) % n_points
                corr_val += h[i] * (1 - h[j]) * dx
            if corr_val > max_overlap:
                max_overlap = corr_val
                best_shift = k
        
        # Optimize h to minimize overlap for best_shift
        # Convert optimization into convex formulation
        h_var = Variable(n_points)
        
        # Constraints for h
        constraints = [h_var >= 0, h_var <= 1]
        constraints.append(sum(h_var) == n_points / 2)  # Integral constraint
        
        # Approximate the overlap for fixed shift
        objective_expr = 0
        k_idx = int(best_shift * n_points / 2)
        for i in range(n_points):
            j = (i + k_idx) % n_points
            objective_expr += h_var[i] * (1 - h_var[j])
        
        objective = Minimize(objective_expr)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, feastol=1e-8, abstol=1e-8, reltol=1e-8)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_new = np.array(h_var.value).flatten()
                # Project back into constraints
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            else:
                # If optimization fails, perturb slightly
                h = h + np.random.normal(0, 1e-4, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Retry with simple projection
            h = h + np.random.normal(0, 1e-4, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
        
        # Update best if improved
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            stagnation_count = 0
            last_improvement = iteration
        else:
            stagnation_count += 1
            
        # Early termination
        if best_c5 < 0.3807:
            break
            
        # Handle stagnation
        if stagnation_count > stagnation_limit:
            # Perturb solution and restart
            h = best_h + np.random.normal(0, 1e-3, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0
    
    # Final refinement with local optimization
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
