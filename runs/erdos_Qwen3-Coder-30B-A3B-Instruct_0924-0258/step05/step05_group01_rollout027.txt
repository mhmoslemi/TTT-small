import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale optimization with adaptive refinement
    scales = [35, 50, 75, 100]
    current_scale = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale < len(scales):
        scale = scales[current_scale]
        # Interpolate to target scale
        if scale > n_points:
            h_current = np.interp(np.linspace(0, 2, scale), np.linspace(0, 2, n_points), best_h)
            n_points = scale
        else:
            h_current = best_h
            n_points = scale
            
        dx = 2.0 / n_points
        
        # Gradient-based local optimization
        max_iter = 100
        lr = 0.01
        previous_loss = float('inf')
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute C5 and gradient
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            max_val = corr[max_idx]
            
            # Reconstruct gradient related to the maximum overlap shift
            gradient = np.zeros_like(h_current)
            shift = max_idx - (n_points - 1)
            for i in range(len(h_current)):
                if 0 <= i - shift < len(h_current):
                    gradient[i] += (1 - h_current[i - shift])
            
            # Update step
            h_new = h_current - lr * gradient
            
            # Apply constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update
            h_current = h_new
            
            new_loss = evaluate_erdos_solution(h_current, dx)
            
            if new_loss < best_c5:
                best_h = h_current.copy()
                best_c5 = new_loss
            
            # Adaptive learning rate
            if i > 0 and abs(new_loss - previous_loss) < 1e-8:
                lr *= 0.95
            previous_loss = new_loss
            
        current_scale += 1
    
    # Final validation
    if time() - start_time < budget_s * 0.1:
        try:
            h_final = best_h.copy()
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
