import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Read initial if available, otherwise initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize Lagrange multipliers
    lambda_vals = np.ones(2 * n_points - 1) * 0.1
    mu = 1.0
    penalty_weight = 1e3
    
    # Iteration parameters
    max_iter = 500
    tolerance = 1e-4
    
    # Precompute correlation matrices
    A_matrices = []
    for k in range(2 * n_points - 1):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j_start = max(0, k - n_points + 1)
            j_end = min(n_points, k + 1)
            if j_start <= i < j_end:
                A[i, j_start + (i - (k - n_points + 1))] = 1
        A_matrices.append(A)
    
    for iter_num in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Create convex subproblem
        h = Variable(n_points)
        
        # Objective: minimize max overlap + penalty terms
        current_c5 = evaluate_erdos_solution(best_h, dx)
        lagrangian = 0
        
        # Add lagrangian terms for overlap constraints
        for k, A in enumerate(A_matrices):
            lagrangian += lambda_vals[k] * (h.T @ A @ (1 - h) - current_c5 * dx)
        
        # Add integral constraint penalty
        lagrangian += mu * (sum(h) - n_points / 2)
        
        # Convex optimization problem
        constraints = [h >= 0, h <= 1, sum(h) == n_points / 2]
        prob = Problem(Minimize(lagrangian), constraints)
        
        try:
            prob.solve(solver=OSQP, warm_start=True, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_vals = h.value
            else:
                h_vals = best_h
        except:
            h_vals = best_h
            
        # Compute actual max overlap and update multipliers
        actual_c5 = evaluate_erdos_solution(h_vals, dx)
        prev_c5 = best_c5
        
        # Update Lagrange multipliers
        for k, A in enumerate(A_matrices):
            overlap = h_vals.T @ A @ (1 - h_vals)
            lambda_vals[k] = max(0, lambda_vals[k] + penalty_weight * (overlap - actual_c5 * dx))
            
        # Update mu for integral constraint
        integral_error = np.sum(h_vals) - n_points / 2
        mu = max(0, mu + penalty_weight * integral_error)
        
        # Project onto feasible set
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        
        # Check for improvement
        if actual_c5 < best_c5:
            best_c5 = actual_c5
            best_h = h_vals.copy()
        
        # Adaptive penalty adjustment
        if abs(integral_error) > tolerance:
            penalty_weight *= 2
        else:
            penalty_weight *= 0.5
            
        # Early stopping
        if best_c5 < 0.3807:
            break
    
    # Final local refinement with SLSQP
    remaining_time = max_time - (time() - start_time)
    
    if remaining_time > 30:
        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
