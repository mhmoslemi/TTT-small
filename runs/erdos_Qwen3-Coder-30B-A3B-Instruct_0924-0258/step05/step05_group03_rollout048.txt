import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Initialize best solution
    best_h = initial_h.copy()
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # CCP-style optimization
    max_iter = 100
    stagnation_count = 0
    last_best = best_c5
    shift_samples = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Inner loop: find worst shift k for current h
        max_overlap_val = -np.inf
        best_shift = 0
        
        # Sample shifts
        shifts = np.linspace(0, 2, shift_samples)
        for k in shifts:
            # Compute overlap for shift k
            corr = np.correlate(best_h, 1 - np.roll(best_h, int(k * n_points / 2)), mode='full')
            overlap = np.max(corr) * dx
            if overlap > max_overlap_val:
                max_overlap_val = overlap
                best_shift = k
        
        # Outer loop: optimize h to reduce overlap at k*
        # Create CVXPY variables
        h = Variable(n_points)
        
        # Constraint that h sums to n_points / 2
        constraints = [h >= 0, h <= 1, sum(h) == n_points / 2]
        
        # Define objective as overlap at best_shift
        # Note: Using approximate correlation to keep this convex
        # Create shift vector
        k_int = int(best_shift * n_points / 2)
        if k_int == 0:
            # No shift
            shift_h = best_h
        else:
            shift_h = np.roll(best_h, k_int)
        
        # Approximate objective using linear terms
        obj = 0
        for i in range(n_points):
            obj += h[i] * (1 - shift_h[i])
        
        # Minimize this (linear objective)
        prob = Problem(Minimize(obj), constraints)
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status == 'optimal':
                new_h = np.array(h.value).flatten()
                # Project back to valid range
                new_h = np.clip(new_h, 0, 1)
                # Re-normalize
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                # Evaluate new solution
                new_c5 = evaluate_erdos_solution(new_h, dx)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
                    stagnation_count = 0
                    last_best = best_c5
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
        except:
            stagnation_count += 1
            
        # Perturb if stuck
        if stagnation_count > 30:
            # Random small noise
            noise = np.random.normal(0, 0.01, n_points)
            perturbed_h = best_h + noise
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
            
            new_c5 = evaluate_erdos_solution(perturbed_h, dx)
            if new_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = new_c5
                
            stagnation_count = 0
            
        # Early termination if target achieved
        if best_c5 < 0.3807:
            break

    # Final local refinement using SLSQP
    def obj(x):
        return evaluate_erdos_solution(x, dx)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
