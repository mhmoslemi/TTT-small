from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use genetic algorithm with local refinement
    population_size = 50
    generations = 1000
    elitism = int(0.3 * population_size)
    mutation_rate = 0.1
    crossover_alpha_range = (0.4, 0.6)
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def create_initial_population():
        pop = []
        if 'initial_h_values' in globals():
            pop.append(np.array(initial_h_values))
        pop.append(h_start)
        while len(pop) < population_size:
            ind = create_individual()
            pop.append(ind)
        return pop
    
    population = create_initial_population()
    
    # Update best solution
    for ind in population:
        c5 = evaluate_erdos_solution(ind, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = ind.copy()
    
    # Genetic algorithm main loop
    generation = 0
    stagnation_count = 0
    max_stagnation = 100
    
    while time() - start_time < budget_s * 0.95 and generation < generations:
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = evaluate_erdos_solution(ind, dx)
            fitness.append((c5, ind))
        
        fitness.sort(key=lambda x: x[0])
        best_in_gen = fitness[0][0]
        if best_in_gen < best_c5:
            best_c5 = best_in_gen
            best_h = fitness[0][1].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        if stagnation_count > max_stagnation:
            # Restart with random solutions
            population = create_initial_population()
            stagnation_count = 0
        
        # Selection
        elite = [ind for _, ind in fitness[:elitism]]
        
        # Crossover and mutation to generate new population
        new_pop = elite.copy()
        
        while len(new_pop) < population_size:
            # Tournament selection for parents
            p1 = random.choice(elite)
            p2 = random.choice(elite)
            
            # Crossover
            alpha = random.uniform(*crossover_alpha_range)
            child = alpha * p1 + (1 - alpha) * p2
            
            # Mutate
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += random.gauss(0, 0.02)
            
            # Ensure constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)

        if generation % 10 == 0:

                # Use L-BFGS-B to refine
                bounds = [(0, 1) for _ in range(n_points)]

                        lambda x: evaluate_erdos_solution(x, dx),
                        ind,
                        method='L-BFGS-B',
                        bounds=bounds,
                        constraints=cons,
                        options={'maxiter': 50}
                    )
                    if res.success:
                        refined = res.x
                        refined_c5 = evaluate_erdos_solution(refined, dx)

    # Final check and return
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
