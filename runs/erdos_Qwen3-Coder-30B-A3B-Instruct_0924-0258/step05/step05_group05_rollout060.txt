import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a symmetric triangular initial solution
        mid = n_points // 2
        h_start = np.minimum(np.arange(n_points), 2*mid - np.arange(n_points))
        h_start = h_start / (mid)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Allow for some local refinement with different methods
    max_iter = 200
    stagnation_count = 0
    max_stagnation = 50
    
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # L-BFGS-B local optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0

        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'gtol': 1e-8})
            if res.success:
                h_candidate = res.x
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
        except:
            stagnation_count += 1
            
        # Random perturbation and normalization if stagnating or first attempt
        if stagnation_count > 0 or iter_count == 0:
            h_pert = best_h + np.random.normal(0, 0.01, n_points)  
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            
            c5_pert = evaluate_erdos_solution(h_pert, dx)
            if c5_pert < best_c5:
                best_h = h_pert
                best_c5 = c5_pert
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Gradient ascent/decay on a selected lag if improving
        if iter_count % 10 == 0 and iter_count > 0:
            # Just exploration: try different structure
            h_new = np.copy(best_h)
            # Slight smoothing in one direction
            h_new = np.convolve(h_new, [0.25, 0.5, 0.25], mode='same')
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Early termination
        if best_c5 < 0.3807:
            break

    # Final verification and cleanup
    best_h = np.clip(best_h, 0, 1)
    if np.sum(best_h) == 0:
        best_h = np.ones_like(best_h) * 0.5
    else:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
