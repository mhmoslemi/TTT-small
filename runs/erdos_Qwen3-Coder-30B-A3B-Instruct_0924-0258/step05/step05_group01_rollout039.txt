import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Evolutionary algorithm with multi-resolution refinement
    pop_size = 50
    max_generations = 1000
    mutation_rate = 0.15
    crossover_rate = 0.8
    tournament_size = 3
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evolutionary loop
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            # Reset with better initialization if no improvement
            population = []
            for _ in range(pop_size):
                h = best_h + np.random.normal(0, 0.03, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                population.append(h)
            no_improvement_count = 0
            
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % pop_size] if i + 1 < pop_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
                
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.03)
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.03)
            
            # Ensure constraints for children
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_population.extend([child1, child2])
            
        population = new_population[:pop_size]
        generation += 1
        
        # Multi-resolution refinement (upscale every 100 generations)
        if generation % 100 == 0 and n_points < 50:
            old_n = n_points
            n_points = min(n_points + 5, 50)
            dx = 2.0 / n_points
            
            # Interpolate and rescale for all individuals
            new_population = []
            for h in population:
                h_interpolated = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, old_n), h)
                h_interpolated = h_interpolated * (n_points/2) / np.sum(h_interpolated)
                new_population.append(h_interpolated)
            population = new_population
    
    # Final optimization at highest resolution if needed
    if time() - start_time < budget_s * 0.1:
        # Local refinement on best solution
        try:
            h_refined = best_h.copy()
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            
            # L-BFGS optimization for local refinement
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Random restarts for better convergence
            for _ in range(5): 
                h_restart = h_refined + np.random.normal(0, 0.01, n_points)
                h_restart = np.clip(h_restart, 0, 1)
                h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                
                try:
                    res_opt = minimize(objective, h_restart, method='L-BFGS-B', 
                                    bounds=bounds, constraints=cons_sum, 
                                    options={'maxiter': 100})
                    if res_opt.success:
                        h_refined_now = res_opt.x
                        c5_refined = objective(h_refined_now)
                        if c5_refined < best_c5:
                            best_h = h_refined_now
                            best_c5 = c5_refined
                except:
                    pass
                    
        except Exception:
            pass
    
    # Ensure final constraint satisfaction
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
