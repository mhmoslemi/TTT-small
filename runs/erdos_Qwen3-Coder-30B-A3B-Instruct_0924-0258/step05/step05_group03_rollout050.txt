import numpy as np
from cvxpy import *
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 35
        h_init = np.ones(n_points) * 0.5
    
    # Initialize h
    h = h_init.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    # Sampling points for shifts
    shift_samples = np.linspace(0, 2, 100)
    
    # Iterative cutting plane approach
    max_iter = 50
    prev_c5 = float('inf')
    early_stop = False
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
        
        # Evaluate current overlaps
        c5_values = []
        for k in shift_samples:
            k_idx = int(k * n_points / 2)
            if k_idx < n_points:
                corr_val = np.sum(h[:n_points-k_idx] * (1 - h[k_idx:]))
                c5_values.append(corr_val * 2.0 / n_points)
        
        # Find worst case shift
        max_overlap = np.max(c5_values)
        worst_idx = np.argmax(c5_values)
        best_shift = shift_samples[worst_idx]
        
        # Check termination
        if abs(prev_c5 - max_overlap) < 1e-5:
            break
        prev_c5 = max_overlap
        
        # Some tolerance check
        if max_overlap < 0.3807:
            early_stop = True
            break
        
        # Local refinement with SLSQP on best solution
        def obj(h_vec):
            return evaluate_erdos_solution(h_vec, 2.0 / n_points)
        
        def sum_constraint(h_vec):
            return np.sum(h_vec) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': sum_constraint}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 50, 'ftol': 1e-7})
            if res.success:
                h = res.x
        except:
            pass
        
        if early_stop:
            break
    
    # Final local refinement
    try:
        def obj(h_vec):
            return evaluate_erdos_solution(h_vec, 2.0 / n_points)
        
        def sum_constraint(h_vec):
            return np.sum(h_vec) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': sum_constraint}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            h = res.x
    except:
        pass
    
    final_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    return (list(h), final_c5, n_points)
