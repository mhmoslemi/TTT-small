import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        n_points = len(h_start)
    else:

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    max_generations = 1000
    
    # Initialize population
    def create_individual():
        h = np.random.random(n_points)
        h = h * (n_points/2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    if 'initial_h_values' in globals():
        population = [h_start.copy()]
        while len(population) < pop_size:
            individual = create_individual()
            population.append(individual)
    else:
        population = [create_individual() for _ in range(pop_size)]
    
    # Genetic Algorithm main loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
        
        # Selection
        def tournament_selection():
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            tournament.sort(key=lambda x: x[1])
            return tournament[0][0]
        
        # Crossover and mutation
        new_population = [population[min_fitness_idx].copy()]  # Elitism
        
        while len(new_population) < pop_size:
            if random.random() < crossover_rate:
                parent1 = tournament_selection()
                parent2 = tournament_selection()
                
                # Single-point crossover
                crossover_point = random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = tournament_selection()
            
            # Mutation
            if random.random() < mutation_rate:
                for i in range(n_points):
                    if random.random() < 0.1:
                        child[i] += np.random.normal(0, 0.02)
                
                # Clip values
                child = np.clip(child, 0, 1)
                
                # Rescale to meet integral constraint
                child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Resolution adaptation at 80% time
    if time() - start_time < budget_s * 0.8:
        n_points_new = 40
        if n_points_new > n_points:
            # Interpolate best solution
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_points_new)
            h_interp = np.interp(x_new, x_old, best_h)
            h_current = h_interp

        dx = 2.0 / n_points
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Local refinement with SLSQP
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, h_current, method='SLSQP',

                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Final refinement with SLSQP
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
