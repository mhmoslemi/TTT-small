import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Initialize with a triangular function
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start / np.sum(h_start) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution with Simulated Annealing
    scales = [35, 50, 75, 100]
    current_scale_idx = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale_idx < len(scales):
        scale = scales[current_scale_idx]
        if scale > n_points:
            # Interpolate to higher resolution
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            h_interp = np.interp(x_new, x_orig, best_h)
            h_current = h_interp
            n_points = scale
        else:
            h_current = best_h
            n_points = scale
            
        dx = 2.0 / n_points
        
        # Simulated Annealing parameters
        T0 = 0.1
        alpha = 0.99
        max_iter = int(1000 * (scale / 35))
        T = T0
        h_best = h_current.copy()
        c5_best = evaluate_erdos_solution(h_current, dx)
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb the current solution
            h_new = h_current.copy()
            # Flip some values
            flip_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            for idx in flip_indices:
                h_new[idx] = 1 - h_new[idx]
            
            # Ensure constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            if c5_new < c5_best or np.random.rand() < np.exp(-(c5_new - c5_best) / T):
                h_current = h_new
                c5_current = c5_new
                if c5_new < c5_best:
                    h_best = h_new
                    c5_best = c5_new
            
            # Cooling
            T = alpha * T
            
            # Update best
            if c5_best < best_c5:
                best_h = h_best.copy()
                best_c5 = c5_best
                
        current_scale_idx += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (list(best_h), best_c5, n_points)
