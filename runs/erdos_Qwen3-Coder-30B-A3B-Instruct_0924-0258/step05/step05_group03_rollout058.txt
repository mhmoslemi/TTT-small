import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 35
        h0 = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points/2
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Sample shifts from [0, 2] for constraints
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    
    # Optimization setup
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    # Iterative cutting plane approach using convex relaxation
    max_iterations = 50
    iteration = 0
    stagnant = 0
    old_c5 = best_c5 + 1.0
    
    while (time() - start_time < budget_s * 0.95 and 
           iteration < max_iterations and 
           stagnant < 50 and 
           best_c5 > 0.3807):
        
        # Evaluate current overlaps at sampled shifts
        overlaps = []
        for k in shifts:
            # Compute integral over all shifts
            corr = np.correlate(best_h, 1 - np.roll(best_h, int(k * n_points / 2)), mode='full')
            overlap = np.max(corr) * (2.0 / n_points)
            overlaps.append(overlap)
        
        # Find worst overlap
        max_overlap_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_overlap_idx]
        k_best = shifts[max_overlap_idx]
        
        # If not improving significantly, break early
        if abs(old_c5 - max_overlap) < 1e-6:
            stagnant += 1
        else:
            stagnant = 0
        old_c5 = max_overlap
        
        # Apply cutting plane - exact solution approach
        # Set up CVXPY problem
        h_vars = cp.Variable(n_points)
        c5_var = cp.Variable()
        
        # Objective
        objective = cp.Minimize(c5_var)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2,
            c5_var >= cp.maximum([cp.sum(cp.multiply(h_vars, cp.roll(1 - h_vars, int(k * n_points / 2))) * (2.0 / n_points)) 
                                  for k in shifts])
        ]
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                c5_new = c5_var.value
            else:
                h_new = best_h.copy()
                c5_new = best_c5
        except:
            h_new = best_h.copy()
            c5_new = best_c5
            
        # Update best solution if improved
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            stagnant = 0
            
        iteration += 1
    
    # Final polishing with local optimizer
    try:
        def obj(h):
            return evaluate_erdos_solution(h, 2.0 / n_points)

        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0

        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    # Ensure constraints are met
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
