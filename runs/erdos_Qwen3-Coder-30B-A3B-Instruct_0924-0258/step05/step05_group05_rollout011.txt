import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create symmetric triangular initial function
        mid = n_points // 2
        h_start = np.minimum(np.arange(n_points), 2*mid - np.arange(n_points))
        h_start = h_start / (mid)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    max_iter = 1000
    patience = 100
    stagnation_count = 0
    
    h_current = h_start.copy()
    
    # Main optimization loop
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convex relaxation to get a better starting point
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
                
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Try gradient-based optimization
            res = minimize(objective, h_current, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-8})
            
            if res.success:
                h_candidate = res.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
                candidate_c5 = objective(h_candidate)
                
                if candidate_c5 < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = candidate_c5
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                h_current = h_candidate
            else:
                # Fall back to random perturbation
                h_perturbed = h_current + np.random.normal(0, 0.01, n_points)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
                c5_perturbed = objective(h_perturbed)
                
                if c5_perturbed < best_c5:
                    best_h = h_perturbed.copy()
                    best_c5 = c5_perturbed
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                h_current = h_perturbed
                
        except:
            # Fallback
            h_perturbed = h_current + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            c5_perturbed = objective(h_perturbed)
            
            if c5_perturbed < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_perturbed
                stagnation_count = 0
            else:
                stagnation_count += 1
            h_current = h_perturbed
            
        # Early stopping condition
        if stagnation_count > patience:
            break
            
        # Occasionally perform a full optimization
        if i % 10 == 0:
            try:
                res = minimize(objective, h_current, method='L-BFGS-B', 
                               bounds=bounds, options={'maxiter': 100, 'ftol': 1e-9})
                if res.success:
                    h_candidate = res.x
                    h_candidate = np.clip(h_candidate, 0, 1)
                    h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
                    candidate_c5 = objective(h_candidate)
                    
                    if candidate_c5 < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = candidate_c5
                        stagnation_count = 0
                    h_current = h_candidate
            except:
                pass
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 100, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
