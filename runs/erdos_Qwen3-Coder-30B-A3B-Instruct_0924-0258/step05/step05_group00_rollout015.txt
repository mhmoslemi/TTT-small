import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization approach
    resolutions = [35, 40, 50, 60]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Cutting-plane optimization for improved convergence
        max_iterations = 50
        for iteration in range(max_iterations):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Compute current correlation to find worst case
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            max_k = max_idx - (len(h_current) - 1)
            
            # Add a constraint tightening around the worst overlap
            # Save the current best bound for this constraint
            current_bound = best_c5 * dx + 1e-10
            
            # Create optimization with an upper bound constraint
            try:
                # Define optimization variables
                h_opt = cp.Variable(n_points)
                
                # Constraints
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1,
                ]
                
                # Add the overlap constraint for the current shift k
                if max_k >= 0:
                    # This is alignment: h[i] * (1 - h[i + max_k])
                    if max_k < n_points:
                        overlap_expr = cp.sum(cp.multiply(h_opt[:n_points-max_k], 1 - h_opt[max_k:]))
                    else:
                        overlap_expr = 0  # No meaningful overlap for large k
                else:
                    # Reverse shift case
                    max_k = -max_k
                    if max_k < n_points:
                        overlap_expr = cp.sum(cp.multiply(h_opt[max_k:], 1 - h_opt[:n_points-max_k]))
                    else:
                        overlap_expr = 0
                
                constraints.append(overlap_expr <= current_bound)
                
                # Solve optimization problem
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_new = h_opt.value
                        c5_new = evaluate_erdos_solution(h_new, dx)
                        if c5_new < best_c5:
                            best_h = h_new
                            best_c5 = c5_new
                            h_current = h_new.copy()
                except:
                    pass
            except:
                pass
            
            # Occasionally use SLSQP for local refinement
            if iteration % 10 == 0:
                h_restart = h_current + np.random.normal(0, 0.01, n_points)
                h_restart = np.clip(h_restart, 0, 1)
                h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                try:
                    res = minimize(objective, h_restart, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-8})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                            h_current = h_refined.copy()
                except:
                    pass
        
        h_start = best_h.copy()
    
    # Final refinement using SLSQP
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
