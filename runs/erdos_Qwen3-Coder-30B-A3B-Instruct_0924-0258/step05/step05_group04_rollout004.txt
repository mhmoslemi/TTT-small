import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    max_iter = 5000
    mu = 1e-3
    tol = 1e-6
    rho = 1.0
    
    h = h_start.copy()
    h_prev = h.copy()
    z = np.zeros(2 * n_points - 1)
    lambda_ = np.zeros(2 * n_points - 1)
    
    # Initial constraints
    sum_constraint = n_points / 2.0
    bounds = [(0, 1) for _ in range(n_points)]
    
    iteration = 0
    converged = False
    
    def proj_simplex(v, z=1.0):
        """Project onto simplex: sum(v) = z"""
        v = np.array(v)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - z
        idx = np.where(u > cssv / np.arange(1, len(u) + 1))[0]
        if len(idx) == 0:
            return np.zeros_like(v)
        rho_idx = idx[-1]
        theta = cssv[rho_idx] / (rho_idx + 1)
        return np.maximum(v - theta, 0)
    
    # Precompute shifted versions
    ones = np.ones(n_points)
    
    while iteration < max_iter and (time() - start_time) < max_time * 0.95:
        iteration += 1
        
        # Update h by solving subproblem
        h_old = h.copy()
        
        # Compute -z + λ  
        outer_sum = np.zeros(2 * n_points - 1)
        for i in range(n_points):
            for j in range(n_points):
                if i + j < 2 * n_points - 1:
                    outer_sum[i + j] += (1 - h_old[j]) * h_old[i]
        
        # Current error term
        err = -z + lambda_ - mu * outer_sum
        
        # Quadratic subproblem for h
        def obj(h_vec):
            h_vec = np.array(h_vec)
            # Compute overlap constraint contributions
            h1 = np.concatenate([h_vec, np.zeros(n_points - 1)])
            h2 = np.concatenate([np.zeros(n_points - 1), h_vec])
            excl1 = np.concatenate([h_vec, np.zeros(n_points - 1)])
            excl2 = np.concatenate([np.zeros(n_points - 1), 1 - h_vec])
            corr = np.correlate(excl1, excl2, mode='full')
            corr_cut = corr[n_points - 1:] * dx
            penalty_term = np.sum((corr_cut - err) ** 2) / 2
            # Add regularization for simplicity
            reg = np.sum((h_vec - h_old) ** 2) / 2
            return penalty_term + reg / (1e-6 + mu)
        
        # Using SLSQP to solve for h
        def constraint_sum(h_vec):
            return np.sum(h_vec) - sum_constraint
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        try:
            res = minimize(obj, h_old, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                h = res.x
            else:
                h = h_old
        except:
            h = h_old

        # Project h onto simplex sum = 1  
        h = proj_simplex(h, sum_constraint)
        
        # Update auxiliary variable z = prox_z (h - lambda/mu)
        h_shifted_ones = np.concatenate([h, np.zeros(n_points - 1)])
        ones_shifted = np.concatenate([np.zeros(n_points - 1), ones])
        overlap_eval = np.correlate(h_shifted_ones, ones_shifted, mode='full')
        overlap_cut = overlap_eval[n_points - 1:] * dx
        
        # Proximal update for z
        z_new = overlap_cut - lambda_ / mu
        
        # Update z to lie in [0, 1] (project onto [0,1]^N)
        z_new = np.clip(z_new, 0, 1)
        
        z = z_new.copy()
        
        # Update dual variable
        lambda_ += mu * (overlap_cut - z)
        
        # Check for convergence using duality gap
        primal_residual = np.sum((h - h_old)**2) ** 0.5
        dual_residual = np.sum((z - overlap_cut)**2) ** 0.5
        if primal_residual < tol and dual_residual < tol:
            converged = True
            break
            
        # Periodic convergence check
        if iteration % 100 == 0:
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            if current_c5 < 0.3807:
                break
    
    # Final optimization with local refinement
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 20:
        h_init = best_h.copy()
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - sum_constraint
        
        cons = {'type': 'eq', 'fun': con_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, h_init, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = obj(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
