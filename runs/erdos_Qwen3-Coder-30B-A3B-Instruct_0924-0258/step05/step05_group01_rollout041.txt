import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary tabu search
    # Initialize binary representation
    h_binary = np.round(best_h).astype(int)
    # Ensure exactly n_points/2 ones
    current_ones = np.sum(h_binary)
    target_ones = n_points // 2
    if current_ones > target_ones:
        # Zero out some ones
        zero_indices = np.where(h_binary == 1)[0]
        np.random.shuffle(zero_indices)
        for idx in zero_indices[:current_ones - target_ones]:
            h_binary[idx] = 0
    elif current_ones < target_ones:
        # Set some zeros to ones
        one_indices = np.where(h_binary == 0)[0]
        np.random.shuffle(one_indices)
        for idx in one_indices[:target_ones - current_ones]:
            h_binary[idx] = 1
    
    best_binary = h_binary.copy()
    best_c5_binary = evaluate_erdos_solution(best_binary, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 10
    max_no_improvement = 100
    no_improvement_count = 0
    shaking_frequency = 50
    
    def get_neighbors(h):
        """Generate neighbors by flipping 1 bit to 0 and 0 bit to 1"""
        neighbors = []
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            for i in ones:
                for j in zeros:
                    neighbor = h.copy()
                    neighbor[i] = 0
                    neighbor[j] = 1
                    neighbors.append(neighbor)
        return neighbors
    
    def apply_shaking(h, ratio=0.05):
        """Random restart with some bits changed"""
        num_bits = max(1, int(ratio * len(h)))
        h_new = h.copy()
        indices = np.random.permutation(len(h))[:num_bits]
        for idx in indices:
            h_new[idx] = 1 - h_new[idx]
        # Balance the count again
        current_ones = np.sum(h_new)
        target_ones = n_points // 2
        if current_ones > target_ones:
            zero_indices = np.where(h_new == 1)[0]
            np.random.shuffle(zero_indices)
            for idx in zero_indices[:current_ones - target_ones]:
                h_new[idx] = 0
        elif current_ones < target_ones:
            one_indices = np.where(h_new == 0)[0]
            np.random.shuffle(one_indices)
            for idx in one_indices[:target_ones - current_ones]:
                h_new[idx] = 1
        return h_new
    
    current_h = best_binary.copy()
    current_c5 = best_c5_binary
    
    iteration = 0
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        
        neighbors = get_neighbors(current_h)
        if not neighbors:
            break
        
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for neighbor in neighbors:
            # Check if neighbor is tabu
            move = tuple(sorted([np.where(current_h == 1)[0][np.where(current_h == 1)[0] == np.where(neighbor == 1)[0][0]][0] if np.any(current_h == 1) and np.any(neighbor == 1) else 0,
                                np.where(current_h == 0)[0][np.where(current_h == 0)[0] == np.where(neighbor == 0)[0][0]][0] if np.any(current_h == 0) and np.any(neighbor == 0) else 1]))
            if move in tabu_list and best_neighbor_c5 >= current_c5:
                # Aspiration: if it's better than current, override tabu
                pass
            elif move in tabu_list and best_neighbor_c5 < current_c5:
                continue  # Skip tabu moves if not better
            
            c5 = evaluate_erdos_solution(neighbor, dx)
            if c5 < best_neighbor_c5:
                best_neighbor = neighbor
                best_neighbor_c5 = c5
                
        if best_neighbor is not None:
            current_h = best_neighbor
            current_c5 = best_neighbor_c5
            
            # Update tabu list
            tabu_list.append(move)
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
            
            if current_c5 < best_c5_binary:
                best_c5_binary = current_c5
                best_binary = current_h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Shaking
        if no_improvement_count >= shaking_frequency:
            current_h = apply_shaking(current_h)
            current_c5 = evaluate_erdos_solution(current_h, dx)
            no_improvement_count = 0
            tabu_list.clear()
            
        # Check for stopping conditions
        if no_improvement_count >= max_no_improvement:
            break
            
    # Rescale to original continuous representation (if needed) for final evaluation
    h_final = best_binary.astype(float)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), c5_final, n_points)
