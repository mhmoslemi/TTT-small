import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Start with a simple hill climbing approach with restarts
    max_restarts = 50
    max_iter = 500
    current_h = best_h.copy()
    
    # Use a simpler restart mechanism
    for restart in range(max_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation
        h_restart = current_h + np.random.normal(0, 0.02, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        # Local optimization using L-BFGS-B
        def objective(h):
            return evaluate_erdos_solution(h, dx)

            return np.sum(h) - n_points / 2.0
        
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res_opt = minimize(objective, h_restart, method='L-BFGS-B',
                            bounds=bounds, constraints=cons_sum,
                            options={'maxiter': 100})
            if res_opt.success:
                h_refined = res_opt.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
            
        # Reinitialize with new random sample if no improvement
        current_h = np.random.rand(n_points)
        current_h = current_h * (n_points/2) / np.sum(current_h)

    if time() - start_time < budget_s * 0.1:
        # Light local search at final resolution
        h_final = best_h.copy()
        try:
            bounds = [(0, 1) for _ in range(n_points)]

            res_opt = minimize(objective, h_final, method='L-BFGS-B',

                            options={'maxiter': 200})
            if res_opt.success:
                h_refined = res_opt.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
