import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h from initial_h_values if available, otherwise create a symmetric construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    # Normalize h to satisfy constraint sum(h) * dx = 1 (i.e., sum(h) = n_points / 2)
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    # Adaptive parameters that we'll use for the optimization process
    max_iter = 500
    min_improvement = 1e-6
    stagnation_limit = 50
    shift_samples = 50  # number of shift samples when finding max overlap
    perturbation_strength = 0.02
    
    improvement_count = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Step 1: Find the shift k that maximizes the overlap for current h
        max_overlap = -float('inf')
        best_k = 0
        
        # Evaluate overlap over a range of shift values
        shift_range = np.linspace(0, 2, shift_samples)
        for k in shift_range:
            # Compute overlap for this shift k
            overlap_sum = 0
            for i in range(n_points):
                j = i + int(k * n_points / 2)
                if j < n_points:
                    overlap_sum += h[i] * (1 - h[j])
                else:
                    # Handle wraparound or just don't include
                    break
                    
            if overlap_sum > max_overlap:
                max_overlap = overlap_sum
                best_k = k
        
        # Convert max_overlap to actual C5 value (account for dx)
        current_c5 = max_overlap * dx
        
        # Update the best solution found
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            improvement_count = 0
        else:
            improvement_count += 1
            
        # Count stagnation
        if improvement_count > stagnation_limit:
            # Perturb h slightly to escape local minima
            noise = np.random.normal(0, perturbation_strength, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            improvement_count = 0
            continue
            
        # Step 2: Optimize h to reduce the overlap at k = best_k using a convex problem
        # Use CVXPY to model this as a convex optimization
        h_var = cp.Variable(n_points)
        constraints = [h_var[i] >= 0 for i in range(n_points)]
        constraints += [h_var[i] <= 1 for i in range(n_points)]
        # Constraint on sum
        constraints += [cp.sum(h_var) == n_points / 2]
        
        # Try to use an existing method as the solver handles the sure need
        # We are solving a simplification of the problem, 
        # Goal: minimize (h * (1 - shift_k(h))) which is linear in h for fixed shift_k
        # Therfore, we can minimize the overlap function using linear constraints
        def build_overlap_expression(shift_idx):
            # Compute the expected term for this shift (fixed)
            # This only works approximately due to discrete nature of h
            # For best_k (use nearest index), compute specific overlap
            # We're going to do direct linear approximation for gradient
            # Simplify: the function to minimize is quadratic in h
            return cp.sum(cp.multiply(h_var, 1 - h_var))  # this is roughtly area
            
        # An easier approach: use a simple gradient-based step, not full SDP logic
        # Implement a simple Newton-like update to project h to reduce objective
        def compute_gradient(h_vec, shift):
            grad = np.zeros(n_points)
            # Compute directional gradient for the specific overlap term
            # For any shift k, derivative w.r.t to h[i] is (1 - h[i + shift])
            # Since shift is in continuous domain, we linearize the operation
            for i in range(n_points):
                j = i + int(shift * n_points / 2) 
                if j < n_points:
                    grad[i] = 1 - h_vec[j]
            return grad

        # Gradients for best shift
        grad = compute_gradient(h, best_k)
        # Take step
        step_size = 0.01
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Early stopping
        if best_c5 < 0.3807:
            break
            
    # Final refinement with scipy minimize, only using one iteration if time runs out
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-9})
        if res.success:
            final_h = res.x
            final_c5 = obj(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h
    except:
        pass

    return (list(best_h), best_c5, n_points)
