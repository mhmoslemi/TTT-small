import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Initialize
    h = initial_h.copy()
    h = h * (n_points / 2) / np.sum(h)  # Normalize to sum to 1
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Convex-concave procedure
    max_iter = 500
    tolerance = 1e-6
    last_c5 = best_c5 + 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find worst shift k* (based on current h)
        shifts = np.linspace(0, 2, 100)
        max_correlations = []
        for k in shifts:
            # Calculate correlation at shift k
            k_int = int(k / dx)
            if k_int >= 0 and k_int < n_points:
                corr = np.sum(h * np.roll(1 - h, k_int))
                max_correlations.append(corr)
            else:
                max_correlations.append(0)
        
        k_star = shifts[np.argmax(max_correlations)]
        k_star_int = int(k_star / dx)
        
        # Construct convex optimization problem
        h_new = cp.Variable(n_points)
        constraints = [h_new >= 0, h_new <= 1, cp.sum(h_new) == n_points / 2]
        
        # Objective: minimize correlation at k_star
        if k_star_int >= 0 and k_star_int < n_points:
            # Shifted version of 1-h
            shifted_ones_minus_h = np.roll(1 - h, k_star_int)
            objective = cp.Minimize(cp.sum(cp.multiply(h_new, shifted_ones_minus_h)))
        else:
            # This shouldn't happen with integer shift but just in case
            objective = cp.Minimize(cp.sum(h_new))
        
        problem = cp.Problem(objective, constraints)
        
        # Solve
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            h_new_val = h_new.value
            if h_new_val is not None:
                h = h_new_val
                current_c5 = evaluate_erdos_solution(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        except:
            pass
            
        # Check convergence
        if abs(last_c5 - best_c5) < tolerance:
            break
        last_c5 = best_c5
    
    # Final polishing
    def obj(x):
        return evaluate_erdos_solution(x, dx)
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass

    return (list(best_h), best_c5, n_points)
