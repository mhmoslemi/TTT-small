import numpy as np
from scipy.signal import correlate
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Triangular function as starting point
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start / np.sum(h_start) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Scale levels
    scales = [35, 50, 75, 100]
    
    for scale in scales:
        if time() - start_time > budget_s * 0.95:
            break
            
        if scale > n_points:
            # Interpolate to higher resolution
            x_fine = np.linspace(0, 2, scale)
            x_coarse = np.linspace(0, 2, n_points)
            h_coarse = best_h
            h_fine = np.interp(x_fine, x_coarse, h_coarse)
            n_points = scale
            dx = 2.0 / n_points
        else:
            h_fine = best_h
            n_points = scale
            dx = 2.0 / n_points

        # Gradient-based optimization
        max_iter = 100
        learning_rate = 0.01
        prev_c5 = best_c5 + 1.0
        patience = 0
        max_patience = 10

        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute correlation to find max overlap
            corr = correlate(h_fine, 1 - h_fine, mode='full')
            max_idx = np.argmax(corr)
            max_corr = corr[max_idx]
            
            # Gradient w.r.t. h for this shift
            # Not exactly the gradient of max overlapping, but a reasonable approximation
            # We use a smoothed gradient approximation
            grad = np.zeros_like(h_fine)
            # For the peak correlation index, compute forward difference gradient
            k = max_idx - (len(h_fine) - 1)  # convert to shift
            shift_idx = max_idx
            for j in range(len(h_fine)):
                if j+k >= 0 and j+k < len(h_fine):
                    grad[j] += (1 - h_fine[j+k])
            # Apply gradient descent step with clipped values
            h_new = h_fine - learning_rate * grad
            h_new = np.clip(h_new, 0, 1)
            
            # Rescale to maintain sum constraint
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                patience = 0
                learning_rate = min(learning_rate * 1.05, 0.1)
            else:
                patience += 1
                if patience > max_patience:
                    break
                
            h_fine = h_new
            
    return (list(best_h), best_c5, n_points)
