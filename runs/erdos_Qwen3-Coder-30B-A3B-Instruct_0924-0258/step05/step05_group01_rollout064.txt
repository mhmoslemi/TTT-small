import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary algorithm parameters
    pop_size = 50
    max_gens = 1000
    mutation_rate = 0.15
    elite_size = 5
    
    # Population initialization
    population = []
    for _ in range(pop_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Main evolutionary loop
    gen = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time() - start_time < budget_s * 0.9 and gen < max_gens:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Early stopping
        if no_improvement_count >= max_no_improvement:
            # Reset with new random solutions
            for i in range(elite_size, pop_size):
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                population[i] = h
            no_improvement_count = 0
        
        # Selection - tournament
        new_pop = population[:elite_size]
        while len(new_pop) < pop_size:
            # Tournament selection
            tournament_size = 3
            tournament_indices = random.sample(range(pop_size), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            parent = population[winner_idx].copy()
            
            # Crossover with random parent
            if random.random() < 0.7:
                other_parent_idx = random.randint(0, pop_size - 1)
                other_parent = population[other_parent_idx]
                
                # One-point crossover
                crossover_point = random.randint(1, n_points - 1)
                child = np.concatenate([parent[:crossover_point], other_parent[crossover_point:]])
            else:
                child = parent
            
            # Mutation
            if random.random() < mutation_rate:
                mutation_indices = random.sample(range(n_points), max(1, int(n_points * mutation_rate)))
                for idx in mutation_indices:
                    child[idx] += np.random.normal(0, 0.03)
                child = np.clip(child, 0, 1)
            
            # Rescale to sum constraint
            child = child * (n_points/2) / np.sum(child)
            
            new_pop.append(child)
        
        population = new_pop
        gen += 1
    
    # Final optimization with local search
    if time() - start_time < budget_s * 0.1:
        h_final = best_h.copy()
        dx_final = 2.0 / n_points
        
        def objective(h):
            return evaluate_erdos_solution(h, dx_final)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        # Local optimization on best solution
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res_opt = minimize(objective, h_final, method='SLSQP', 
                            bounds=bounds, constraints=cons_sum, 
                            options={'maxiter': 200, 'ftol': 1e-8})
            if res_opt.success:
                h_refined = res_opt.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
