import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()

    shift_samples = 100
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    # Initialize
    h_current = h_init.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Cutting plane method
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample shifts
        shifts = np.linspace(0, 2, shift_samples, endpoint=False)
        
        # Compute overlaps for all shifts
        overlaps = []
        for k in shifts:
            # Compute overlap integral with shift k
            overlap = 0.0
            for i in range(n_points):
                j = min(i + int(k/dx), n_points-1)
                if j < n_points:
                    overlap += h_current[i] * (1 - h_current[j])
            overlaps.append(overlap * dx)
        
        # Find the maximum overlap shift
        max_idx = np.argmax(overlaps)
        max_shift = shifts[max_idx]
        max_overlap = overlaps[max_idx]
        
        # Add constraint to our model
        # We only optimize C5 (the upper bound) and don't recompute the full problem each time
        # but we can just track the maximum overlap
        
        # Update best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h_current.copy()
        
        if best_c5 < target_c5:
            break
            
        # Slightly try to improve by local search
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try to improve current solution
        try:
            res = minimize(obj, h_current, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 50, 'ftol': 1e-8})
            if res.success and res.fun < obj(h_current):
                h_current = res.x
        except:
            pass
            
        # Try a small perturbation to escape local minima
        if iteration % 10 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h_current = np.clip(h_current + noise, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Final refinement using SLSQP
    try:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
