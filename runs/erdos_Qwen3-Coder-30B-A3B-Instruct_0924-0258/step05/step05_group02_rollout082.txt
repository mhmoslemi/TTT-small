import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm with local refinement
    pop_size = 50
    generations = 1000
    elitism = int(0.3 * pop_size)
    mutation_rate = 0.1
    crossover_weight_range = (0.4, 0.6)
    
    # Initialize population
    population = []
    population.append(best_h.copy())
    
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    def mutate(h):
        h_mut = h.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                h_mut[i] += np.random.normal(0, 0.02)
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * (n_points/2) / np.sum(h_mut)
        return h_mut
    
    def crossover(h1, h2):
        alpha = random.uniform(*crossover_weight_range)
        h_child = alpha * h1 + (1 - alpha) * h2
        h_child = h_child * (n_points/2) / np.sum(h_child)
        h_child = np.clip(h_child, 0, 1)
        return h_child
    
    # GA iterations
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness_scores = [fitness(h) for h in population]
        ranked_idx = sorted(range(len(fitness_scores)), key=lambda i: fitness_scores[i])
        
        # Keep elite
        elite = [population[i] for i in ranked_idx[:elitism]]
        
        # Create new population
        new_pop = elite.copy()
        
        # Crossover
        while len(new_pop) < pop_size:
            parent1 = random.choice(elite)
            parent2 = random.choice(elite)
            child = crossover(parent1, parent2)
            new_pop.append(child)
        
        # Mutation
        for i in range(len(new_pop)):
            if random.random() < 0.3:  # Mutation rate
                new_pop[i] = mutate(new_pop[i])
        
        population = new_pop
        
        # Local refinement with gradient descent
        for i in range(min(10, len(population))):  # Refine top candidates
            h_refine = population[i].copy()
            try:
                cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
                bounds = [(0, 1) for _ in range(n_points)]
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                res = minimize(
                    objective,
                    h_refine,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons_sum,
                    options={'maxiter': 50, 'ftol': 1e-6}
                )
                
                if res.success:
                    h_opt = res.x
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt
                        best_c5 = c5_opt
                        
            except Exception:
                continue
                
        # Check best so far
        current_best = min([fitness(h) for h in population])
        if current_best < best_c5:
            idx = np.argmin([fitness(h) for h in population])
            best_h = population[idx].copy()
            best_c5 = current_best
    
    # Final evaluation with refinement
    try:
        final_h = best_h.copy()
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points/2) / np.sum(final_h)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
