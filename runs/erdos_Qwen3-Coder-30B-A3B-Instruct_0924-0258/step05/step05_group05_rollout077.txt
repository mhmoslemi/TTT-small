import numpy as np
from scipy.optimize import minimize
from time import time
import hashlib

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def hash_state(h):
    # Create a hash of the sorted values to detect repeated states
    sorted_h = np.sort(h)
    return hashlib.md5(sorted_h.tobytes()).hexdigest()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = min(50, max(10, n_points // 5))
    max_iter = 1000
    max_stagnation = 100
    mutation_scale = 0.04
    mutation_budget = max(1, int(0.3 * n_points))
    
    # Keep track of stagnation
    stagnation_counter = 0
    current_best = best_c5
    
    # Initial state and tabu list
    tabu_list = [hash_state(best_h)]
    
    # Main tabu search loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate new candidate
        h_candidate = best_h.copy()
        indices = np.random.choice(n_points, size=mutation_budget, replace=False)
        
        for i in indices:
            # Add Gaussian noise
            h_candidate[i] += np.random.normal(0, mutation_scale)
        
        # Clip to [0,1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Normalize to sum to n_points/2
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Check if this is better and not tabu
        candidate_hash = hash_state(h_candidate)
        
        if c5_candidate < best_c5:
            # Update best solution
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            current_best = c5_candidate
            stagnation_counter = 0
            
            # Add to tabu list
            if len(tabu_list) >= tabu_tenure:
                tabu_list.pop(0)
            tabu_list.append(candidate_hash)
            
            # Local refinement 
            if iter_num % 10 == 0 and iter_num > 0:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, best_h, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 200, 'ftol': 1e-9})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
                    
        else:
            # Accept even if worse if not in tabu list and not too stale
            if candidate_hash not in tabu_list:
                # Add to tabu list
                if len(tabu_list) >= tabu_tenure:
                    tabu_list.pop(0)
                tabu_list.append(candidate_hash)
            
            # Update stagnation
            stagnation_counter += 1
            
            # Increase mutation if stagnated
            if stagnation_counter >= max_stagnation:
                mutation_scale = min(0.2, mutation_scale * 1.2)
                stagnation_counter = 0
    
    # Final local optimization if time permits
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
