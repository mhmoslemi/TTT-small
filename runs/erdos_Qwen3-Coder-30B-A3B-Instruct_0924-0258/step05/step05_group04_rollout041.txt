import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # SCP parameters
    max_iter = 500
    mu = 1e-3
    C5_init = 0.4
    c5_tolerance = 1e-6
    batch_size = 5
    
    # Create overlap constraint matrices
    def get_overlap_matrix(n, k):
        # Get the overlap matrix for shift k
        if k == 0:
            return np.eye(n)
        elif k > 0:
            # Shift right by k
            mat = np.zeros((n, n))
            for i in range(n):
                mat[i, (i+k) % n] = 1
            return mat
        else:
            # Shift left by |k|
            mat = np.zeros((n, n))
            for i in range(n):
                mat[i, (i+k)%n] = 1
            return mat
    
    # Precompute all shift matrices
    shift_matrices = [get_overlap_matrix(n_points, k) for k in range(-n_points+1, n_points)]

    # Main SCP iteration
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        current_c5 = evaluate_erdos_solution(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
            
        if best_c5 < 0.3807:
            break
            
        # Formulate QCQP subproblem based on current iterate
        h_vars = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum_squares(h_vars) * 0)  # Placeholder objective
        
        constraints = []
        
        # Add constraint for each shift
        for k in range(-n_points + 1, n_points):
            if k == 0:
                continue
            # h * (1 - shift_k(h)) <= C5
            shift_mat = shift_matrices[k + n_points - 1]
            shifted_h = shift_mat @ h_vars
            overlap_constraint = cp.sum(cp.multiply(h_vars, 1 - shifted_h)) <= C5_init + mu
            
            # Make sure h in [0, 1]
            constraints.extend([h_vars >= 0, h_vars <= 1])
            constraints.append(overlap_constraint)
            
        # Mass conservation constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Define and solve problem
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.OSQP, max_iter=1000, eps_abs=1e-6, eps_rel=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                if h_new is not None:
                    # Some numerical clipping might be needed
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    current_eval = evaluate_erdos_solution(h_new, dx)
                    
                    if current_eval < best_c5:
                        best_c5 = current_eval
                        best_h = h_new.copy()
        except:
            pass
        
        # Update mu and C5_init for next iteration
        if np.abs(best_c5 - C5_init) < c5_tolerance:
            mu *= 2
        else:
            mu = max(1e-6, mu / 2)
            
        C5_init = best_c5 * 1.05  # Slack factor
        
    # Final SLSQP refinement
    remaining_time = max_time - (time() - start_time)
    
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
