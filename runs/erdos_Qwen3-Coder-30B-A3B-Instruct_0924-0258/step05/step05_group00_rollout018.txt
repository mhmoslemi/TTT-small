import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization approach
    resolutions = [35, 40, 50, 60]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Cutting-plane optimization
        max_iterations = 100
        iterations = 0
        
        while time() - start_time < budget_s * 0.95 and iterations < max_iterations:
            # Find current maximum overlap
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            k_max = np.argmax(corr)
            max_overlap = corr[k_max] * dx
            
            # Check if we're improving
            if max_overlap < best_c5:
                best_h = h_current.copy()
                best_c5 = max_overlap
            
            # Create optimization problem with current constraints
            try:
                # Variables
                h_opt = cp.Variable(n_points)
                
                # Constraints
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add specific overlap constraint
                k_shift = k_max - (n_points - 1)
                if k_shift < 0:
                    # Compute the specific overlap
                    overlap_expr = cp.sum(cp.multiply(h_opt[:n_points+k_shift], 
                                                      1 - h_opt[-k_shift:]))
                else:
                    overlap_expr = cp.sum(cp.multiply(h_opt[k_shift:], 
                                                      1 - h_opt[:n_points-k_shift]))
                
                # Add constraint for current max overlap
                constraints.append(overlap_expr <= max_overlap * dx + 1e-8)
                
                # Solve
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6)
                
                if prob.status == cp.OPTIMAL:
                    h_new = h_opt.value
                    # Check if we improved
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
                        h_current = h_new.copy()
                    else:
                        break
                else:
                    break
                    
            except Exception:
                break
                
            iterations += 1
        
        h_start = best_h.copy()
    
    # Final refinement with SLSQP
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        # SLSQP refinement
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
