import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Start with coarse grid
        n_points = 20
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Refinement loop
    max_iter = 1000
    iter_count = 0
    no_improvement_count = 0
    tolerance = 1e-6
    
    # Coarse grid optimization first
    def coarse_optimization(h_init):
        h = h_init.copy()
        # Convert to probability distribution, preserve exact sum as 17.5
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Try to minimize with constraints
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(
                obj,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-7}
            )
            if res.success:
                return np.clip(res.x, 0, 1), evaluate_erdos_solution(res.x, dx)
        except Exception:
            pass
        return h, obj(h)
    
    h_coarse, c5_coarse = coarse_optimization(h_start)
    if c5_coarse < best_c5:
        best_h = h_coarse
        best_c5 = c5_coarse

    # Upsample to target size (35) and re-optimize
    if n_points < 35:
        h_upsampled = np.interp(np.linspace(0, 2, 35), np.linspace(0, 2, n_points), best_h)
        h_upsampled = np.clip(h_upsampled, 0, 1)
        h_upsampled = h_upsampled * 17.5 / np.sum(h_upsampled)
        best_h = h_upsampled
        best_c5 = evaluate_erdos_solution(best_h, 2.0/35)

    # Adaptive refinement loop
    n_points = 35
    dx = 2.0 / n_points
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute current overlaps
        overlaps = []
        for k in range(1, n_points):
            overlap = np.sum(best_h[:-k] * (1 - best_h[k:])) * dx
            overlaps.append(overlap)
        
        max_k = np.argmax(overlaps)
        max_overlap = max(overlaps)
        
        if max_overlap <= best_c5 + tolerance:
            no_improvement_count += 1
            if no_improvement_count > 50:
                break
        else:
            no_improvement_count = 0
            
        # Add constraint that the maximum overlap must be less than current value minus epsilon
        epsilon = 0.00001
        h = best_h.copy()
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint(h):
            # Add the constraint that max overlap decreases
            return max_overlap - epsilon - np.sum(h[:-max_k-1] * (1 - h[max_k+1:])) * dx
        
        cons = {'type': 'ineq', 'fun': constraint}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(
                obj,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-7}
            )
            if res.success:
                h_new = np.clip(res.x, 0, 1)
                h_new = h_new * 17.5 / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    no_improvement_count = 0
        except Exception:
            pass
            
        # Perturb if no improvement
        if no_improvement_count > 10:
            h_perturbed = best_h + np.random.normal(0, 0.05, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * 17.5 / np.sum(h_perturbed)
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
            no_improvement_count = 0
            
        iter_count += 1
    
    # Final refinement
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * 17.5 / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_final
    
    return (list(best_h), best_c5, n_points)
