import numpy as np
from scipy.signal import correlate
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Block-based approach
    # Start with sqrt(n_points) blocks
    M = int(np.sqrt(n_points))
    if M < 2:
        M = 2
    block_size = n_points // M
    
    # Ensure block_size is integer and M*block_size == n_points
    if M * block_size != n_points:
        block_size = n_points // M
        M = n_points // block_size
    
    # Current configuration: which blocks are on (1) or off (0)
    # Initialize with a reasonable guess
    block_config = np.ones(M)
    active_blocks = M // 2
    block_config[:active_blocks] = 1
    block_config[active_blocks:] = 0
    np.random.shuffle(block_config)
    
    # Convert block configuration to actual h vector
    h = np.zeros(n_points)
    for i in range(M):
        start_idx = i * block_size
        end_idx = start_idx + block_size
        h[start_idx:end_idx] = block_config[i]
        
    # Scale to meet integral constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization loop
    max_iters = 500
    for iteration in range(max_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try perturbing block configuration
        new_block_config = block_config.copy()
        
        # Swap two random blocks
        indices = list(range(M))

        h_new = np.zeros(n_points)

            h_new[start_idx:end_idx] = new_block_config[i]

        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * target_sum / current_sum

        c5_new = evaluate_erdos_solution(h_new, dx)

            best_c5 = c5_new
            best_h = h_new.copy()
            block_config = new_block_config.copy()
            
            if best_c5 < 0.3807:
                break
        else:
            # Metropolis-like acceptance
            if random.random() < 0.1:
                # Accept worse solution with some probability
                block_config = new_block_config.copy()
                h = h_new.copy()
    
    # Final refinement using local search
    remaining_time = budget_s - (time() - start_time)
