import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Triangular initialization
        x = np.linspace(0, 2, 35)
        h_start = 1 - np.abs(x - 1)
        h_start = np.clip(h_start, 0, 1)
        n_points = 35
    
    # Normalize to sum to n_points / 2
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Scale levels (smaller first for fast exploration)
    scales = [35, 50, 75]
    
    for scale in scales:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to current scale if needed
        if scale > n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            h_temp = np.interp(x_new, x_orig, best_h)
            n_points = scale
        else:
            h_temp = best_h
        
        dx = 2.0 / n_points
        
        # Gradient-based optimization at this scale
        max_iter = 50
        lr = 0.02
        for _ in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Evaluate correlation to find max
            corr = np.correlate(h_temp, 1 - h_temp, mode='full')
            max_k = np.argmax(corr)
            max_val = corr[max_k]
            
            # Compute gradient w.r.t. h for this max
            # This is a simplified gradient for illustration
            # Actual gradient would require more complex derivative
            grad = np.zeros_like(h_temp)
            # For shift k, the correlation is sum(h[i] * (1 - h[i+k]))
            # Gradient is -h[i+k] for h[i] term
            if max_k < len(h_temp):
                for i in range(len(h_temp) - max_k):
                    grad[i] -= (1 - h_temp[i + max_k])
            else:
                k = max_k - len(h_temp) + 1
                for i in range(len(h_temp) - k):
                    grad[i + k] -= (1 - h_temp[i])
            
            # Update step with learning rate
            h_new = h_temp - lr * grad
            # Clipping and rescaling
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_c5 = evaluate_erdos_solution(h_new, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
            h_temp = h_new
            
        # Early stopping if no significant improvement
        if abs(best_c5 - evaluate_erdos_solution(h_temp, dx)) < 1e-6:
            break
    
    # Final refinement using direct optimization if time permits
    if time() - start_time < budget_s * 0.9:
        try:
            # Do a coarse optimization to fine tune
            h_final = best_h.copy()
            bounds = [(0, 1) for _ in range(n_points)]
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            def constraint_sum(h):
                return np.sum(h) - n_points/2.0
                
            # Use gradient-free approach to improve final solution
            try:
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                res_opt = minimize(obj, h_final, method='SLSQP', 
                                bounds=bounds, constraints=cons_sum, 
                                options={'maxiter': 100, 'ftol': 1e-8})
                if res_opt.success:
                    h_refined = res_opt.x
                    c5_refined = obj(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
