import numpy as np
from scipy.optimize import minimize
from time import time
import hashlib

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    max_iter = 1000
    max_stagnation = 100
    mutation_scale = 0.04
    tabu_tenure = min(50, n_points // 2)
    tabu_list = []
    stagnation_counter = 0
    
    # Initialize best solution
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Hash function for tabu list
    def hash_solution(h):
        sorted_h = np.sort(h)
        return hashlib.md5(sorted_h.tobytes()).hexdigest()
    
    # Main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate perturbed solution
        h_new = current_h.copy()
        n_perturb = int(0.3 * n_points)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, mutation_scale)
        
        # Repair solution
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Check tabu list
        h_hash = hash_solution(h_new)
        is_tabu = h_hash in tabu_list
        
        # Accept or reject based on improvement and tabu status
        if c5_new < current_c5 or not is_tabu:
            current_h = h_new
            current_c5 = c5_new
            
            # Add to tabu list
            if len(tabu_list) >= tabu_tenure:
                tabu_list.pop(0)
            tabu_list.append(h_hash)
            
            # Update best if improved
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                stagnation_counter = 0
                if best_c5 < 0.3807:
                    break
            else:
                stagnation_counter += 1
        else:
            stagnation_counter += 1
        
        # Adjust mutation if stagnating
        if stagnation_counter > max_stagnation:
            mutation_scale = min(0.5, mutation_scale * 1.2)
            stagnation_counter = 0
    
    # Local refinement if time permits
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30 and best_c5 > 0.3807:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
