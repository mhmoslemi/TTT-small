import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary algorithm parameters
    pop_size = 50
    max_gens = 1000
    mutation_rate = 0.15
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Main evolutionary loop
    gen = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time() - start_time < budget_s * 0.95 and gen < max_gens:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Find best in current population
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            # Reinitialize with better start
            h_start = np.ones(n_points) * 0.5
            for i in range(pop_size):
                if i == 0:
                    h = h_start + np.random.normal(0, 0.1, n_points)
                else:
                    h = np.random.random(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                population[i] = h
            no_improvement_count = 0

        # Selection (tournament)
        selected = []
        tournament_size = 3
        for _ in range(pop_size):
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0].copy())
        
        # Crossover and Mutation
        new_pop = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % pop_size] if i+1 < pop_size else selected[0]
            
            if random.random() < crossover_rate:
                # One-point crossover
                cx_point = random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:cx_point], parent2[cx_point:]])
                child2 = np.concatenate([parent2[:cx_point], parent1[cx_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if random.random() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                    child2[j] += np.random.normal(0, 0.05)
            
            # Clip and normalize
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_pop.extend([child1, child2])
            
        population = new_pop[:pop_size]
        gen += 1
    
    # Final validation
    try:
        final_h = best_h.copy()
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points/2) / np.sum(final_h)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    except:
        pass
    
    # Additional local refinement
    if time() - start_time < budget_s * 0.98:
        try:
            # Try Newton-based refinement on best solution
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            x0 = best_h.copy()
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res_opt = minimize(
                lambda h: evaluate_erdos_solution(h, dx),
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if res_opt.success:
                refined_h = res_opt.x
                refined_c5 = evaluate_erdos_solution(refined_h, dx)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
