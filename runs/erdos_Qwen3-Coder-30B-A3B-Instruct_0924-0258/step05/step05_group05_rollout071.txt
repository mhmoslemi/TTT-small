import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # SDP relaxation approach
    # Try a direct optimization approach
    max_iter = 1000
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try to improve with local optimization
        h_current = best_h.copy()
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_current, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_refined = project_h(res.x)
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    if best_c5 < 0.3807:
                        break
        except:
            pass
        
        # Perturb current solution and try again
        h_perturbed = h_current + np.random.normal(0, 0.02, n_points)
        h_perturbed = project_h(h_perturbed)
        c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
        if c5_perturbed < best_c5:
            best_h = h_perturbed
            best_c5 = c5_perturbed
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 60:
        try:
            res = minimize(evaluate_erdos_solution, best_h, args=(dx,), 
                           method='L-BFGS-B', bounds=[(0,1) for _ in range(n_points)], 
                           options={'maxiter': 200, 'ftol': 1e-10})
            if res.success:
                h_final = project_h(res.x)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
