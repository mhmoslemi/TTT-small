import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Check for initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid approach: first solve with lower resolution
    n_low = 20
    dx_low = 2.0 / n_low
    
    # Define coarse grid optimization
    def coarse_optimize():
        h_low = cp.Variable(n_low)
        constraints = [
            cp.sum(h_low) == n_low / 2.0,
            h_low >= 0,
            h_low <= 1
        ]
        
        # Add constraints for a subset of shifts (to make it tractable)
        for k in range(1, min(5, n_low)):
            corr_sum = cp.sum(cp.multiply(h_low[:n_low-k], 1 - h_low[k:]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_low)), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-7)
            if prob.status == cp.OPTIMAL:
                return h_low.value
        except Exception:
            pass
        return None

    # Interpolation from coarse to fine grid
    def upsample(h_low):
        h_high = np.interp(np.linspace(0, 2, n_points), 
                          np.linspace(0, 2, n_low), h_low)
        h_high = h_high * (n_points/2) / np.sum(h_high)
        return np.clip(h_high, 0, 1)

    # Mutable parameters
    lr = 0.01
    mutation_rate = 0.1
    mutation_strength = 0.05
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Try to solve coarse grid
    h_low = coarse_optimize()
    if h_low is not None:
        try:
            h_refined = upsample(h_low)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
        except Exception:
            pass

    # Fine grid optimization
    max_iterations = 1000
    iteration = 0
    
    # Gradient computation (numerical)
    def compute_gradient(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (np.max(corr_plus) - np.max(corr)) / 1e-5
        return grad * dx

    # Adaptive optimization for fine grid
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        try:
            # Refine current solution
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx),
                best_h.copy(),
                method='SLSQP',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 30, 'ftol': 1e-7}
            )
            
            if res.success:
                h_new = res.x
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1

            # Occasionally perturb the solution to escape local optima
            if np.random.rand() < mutation_rate or no_improvement_count > 50:
                h_mutation = best_h.copy()
                noise = np.random.randn(n_points) * mutation_strength
                h_mutation += noise
                h_mutation = np.clip(h_mutation, 0, 1)
                h_mutation = h_mutation * (n_points/2) / np.sum(h_mutation)
                c5_mutation = evaluate_erdos_solution(h_mutation, dx)
                if c5_mutation < best_c5:
                    best_h = h_mutation
                    best_c5 = c5_mutation
                    no_improvement_count = 0

            iteration += 1
            
            if no_improvement_count >= max_no_improvement:
                break
                
        except Exception:
            no_improvement_count += 1
            iteration += 1
            if no_improvement_count >= max_no_improvement:
                break

    # Final validation and normalization
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), c5_final, n_points)
