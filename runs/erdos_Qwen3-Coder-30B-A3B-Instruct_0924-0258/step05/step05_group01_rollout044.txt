import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary tabu search
    # Convert to binary
    def to_binary(h):
        h_bin = (h > 0.5).astype(int)
        # Adjust to meet sum constraint
        current_sum = np.sum(h_bin)
        target_sum = n_points // 2
        if current_sum > target_sum:
            # Need to flip some 1s to 0s
            ones_indices = np.where(h_bin == 1)[0]
            flip_indices = np.random.choice(ones_indices, current_sum - target_sum, replace=False)
            h_bin[flip_indices] = 0
        elif current_sum < target_sum:
            # Need to flip some 0s to 1s
            zeros_indices = np.where(h_bin == 0)[0]
            flip_indices = np.random.choice(zeros_indices, target_sum - current_sum, replace=False)
            h_bin[flip_indices] = 1
        return h_bin
    
    h_best = to_binary(best_h)
    c5_best = evaluate_erdos_solution(h_best, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 10
    max_no_improve = 100
    no_improve_count = 0
    max_shake = 50
    shake_count = 0
    
    # Neighborhood generation
    def get_neighbors(h):
        neighbors = []
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            for i in ones:
                for j in zeros:
                    h_new = h.copy()
                    h_new[i] = 0
                    h_new[j] = 1
                    neighbors.append(h_new)
        return neighbors
    
    # Start tabu search
    while time() - start_time < budget_s * 0.95:
        # Get neighbors
        neighbors = get_neighbors(h_best)
        
        if not neighbors:
            break
            
        # Evaluate neighbors
        c5_values = []
        for h_n in neighbors:
            c5_n = evaluate_erdos_solution(h_n, dx)
            c5_values.append(c5_n)
        
        # Get the best neighbor not in tabu list
        best_idx = np.argmin(c5_values)
        best_c5_neighbor = c5_values[best_idx]
        best_h_neighbor = neighbors[best_idx]
        
        # Check if it's better than current best
        if best_c5_neighbor < c5_best:
            # Accept improvement
            h_best = best_h_neighbor
            c5_best = best_c5_neighbor
            no_improve_count = 0
            shake_count = 0
        else:
            # Apply aspiration criterion if drastically better
            if best_c5_neighbor < c5_best - 0.001:
                h_best = best_h_neighbor
                c5_best = best_c5_neighbor
                no_improve_count = 0
                shake_count = 0
            else:
                no_improve_count += 1
                shake_count += 1
        
        # Add current best to tabu list
        current_flip = np.where(h_best != best_h_neighbor)[0]
        if len(current_flip) > 0:
            tabu_list.append(current_flip[0])
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
        
        # Shaking
        if shake_count >= max_shake:
            # Random restart - set random binary solution that meets sum constraint
            h_random = np.zeros(n_points)
            indices = np.random.choice(n_points, n_points//2, replace=False)
            h_random[indices] = 1
            h_best = h_random.astype(int)
            c5_best = evaluate_erdos_solution(h_best, dx)
            no_improve_count = 0
            shake_count = 0
            tabu_list = []
        
        # Early termination if no improvement for a while
        if no_improve_count > max_no_improve:
            break
    
    # Final check and return
    return (list(h_best.astype(float)), c5_best, n_points)
