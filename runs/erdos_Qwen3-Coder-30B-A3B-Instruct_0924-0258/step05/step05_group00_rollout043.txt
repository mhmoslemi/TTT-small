import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialization
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        n_points = len(h_current)
    else:
        n_points = 35
        h_current = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Multi-resolution approach
    resolutions = [35, 40, 50, 60]
    for res in resolutions:
        if time() - start_time > budget_s * 0.9:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_current)
            h_current = h_interp
        else:
            h_current = h_current[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Cutting-plane method
        max_iterations = 100
        for iteration in range(max_iterations):
            if time() - start_time > budget_s * 0.9:
                break
                
            # Compute overlap
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            max_k = max_idx - (len(h_current) - 1)
            
            # Add constraint for the max overlap shift
            if max_k >= 0:
                overlap_sum = np.sum(h_current[:-max_k] * (1 - h_current[max_k:]))
            else:
                overlap_sum = np.sum(h_current[-max_k:] * (1 - h_current[:max_k]))
            
            # Constraint tightening approach
            if overlap_sum > best_c5 * dx * 1.001:  # Allow small tolerance
                # Refine by trying local search
                h_trial = h_current + np.random.normal(0, 0.01, n_points)
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points/2) / np.sum(h_trial)
                
                try:
                    def objective(h):
                        return evaluate_erdos_solution(h, dx)
                    
                    def constraint_sum(h):
                        return np.sum(h) - n_points / 2.0
                    
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    bounds = [(0, 1) for _ in range(n_points)]
                    res = minimize(objective, h_trial, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-8})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
            else:
                break  # No significant improvement, exit early
                
        h_current = best_h.copy()  # Revert to best solution
    
    # Final refinement at highest resolution
    if time() - start_time < budget_s * 0.95:
        try:
            h_final = best_h.copy()
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
