import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Keep a diverse population
    pop_size = 50
    population = []
    for _ in range(pop_size):
        # Random initialization with constraint enforcement
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
    
    # Add the best current solution to the population
    population[0] = best_h.copy()
    
    generations = 1000
    no_improvement_count = 0
    max_no_improvement = 100
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate population
        fitness = [objective(h) for h in population]
        
        # Selection - tournament
        new_population = []
        for _ in range(pop_size):
            # Tournament selection
            idx1, idx2 = random.sample(range(pop_size), 2)
            if fitness[idx1] < fitness[idx2]:
                parent = population[idx1]
            else:
                parent = population[idx2]
            new_population.append(parent.copy())
        
        # Crossover and Mutation
        for i in range(0, pop_size, 2):
            if i+1 < pop_size:
                # One-point crossover
                crossover_point = random.randint(1, n_points-1)
                child1 = np.concatenate([new_population[i][:crossover_point], new_population[i+1][crossover_point:]])
                child2 = np.concatenate([new_population[i+1][:crossover_point], new_population[i][crossover_point:]])
            else:
                child1 = new_population[i].copy()
                child2 = new_population[i].copy()
            
            # Mutation: Gaussian noise
            mutation_rate = 0.15
            for j in range(n_points):
                if random.random() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.03)
                    child2[j] += np.random.normal(0, 0.03)
            
            # Clip and normalize
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            # Replace with children
            population[i] = child1
            if i+1 < pop_size:
                population[i+1] = child2
        
        # Update best
        current_fitness = [objective(h) for h in population]
        min_fitness_idx = np.argmin(current_fitness)
        if current_fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = current_fitness[min_fitness_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count > max_no_improvement:
            # Reset to parent initialization in case of stagnation
            if 'initial_h_values' in globals():
                h_reset = np.array(initial_h_values)
                population[0] = h_reset.copy()
                no_improvement_count = 0
                # Do another round of local search on current best
                h_restart = best_h + np.random.normal(0, 0.02, n_points)
                h_restart = np.clip(h_restart, 0, 1)
                h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                try:
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    bounds = [(0, 1) for _ in range(n_points)]
                    res_opt = minimize(objective, h_restart, method='SLSQP', 
                                    bounds=bounds, constraints=cons_sum, 
                                    options={'maxiter': 100, 'ftol': 1e-8})
                    if res_opt.success:
                        h_refined = res_opt.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
    
    # Final refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    # Final validation 
    try:
        final_validation = evaluate_erdos_solution(best_h, dx)
        if final_validation < best_c5:
            best_c5 = final_validation
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
