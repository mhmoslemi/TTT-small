import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Convert to binary using rounding with adjustment
    h_bin = np.round(h_start).astype(int)
    current_sum = np.sum(h_bin)
    target_sum = n_points // 2
    
    # Adjust to ensure exactly target_sum ones
    if current_sum > target_sum:
        # Turn some 1s to 0s
        ones_indices = np.where(h_bin == 1)[0]
        to_flip = current_sum - target_sum
        flip_indices = np.random.choice(ones_indices, to_flip, replace=False)
        h_bin[flip_indices] = 0
    elif current_sum < target_sum:
        # Turn some 0s to 1s
        zeros_indices = np.where(h_bin == 0)[0]
        to_flip = target_sum - current_sum
        flip_indices = np.random.choice(zeros_indices, to_flip, replace=False)
        h_bin[flip_indices] = 1
    
    best_h = h_bin.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Tabu search parameters
    tabu_tenure = 10
    tabu_list = []
    max_no_improvement = 100
    no_improvement_count = 0
    shaking_frequency = 50
    
    # Precompute neighbor generation
    neighbor_cache = {}
    
    # Main tabu search loop
    while time() - start_time < budget_s * 0.95:
        # Check for shaking
        if no_improvement_count >= shaking_frequency:
            # Perform shaking
            h_shake = best_h.copy()
            # Flip 5% of the bits
            n_flip = max(1, n_points // 20)
            flip_indices = np.random.choice(n_points, n_flip, replace=False)
            for idx in flip_indices:
                h_shake[idx] = 1 - h_shake[idx]
            # Ensure exact sum constraint
            current_sum = np.sum(h_shake)
            if current_sum > target_sum:
                ones_indices = np.where(h_shake == 1)[0]
                to_flip = current_sum - target_sum
                flip_indices = np.random.choice(ones_indices, to_flip, replace=False)
                h_shake[flip_indices] = 0
            elif current_sum < target_sum:
                zeros_indices = np.where(h_shake == 0)[0]
                to_flip = target_sum - current_sum
                flip_indices = np.random.choice(zeros_indices, to_flip, replace=False)
                h_shake[flip_indices] = 1
            h_current = h_shake
            no_improvement_count = 0
        else:
            # Generate neighbor by flipping one 1 to 0 and one 0 to 1
            ones_indices = np.where(best_h == 1)[0]
            zeros_indices = np.where(best_h == 0)[0]
            
            if len(ones_indices) > 0 and len(zeros_indices) > 0:
                i1 = np.random.choice(ones_indices)
                i0 = np.random.choice(zeros_indices)
                
                # Create neighbor
                h_current = best_h.copy()
                h_current[i1] = 0
                h_current[i0] = 1
                
                # Cache neighbor for evaluation
                neighbor_key = tuple(h_current)
                if neighbor_key not in neighbor_cache:
                    neighbor_cache[neighbor_key] = evaluate_erdos_solution(h_current, dx)
                current_c5 = neighbor_cache[neighbor_key]
            else:
                # Fallback if constraint violated
                h_current = best_h.copy()
                current_c5 = best_c5
        
        # Evaluate current neighbor
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        # Check if it's better (or aspirational)
        is_aspiration = current_c5 < best_c5 and tuple(h_current) in tabu_list
        
        # Evaluate the local search move
        if current_c5 < best_c5 or is_aspiration:
            best_h = h_current.copy()
            best_c5 = current_c5
            no_improvement_count = 0
            
            # Update tabu list (remove oldest and add new)
            if len(tabu_list) >= tabu_tenure:
                tabu_list.pop(0)
            tabu_list.append(tuple(h_current))
        else:
            no_improvement_count += 1
            
        # Terminate if no improvement for too long
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Ensure we meet constraints exactly
    final_sum = np.sum(best_h)
    if final_sum != target_sum:
        # Adjust if needed
        ones_indices = np.where(best_h == 1)[0]
        zeros_indices = np.where(best_h == 0)[0]
        to_fix = int(final_sum - target_sum)
        if to_fix > 0:
            # Remove some ones
            indices_to_flip = np.random.choice(ones_indices, to_fix, replace=False)
            best_h[indices_to_flip] = 0
        elif to_fix < 0:
            # Add some ones
            indices_to_flip = np.random.choice(zeros_indices, -to_fix, replace=False)
            best_h[indices_to_flip] = 1
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
