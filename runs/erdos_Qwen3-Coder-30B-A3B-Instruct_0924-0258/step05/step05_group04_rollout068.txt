import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # SCP parameters
    max_iter = 500
    mu = 1e-3
    c5_prev = 1.0
    tolerance = 1e-6
    
    # Main SCP loop
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        current_c5 = evaluate_erdos_solution(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
            
        if abs(current_c5 - c5_prev) < tolerance:
            break
        c5_prev = current_c5
        
        # Compute all correlations for current h
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        corr_values = corr_full[len(best_h)-1:]  # Only positive shifts
        max_overlap_idx = np.argmax(corr_values)
        max_overlap = corr_values[max_overlap_idx] * dx
        
        if max_overlap <= current_c5 + 1e-10:
            continue
            
        # Formulate CVXPY problem
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(cp.Variable())
        
        # Constraints: h_var[i] in [0,1]
        constraints = [h_var >= 0, h_var <= 1]
        
        # Constraint for the maximum overlap
        h_shifted = np.roll(best_h, -max_overlap_idx)
        overlap_expr = cp.sum(cp.multiply(h_var, 1 - h_shifted)) * dx
        constraints.append(overlap_expr <= cp.Variable())
        
        # Sum constraint
        constraints.append(cp.sum(h_var) * dx == 1.0)
        
        # Solve
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.OSQP, verbose=False, eps_abs=1e-6, eps_rel=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h_new = np.clip(h_new, 0, 1)
                    # Normalize
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    best_h = h_new
        except:
            pass
            
        # Update μ
        if max_overlap > current_c5 + 1e-10:
            mu *= 2
            
        # Early termination
        if best_c5 < 0.3807:
            break
    
    # Final refinement with NLP solver
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_init = best_h.copy()
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
