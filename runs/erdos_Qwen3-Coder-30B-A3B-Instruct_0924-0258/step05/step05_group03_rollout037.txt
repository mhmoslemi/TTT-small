import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize initial solution
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Parameters
    max_iterations = 500
    stagnation_limit = 50
    shift_samples = 50
    
    # Initialize
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    last_improvement = 0
    restart_count = 0
    
    # Helper function to compute gradient
    def compute_gradient(h, k, dx):
        grad = np.zeros_like(h)
        n = len(h)
        for i in range(n):
            grad[i] = (1 - h[(i + k) % n]) * dx
            grad[(i - k) % n] -= h[i] * dx
        return grad
    
    # Backtracking line search
    def backtracking_line_search(h, grad, alpha_init=1.0, rho=0.5, c=1e-4):
        alpha = alpha_init
        current_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
        while True:
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_c5 = evaluate_erdos_solution(h_new, 2.0 / n_points)
            if new_c5 <= current_c5 - c * alpha * np.dot(grad, grad):
                return alpha, h_new
            alpha *= rho
            if alpha < 1e-10:
                return alpha, h_new
    
    iteration = 0
    while iteration < max_iterations and time() - start_time < budget_s * 0.95:
        # Sample shifts
        shifts = np.random.choice(n_points, shift_samples, replace=False)
        max_overlap = 0
        best_shift = 0
        
        # Evaluate overlap for each sampled shift
        for k in shifts:
            corr = np.correlate(h, 1 - h, mode='full')
            overlap = corr[n_points - 1 + k] * (2.0 / n_points)
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = k
        
        # Compute gradient for the worst shift
        grad = compute_gradient(h, best_shift, 2.0 / n_points)
        
        # Line search
        alpha, h_new = backtracking_line_search(h, grad)
        
        # Update
        h = h_new
        c5 = evaluate_erdos_solution(h, 2.0 / n_points)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Stagnation check
        if iteration - last_improvement > stagnation_limit:
            # Perturb solution
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = iteration
            restart_count += 1
            
        iteration += 1
        
        # Early termination
        if best_c5 < 0.3807:
            break
    
    # Final refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
        
    return (list(best_h), best_c5, n_points)
