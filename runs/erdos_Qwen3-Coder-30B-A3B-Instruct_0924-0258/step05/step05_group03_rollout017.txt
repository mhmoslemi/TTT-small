import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Ensure initial solution is feasible
    h = h * (n_points / 2) / np.sum(h)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    max_iter = 2000
    tolerance = 1e-6
    stagnation_limit = 50
    stagnation_count = 0
    
    # Convex-concave procedure
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find worst shift k using grid search
        shifts = np.linspace(0, 2, 100)
        max_correlations = []
        
        for k in shifts:
            # Align h with shifted version (manually compute correlation)
            k_idx = int(k / dx)
            if k_idx >= n_points:
                k_idx = n_points - 1
                
            # Compute overlap for this shift
            corr = 0.0
            for i in range(n_points):
                j = i + k_idx
                if j < n_points:
                    corr += h[i] * (1 - h[j])
            max_correlations.append(corr * dx)
            
        k_star = shifts[np.argmax(max_correlations)]
        k_idx = int(k_star / dx)
        
        # Solve convex optimization problem for h given fixed k*
        # Minimize sum h[i] * (1 - h[i + k]) for fixed k
        h_new = cp.Variable(n_points)
        constraints = [h_new[i] >= 0 for i in range(n_points)]
        constraints.extend([h_new[i] <= 1 for i in range(n_points)])
        constraints.append(cp.sum(h_new) == n_points / 2)
        
        # Objective: minimize sum of h[i] * (1 - h[i + k])
        objective = 0
        for i in range(n_points):
            j = i + k_idx
            if j < n_points:
                objective += h_new[i] * (1 - h_new[j])
        
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_new_vals = h_new.value
                h_new_vals = np.clip(h_new_vals, 0, 1)
                h_new_vals = h_new_vals * (n_points / 2) / np.sum(h_new_vals)
                h = h_new_vals
                
                # Evaluate new C5
                new_c5 = evaluate_erdos_solution(h, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                    
                if best_c5 < 0.3807:
                    break
                
                if stagnation_count > stagnation_limit:
                    # Reinitialize with slightly perturbed solution
                    h = best_h + np.random.normal(0, 0.001, n_points)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    stagnation_count = 0
                    
        except:
            # Fallback to gradient-based optimization
            try:
                def obj(x):
                    return evaluate_erdos_solution(x, dx)

                def constraint_sum(x):
                    return np.sum(x) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(obj, h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h = res.x
                    new_c5 = obj(h)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                        stagnation_count = 0
                    else:
                        stagnation_count += 1
            except:
                # As a last resort, just reinitialize
                h = np.random.random(n_points)
                h = h * (n_points / 2) / np.sum(h)
    
    # Final polishing with local optimization
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass

    return (list(best_h), best_c5, n_points)
