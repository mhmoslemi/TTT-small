import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
import warnings
warnings.filterwarnings('ignore')

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Initial samples
    n_init = 20
    init_samples = []
    init_values = []
    
    # Add parent if exists
    if 'initial_h_values' in globals():
        init_samples.append(h_start)
        init_values.append(objective(h_start))
    
    # Generate random initial points
    for _ in range(n_init):
        h = np.random.random(n_points)
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        init_samples.append(h)
        init_values.append(objective(h))
    
    # Fit initial GP
    X_init = np.array(init_samples)
    y_init = np.array(init_values)
    
    # Use Gaussian Process for surrogate model
    kernel = C(1.0, (1e-3, 1e3)) * RBF(1.0, (1e-2, 1e2))
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-6, 
                                  normalize_y=True, n_restarts_optimizer=2)
    gp.fit(X_init, y_init)
    
    # Optimizer loop
    n_evaluations = len(init_samples)
    max_evaluations = 500  # reasonable limit within time budget
    
    while time() - start_time < budget_s * 0.95 and n_evaluations < max_evaluations:
        # Generate candidates
        n_candidates = 50
        candidates = []
        
        for _ in range(n_candidates):
            # Sample from GP with acquisition function (upper confidence bound)
            x = np.random.random(n_points)
            x = x * (n_points/2) / np.sum(x)
            x = np.clip(x, 0, 1)
            candidates.append(x)
        
        # Evaluate all candidates
        x_candidates = np.array(candidates)
        pred_mean, pred_std = gp.predict(x_candidates, return_std=True)
        
        # Sort by predicted value and use uncertainty for diversity
        sorted_idx = np.argsort(pred_mean)
        selected_candidates = []
        
        # Select top candidates with some diversity
        for i in range(min(10, len(sorted_idx))):
            idx = sorted_idx[i]
            selected_candidates.append(x_candidates[idx])
            
        # Evaluate selected candidates
        for h_candidate in selected_candidates:
            # Ensure feasibility
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
            
            c5_val = objective(h_candidate)
            n_evaluations += 1
            
            if c5_val < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_val
                
            # Add to training data
            X_init = np.vstack([X_init, h_candidate])
            y_init = np.append(y_init, c5_val)
            
            # Retrain GP
            try:
                gp.fit(X_init, y_init)
            except:
                # Fallback if GP fails to fit
                pass
                
        # Occasionally increase resolution
        if n_evaluations % 25 == 0 and n_points < 80:
            n_points += 5
            dx = 2.0 / n_points
            # Interpolate current best
            h_new = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
            best_h = h_new.copy()
            best_c5 = evaluate_erdos_solution(best_h, dx)
            # Reset for better exploration
            init_samples = []
            init_values = []
            for _ in range(10):
                h = np.random.random(n_points)
                h = h * (n_points/2) / np.sum(h)
                h = np.clip(h, 0, 1)
                init_samples.append(h)
                init_values.append(objective(h))
            X_init = np.array(init_samples)
            y_init = np.array(init_values)
            gp.fit(X_init, y_init)
    
    return (list(best_h), best_c5, n_points)
