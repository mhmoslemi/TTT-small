import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.linspace(0, 1, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    population_size = 30
    mutation_factor = 0.8
    crossover_prob = 0.8
    max_generations = 600
    
    # Initialize population
    population = []
    for i in range(population_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evolution loop
    stagnation_counter = 0
    current_best = best_c5
    
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
            if best_c5 < 0.3807:
                break
        
        # Check stagnation
        if fitness[min_fitness_idx] < current_best:
            current_best = fitness[min_fitness_idx]
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Selection and reproduction
        new_population = [population[min_fitness_idx]]
        
        for _ in range(population_size - 1):
            # Tournament selection
            idx1, idx2, idx3 = np.random.choice(len(population), 3, replace=False)
            if fitness[idx1] < fitness[idx2]:
                parent1 = population[idx1]
            else:
                parent1 = population[idx2]
            if fitness[idx3] < fitness[idx1]:
                parent2 = population[idx3]
            else:
                parent2 = population[idx1]
                
            # Differential mutation
            mutant = parent1 + mutation_factor * (parent2 - population[idx3])
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            crossover_mask = np.random.random(n_points) < crossover_prob
            if np.any(crossover_mask):
                child = np.where(crossover_mask, mutant, population[idx3])
            else:
                child = mutant.copy()
                
            # Repair to satisfy constraint
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        
        # Local refinement every 10 generations
        if gen % 10 == 0:
            refined_population = []
            for i in range(min(5, len(population))):
                h_init = population[i]
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_init, method='L-BFGS-B', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'ftol': 1e-8})
                    if res.success:
                        h_refined = res.x
                        refined_population.append(h_refined)
                        continue
                except:
                    pass
                refined_population.append(h_init)
            population = refined_population
    
    # Final refinement using full optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
