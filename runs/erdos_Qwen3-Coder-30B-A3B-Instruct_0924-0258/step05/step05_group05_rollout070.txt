from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    # First attempt to solve a relaxed problem using CVXPY
    try:
        # Create SDP relaxation - this is a simplified version for this case
        # We'll use a hybrid approach: use SDP for a good starting point, 
        # then refine with gradient-based optimization
        
        # Initialize with the best known solution if available
        h_current = h_start.copy()
        
        # Gradient-based refinement with projection
        def objective(h):
            return evaluate_erdos_solution(h, dx)

        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        # Bounds for each h[i]
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}

        max_iter = 150

        res = minimize(objective, h_refined, method=method,
