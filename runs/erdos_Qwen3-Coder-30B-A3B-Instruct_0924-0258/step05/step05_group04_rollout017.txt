import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # QP-based sequential convex programming approach
    max_iter = 500
    mu = 1e-3
    c5_upper = 0.4
    tolerance = 1e-6
    
    # Early exit for extremely good solutions
    if best_c5 < 0.3807:
        return (list(best_h), best_c5, n_points)
    
    h_current = h_start.copy()
    iter_count = 0
    
    while time() - start_time < max_time * 0.95 and iter_count < max_iter:
        iter_count += 1
        # Calculate overlap constraints
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        overlaps = corr * dx
        max_overlap = np.max(overlaps)
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h_current.copy()
            
        if best_c5 < 0.3807:
            break
            
        # Formulate QP: minimize C5 
        # Subject to linearized constraints around h_current
        h = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Objective: minimize C5
        objective = cp.Minimize(C5)
        
        # Linearized constraints at current point
        constraints = []
        
        # Add integral constraint
        constraints.append(cp.sum(h) == n_points / 2.0)
        
        # Add box constraints
        for i in range(n_points):
            constraints.append(h[i] <= 1)
            constraints.append(h[i] >= 0)
            
        # Add overlap constraints using linearization
        # For each k, we have: sum(h[i] * (1 - h[i+k])) * dx <= C5
        # Linearize as: sum(h[i] * (1 - h_current[i+k])) * dx <= C5
        # This is a rough approximation, but allows solving QP
        # We look to satisfiy all overlaps simultaneously so we add penalties
        
        # Create constraints based on original overlaps calculation 
        # We generate the linearized constraints by focusing on violations
        h_conv = np.concatenate([h_current, h_current[:n_points-1]])
        for k in range(1, 2 * n_points - 1):
            # Linearized quadratic term
            # Use linearization at h_current
            lhs = 0
            for i in range(n_points):
                lhs += h[i] * (1 - h_current[k-i]) * dx
            constraints.append(lhs <= C5)
        
        # Add penalty to prevent getting stuck
        penalty = mu * cp.sum_squares(h - h_current)
        objective = cp.Minimize(C5 + penalty)
        
        # Solve QP
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.OSQP, 
                       eps_abs=1e-6, 
                       eps_rel=1e-6,
                       max_iter=2000)
            
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_new = h.value
                c5_new = C5.value
                h_current = h_new.copy()
                if c5_new < c5_upper:
                    c5_upper = c5_new
                mu *= 2  # Increase penalty on subsequent iterations
            else:
                # Fallback to a gradient-like step
                h_current = h_current + 0.01 * (h_current - 0.5)
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points / 2) / np.sum(h_current)
                mu = max(1e-6, mu * 0.99)
        except:
            h_current = h_current + 0.01 * (h_current - 0.5)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            mu = max(1e-6, mu * 0.99)
    
    # Final local refinement
    h_init = best_h.copy()
    dx = 2.0 / n_points

    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, h_init, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
