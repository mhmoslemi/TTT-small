import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 50
    max_generations = 1000
    mutation_rate = 0.15
    crossover_rate = 0.8
    tournament_size = 3
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    def tournament_selection(pop, fitnesses):
        indices = np.random.choice(len(pop), size=tournament_size, replace=False)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        return pop[best_idx]
    
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        crossover_point = random.randint(1, n_points-1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                h[i] += np.random.normal(0, 0.03)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Evolutionary loop
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time() - start_time < budget_s * 0.9 and generation < max_generations:
        # Evaluate fitness
        fitnesses = [fitness(ind) for ind in population]
        current_best_idx = np.argmin(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        
        if current_best_fitness < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = current_best_fitness
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count > max_no_improvement:
            # Reset with a fresh population starting from best solution
            best_h = best_h * (n_points/2) / np.sum(best_h)
            population = []
            for _ in range(pop_size):
                h = best_h + np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                population.append(h)
            no_improvement_count = 0
        else:
            # Generate new population
            new_population = []
            # Elitism: Keep the best individual
            new_population.append(population[current_best_idx])
            while len(new_population) < pop_size:
                parent1 = tournament_selection(population, fitnesses)
                parent2 = tournament_selection(population, fitnesses)
                child1, child2 = crossover(parent1, parent2)
                child1 = mutate(child1)
                child2 = mutate(child2)
                new_population.extend([child1, child2])
            population = new_population[:pop_size]
        
        generation += 1
    
    # Final optimization
    if time() - start_time < budget_s * 0.1:
        try:
            h_final = best_h.copy()
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
