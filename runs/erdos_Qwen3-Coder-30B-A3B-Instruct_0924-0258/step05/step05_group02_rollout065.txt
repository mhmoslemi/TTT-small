import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid approach parameters
    n_low = 20
    dx_low = 2.0 / n_low
    
    # Coarse grid relaxation
    try:
        h_low = cp.Variable(n_low)
        constraints = [
            cp.sum(h_low) == n_low / 2.0,
            h_low >= 0,
            h_low <= 1
        ]
        
        # Add constraints for some shifts to reduce complexity
        for k in range(1, min(10, n_low)):
            if k < n_low // 2:
                corr_sum = cp.sum(cp.multiply(h_low[:n_low-k], 1 - h_low[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_low[k:], 1 - h_low[:n_low-k]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
            
        prob = cp.Problem(cp.Minimize(cp.sum(h_low)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        
        if prob.status == cp.OPTIMAL:
            h_coarse = h_low.value
            # Upsample to fine grid
            h_fine = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, n_low), 
                               h_coarse)
            h_fine = np.clip(h_fine, 0, 1)
            h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
            
            # Refinement using SLSQP
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
                
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_fine, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-7}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
    except Exception:
        pass

    # Fine-tune with gradient-based optimization
    h_current = best_h.copy()
    iterations_no_improvement = 0
    max_iterations_no_improvement = 100
    step_size = 0.01
    
    def compute_gradient(h, dx):
        # Compute gradient of C5 w.r.t. h using finite differences
        n = len(h)
        grad = np.zeros(n)
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            h_minus = h.copy()
            h_minus[i] -= 1e-6
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            grad[i] = (c5_plus - c5_minus) / (2e-6)
        return grad
    
    while time() - start_time < budget_s * 0.9:
        # Compute current gradient
        try:
            grad = compute_gradient(h_current, dx)
        except:
            break
            
        # Update using gradient descent
        h_new = h_current - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            iterations_no_improvement = 0
        else:
            iterations_no_improvement += 1
            # Adaptive learning rate decay
            step_size *= 0.99
            
        h_current = h_new.copy()
        
        # Mutation with probability based on gradient magnitude
        if np.random.rand() < 0.1 or iterations_no_improvement > 50:
            mutation_magnitude = 0.05 * np.mean(np.abs(grad))
            mutation = np.random.normal(0, mutation_magnitude, n_points)
            h_mutation = h_current + mutation
            h_mutation = np.clip(h_mutation, 0, 1)
            h_mutation = h_mutation * (n_points / 2) / np.sum(h_mutation)
            c5_mutation = evaluate_erdos_solution(h_mutation, dx)
            if c5_mutation < best_c5:
                best_h = h_mutation
                best_c5 = c5_mutation
                iterations_no_improvement = 0
                
        if iterations_no_improvement > max_iterations_no_improvement:
            break
    
    # Final normalization and validation
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
