import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with parent construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Algorithm parameters
    max_iter = 100
    shift_samples = 100
    tolerance = 1e-6
    perturbation_magnitude = 0.01
    
    # Initialize
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    # Iterative improvement loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Step 1: Compute shifts
        shifts = np.linspace(0, 2, shift_samples, endpoint=False)
        overlaps = []
        for k in shifts:
            # Compute shifted correlation for this k
            shift_idx = int(k * n_points / 2)
            if shift_idx == 0:
                corr = np.correlate(h, 1-h, mode='full')
            else:
                # Manually compute circular correlation for small shift
                h_shifted = np.roll(h, shift_idx)
                corr = np.correlate(h, 1-h_shifted, mode='full')
                
            # Valid range of correlation indices for this shift
            start_idx = shift_idx
            end_idx = shift_idx + n_points
            overlap_at_k = np.max(corr[start_idx:end_idx]) * (2.0 / n_points)
            overlaps.append(overlap_at_k)
        
        # Find max overlap and corresponding shift
        max_overlap_idx = np.argmax(overlaps)
        k_star = shifts[max_overlap_idx]
        
        # Step 2: Optimization - solve for optimal h given k*
        # Create optimization variables
        h_var = Variable(n_points)
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, 
                       sum(h_var) == n_points / 2]
        
        # Objective: minimize overlap at k*
        # Therefore minimize sum(h[i] * (1-h[i+k])) * dx  
        # This is a bilinear form to approximate
        obj_expr = 0
        for i in range(n_points):
            # Periodic shift for k*
            j = (i + int(k_star * n_points / 2)) % n_points
            obj_expr += h_var[i] * (1 - h_var[j])
            
        objective = Minimize(obj_expr)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False, max_iters=1000)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_new = np.array(h_var.value).flatten()
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Update if better
                c5_new = evaluate_erdos_solution(h_new, 2.0 / n_points)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                else:
                    # Perturbation if not improving
                    h = h_new.copy()
                    noise = np.random.normal(0, perturbation_magnitude, n_points)
                    h += noise
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    
                # Early termination
                if best_c5 < 0.3807:
                    break
            else:
                # Fall back to basic strategy
                h = h_new.copy()
                noise = np.random.normal(0, perturbation_magnitude, n_points)
                h += noise
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback if optimization fails
            noise = np.random.normal(0, perturbation_magnitude, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final polishing using local optimization
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
