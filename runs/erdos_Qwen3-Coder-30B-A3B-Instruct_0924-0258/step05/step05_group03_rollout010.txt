import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize to have sum = n_points / 2
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Current best solution
    h_current = initial_h.copy()
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Termination parameters
    max_iterations = 500
    stagnation_limit = 50
    iteration = 0
    stagnation_count = 0
    last_best = best_c5
    
    while iteration < max_iterations and time() - start_time < budget_s * 0.95:
        # Sample some shifts to identify the most problematic one
        shift_samples = np.random.choice(n_points, size=min(50, n_points), replace=False)
        
        max_overlap_val = -1
        best_shift = 0
        
        # Find shift that produces maximum overlap
        for k in shift_samples:
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h_current[i] * (1 - h_current[j])
            overlap *= dx
            if overlap > max_overlap_val:
                max_overlap_val = overlap
                best_shift = k
        
        # Compute gradient for this shift
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_shift) % n_points
            gradient[i] += (1 - h_current[j]) * dx
            gradient[j] -= h_current[i] * dx
        
        # Line search for step size
        alpha = 1.0
        step_size_factor = 0.5
        max_steps = 10
        
        # Try different step sizes
        for _ in range(max_steps):
            h_trial = h_current - alpha * gradient
            # Project to feasible region
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            trial_c5 = evaluate_erdos_solution(h_trial, dx)
            
            if trial_c5 < best_c5:
                h_current = h_trial
                best_c5 = trial_c5
                break
            else:
                alpha *= step_size_factor
                
        # Update best solution if improved
        if best_c5 < last_best:
            last_best = best_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Early termination if target met
        if best_c5 <= 0.3807:
            break
            
        # Restart if stuck
        if stagnation_count >= stagnation_limit:
            # Add small random noise
            noise = np.random.normal(0, 0.01, n_points)
            h_current = h_current + noise
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            stagnation_count = 0

        iteration += 1
        
    # Final refinement using scipy
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, h_current, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        
        if res.success:
            h_current = res.x
            best_c5 = obj(h_current)
    except:
        pass
    
    return (list(h_current), best_c5, n_points)
