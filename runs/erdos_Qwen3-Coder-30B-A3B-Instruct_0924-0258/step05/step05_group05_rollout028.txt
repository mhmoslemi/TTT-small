import numpy as np
from scipy.optimize import minimize
from time import time
import hashlib

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_tenure = max(10, int(0.2 * n_points))
    max_iter = 1000
    max_stagnation = 100
    mutation_scale = 0.04
    perturb_fraction = 0.3
    refinement_interval = 10
    
    # State tracking
    tabu_list = []
    stagnation_counter = 0
    improvement_count = 0
    current_best = best_c5
    
    def hash_state(h):
        # Create a hash for the state to detect tabu
        sorted_h = np.sort(h)
        return hashlib.md5(sorted_h.tobytes()).hexdigest()
    
    def perturb_solution(h, scale):
        # Create a neighbor by perturbing a fraction of elements
        h_new = h.copy()
        n_perturb = max(1, int(perturb_fraction * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        for i in indices:
            # Add Gaussian noise
            h_new[i] += np.random.normal(0, scale)
        
        # Ensure bounds and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        return h_new
    
    # Main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate
        h_candidate = perturb_solution(best_h, mutation_scale)
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Check if it's tabu
        candidate_hash = hash_state(h_candidate)
        is_tabu = candidate_hash in tabu_list
        if not is_tabu or c5_candidate < current_best:
            # Accept this solution
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            
            # Update tabu list
            tabu_list.append(candidate_hash)
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
                
            # Accept improvement
            if c5_candidate < current_best:
                current_best = c5_candidate
                stagnation_counter = 0
                improvement_count += 1
                
                # Local refinement every few improvements
                if improvement_count % refinement_interval == 0 and time() - start_time < budget_s * 0.9:
                    try:
                        def objective(h):
                            return evaluate_erdos_solution(h, dx)
                        
                        def constraint_sum(h):
                            return np.sum(h) - n_points / 2.0
                        
                        cons_sum = {'type': 'eq', 'fun': constraint_sum}
                        bounds = [(0, 1) for _ in range(n_points)]
                        
                        res = minimize(objective, best_h, method='SLSQP', 
                                       bounds=bounds, constraints=cons_sum, 
                                       options={'maxiter': 100, 'ftol': 1e-8})
                        if res.success:
                            h_refined = res.x
                            c5_refined = objective(h_refined)
                            if c5_refined < best_c5:
                                best_h = h_refined
                                best_c5 = c5_refined
                    except:
                        pass
            else:
                stagnation_counter += 1
        else:
            stagnation_counter += 1
            
        # Adjust mutation scale based on stagnation
        if stagnation_counter > max_stagnation:
            mutation_scale = min(0.5, mutation_scale * 1.2)
            stagnation_counter = 0
    
    return (list(best_h), best_c5, n_points)
