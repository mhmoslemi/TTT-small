import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization with cutting plane approach
    resolutions = [35, 40, 50, 60]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Cutting-plane optimization loop
        max_iter = 100
        iter_count = 0
        while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
            iter_count += 1
            
            # Evaluate current overlap
            current_c5 = evaluate_erdos_solution(h_current, dx)
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
            
            # Compute all correlations to find worst case
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            k_shift = max_idx - (len(h_current) - 1)
            
            # Create constraint: corr[k_shift] <= best_c5 * dx
            # We'll solve a convex relaxation to improve h
            
            # Optimization with constraints
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add constraint for current worst k
            if k_shift >= 0:
                # Add constraint on the shift
                corr_constraint = cp.sum(cp.multiply(h_opt[:n_points-k_shift], 1 - h_opt[k_shift:]))
            else:
                # k_shift < 0, we need to unroll
                corr_constraint = cp.sum(cp.multiply(h_opt[-k_shift:], 1 - h_opt[:n_points+k_shift]))
            
            constraints.append(corr_constraint <= best_c5 * dx + 1e-6)
            
            # Minimize largest overlap (in theoretical terms)
            # But this isn't easily expressible. Instead, just re-solve with that constraint.
            # For practicality, we try local search
                
            # Local refinement using Nelder-Mead (some robustness) 
            h_refined = h_current.copy()
            
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(objective, h_refined, method='Nelder-Mead',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 100, 'adaptive': True})
                
                if res.success:
                    h_update = res.x
                    c5_update = objective(h_update)
                    if c5_update < best_c5:
                        best_h = h_update
                        best_c5 = c5_update
                        h_current = h_update
                        continue
            except Exception:
                pass
                
            # Small random perturb and normalize
            h_perturb = h_current + np.random.normal(0, 0.01, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
            
            new_c5 = evaluate_erdos_solution(h_perturb, dx)
            if new_c5 < best_c5:
                best_h = h_perturb.copy()
                best_c5 = new_c5
                h_current = h_perturb.copy()
                
            # Ensure that we don't waste all time
            if iter_count > 150:
                break

        h_start = best_h.copy()
    
    # Final refinement with SLSQP
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        res = minimize(objective, best_h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 300, 'ftol': 1e-9})
        
        if res.success:
            final_h = res.x
            final_c5 = objective(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
