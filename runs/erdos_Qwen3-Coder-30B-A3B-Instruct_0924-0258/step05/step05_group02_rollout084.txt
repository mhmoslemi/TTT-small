import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initialize population
    pop_size = 50
    n_points = 35
    dx = 2.0 / n_points
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    else:
        h_start = np.ones(n_points) * 0.5
    
    # Population initialization
    population = [h_start.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    # Generational evolution
    max_generations = 1000
    elitism = pop_size // 3
    mutation_rate = 0.1
    
    # Track best solution across generations
    generation_count = 0
    
    # Main evolution loop
    while time() - start_time < budget_s * 0.95 and generation_count < max_generations:
        # Evaluate all individuals in population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        new_pop = population[:elitism]
        
        # Generate offspring
        while len(new_pop) < pop_size:
            # Crossover
            parent1 = population[random.randint(0, elitism - 1)]
            parent2 = population[random.randint(0, elitism - 1)]
            
            # Blend crossover with weights in [0.4, 0.6]
            alpha = random.uniform(0.4, 0.6)
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += random.gauss(0, 0.02)
            
            # Ensure constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            # Add to population
            new_pop.append(child)
        
        population = new_pop[:]
        generation_count += 1
    
    # Final gradient refinement on best
    if best_h is not None:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Refine with optimization
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Ensure final constraints
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points/2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
