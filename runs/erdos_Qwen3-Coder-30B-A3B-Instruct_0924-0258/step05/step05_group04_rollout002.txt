import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    max_iter = 5000
    mu = 1e-3
    tol = 1e-6
    admm_iter = 0
    
    # Initialize ADMM variables
    h = h_start.copy()
    h_prev = h.copy()
    u = np.zeros(2 * n_points - 1)
    lam = np.zeros(2 * n_points - 1)
    
    # ADMM iterations
    while admm_iter < max_iter and time() - start_time < max_time * 0.95:
        admm_iter += 1
        
        # Update h by solving quadratic subproblem
        # minimize (1/2)*||h - h_prev||^2 + mu*||z - (h*(1-h) - C5_threshold)||^2
        # subject to 0 <= h <= 1, sum(h) = n_points / 2
        
        # We'll make this simpler by directly minimizing with constraints
        # Create objective function for h update
        def obj_h(h_h):
            # Compute overlaps
            corr = np.correlate(h_h, 1 - h_h, mode='full') 
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        def constr_sum(h_h):
            return np.sum(h_h) - n_points / 2.0
            
        # Bounds
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constr_sum}
        
        # Avoid full optimization every iteration, just often enough
        if admm_iter % 10 == 0:
            try:
                # Solve the constrained optimization to get new h
                res = minimize(obj_h, h, method='SLSQP', bounds=bounds, 
                               constraints=cons, options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_new = res.x
                    h = h_new
                    
                    # Update best solution
                    c5_new = evaluate_erdos_solution(h, dx)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                        
            except:
                pass
                
        # Early stopping
        if best_c5 < 0.3807:
            break
    
    # Final fine-tuning with SLSQP
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        try:
            def obj_final(h_h): 
                return evaluate_erdos_solution(h_h, dx)
            def constr_final(h_h):
                return np.sum(h_h) - n_points / 2.0
            
            cons_final = {'type': 'eq', 'fun': constr_final}
            bounds_final = [(0, 1) for _ in range(n_points)]
            
            res_final = minimize(obj_final, best_h.copy(), method='SLSQP',
                                bounds=bounds_final, constraints=cons_final,
                                options={'maxiter': 200, 'ftol': 1e-9})
            
            if res_final.success:
                h_final = res_final.x
                c5_final = obj_final(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
                    
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
