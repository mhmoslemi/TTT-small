import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Smooth approximation parameters
    tau = 1.0
    max_iter = 3000
    tol = 1e-7
    step_size = 1.0
    stagnation_count = 0
    last_improvement = 0
    
    def smooth_max_overlaps(h, tau):
        """Compute smooth maximum of overlap integrals"""
        # For a step function h, compute overlaps at all possible shifts
        # This computes a convolution-like operation with h(x) * (1 - h(x+k))
        overlaps = []
        for k in range(2 * n_points - 1):
            # Compute overlap for shift k
            if k < n_points:
                h_shifted = np.concatenate([np.zeros(k), h[:n_points-k]])
            else:
                h_shifted = np.concatenate([h[k-n_points+1:], np.zeros(2*n_points-k-1)])
            overlap = np.sum(h * (1 - h_shifted)) * dx
            overlaps.append(overlap)
        # Smooth max using log-sum-exp
        return np.log(np.sum(np.exp(tau * np.array(overlaps)))) / tau

    def compute_gradient(h, tau):
        """Compute gradient of smooth max with respect to h"""
        # This requires numerical computation of gradient from the correlation
        # For computational efficiency, we compute drift based on minimum overlap terms
        # Two implementations: one based on full shifts for small n_points, another
        # via sweeps
        
        grad = np.zeros(n_points)
        
        # Compute all overlaps
        all_overlaps = []
        for k in range(2 * n_points - 1):
            # compute h * (1 - h_{.,k})
            if k < n_points:
                h_shifted = np.concatenate([np.zeros(k), h[:n_points-k]])
            else:
                h_shifted = np.concatenate([h[k-n_points+1:], np.zeros(2*n_points-k-1)])
            overlap_val = np.sum(h * (1 - h_shifted)) * dx
            all_overlaps.append(overlap_val)
            
        all_overlaps = np.array(all_overlaps)
        exp_vals = np.exp(tau * all_overlaps)
        Z = np.sum(exp_vals)
        weights = exp_vals / Z
        
        # Gradient contribution from each term
        for k in range(2 * n_points - 1):
            if k < n_points:
                h_shifted = np.concatenate([np.zeros(k), h[:n_points-k]])
            else:
                h_shifted = np.concatenate([h[k-n_points+1:], np.zeros(2*n_points-k-1)])
                
            # Gradient of overlap, weighted by the weight of this term
            d_overlaps_dh = 1 - h_shifted
            # For the k-th term, we simulate derivative w.r.t. h
            if k < n_points:
                d_overlaps_dh_shifted = np.concatenate([np.zeros(k), np.ones(n_points-k)])
            else:
                d_overlaps_dh_shifted = np.concatenate([np.ones(2*n_points-k-1), np.zeros(k-n_points+1)])
                
            local_grad = (1 - h_shifted) * (1 - d_overlaps_dh_shifted) * dx
            # shape of local_grad is n_points, we must sum over shifts weighted by weight
            grad += weights[k] * local_grad

        return grad * tau

    # Gradient descent loop
    for i in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Compute function and gradient
        c5_smooth = smooth_max_overlaps(h_current, tau)
        try:
            grad = compute_gradient(h_current, tau)
        except:
            # fallback if gradient computation fails
            grad = np.random.randn(n_points) * 1e-3
            
        # Update h with L-BFGS-B method, which requires bounds and constraint handling
        # We'll use scipy minimize with bounds, despite non-smoothness, and be robust
        h_new = h_current - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        # Re-normalize to enforce sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stagnation_count = 0
            last_improvement = i
        else:
            stagnation_count += 1
            
        # Update h
        h_current = h_new
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        
        # Adjust hyperparameters
        if i % 100 == 0 and i > 0:
            tau *= 0.99  # Decrease temperature
        if stagnation_count > 100:
            tau = min(100.0, tau * 1.1)  # Increase smoothing if stagnating
            step_size = max(1e-6, step_size * 0.5)  # Reduce stepsize
            stagnation_count = 0
        if np.linalg.norm(grad) < 1e-6:
            break
            
        # Early escape if very good solution found
        if best_c5 < 0.3807:
            break
            
    # Final local refinement with tighter tolerance
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        try:
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            def con(h):
                return np.sum(h) - n_points / 2.0
            cons = {'type': 'eq', 'fun': con}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, best_h.copy(), method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = obj(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
