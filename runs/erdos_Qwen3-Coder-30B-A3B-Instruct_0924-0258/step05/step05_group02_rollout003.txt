import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial if available, otherwise create starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def project_to_feasible(h):
        """Ensure h is in [0,1] and integral equals 1."""
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Coarse grid optimization
    n_coarse = 20
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.ones(n_coarse) * 0.5
    
    try:
        # Use CVXPY to solve using convex relaxation for subset of constraints
        h_relaxed = cp.Variable(n_coarse)
        constraints = [
            cp.sum(h_relaxed) == n_points / 2.0,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        
        # Add some constraints (sample subset for efficiency)
        k_subset = min(5, n_coarse // 2)
        for k in range(1, k_subset):  
            if k < n_coarse // 2:
                corr_sum = cp.sum(cp.multiply(h_relaxed[:n_coarse-k], 1 - h_relaxed[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_coarse-k]))
            constraints.append(corr_sum <= best_c5 * dx_coarse + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            h_coarse = h_relaxed.value
    except Exception:
        pass
    
    # Interpolate coarse solution to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
    h_fine = project_to_feasible(h_fine)
    
    # Local refinement using SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    max_iter = 100
    iter_count = 0
    early_stop = 0
    
    while time() - start_time < budget_s * 0.8 and iter_count < max_iter:
        h_current = h_fine.copy()
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                objective,
                h_current,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 10, 'ftol': 1e-8}
            )
            if res.success:
                h_refined = res.x
                h_refined = project_to_feasible(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    h_fine = h_refined
                    best_c5 = c5_refined
                    early_stop = 0
                else:
                    early_stop += 1
                    if early_stop > 50:
                        break
            else:
                early_stop += 1
        except Exception:
            early_stop += 1
            if early_stop > 50:
                break
        
        iter_count += 1
    
    # Global restart
    restart_count = 0
    while time() - start_time < budget_s * 0.95 and restart_count < 30:
        try:
            h_restart = np.random.rand(n_points)
            h_restart = project_to_feasible(h_restart)
            c5_restart = evaluate_erdos_solution(h_restart, dx)
            if c5_restart < best_c5:
                h_fine = h_restart
                best_c5 = c5_restart
                restart_count = 0
            else:
                restart_count += 1
        except Exception:
            restart_count += 1

    # Final validation
    h_final = project_to_feasible(h_fine)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(h_final), best_c5, n_points)
