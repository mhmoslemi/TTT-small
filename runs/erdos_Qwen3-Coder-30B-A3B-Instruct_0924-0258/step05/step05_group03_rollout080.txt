import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial solution
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Use convex optimization with cutting planes
    # Discretize shifts for constraints
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    
    # Initialize h
    h_values = initial_h.copy()
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    
    # Convex optimization variables
    h = Variable(n_points)
    C5 = Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, sum(h) == n_points / 2, C5 >= 0]
    
    # Add initial constraint based on starting solution
    dx = 2.0 / n_points
    initial_c5 = evaluate_erdos_solution(h_values, dx)
    
    # Cutting plane approach
    iter_count = 0
    stagnation_count = 0
    best_c5 = initial_c5
    best_h = h_values.copy()
    
    # Build constraints for all shifts
    all_shift_constraints = []
    
    while time.time() - start_time < budget_s * 0.95 and stagnation_count < 50:
        # Sample shifts evenly
        shift_indices = np.linspace(0, n_points - 1, n_shifts, dtype=int, endpoint=False)
        shift_indices = np.minimum(shift_indices, n_points - 1)
        
        # Compute overlaps for current h
        overlap_values = []
        for shift_idx in shift_indices:
            shift = shift_idx * dx
            # Compute overlap integral
            if shift <= 2:
                # Use np.correlate with proper indexing
                corr = np.correlate(h_values, 1 - np.roll(h_values, shift_idx), mode='full')
                max_corr = np.max(corr)
                overlap = max_corr * dx
                overlap_values.append(overlap)
            else:
                overlap = np.inf
                overlap_values.append(overlap)
        
        # Find worst shift
        worst_idx = np.argmax(overlap_values)
        worst_shift_idx = shift_indices[worst_idx]
        max_overlap = overlap_values[worst_idx]
        
        # Add constraint: ensure max overlap constraint holds
        # For now, approximate with a smooth version using integration
        # Using correlation more directly
        
        # Overlap for shift k is integral of h(x)*(1-h(x+k)) over [0,2]
        # Perhaps use a direct (but inexact) method for constraints:
        # This page's approximation is to assume overlap can be evaluated via correlation
        
        # So we add C5 >= integral_h(x)(1-h(x+k))dx for the worst shift
        # We directly use the correlation as proxy for the integral
        corr = np.correlate(h_values, 1 - np.roll(h_values, worst_shift_idx), mode='full')
        max_corr = np.max(corr) 
        if max_corr * dx > C5.value:
            # Approximate the constraint on C5
            constraints.append(C5 >= max_corr * dx)
        else:
            constraints.append(C5 >= max_overlap)
            
        # Add incremental constraint using shift index as relative
        try:
            prob = Problem(Minimize(C5), constraints)
            prob.solve(solver=ECOS, verbose=False, max_iters=100)
            if prob.status in (Optimal, AlmostOptimal):
                new_c5 = C5.value
                # Update h
                h_values = [h[i].value for i in range(n_points)]
                h_values = np.array(h_values)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_values.copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                # Fallback to local search
                break
        except:
            # Fallback to direct optimization
            break
                        
        # Local refinement using SLSQP if time allows
        try:
            def obj(x):
                return evaluate_erdos_solution(x, dx)
            def constraint_sum(x):
                return np.sum(x) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(obj, h_values, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_values = res.x
                new_c5 = obj(h_values)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_values.copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
        except:
            stagnation_count += 1
        
        iter_count += 1
        
        # Early exit if target met
        if best_c5 < 0.3807:
            break
            
    # Final local refinement step
    try: 
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            final_h = res.x
            final_c5 = obj(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()
    except:
        pass
        
    return (list(best_h), best_c5, n_points)
