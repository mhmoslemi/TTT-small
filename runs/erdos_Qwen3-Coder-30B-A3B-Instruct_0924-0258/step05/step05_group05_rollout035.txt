import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create initial step function (symmetric with peak at center)
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Ensure the initial solution is feasible
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    h_start = np.clip(h_start, 0, 1)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Refinement parameters
    max_iter = 1000
    stagnation_limit = 100
    epsilon = 1e-8
    
    # Optimization setup
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Multi-start local optimization
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation of best solution
        h_pert = best_h + np.random.normal(0, 0.01, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        # Local optimization
        try:
            res = minimize(objective, h_pert, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 100, 'ftol': epsilon})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
                    if best_c5 < 0.3807:
                        break
        except:
            pass
    
    # Final fine-tuning
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            res = minimize(objective, best_h, method='L-BFGS-B', 
                          bounds=bounds, options={'maxiter': 200})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
