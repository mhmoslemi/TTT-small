import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Load initial solution or create default
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize initial solution
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Variables
    h = initial_h.copy()
    dx = 2.0 / n_points
    
    # Shift sampling for approximate maximization
    k_samples = np.linspace(0, 2, 100)
    
    # Iterative optimization
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    stagnation_count = 0
    max_stagnation = 50
    
    while time() - start_time < budget_s * 0.95:
        # Inner loop: Find best k for current h
        max_overlap = -np.inf
        best_k = 0
        
        for k in k_samples:
            # Compute overlap for shift k
            shift_idx = int(k / dx)
            if shift_idx >= n_points:
                overlap_val = 0
            else:
                overlap_val = np.sum(h * np.roll(1 - h, shift_idx)) * dx
                
            if overlap_val > max_overlap:
                max_overlap = overlap_val
                best_k = k
        
        # Outer loop: optimize h for fixed best_k
        # Use convex relaxation approach with CVXPY
        # We want to minimize overlap(k*) = sum h[i] * (1 - h[i + k*])
        
        # Using a small SDP-based relaxation for improved convergence
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # For simplicity, let's stick to gradient-based local optimization:
        # We'll compute the gradient numerically and apply a simple update
        
        # Simple optimization: use scipy minimize on small subspace
        def obj(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Local optimization using SLSQP
        h_new = h.copy()
        try:
            res = minimize(obj, h_new, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_new = res.x
                c5_new = obj(h_new)
                if c5_new < best_c5:
                    h = h_new.copy()
                    best_c5 = c5_new
                    best_h = h.copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
        except Exception as e:
            stagnation_count += 1
            pass
        
        if stagnation_count > max_stagnation:
            # Perturb the solution to escape local minima
            perturb_strength = 0.02
            h += np.random.normal(0, perturb_strength, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0
        
        # Early termination if target achieved
        if best_c5 < 0.3807:
            break
    
    # Final refinement
    try:
        def final_obj(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        def final_constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': final_constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(final_obj, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = final_obj(best_h)
    except Exception as e:
        pass
    
    return (list(best_h), best_c5, n_points)
