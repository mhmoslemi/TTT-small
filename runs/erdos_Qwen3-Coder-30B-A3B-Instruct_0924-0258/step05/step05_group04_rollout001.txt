import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

            # To handle the constraints differently, we'll use projected gradient

            return evaluate_erdos_solution(h_vec, dx)
