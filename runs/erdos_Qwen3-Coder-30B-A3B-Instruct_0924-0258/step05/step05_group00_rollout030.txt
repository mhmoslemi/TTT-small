import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def normalize_h(h):
        return h * (n_points/2) / np.sum(h)
    
    def mutate_individual(h, noise_level=0.02):
        h_mut = h.copy()
        indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        for i in indices:
            h_mut[i] += np.random.normal(0, noise_level)
        h_mut = np.clip(h_mut, 0, 1)
        return normalize_h(h_mut)
    
    def crossover(parent1, parent2):
        point = np.random.randint(1, n_points-1)
        child = np.concatenate([parent1[:point], parent2[point:]])
        return normalize_h(child)
    
    # Parameters
    population_size = 50
    generations = 1000
    tournament_size = 5
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        population_size -= 1
    
    for _ in range(population_size):
        h = np.random.random(n_points)
        h = normalize_h(h)
        population.append(h)
    
    # Evolutionary algorithm
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
        sorted_indices = np.argsort(fitness)
        
        # Update best solution
        if fitness[sorted_indices[0]] < best_c5:
            best_h = population[sorted_indices[0]].copy()
            best_c5 = fitness[sorted_indices[0]]
        
        # Selection and reproduction
        new_population = [population[sorted_indices[0]].copy()]
        
        while len(new_population) < population_size:
            # Tournament selection
            tournament_indices = np.random.choice(population_size, size=tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            if np.random.random() < crossover_rate:
                parent1 = population[winner_idx]
                parent2 = population[np.random.choice(population_size)]
                child = crossover(parent1, parent2)
            else:
                child = population[winner_idx].copy()
            
            # Mutation
            child = mutate_individual(child)
            new_population.append(child)
        
        population = new_population
    
    # Resolution adaptation
    if time() - start_time < budget_s * 0.8 and best_c5 > 0.3808:
        n_points = 40
        dx = 2.0 / n_points
        
        # Interpolate best solution
        x_original = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_original, best_h)
        best_h = normalize_h(h_interp)
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
        # Lower resolution GA
        population_size = 20
        generations = 300
        population = []
        
        if 'initial_h_values' in globals():
            population.append(best_h.copy())
            population_size -= 1
            
        for _ in range(population_size):
            h = np.random.random(n_points)
            h = normalize_h(h)
            population.append(h)
        
        for gen in range(generations):
            if time() - start_time > budget_s * 0.95:
                break
                
            fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
            sorted_indices = np.argsort(fitness)
            
            if fitness[sorted_indices[0]] < best_c5:
                best_h = population[sorted_indices[0]].copy()
                best_c5 = fitness[sorted_indices[0]]
            
            new_population = [population[sorted_indices[0]].copy()]
            
            while len(new_population) < population_size:
                tournament_indices = np.random.choice(population_size, size=tournament_size, replace=False)
                tournament_fitness = [fitness[i] for i in tournament_indices]
                winner_idx = tournament_indices[np.argmin(tournament_fitness)]
                
                if np.random.random() < crossover_rate:
                    parent1 = population[winner_idx]
                    parent2 = population[np.random.choice(population_size)]
                    child = crossover(parent1, parent2)
                else:
                    child = population[winner_idx].copy()
                
                child = mutate_individual(child)
                new_population.append(child)
            
            population = new_population
    
    # Final local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds,
                       constraints=cons_sum, options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            best_h = res.x
            best_c5 = objective(best_h)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
