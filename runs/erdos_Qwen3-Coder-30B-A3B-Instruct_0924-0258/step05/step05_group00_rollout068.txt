import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach
    resolutions = [35, 40, 50, 60]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to new resolution if needed
        if res > n_points:
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Cutting-plane approach to refine
        max_iter = 100
        for iter in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Calculate overlaps
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            k_max = max_idx - (n_points - 1)
            
            # Use a slightly relaxed approach for better convergence
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            # Constraint that the worst overlap should stay within our bound
            try:
                # Create a linear constraint for the current maximum overlap
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add bottleneck constraint  
                shift = k_max
                if shift >= 0:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-shift], 1 - h_opt[shift:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[-shift:], 1 - h_opt[:n_points+shift]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-8)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6)
                
                if prob.status == cp.OPTIMAL:
                    h_new = h_opt.value
                    new_c5 = evaluate_erdos_solution(h_new, dx)
                    if new_c5 < best_c5:
                        best_h = h_new
                        best_c5 = new_c5
                        h_current = h_new
                else:
                    h_current = h_current + np.random.normal(0, 0.001, n_points)
                    h_current = np.clip(h_current, 0, 1)
                    h_current = h_current * (n_points/2) / np.sum(h_current)
                    
            except Exception:
                h_current = h_current + np.random.normal(0, 0.001, n_points) 
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points/2) / np.sum(h_current)
        
        h_start = best_h.copy()
    
    # Final refinement with local optimization
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        # Local optimization with bounds
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final
    except Exception:
        pass
    
    # Ensure final validation
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_final
    
    return (list(best_h), best_c5, n_points)
