import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a symmetric triangular initial guess
        mid = n_points // 2
        h_start = np.minimum(np.arange(n_points), 2*mid - np.arange(n_points))
        h_start = h_start / (mid)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    max_iter = 1000
    stagnation_limit = 100
    min_improvement = 1e-8
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Local optimization step
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                # Repair to maintain constraints
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    if best_c5 < 0.3807:
                        break
        except:
            pass
        
        # Quick heuristic improvement using gradient descent
        # Use finite differences to estimate gradient
        def finite_diff_grad(h, eps=1e-4):
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_minus = h.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points/2) / np.sum(h_minus)
                grad[i] = (evaluate_erdos_solution(h_plus, dx) - 
                          evaluate_erdos_solution(h_minus, dx)) / (2 * eps)
            return grad
        
        # Simple gradient-based update
        if iteration % 5 == 0:
            grad = finite_diff_grad(best_h)
            # Simple line search with step size
            step_size = 1e-3
            h_new = best_h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        
        # Check for stagnation
        if iteration > 0 and iteration % 10 == 0:
            current_c5 = evaluate_erdos_solution(best_h, dx)
            if np.abs(current_c5 - best_c5) < min_improvement:
                # Try more aggressive exploration
                # Random perturbation
                perturbation = np.random.normal(0, 0.05, n_points)
                h_perturbed = best_h + perturbation
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
                c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
                if c5_perturbed < best_c5:
                    best_h = h_perturbed
                    best_c5 = c5_perturbed
            else:
                # Reset the stagnation counter
                pass

    # Final local refinement one more time
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
