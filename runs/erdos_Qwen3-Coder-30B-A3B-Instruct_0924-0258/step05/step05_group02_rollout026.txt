import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Determine resolution
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize with existing solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        # Create a good starting guess: piecewise constant with one peak
        h_start = np.ones(n_points) * 0.5
        # Add some structure
        h_start[5:15] = 0.8
        h_start[20:30] = 0.2
    
    # Normalize to sum to 1 (since ∫hdx = 1, and dx = 2/n)
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Alternative search: use coarse to fine resolution
    # Start with smaller grid (n=20) to get a rough solution quickly
    n_low = 20
    dx_low = 2.0 / n_low

    # Efficient initial solving on low-res grid
    try:
        h_low = cp.Variable(n_low)
        constraints = [
            cp.sum(h_low) == n_points / 2.0 * dx_low / dx,  # conservative, will fix
            h_low >= 0,
            h_low <= 1
        ]
        # Add key constraints for a few shifts, use soft constraints to stay in feasible set
        for k in range(1, min(10, n_low)):
            if k < n_low // 2:
                corr_sum = cp.sum(cp.multiply(h_low[:n_low-k], 1 - h_low[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_low[k:], 1 - h_low[:n_low-k]))
            # Limit overlap of k-shifted correlation, add soft penalty in proxy for empircal bound  
            constraints.append(corr_sum <= best_c5 * dx + 1e-5)

        # Solve with CVXPY for approximate solution
        prob = cp.Problem(cp.Minimize(cp.sum(h_low)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL and prob.value < np.inf:
            h_low_sol = h_low.value
            h_low_sol = np.clip(h_low_sol, 0, 1)
            # Interpolate to target resolution
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                 np.linspace(0, 2, n_low), h_low_sol)
            # Renormalize
            h_interp = h_interp * (n_points/2) / np.sum(h_interp)
            h_interp = np.clip(h_interp, 0, 1)
            c5_interp = evaluate_erdos_solution(h_interp, dx)
            if c5_interp < best_c5:
                best_h = h_interp
                best_c5 = c5_interp
    except Exception as e:
        # fall back to previous heuristics
        pass

    # Fine-grid optimization with local refinement
    max_iter = 1000
    iter_count = 0
    no_improvement_count = 0

    # Gradient-free optimization and perturbations on better solutions
    def perturb(h, factor=0.1):
        h_pert = h + np.random.normal(0, factor, len(h))
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        return h_pert

    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Try local refinement using SLSQP
        try:
            cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 10, 'ftol': 1e-8}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
        except Exception:
            no_improvement_count += 1

        if no_improvement_count > 10:
            # Perturb and restart
            no_improvement_count = 0
            h_new = perturb(best_h, 0.05)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

        iter_count += 1

    # Final check of solution
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), c5_final, n_points)
