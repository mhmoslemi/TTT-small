import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Use initial construction if available

        n_points = 35

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Normalize initial solution
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Helper functions
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    def construct_constraints(h, max_overlap):
        # Only add constraints for dominant shifts
        corr = np.correlate(h, 1 - h, mode='full')
        k_shifts = np.argsort(-np.abs(corr))[:min(10, len(corr)//2)]
        return [k_shifts, corr[k_shifts]]
    
    # Iterative refinement with expanding constraints
    max_iter = 200
    iter_count = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute current overlaps
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_overlap_idx = np.argmax(np.abs(corr))
        max_overlap = corr[max_overlap_idx] * dx
        
        # Add new constraint: reduce the maximum overlap by epsilon
        epsilon = 1e-6
        k_shift = max_overlap_idx - (len(corr) // 2)
        
        # Compute new bounds for optimization
        try:
            # Construct the constraint matrix
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            # Create a version of the objective with the current max overlap constraint
            def objective_with_constraint(h):
                return evaluate_erdos_solution(h, dx)

            def shift_constraint(h):
                # Compute overlap at the dominant shift k
                if k_shift >= 0:
                    overlap = np.sum(h[:-k_shift] * (1 - h[k_shift:]))
                else:
                    overlap = np.sum(h[-k_shift:] * (1 - h[:k_shift]))
                return overlap * dx - (max_overlap - epsilon)
            
            # Add constraint to optimization problem
            cons_shift = {'type': 'ineq', 'fun': shift_constraint}
            
            # Run local optimization
            res = minimize(
                objective_with_constraint, 
                best_h.copy(), 
                method='SLSQP', 
                bounds=bounds,

            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
