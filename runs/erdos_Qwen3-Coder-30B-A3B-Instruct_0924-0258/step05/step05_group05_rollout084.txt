import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.linspace(0, 1, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential evolution parameters
    population_size = 30
    mutation_factor = 0.8
    crossover_rate = 0.8
    max_generations = 500
    
    # Initialize population
    population = []
    for i in range(population_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evolution loop
    stagnation_counter = 0
    current_best = best_c5
    
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness of population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
            if best_c5 < 0.3807:
                break
        
        # Check stagnation
        if fitness[min_fitness_idx] < current_best:
            current_best = fitness[min_fitness_idx]
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Selection and reproduction
        new_population = []
        
        # Elitism: keep best two
        sorted_indices = np.argsort(fitness)
        new_population.extend([population[i] for i in sorted_indices[:2]])
        
        # Differential evolution
        while len(new_population) < population_size:
            # Select three distinct individuals
            idxs = random.sample(range(population_size), 3)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = a + mutation_factor * (b - c)
            
            # Crossover
            crossover_mask = np.random.random(n_points) < crossover_rate
            if not np.any(crossover_mask):
                crossover_mask[random.randint(0, n_points-1)] = True
            
            trial = np.where(crossover_mask, mutant, population[random.choice(range(population_size))])
            
            # Repair
            trial = np.clip(trial, 0, 1)
            if np.sum(trial) == 0:
                trial = np.ones_like(trial) * 0.5
            else:
                trial = trial * (n_points / 2) / np.sum(trial)
            
            new_population.append(trial)
        
        population = new_population
    
    # Final local refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
                
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons, 
                           options={'ftol': 1e-9, 'maxiter': 100})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
