import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
        
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Convex-concave procedure
    max_iter = 1000
    tol = 1e-6
    last_c5 = float('inf')
    
    # Sample shifts to find worst case
    shift_samples = np.linspace(0, 2, 100)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate overlaps for all shifts
        overlaps = []
        for k in shift_samples:
            # Compute correlation for this shift
            corr = np.correlate(h, 1-h, mode='full')
            # Find the correlation at shift k (in terms of indices)
            idx = int(k / (2.0 / n_points)) + n_points - 1
            if 0 <= idx < len(corr):
                overlaps.append(corr[idx] * (2.0 / n_points))
            else:
                # Handle boundary case
                overlaps.append(0)
        
        # Find worst-case shift
        max_idx = np.argmax(overlaps)
        k_max = shift_samples[max_idx]
        
        # Build convex subproblem at worst shift
        # We're minimizing sum(h_new[i] * (1 - h[i + k_max]))
        # Note: we implement the constraint that rows of the matrix 
        # are h[i + k_max] + h[i] * (1 - h[i + k_max]) to maximize
        # our chance of getting good values out
        
        # Instead, we will optimize directly using the fact:
        # min_h sum h[i] (1 - h[i+k]) s.t. sum h[i] = 1
        # Using the fixed k, it's a convex problem.
        
        # Create subproblem: minimize sum(h[i](1 - h[i+k]))
        # where the shifted expression is to be interpreted as cyclic
        
        # Optimize directly by treating as a convex optimization
        # The objective is approximately linear in h when k is fixed
        
        # Prepare matrix for quadratic form approximation or solve directly
        
        # Alternative: form a new problem with matrix representation, 
        # maybe even use Lagrange multipliers or the convexity directly
        
        # Use LBFGS with bounds on h to solve this
        def obj(h_new):
            # Need to compute sum h_new[i] * (1 - h[i + k_max])
            # Shift the indices properly
            shift = int(k_max / (2.0 / n_points))
            h_shifted = np.roll(h, shift)
            return np.sum(h_new * (1 - h_shifted))
            
        def constraint_sum(h_new):
            return np.sum(h_new) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            # Solve with L-BFGS-B or trust-constr
            res = minimize(obj, h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50})
            if res.success:
                h_new = res.x
                h = h_new
            else:
                # Fallback: small change
                pass
        except:
            pass
        
        # Recompute C5
        c5 = evaluate_erdos_solution(h, 2.0 / n_points)
        
        if abs(c5 - last_c5) < tol:
            break
        last_c5 = c5
        
        if c5 < 0.3807:
            break
            
    # Final polishing with SLSQP
    try:
        def obj_final(h_new):
            return evaluate_erdos_solution(h_new, 2.0 / n_points)
        def constraint_sum_final(h_new):
            return np.sum(h_new) - n_points / 2.0
            
        cons_sum_final = {'type': 'eq', 'fun': constraint_sum_final}
        bounds_final = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj_final, h, method='SLSQP', 
                       bounds=bounds_final, constraints=cons_sum_final, 
                       options={'maxiter': 50, 'ftol': 1e-9})
        if res.success:
            h = res.x
            c5 = obj_final(h)
    except:
        pass
        
    return (list(h), c5, n_points)
