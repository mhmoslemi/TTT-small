import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)

        k_samples = np.random.randint(0, n_points, 50)

        for k in k_samples:
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points

                worst_k = k
                
        # Compute exact gradient for the worst k
        grad = np.zeros(n_points)

            j = (i + worst_k) % n_points
            grad[i] += 1 - h[j]

        if abs(c5_prev - best_c5) < 1e-8:
