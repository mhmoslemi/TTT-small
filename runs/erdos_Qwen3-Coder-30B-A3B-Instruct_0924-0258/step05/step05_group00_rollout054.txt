import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive Stochastic Descent
    n_iter = 500
    step_size = 0.1
    decay = 0.99
    
    for i in range(n_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb h
        noise = np.random.normal(0, step_size, n_points)
        h_new = best_h + noise
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Decay step size
        step_size *= decay
    
    # Quadratic Constraint Programming Refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add overlap constraints for critical lags
        k_vals = np.linspace(0.01, 1.99, min(20, n_points//2))
        for k in k_vals:
            if k < 1:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-int(k/dx)], 1 - h_opt[int(k/dx):]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[int(k/dx)-n_points+1:], 1 - h_opt[:n_points-(int(k/dx)-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final refinement
    try:
        h_final = np.clip(best_h, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
