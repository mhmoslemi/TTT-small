import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # SCP parameters
    max_iter = 500
    mu = 1e-3
    C5_nih = 0.4
    batch_size = 5
    
    def create_constraints(h, C5_estimate):
        """Create QCQP constraints based on current h and estimate"""
        constraints = []
        # For every possible shift k
        for k in range(n_points):
            # Build the shifted version of h
            h_shifted = np.roll(h, k)
            # The overlap integral with shift k
            overlap = np.sum(h * (1 - h_shifted)) * dx
            # Constraint that overlap should not exceed C5_estimate
            constraints.append(overlap <= C5_estimate)
        # Sum constraint
        constraints.append(cp.sum(h) == n_points / 2)
        return constraints
    
    # Main SCP loop
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Get current overlap values
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_overlap = np.max(corr) * dx
        
        current_c5 = max_overlap
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        
        if best_c5 < 0.3807:
            break
            
        # Formulate convex QP based on current solution
        h_var = cp.Variable(n_points)
        
        # Objective: minimize C5 (this is an estimate)
        # We'll use a constraint-based formulation where we add penalty terms
        # Build constraints 
        constraints = []
        
        # Add constraints for selected shifts to avoid complexity
        shifts = np.random.choice(n_points, min(batch_size, n_points), replace=False)
        for k in shifts:
            # Slide h by k positions using cyclic shift
            h_shifted = np.roll(best_h, k)
            overlap = np.sum(best_h * (1 - h_shifted)) * dx
            if overlap > best_c5 * 1.01:
                constraints.append(cp.sum(h_var * (1 - cp.roll(h_var, k))) * dx <= best_c5)
        
        # Add sum constraint
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Add bounds
        for i in range(n_points):
            constraints.append(h_var[i] >= 0)
            constraints.append(h_var[i] <= 1)
        
        # Objective function
        objective = cp.Minimize(cp.sum(h_var * (1 - cp.roll(h_var, 1))) * dx)
        
        # Solve QP
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.OSQP, warm_start=True, max_iter=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h_new = np.clip(h_new, 0, 1)
                    # Renormalize
                    h_new = h_new * (n_points / 2) / np.sum(h_new) 
                    new_c5 = evaluate_erdos_solution(h_new, dx)
                    if new_c5 < best_c5:
                        best_h = h_new
                        best_c5 = new_c5
        except:
            pass
        
        # Local refinement using SLSQP
        if iteration % 5 == 0:
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            def con_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons = {'type': 'eq', 'fun': con_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = obj(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    # Final local refinement with fine tolerance
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
