def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points

    best_c5 = evaluate_erdos_solution(h_start, dx)

            # interpolate to finer resolution
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            current_h = np.interp(x_new, x_orig, current_h)

            dx = 2.0 / n_points

            current_h = current_h[:scale]

                break

            corr = np.correlate(current_h, 1 - current_h, mode='full')

            eps = 1e-4

            new_h = new_h * (n_points/2) / np.sum(new_h)

                best_h = new_h.copy()

            if new_c5 < loss_prev:
                lr = min(lr * 1.1, 0.1)

            current_h = new_h

        except:
            pass
    
    return (list(best_h), best_c5, n_points)
