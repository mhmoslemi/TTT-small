import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # If no parent construction, use a simple construction
        n_points = 35
        h = np.ones(n_points) * 0.5
        
    # Initialize with a normalized version that meets constraints
    h = h * (n_points / 2) / np.sum(h)
    
    # Convex-Concave Procedure (CCP)
    max_iter = 500
    tol = 1e-6
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    best_h = h.copy()
    
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find worst-case shift by evaluating all possible shifts
        # Use a coarse grid to find the worst shift
        shifts = np.linspace(0, 2, 100)
        worst_shift = 0
        max_corr = -np.inf
        
        for shift in shifts:
            # Compute correlation at this shift
            corr = np.correlate(h, 1 - np.roll(h, int(shift / (2.0 / n_points))), mode='full')
            # Get the correlation for the given shift
            shift_idx = int(shift / (2.0 / n_points)) + len(h) - 1
            if 0 <= shift_idx < len(corr):
                corr_val = corr[shift_idx]
                if corr_val > max_corr:
                    max_corr = corr_val
                    worst_shift = shift
        
        # Now solve the convex subproblem: minimize sum h[i] * (1 - h[i + k])
        # This becomes a linear optimization with constraints
        # We solve a relaxed version using cvxpy
        h_new = cp.Variable(n_points)
        objective = cp.Minimize(h_new.T @ (1 - np.roll(h, int(worst_shift / (2.0 / n_points)))))
        constraints = [
            cp.sum(h_new) == n_points / 2,
            h_new >= 0,
            h_new <= 1
        ]
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h_new_val = h_new.value
            if h_new_val is not None:
                h = np.array(h_new_val)
                # Project to ensure constraints
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                c5 = evaluate_erdos_solution(h, 2.0 / n_points)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                if abs(best_c5 - c5) < tol:
                    break
        except:
            pass
    
    # Final polishing
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
