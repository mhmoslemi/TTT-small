import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA Parameters
    population_size = 50
    max_generations = 1000
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    def normalize_h(h):
        s = np.sum(h)
        if s > 0:
            h = h * (n_points / 2) / s
        return np.clip(h, 0, 1)
    
    def create_individual():
        h = np.random.rand(n_points)
        return normalize_h(h)
    
    def evaluate_individual(h):
        return evaluate_erdos_solution(h, dx)
    
    population = [create_individual() for _ in range(population_size)]
    fitness = [evaluate_individual(ind) for ind in population]
    
    best_in_run = best_c5
    no_improvement_count = 0
    max_no_improvement = 50
    
    # GA main loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = [evaluate_individual(ind) for ind in population]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
        # Selection with tournament
        def tournament_selection():
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            tournament.sort(key=lambda x: x[1])
            return tournament[0][0]
        
        # Create new population
        new_population = []
        
        # Elite preservation
        elite_indices = np.argsort(fitness)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            if random.random() < crossover_rate:
                parent1 = tournament_selection()
                parent2 = tournament_selection()
                
                # Single-point crossover
                crossover_point = random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutation
                for child in [child1, child2]:
                    for i in range(n_points):
                        if random.random() < mutation_rate:
                            child[i] += np.random.normal(0, 0.02)
                    
                    # Repair
                    child = normalize_h(child)
                    new_population.append(child)
                    
                    if len(new_population) >= population_size:
                        break
            else:
                # Just mutate a random individual
                ind = tournament_selection()
                child = ind.copy()
                for i in range(n_points):
                    if random.random() < mutation_rate:
                        child[i] += np.random.normal(0, 0.02)
                child = normalize_h(child)
                new_population.append(child)
        
        population = new_population[:population_size]
    
    # Resolution adaptation
    if time() - start_time < budget_s * 0.8 and best_c5 > 0.3808:
        try:
            n_points = 40
            dx = 2.0 / n_points
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, len(best_h)), 
                               best_h)
            best_h = normalize_h(h_interp)
            best_c5 = evaluate_erdos_solution(best_h, dx)
            
            # Second round GA with smaller population
            population_size = 20
            max_generations = 300
            population = [best_h.copy()]
            population.extend([create_individual() for _ in range(population_size - 1)])
            fitness = [evaluate_individual(ind) for ind in population]
            
            # GA again
            for generation in range(max_generations):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                fitness = [evaluate_individual(ind) for ind in population]
                current_best_idx = np.argmin(fitness)
                current_best_c5 = fitness[current_best_idx]
                if current_best_c5 < best_c5:
                    best_c5 = current_best_c5
                    best_h = population[current_best_idx].copy()
                
                # Selection and reproduction
                def tournament_selection():
                    tournament = random.sample(list(zip(population, fitness)), tournament_size)
                    tournament.sort(key=lambda x: x[1])
                    return tournament[0][0]
                
                new_population = [population[current_best_idx].copy()]
                
                while len(new_population) < population_size:
                    if random.random() < crossover_rate:
                        parent1 = tournament_selection()
                        parent2 = tournament_selection()
                        
                        crossover_point = random.randint(1, n_points - 1)
                        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                        
                        for child in [child1, child2]:
                            for i in range(n_points):
                                if random.random() < mutation_rate:
                                    child[i] += np.random.normal(0, 0.02)
                            child = normalize_h(child)
                            new_population.append(child)
                            
                            if len(new_population) >= population_size:
                                break
                    else:
                        ind = tournament_selection()
                        child = ind.copy()
                        for i in range(n_points):
                            if random.random() < mutation_rate:
                                child[i] += np.random.normal(0, 0.02)
                        child = normalize_h(child)
                        new_population.append(child)
                
                population = new_population[:population_size]
        except Exception:
            pass
    
    # Local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        res = minimize(objective, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons_sum, 
                      options={'maxiter': 200, 'ftol': 1e-8})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
