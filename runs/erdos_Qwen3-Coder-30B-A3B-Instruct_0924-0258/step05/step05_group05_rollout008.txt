import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a smooth ramp to start with
        h_start = np.linspace(0, 1, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population sizes and parameters
    population_size = 30
    mutation_factor = 0.8
    crossover_prob = 0.8
    max_generations = 600
    refinement_interval = 10
    
    # Initialize population
    population = [h_start.copy()]
    fitness = [best_c5]
    
    # Generate diverse initial population
    for i in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
        
    # Diff-Evolution Loop
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort based on fitness and keep top 50%
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size//2]
        top_pop = [population[i] for i in top_indices]
        top_fit = [fitness[i] for i in top_indices]
        
        # Update best if found
        current_best = min(top_fit)
        if current_best < best_c5:
            best_h = top_pop[0].copy()
            best_c5 = current_best
            if best_c5 < 0.3807:
                break
        
        # Create new population
        new_pop = top_pop.copy()
        
        # Differential evolution
        for _ in range(population_size - len(top_pop)):
            # Select three distinct individuals
            indices = random.sample(range(len(top_pop)), 3)
            a, b, c = top_pop[indices[0]], top_pop[indices[1]], top_pop[indices[2]]
            
            # Mutation
            mutant = a + mutation_factor * (b - c)
            
            # Crossover
            trial = np.where(np.random.random(n_points) < crossover_prob, mutant, 
                             population[random.choice(range(len(population)))])
            
            # Repair to bounds and sum constraint
            trial = np.clip(trial, 0, 1)
            trial = trial * (n_points/2) / np.sum(trial)
            
            new_pop.append(trial)
        
        # Evaluate new population
        fitness = [evaluate_erdos_solution(h, dx) for h in new_pop]
        
        # Local refinement
        if gen % refinement_interval == 0 and len(new_pop) > 0:
            for i in range(min(5, len(new_pop))):
                h_init = new_pop[i]
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_init, method='L-BFGS-B', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-8})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < fitness[i]:
                            new_pop[i] = h_refined
                            fitness[i] = c5_refined
                except:
                    pass
    
    # Final refinement of best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
