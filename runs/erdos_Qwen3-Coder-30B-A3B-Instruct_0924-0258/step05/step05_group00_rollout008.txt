import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm
    population_size = 50
    max_generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    
    # Population initialization
    def init_population(size, n):
        if 'initial_h_values' in globals():
            pop = [h_start.copy()]
            pop.extend([np.random.random(n) for _ in range(size - 1)])
        else:
            pop = [np.random.random(n) for _ in range(size)]
        return pop

    def normalize(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def select_tournament(pop, fitnesses, k):
        indices = np.random.choice(len(pop), k)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        return pop[best_idx]
    
    def crossover(p1, p2):
        if np.random.random() > crossover_rate:
            return p1.copy(), p2.copy()
        point = np.random.randint(1, len(p1))
        child1 = np.concatenate([p1[:point], p2[point:]])
        child2 = np.concatenate([p2[:point], p1[point:]])
        return child1, child2

    def mutate(h):
        h = h.copy()
        indices = np.random.choice(len(h), int(len(h) * mutation_rate), replace=False)
        h[indices] += np.random.normal(0, 0.02, len(indices))
        return normalize(h)
    
    population = init_population(population_size, n_points)
    for g in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(normalize(ind), dx) for ind in population]
        
        # Track best
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitnesses[best_idx]
        
        # Selection + crossover + mutation
        new_population = []
        for _ in range(population_size // 2):
            parent1 = select_tournament(population, fitnesses, tournament_size)
            parent2 = select_tournament(population, fitnesses, tournament_size)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        population = new_population
    
    # Resolution adaptation: increase if not good enough
    if best_c5 > 0.3808 and n_points < 40:
        n_points = 40

                break
            
            fitnesses = [evaluate_erdos_solution(normalize(ind), dx) for ind in population]
            
            # Track best
            best_idx = np.argmin(fitnesses)
            if fitnesses[best_idx] < best_c5:
                best_h = population[best_idx].copy()
                best_c5 = fitnesses[best_idx]
            
            # Selection + crossover + mutation
            new_population = []
            for _ in range(pop_size // 2):
                parent1 = select_tournament(population, fitnesses, tournament_size)
                parent2 = select_tournament(population, fitnesses, tournament_size)
                child1, child2 = crossover(parent1, parent2)
                child1 = mutate(child1)
                child2 = mutate(child2)
                new_population.extend([child1, child2])
            
            population = new_population
    
    # Final local refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
