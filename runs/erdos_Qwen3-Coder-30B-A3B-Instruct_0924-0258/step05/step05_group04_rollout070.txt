import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)

    # Initialize Lagrange multipliers and parameters
    lambda_vars = np.ones(2 * n_points - 1) * 0.1
    mu = 1.0
    rho = 1e3
    max_iter = 500

    # Precompute correlation matrices
    A_matrices = []
    for k in range(1, 2 * n_points - 1):
        A = np.zeros((n_points, n_points))
        if k < n_points:
            A[np.arange(k, n_points), np.arange(0, n_points - k)] = 1
        else:
            A[np.arange(0, n_points - (k - n_points + 1)), np.arange(k - n_points + 1, n_points)] = 1
        A_matrices.append(A)

    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break

        # Solve convex subproblem using CVXPY
        h_var = cp.Variable(n_points)
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]

        # Form the Lagrangian
        lagrangian = 0
        for k, A in enumerate(A_matrices):
            # Access the overlap constraint
            overlap_val = cp.sum(cp.multiply(h_var, A @ (1 - h_var)))
            lagrangian += lambda_vars[k] * (overlap_val - best_c5 * dx)

        # Add the integral constraint penalty
        integral_penalty = mu * (cp.sum(h_var) - n_points / 2)
        lagrangian += integral_penalty

        # Minimize the Lagrangian
        objective = cp.Minimize(lagrangian)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.OSQP)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if np.all(h_new >= 0) and np.all(h_new <= 1):
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                else:
                    h_new = h_start.copy()
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
            else:
                h_new = h_start.copy()
                h_new = h_new * (n_points / 2) / np.sum(h_new)
        except:
            h_new = h_start.copy()
            h_new = h_new * (n_points / 2) / np.sum(h_new)

        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        # Update Lagrange multipliers
        for k, A in enumerate(A_matrices):
            overlap_val = np.sum(h_new * (A @ (1 - h_new)))
            lambda_vars[k] = max(0, lambda_vars[k] + rho * (overlap_val - best_c5 * dx))

        # Update mu
        integral_violation = np.abs(np.sum(h_new) - n_points / 2)
        mu = max(0, mu + rho * (np.sum(h_new) - n_points / 2))

        # Adjust rho based on constraint violations
        if integral_violation > 1e-4:
            rho *= 2
        else:
            rho *= 0.5

        # Projection step
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)

        # Early stopping
        if best_c5 < 0.3807:
            break

    # Final local refinement with SLSQP
    if time() - start_time < max_time - 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                           constraints=cons_sum, options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass

    return (list(best_h), best_c5, n_points)
