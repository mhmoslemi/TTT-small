import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Lagrangian relaxation approach with iterative convex optimization
    max_iter = 500
    rho = 1e3
    lambda_init = 0.1
    
    # Precompute shift matrices
    A_list = []
    for k in range(2 * n_points - 1):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = i + k - (n_points - 1)
            if 0 <= j < n_points:
                A[i, j] = 1
        A_list.append(A)
    
    # Iterative optimization loop
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Update multipliers
        current_c5 = evaluate_erdos_solution(best_h, dx)
        current_h = best_h.copy()
        
        # Solving convex problem using CVXPY
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(h_var - current_h))
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Add penalty for overlap constraints
        overlap_constraints = []
        overlaps = []
        for k, A in enumerate(A_list):
            overlap = cp.sum(cp.multiply(h_var, A @ (1 - h_var))) * dx
            overlaps.append(overlap)
            # Add slack constraint with Lagrangian terms, but keep it in form:
            # overlap <= current_c5 * dx + slack
            # But for practical solving, we tighten the constraint
            overlap_constraints.append(overlap <= current_c5 * dx + 1e-6)
            
        constraints.extend(overlap_constraints)
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.OSQP, verbose=False, 
                       eps_abs=1e-6, eps_rel=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                new_c5 = evaluate_erdos_solution(h_new, dx)
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
        except:
            pass

        # Local refinement
        try:
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            def con(h):
                return np.sum(h) - n_points / 2.0
            
            cons = {'type': 'eq', 'fun': con}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 50, 'ftol': 1e-7})
            if res.success:
                h_refined = res.x
                c5_refined = obj(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
            
        # Early stopping
        if best_c5 < 0.3807:
            break
    
    # Final local refinement with tight tolerance
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 20:
        try:
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            def con(h):
                return np.sum(h) - n_points / 2.0
            
            cons = {'type': 'eq', 'fun': con}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = obj(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
