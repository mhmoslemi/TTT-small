import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try to find a good integer block-based solution
    def construct_block_solution(n_points, target_sum):
        # Start with blocks of roughly sqrt(n_points) size
        M = max(2, int(np.sqrt(n_points)))
        block_size = n_points // M
        
        if block_size * M != n_points:
            # Adjust to make it work
            block_size = n_points // M
            M = n_points // block_size
            
        # Ensure block_size is integer
        if block_size * M != n_points:
            block_size = n_points // M
            M = n_points // block_size
            
        h = np.zeros(n_points)
        
        # Create block ramp-up pattern
        blocks_ones = int(target_sum / block_size)
        if blocks_ones > M:
            blocks_ones = M
            
        # Ensure the total sum matches
        if blocks_ones * block_size > target_sum:
            blocks_ones = int(target_sum / block_size)
        elif blocks_ones * block_size < target_sum:
            blocks_ones = min(M, blocks_ones + 1)
            
        # Set the blocks
        for i in range(blocks_ones):
            start = i * block_size
            end = start + block_size
            h[start:end] = 1.0
            
        # Scale to correct sum
        actual_sum = np.sum(h)
        if actual_sum > 0:
            h = h * target_sum / actual_sum
            
        return h
    
    # Try block-based solutions
    target_sum = n_points / 2.0
    h_blocks = construct_block_solution(n_points, target_sum)

    return (list(best_h), best_c5, n_points)
