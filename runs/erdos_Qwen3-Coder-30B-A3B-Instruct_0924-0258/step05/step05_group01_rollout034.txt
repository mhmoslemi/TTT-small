import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a triangular construction as initial guess
        x = np.linspace(0, 2, 35)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start / np.sum(h_start) * 17.5  # Scale to sum to 17.5 (n/2)
        n_points = 35
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution refinement with Simulated Annealing
    scales = [35, 50, 75, 100]
    current_scale_idx = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale_idx < len(scales):
        scale = scales[current_scale_idx]
        
        # Interpolate to target scale
        if scale > n_points:
            h_new = np.interp(np.linspace(0, 2, scale), np.linspace(0, 2, n_points), best_h)
            n_points = scale
        else:
            h_new = best_h.copy()
            n_points = scale
        
        dx = 2.0 / n_points
        
        # Simulated Annealing
        T0 = 0.1
        alpha = 0.99
        max_iter = int(1000 * (scale / 35.0))  # Scale iterations with resolution
        
        h_current = h_new.copy()
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        T = T0
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate neighbor by slight perturbation
            h_proposed = h_current.copy()
            # Flip some values with 10% probability
            flip_mask = np.random.random(n_points) < 0.1
            if np.any(flip_mask):
                # Apply normally distributed perturbation to selected points
                perturb = np.random.normal(0, 0.05, n_points)
                h_proposed += perturb * flip_mask
                
                # Clip to [0,1]
                h_proposed = np.clip(h_proposed, 0, 1)
                
                # Rescale to exact sum
                h_proposed = h_proposed * (n_points / 2) / np.sum(h_proposed)
                
                c5_proposed = evaluate_erdos_solution(h_proposed, dx)
                
                # Accept or reject based on SA probability
                if c5_proposed < c5_current or np.random.random() < np.exp(-(c5_proposed - c5_current) / T):
                    h_current = h_proposed
                    c5_current = c5_proposed
                    
            T *= alpha
            
        # Update best solution if improved
        if c5_current < best_c5:
            best_h = h_current
            best_c5 = c5_current
            
        current_scale_idx += 1
    
    # Final refinement using L-BFGS if we didn't exhaust time
    if time() - start_time < budget_s * 0.9:
        # Try local optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        c5_refined = objective(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    return (list(best_h), best_c5, n_points)
