import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with available construction or default
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a directed search to explore better regions
    # Start from a coarse level and refine
    resolutions = [35, 40, 50, 60, 70]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Handle resolution changes
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Initially try a few gradient-free methods to escape local optima
        max_iterations = 100
        iteration = 0
        improvement_counter = 0
        
        while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
            # Joint optimization using multiple approaches
            # Approach 1: random perturbations
            h_trial = h_current + np.random.normal(0, 0.02, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points/2) / np.sum(h_trial)
            
            new_c5 = evaluate_erdos_solution(h_trial, dx)
            
            if new_c5 < best_c5:
                best_h = h_trial
                best_c5 = new_c5
                improvement_counter = 0
                h_current = h_trial.copy()
            else:
                improvement_counter += 1

            # Approach 2: deflation of overlap terms
            # Check if we can locally improve to reduce overlap
            h_updated = h_current.copy()
            for _ in range(3):
                # Simple neighborhood descent
                local_c5 = evaluate_erdos_solution(h_updated, dx)
                if time() - start_time > budget_s * 0.95:
                    break
                improvement_found = False
                for i in range(n_points):
                    for delta in [-0.02, 0.02]:
                        h_test = h_updated.copy()
                        h_test[i] += delta
                        h_test[i] = np.clip(h_test[i], 0, 1)
                        # Renormalize
                        h_test = h_test * (n_points/2) / np.sum(h_test)
                        test_c5 = evaluate_erdos_solution(h_test, dx)
                        if test_c5 < local_c5:
                            h_updated = h_test
                            local_c5 = test_c5
                            improvement_found = True
                            break
                    if improvement_found:
                        break
                if not improvement_found:
                    break
            if evaluate_erdos_solution(h_updated, dx) < best_c5:
                best_h = h_updated
                best_c5 = evaluate_erdos_solution(h_updated, dx)
                
            iteration += 1

    # Final QCQP approach for tight bound validation
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Approximate constraints for performance: use a subset
        # of possible shifts
        stride = max(1, n_points // 15)
        for k in range(stride, n_points, stride):
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_sol = h_opt.value
            h_feasible = np.clip(h_sol, 0, 1)
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final safe refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
        
    return (list(best_h), best_c5, n_points)
