import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a triangular function as starting point
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution with simulated annealing
    resolutions = [35, 50, 75, 100]
    current_res = 0
    
    while time() - start_time < budget_s * 0.95 and current_res < len(resolutions):
        res = resolutions[current_res]
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), best_h)
            h_current = h_interp
            n_points = res
            dx = 2.0 / n_points
        else:
            h_current = best_h
            n_points = res
            dx = 2.0 / n_points
            
        # Simulated Annealing
        T = 0.1
        alpha = 0.99
        max_iter = int(1000 * (1 + current_res))  # More iterations at lower resolutions
        
        h_best = h_current.copy()
        c5_best = evaluate_erdos_solution(h_current, dx)
        
        h_temp = h_current.copy()
        c5_temp = c5_best
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Random perturbation
            h_new = h_temp.copy()
            flip_prob = 0.1
            for j in range(n_points):
                if np.random.random() < flip_prob:
                    # Perturb by small amount
                    delta = np.random.normal(0, 0.05)
                    h_new[j] += delta
                    # Clip to [0,1]
                    h_new[j] = np.clip(h_new[j], 0, 1)
            
            # Ensure sum constraint
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            if c5_new < c5_temp or np.random.random() < math.exp(-(c5_new - c5_temp) / T):
                h_temp = h_new
                c5_temp = c5_new
            
            # Update best
            if c5_temp < c5_best:
                h_best = h_temp
                c5_best = c5_temp
                
            # Cooling
            T = alpha * T
            
        # Update global best
        if c5_best < best_c5:
            best_h = h_best.copy()
            best_c5 = c5_best
            
        current_res += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, len(best_h))
