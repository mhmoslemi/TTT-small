import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # SCP parameters
    max_iter = 500
    mu = 1e-3
    C5_nih = 0.4
    penalty_factor = 2.0
    
    # Create all shifted versions of h
    def get_shifted_h(h, shift):
        return np.roll(h, shift)
    
    def compute_all_overlaps(h):
        overlaps = []
        for k in range(n_points):
            shifted_h = get_shifted_h(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            overlaps.append(overlap)
        return np.array(overlaps)
    
    def solve_subproblem(h_current, C5_bound):
        # Create convex approximation using current h estimates
        h = cp.Variable(n_points)
        constraints = []
        obj = cp.Minimize(C5_bound)
        
        # Add constraint that all overlaps are bounded by C5_bound
        for k in range(n_points):
            shifted_h = get_shifted_h(h_current, k)
            # Linearized constraint
            linearized_overlap = cp.sum(cp.multiply(h, 1 - shifted_h)) * dx
            constraints.append(linearized_overlap <= C5_bound)
            
        # Add integral constraint
        constraints.append(cp.sum(h) == n_points / 2.0)
        # Add bounds
        constraints.extend([h >= 0, h <= 1])
        
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.OSQP, verbose=False, 
                      eps_abs=1e-6, eps_rel=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h.value
        except:
            pass
        return h_current.copy()
    
    # Main SCP loop
    gen = 0
    while time() - start_time < max_time * 0.95 and gen < max_iter:
        gen += 1
        overlaps = compute_all_overlaps(best_h)
        current_max_overlap = np.max(overlaps)
        
        if current_max_overlap < best_c5:
            best_c5 = current_max_overlap
            best_h = best_h.copy()
            
        if current_max_overlap <= C5_nih + 1e-4:
            # Converged
            break
        
        # Update C5_nih
        C5_nih = current_max_overlap
        
        # If constraint violation detected, increase penalty
        if current_max_overlap > C5_nih:
            mu *= penalty_factor
        
        # Solve subproblem with current best
        refined_h = solve_subproblem(best_h, C5_nih)
        
        # Project refined solution to feasible region
        refined_h = np.clip(refined_h, 0, 1)
        refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
        
        # Update best solution
        if evaluate_erdos_solution(refined_h, dx) < best_c5:
            best_h = refined_h.copy()
            best_c5 = evaluate_erdos_solution(refined_h, dx)
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    # Final refinement
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
