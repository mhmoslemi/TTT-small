import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Population-based optimization using evolutionary approach
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    pop_size = 50
    elitism = int(0.3 * pop_size)
    mutation_rate = 0.1
    max_generations = 1000
    
    def normalize_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    # Generate initial population
    population = [normalize_h(np.random.rand(n_points)) for _ in range(pop_size)]
    population[0] = best_h.copy()  # Include initial solution
    
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = [fitness(h) for h in population]
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        if fitnesses[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitnesses[0]
        
        # Select elite
        elite = population[:elitism]
        
        # Generate offspring
        offspring = []
        while len(offspring) < pop_size - elitism:
            # Crossover
            parent1, parent2 = random.sample(elite, 2)
            alpha = random.uniform(0.4, 0.6)
            child = alpha * parent1 + (1 - alpha) * parent2
            child = normalize_h(child)
            
            # Mutation
            if random.random() < mutation_rate:
                for i in range(n_points):
                    if random.random() < 0.1:
                        child[i] += np.random.normal(0, 0.02)
                child = normalize_h(child)
            
            offspring.append(child)
        
        # Create new population
        population = elite + offspring
        
        # Gradual local refinement
        if gen % 5 == 0 and gen > 0:
            for i in range(min(10, len(population))):
                h_candidate = population[i].copy()
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(obj, h_candidate, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < fitnesses[i]:
                        population[i] = h_refined
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                            
    # Final refinement of best solution
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
