import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.linspace(0, 1, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential evolution parameters
    population_size = 30
    mutation_factor = 0.8
    crossover_prob = 0.8
    max_generations = 400
    local_refinement_freq = 10
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evolution loop
    stagnation_counter = 0
    current_best = best_c5
    
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
            if best_c5 < 0.3807:
                break
        
        # Check stagnation
        if fitness[min_fitness_idx] < current_best:
            current_best = fitness[min_fitness_idx]
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        if stagnation_counter > 20:
            # Increase mutation factor and decrease crossover
            mutation_factor = min(1.5, mutation_factor + 0.1)
            crossover_prob = max(0.5, crossover_prob - 0.05)
        else:
            # Gradually restore parameters
            mutation_factor = max(0.5, mutation_factor - 0.02)
            crossover_prob = min(0.9, crossover_prob + 0.02)
        
        # Create new population
        new_population = [population[min_fitness_idx]]  # Elitism
        
        for _ in range(population_size - 1):
            # Tournament selection
            idx1, idx2, idx3 = random.sample(range(population_size), 3)
            a, b, c = population[idx1], population[idx2], population[idx3]
            
            # Differential mutation
            mutant = a + mutation_factor * (b - c)
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            if random.random() < crossover_prob:
                crossover_mask = np.random.random(n_points) < 0.2
                mutant = np.where(crossover_mask, population[idx1], mutant)
            
            # Repair
            mutant = mutant * (n_points/2) / np.sum(mutant)
            
            new_population.append(mutant)
        
        population = new_population
        
        # Local refinement
        if gen % local_refinement_freq == 0:
            for i in range(min(5, len(population))):
                h_init = population[i]
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    # Add small regularization to avoid local minima
                    def regularized_objective(h):
                        return objective(h) + 0.001 * np.sum(np.diff(h)**2)
                    
                    # Use L-BFGS-B for better convergence
                    res = minimize(regularized_objective, h_init, 
                                   method='L-BFGS-B', bounds=bounds, 
                                   options={'maxiter': 100, 'ftol': 1e-8})
                    
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < fitness[i]:
                            population[i] = h_refined
                except:
                    pass
    
    # Final refinement if time permits
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
