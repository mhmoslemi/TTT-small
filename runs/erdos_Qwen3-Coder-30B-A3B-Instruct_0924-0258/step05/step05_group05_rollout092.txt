import numpy as np
from scipy.signal import correlate
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    if best_c5 <= 0.3807:
        return (list(best_h), best_c5, n_points)
    
    # Block-based optimization approach
    # Define block structure
    M = int(np.sqrt(n_points))  # number of blocks
    if M < 2:
        M = 2
    block_size = n_points // M
    if block_size * M != n_points:
        block_size = n_points // M
        M = n_points // block_size
    
    # Initialize blocks
    block_config = np.zeros(M, dtype=int)
    num_active_blocks = n_points // (2 * block_size)
    if num_active_blocks > 0:
        active_indices = np.random.choice(M, num_active_blocks, replace=False)
        block_config[active_indices] = 1
    
    def config_to_h(block_config):
        """Convert block configuration to h vector"""
        h = np.zeros(n_points)
        for i in range(M):
            start_idx = i * block_size
            end_idx = start_idx + block_size
            h[start_idx:end_idx] = block_config[i]
        return h
    
    def compute_config_value(block_config):
        """Compute overlap for given block config"""
        h = config_to_h(block_config)
        return evaluate_erdos_solution(h, dx)
    
    def perturb_block_config(block_config):
        """Perturb block configuration slightly"""
        new_config = block_config.copy()
        # Randomly flip one block
        flip_idx = random.randint(0, M - 1)
        new_config[flip_idx] = 1 - new_config[flip_idx]
        
        # Ensure sum constraint
        current_sum = np.sum(new_config)
        target_sum = n_points // (2 * block_size)
        if current_sum < target_sum:
            # Turn on more blocks
            off_blocks = np.where(new_config == 0)[0]
            if len(off_blocks) > 0:
                to_turn_on = target_sum - current_sum
                added = min(to_turn_on, len(off_blocks))
                on_indices = np.random.choice(off_blocks, added, replace=False)
                new_config[on_indices] = 1
        elif current_sum > target_sum:
            # Turn off some blocks
            on_blocks = np.where(new_config == 1)[0]
            if len(on_blocks) > 0:
                to_turn_off = current_sum - target_sum
                added = min(to_turn_off, len(on_blocks))
                off_indices = np.random.choice(on_blocks, added, replace=False)
                new_config[off_indices] = 0
                
        return new_config
    
    # Optimization main loop
    iteration = 0
    max_iterations = 1000
    stagnation_count = 0
    max_stagnation = 100
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        iteration += 1
        current_c5 = compute_config_value(block_config)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = config_to_h(block_config)
            if best_c5 <= 0.3807:
                break
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Perturb the configuration
        new_config = perturb_block_config(block_config)
        new_c5 = compute_config_value(new_config)
        
        # Accept if better
        if new_c5 < current_c5:
            block_config = new_config
        # Occasionally accept worse solutions with probability
        elif random.random() < 0.05:
            block_config = new_config
            
        # Break early if stagnating too much
        if stagnation_count > max_stagnation:
            break
            
    # Apply local refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        # Refine using gradient-free approach with several restarts
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            # Random restart
            temp_h = config_to_h(block_config)
            temp_h += np.random.normal(0, 0.01, n_points)
            temp_h = np.clip(temp_h, 0, 1)
            temp_h = temp_h * (n_points/2) / np.sum(temp_h)
            temp_c5 = evaluate_erdos_solution(temp_h, dx)
            if temp_c5 < best_c5:
                best_h = temp_h
                best_c5 = temp_c5
    
    return (list(best_h), best_c5, n_points)
