import numpy as np
from scipy.optimize import minimize
from time import time
import hashlib

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = min(50, max(5, n_points // 4))
    max_stagnation = 100
    max_iter = 1000
    mutation_scale = 0.04
    mutate_ratio = 0.3
    
    # Initialize
    current_h = h_start.copy()
    current_c5 = best_c5
    stagnation_count = 0
    
    def hash_state(h):
        # Create a hashable representation
        sorted_h = np.sort(h)
        return hashlib.md5(sorted_h.tobytes()).hexdigest()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb current solution
        num_mutate = int(n_points * mutate_ratio)
        mutate_indices = np.random.choice(n_points, num_mutate, replace=False)
        new_h = current_h.copy()
        
        for i in mutate_indices:
            # Add Gaussian noise
            new_h[i] += np.random.normal(0, mutation_scale)
        
        # Clip to [0,1]
        new_h = np.clip(new_h, 0, 1)
        
        # Scale to correct sum
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Check tabu
        state_hash = hash_state(new_h)
        is_tabu = state_hash in tabu_list
        
        # Accept if better or not tabu
        if new_c5 < current_c5 or not is_tabu:
            current_h = new_h
            current_c5 = new_c5
            
            # Update best
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                if best_c5 < 0.3807:
                    break
                    
            # Add to tabu list
            tabu_list.append(state_hash)
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
            
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Adjust mutation scale based on stagnation
        if stagnation_count > max_stagnation:
            mutation_scale = min(0.5, mutation_scale * 1.2)
            stagnation_count = 0
        else:
            mutation_scale = max(0.005, mutation_scale * 0.99)
            
        # Occasionally apply local refinement
        if iteration % 10 == 0 and iteration > 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, current_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    refined_h = res.x
                    refined_c5 = objective(refined_h)
                    if refined_c5 < current_c5:
                        current_h = refined_h
                        current_c5 = refined_c5
                        if refined_c5 < best_c5:
                            best_h = refined_h
                            best_c5 = refined_c5
                            if best_c5 < 0.3807:
                                break
            except:
                pass
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
