import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ASD optimization with adaptive step size
    max_iters = 500  # number of iterations for ASD
    step_size = 0.1
    decay_rate = 0.995
    
    for iter_count in range(max_iters):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Perturb in each dimension
        h_perturbed = best_h + np.random.normal(0, step_size, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Re-normalize to ensure sum equals n_points / 2
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Evaluate the overlap
        c5_new = evaluate_erdos_solution(h_perturbed, dx)
        
        # Accept improvement
        if c5_new < best_c5:
            best_h = h_perturbed
            best_c5 = c5_new
            
        # Apply decay
        step_size *= decay_rate
    
    # Convex relaxation with QCP for final refinement
    try:
        n = n_points
        h_opt = cp.Variable(n)
        constraints = [
            cp.sum(h_opt) == n / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add overlap constraints for some sample indices
        # Avoiding too many constraints for performance
        lag_indices = np.linspace(1, n - 1, min(20, n - 1), dtype=int)
        for k in lag_indices:
            # Compute the correlation term for lag k
            # This is done using the full convolution approach  
            corr = cp.sum(cp.multiply(h_opt[:n-k], 1 - h_opt[k:]))
            constraints.append(corr <= best_c5 * dx + 1e-8)
            
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-8, abstol=1e-8)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final refinement
    try:
        h_final = np.clip(best_h, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
