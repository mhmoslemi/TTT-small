import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
        pop_size = 50
    else:
        initial_h = np.ones(35) * 0.5
        n_points = 35
        pop_size = 50
    
    # GA parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    max_gens = 1000
    
    # Resolution schedule
    resolutions = [35, 40]
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def normalize_individual(h):
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def tournament_selection(pop, fitnesses):
        selected = []
        for _ in range(pop_size):
            indices = np.random.choice(len(pop), tournament_size)
            best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
            selected.append(pop[best_idx])
        return selected
    
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        child1 = normalize_individual(child1)
        child2 = normalize_individual(child2)
        return child1, child2
    
    def mutate(individual):
        if random.random() > mutation_rate:
            return individual.copy()
        
        mutated = individual.copy()
        indices = np.random.choice(n_points, int(mutation_rate * n_points), replace=False)
        for i in indices:
            mutated[i] += np.random.normal(0, 0.02)
        
        mutated = normalize_individual(mutated)
        return mutated
    
    # Main optimization loop
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(initial_h, 2.0 / n_points)
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            h_interp = np.interp(np.linspace(0, 2, res),
                               np.linspace(0, 2, n_points),
                               best_h)
            n_points = res
            best_h = h_interp.copy()
            best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
            
        dx = 2.0 / n_points
        
        # Adjust population size for resolution
        pop_size = 50 if res == 35 else 20
        
        population = [create_individual() for _ in range(pop_size)]
        population[0] = best_h.copy()  # Seed with best found so far
        
        gen = 0
        stagnation_count = 0
        last_best_c5 = best_c5
        
        while gen < max_gens and time() - start_time < budget_s * 0.95:
            fitnesses = [evaluate_erdos_solution(ind, dx) for ind in population]
            new_best_c5 = min(fitnesses)
            
            if new_best_c5 < best_c5:
                best_c5 = new_best_c5
                best_h = population[np.argmin(fitnesses)].copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
            
            if stagnation_count >= 50:
                # Reset population if no improvement
                population = [create_individual() for _ in range(pop_size)]
                population[0] = best_h.copy()
                stagnation_count = 0
                
            # Selection
            selected = tournament_selection(population, fitnesses)
            
            # Crossover and Mutation
            new_population = []
            for i in range(0, pop_size, 2):
                if i + 1 < pop_size:
                    child1, child2 = crossover(selected[i], selected[i+1])
                else:
                    child1 = selected[i].copy()
                    child2 = selected[i].copy()
                
                child1 = mutate(child1)
                child2 = mutate(child2)
                
                new_population.extend([child1, child2])
            
            population = new_population[:pop_size]
            gen += 1
    
    # Final local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / n_points)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 100, 'ftol': 1e-8})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
