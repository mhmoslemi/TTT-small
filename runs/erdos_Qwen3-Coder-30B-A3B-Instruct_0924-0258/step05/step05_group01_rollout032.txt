import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Use triangle-like construction
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 1 - np.abs(x - 1)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution Simulated Annealing
    scales = [35, 50, 75, 100]
    current_scale_idx = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale_idx < len(scales):
        scale = scales[current_scale_idx]
        if scale > n_points:
            # Upscale
            x_fine = np.linspace(0, 2, scale)
            x_coarse = np.linspace(0, 2, n_points)
            h_fine = np.interp(x_fine, x_coarse, best_h)
            n_points = scale
            dx = 2.0 / n_points
        else:
            h_fine = best_h
            n_points = scale
            dx = 2.0 / n_points
        
        # Simulated Annealing at current scale
        T = 0.1
        alpha = 0.99
        max_iter = int(2000 * (budget_s * 0.95 - (time() - start_time)) / (budget_s * 0.95))
        if max_iter < 100:
            max_iter = 100
            
        h_current = h_fine.copy()
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        for _ in range(max_iter):
            if time() - start_time >= budget_s * 0.95:
                break
                
            # Make perturbation
            h_new = h_current.copy()
            for i in range(n_points):
                if np.random.random() < 0.1:
                    # Random perturbation
                    delta = np.random.normal(0, 0.1)
                    h_new[i] += delta
                    if np.random.random() < 0.3:
                        h_new[i] += delta*2
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            if c5_new < c5_current or np.random.random() < np.exp(-(c5_new - c5_current) / T):
                h_current = h_new
                c5_current = c5_new
            
            # Cool down
            T *= alpha
            
            # Update best solution
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                
        current_scale_idx += 1
    
    # Final refinement
    if time() - start_time < budget_s * 0.1:
        # Local optimization using L-BFGS
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        # Simple heuristic refinement with gradient-free method
        h_refined = best_h.copy()
        for _ in range(100):
            if time() - start_time >= budget_s * 0.99:
                break
            h_new = h_refined + np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = objective(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                h_refined = h_new
    
    return (list(best_h), best_c5, n_points)
