import numpy as np
from cvxpy import *
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try SDP relaxation approach
    if n_points <= 50:  # Only for smaller problems
        try:
            # SDP relaxation approach
            h_sdp = h_start.copy()
            
            # Set up optimization variables
            h = Variable(n_points)
            # Objective: just a placeholder as we are working with the relaxed version
            
            # Constraints
            constraints = [h >= 0, h <= 1, sum(h) == n_points/2.0]
            
            # Use a simple gradient approach instead of full SDP for efficiency
            def optimize_with_gradients(h_init):
                h_current = h_init.copy()
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                # Consistent bounds and constraints
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                # Try different optimization methods
                methods = ['L-BFGS-B', 'TNC', 'SLSQP']
                for method in methods:
                    if time() - start_time > budget_s * 0.9:
                        break
                    try:
                        res = minimize(objective, h_current, method=method, 
                                       bounds=bounds, constraints=cons_sum,
                                       options={'maxiter': 100, 'ftol': 1e-8})
                        if res.success:
                            h_refined = res.x
                            c5_refined = objective(h_refined)
                            if c5_refined < best_c5:
                                return h_refined, c5_refined
                    except:
                        continue
                        
                return h_current, objective(h_current)
            
            # Multi-start approach
            for _ in range(5):
                if time() - start_time > budget_s * 0.9:
                    break
                # Perturb h_start
                h_perturb = h_start + np.random.normal(0, 0.02, n_points)
                h_perturb = np.clip(h_perturb, 0, 1)
                h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
                
                h_new, c5_new = optimize_with_gradients(h_perturb)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            
            # Final refinement using good starting point
            if time() - start_time < budget_s * 0.9:
                h_final, c5_final = optimize_with_gradients(best_h)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
                    
        except Exception as e:
            # Fallback to a different approach if SDP fails
            pass
    
    # If we didn't get better result with SDP, run additional local optimization
    if best_c5 >= 0.38085:
        # Enhanced local optimization with random restarts
        for i in range(30):
            if time() - start_time > budget_s * 0.9:
                break
            # Random start
            h_random = np.random.uniform(0, 1, n_points)
            h_random = h_random * (n_points/2) / np.sum(h_random)
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, h_random, method='L-BFGS-B', 
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        if best_c5 < 0.3807:
                            break
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
