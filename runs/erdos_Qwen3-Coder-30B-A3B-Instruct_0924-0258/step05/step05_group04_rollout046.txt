import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Get initial solution if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize problem matrices
    # Create shift matrices for overlap computation
    A_full = np.zeros((2*n_points-1, n_points))
    for k in range(2*n_points-1):
        shift = k - (n_points - 1)
        if shift >= 0:
            A_full[k, shift:] = 1
        else:
            A_full[k, :shift] = 1
    
    # Helper to build shifted matrix
    def build_shift_matrix(shift):
        if shift >= 0:
            mat = np.zeros((n_points, n_points))
            mat[np.arange(n_points-shift), np.arange(shift, n_points)] = 1
        else:
            mat = np.zeros((n_points, n_points))
            mat[np.arange(-shift, n_points), np.arange(n_points+shift)] = 1
        return mat
    
    # Lagrangian relaxation with convex optimization
    max_iter = 500
    rho = 1e3
    tol = 1e-4
    
    h_current = h_start.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Initialize multipliers
    lambda_multipliers = np.ones(2*n_points-1) * 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Warm-start from current solution
        h_prev = h_current.copy()
        
        # Create the convex optimization problem
        h_var = cp.Variable(n_points)
        
        # Objective includes Lagrangian terms
        # Since we are minimizing the maximal overlap, we will use a proxy
        # We'll compute the maximum overlap more directly
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add Lagrangian terms for overlap constraints
        # Simplified: We use direct computations for overlap to build proxies
        # For this approach, solve a clean convex problem with constraints related to overlaps
        # We are computing directly the C5 constraint with the relaxed Lagrangian formulation
        
        # Use CVXPY to solve convex approximation
        prob = cp.Problem(cp.Minimize(cp.sum(h_var)), constraints)
        
        try:
            prob.solve(solver=cp.OSQP, warm_start=True, max_iter=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h_current = np.array(h_new)
                    h_current = np.clip(h_current, 0, 1)
                    # Normalize to sum to n_points/2
                    h_current = h_current * (n_points / 2) / np.sum(h_current)
            else:
                h_current = h_prev
        except:
            h_current = h_prev
            
        # Compute new C5
        current_c5 = evaluate_erdos_solution(h_current, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            
        # Check termination condition
        if best_c5 < 0.3807:
            break

        if iteration % 20 == 0 and time() - start_time < max_time * 0.9:

            def obj(h):
                return evaluate_erdos_solution(h, dx)
            def con(h):
                return np.sum(h) - n_points / 2.0
            
            cons = {'type': 'eq', 'fun': con}
            bounds = [(0, 1) for _ in range(n_points)]
            try:
                res = minimize(obj, h_current, method='SLSQP', bounds=bounds, 
                               constraints=cons, options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_local = res.x
                    f_local = obj(h_local)
                    if f_local < best_c5:
                        best_c5 = f_local
                        best_h = h_local.copy()
            except:
                pass
    
    # Final local refinement step
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_refine_start = best_h.copy()
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        def con(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': con}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, h_refine_start, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = obj(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
