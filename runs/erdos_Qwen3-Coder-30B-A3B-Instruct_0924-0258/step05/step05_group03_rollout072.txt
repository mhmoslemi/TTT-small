import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Discretization
    dx = 2.0 / n_points
    
    # Target bound
    target_bound = 0.3807
    
    # Optimization parameters
    max_iter = 100
    stagnation_limit = 50
    shift_samples = 100
    
    # History for early stopping
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # 1. Evaluate current h and find worst k
        shift_indices = np.linspace(0, 2, shift_samples, endpoint=False)
        overlaps = []
        for k_idx in shift_indices:
            k = k_idx * n_points / 2  # convert to discrete shift
            # Compute overlap for given k
            corr = np.correlate(h, 1 - h, mode='full')
            # In numpy, corr has length 2*n-1, and index n-1 corresponds to k=0
            # For k we need to shift by k points, convert continuous shift to discrete
            k_int = int(round(k))
            if k_int < 0:
                corr = np.roll(corr, k_int)
            elif k_int > 0:
                corr = np.roll(corr, -k_int)
                
            # Take actual correlation at the shift
            overlap_val = corr[n_points - 1] * dx
            overlaps.append(overlap_val)
            
        # Find worst k
        worst_k_idx = np.argmax(overlaps)
        worst_k = shift_indices[worst_k_idx]
            
        # 2. Apply convex-concave procedure
        # To optimize h with fixed k*, we solve the convex subproblem:
        # minimize sum_i h[i] * (1 - h[i + k*])
        # subject to sum(h) = n/2, 0 <= h[i] <= 1
        
        # More robust: compute the integral directly
        # We identify the specific shift as integer offset and compute overlap
        k_int = int(round(worst_k * n_points / 2))
        # Compute the overlap as a bilinear term on h, which is non-convex,
        # so we try a different optimization approach
        
        # Optimization issue in previous approach, try convex relaxation
        # Let's just optimize via direct gradient descent through approximated gradient
        # Sample shift uniformly and use optimization
        
        # Try to locally refine h using NLopt or scipy approach
        # Use a robust optimization subproblem in numpy
        def objective(h_vec):
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            # For the worst k, align it
            shift_idx = int(round(worst_k * n_points / 2)) + n_points - 1
            return corr[shift_idx] * dx
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
            
        # Add a small epsilon to prevent boundary issues
        bounds = [(1e-6, 1-1e-6) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Use L-BFGS-B method in scipy for better handling of constraints
        try:
            res = minimize(objective, h.copy(), method='L-BFGS-B',
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 50, 'ftol': 1e-9})
            if res.success:
                h_new = res.x.copy()
                # Clip to feasible region
                h_new = np.clip(h_new, 0, 1)
                # Renormalize
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                # Try to be more robust to small improvements by projecting to weak feasibility
                new_c5 = evaluate_erdos_solution(h_new, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                h = h_new.copy()
            else:
                # No improvement, slightly perturb and try again
                h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                stagnation_count += 1
        except Exception as e:
            # Error in optimization, perturb slightly
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count += 1

        # Early stopping
        if best_c5 < target_bound:
            break
            
        # If no improvement for too long, reinitialize or try strategies
        if stagnation_count > stagnation_limit:
            # Random perturbation
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0

    # Final refinement with SLSQP
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-10})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
