import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    c5_current = evaluate_erdos_solution(h_current, dx)
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # ADMM parameters
    max_iter = 5000
    mu = 1e-3
    tol = 1e-6
    rho = 1.0
    
    # Initialize ADMM variables
    h = h_current.copy()
    h_prev = h.copy()
    z = np.zeros(2 * n_points - 1)
    lambda_ = np.zeros(2 * n_points - 1)
    
    # Main ADMM loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update h via proximal operator
        # Solve: min_h 0.5 * ||h - (z - lambda/mu)||^2 + mu/2 * ||conv(h, 1-h) - z||^2
        # subject to 0 <= h <= 1, sum(h) = n_points/2
        
        def obj(h):
            conv_term = np.correlate(h, 1 - h, mode='full')
            penalty = 0.5 * mu * np.sum((conv_term - z)**2)
            return penalty
        
        def con(h):
            return np.sum(h) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con}
        
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_new = res.x
                # Basic projection to interval [0,1]
                h_new = np.clip(h_new, 0, 1)
                # Fix constraint
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            else:
                h = h_prev
        except:
            h = h_prev
            
        # Update z (proximal operator of negative convex function)
        z_prev = z.copy()
        z = np.clip(z + lambda_ - mu * np.correlate(h, 1 - h, mode='full'), 
                   -np.inf, np.inf)
        
        # Update dual variable
        lambda_ = lambda_ + rho * (h - z)
        
        # Update current estimate
        c5_new = evaluate_erdos_solution(h, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Check convergence
        if iteration % 50 == 0:
            residual = np.linalg.norm(h - h_prev) + np.linalg.norm(z - z_prev)
            if residual < tol:
                break
                
        h_prev = h.copy()
        
        if best_c5 < 0.3807:
            break
    
    # Final refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
