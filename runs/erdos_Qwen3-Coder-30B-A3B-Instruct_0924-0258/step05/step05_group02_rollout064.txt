import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Integer constraint handling via rounding and projection
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Coordinate descent refinement with random perturbations
    max_iter = 1000
    local_iter = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time() - start_time < budget_s * 0.95 and local_iter < max_iter:
        h_new = best_h.copy()
        
        # Random perturbation
        perturbation_magnitude = 0.05
        noise = np.random.normal(0, perturbation_magnitude, n_points)
        h_new += noise
        
        # Project and refine
        h_new = project_to_feasible(h_new)
        
        try:
            # Local optimization step
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_new, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 50, 'ftol': 1e-7})
            
            if res.success:
                h_opt = res.x
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
                
        except Exception:
            no_improvement_count += 1
            pass
        
        # Occasionally perform more aggressive search
        if local_iter % 20 == 0:
            h_aggressive = np.random.rand(n_points)
            h_aggressive = project_to_feasible(h_aggressive)
            try:
                res = minimize(evaluate_erdos_solution, h_aggressive, args=(dx,), 
                              method='SLSQP', bounds=[(0,1) for _ in range(n_points)],
                              constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0},
                              options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_aggr = res.x
                    c5_aggr = evaluate_erdos_solution(h_aggr, dx)
                    if c5_aggr < best_c5:
                        best_h = h_aggr
                        best_c5 = c5_aggr
            except Exception:
                pass
        
        if no_improvement_count >= max_no_improvement:
            # Random restart
            h_restart = np.random.rand(n_points)
            h_restart = project_to_feasible(h_restart)
            best_h = h_restart
            best_c5 = evaluate_erdos_solution(h_restart, dx)
            no_improvement_count = 0
        
        local_iter += 1
    
    # Final validation and projection
    final_h = project_to_feasible(best_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h
    
    return (list(best_h), best_c5, n_points)
