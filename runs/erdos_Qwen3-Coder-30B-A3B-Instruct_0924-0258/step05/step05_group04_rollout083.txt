import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Smooth approximation parameters
    tau = 1.0  # temperature for smooth max
    smoothing_decay = 0.99
    max_iter = 3000
    tolerance = 1e-7
    
    # Initialize optimization
    h = h_start.copy()
    h = h * (n_points / 2) / np.sum(h)  # ensure sum constraint
    
    # Gradient-based optimization with L-BFGS-B
    def smooth_max_overlap(h, tau):
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        # Use smooth max: log(sum(exp(tau * x))) / tau
        smooth_max = np.log(np.sum(np.exp(tau * corr))) / tau
        return smooth_max * dx
    
    def smooth_max_overlap_grad(h, tau):
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        # Gradient of smooth max w.r.t. h
        exp_corr = np.exp(tau * corr)
        Z = np.sum(exp_corr)
        # Gradient of correlation with respect to h
        # For correlation of h with (1-h), we need derivatives
        # Not exact but using approximate gradient formula
        # This simplifies assuming that indices work out symmetrically
        # In practice, it's very heuristic due to non-differentiable max operation
        n = len(h)
        grad = np.zeros(n)
        # Reconstructing correlation gradients part
        for i in range(n):
            # For current h[i], its influence on corr values
            # corr[k] = sum_{j} h[j] * (1 - h[j+k]) where all indices_mod n
            sum_grad = 0.0
            # From correlations involving h[i]
            for k in range(n):
                if (i+k) < n:
                    sum_grad += (1 - h[i+k]) * exp_corr[k]
                else:
                    # Apply circular condition
                    j = (i+k) % n
                    sum_grad += (1 - h[j]) * exp_corr[k]
            # Note: this is approximate gradient for smooth version
            grad[i] = sum_grad / (Z * tau)
        return grad * dx
    
    def objective(h):
        return smooth_max_overlap(h, tau)
    
    def grad(h):
        return smooth_max_overlap_grad(h, tau) 
    
    # Iteration loop
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.99:
            break
            
        # Update tau
        if iteration % 100 == 0 and iteration > 0:
            tau *= smoothing_decay
            
        # L-BFGS-B optimization step with box constraints
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            result = minimize(objective, h, method='L-BFGS-B', jac=grad,
                              bounds=bounds, options={'maxiter': 50})
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)  # keep sum constraint
            else:
                # fallback on simple gradient descent if L-BFGS fails
                g = grad(h)
                step_size = 1e-3
                h = h - step_size * g
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # fallback on simple step
            g = grad(h)
            step_size = 1e-3
            h = h - step_size * g
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final refinement with more precise solver
    if time() - start_time < max_time * 0.95:
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            def con(h):
                return np.sum(h) - n_points / 2.0
            
            cons = {'type': 'eq', 'fun': con}
            res = minimize(obj, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = obj(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
