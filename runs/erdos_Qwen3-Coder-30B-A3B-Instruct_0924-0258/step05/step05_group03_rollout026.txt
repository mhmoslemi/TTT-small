import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize initial h to satisfy constraint
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Sample shifts for overlap constraints
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    
    max_iter = 50
    last_c5 = float('inf')
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(initial_h, 2.0 / n_points)
    
    # Cut-plane approach with CVXPY
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all overlaps to find maximum
        overlaps = []
        for k in shifts:
            # Compute overlap for shift k
            if k == 0:
                overlap = np.sum(initial_h * (1 - initial_h)) * (2.0 / n_points)
            else:
                # Handle the convolution manually
                k_idx = int(k / (2.0 / n_points))  # Index in the discretized domain
                if k_idx >= n_points:
                    k_idx = n_points - 1
                # Left part of h convolved with right part of (1-h)
                h_shifted = np.concatenate([initial_h[k_idx:], np.zeros(k_idx)])
                left_h = np.concatenate([np.zeros(k_idx), initial_h[:-k_idx]])
                overlap = np.sum(left_h * (1 - h_shifted)) * (2.0 / n_points)
            overlaps.append(overlap)
        
        max_k_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_k_idx]
        max_shift = shifts[max_k_idx]
        
        # Add constraint on this overlap to the problem
        # Create a more refined CVX optimization
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Add overlap constraint
        k_idx = int(max_shift / (2.0 / n_points))
        if k_idx >= n_points:
            k_idx = n_points - 1
            
        # Evaluate overlap constraint
        if k_idx == 0:
            overlap_expr = cp.sum(h_vars * (1 - h_vars))
        else:
            # Convolution-like constraint
            h_shifted = cp.hstack([h_vars[k_idx:], cp.Constant(0.0) * cp Variable(n_points - k_idx)])
            left_h = cp.hstack([cp.Constant(0.0) * cp.Variable(k_idx), h_vars[:n_points - k_idx]])
            overlap_expr = cp.sum(left_h * (1 - h_shifted))
            
        constraints.append(overlap_expr <= max_overlap * n_points / 2)
        
        # Objective: maximize C5 (which is minimize -C5)
        # We don't actually optimize C5 directly, so we focus on satisfying the constraints
        # Instead, try local search on promising candidates
        
        # Instead of using CVXPY here, employ a local search technique
        # This is more direct and preserves the spirit of the problem
        
        # Direct approach: improve locally using SLSQP
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, initial_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                better_h = res.x
                better_c5 = obj(better_h)
                if better_c5 < best_c5:
                    best_h = better_h
                    best_c5 = better_c5
        except:
            pass
            
        # Local refinement to get the final results
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                final_h = res.x
                final_c5 = obj(final_h)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            final_h = best_h
            final_c5 = best_c5
        
        # Situations for early stopping
        if abs(last_c5 - best_c5) < 1e-5:
            break
        last_c5 = best_c5
        
        if best_c5 < 0.3807:
            break
        
        # Refine initial
        initial_h = best_h.copy()
    
    # Final polishing
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-10})
        if res.success:
            final_h = res.x
            final_c5 = obj(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap
