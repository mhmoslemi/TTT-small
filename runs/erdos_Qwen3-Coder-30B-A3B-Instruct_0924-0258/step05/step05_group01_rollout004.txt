import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

        # Create a triangular function as starting point
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start / np.sum(h_start) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Scale levels to refine
    scales = [35, 50, 75, 100]
    current_scale_idx = 0
    
    # Gradient-based refinement with momentum
    def gradient_step(h, learning_rate=0.01, momentum=0.9):
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Compute gradient
        grad = np.zeros_like(h)
        # Gradient is approximately 1 - 2*h for the specific shift that gives max overlap
        shift = max_idx - (len(h) - 1)
        if shift >= 0:
            grad[shift:] += (1 - 2 * h[:-shift]) if shift < len(h) else 0
        else:
            grad[:shift] += (1 - 2 * h[-shift:]) if -shift < len(h) else 0
        
        # Adjust by the gradient
        h_new = h - learning_rate * grad
        return np.clip(h_new, 0, 1)
    
    while time() - start_time < budget_s * 0.95 and current_scale_idx < len(scales):
        target_n = scales[current_scale_idx]
        
        # Interpolate or downsample to target resolution
        if target_n > n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_target = np.linspace(0, 2, target_n)
            h_interpolated = np.interp(x_target, x_orig, best_h)
            n_points = target_n
        else:
            h_interpolated = best_h
        
        dx = 2.0 / n_points
        h_current = h_interpolated.copy()
        
        # Gradient descent iterations
        learning_rate = 0.01
        best_iter_c5 = float('inf')
        momentum = 0.9
        velocity = np.zeros_like(h_current)
        
        for iteration in range(50):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Gradient step
            grad = np.zeros_like(h_current)
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            shift = max_idx - (len(h_current) - 1)
            
            # Compute gradient approximation
            if shift >= 0:
                grad[shift:] += (1 - 2 * h_current[:-shift]) if shift < len(h_current) else 0
            else:
                grad[:shift] += (1 - 2 * h_current[-shift:]) if -shift < len(h_current) else 0
            
            # Update with momentum
            velocity = momentum * velocity - learning_rate * grad
            h_new = h_current + velocity
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)

            c5_new = evaluate_erdos_solution(h_new, dx)

                h_current = h_new.copy()
                best_iter_c5 = c5_new
            
            # Reduce learning rate if no improvement
            if iteration > 0 and c5_new > best_iter_c5:
                learning_rate *= 0.9
            
        # Update best solution
        if best_iter_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = best_iter_c5
        
        current_scale_idx += 1
    
    # Final refinement with dedicated local optimization
    if time() - start_time < budget_s * 0.9:

            # Use SLSQP with starting point from best_h, but with a potentially different number of points
            # For simplicity, stick with current best_h and compute directly
            h_final = best_h.copy()
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
