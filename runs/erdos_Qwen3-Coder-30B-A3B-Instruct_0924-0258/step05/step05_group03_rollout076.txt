import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from provided construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Sample shifts for constraint tightening
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    
    # Initial solution
    h_current = initial_h.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Convex relaxation approach with cutting planes
    best_c5 = evaluate_erdos_solution(h_current, 2.0 / n_points)
    best_h = h_current.copy()
    
    # Use CVXPY for convex optimization
    # Variables
    h_vars = cp.Variable(n_points)
    c5_var = cp.Variable()
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points / 2,
        c5_var >= 0
    ]
    
    # Extend shifts to handle all cases
    shift_indices = np.arange(n_points)
    
    # Cutting plane iterations
    max_iter = 100
    iter_count = 0
    last_improvement = float('inf')
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Evaluate current overlaps
        overlaps = []
        for k in shifts:
            # Since we're dealing with discrete samples, approximate the integral
            k_idx = int(np.round(k * n_points / 2.0))  # Map k from [0,2] to [0,n]
            if k_idx >= n_points: k_idx = n_points - 1
            
            # Compute overlap for this shift
            overlap = 0.0
            for i in range(n_points - k_idx):
                overlap += h_current[i] * (1 - h_current[i + k_idx])
            
            # Add the boundary term as in correlation
            overlaps.append(overlap * (2.0 / n_points))
        
        max_overlap_idx = np.argmax(overlaps)
        max_overlap_val = overlaps[max_overlap_idx] 
        k_opt = shifts[max_overlap_idx]
        
        # Add constraint for current max overlap
        k_idx = int(np.round(k_opt * n_points / 2.0))
        if k_idx >= n_points: k_idx = n_points - 1
        
        # Hard constraint for this shift
        shift_constraint = 0.0
        for i in range(n_points - k_idx):
            shift_constraint += h_vars[i] * (1 - h_vars[i + k_idx])
        shift_constraint *= (2.0 / n_points)
        constraints.append(shift_constraint <= c5_var)
        
        # Ensure it does not leak below existing bound
        if c5_var.value is None or c5_var.value > max_overlap_val + 1e-6:
            c5_var.value = max_overlap_val + 1e-6
        else:
            c5_var.value = max_overlap_val
            
        # Re-optimize with new constraint
        prob = cp.Problem(cp.Minimize(c5_var), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_vars.value
                if h_new is not None:
                    c5_new = c5_var.value
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                
                if np.allclose(h_current, h_new, rtol=1e-6):
                    break  # No movement, converged
                h_current = h_new.copy()
                
        except Exception as e:
            # Fallback to previous best
            break
            
        # Early termination
        if best_c5 < 0.3807:
            break
            
        iter_count += 1

    # Final local refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    # Ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
