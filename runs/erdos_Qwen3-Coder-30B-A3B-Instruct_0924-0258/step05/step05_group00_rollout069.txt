import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach with cutting-plane refinement
    resolutions = [35, 40, 50, 60]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Cutting-plane optimization loop
        max_iterations = 100
        for iteration in range(max_iterations):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Find current maximum overlap and worst shift
            correlation = np.correlate(h_current, 1 - h_current, mode='full')
            max_corr_idx = np.argmax(correlation)
            max_corr_val = correlation[max_corr_idx]
            
            # Calculate which shift k caused this maximum overlap
            k = max_corr_idx - (len(h_current) - 1)
            
            # We will constraint this overlap using a convex relaxation
            h_vars = cp.Variable(n_points)
            constraints = [
                cp.sum(h_vars) == n_points / 2.0,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Add the overlap constraint for the worst shift
            if k >= 0:
                overlap_constraint = cp.sum(cp.multiply(h_vars[:n_points-k], 1 - h_vars[k:]))
            else:
                overlap_constraint = cp.sum(cp.multiply(h_vars[-k:], 1 - h_vars[:n_points+k]))
            
            constraints.append(overlap_constraint <= best_c5 * dx + 1e-8)
            
            # Also add constraints for some other significant shifts
            # Generate alternative shifts that may be critical
            shift_indices = np.arange(-n_points//2, n_points//2 + 1, max(1, n_points//20))
            for shift in shift_indices:
                if abs(shift) == 0 or abs(shift) >= n_points//2: continue
                if shift > 0:
                    overlap_sum = cp.sum(cp.multiply(h_vars[:n_points-shift], 1 - h_vars[shift:]))
                else:
                    overlap_sum = cp.sum(cp.multiply(h_vars[-shift:], 1 - h_vars[:n_points+shift]))
                constraints.append(overlap_sum <= best_c5 * dx + 1e-8)
            
            # Set up and solve convex problem
            try:
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_new = h_vars.value
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
                        h_current = h_new
                else:
                    break  # Stop if infeasible
            except:
                break  # Stop on any error
            
            # Random perturbation to escape local minima
            h_perturb = h_current + np.random.normal(0, 0.01, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
            
            # Local refinement with SLSQP
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(objective, h_perturb, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        h_current = h_refined
            except:
                pass
                
        h_start = best_h.copy()
    
    # Final optimization with high-resolution local search
    try:
        h_final = best_h.copy()
        dx = 2.0 / len(h_final)
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (len(h_final)/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    # Final SLSQP refinement
    try:
        h_slsqp = best_h.copy()
        dx = 2.0 / len(h_slsqp)
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - len(h) / 2.0
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(len(h_slsqp))]
        res = minimize(objective, h_slsqp, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100})
        if res.success:
            h_optimal = res.x
            c5_optimal = objective(h_optimal)
            if c5_optimal < best_c5:
                best_h = h_optimal
                best_c5 = c5_optimal
    except:
        pass
        
    return (list(best_h), best_c5, len(best_h))
