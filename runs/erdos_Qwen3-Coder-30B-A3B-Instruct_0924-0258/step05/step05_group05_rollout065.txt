import numpy as np
from scipy.signal import correlate
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Block-based approach
    M = int(np.sqrt(n_points))  # number of blocks
    if n_points % M != 0:
        M = int(np.sqrt(n_points)) + 1
        
    block_size = n_points // M
    if block_size * M != n_points:
        block_size = n_points // M
        M = n_points // block_size
    
    def h_from_blocks(blocks):
        # blocks: array of 0/1 indicating which blocks are on
        h = np.zeros(n_points)
        for i in range(M):
            if blocks[i] == 1:
                h[i*block_size:(i+1)*block_size] = 1
        # Normalize to meet sum constraint
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def get_random_blocks():
        blocks = np.zeros(M, dtype=int)
        on_blocks = n_points // (2 * block_size)
        if on_blocks * block_size * 2 > n_points:
            on_blocks = n_points // block_size
        indices = np.random.choice(M, on_blocks, replace=False)
        blocks[indices] = 1
        return blocks
    
    def mutate_blocks(blocks, mutation_rate=0.1):
        mutated = blocks.copy()
        for i in range(M):
            if np.random.random() < mutation_rate:
                mutated[i] = 1 - mutated[i]
        # Keep exactly n_points/2 integral
        current_sum = np.sum(mutated) * block_size
        target_sum = n_points / 2
        diff = target_sum - current_sum
        if diff != 0:
            # Adjust by flipping blocks
            num_flips = int(np.abs(diff) // block_size)
            if num_flips > 0:
                if diff > 0:
                    # Need to turn on more blocks
                    off_indices = np.where(mutated == 0)[0]
                    if len(off_indices) >= num_flips:
                        flip_indices = np.random.choice(off_indices, num_flips, replace=False)
                        mutated[flip_indices] = 1
                else:
                    # Need to turn off more blocks
                    on_indices = np.where(mutated == 1)[0]
                    if len(on_indices) >= num_flips:
                        flip_indices = np.random.choice(on_indices, num_flips, replace=False)
                        mutated[flip_indices] = 0
        return mutated
    
    def partition_blocks():
        # Try to find a good partition with fewer blocks
        new_M = min(M, max(2, int(np.sqrt(n_points))))
        new_block_size = n_points // new_M
        if new_M * new_block_size != n_points:
            new_M = n_points // new_block_size
        return new_M, new_block_size
    
    blocks = get_random_blocks()
    current_blocks = blocks.copy()
    current_c5 = evaluate_erdos_solution(h_from_blocks(blocks), dx)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_from_blocks(blocks)
    
    iterations = 0
    stagnation = 0
    max_stagnation = 50
    
    while time() - start_time < budget_s * 0.95:
        iterations += 1
        
        # Try mutation
        mutated = mutate_blocks(current_blocks)
        candidate = h_from_blocks(mutated)
        candidate_c5 = evaluate_erdos_solution(candidate, dx)
        
        if candidate_c5 < current_c5:
            current_blocks = mutated
            current_c5 = candidate_c5
            stagnation = 0
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = candidate
            if best_c5 < 0.3807:
                break
        else:
            stagnation += 1
            
        if stagnation > max_stagnation:
            # Try new configuration
            current_blocks = get_random_blocks()
            current_c5 = evaluate_erdos_solution(h_from_blocks(current_blocks), dx)
            stagnation = 0
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_from_blocks(current_blocks)
                
        # Occasionally reduce block size to increase flexibility
        if iterations % 50 == 0:
            new_M, new_block_size = partition_blocks()
            if new_M < M:
                original_blocks = current_blocks.copy()
                current_blocks = np.zeros(new_M, dtype=int)
                # Repartition
                for i in range(new_M):
                    start_idx = i * new_block_size
                    end_idx = (i + 1) * new_block_size
                    # Check average value of original blocks
                    avg_value = np.mean(h_from_blocks(original_blocks)[start_idx:end_idx])
                    if avg_value > 0.5:
                        current_blocks[i] = 1
                current_c5 = evaluate_erdos_solution(h_from_blocks(current_blocks), dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_from_blocks(current_blocks)
    
    # Final local refinement
    if time() - start_time < budget_s - 10:
        refine_h = best_h.copy()
        dt = 1e-5
        best_refined = best_c5
        
        # Gradient descent-style refinement
        for _ in range(200):
            if time() - start_time > budget_s - 5:
                break
            new_h = refine_h.copy()
            # Adjust by small perturbations
            for i in range(n_points):
                if np.random.random() < 0.3:
                    pert = np.random.normal(0, dt)
                    new_h[i] = max(0, min(1, new_h[i] + pert))
            
            # Enforce constraint sum(h) = n_points/2
            h_sum = np.sum(new_h)
            new_h = new_h * (n_points / 2) / h_sum
            
            new_c5 = evaluate_erdos_solution(new_h, dx)
            if new_c5 < best_refined:
                best_refined = new_c5
                refine_h = new_h
                
        if best_refined < best_c5:
            best_c5 = best_refined
            best_h = refine_h
        elif best_c5 < 0.3807:
            best_h = refine_h
    
    if best_c5 < 0.3807:
        # Try to improve with line search or local search
        success_count = 0
        for _ in range(20):
            if time() - start_time > budget_s - 5:
                break
                
            # Generate neighbor
            h_neighbor = best_h.copy()
            for i in range(n_points):
                if np.random.random() < 0.05:
                    h_neighbor[i] += np.random.normal(0, 0.01)
            
            # Bound
            h_neighbor = np.clip(h_neighbor, 0, 1)
            # Normalize
            h_sum = np.sum(h_neighbor)
            h_neighbor = h_neighbor * (n_points / 2) / h_sum
            
            neighbor_c5 = evaluate_erdos_solution(h_neighbor, dx)
            if neighbor_c5 < best_c5:
                best_c5 = neighbor_c5
                best_h = h_neighbor
                success_count += 1
                if best_c5 < 0.3807:
                    break
        
    return (list(best_h), best_c5, n_points)
