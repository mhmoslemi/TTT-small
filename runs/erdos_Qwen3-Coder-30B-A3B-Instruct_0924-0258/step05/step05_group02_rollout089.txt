import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a hybrid approach: local search with genetic-like mutation
    def mutate_h(h, rate=0.1):
        h_new = h.copy()
        mask = np.random.random(n_points) < rate
        h_new[mask] = np.random.random(np.sum(mask))
        return h_new
    
    def normalize_h(h):
        return h * (n_points/2) / np.sum(h)
    
    max_restarts = 100
    restart_count = 0
    max_no_improvement = 20
    no_improvement = 0
    
    while time() - start_time < budget_s * 0.95:
        # Random restart
        if restart_count > max_restarts or no_improvement > max_no_improvement:
            h_candidate = np.random.rand(n_points)
            h_candidate = normalize_h(h_candidate)
            restart_count = 0
            no_improvement = 0
        else:
            # Local search with perturbation
            h_candidate = best_h.copy()
            h_candidate += np.random.normal(0, 0.05, n_points)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h(h_candidate)
        
        # Optimization refinement using SLSQP
        try:
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx),
                h_candidate,
                method='SLSQP',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 100, 'ftol': 1e-8},
                constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    no_improvement = 0
                else:
                    restart_count += 1
                    no_improvement += 1
            else:
                restart_count += 1
                no_improvement += 1
        except Exception:
            restart_count += 1
            no_improvement += 1
        
        if time() - start_time > budget_s * 0.9:
            break
    
    # Final refinement with stricter optimization
    try:
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda h: evaluate_erdos_solution(h, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-9}
        )
        if res.success:
            h_final = res.x
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Ensure final solution respects constraint exactly
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
