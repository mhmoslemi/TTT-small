import numpy as np
from time import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with existing construction or triangular
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.linspace(0, 1, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale approach
    scales = [35, 50, 75, 100]
    
    for scale in scales:
        if scale > n_points:
            # Interpolate to higher resolution
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            h_new = np.interp(x_new, x_orig, best_h)
            n_points = scale
            dx = 2.0 / n_points
        else:
            h_new = best_h.copy()
            n_points = scale
            dx = 2.0 / n_points
            
        h_current = h_new.copy()
        
        # Gradient-based optimization
        max_iter = 100
        alpha = 0.01
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute cross-correlation
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            max_val = corr[max_idx]
            
            # Compute gradient for the dominant shift
            h_grad = np.zeros_like(h_current)
            shifted_h = np.roll(1 - h_current, max_idx - len(h_current) + 1)
            h_grad += shifted_h
            
            # Apply gradient and update
            h_new = h_current - alpha * h_grad
            h_new = np.clip(h_new, 0, 1)
            
            # Rescale to constraint
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            h_current = h_new
            
            # Validation
            c5_current = evaluate_erdos_solution(h_current, dx)
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                
        if time() - start_time > budget_s * 0.95:
            break
            
    return (list(best_h), best_c5, n_points)
