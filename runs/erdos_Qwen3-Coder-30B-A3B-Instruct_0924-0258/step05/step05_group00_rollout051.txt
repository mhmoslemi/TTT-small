import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive Stochastic Descent
    max_iter = 500
    step_size = 0.1
    decay_factor = 0.99
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Stochastic perturbation
        noise = np.random.normal(0, step_size, n_points)
        h_perturbed = best_h + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Normalize
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        c5_candidate = evaluate_erdos_solution(h_perturbed, dx)
        if c5_candidate < best_c5:
            best_h = h_perturbed
            best_c5 = c5_candidate
            
        step_size *= decay_factor
    
    # Quadratic Constraint Programming refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample key lags
        lags = np.arange(1, n_points//2 + 1)
        for k in lags:
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-8)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final refinement
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), c5_final, n_points)
