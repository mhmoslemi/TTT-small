import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)

    # Prepare the correlation matrix for constraints
    # We'll use a slightly different approach to optimization
    # Use a combination of gradient descent, domain projection, and local refinement

    # Optimizing using continuous relaxation with projection
    h_current = h_start.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)  # Ensure sum = n/2
    
    # Learning rate and decay
    lr = 0.01
    decay = 0.995
    
    # Main optimization loop
    iteration = 0
    max_iterations = 50000
    stagnation_count = 0
    best_c5_so_far = best_c5
    prev_c5 = best_c5
    
    while time() - start_time < max_time * 0.95 and iteration < max_iterations:
        iteration += 1
        # Compute correlation for overlap
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_overlap = np.max(corr) * dx
        
        # Compute gradient estimate by finite differences
        eps = 1e-4
        grad = np.zeros_like(h_current)
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = np.max(corr_plus) * dx
            grad[i] = (max_plus - max_overlap) / eps
        
        # Apply gradient descent step
        h_new = h_current - lr * grad
        # Project to feasible region [0,1] and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5_so_far:
            best_c5_so_far = new_c5
            best_h = h_new.copy()
        
        # Check if we improved
        if new_c5 < prev_c5:
            h_current = h_new
            prev_c5 = new_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count > 500:  # Local minimum reached
                # Perturb solution to escape
                h_current = h_current + np.random.normal(0, 0.001, n_points)
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points / 2) / np.sum(h_current)
                stagnation_count = 0
                lr = max(1e-6, lr * decay)
        
        # Early stopping for target
        if best_c5_so_far < 0.3807:
            break

    # Local refine final solution
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5_so_far:
                    best_h = h_final
                    best_c5_so_far = c5_final
        except:
            pass

    return (list(best_h), best_c5_so_far, n_points)
