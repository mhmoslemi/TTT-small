import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Ensure sum equals n_points/2 and h in [0,1]
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Optimization parameters
    max_iter = 100
    early_stop_threshold = 0.3807
    tolerance = 1e-6
    shift_sampling = 100
    
    # Track best solution
    h_best = initial_h.copy()
    c5_best = evaluate_erdos_solution(h_best, 2.0 / n_points)
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find optimal shift k for current h
        shifts = np.linspace(0, 2, shift_sampling)
        max_overlap = -np.inf
        best_shift = 0
        
        for k in shifts:
            idx = int(k * n_points / 2)
            if idx >= n_points:
                continue
            # Compute overlap for this shift
            overlap = 0
            for i in range(n_points):
                j = (i + idx) % n_points
                overlap += h_best[i] * (1 - h_best[j])
            overlap *= 2.0 / n_points
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = k
        
        # Update objective with best shift
        # Minimize the overlap w.r.t. h (convex approx)
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Define objective
        idx_shift = int(best_shift * n_points / 2)
        objective = 0
        for i in range(n_points):
            j = (i + idx_shift) % n_points
            objective += h_var[i] * (1 - h_var[j])
        
        objective *= 2.0 / n_points
        
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = np.array(h_var.value).flatten()
                if np.all(h_new >= 0) and np.all(h_new <= 1):
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    c5_new = evaluate_erdos_solution(h_new, 2.0 / n_points)
                    if c5_new < c5_best:
                        h_best = h_new.copy()
                        c5_best = c5_new
        except:
            pass
            
        # Early termination
        if c5_best < early_stop_threshold:
            break
            
    # Final refinement with SLSQP
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, h_best, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            h_best = res.x
            c5_best = obj(h_best)
    except:
        pass

    return (list(h_best), c5_best, n_points)
