import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create triangular function
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start / np.sum(h_start) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def accept_prob(old_cost, new_cost, temp):
        if new_cost < old_cost:
            return 1.0
        return math.exp(-(new_cost - old_cost) / temp)
    
    def perturb(h, rate=0.1):
        h_new = h.copy()
        mask = np.random.random(len(h)) < rate
        h_new[mask] = 1 - h_new[mask]
        return h_new
    
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Multi-resolution scheme
    scales = [35, 50, 75]
    current_scale = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale < len(scales):
        scale = scales[current_scale]
        
        # Coarsen to current scale
        if scale > n_points:
            # Interpolate to higher resolution
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            h_coarse = np.interp(x_new, x_orig, best_h)
        else:
            h_coarse = best_h[:scale] if scale < n_points else best_h.copy()
            h_coarse = h_coarse * (scale / n_points)
        
        n_points = scale
        dx = 2.0 / n_points
        
        # Simulated Annealing
        current_h = h_coarse.copy()
        current_c5 = evaluate_erdos_solution(current_h, dx)
        
        temp = 0.1
        alpha = 0.99
        max_iter = int(800 * (scale / 35))
        
        for _ in range(max_iter):
            if time() - start_time >= budget_s * 0.95:
                break
                
            # Perturb solution
            new_h = perturb(current_h)
            new_h = project(new_h)
            new_c5 = evaluate_erdos_solution(new_h, dx)
            
            # Accept or reject
            if accept_prob(current_c5, new_c5, temp) > np.random.random():
                current_h = new_h
                current_c5 = new_c5
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    
            # Cool down
            temp *= alpha
            
        current_scale += 1
    
    # Final refinement at finest scale
    if time() - start_time < budget_s * 0.9:
        try:
            refined_h = project(best_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
