import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h_start = 1 - np.abs(x - 1)  # triangular function
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale optimization
    scales = [35, 50, 75, 100]
    current_scale_idx = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale_idx < len(scales):
        target_n = scales[current_scale_idx]
        if target_n > n_points:
            # Interpolate to higher resolution
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, target_n)
            h_new = np.interp(x_new, x_old, best_h)
            n_points = target_n
            dx = 2.0 / n_points
        else:
            h_new = best_h
            n_points = target_n
            dx = 2.0 / n_points
            
        # Local optimization using gradient-free method
        max_iter = 100
        h_current = h_new.copy()
        
        # Try multiple restarts
        for restart in range(10):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_init = h_current.copy()
            
            # Random perturbation
            noise = np.random.normal(0, 0.03, n_points)
            h_perturbed = h_init + noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            
            # Minimize using COBYLA
            cons_eq = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(evaluate_erdos_solution, h_perturbed, 
                              args=(dx,), method='COBYLA',
                              constraints=cons_eq, bounds=bounds,
                              options={'maxiter': 100, 'rhobeg': 0.01})
                
                if res.success:
                    h_refined = res.x
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        h_current = h_refined
            except:
                pass
            
        current_scale_idx += 1
    
    # Final refinement
    if time() - start_time < budget_s * 0.1:
        # Refine at finest resolution
        h_final = best_h.copy()
        try:
            cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(evaluate_erdos_solution, h_final, 
                          args=(dx,), method='SLSQP',
                          constraints=cons, bounds=bounds,
                          options={'maxiter': 100})
            if res.success:
                h_final = res.x
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
