import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        # The correlation values correspond to shifts from -n+1 to n-1

                max_corr = corr[k_idx]

            if 0 <= j < n_points:

        # So, we solve:
        # min sum h_i * (1 - h[i+shift]) s.t. 0 <= h_i <= 1, sum h_i = n/2
        # Which is equivalent to max sum h_i * h[i+shift] - sum h_i

            # Check convergence
            diff = np.sum((h - h_new)**2)
            h = h_new
            if diff < toler:

    try:

            return evaluate_erdos_solution(h_vec, dx)

            final_c5 = obj(best_h)
            if final_c5 < best_c5:

    except:
        pass
    
    return (list(best_h), best_c5, n_points)
