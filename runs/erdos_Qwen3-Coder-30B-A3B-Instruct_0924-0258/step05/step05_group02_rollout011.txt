import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    # Try to optimize using different approaches
    max_iter = 150
    iter_count = 0
    patience = 0
    max_patience = 30
    
    while iter_count < max_iter and time() - start_time < budget_s * 0.95:
        iter_count += 1
        
        # Random perturbation
        h_perturbed = best_h + np.random.normal(0, 0.03, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Try optimization with different methods
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            # Try different optimization methods
            methods = ['SLSQP', 'L-BFGS-B', 'TNC']
            for method in methods:
                try:
                    res = minimize(
                        lambda h: evaluate_erdos_solution(h, dx), 
                        h_perturbed, 
                        method=method, 
                        bounds=bounds, 
                        constraints=cons_sum,
                        options={'maxiter': 100, 'ftol': 1e-7}
                    )
                    if res.success:
                        h_opt = res.x
                        c5_opt = evaluate_erdos_solution(h_opt, dx)
                        if c5_opt < best_c5:
                            best_h = h_opt
                            best_c5 = c5_opt
                            patience = 0
                            break
                except Exception:
                    continue
            else:
                # If optimization failed, use perturbed solution
                c5_new = evaluate_erdos_solution(h_perturbed, dx)
                if c5_new < best_c5:
                    best_h = h_perturbed
                    best_c5 = c5_new
                    patience = 0
        except Exception:
            pass
        
        # Check for improvement
        patience += 1
        if patience > max_patience:
            break
    
    # Final refinement with trust-constr if available
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        res = minimize(
            lambda h: evaluate_erdos_solution(h, dx), 
            best_h, 
            method='trust-constr', 
            bounds=bounds, 
            constraints=cons_sum,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if res.success:
            h_refined = res.x
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Ensure final constraints
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
