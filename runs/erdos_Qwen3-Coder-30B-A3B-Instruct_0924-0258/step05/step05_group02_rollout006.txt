import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based approach with genetic operators
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        return h
    
    def crossover(h1, h2):
        alpha = random.uniform(0.4, 0.6)
        child = alpha * h1 + (1 - alpha) * h2
        child = child * (n_points/2) / np.sum(child)
        child = np.clip(child, 0, 1)
        return child
    
    def mutate(h, rate=0.1):
        h = h.copy()
        for i in range(n_points):
            if random.random() < rate:
                h[i] += np.random.normal(0, 0.02)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def refine_with_optimization(h):
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(
                objective, 
                h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-7}
            )
            if res.success:
                return res.x
        except:
            pass
        return h
    
    pop_size = 50
    elitism = int(0.3 * pop_size)
    max_generations = 1000
    generation = 0
    evaluated = 0
    stagnation_counter = 0
    max_stagnation = 50
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        evaluated += 1
    
    while len(population) < pop_size:
        ind = create_individual()
        population.append(ind)
        evaluated += 1
    
    # Evaluate initial population
    fitness = []
    for ind in population:
        c5 = evaluate_erdos_solution(ind, dx)
        fitness.append((c5, ind))
        if c5 < best_c5:
            best_c5 = c5
            best_h = ind.copy()
    
    fitness.sort()
    best_in_generation = best_c5
    
    while time() - start_time < budget_s * 0.95 and evaluated < max_generations:
        # Selection
        elite = [ind for _, ind in fitness[:elitism]]
        
        # Generate offspring
        offspring = []
        for _ in range(pop_size - len(elite)):
            if len(elite) >= 2:
                parent1, parent2 = random.sample(elite, 2)
                child = crossover(parent1, parent2)
            else:
                child = create_individual()
            child = mutate(child)
            offspring.append(child)
            evaluated += 1
        
        # Refine top individuals with optimization
        refined = []
        top_indices = sorted(range(len(fitness)), key=lambda i: fitness[i][0])[:10]
        for i in top_indices:
            refined_ind = refine_with_optimization(fitness[i][1])
            c5_refined = evaluate_erdos_solution(refined_ind, dx)
            refined.append((c5_refined, refined_ind))
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = refined_ind.copy()
        
        # Combine new population
        new_population = elite + offspring
        new_fitness = [(evaluate_erdos_solution(ind, dx), ind) for ind in new_population]
        new_fitness.extend(refined)
        new_fitness.sort()
        
        fitness = new_fitness[:pop_size]
        if fitness[0][0] < best_in_generation:
            best_in_generation = fitness[0][0]
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        if stagnation_counter > max_stagnation:
            # Introduce diversity
            for _ in range(5):
                new_pop = create_individual()
                new_fitness.append((evaluate_erdos_solution(new_pop, dx), new_pop))
            new_fitness.sort()
            fitness = new_fitness[:pop_size]
            stagnation_counter = 0
        
        generation += 1
    
    # Final refinement of best solution
    try:
        final_h = refine_with_optimization(best_h)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h
    except:
        pass
    
    # Ensure constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
