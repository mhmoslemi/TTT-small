import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Use a triangular function as starting point
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale refinement with gradient-based optimization
    scales = [35, 50, 75]
    current_scale = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale < len(scales):
        scale = scales[current_scale]
        if scale > n_points:
            # Interpolate to higher resolution
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            h_new = np.interp(x_new, x_old, best_h)
            n_points = scale
            dx = 2.0 / n_points
        else:
            h_new = best_h
            n_points = scale
            dx = 2.0 / n_points
        
        # Gradient-based local optimization
        max_iter = 100
        lr = 0.01
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute overlap and gradient approximation
            corr = np.correlate(h_new, 1 - h_new, mode='full')
            max_idx = np.argmax(corr)
            max_corr = corr[max_idx]
            
            # Estimate gradient by perturbing at the max overlap shift
            h_perturbed = h_new.copy()
            h_perturbed[max_idx % n_points] += 1e-4
            corr_pert = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            grad = (corr_pert[max_idx] - max_corr) / 1e-4
            
            # Simple gradient step
            h_new = h_new - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                lr = max(1e-5, lr * 0.99)  # Adaptive learning rate
            else:
                lr = max(1e-5, lr * 0.995)
                
        current_scale += 1
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
