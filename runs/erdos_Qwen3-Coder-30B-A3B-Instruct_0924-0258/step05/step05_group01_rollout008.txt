import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a triangular-shaped initial function
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 1.0 - np.abs(x - 1.0)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution simulated annealing
    scale_levels = [35, 50, 75, 100]
    
    for scale in scale_levels:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate or coarsen to current scale
        if scale > n_points:
            # Upsample
            x_new = np.linspace(0, 2, scale)
            x_old = np.linspace(0, 2, n_points)
            h_current = np.interp(x_new, x_old, best_h)
            n_points = scale
        else:
            # Downsample with averaging
            h_current = np.array([np.mean(best_h[i:i+int(n_points/scale)]) 
                                for i in range(0, n_points, int(n_points/scale))])
            n_points = scale
        dx = 2.0 / n_points
        
        # Simulated Annealing
        T = 0.1
        alpha = 0.99
        max_iter = int(5000 * (scale / 35.0))
        
        h_temp = h_current.copy()
        c5_temp = evaluate_erdos_solution(h_temp, dx)
        
        if c5_temp < best_c5:
            best_h = h_temp.copy()
            best_c5 = c5_temp
        
        iter_count = 0
        while iter_count < max_iter and time() - start_time < budget_s * 0.95:
            # Perturb solution
            h_proposal = h_temp.copy()
            # Flip some elements
            n_flip = max(1, int(0.1 * n_points))
            flip_indices = np.random.choice(n_points, size=n_flip, replace=False)
            for idx in flip_indices:
                h_proposal[idx] = 1 - h_proposal[idx]
                
            # Ensure constraints
            h_proposal = np.clip(h_proposal, 0, 1)
            h_proposal = h_proposal * (n_points/2) / np.sum(h_proposal)
            
            c5_proposal = evaluate_erdos_solution(h_proposal, dx)
            
            # Accept or reject
            if c5_proposal < c5_temp or np.random.rand() < math.exp(-(c5_proposal - c5_temp) / T):
                h_temp = h_proposal
                c5_temp = c5_proposal
                
                if c5_temp < best_c5:
                    best_h = h_temp.copy()
                    best_c5 = c5_temp
            
            T = alpha * T
            iter_count += 1
    
    return (list(best_h), best_c5, n_points)
