import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start / np.sum(h_start) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution SA with geometric cooling
    resolutions = [35, 50, 75, 100]
    temperature_schedule = [0.1, 0.05, 0.01, 0.005]
    
    for res_idx, (res, T0) in enumerate(zip(resolutions, temperature_schedule)):
        if res > n_points:
            # Interpolate to higher resolution
            h_new = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), best_h)
            n_points = res
            dx = 2.0 / n_points
        else:
            h_new = best_h
            n_points = res
            dx = 2.0 / n_points
        
        # Simulated Annealing implementation
        h_current = h_new.copy()
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        # Tweak iterations based on resolution
        max_iter = int(2000 * (res / 35) ** 1.2)
        T = T0
        alpha = 0.995
        
        for _ in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb solution
            h_trial = h_current.copy()
            for i in range(n_points):
                if np.random.random() < 0.1:
                    # Add small random change
                    delta = np.random.normal(0, 0.03)
                    h_trial[i] += delta
                    # Clip and rescale
                    h_trial = np.clip(h_trial, 0, 1)
                    h_trial = h_trial * (n_points/2) / np.sum(h_trial)
            
            trial_c5 = evaluate_erdos_solution(h_trial, dx)
            
            # Accept or reject
            if trial_c5 < current_c5 or np.random.random() < math.exp((current_c5 - trial_c5) / T):
                h_current = h_trial
                current_c5 = trial_c5
            
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
            
            # Cool down
            T *= alpha
        
        # Rescale to precise sum constraint
        best_h = best_h * (n_points/2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
