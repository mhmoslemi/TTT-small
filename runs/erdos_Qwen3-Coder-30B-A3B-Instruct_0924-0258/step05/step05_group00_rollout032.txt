import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm
    pop_size = 50
    max_gens = 1000
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    def initialize_population(size, n):
        population = []
        if 'initial_h_values' in globals():
            # Start with initial solution
            h_ref = np.array(initial_h_values)
            h_ref = h_ref * (n/2) / np.sum(h_ref)
            population.append(h_ref.copy())
            size -= 1
        
        for _ in range(size):
            h = np.random.random(n)
            h = h * (n/2) / np.sum(h)
            population.append(h)
        return population
    
    def tournament_selection(population, fitnesses):
        selected = []
        for _ in range(len(population)):
            idxs = np.random.choice(len(population), tournament_size)
            best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
            selected.append(population[best_idx].copy())
        return selected
    
    def crossover(parent1, parent2):
        if np.random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = np.random.randint(1, len(parent1))
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(individual, rate=0.1, sigma=0.02):
        mutated = individual.copy()
        indices = np.random.choice(len(mutated), int(len(mutated) * rate), replace=False)
        mutated[indices] += np.random.normal(0, sigma, len(indices))
        mutated = np.clip(mutated, 0, 1)
        return mutated * (len(mutated)/2) / np.sum(mutated)
    
    # GA loop
    population = initialize_population(pop_size, n_points)
    generation = 0
    
    while time() - start_time < budget_s * 0.8 and generation < max_gens:
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(ind, dx) for ind in population]
        
        # Update best solution
        min_fit_idx = np.argmin(fitnesses)
        if fitnesses[min_fit_idx] < best_c5:
            best_h = population[min_fit_idx].copy()
            best_c5 = fitnesses[min_fit_idx]
        
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Crossover and mutation
        new_population = []
        for i in range(0, len(selected), 2):
            if i + 1 < len(selected):
                child1, child2 = crossover(selected[i], selected[i+1])
                new_population.append(mutate(child1, mutation_rate))
                new_population.append(mutate(child2, mutation_rate))
            else:
                new_population.append(mutate(selected[i], mutation_rate))
        
        population = new_population[:pop_size]
        generation += 1
    
    # Higher resolution refinement
    if time() - start_time < budget_s * 0.95 and best_c5 > 0.3808:
        n_points = 40
        dx = 2.0 / n_points
        h_current = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, len(best_h)), 
                              best_h)
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Local optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h_current, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    # Final refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
