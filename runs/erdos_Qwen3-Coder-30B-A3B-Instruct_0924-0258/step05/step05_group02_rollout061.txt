import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a hybrid approach: local search + convex relaxation
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def local_search(h_initial):
        # Use L-BFGS-B optimizer
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        res = minimize(
            lambda h: evaluate_erdos_solution(h, dx), 
            h_initial, 
            method='L-BFGS-B', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        return res.x if res.success else h_initial
    
    # Try different initialization strategies
    strategies = []
    if 'initial_h_values' in globals():
        strategies.append(('parent', best_h))
    
    # Random initialization
    h_rand = np.random.rand(n_points)
    h_rand = h_rand * (n_points/2) / np.sum(h_rand)
    strategies.append(('random', h_rand))
    
    # Step function initialization
    h_step = np.zeros(n_points)
    mid = n_points // 2
    h_step[:mid] = 1
    h_step = h_step * (n_points/2) / np.sum(h_step)
    strategies.append(('step', h_step))
    
    # Improved initialization
    h_improved = np.zeros(n_points)
    half = n_points // 2
    h_improved[:half] = 1
    h_improved = h_improved * (n_points/2) / np.sum(h_improved)
    strategies.append(('improved', h_improved))
    
    # Try all strategies in order, but stop early if budget
    for name, h_start in strategies:
        if time() - start_time > budget_s * 0.95:
            break
            
        h_refined = local_search(h_start)
        c5_new = evaluate_erdos_solution(h_refined, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_refined.copy()
        
        # Also try intermediate refinement via CVXPY if allowed
        try:
            h_relaxed = cp.Variable(n_points)
            cons_sum = cp.sum(h_relaxed) == n_points / 2.0
            cons_bounds = [h_relaxed >= 0, h_relaxed <= 1]
            
            # Use a smaller subset of constraints to keep it manageable
            # Only keep constraints for small lags
            constraints = [cons_sum] + cons_bounds
            for k in range(1, min(10, n_points//2 + 1)):
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_relaxed[:n_points-k], 1 - h_relaxed[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_points-k]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-8)
            
            prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-8, max_iters=100)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_relaxed.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass
    
    # Final refinement with multiple restarts
    restart_count = 0
    max_restarts = 20
    while time() - start_time < budget_s * 0.98 and restart_count < max_restarts:
        h_restart = np.random.rand(n_points)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        h_restart = np.clip(h_restart, 0, 1)
        h_refined = local_search(h_restart)
        c5_new = evaluate_erdos_solution(h_refined, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_refined.copy()
        restart_count += 1
    
    # Ensure final refinement adheres to constraints
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
