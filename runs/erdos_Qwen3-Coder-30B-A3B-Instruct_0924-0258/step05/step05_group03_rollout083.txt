import numpy as np
from cvxpy import *
import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 35
        h0 = np.ones(n_points) * 0.5
        
    # Ensure sum constraint
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Parameters
    dx = 2.0 / n_points
    max_shifts = 100
    shift_samples = np.linspace(0, 2, max_shifts, endpoint=False)
    
    # Variables
    h = Variable(n_points)
    C5 = Variable()
    
    # Create constraints
    constraints = [h >= 0, h <= 1, sum(h) == n_points / 2, C5 >= 0]
    
    # Add initial constraints from shifts
    for shift in shift_samples:
        shift_idx = int(shift / dx)
        if shift_idx >= 0 and shift_idx < n_points:
            # Compute overlap constraint
            overlap = sum(h[i] * (1 - h[i + shift_idx]) for i in range(n_points - shift_idx)) * dx
            constraints.append(overlap <= C5)
    
    # Objective
    objective = Minimize(C5)
    
    # Build problem
    prob = Problem(objective, constraints)
    
    # Cutting plane iterations
    max_iterations = 50
    last_c5 = float('inf')
    stagnation = 0
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        try:
            # Solve relaxed problem
            prob.solve(solver=ECOS, max_iters=1000, verbose=False)
            
            if prob.status != 'optimal':
                break
                
            h_val = h.value
            current_c5 = C5.value
            
            if abs(current_c5 - last_c5) < 1e-6:
                stagnation += 1
            else:
                stagnation = 0
                last_c5 = current_c5
                
            if current_c5 < 0.3807:
                break
                
            if stagnation > 50:
                break
                
            # Evaluate overlaps for current solution
            overlaps = []
            for shift in shift_samples:
                shift_idx = int(shift / dx)
                if shift_idx >= 0 and shift_idx < n_points:
                    overlap = sum(h_val[i] * (1 - h_val[i + shift_idx]) for i in range(n_points - shift_idx)) * dx
                    overlaps.append(overlap)
            
            # Add constraint for most violated shift
            max_overlap_idx = np.argmax(overlaps)
            max_shift = shift_samples[max_overlap_idx]
            max_shift_idx = int(max_shift / dx)
            if max_shift_idx >= 0 and max_shift_idx < n_points:
                overlap = sum(h[i] * (1 - h[i + max_shift_idx]) for i in range(n_points - max_shift_idx)) * dx
                constraints.append(overlap <= C5)
                
        except:
            break
    
    # Final refinement with local optimizer to improve solution
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, h_val, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            h_val = res.x
            final_c5 = obj(h_val)
        else:
            final_c5 = current_c5
    except:
        final_c5 = current_c5
    
    return (list(h_val), final_c5, n_points)
