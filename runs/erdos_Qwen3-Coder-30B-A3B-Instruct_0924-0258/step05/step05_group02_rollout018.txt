import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid approach to get better starting point
    n_coarse = 20
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.ones(n_coarse) * 0.5
    
    def solve_coarse():
        h_var = cp.Variable(n_coarse)
        constraints = [
            cp.sum(h_var) == n_coarse / 2.0,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add several key constraints to make problem more tractable
        for k in range(1, min(8, n_coarse)):
            if k < n_coarse // 2:
                corr_sum = cp.sum(cp.multiply(h_var[:n_coarse-k], 1 - h_var[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_var[k:], 1 - h_var[:n_coarse-k]))
            constraints.append(corr_sum <= 0.4 * dx_coarse + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-7)
            if prob.status == cp.OPTIMAL:
                return h_var.value
        except:
            pass
        return h_coarse
    
    # Get coarse resolution solution
    h_coarse_sol = solve_coarse()
    if h_coarse_sol is not None:
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, n_coarse), 
                            h_coarse_sol)
        h_interp = np.clip(h_interp, 0, 1)
        h_interp = h_interp * (n_points/2) / np.sum(h_interp)
        h_start = h_interp
    
    # Fine-tuning with local optimization
    max_iter = 1000
    iter_count = 0
    no_improve_count = 0
    max_no_improve = 100
    
    h_current = h_start.copy()
    current_c5 = evaluate_erdos_solution(h_current, dx)
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        h_prev = h_current.copy()
        prev_c5 = current_c5
        
        # Try to improve via gradient-based local search
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            def jac(h):
                # Approximate gradient
                eps = 1e-6
                grad = np.zeros_like(h)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                    
                    h_minus = h.copy()
                    h_minus[i] -= eps
                    h_minus = np.clip(h_minus, 0, 1)
                    h_minus = h_minus * (n_points/2) / np.sum(h_minus)
                    
                    grad[i] = (obj(h_plus) - obj(h_minus)) / (2 * eps)
                
                return grad
            
            res = minimize(obj, h_current, method='L-BFGS-B', 
                          bounds=bounds, jac=jac, 
                          options={'maxiter': 50, 'ftol': 1e-9})
            
            if res.success:
                h_current = res.x
                # Ensure constraints still hold
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points/2) / np.sum(h_current)
                current_c5 = evaluate_erdos_solution(h_current, dx)
                
                if current_c5 < best_c5:
                    best_h = h_current.copy()
                    best_c5 = current_c5
                    no_improve_count = 0
                else:
                    no_improve_count += 1
            else:
                no_improve_count += 1
        except Exception:
            no_improve_count += 1
        
        # Occasionally apply mutations to escape local optima
        if iter_count % 20 == 0 and no_improve_count > 10:
            mutation_strength = 0.03
            h_mutated = h_current + np.random.normal(0, mutation_strength, n_points)
            h_mutated = np.clip(h_mutated, 0, 1)
            h_mutated = h_mutated * (n_points/2) / np.sum(h_mutated)
            mutated_c5 = evaluate_erdos_solution(h_mutated, dx)
            if mutated_c5 < current_c5:
                h_current = h_mutated
                current_c5 = mutated_c5
                no_improve_count = 0
        
        iter_count += 1
        if no_improve_count > max_no_improve:
            break
    
    # Final check and refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
