import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use default initialization if no initial_h_values available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimized multi-resolution strategy
    n_coarse = min(20, n_points)  # Coarse grid resolution
    dx_coarse = 2.0 / n_coarse
    
    # Initial coarse h
    h_coarse = np.ones(n_coarse) * 0.5
    h_coarse = h_coarse * (n_coarse/2) / np.sum(h_coarse)
    
    # Convex relaxation on coarse grid
    try:
        h_relaxed = cp.Variable(n_coarse)
        constraints = [
            cp.sum(h_relaxed) == n_coarse / 2.0,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        
        # Add constraints for some shifts
        for k in range(1, min(15, n_coarse)):  
            if k < n_coarse // 2:
                corr_sum = cp.sum(cp.multiply(h_relaxed[:n_coarse-k], 1 - h_relaxed[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_coarse-k]))
            constraints.append(corr_sum <= best_c5 * dx_coarse + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        
        if prob.status == cp.OPTIMAL:
            h_coarse = h_relaxed.value
            h_coarse = np.clip(h_coarse, 0, 1)
            h_coarse = h_coarse * (n_coarse/2) / np.sum(h_coarse)
    except Exception:
        pass
    
    # Interpolate to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_points/2) / np.sum(h_fine)
    
    # Local refinement with SLSQP
    best_h = h_fine.copy()
    best_c5 = evaluate_erdos_solution(h_fine, dx)
    
    max_iter = 200
    iter_count = 0
    no_improvement = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Local optimization step
        bounds = [(0, 1) for _ in range(n_points)]
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        try:
            res = minimize(
                obj, 
                best_h.copy(), 
                method='SLSQP', 
                bounds=bounds,
                options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-6}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    no_improvement = 0
                else:
                    no_improvement += 1
            else:
                no_improvement += 1
        except Exception:
            no_improvement += 1
        
        # Adaptive restart if no improvement
        if no_improvement > 30:
            h_restart = np.random.rand(n_points)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            c5_restart = evaluate_erdos_solution(h_restart, dx)
            if c5_restart < best_c5:
                best_h = h_restart
                best_c5 = c5_restart
            no_improvement = 0
            
        iter_count += 1
    
    # Final refinement
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            evaluate_erdos_solution, 
            best_h.copy(), 
            method='SLSQP', 
            bounds=bounds,
            args=(dx,),
            options={'maxiter': 20, 'ftol': 1e-9}
        )
        if res.success:
            final_h = res.x
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
    
    # Final constraints check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), float(final_c5), int(n_points))
