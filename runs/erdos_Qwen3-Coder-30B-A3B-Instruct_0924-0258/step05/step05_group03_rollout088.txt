import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize initial h to satisfy constraint
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    h = initial_h.copy()
    
    # Optimization parameters
    max_iter = 500
    patience = 50
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    best_h = h.copy()
    stagnation_count = 0
    
    # Sample more shifts for gradient computation
    shift_samples = min(50, n_points)
    
    # Tracking variables
    last_c5 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Sample shifts to compute gradients
        shifts = np.random.choice(n_points, shift_samples, replace=False)
        
        # Find shift with maximum overlap
        max_overlap = -1
        best_shift = 0
        for k in shifts:
            # Compute overlap for this shift
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlap *= 2.0 / n_points
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = k
        
        # Compute gradient for the worst shift
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_shift) % n_points
            gradient[i] += (1 - h[j]) * 2.0 / n_points  # Gradient from h[i]
            gradient[j] -= h[i] * 2.0 / n_points        # Gradient from h[j]
        
        # Gradient descent step
        step_size = 0.01
        h_new = h - step_size * gradient
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, 2.0 / n_points)
        
        # Adaptive update
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            stagnation_count = 0
            if c5_new < 0.3807:
                break
        else:
            stagnation_count += 1
            if stagnation_count > patience:
                # Perturb the solution if stuck
                h = h + np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                stagnation_count = 0
        
        # Early termination if improvement
        if best_c5 < last_c5:
            last_c5 = best_c5
    
    # Final refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
