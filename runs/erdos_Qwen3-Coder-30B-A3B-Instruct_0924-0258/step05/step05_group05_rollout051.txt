import numpy as np
from scipy.optimize import minimize
from time import time
import hashlib

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def hash_state(h):
    # Create a hash of the sorted values to detect repeated states
    sorted_h = np.sort(h)
    return hashlib.md5(sorted_h.tobytes()).hexdigest()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_tenure = max(1, int(0.2 * n_points))
    max_iter = 1000
    max_stagnation = 100
    mutation_scale = 0.04
    mutation_budget = max(1, int(0.3 * n_points))
    
    # Initialize
    tabu_list = []
    current_h = h_start.copy()
    current_c5 = best_c5
    stagnation_counter = 0
    
    # Main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create new candidate by perturbing some elements
        candidate_h = current_h.copy()
        indices_to_mutate = np.random.choice(n_points, mutation_budget, replace=False)
        
        for i in indices_to_mutate:
            # Add Gaussian noise
            candidate_h[i] += np.random.normal(0, mutation_scale)
        
        # Repair: clamp to [0,1]
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Repair: ensure sum = n_points / 2 exactly
        scale_factor = (n_points / 2) / np.sum(candidate_h)
        candidate_h *= scale_factor
        
        # Evaluate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Check tabu status
        candidate_hash = hash_state(candidate_h)
        is_tabu = candidate_hash in tabu_list
        
        # Accept condition: better solution or not tabu
        if candidate_c5 < current_c5 or not is_tabu:
            current_h = candidate_h
            current_c5 = candidate_c5
            
            # Update best solution if needed
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                if best_c5 < 0.3807:
                    break
            
            # Update tabu list
            tabu_list.append(candidate_hash)
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
            
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        
        # Adaptive changes
        if stagnation_counter > max_stagnation:
            mutation_scale = min(0.5, mutation_scale * 1.5)
            stagnation_counter = 0
        else:
            mutation_scale = max(0.005, mutation_scale * 0.99)
    
    # Final local refinement if time allows
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
