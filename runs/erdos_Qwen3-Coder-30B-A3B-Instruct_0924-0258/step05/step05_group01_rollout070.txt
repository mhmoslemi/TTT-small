import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
import warnings
warnings.filterwarnings('ignore')

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = project_to_feasible(h_start.copy(), n_points)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Initial candidates
    initial_candidates = []
    initial_values = []
    
    # Add parent if available
    if 'initial_h_values' in globals():
        initial_candidates.append(best_h.copy())
        initial_values.append(best_c5)
    
    # Generate random candidates
    m = 20
    for i in range(m):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        c5 = evaluate_erdos_solution(h, dx)
        initial_candidates.append(h)
        initial_values.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Train initial GP
    X_train = np.array(initial_candidates)
    y_train = np.array(initial_values)
    
    # GP kernel
    kernel = C(1.0, (1e-3, 1e3)) * RBF(1.0, (1e-2, 1e2))
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-6, n_restarts_optimizer=5,
                                  normalize_y=False, noise_level=1e-10)
    gp.fit(X_train, y_train)
    
    # SMBO loop
    max_iter = 1000
    eval_count = len(initial_candidates)
    resolution_step = 0
    resolution_limit = 100
    
    for iteration in range(max_iter):
        if time() - start_time >= budget_s * 0.95:
            break
            
        # Sample new candidates
        n_suggest = 50
        candidates = []
        
        # Generate samples
        for _ in range(n_suggest):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            candidates.append(h)
        
        # Predict with GP
        X_suggest = np.array(candidates)
        preds = gp.predict(X_suggest, return_std=True)
        means = preds[0]
        stds = preds[1]
        
        # Score based on EI (Expected Improvement)
        # For simplicity, we directly use the mean prediction
        score = means
        top_idx = np.argsort(score)[:10]  # Top 10
        
        # Evaluate selected candidates
        for i in top_idx:
            h = candidates[i]
            h = project_to_feasible(h, n_points)
            c5 = evaluate_erdos_solution(h, dx)
            eval_count += 1
            
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            
            # Update GP with new data
            X_train = np.vstack([X_train, h])
            y_train = np.append(y_train, c5)
            gp.fit(X_train, y_train)
            
        # Adaptive resolution
        if eval_count > 30 and resolution_step < 5:
            if resolution_step * 10 < eval_count:
                resolution_step += 1
                n_points += 5
                if n_points > resolution_limit:
                    n_points = resolution_limit
                    
                dx = 2.0 / n_points
                # Interpolate existing solution
                h_new = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
                best_h = project_to_feasible(h_new, n_points)
                best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
