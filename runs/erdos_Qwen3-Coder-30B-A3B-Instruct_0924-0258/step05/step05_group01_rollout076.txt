import numpy as np
from scipy.signal import convolve
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create triangular-like function for good initial guess
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 1 - np.abs(x - 1)  # Triangular function
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Refinement levels: coarse to fine
    scales = [35, 50, 75, 100]
    scale_idx = 0
    
    while time() - start_time < budget_s * 0.95 and scale_idx < len(scales):
        curr_scale = scales[scale_idx]
        
        if curr_scale > n_points:
            # Interpolate to higher resolution
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, curr_scale)
            h_current = np.interp(x_new, x_orig, best_h)
            n_points = curr_scale
        else:
            h_current = best_h.copy()
            n_points = curr_scale
            
        dx = 2.0 / n_points
        
        # Gradient-based local optimization
        max_iters = 100
        lr = 0.01
        last_improvement = 0
        stagnation_count = 0
        
        for iteration in range(max_iters):
            if time() - start_time >= budget_s * 0.95:
                break
            
            # Evaluate current state
            current_c5 = evaluate_erdos_solution(h_current, dx)
            
            # Find the maximum overlap and its corresponding shift
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_shift = np.argmax(corr)
            
            # Compute approximate gradient via finite differences
            epsilon = 1e-4
            grad = np.zeros_like(h_current)
            for i in range(len(h_current)):
                h_plus = h_current.copy()
                h_plus[i] += epsilon
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                grad[i] = (corr_plus[max_shift] - corr[max_shift]) / epsilon
            
            # Update with clipped gradient descent
            h_new = h_current - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate new solution
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            if new_c5 < current_c5:
                h_current = h_new
                best_c5 = min(best_c5, new_c5)
                last_improvement = iteration
                stagnation_count = 0
            else:
                stagnation_count += 1
                if stagnation_count > 10:
                    lr *= 0.9
            
        # Update best solution
        if evaluate_erdos_solution(h_current, dx) < best_c5:
            best_h = h_current.copy()
            best_c5 = evaluate_erdos_solution(h_current, dx)
            
        scale_idx += 1

    return (list(best_h), best_c5, n_points)
