import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create symmetric triangular function
        mid = n_points // 2
        h_start = np.minimum(np.arange(n_points), 2*mid - np.arange(n_points))
        h_start = h_start / (mid)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convex relaxation parameters
    lambda_reg = 0.9  # Weight for convex approximation
    
    # Gradient descent parameters
    max_iter = 100
    tol = 1e-8
    armijo_beta = 0.5
    armijo_sigma = 0.1
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try convex optimization approach
        try:
            # Create a relaxed version of the problem
            h_vars = cp.Variable(n_points)
            # Minimize max overlap using convex relaxation
            # Lower bound on the overlap is semi-definite representable
            # For now, use gradient-based local refinement
            
            h_current = best_h.copy()
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            # Use L-BFGS-B to refine
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                objective, 
                h_current, 
                method='L-BFGS-B',
                bounds=bounds, 
                constraints=cons_sum,
                options={'ftol': tol, 'gtol': tol}
            )
            
            if res.success:
                h_refined = res.x
                # Ensure feasibility
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = objective(h_refined)
                
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    if best_c5 < 0.3807:
                        break
                        
        except Exception as e:
            pass
        
        # Perturb the current solution and explore locally
        try:
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
                if best_c5 < 0.3807:
                    break
                    
        except Exception as e:
            pass
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:

            def objective(h):
                return evaluate_erdos_solution(h, dx)

                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='L-BFGS-B',
                           bounds=bounds, constraints=cons_sum,
                           options={'ftol': 1e-9, 'gtol': 1e-9})

                h_final = res.x

                c5_final = objective(h_final)

                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
