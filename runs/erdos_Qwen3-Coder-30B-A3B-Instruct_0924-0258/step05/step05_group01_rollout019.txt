import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Convert to binary if needed
    if np.all(np.logical_or(h_start == 0, h_start == 1)):
        h_best = h_start.copy()
    else:
        # Round to nearest binary values
        h_rounded = np.round(h_start)
        # Adjust to satisfy sum constraint
        current_sum = np.sum(h_rounded)
        target_sum = n_points / 2
        diff = int(target_sum - current_sum)
        
        if diff > 0:
            # Need to increase number of 1s
            zero_indices = np.where(h_rounded == 0)[0]
            if len(zero_indices) >= diff:
                h_rounded[zero_indices[:diff]] = 1
        elif diff < 0:
            # Need to decrease number of 1s
            one_indices = np.where(h_rounded == 1)[0]
            if len(one_indices) >= -diff:
                h_rounded[one_indices[:(-diff)]] = 0
        
        h_best = h_rounded
    
    # Ensure sum constraint
    current_sum = np.sum(h_best)
    target_sum = n_points / 2
    diff = int(target_sum - current_sum)
    
    if diff > 0:
        zero_indices = np.where(h_best == 0)[0]
        if len(zero_indices) >= diff:
            h_best[zero_indices[:diff]] = 1
    elif diff < 0:
        one_indices = np.where(h_best == 1)[0]
        if len(one_indices) >= -diff:
            h_best[one_indices[:(-diff)]] = 0
    
    dx = 2.0 / n_points
    c5_best = evaluate_erdos_solution(h_best, dx)
    
    # Tabu search parameters
    tabu_tenure = 10
    tabu_list = []
    no_improvement_count = 0
    max_no_improvement = 100
    shaking_threshold = 50
    
    # Main tabu search loop
    while time() - start_time < budget_s * 0.95:
        # Generate neighbors by flipping exactly one 1 to 0 and one 0 to 1
        one_indices = np.where(h_best == 1)[0]
        zero_indices = np.where(h_best == 0)[0]
        
        if len(one_indices) == 0 or len(zero_indices) == 0:
            # Restart with a new binary construction
            h_new = np.zeros(n_points)
            ones_indices = np.random.choice(n_points, n_points // 2, replace=False)
            h_new[ones_indices] = 1
            h_best = h_new
            c5_best = evaluate_erdos_solution(h_best, dx)
            no_improvement_count = 0
            tabu_list = []
            continue
            
        # Generate a neighbor
        i1 = np.random.choice(one_indices)
        i0 = np.random.choice(zero_indices)
        
        # Create a neighbor
        h_neighbor = h_best.copy()
        h_neighbor[i1] = 0
        h_neighbor[i0] = 1
        
        # Check if the move is tabu
        move = (i1, i0)
        is_tabu = move in tabu_list or (i0, i1) in tabu_list
        
        # Evaluate neighbor
        c5_neighbor = evaluate_erdos_solution(h_neighbor, dx)
        
        # Check aspiration criteria
        aspiration = c5_neighbor < c5_best
        
        # Accept the neighbor if it's not tabu or satisfies aspiration
        if not is_tabu or aspiration:
            h_best = h_neighbor
            c5_best = c5_neighbor
            
            # Add move to tabu list
            tabu_list.append(move)
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
                
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Shake if no improvement for too long
        if no_improvement_count >= shaking_threshold:
            # Randomly perturb by flipping some bits
            num_flip = max(1, int(n_points * 0.05))  # Flip 5% of bits
            indices_to_flip = np.random.choice(n_points, num_flip, replace=False)
            h_best[indices_to_flip] = 1 - h_best[indices_to_flip]
            
            # Ensure sum constraint
            current_sum = np.sum(h_best)
            diff = int(n_points / 2 - current_sum)
            
            if diff > 0:
                zero_indices = np.where(h_best == 0)[0]
                if len(zero_indices) >= diff:
                    h_best[zero_indices[:diff]] = 1
            elif diff < 0:
                one_indices = np.where(h_best == 1)[0]
                if len(one_indices) >= -diff:
                    h_best[one_indices[:(-diff)]] = 0
                    
            c5_best = evaluate_erdos_solution(h_best, dx)
            no_improvement_count = 0
            tabu_list = []
    
    # Final validation
    if np.any(h_best < 0) or np.any(h_best > 1):
        # Reconstruct with valid bounds
        h_best = np.clip(h_best, 0, 1)
        h_best = h_best * (n_points / 2) / np.sum(h_best)
    
    c5_best = evaluate_erdos_solution(h_best, dx)
    
    return (list(h_best), c5_best, n_points)
