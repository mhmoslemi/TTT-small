import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid for initial relaxation (20 points)
    n_coarse = 20
    dx_coarse = 2.0 / n_coarse
    
    # Optimization loop
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Phase 1: Coarse grid relaxation
        if iteration % 5 == 1:
            h_coarse = np.interp(np.linspace(0, 2, n_coarse), 
                               np.linspace(0, 2, n_points), 
                               best_h)
            h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
            
            try:
                h_relaxed = cp.Variable(n_coarse)
                constraints = [
                    cp.sum(h_relaxed) == n_coarse / 2.0,
                    h_relaxed >= 0,
                    h_relaxed <= 1
                ]
                # Add some constraints for k shifts up to a reasonable bound
                for k in range(1, min(10, n_coarse)):  
                    if k < n_coarse // 2:
                        corr_sum = cp.sum(cp.multiply(h_relaxed[:n_coarse-k], 1 - h_relaxed[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_coarse-k]))
                    constraints.append(corr_sum <= best_c5 * dx_coarse + 1e-6)
                
                prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-7)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_relaxed.value
                    # Interpolate to original resolution
                    h_interp = np.interp(np.linspace(0, 2, n_points), 
                                       np.linspace(0, 2, n_coarse), 
                                       h_feasible)
                    h_interp = h_interp * (n_points / 2) / np.sum(h_interp)
                    h_interp = np.clip(h_interp, 0, 1)
                    c5_feasible = evaluate_erdos_solution(h_interp, dx)
                    if c5_feasible < best_c5:
                        best_h = h_interp
                        best_c5 = c5_feasible
                        no_improvement_count = 0
                    else:
                        no_improvement_count += 1
                else:
                    no_improvement_count += 1
            except Exception:
                no_improvement_count += 1

        # Phase 2: Fine grid local search
        else:
            # Gradient descent with adaptive mutation
            try:
                h_current = best_h.copy()
                
                # Compute gradient approximatively by finite differences
                eps = 1e-5
                grad = np.zeros_like(h_current)
                c5_current = evaluate_erdos_solution(h_current, dx)
                
                for i in range(len(h_current)):
                    h_plus = h_current.copy()
                    h_plus[i] += eps
                    h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                    h_plus = np.clip(h_plus, 0, 1)
                    c5_plus = evaluate_erdos_solution(h_plus, dx)
                    grad[i] = (c5_plus - c5_current) / eps
                
                # Update with gradient descent
                eta = 0.01
                h_new = h_current - eta * grad
                h_new = h_new * (n_points/2) / np.sum(h_new)
                h_new = np.clip(h_new, 0, 1)
                
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
                    
                # Occasionally apply mutations
                if np.random.rand() < 0.1 and iteration > 10:
                    mutation_strength = 0.05
                    mask = np.random.rand(len(h_current)) < 0.1
                    noise = np.random.randn(len(h_current)) * mutation_strength
                    h_mutated = h_current + noise * mask
                    h_mutated = h_mutated * (n_points/2) / np.sum(h_mutated)
                    h_mutated = np.clip(h_mutated, 0, 1)
                    c5_mutated = evaluate_erdos_solution(h_mutated, dx)
                    if c5_mutated < best_c5:
                        best_h = h_mutated
                        best_c5 = c5_mutated
                        no_improvement_count = 0
                        
            except Exception:
                no_improvement_count += 1
                
        # Check for early termination
        if no_improvement_count > max_no_improvement:
            break
            
        # Ensure we don't spend too long
        if time() - start_time > budget_s * 0.98:
            break

    # Final validation
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), c5_final, n_points)
