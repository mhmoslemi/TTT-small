import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    M = int(np.sqrt(n_points))  # Number of blocks
    if n_points % M != 0:
        M = int(np.sqrt(n_points)) + 1
    block_size = n_points // M
    
    if block_size * M != n_points:
        block_size = n_points // M
        M = n_points // block_size

    if 'initial_h_values' in globals():
