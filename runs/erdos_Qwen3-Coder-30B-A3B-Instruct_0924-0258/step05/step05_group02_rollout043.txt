import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a simple construction
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def project_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Coarse optimization first to get a better starting point
    # Use a smaller grid for coarse optimization
    n_coarse = 20
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.ones(n_coarse) * 0.5
    
    # Convex relaxation for coarse case
    try:
        h_relaxed = cp.Variable(n_coarse)
        constraints = [
            cp.sum(h_relaxed) == n_coarse / 2.0,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        
        # Add a reasonable subset of overlap constraints
        for k in range(1, min(10, n_coarse)):
            if k < n_coarse // 2:
                corr_sum = cp.sum(cp.multiply(h_relaxed[:n_coarse-k], 1 - h_relaxed[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_coarse-k]))
            constraints.append(corr_sum <= best_c5 * dx_coarse + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_relaxed.value
            # Interpolate to high resolution
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, n_coarse), 
                               h_feasible)
            h_coarse = project_feasible(h_interp)
    except Exception:
        pass
    
    # Refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Gradient-enhanced search
    max_iter = 2000
    patience = 0
    max_patience = 100
    mutation_freq = 50
    mutation_strength = 0.05
    lr = 0.01
    
    # Sample the gradient manually
    def compute_gradient(h, dx):
        # Approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h)
        base_val = objective(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus)
            grad[i] = (objective(h_plus) - base_val) / eps
        return grad
    
    h_current = h_coarse.copy()
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.99:
            break
            
        # Compute gradient and update
        try:
            grad = compute_gradient(h_current, dx)
            h_new = h_current - lr * grad
            h_new = project_feasible(h_new)
            
            new_val = objective(h_new)
            if new_val < best_c5:
                best_c5 = new_val
                best_h = h_new.copy()
                patience = 0
            else:
                patience += 1
                # Occasionally perturb to escape local minima
                if i % mutation_freq == 0 and patience > 10:
                    h_new = h_current + mutation_strength * np.random.randn(len(h_current))
                    h_new = project_feasible(h_new)
                    new_val = objective(h_new)
                    if new_val < best_c5:
                        best_c5 = new_val
                        best_h = h_new.copy()
                        patience = 0
            
            h_current = h_new.copy()
            
            if patience > max_patience:
                break
                
        except Exception:
            patience += 1
            if patience > max_patience:
                break
                
    # Final validation
    final_h = project_feasible(best_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
