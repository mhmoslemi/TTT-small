import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        n_points = 20

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid optimization
    if n_points < 35:
        # upsample coarse solution
        h_upsampled = np.interp(np.linspace(0, 2, 35), np.linspace(0, 2, n_points), h_start)
        h_upsampled = h_upsampled * (35/2) / np.sum(h_upsampled)
        h_upsampled = np.clip(h_upsampled, 0, 1)
        best_h = h_upsampled
        best_c5 = evaluate_erdos_solution(best_h, 2.0/35)
        n_points = 35
        dx = 2.0 / n_points
    
    # Fix initial point to ensure constraint satisfaction
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Refinement loop
    max_iterations = 100
    improvement_count = 0
    
    while time() - start_time < budget_s * 0.95:
        # Compute all correlations and find the maximum
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        max_overlap = corr[max_k] * dx
        
        # Check if we improved
        if max_overlap < best_c5 - 1e-8:
            best_c5 = max_overlap
            improvement_count = 0
        else:
            improvement_count += 1
        
        # If no improvement after many steps, perturb the solution
        if improvement_count >= 50:
            h_perturbed = best_h + np.random.normal(0, 0.05, n_points)

            best_c5 = c5_perturbed
            improvement_count = 0

        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0

                # For a specific shift k, compute overlap

                max_corr_k = np.max(corr_k)
                return max_corr_k * dx - best_c5
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}

            res = minimize(
                objective,
                best_h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,

                new_c5 = evaluate_erdos_solution(new_h, dx)

                    best_c5 = new_c5

    # Final verification and correction
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
