import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Get initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse optimization phase
    n_low = 20
    h_low = np.ones(n_low) * 0.5
    dx_low = 2.0 / n_low
    
    # First, optimize a coarse grid
    try:
        h_relaxed = cp.Variable(n_low)
        constraints = [
            cp.sum(h_relaxed) == n_low / 2.0,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        
        # Add constraints for shifts to reduce non-convexity
        for k in range(1, min(20, n_low//2)):  
            if k < n_low // 2:
                corr_sum = cp.sum(cp.multiply(h_relaxed[:n_low-k], 1 - h_relaxed[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_low-k]))
            constraints.append(corr_sum <= best_c5 * dx_low + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        
        if prob.status == cp.OPTIMAL and prob.value < np.inf:
            h_low = h_relaxed.value
    except Exception:
        pass
    
    # Upsample coarse grid to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, n_low), h_low)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_points/2) / np.sum(h_fine)
    
    # Fine grid refinement with local optimization
    def loss_func(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Gradient descent + local search
    h_current = h_fine.copy()
    c5_current = loss_func(h_current)
    if c5_current < best_c5:
        best_h = h_current.copy()
        best_c5 = c5_current
    
    max_iter = 5000
    iter_count = 0
    patience = 0
    n_mutate = 10  # number of points to mutate per iteration
    
    while time() - start_time < budget_s * 0.98 and iter_count < max_iter:
        iter_count += 1
        try:
            # Compute gradient using finite differences
            epsilon = 1e-5
            grad = np.zeros_like(h_current)
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += epsilon
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                f_plus = loss_func(h_plus)
                
                h_minus = h_current.copy()
                h_minus[i] -= epsilon
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points/2) / np.sum(h_minus)
                f_minus = loss_func(h_minus)
                
                grad[i] = (f_plus - f_minus) / (2 * epsilon)
            
            # Update with learning rate
            lr = 0.01 / (1.0 + iter_count / 1000.0)
            h_new = h_current - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = loss_func(h_new)
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                patience = 0
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
            else:
                patience += 1
                
        except Exception:
            patience += 1
            
        # Adaptive mutation to escape local optima
        if patience > 50 or iter_count % 200 == 0:
            idx_mutate = np.random.choice(n_points, n_mutate, replace=False)
            mutation_strength = 0.05 * (1.0 - min(1.0, iter_count / 2000.0))
            for i in idx_mutate:
                h_current[i] += np.random.normal(0, mutation_strength)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
            c5_current = loss_func(h_current)
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
            patience = 0
            
        if patience > 100:
            break
    
    # Final validation
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
