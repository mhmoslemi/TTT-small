import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.linspace(0, 1, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # DE parameters
    population_size = 30
    mutation_factor = 0.8
    crossover_prob = 0.8
    max_gens = 500
    
    # Initialize population
    population = []
    fitness = []
    
    # Generate initial population
    for i in range(population_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Differential evolution loop
    for gen in range(max_gens):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        if sorted_fit[0] < best_c5:
            best_h = sorted_pop[0].copy()
            best_c5 = sorted_fit[0]
            if best_c5 < 0.3807:
                break
        
        # Create new population
        new_pop = [sorted_pop[0], sorted_pop[1]]  # Elitism
        
        for i in range(2, population_size):
            # Select three different individuals
            a, b, c = random.sample(range(len(sorted_pop)), 3)
            if a == i: a = (a + 1) % len(sorted_pop)
            if b == i: b = (b + 1) % len(sorted_pop)
            if c == i: c = (c + 1) % len(sorted_pop)
            
            # Mutation
            mutant = sorted_pop[a] + mutation_factor * (sorted_pop[b] - sorted_pop[c])
            
            # Crossover
            cross_points = np.random.random(n_points) < crossover_prob
            if not np.any(cross_points):
                cross_points[np.random.randint(0, n_points)] = True
            
            child = np.where(cross_points, mutant, sorted_pop[i])
            
            # Repair constraints
            child = np.clip(child, 0, 1)
            if np.sum(child) == 0:
                child = np.ones_like(child) * 0.5
            else:
                child = child * (n_points/2) / np.sum(child)
            
            new_pop.append(child)
        
        # Evaluate new population
        population = new_pop
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Final refinement with local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Refine best solution found
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
