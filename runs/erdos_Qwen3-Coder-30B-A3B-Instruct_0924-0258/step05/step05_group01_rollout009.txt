import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Initialize with triangular function
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start / np.sum(h_start) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution Simulated Annealing
    scales = [35, 50, 75, 100]
    temp_init = 0.1
    cooling_rate = 0.99
    max_iter_per_scale = int(0.8 * budget_s / len(scales))
    
    current_h = h_start.copy()
    
    for scale in scales:
        if scale > n_points:
            # Interpolate to higher resolution
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            current_h = np.interp(x_new, x_orig, current_h)
            n_points = scale
            dx = 2.0 / n_points
        
        # Simulated Annealing at this scale
        temp = temp_init
        current_c5 = evaluate_erdos_solution(current_h, dx)
        
        # Optimization loop
        iter_count = 0
        while iter_count < max_iter_per_scale and time() - start_time < budget_s * 0.95:
            # Generate neighbor
            h_new = current_h.copy()
            # Flip some values randomly
            flip_mask = np.random.rand(n_points) < 0.1
            h_new[flip_mask] = 1 - h_new[flip_mask]
            
            # Ensure constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            # Metropolis criterion
            if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / temp):
                current_h = h_new
                current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
            
            iter_count += 1
            temp *= cooling_rate
        
        if time() - start_time > budget_s * 0.95:
            break
    
    # Final validation at highest resolution
    if time() - start_time < budget_s * 0.05:
        try:
            final_h = best_h.copy()
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points/2) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
