import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    random.seed(seed)
    start_time = time()

        n_points = len(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    pop_size = 50
    elitism = int(0.3 * pop_size)
    mutation_rate = 0.1
    max_generations = 1000
    generation = 0
    
    # Population initialization
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    if 'initial_h_values' in globals():
        population = [h_start.copy()]
        population.extend([create_individual() for _ in range(pop_size - 1)])
    else:
        population = [create_individual() for _ in range(pop_size)]
    
    # Evolutionary algorithm with local refinement
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection
        elite = population[:elitism]
        
        # Crossover and mutation
        new_population = elite[:]
        
        # Generate offspring through crossover
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = elite[random.randint(0, len(elite)-1)]
            parent2 = elite[random.randint(0, len(elite)-1)]

            alpha = random.uniform(0.4, 0.6)
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += random.gauss(0, 0.02)
            
            # Ensure bounds and constraint
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        generation += 1
        
        # Local refinement step for top individuals
        if generation % 5 == 0:
            top_candidates = population[:10]
            for h in top_candidates:
                # Use SLSQP for local refinement
                def obj(h_vec):
                    return evaluate_erdos_solution(h_vec, dx)
                
                cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,

    best_h = best_h * (n_points/2) / np.sum(best_h)
    c5_final = evaluate_erdos_solution(best_h, dx)
    
    if c5_final < best_c5:
        best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
