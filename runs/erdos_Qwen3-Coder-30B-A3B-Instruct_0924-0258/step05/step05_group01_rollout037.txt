import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def create_individual():
        h = np.random.uniform(0, 1, n_points)
        return project_to_feasible(h)
    
    def mutate(h, rate=0.1):
        h = h.copy()
        for i in range(n_points):
            if random.random() < rate:
                h[i] += np.random.normal(0, 0.05)
        return project_to_feasible(h)
    
    def crossover(h1, h2):
        point = random.randint(1, n_points-1)
        h = np.concatenate([h1[:point], h2[point:]])
        return project_to_feasible(h)
    
    def evaluate_individual(h):
        return evaluate_erdos_solution(h, dx)
    
    # EA parameters
    pop_size = 50
    generations = 1000
    mutation_rate = 0.1
    
    population = [create_individual() for _ in range(pop_size)]
    fitness = [evaluate_individual(ind) for ind in population]
    
    best_in_run = float('inf')
    best_individual = None
    
    gen_count = 0
    while time() - start_time < budget_s * 0.95 and gen_count < generations:
        # Selection: tournament
        new_population = []
        for _ in range(pop_size):
            tournament = random.sample(list(zip(population, fitness)), 5)
            winner = min(tournament, key=lambda x: x[1])
            new_population.append(winner[0])
        
        # Crossover and mutation
        next_pop = []
        for i in range(0, pop_size, 2):
            parent1 = new_population[i]
            parent2 = new_population[(i+1) % pop_size]
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            next_pop.extend([mutate(child1, mutation_rate), mutate(child2, mutation_rate)])
        
        population = next_pop[:pop_size]
        fitness = [evaluate_individual(ind) for ind in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        current_best = fitness[current_best_idx]
        
        if current_best < best_in_run:
            best_in_run = current_best
            best_individual = population[current_best_idx].copy()
        
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[current_best_idx].copy()
        
        gen_count += 1
    
    # Final validation and refinement
    if time() - start_time < budget_s * 0.1:
        try:
            h_final = project_to_feasible(best_h)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
