import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Custom evolutionary approach with restarts
    pop_size = 50
    generations = 1000
    mutation_rate = 0.15
    elite_size = 5
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points) * 0.5 + 0.25  # Random in [0.25, 0.75]
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Add the initial solution
    population[0] = best_h.copy()
    
    # Evolution loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = objective(h)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Select elite
        elite = population[:elite_size]
        
        # Create new population
        new_population = elite[:]
        
        # Generate offspring through crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            idx1 = np.random.randint(0, pop_size//2)
            idx2 = np.random.randint(0, pop_size//2)
            parent1 = population[idx1] if fitness[idx1] < fitness[idx2] else population[idx2]
            
            idx3 = np.random.randint(0, pop_size//2)
            idx4 = np.random.randint(0, pop_size//2)
            parent2 = population[idx3] if fitness[idx3] < fitness[idx4] else population[idx4]
            
            # Crossover (order-based)
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mut_indices = np.random.choice(n_points, size=int(mutation_rate * n_points), replace=False)
                for i in mut_indices:
                    child[i] += np.random.normal(0, 0.05)
            
            # Clip and rescale
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Final refinement with scipy
    if time() - start_time < budget_s * 0.9:
        # Try several restarts with optimized initialization
        for _ in range(20):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Perturb the best solution
            h_test = best_h + np.random.normal(0, 0.01, n_points)
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points/2) / np.sum(h_test)
            
            # Local optimization
            try:
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res_opt = minimize(objective, h_test, method='SLSQP', 
                                bounds=bounds, constraints=cons_sum, 
                                options={'maxiter': 100, 'ftol': 1e-8})
                if res_opt.success:
                    h_refined = res_opt.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass

    return (list(best_h), best_c5, n_points)
