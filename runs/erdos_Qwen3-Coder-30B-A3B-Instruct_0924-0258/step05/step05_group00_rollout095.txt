import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Normalize initial h
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach
    resolutions = [35, 40, 50, 60]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to new resolution
        if res > n_points:
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            h_interp = np.interp(x_new, x_old, best_h)
        else:
            h_interp = best_h[:res].copy()
            if res < n_points:
                h_interp = np.pad(h_interp, (0, n_points - res), 'constant')
            else:
                h_interp = h_interp[:n_points]
        
        n_points = len(h_interp)
        dx = 2.0 / n_points
        
        # Normalize to correct sum
        h_interp = h_interp * (n_points/2) / np.sum(h_interp)
        
        # Convex relaxation with SDP
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add overlap constraints
            for k in range(1, n_points//2 + 1):
                if k < n_points:
                    # For shift k, compute correlation over valid indices
                    corr_term = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    constraints.append(corr_term <= best_c5 * dx + 1e-8)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass
        
        # Local refinement with SLSQP
        max_restarts = 10
        for i in range(max_restarts):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Random perturbation with less noise
            h_restart = h_interp + np.random.normal(0, 0.005, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-8})
                
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final validation
    try:
        h_final = np.clip(best_h, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
