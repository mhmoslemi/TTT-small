import numpy as np
from scipy.optimize import minimize
from time import time
from hashlib import sha256

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    max_iter = 1000
    max_stagnation = 100
    tabu_tenure = min(50, max(5, n_points // 4))
    mutation_scale = 0.04
    tabu_list = []
    stagnation_counter = 0
    current_best = best_c5
    
    # Hash function for tabu list
    def hash_state(h):
        return sha256(np.sort(h).tobytes()).hexdigest()
    
    # Perturbation function
    def perturb(h, scale=mutation_scale):
        h_new = h.copy()
        # Perturb 30% of elements
        indices = np.random.choice(n_points, size=int(0.3 * n_points), replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, scale)
        # Repair
        h_new = np.clip(h_new, 0, 1)
        # Scale to correct sum
        h_new = h_new * (n_points/2) / np.sum(h_new)
        return h_new
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate new candidate
        h_candidate = perturb(best_h, mutation_scale)
        
        # Check tabu list
        candidate_hash = hash_state(h_candidate)
        if candidate_hash in tabu_list:
            # Accept with low probability or if better than all tabu
            if np.random.random() > 0.9:
                pass  # proceed anyway
            else:
                continue
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Update best if improvement
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            stagnation_counter = 0
            current_best = best_c5
            
            # Local refinement every 10 improvements
            if len(tabu_list) % 10 == 0:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_candidate, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-8})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
        else:
            stagnation_counter += 1
            
        # Update tabu list
        tabu_list.append(candidate_hash)
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
        
        # Adjust mutation if stagnating
        if stagnation_counter > max_stagnation:
            mutation_scale *= 2
            stagnation_counter = 0
            
        # Early termination if good enough
        if best_c5 < 0.3807:
            break
    
    # Final refinement with some better parameters
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
