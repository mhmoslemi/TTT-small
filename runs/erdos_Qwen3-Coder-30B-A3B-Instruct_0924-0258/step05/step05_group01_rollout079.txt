import numpy as np

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Scale levels: progressively refine resolution
    scales = [35, 50, 75, 100]
    
    for i, scale in enumerate(scales):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to the desired scale
        if scale > n_points:
            h_current = np.interp(np.linspace(0, 2, scale), np.linspace(0, 2, n_points), best_h)
            n_points = scale
            dx = 2.0 / n_points
        else:
            h_current = best_h
            n_points = scale
            dx = 2.0 / n_points
            
        # Gradient descent optimizer with adaptive learning rate
        max_iter = 100 + i * 50
        alpha = 0.01
        last_improvement = 0
        improvement_threshold = 1e-6
        
        for iter_num in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Evaluate current objective
            current_c5 = evaluate_erdos_solution(h_current, dx)
            
            # Find the shift that contributes to the max
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            k_max = np.argmax(corr)
            
            # Gradient estimation by finite differences
            eps = 1e-5
            grad = np.zeros_like(h_current)
            
            for j in range(len(h_current)):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                
                c_plus = evaluate_erdos_solution(h_plus, dx)
                c_minus = evaluate_erdos_solution(h_minus, dx)
                
                if abs(c_plus - c_minus) > 1e-10:
                    grad[j] = (c_plus - c_minus) / (2 * eps)
            
            # Apply gradient step
            h_new = h_current - alpha * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Check if this improves
            new_c5 = evaluate_erdos_solution(h_new, dx)
            if new_c5 < current_c5:
                h_current = h_new
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_current.copy()
                last_improvement = 0
            else:
                last_improvement += 1
                if last_improvement > 20:
                    # Reduce step size if not improving
                    alpha *= 0.9
                    last_improvement = 0
                    
    return (list(best_h), best_c5, n_points)
