import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Load initial solution
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Sample shifts
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    
    # Initialize
    current_h = initial_h.copy()
    current_h = current_h * (n_points / 2) / np.sum(current_h)
    best_h = current_h.copy()
    best_c5 = evaluate_erdos_solution(current_h, 2.0 / n_points)
    
    # Iterative process
    iteration = 0
    max_iter = 50
    stagnation_count = 0
    last_best = best_c5
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iter:
        # Calculate all correlations for current h
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        # Take the part with valid shifts
        shift_idx = np.arange(len(corr) - n_points + 1, len(corr))
        corr_values = corr[shift_idx]
        
        # Find the worst shift
        max_idx = np.argmax(corr_values)
        max_correlation = corr_values[max_idx]
        max_shift = shifts[max_idx]
        
        # Add this constraint: correlation <= C5
        # For now we do a sequential approach for efficiency
        
        # Use convex optimization with this constraint
        h = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraints
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2, C5 >= 0]
        
        # For the worst shift, we add the constraint that the overlap does not exceed the current C5
        # This is a one-time constraint from this specific shift and will get updated in next iteration
        # We use the numerical value from computation above
        # Instead, we do a simpler but iterative approach
        # Let's optimize with the current best h and constraint
        # So the problem becomes:
        # Minimize C5
        # Subject to:
        # h[i] >= 0, h[i] <= 1, sum(h) = n_points / 2
        # For current max_shift: integral h(x) * (1 - h(x + max_shift)) dx <= C5
        
        # Temporarily fix h to current_h and check the overlap
        # Cast to normalize initial solution
        current_h_normalized = current_h.copy()
        current_h_normalized = current_h_normalized * (n_points / 2) / np.sum(current_h_normalized)
        
        # Now refine it by solving a local optimization problem
        # Use SLSQP method with standard constraints
        
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
            
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, current_h_normalized, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8})
            if res.success:
                new_h = res.x
                new_c5 = obj(new_h)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                current_h = new_h.copy()
            else:
                current_h = current_h_normalized.copy()
        except Exception as e:
            current_h = current_h_normalized.copy()
            
        # Scheme: every few iterations, try a global refinement with SLSQP again
        if iteration % 5 == 0:
            try:
                res = minimize(obj, current_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9})
                if res.success:
                    new_h = res.x
                    new_c5 = obj(new_h)
                    if new_c5 < best_c5:
                        best_h = new_h.copy()
                        best_c5 = new_c5
                        stagnation_count = 0
                    else:
                        stagnation_count += 1
                    current_h = new_h.copy()
            except Exception as e:
                pass
        
        # Check early termination
        if best_c5 < 0.3807:
            break
            
        iteration += 1
        
        # Update tracker
        if best_c5 < last_best:
            last_best = best_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count > 30:
                break
    
    # Final polishing
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0

        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-10})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
