import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1
    
    # Binary tabu search
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_best, 2.0/n_points)
    
    # Initial binary conversion
    h_best = np.round(h_best).astype(int)
    h_best = np.clip(h_best, 0, 1)
    target_sum = n_points // 2
    actual_sum = np.sum(h_best)
    if actual_sum != target_sum:
        # Adjust to achieve exact sum
        indices_zero = np.where(h_best == 0)[0]
        indices_one = np.where(h_best == 1)[0]
        if actual_sum < target_sum:
            # Need to add more ones
            n_add = target_sum - actual_sum
            for i in range(n_add):
                if indices_zero.size > 0:
                    idx = indices_zero[0]
                    h_best[idx] = 1
                    indices_zero = np.delete(indices_zero, 0)
        else:
            # Need to remove ones
            n_remove = actual_sum - target_sum
            for i in range(n_remove):
                if indices_one.size > 0:
                    idx = indices_one[-1]
                    h_best[idx] = 0
                    indices_one = indices_one[:-1]
    
    tabu_list = []
    tabu_tenure = 10
    max_no_improve = 100
    no_improve_count = 0
    shaking_freq = 50
    
    # Convert to binary + exact sum adjustment
    h_current = h_best.copy()
    dx = 2.0 / n_points
    
    while time() - start_time < budget_s * 0.9:
        h_current = h_best.copy()
        current_c5 = c5_best
        
        # Local neighbor generation and evaluation
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        # Generate neighbors by flipping one 1 to 0 and one 0 to 1
        ones = np.where(h_current == 1)[0]
        zeros = np.where(h_current == 0)[0]
        
        if len(ones) > 0 and len(zeros) > 0:
            for i in range(min(10, len(ones) * len(zeros))):
                # Pick one 1 and one 0
                idx_one = np.random.choice(ones)
                idx_zero = np.random.choice(zeros)
                # Create neighbor
                neighbor = h_current.copy()
                neighbor[idx_one] = 0
                neighbor[idx_zero] = 1
                
                # Evaluate neighbor
                c5 = evaluate_erdos_solution(neighbor, dx)
                if c5 < best_neighbor_c5:
                    best_neighbor = neighbor
                    best_neighbor_c5 = c5
        
        # Check tabu
        if best_neighbor is not None:
            # Determine move
            flip_one = np.where(h_current == 1)[0]
            flip_zero = np.where(h_current == 0)[0]
            move = (np.random.choice(flip_one), np.random.choice(flip_zero))
            
            if move in tabu_list:
                # Check aspiration
                if best_neighbor_c5 < c5_best:
                    pass  # Allow tabu move
                else:
                    continue  # Skip tabu
            else:
                # Add to tabu list
                tabu_list.append(move)
                if len(tabu_list) > tabu_tenure:
                    tabu_list.pop(0)
            
            # Update current solution
            h_current = best_neighbor
            current_c5 = best_neighbor_c5
            
            # Update global best
            if current_c5 < c5_best:
                h_best = h_current.copy()
                c5_best = current_c5
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            
        # Shaking
        if no_improve_count >= shaking_freq:
            # Perturb the solution
            h_current = h_best.copy()
            idxs_ones = np.where(h_current == 1)[0]
            idxs_zeros = np.where(h_current == 0)[0]
            n_flip = max(1, n_points // 20)
            flip_indices = []
            for _ in range(n_flip):
                if len(idxs_ones) > 0:
                    idx = np.random.choice(idxs_ones)
                    flip_indices.append(idx)
                    idxs_ones = np.delete(idxs_ones, np.where(idxs_ones == idx)[0])
                if len(idxs_zeros) > 0:
                    idx = np.random.choice(idxs_zeros)
                    flip_indices.append(idx)
                    idxs_zeros = np.delete(idxs_zeros, np.where(idxs_zeros == idx)[0])
            
            # Flip the selected bits
            for idx in flip_indices:
                h_current[idx] = 1 - h_current[idx]
            
            # Ensure sum constraint
            actual_sum = np.sum(h_current)
            if actual_sum != target_sum:
                if actual_sum < target_sum:
                    # Add ones
                    idxs_zeros = np.where(h_current == 0)[0]
                    for _ in range(target_sum - actual_sum):
                        if idxs_zeros.size > 0:
                            idx = idxs_zeros[0]
                            h_current[idx] = 1
                            idxs_zeros = idxs_zeros[1:]
                else:
                    # Remove ones
                    idxs_ones = np.where(h_current == 1)[0]
                    for _ in range(actual_sum - target_sum):
                        if idxs_ones.size > 0:
                            idx = idxs_ones[-1]
                            h_current[idx] = 0
                            idxs_ones = idxs_ones[:-1]
            
            # Re-evaluate
            current_c5 = evaluate_erdos_solution(h_current, dx)
            if current_c5 < c5_best:
                h_best = h_current.copy()
                c5_best = current_c5
                no_improve_count = 0
            else:
                no_improve_count += 1
            
        # Early stop
        if no_improve_count >= max_no_improve:
            break
    
    # Final validation (ensure binary)
    h_final = np.round(h_best).astype(int)
    h_final = np.clip(h_final, 0, 1)
    actual_sum = np.sum(h_final)
    if actual_sum != target_sum:
        idxs_zeros = np.where(h_final == 0)[0]
        idxs_ones = np.where(h_final == 1)[0]
        if actual_sum < target_sum:
            n_add = target_sum - actual_sum
            for _ in range(n_add):
                if idxs_zeros.size > 0:
                    idx = idxs_zeros[0]
                    h_final[idx] = 1
                    idxs_zeros = np.delete(idxs_zeros, 0)
        else:
            n_remove = actual_sum - target_sum
            for _ in range(n_remove):
                if idxs_ones.size > 0:
                    idx = idxs_ones[-1]
                    h_final[idx] = 0
                    idxs_ones = idxs_ones[:-1]
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
