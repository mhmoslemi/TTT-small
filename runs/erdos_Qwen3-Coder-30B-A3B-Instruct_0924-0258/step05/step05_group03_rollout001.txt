import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 35
        h0 = np.ones(n_points) * 0.5
        
    # Ensure sum constraint
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Parameter configuration
    max_iter = 50
    shift_samples = 100
    shift_step = 2.0 / shift_samples
    
    # Initial handling
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Cut-plane method with CVXPY
    # Variables
    h = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
        C5 >= 0,
    ]
    
    # Objective
    objective = cp.Minimize(C5)
    
    # Optimization loop with cutting planes
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(best_h, dx)
        
        # Sample shifts and add constraints
        shift_indices = np.arange(shift_samples)
        shifts = shift_indices * shift_step
        
        # Sample overlaps to add one constraint at a time
        overlaps = []
        for k in shifts:
            # k in [0, 2], need to convert to indices
            k_idx = int(k / dx)
            if k_idx < n_points:
                # Compute overlap at this shift
                corr_val = np.sum(best_h[:n_points-k_idx] * (1 - best_h[k_idx:]))
                overlaps.append(corr_val * dx)
            else:
                overlaps.append(0)
                
        max_overlap = max(overlaps)
        
        # Add constraint (one at a time, focusing on max)
        constraints.append(C5 >= max_overlap)
        
        # Solve convex relaxation
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
        except:
            # Fallback to previous solution
            break
            
        if prob.status == cp.OPTIMAL:
            h_val = h.value
            if np.all(0 <= h_val) and np.all(h_val <= 1):
                # Ensure sum constraint
                h_val = h_val * (n_points / 2) / np.sum(h_val)
                new_c5 = evaluate_erdos_solution(h_val, dx)
                if new_c5 < best_c5:
                    best_h = h_val.copy()
                    best_c5 = new_c5
                
                # Early termination for target
                if best_c5 < 0.3807:
                    break
        else:
            break
            
        # Clean up constraints for next iteration
        constraints = constraints[:4]  # Keep only original constraints
        
    # Local refinement with SLSQP
    def obj(x):
        return evaluate_erdos_solution(x, dx)
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
