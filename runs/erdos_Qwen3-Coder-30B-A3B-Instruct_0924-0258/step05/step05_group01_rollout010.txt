import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary algorithm parameters
    pop_size = 50
    max_generations = 1000
    mutation_rate = 0.15
    crossover_rate = 0.8
    tournament_size = 3
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Add parent if available
    if 'initial_h_values' in globals():
        population[0] = h_start.copy()
    
    # Main EA loop
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time() - start_time < budget_s * 0.9 and generation < max_generations:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Reset if no improvement
        if no_improvement_count > max_no_improvement:
            if 'initial_h_values' in globals():
                population[0] = h_start.copy()
            else:
                population[0] = np.ones(n_points) * 0.5
            no_improvement_count = 0
        
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < pop_size else selected[0]
            
            if np.random.random() < crossover_rate:
                # One-point crossover
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.random() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                    child2[j] += np.random.normal(0, 0.05)
            
            # Ensure feasibility
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        # Keep population size consistent
        population = new_population[:pop_size]
        
        # Resolution upgrade every 100 generations
        if generation % 100 == 0 and generation > 0:
            if n_points < 50:
                n_points += 5
                dx = 2.0 / n_points
                # Interpolate population to new resolution
                new_population = []
                for h in population:
                    h_new = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
                    h_new = h_new * (n_points/2) / np.sum(h_new)
                    new_population.append(h_new)
                population = new_population
        
        generation += 1
    
    # Final optimization with local search
    if time() - start_time < budget_s * 0.1:
        try:
            # Use best solution as starting point
            h_opt = best_h.copy()
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points/2) / np.sum(h_opt)
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < best_c5:
                best_c5 = c5_opt
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
