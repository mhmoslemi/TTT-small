import numpy as np
from time import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with triangular shape
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution simulation annealing with cooling schedule
    scale_levels = [35, 50, 75, 100]
    scale_levels = [s for s in scale_levels if s <= 200]  # Limit to reasonable size
    
    current_best_h = h_start.copy()
    current_best_c5 = best_c5
    
    for scale in scale_levels:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to current scale
        if scale > n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            h_interp = np.interp(x_new, x_orig, current_best_h)
            h_current = h_interp
        else:
            h_current = current_best_h.copy()
            
        n_points = scale
        dx = 2.0 / n_points
        
        # Simulated Annealing parameters
        T0 = 0.1
        alpha = 0.99
        T = T0
        max_iter = int(3000 * (200 / scale))  # Scale iterations with resolution
        
        current_h = h_current.copy()
        current_c5 = evaluate_erdos_solution(current_h, dx)
        
        # SA with bit flipping
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate neighbor: flip some elements
            h_new = current_h.copy()
            flip_indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
            for i in flip_indices:
                # Flip a value with entropy
                perturbation = np.random.normal(0, 0.05)
                h_new[i] += perturbation
                
            # Clamp to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Rescale to meet constraint sum(h) = n/2
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate new solution
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            delta = c5_new - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                current_h = h_new
                current_c5 = c5_new
                
                if current_c5 < current_best_c5:
                    current_best_h = current_h.copy()
                    current_best_c5 = current_c5
                    
                    if current_best_c5 < best_c5:
                        best_h = current_best_h.copy()
                        best_c5 = current_best_c5
            
            # Cooling schedule
            T = alpha * T
            T = max(T, 1e-6)
    
    # Final validation
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, len(best_h))
