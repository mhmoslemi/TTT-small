import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Check if initial_h_values is available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Early termination if we already have a good solution
    current_best_h = initial_h.copy()
    current_c5 = evaluate_erdos_solution(current_best_h, 2.0 / n_points)
    
    # If already good, return early
    if current_c5 < 0.3807:
        return (list(current_best_h), current_c5, n_points)
    
    # Generate a set of shift points for constraint generation
    shift_points = np.linspace(0, 2, 100)
    
    # Solve via convex relaxation with cutting planes
    max_iterations = 100
    improvement_threshold = 1e-5
    
    # Use a smaller number of points for faster computation
    n = n_points
    
    # Initialize variables for CVXPY
    h_vars = cp.Variable(n)
    C5_var = cp.Variable()
    
    # Constraints for h
    constraints = [h_vars >= 0, h_vars <= 1]
    
    # Sum constraint
    constraints.append(cp.sum(h_vars) == n / 2.0)
    
    # Add initial constraints from the initial solution
    for k in shift_points:
        shift_idx = int(k * n / 2.0)
        if shift_idx < n:
            # Compute the overlap constraint as linear inequality
            # This is an approximation for the original constraint
            # We use the current h to get an estimate of overlap
            overlap = 0.0
            for i in range(n):
                j = (i + shift_idx) % n
                overlap += h_vars[i] * (1 - h_vars[j])
            constraints.append(overlap <= C5_var)
    
    # Objective
    objective = cp.Minimize(C5_var)
    
    # Solve initial problem with a few constraints
    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(solver=cp.ECOS, verbose=False)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            current_h = h_vars.value
            current_c5 = C5_var.value
            # Round to avoid numerical issues
            current_h = np.clip(current_h, 0, 1)
            # Normalize
            current_h = current_h * (n / 2) / np.sum(current_h)
        else:
            # Fallback to initial if CVXPY fails
            current_h = initial_h.copy()
            current_c5 = evaluate_erdos_solution(current_h, 2.0 / n_points)
    except:
        current_h = initial_h.copy()
        current_c5 = evaluate_erdos_solution(current_h, 2.0 / n_points)
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    if best_c5 < 0.3807:
        return (list(best_h), best_c5, n_points)
    
    # Iterative refinement
    last_c5 = current_c5 + 1e3
    iteration = 0
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Evaluate single shift to get the max overlap direction
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_overlap_idx = np.argmax(corr)
        max_overlap_shift = max_overlap_idx - (n - 1)
        
        # Add constraint for this shift direction
        # Determine indices to use for this constraint
        shift = max_overlap_shift
        if shift < 0:
            h1 = current_h[-shift:]
            h2 = current_h[:shift]
        elif shift >= 0:
            h1 = current_h[:-shift]
            h2 = current_h[shift:]
        # Pad with zeros if needed
        if shift > 0:
            h1_padded = np.pad(h1, (0, shift), mode='constant', constant_values=0)
            h2_padded = np.pad(h2, (shift, 0), mode='constant', constant_values=0)
        else:
            h1_padded = np.pad(h1, (-shift, 0), mode='constant', constant_values=0)
            h2_padded = np.pad(h2, (0, -shift), mode='constant', constant_values=0)
        
        # Constraint: sum h[i] * (1 - h[i+shift]) <= C5_var
        overlap = np.sum(h1_padded * (1 - h2_padded))
        
        # Recompile problem with this constraint
        h_vars = cp.Variable(n)
        C5_var = cp.Variable()
        
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n / 2.0,
            cp.sum(cp.multiply(h_vars[:-shift], 1 - h_vars[shift:])) <= C5_var,
        ]
        
        # Add previous constraints for stability
        objective = cp.Minimize(C5_var)
        problem = cp.Problem(objective, constraints)
        
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_vars.value
                new_c5 = C5_var.value
                new_h = np.clip(new_h, 0, 1)
                # Normalize
                new_h = new_h * (n / 2) / np.sum(new_h)
                if new_c5 < current_c5:
                    current_h = new_h.copy()
                    current_c5 = new_c5
                    if new_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
        except:
            pass
        
        if abs(last_c5 - current_c5) < improvement_threshold:
            break
        last_c5 = current_c5
        iteration += 1
    
    # Final local optimization
    def obj(h):
        return evaluate_erdos_solution(h, 2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - n / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n)]
    
    try:
        res = minimize(obj, current_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            current_h = res.x
            current_c5 = obj(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    except:
        pass
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    return (list(best_h), final_c5, n_points)
