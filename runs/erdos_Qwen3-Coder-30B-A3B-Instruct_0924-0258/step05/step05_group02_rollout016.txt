import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse optimization first
    n_coarse = 20
    dx_coarse = 2.0 / n_coarse
    
    def coarse_relaxation(h_coarse):
        # Relaxation: maintain sum constraint and upper bound on C5
        try:
            h_relaxed = cp.Variable(n_coarse)
            constraints = [
                cp.sum(h_relaxed) == n_coarse / 2.0,
                h_relaxed >= 0,
                h_relaxed <= 1
            ]
            
            # Constraint for a subset of shifts - not all but enough to constrain
            for k in range(1, min(10, n_coarse)):  
                if k < n_coarse // 2:
                    corr_sum = cp.sum(cp.multiply(h_relaxed[:n_coarse-k], 1 - h_relaxed[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_coarse-k]))
                constraints.append(corr_sum <= best_c5 * dx_coarse + 1e-6)
            
            prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-7)
            if prob.status == cp.OPTIMAL:
                return h_relaxed.value
        except Exception:
            pass
        return h_coarse
    
    # Refinement loop with adaptive gradient-based steps
    # Start with coarse grid
    h_coarse = np.ones(n_coarse) * (n_coarse / 2.0) / n_coarse
    h_coarse = coarse_relaxation(h_coarse)
    
    # Interpolate to high-resolution
    h_interpolated = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, n_coarse), 
                              h_coarse)
    h_interpolated = h_interpolated * (n_points/2) / np.sum(h_interpolated)
    h_interpolated = np.clip(h_interpolated, 0, 1)
    
    # Start with this interpolated solution
    h_current = h_interpolated.copy()
    c5_current = evaluate_erdos_solution(h_current, dx)
    
    if c5_current < best_c5:
        best_h = h_current.copy()
        best_c5 = c5_current
        
    # Fine-grained optimization
    opt_steps = 0
    no_improvement_iters = 0
    max_no_improvement = 100
    lr = 0.02
    mutation_rate = 0.1
    
    def compute_gradient(h):
        # Gradient estimation using finite differences for simplicity
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad
    
    while time() - start_time < budget_s * 0.95:
        opt_steps += 1
        
        # Get step direction
        grad = compute_gradient(h_current)
        h_new = h_current - lr * grad
        
        # Apply constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept improvement or apply mutation
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            no_improvement_iters = 0
        else:
            no_improvement_iters += 1
            # Apply mutation with increasing probability if stuck
            if np.random.random() < mutation_rate + (no_improvement_iters / 1000.0):
                mutation = np.random.normal(0, 0.05, n_points)
                h_mutation = h_current + mutation
                h_mutation = np.clip(h_mutation, 0, 1)
                h_mutation = h_mutation * (n_points/2) / np.sum(h_mutation)
                c5_mutation = evaluate_erdos_solution(h_mutation, dx)
                if c5_mutation < c5_current:
                    h_current = h_mutation
                    c5_current = c5_mutation
                    no_improvement_iters = 0
        
        # Update best
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        # Adaptive learning rate
        if no_improvement_iters > 30:
            lr = max(lr * 0.95, 1e-5)
            
        if no_improvement_iters > max_no_improvement:
            break
            
    # Final adjustment
    try:
        h_final = np.clip(best_h, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
