import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

    # Convert to binary if needed, respecting sum constraint
    h_best = h_start.copy()
    dx = 2.0 / n_points
    current_sum = np.sum(h_best)
    target_sum = n_points / 2.0
    
    # Round and scale to satisfy constraint
    h_binary = (h_best > 0.5).astype(float)
    scale_factor = target_sum / np.sum(h_binary)
    h_binary = np.round(h_binary * scale_factor)
    if np.sum(h_binary) != target_sum:
        # Adjust to ensure exact sum
        diff = int(target_sum - np.sum(h_binary))
        if diff > 0:
            # Add ones
            ones_indices = np.where(h_binary == 0)[0]
            for i in range(diff):
                if i < len(ones_indices):
                    h_binary[ones_indices[i]] = 1
        else:
            # Remove ones
            ones_indices = np.where(h_binary == 1)[0]
            for i in range(-diff):
                if i < len(ones_indices):
                    h_binary[ones_indices[i]] = 0
    h_best = h_binary
    
    c5_best = evaluate_erdos_solution(h_best, dx)
    
    # Tabu search for binary variables
    tabu_list = []
    tabu_tenure = 10
    max_no_improvement = 100
    no_improvement_count = 0
    
    best_h = h_best.copy()
    best_c5 = c5_best
    
    n_iter = 0
    while time() - start_time < budget_s * 0.95:
        n_iter += 1
        
        # Generate neighbor by flipping one 1 to 0 and one 0 to 1
        h_neighbor = best_h.copy()
        
        # Select two indices to flip
        ones_indices = np.where(h_neighbor == 1)[0]
        zeros_indices = np.where(h_neighbor == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            # If not enough 1s or 0s, adjust
            if len(ones_indices) == 0:
                # Make some 1s to 0s
                indices = np.random.choice(n_points, int(n_points/4), replace=False)
                for i in indices:
                    if h_neighbor[i] == 1:
                        h_neighbor[i] = 0
            else:
                # Make some 0s to 1s
                indices = np.random.choice(n_points, int(n_points/4), replace=False)
                for i in indices:
                    if h_neighbor[i] == 0:
                        h_neighbor[i] = 1

            ones_indices = np.where(h_neighbor == 1)[0]
            zeros_indices = np.where(h_neighbor == 0)[0]
        
        # Flip one 1 to 0 and one 0 to 1
        idx_1 = np.random.choice(ones_indices)
        idx_0 = np.random.choice(zeros_indices)
        
        # Check if this move is tabu
        if (idx_1, idx_0) in tabu_list or (idx_0, idx_1) in tabu_list:
            # Check aspiration
            aspiration = False
            # For now, just proceed with alternative approach if tabu
            if np.sum(h_neighbor) == n_points/2:
                h_neighbor[idx_1] = 0
                h_neighbor[idx_0] = 1
            else:
                # Try to find a valid pair
                valid_pairs = [(i,j) for i in ones_indices for j in zeros_indices if i != j]
                if valid_pairs:
                    idx_1, idx_0 = valid_pairs[np.random.randint(len(valid_pairs))]
                    h_neighbor[idx_1] = 0
                    h_neighbor[idx_0] = 1
                else:
                    # Fallback: make a single random move
                    h_neighbor = best_h.copy()

                        if len(ones) > 0:

                            h_neighbor[idx_to_zero] = 0
                    elif np.sum(h_neighbor) < n_points/2:
                        zeros = np.where(h_neighbor == 0)[0]
                        if len(zeros) > 0:
                            idx_to_one = np.random.choice(zeros)
                            h_neighbor[idx_to_one] = 1
        else:
            # Do the flip
            h_neighbor[idx_1] = 0
            h_neighbor[idx_0] = 1
        
        # Evaluate neighbor
        c5_neighbor = evaluate_erdos_solution(h_neighbor, dx)
        
        # Update tabu list
        tabu_list.append((idx_1, idx_0))
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
        
        # Check if neighbor improves
        if c5_neighbor < best_c5:
            best_h = h_neighbor.copy()
            best_c5 = c5_neighbor
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Shaking - restart if no improvement
        if no_improvement_count > 50:
            no_improvement_count = 0
            # Perturb the best solution
            h_perturbed = best_h.copy()
            n_flip = max(1, int(n_points * 0.05))  # 5% of points
            flip_indices = np.random.choice(n_points, n_flip, replace=False)
            for idx in flip_indices:
                h_perturbed[idx] = 1 - h_perturbed[idx]
            
            # Ensure sum constraint
            current_sum = np.sum(h_perturbed)
            target_sum = n_points / 2.0
            diff = int(target_sum - current_sum)
            if diff > 0:
                zeros = np.where(h_perturbed == 0)[0]
                if len(zeros) >= diff:
                    for i in range(diff):
                        h_perturbed[zeros[i]] = 1
            elif diff < 0:
                ones = np.where(h_perturbed == 1)[0]
                if len(ones) >= -diff:
                    for i in range(-diff):
                        h_perturbed[ones[i]] = 0
            # Update best if improved
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_perturbed
        
        # Check for early exit conditions
        if no_improvement_count > max_no_improvement:
            break
    
    # Final computation
    try:
        final_c5 = evaluate_erdos_solution(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    except:
        pass
    
    # Ensure final solution is valid binary
    best_h = np.round(best_h).astype(int)
    # Ensure sum constraint exactly
    current_sum = np.sum(best_h)
    diff = int(n_points/2) - int(current_sum)
    if diff > 0:
        # Change some 0s to 1s
        zeros = np.where(best_h == 0)[0]
        for i in range(min(diff, len(zeros))):
            best_h[zeros[i]] = 1
    elif diff < 0:
        # Change some 1s to 0s
        ones = np.where(best_h == 1)[0]
        for i in range(min(-diff, len(ones))):
            best_h[ones[i]] = 0
    
    return (list(best_h), best_c5, n_points)
