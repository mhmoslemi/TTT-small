import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use the initial construction if available

    # Normalize the initial solution to satisfy the integral constraint
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    dx = 2.0 / n_points
    shift_samples = np.linspace(0, 2, 100, endpoint=False)
    
    # Convex relaxation approach
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Iterative refinement using cutting planes
    max_iterations = 50
    stagnation_limit = 50
    stagnation_count = 0
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for current solution and identify worst case
        overlaps = []
        for k in shift_samples:
            idx = int(k / dx)
            if idx >= n_points:
                idx = n_points - 1
            overlap = np.sum(best_h * np.roll(1 - best_h, -idx)) * dx
            overlaps.append(overlap)
            
        max_overlap = max(overlaps)
        max_idx = np.argmax(overlaps)
        max_shift = shift_samples[max_idx]
                
        # Add constraint (cutting plane)
        # For convex relaxation, we use information from the current worst shift
        # We filter constraints to prevent saturation in optimization
        
        # We don't optimize the full convex relaxation here but we use the information
        # to improve our guess via local search
        
        # Perform local search from current solution
        def obj(x):
            return evaluate_erdos_solution(x, dx)
            
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})

                new_h = res.x
                new_c5 = obj(new_h)
                if new_c5 < best_c5:

                    best_c5 = new_c5
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
        except:
            stagnation_count += 1
            
        # Early termination if we have reached the target
        if best_c5 < 0.3807:
            break
            
        if stagnation_count > stagnation_limit:
            # Perturb the solution and continue
            noise = np.random.normal(0, 0.01, n_points)
            best_h = np.clip(best_h + noise, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            stagnation_count = 0
    
    # Final local refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
            
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}

            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    # Recheck final solution to ensure constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
