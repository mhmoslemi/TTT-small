import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based evolutionary strategy
    pop_size = 50
    max_gens = 1000
    mutation_rate = 0.15
    crossover_rate = 0.8
    tournament_size = 3
    generations_without_improvement = 0
    max_no_improve = 50
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    current_resolution = n_points
    gen = 0
    while time() - start_time < budget_s * 0.95 and gen < max_gens:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Find best in current generation
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
            generations_without_improvement = 0
        else:
            generations_without_improvement += 1
            
        # Reset if no improvement
        if generations_without_improvement >= max_no_improve:
            h_reset = h_start.copy()
            h_reset = h_reset * (n_points/2) / np.sum(h_reset)
            best_h = h_reset.copy()
            best_c5 = evaluate_erdos_solution(best_h, dx)
            generations_without_improvement = 0
            
        # Selection + reproduction
        new_population = []
        
        # Keep best individual
        new_population.append(population[min_fitness_idx].copy())
        
        while len(new_population) < pop_size:
            # Tournament selection
            idx1 = np.random.choice(pop_size, tournament_size)
            idx2 = np.random.choice(pop_size, tournament_size)
            sel1 = idx1[np.argmin([fitness[i] for i in idx1])]
            sel2 = idx2[np.argmin([fitness[i] for i in idx2])]
            
            # Crossover
            if np.random.random() < crossover_rate:
                point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([population[sel1][:point], population[sel2][point:]])
                child2 = np.concatenate([population[sel2][:point], population[sel1][point:]])
            else:
                child1 = population[sel1].copy()
                child2 = population[sel2].copy()
                
            # Mutation
            for h in [child1, child2]:
                for i in range(n_points):
                    if np.random.random() < mutation_rate:
                        h[i] += np.random.normal(0, 0.03)
                        
                # Enforce bounds and sum constraint
                h = np.clip(h, 0, 1) 
                h = h * (n_points/2) / np.sum(h)
                new_population.append(h)
                
                if len(new_population) >= pop_size:
                    break
        
        population = new_population[:pop_size]
        
        # Resolution increase
        if gen > 0 and gen % 100 == 0 and current_resolution < 50:
            current_resolution += 5
            dx = 2.0 / current_resolution
            # Interpolate and rescale
            new_population = []
            for h in population:
                h_new = np.interp(np.linspace(0, 2, current_resolution), 
                                np.linspace(0, 2, n_points), h)
                h_new = h_new * (current_resolution/2) / np.sum(h_new)
                new_population.append(h_new)
            population = new_population
            n_points = current_resolution
            
        gen += 1

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
