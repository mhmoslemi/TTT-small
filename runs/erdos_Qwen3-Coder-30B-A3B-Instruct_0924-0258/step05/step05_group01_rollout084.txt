import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a triangular function as initial guess
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    current_h = h_start.copy()
    current_c5 = evaluate_erdos_solution(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T0 = 0.1
    alpha = 0.99
    max_iter_per_temp = 500
    temp_steps = 50
    
    # Multi-resolution scale levels
    scales = [35, 50, 75, 100]
    
    for scale in scales:
        if scale > n_points:
            # Interpolate to higher resolution
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            h_interp = np.interp(x_new, x_orig, current_h)
            current_h = h_interp
            n_points = scale
            dx = 2.0 / n_points
        else:
            # Downsample if needed
            if scale < n_points:
                # Average pooling
                stride = n_points // scale
                h_down = np.array([np.mean(current_h[i:i+stride]) 
                                 for i in range(0, n_points, stride)])
                if len(h_down) > scale:
                    h_down = h_down[:scale]
                current_h = h_down
                n_points = scale
                dx = 2.0 / n_points
            else:
                pass  # same size
        
        # Simulated Annealing at current scale
        current_temp = T0
        for _ in range(temp_steps):
            if time() - start_time > budget_s * 0.95:
                break

            for _ in range(max_iter_per_temp):
                if time() - start_time > budget_s * 0.95:
                    break

                # Random neighbor generation
                h_new = current_h.copy()
                # Flip bits with 10% probability
                mask = np.random.random(n_points) < 0.1
                h_new[mask] = 1 - h_new[mask]
                
                # Clip to feasible region
                h_new = np.clip(h_new, 0, 1)
                
                # Rescale to ensure integral constraint
                h_new = h_new * (n_points/2) / np.sum(h_new)
                
                # Evaluate new solution
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                # Accept or reject based on simulated annealing criterion
                delta = c5_new - current_c5
                if delta < 0 or np.random.random() < math.exp(-delta / current_temp):
                    current_h = h_new
                    current_c5 = c5_new
                    
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
            
            # Cooling schedule
            current_temp *= alpha
        
        if time() - start_time > budget_s * 0.95:
            break
    
    # Final refinement with local optimization
    if time() - start_time < budget_s * 0.9:
        final_h = best_h.copy()
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points/2) / np.sum(final_h)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
