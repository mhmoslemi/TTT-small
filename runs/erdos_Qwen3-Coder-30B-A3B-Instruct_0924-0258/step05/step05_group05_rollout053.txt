import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = min(50, max(5, n_points // 5))
    max_iter = 1000
    max_stagnation = 100
    mutation_scale = 0.04
    stagnation_counter = 0
    
    # Initialize the best solution
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Avoid revisiting similar states
    def hash_state(h):
        return hash(tuple(np.round(h, decimals=6)))
    
    # Initial tabu list
    tabu_list.append(hash_state(current_h))
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor solution
        new_h = current_h.copy()
        indices = random.sample(range(n_points), max(1, int(0.3 * n_points)))
        for i in indices:
            new_h[i] += random.gauss(0, mutation_scale)
        
        # Clip and rescale to maintain constraints
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Check tabu list
        new_hash = hash_state(new_h)
        is_tabu = new_hash in tabu_list
        
        # Accept if better
        if new_c5 < current_c5 or not is_tabu:
            current_h = new_h
            current_c5 = new_c5
            
            # Update tabu list
            tabu_list.append(new_hash)
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
            
            # Update best
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                stagnation_counter = 0
                
                # Early exit if target reached
                if best_c5 < 0.3807:
                    break
            else:
                stagnation_counter += 1
        else:
            stagnation_counter += 1
        
        # Adjust mutation rate if stagnating
        if stagnation_counter > max_stagnation:
            mutation_scale = min(0.5, mutation_scale * 1.2)
            stagnation_counter = 0
    
    # Final local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
