import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Set up shift sampling
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    
    # Cutting-plane method to solve convex relaxation
    max_iter = 50
    tol = 1e-5
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        
        # Early termination if target achieved
        if best_c5 < 0.3807:
            break
            
        # Sample shifts to get max overlap
        overlaps = []
        for k in shifts:
            # Compute overlap for shift k
            k_int = int(k * n_points / 2)
            if k_int == 0:
                overlap = np.sum(best_h * (1 - best_h)) * (2.0 / n_points)
            else:
                # Convolution approximated
                h_shifted = np.roll(best_h, k_int)
                overlap = np.sum(best_h * (1 - h_shifted)) * (2.0 / n_points)
            overlaps.append(overlap)
        
        # Find max overlap
        max_idx = np.argmax(overlaps)
        max_shift = shifts[max_idx]
        max_overlap = overlaps[max_idx]
        
        # Solve convex relaxation
        h_vars = cp.Variable(n_points)
        C5 = cp.Variable()
        
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2,
            C5 >= 0
        ]
        
        # Add constraint for current max overlap
        k_int = int(max_shift * n_points / 2)
        if k_int == 0:
            overlap_expr = cp.sum(h_vars * (1 - h_vars))
        else:
            h_shifted = cp.Variable(n_points)
            constraints.append(h_shifted == cp.roll(h_vars, k_int))
            overlap_expr = cp.sum(h_vars * (1 - h_shifted))
            
        constraints.append(overlap_expr <= C5 * n_points / 2)
        
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_vars.value
                if h_new is not None:
                    h_new = np.array(h_new)
                    h_new = np.clip(h_new, 0, 1)
                    # Renormalize
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    best_h = h_new
        except:
            pass
    
    # Final SLSQP refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
