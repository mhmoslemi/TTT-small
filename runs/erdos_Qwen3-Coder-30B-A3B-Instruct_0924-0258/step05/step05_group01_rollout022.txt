import numpy as np
from scipy.optimize import minimize
from time import time
import copy

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Multi-resolution local search with restarts
    max_restarts = 50
    current_n = n_points
    h_current = best_h.copy()
    
    # Try different resolutions
    resolutions = [30, 35, 40, 45, 50]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > current_n:
            # Interpolate to higher resolution
            h_new = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, current_n), h_current)
            current_n = res
            dx = 2.0 / current_n
        else:
            h_new = h_current
            current_n = res
            dx = 2.0 / current_n
            
        # Local optimization with restarts
        restart_count = 0
        h_current = h_new.copy()
        
        while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
            if time() - start_time > budget_s * 0.95:
                break
                
            # Random perturbation
            h_restart = h_current + np.random.normal(0, 0.03, current_n)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (current_n/2) / np.sum(h_restart)
            
            # SLSQP local optimization
            try:
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(current_n)]
                res_opt = minimize(objective, h_restart, method='SLSQP', 
                                bounds=bounds, constraints=cons_sum, 
                                options={'maxiter': 300, 'ftol': 1e-9})
                if res_opt.success:
                    h_refined = res_opt.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        restart_count = 0  # Reset counter on improvement
                        h_current = h_refined.copy()
                else:
                    restart_count += 1
            except Exception:
                restart_count += 1
                
            # Adaptive perturbation scale based on progress
            if restart_count % 10 == 0 and restart_count > 0:
                # Reduce perturbation size if stuck
                if restart_count > 20:
                    h_restart = h_current + np.random.normal(0, 0.01, current_n)
                else:
                    h_restart = h_current + np.random.normal(0, 0.015, current_n)
                h_restart = np.clip(h_restart, 0, 1)
                h_restart = h_restart * (current_n/2) / np.sum(h_restart)
                h_current = h_restart
                
    # Final verification
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (current_n/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, current_n)
