import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    pop_size = 50
    generations = 1000
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_k = 5
    
    # GA with resolution adaptation
    current_resolution = n_points
    max_generations = generations
    
    # Initialize population
    def create_individual():
        h = np.random.rand(current_resolution)
        h = h * (current_resolution/2) / np.sum(h)
        return h
    
    population = [create_individual() for _ in range(pop_size)]
    if 'initial_h_values' in globals():
        population[0] = h_start.copy()
        population[0] = population[0] * (current_resolution/2) / np.sum(population[0])
    
    # Main GA loop
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
        
        # Selection (tournament)
        def tournament_select():
            tournament = random.sample(list(zip(population, fitness)), tournament_k)
            tournament.sort(key=lambda x: x[1])
            return tournament[0][0]
        
        # Crossover (single point)
        def crossover(p1, p2):
            if random.random() > crossover_rate:
                return p1.copy(), p2.copy()
            
            point = random.randint(1, len(p1) - 1)
            child1 = np.concatenate([p1[:point], p2[point:]])
            child2 = np.concatenate([p2[:point], p1[point:]])
            
            # Repair
            child1 = child1 * (current_resolution/2) / np.sum(child1)
            child2 = child2 * (current_resolution/2) / np.sum(child2)
            
            return child1, child2
        
        # Mutation (Gaussian noise)
        def mutate(ind):
            mutated = ind.copy()
            for i in range(len(mutated)):
                if random.random() < mutation_rate:
                    mutated[i] += np.random.normal(0, 0.02)
            mutated = np.clip(mutated, 0, 1)
            mutated = mutated * (current_resolution/2) / np.sum(mutated)
            return mutated
        
        # Create new generation
        new_population = []
        new_population.append(population[min_fitness_idx])  # Elitism
        
        while len(new_population) < pop_size:
            parent1 = tournament_select()
            parent2 = tournament_select()
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
    
    # Resolution adaptation
    if time() - start_time < budget_s * 0.8 and best_c5 > 0.3808:
        # Increase resolution
        current_resolution = 40
        dx = 2.0 / current_resolution
        best_h = np.interp(np.linspace(0, 2, current_resolution),

        h_start = best_h.copy()
        
        # Re-initialize with smaller population
        pop_size = 20

        population = [create_individual() for _ in range(pop_size)]

            if time() - start_time > budget_s * 0.95:
                break
            
            fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
            min_fitness_idx = np.argmin(fitness)
            if fitness[min_fitness_idx] < best_c5:
                best_h = population[min_fitness_idx].copy()
                best_c5 = fitness[min_fitness_idx]
            
            def tournament_select():
                tournament = random.sample(list(zip(population, fitness)), tournament_k)
                tournament.sort(key=lambda x: x[1])
                return tournament[0][0]
            
            def crossover(p1, p2):
                if random.random() > crossover_rate:
                    return p1.copy(), p2.copy()
                point = random.randint(1, len(p1) - 1)
                child1 = np.concatenate([p1[:point], p2[point:]])
                child2 = np.concatenate([p2[:point], p1[point:]])
                child1 = child1 * (current_resolution/2) / np.sum(child1)
                child2 = child2 * (current_resolution/2) / np.sum(child2)
                return child1, child2
            
            def mutate(ind):
                mutated = ind.copy()
                for i in range(len(mutated)):
                    if random.random() < mutation_rate:
                        mutated[i] += np.random.normal(0, 0.02)
                mutated = np.clip(mutated, 0, 1)
                mutated = mutated * (current_resolution/2) / np.sum(mutated)
                return mutated
            
            new_population = []
            new_population.append(population[min_fitness_idx])
            
            while len(new_population) < pop_size:
                parent1 = tournament_select()
                parent2 = tournament_select()
                child1, child2 = crossover(parent1, parent2)
                child1 = mutate(child1)
                child2 = mutate(child2)
                new_population.extend([child1, child2])
            
            population = new_population[:pop_size]
    
    # Final local refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (current_resolution/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - current_resolution / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(current_resolution)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, current_resolution)
