import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a triangular construction
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution with simulated annealing
    scales = [35, 50, 75, 100]
    current_scale_idx = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale_idx < len(scales):
        scale = scales[current_scale_idx]
        
        # Interpolate if needed
        if scale > n_points:
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            h_interp = np.interp(x_new, x_old, best_h)
            h_current = h_interp
            n_points = scale
            dx = 2.0 / n_points
        else:
            h_current = best_h
            n_points = scale
            dx = 2.0 / n_points
            
        # Simulated Annealing
        T = 0.1
        alpha = 0.99
        max_iter = int(1000 * (scale / 35))  # Scale iterations with resolution
        
        for _ in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb solution
            h_new = h_current.copy()
            # Flip some values randomly
            flip_mask = np.random.random(n_points) < 0.1
            h_new[flip_mask] = 1 - h_new[flip_mask]
            
            # Ensure bounds and sum constraint
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            if c5_new < best_c5 or np.random.random() < np.exp(-(c5_new - best_c5) / T):
                h_current = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            
            # Cool down
            T = alpha * T
            
        current_scale_idx += 1
    
    # Final optimization with local refinement
    if time() - start_time < budget_s * 0.1:
        # Simple grid search around best solution
        h_refined = best_h.copy()
        refiner_steps = 500
        for _ in range(refiner_steps):
            if time() - start_time > budget_s:
                break
            # Slight random perturbation
            pert = np.random.normal(0, 0.01, n_points)
            h_trial = h_refined + pert
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points/2) / np.sum(h_trial)
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h_refined = h_trial.copy()
    
    return (list(best_h), best_c5, n_points)
