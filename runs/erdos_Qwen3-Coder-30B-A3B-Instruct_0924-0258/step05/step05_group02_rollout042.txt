import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize: either from previous solution or create new
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid resolution (20 points)
    n_low = 20
    dx_low = 2.0 / n_low
    h_low = np.linspace(0, 1, n_low)
    h_low = h_low * (n_low / 2) / np.sum(h_low)
    
    # Convex relaxation on coarse grid using subset of k values 
    try:
        h_low_var = cp.Variable(n_low)
        constraints = [
            cp.sum(h_low_var) == n_low / 2.0,
            h_low_var >= 0,
            h_low_var <= 1
        ]
        # Add constraints for first few shifts only
        for k in range(1, min(8, n_low // 2)):  
            if k < n_low - k:
                corr_sum = cp.sum(cp.multiply(h_low_var[:n_low-k], 1 - h_low_var[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_low_var[k:], 1 - h_low_var[:n_low-k]))
            constraints.append(corr_sum <= best_c5 * dx_low + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_low_var)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            h_low = h_low_var.value
    except Exception:
        pass
    
    # Upsample to fine resolution
    h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_low), h_low)
    h_fine = h_fine * (n_points/2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)
    
    # Start with the upsampled solution
    h_current = h_fine
    c5_current = evaluate_erdos_solution(h_current, dx)
    if c5_current < best_c5:
        best_h = h_current
        best_c5 = c5_current
    
    # Fine grid refinement with adaptive gradient descent
    max_iter = 5000
    iter_count = 0
    no_improvement = 0
    mutation_count = 0
    learning_rate = 0.01
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute gradient approximation
        grad = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            h_minus = h_current.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
        # Update step
        h_new = h_current - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check improvement
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            h_current = h_new
            best_c5 = c5_new
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Adaptive mutation
        if no_improvement > 50 or np.random.rand() < 0.01:
            mutation_strength = 0.05 * (1.0 if mutation_count == 0 else 1.1)
            mutation = np.random.normal(0, mutation_strength, n_points)
            h_current = h_current + mutation
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
            no_improvement = 0
            mutation_count += 1
            
        # Reduce learning rate over time
        if iter_count % 100 == 0:
            learning_rate *= 0.95
            
        iter_count += 1
        
        # Early stopping if no improvement for too long
        if no_improvement > 100:
            break
    
    # Final check and return best
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
