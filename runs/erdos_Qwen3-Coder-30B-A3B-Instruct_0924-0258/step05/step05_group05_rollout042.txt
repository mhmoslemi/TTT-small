import numpy as np
from scipy.signal import correlate
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        h_start = np.ones(n_points) * 0.5

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try block-based approach
    # Use 5 blocks for better structure
    M = 5
    block_size = n_points // M
    
    # Ensure block_size is integer
    while n_points % M != 0:
        M -= 1
        block_size = n_points // M
    
    # Initialize with best available
    h = np.zeros(n_points)
    
    # Create a block pattern with M blocks where exactly half of points are 1
    ones_count = n_points // 2
    on_blocks = ones_count // block_size
    
    # Set on_blocks blocks to 1
    for i in range(on_blocks):
        start_idx = i * block_size
        end_idx = start_idx + block_size
        h[start_idx:end_idx] = 1.0
    
    # Shuffle the positions to avoid systematic block placement
    np.random.shuffle(h)
    
    # Normalize to integral constraint
    h = h * (n_points/2) / np.sum(h)
    
    # Optimization loop
    iters = 0
    max_iters = 500
    tolerance = 1e-6
    stagnation_count = 0
    max_stagnation = 50
    
    while iters < max_iters and (time() - start_time) < budget_s * 0.95:
        current_c5 = evaluate_erdos_solution(h, dx)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            stagnation_count = 0
            if best_c5 < 0.3807:
                break
        else:
            stagnation_count += 1
            
        # Perturb solution by swapping blocks locally
        new_h = h.copy()
        
        # Try 3 local swaps
        for _ in range(3):
            # Get indices of 1s and 0s
            ones_idx = np.where(new_h > 0.5)[0]
            zeros_idx = np.where(new_h < 0.5)[0]

                block_idx = random.randint(0, M-1)
                block_start = block_idx * block_size
                block_end = min(block_start + block_size, n_points)
                
                # Determine if block is on or off
                if np.sum(new_h[block_start:block_end]) > 0.5 * block_size:
                    # Turn it off
                    new_h[block_start:block_end] = 0.0
                else:
                    # Turn it on
                    new_h[block_start:block_end] = 1.0
                    
                # Renormalize
                new_h = new_h * (n_points/2) / np.sum(new_h)
        
        h = new_h
        
        # Periodically do a gradient-like update
        if iters % 10 == 0 and iters > 0:
            # Compute gradient approximation via small perturbations
            h_pert = h.copy()
            # Add small noise
            perturbation = np.random.normal(0, 0.01, n_points)
            h_pert += perturbation

            h_pert = h_pert * (n_points/2) / np.sum(h_pert)

    remaining_time = budget_s - (time() - start_time)

        def objective(h):
            return evaluate_erdos_solution(h, dx)

                c5_final = objective(h_final)
                if c5_final < best_c5:

                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
