import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Check for initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Convert to binary
    h_best = np.round(h_start).astype(int)
    # Ensure sum is exactly n_points / 2
    current_sum = np.sum(h_best)
    target_sum = n_points // 2
    
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        zeros_indices = np.where(h_best == 0)[0]
        ones_indices = np.where(h_best == 1)[0]
        n_to_flip = current_sum - target_sum
        if len(ones_indices) >= n_to_flip:
            flip_indices = np.random.choice(ones_indices, n_to_flip, replace=False)
            h_best[flip_indices] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        zeros_indices = np.where(h_best == 0)[0]
        ones_indices = np.where(h_best == 1)[0]
        n_to_flip = target_sum - current_sum
        if len(zeros_indices) >= n_to_flip:
            flip_indices = np.random.choice(zeros_indices, n_to_flip, replace=False)
            h_best[flip_indices] = 1
    
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h_best, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 10
    max_no_improvement = 100
    no_improvement_count = 0
    shaking_iter = 50
    
    current_h = h_best.copy()
    current_c5 = best_c5
    
    def get_neighbors(h):
        """Generate neighboring solutions by flipping one 1 and one 0"""
        neighbors = []
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            return neighbors
        
        # Sample some swaps
        n_swaps = min(20, len(ones) * len(zeros))
        for _ in range(n_swaps):
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            neighbor = h.copy()
            neighbor[i] = 0
            neighbor[j] = 1
            neighbors.append((neighbor, i, j))
        return neighbors
    
    iteration = 0
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        neighbors = get_neighbors(current_h)
        if not neighbors:
            break
        
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        best_neighbor_indices = None
        
        for neighbor_h, i, j in neighbors:
            # Check tabu list
            tabu_move = (i, j) if i < j else (j, i)
            if tabu_move in tabu_list:
                # Allow aspiration: if it's much better than current best
                neighbor_c5 = evaluate_erdos_solution(neighbor_h, dx)
                if neighbor_c5 < current_c5 - 1e-10:
                    # Allow this move despite being tabu
                    pass
                else:
                    continue
            
            neighbor_c5 = evaluate_erdos_solution(neighbor_h, dx)
            if neighbor_c5 < best_neighbor_c5:
                best_neighbor_c5 = neighbor_c5
                best_neighbor = neighbor_h
                best_neighbor_indices = (i, j)
        
        # Update current solution
        if best_neighbor is not None and best_neighbor_c5 < current_c5:
            current_h = best_neighbor
            current_c5 = best_neighbor_c5
            no_improvement_count = 0
            if current_c5 < best_c5:
                best_c5 = current_c5
                h_best = current_h.copy()
            
            # Update tabu list
            if len(tabu_list) >= tabu_tenure:
                tabu_list.pop(0)
            tabu_list.append(best_neighbor_indices)
        else:
            no_improvement_count += 1
        
        # Shaking
        if no_improvement_count >= shaking_iter and iteration > shaking_iter:
            # Random restart: flip 5% of bits
            n_flip = max(1, n_points // 20)
            to_flip = np.random.choice(n_points, n_flip, replace=False)
            current_h[to_flip] = 1 - current_h[to_flip]
            # Ensure sum constraint
            current_sum = np.sum(current_h)
            if current_sum != target_sum:
                # Rebalance
                if current_sum < target_sum:
                    # Need to make some 0s into 1s
                    zeros = np.where(current_h == 0)[0]
                    if len(zeros) >= target_sum - current_sum:
                        to_change = np.random.choice(zeros, target_sum - current_sum, replace=False)
                        current_h[to_change] = 1
                else:
                    # Make some 1s into 0s
                    ones = np.where(current_h == 1)[0]
                    if len(ones) >= current_sum - target_sum:
                        to_change = np.random.choice(ones, current_sum - target_sum, replace=False)
                        current_h[to_change] = 0
            current_c5 = evaluate_erdos_solution(current_h, dx)
            no_improvement_count = 0
        
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final validation
    final_sum = np.sum(h_best)
    if abs(final_sum - target_sum) > 0.1:
        # Rebalance if needed
        if final_sum < target_sum:
            zeros = np.where(h_best == 0)[0]
            to_change = np.random.choice(zeros, target_sum - final_sum, replace=False)
            h_best[to_change] = 1
        else:
            ones = np.where(h_best == 1)[0]
            to_change = np.random.choice(ones, final_sum - target_sum, replace=False)
            h_best[to_change] = 0
            
    final_c5 = evaluate_erdos_solution(h_best, dx)
    
    return (list(h_best), final_c5, n_points)
