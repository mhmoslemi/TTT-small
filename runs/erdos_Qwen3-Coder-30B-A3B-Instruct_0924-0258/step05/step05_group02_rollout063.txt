import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    max_iter = 1000
    iter_count = 0
    local_search_iter = 0
    patience = 0
    max_patience = 50
    
    def project_h_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def local_refine(h_init):
        try:
            cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_init, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            if res.success:
                return res.x
        except:
            pass
        return h_init
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1
        
        # Generate new candidate via random perturbation
        h_new = best_h + np.random.normal(0, 0.03, n_points)
        h_new = project_h_to_feasible(h_new)
        
        # Local refinement
        h_refined = local_refine(h_new)
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
            patience = 0
        else:
            patience += 1
        
        # Periodic restart if stuck
        if patience > max_patience:
            h_restart = np.random.rand(n_points)
            h_restart = project_h_to_feasible(h_restart)
            best_h = h_restart
            best_c5 = evaluate_erdos_solution(h_restart, dx)
            patience = 0

        # Occasionally try global search with binary structure
        if iter_count % 50 == 0:
            # Binary approximate with refinement
            h_binary = (best_h > 0.5).astype(float)
            h_binary = project_h_to_feasible(h_binary)
            h_refined = local_refine(h_binary)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined

        # Occasionally apply convex relaxation
        if iter_count % 100 == 0 and time() - start_time < budget_s * 0.9:
            try:
                h_relaxed = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_relaxed) == n_points / 2.0,
                    h_relaxed >= 0,
                    h_relaxed <= 1
                ]
                # Add relaxed constraint
                obj = cp.Minimize(cp.sum(h_relaxed))
                prob = cp.Problem(obj, constraints)
                prob.solve(solver=cp.ECOS, eps=1e-7)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_relaxed.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass

    # Final cleanup
    h_final = project_h_to_feasible(best_h)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), c5_final, n_points)
