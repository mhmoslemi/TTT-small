import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization with cutting-plane approach
    resolutions = [35, 40, 50, 60]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Cutting-plane optimization
        max_iterations = 100
        iteration = 0
        
        while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
            iteration += 1
            
            # Compute overlaps
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_k = np.argmax(corr)
            max_overlap = corr[max_k] * dx
            
            # Check if we can improve
            if max_overlap >= best_c5 * 0.9999:  # No significant improvement
                break
            
            # Add constraint for maximum overlap shift
            try:
                # Use cvxpy for optimization with overlap constraint
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add the constraint for current maximum
                if max_k < n_points:
                    # This corresponds to left-shift in correlation
                    corr_expr = cp.sum(cp.multiply(h_opt[:n_points-max_k], 1 - h_opt[max_k:]))
                else:
                    # This corresponds to right-shift in correlation
                    shift = max_k - n_points + 1
                    corr_expr = cp.sum(cp.multiply(h_opt[shift:], 1 - h_opt[:n_points-shift]))
                    
                constraints.append(corr_expr <= max_overlap * dx + 1e-8)
                
                # Define the objective (minimize max overlap - stable formulation)
                # But since we're already constraining this, we solve the feasibility problem
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
                
                if prob.status == cp.OPTIMAL:
                    h_new = h_opt.value
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
                        h_current = h_new.copy()
                    else:
                        # Local search as backup
                        h_local = h_current + np.random.normal(0, 0.01, n_points)
                        h_local = np.clip(h_local, 0, 1)
                        h_local = h_local * (n_points/2) / np.sum(h_local)
                        
                        def objective(h_vec):
                            return evaluate_erdos_solution(h_vec, dx)
                        
                        def constraint_sum(h_vec):
                            return np.sum(h_vec) - n_points / 2.0
                        
                        cons_sum = {'type': 'eq', 'fun': constraint_sum}
                        bounds = [(0, 1) for _ in range(n_points)]
                        res = minimize(objective, h_local, method='SLSQP', 
                                       bounds=bounds, constraints=cons_sum, 
                                       options={'maxiter': 100, 'ftol': 1e-8})
                        if res.success:
                            h_refined = res.x
                            c5_refined = objective(h_refined)
                            if c5_refined < best_c5:
                                best_h = h_refined
                                best_c5 = c5_refined
                                h_current = h_refined
                else:
                    # Fallback to local search
                    h_local = h_current + np.random.normal(0, 0.01, n_points)
                    h_local = np.clip(h_local, 0, 1)
                    h_local = h_local * (n_points/2) / np.sum(h_local)
                    
                    def objective(h_vec):
                        return evaluate_erdos_solution(h_vec, dx)
                    
                    def constraint_sum(h_vec):
                        return np.sum(h_vec) - n_points / 2.0
                    
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    bounds = [(0, 1) for _ in range(n_points)]
                    res = minimize(objective, h_local, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-8})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                            h_current = h_refined
                            
            except Exception:
                # Local fallback if convex optimization fails
                h_local = h_current + np.random.normal(0, 0.01, n_points)
                h_local = np.clip(h_local, 0, 1)
                h_local = h_local * (n_points/2) / np.sum(h_local)
                
                def objective(h_vec):
                    return evaluate_erdos_solution(h_vec, dx)
                
                def constraint_sum(h_vec):
                    return np.sum(h_vec) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(objective, h_local, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        h_current = h_refined
        
        h_start = best_h.copy()
    
    # Final refinement with a more reliable method
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
