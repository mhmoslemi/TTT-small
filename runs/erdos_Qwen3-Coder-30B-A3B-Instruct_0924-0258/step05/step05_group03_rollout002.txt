import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    h = initial_h.copy()
    h = h * (n_points / 2) / np.sum(h)
    c5_bound = evaluate_erdos_solution(h, 2.0 / n_points)

    max_iter = 50
    min_improvement = 1e-5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all overlaps and find worst one
        overlaps = []
        for shift in shifts:
            # Compute overlap for this shift
            shift_idx = int(shift * n_points / 2)
            if shift_idx == 0:
                overlap = np.sum(h * (1 - h)) * (2.0 / n_points)
            else:
                h_shifted = np.concatenate([h[shift_idx:], h[:shift_idx]])
                overlap = np.sum(h * (1 - h_shifted)) * (2.0 / n_points)
            overlaps.append(overlap)
            
        max_overlap_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_overlap_idx]

        if max_overlap > c5_bound + 1e-10:
