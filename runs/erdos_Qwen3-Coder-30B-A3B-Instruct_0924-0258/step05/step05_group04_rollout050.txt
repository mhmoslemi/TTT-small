import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    max_iter = 5000
    mu = 1e-3
    rho = 1.0
    tol = 1e-6
    stagnation_threshold = 200
    
    # Initialize variables
    h = h_start.copy()
    h_prev = h.copy()
    u = np.zeros(2 * n_points - 1)
    v = np.zeros(2 * n_points - 1)
    
    # Precompute indices for efficient handling
    shift_indices = {}
    for k in range(-n_points + 1, n_points):
        shift_indices[k] = np.arange(max(0, -k), min(n_points, n_points - k))
    
    # Optimization variables
    h_star = h.copy()
    max_overlap = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Update h by solving the subproblem
        # min 0.5 * ||h - h_prev + mu * (u - v)||^2 + mu * ||h - (1-h) * shift + C5thresh||^2
        # subject to 0 <= h <= 1, sum(h) = n_points / 2
        
        # Simplified optimization approach - making small changes based on gradients
        # From a convergent ADMM standpoint, we can just perform gradient-based steps
        h_old = h.copy()
        
        # We'll define the objective and constraints manually
        def obj(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        def con_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, h_old, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h = res.x.copy()
                candidate_c5 = obj(h)
                if candidate_c5 < max_overlap:
                    max_overlap = candidate_c5
                    h_star = h.copy()
        except Exception:
            pass
            
        # Check for improvement
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h_star.copy()
            
        # Adaptive adjustments
        if iteration % 100 == 0 and iteration > 0:
            if abs(max_overlap - best_c5) < 1e-6:
                mu *= 1.5
            else:
                mu = max(1e-6, mu * 0.95)
                
        # Early termination if target met
        if best_c5 < 0.3807:
            break

    # Final refinement with high precision
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_final = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_final, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
            
    return (list(best_h), best_c5, n_points)
