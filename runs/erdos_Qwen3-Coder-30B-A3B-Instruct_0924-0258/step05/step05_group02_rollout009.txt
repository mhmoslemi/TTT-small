import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm with local refinement
    pop_size = 50
    max_generations = 1000
    elitism_rate = 0.3
    mutation_rate = 0.1
    crossover_alpha_range = (0.4, 0.6)
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    generation = 0
    stagnant = 0
    best_found = best_c5
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_found:
            best_found = fitness[0]
            best_h = population[0].copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 20:
            # Restart with new random population
            population = [population[0].copy()]
            for _ in range(pop_size - 1):
                h = np.random.rand(n_points)
                h = h * (n_points/2) / np.sum(h)
                h = np.clip(h, 0, 1)
                population.append(h)
            stagnant = 0
        
        # Selection
        elite_count = max(1, int(elitism_rate * pop_size))
        elite = population[:elite_count]
        
        # Crossover and mutation
        new_population = elite.copy()
        while len(new_population) < pop_size:
            # Crossover
            p1, p2 = np.random.choice(elite, 2, replace=False)
            alpha = np.random.uniform(*crossover_alpha_range)
            child = alpha * p1 + (1 - alpha) * p2
            
            # Mutation
            if np.random.rand() < mutation_rate:
                child += np.random.normal(0, 0.02, n_points)
                child = np.clip(child, 0, 1)
            
            # Normalize to satisfy constraint
            child = child * (n_points/2) / np.sum(child)
            new_population.append(child)
        
        # Local refinement with gradient descent for top 10
        top_candidates = new_population[:10]
        for i, h_candidate in enumerate(top_candidates):
            try:
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(obj, h_candidate, method='SLSQP', bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-7})
                
                if res.success:
                    h_refined = res.x
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        
        population = new_population
        generation += 1
    
    # Final check
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
