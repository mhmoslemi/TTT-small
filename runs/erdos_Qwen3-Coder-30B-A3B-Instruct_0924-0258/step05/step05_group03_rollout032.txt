import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
        h[0] = 0.3
        h[-1] = 0.3
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Continuous optimization approach using CVXPY
    dx = 2.0 / n_points
    
    # Sample shifts uniformly
    shift_samples = np.linspace(0, 2, 100)
    
    # Each constraint corresponds to a shift
    max_iter = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iter and no_improvement_count < 50:
        # Compute overlaps for all sampled shifts
        overlaps = []
        for shift in shift_samples:
            # Compute correlation at this shift
            if shift == 0:
                corr = np.correlate(h, 1 - h, mode='full')
                overlap = np.max(corr) * dx
            else:
                # For non-zero shifts, manually compute
                overlap = 0
                shift_idx = int(shift / dx)
                if shift_idx < n_points:
                    # Left side contribution
                    for i in range(n_points - shift_idx):
                        overlap += h[i] * (1 - h[i + shift_idx])
                    overlap *= dx
                else:
                    overlap = 0
            overlaps.append(overlap)
        
        # Add constraint from max overlap
        max_overlap = max(overlaps)
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Perform local refinement using scipy
        def obj(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try a better solution
        try:
            res = minimize(obj, h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_new = res.x
                c5_new = obj(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    no_improvement_count = 0
        except:
            pass
            
        iteration += 1
        
        # Further polishing with gradient-based optimization
        try:
            res = minimize(obj, best_h, method='L-BFGS-B', 
                           bounds=bounds, 
                           options={'maxiter': 50})
            if res.success:
                h_new = res.x
                c5_new = obj(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        except:
            pass
            
        # Check for early termination
        if best_c5 < 0.3807:
            break
    
    # Final refinement
    try:
        def obj(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
        
    return (list(best_h), best_c5, n_points)
