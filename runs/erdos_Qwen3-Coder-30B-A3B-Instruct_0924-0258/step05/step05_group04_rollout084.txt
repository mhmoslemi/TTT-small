import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def smooth_max(h, dx, tau=1.0):
    """Compute smooth approximation to max overlap"""
    n = len(h)
    max_val = 0.0
    total = 0.0
    for k in range(2 * n - 1):
        corr = 0.0
        for i in range(n):
            if 0 <= i + k - (n - 1) < n:
                corr += h[i] * (1 - h[i + k - (n - 1)])
        total += np.exp(tau * corr) if corr < 100 else np.exp(100 * tau)
        max_val = max(max_val, corr)
        
    return np.log(total) / tau if total > 0 else max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s - 1
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Parameters
    tau = 1.0
    max_iter = 3000
    tolerance = 1e-7
    step_size = 1.0
    stagnation_count = 0
    prev_best = best_c5
    
    def objective(h):
        return smooth_max(h, dx, tau)
    
    def gradient(h):
        # Approximate gradient through finite differences
        eps = 1e-6
        grad = np.zeros_like(h)
        base_val = objective(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            grad[i] = (objective(h_plus) - base_val) / eps
        return grad
    
    # Main optimization loop
    for i in range(max_iter):
        if time() - start_time > max_time:
            break
            
        # Compute gradient
        try:
            grad = gradient(h_current)
        except:
            break
            
        # Update using gradient descent
        h_new = h_current - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate objective
        current_val = objective(h_new)
        current_c5 = evaluate_erdos_solution(h_new, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            
        # Update current step
        h_current = h_new
        
        # Adaptive adjustments
        if abs(current_c5 - prev_best) < 1e-8:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        if stagnation_count > 50:
            tau *= 1.1
            step_size *= 0.9
            stagnation_count = 0
            
        if tau > 1000:
            break
            
        prev_best = current_c5
        
    # Final refinement
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 20 and best_c5 > 0.3807:
        try:
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            def con(h):
                return np.sum(h) - n_points / 2.0
            
            cons = {'type': 'eq', 'fun': con}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 50, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = obj(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
