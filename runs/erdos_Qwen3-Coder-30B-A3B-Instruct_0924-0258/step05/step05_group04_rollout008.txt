import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimize using L-BFGS-B with smooth approximation
    max_iter = 3000
    tolerance = 1e-7
    tau = 1.0
    decay_every = 100
    tau_decay = 0.99
    step_size = 1.0
    
    h_current = h_start.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Smooth max approximation for gradient computation
    def smooth_max(h, tau):
        # Compute all correlations (shifted overlaps)
        corr = np.correlate(h, 1 - h, mode='full')
        # Smooth approximation with temperature tau
        smoothed = np.log(np.sum(np.exp(tau * corr))) / tau
        return smoothed, corr
    
    # Gradient of smooth max w.r.t. h
    def smooth_max_grad(h, corr, tau):
        # Normalize exponential terms
        exp_terms = np.exp(tau * corr)
        sum_exp = np.sum(exp_terms)
        # We compute d(max_smooth)/dh_i
        # This requires computing derivatives of each correlation
        # which is a bit involved but can be done efficiently
        n = len(h)
        grad = np.zeros(n)
        # Correlation for each shift k
        # Ref: derivative of log(sum(exp(tau * corr))) = (sum(exp(tau*corr) * tau * dcorr/dh)) / sum(exp(tau*corr))
        # For correlation with shifted function: corr_k = sum h[i] * (1 - h[i+k])
        # dcorr_k/dh_i = 1 - h[i+k] if i+k < n else 0
        # We compute the gradient of sum over all k of the smooth max
        # But it's easier to compute via finite differences for this problem
        
        # For now, we'll estimate it using finite differences
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(n):
            h_pert = h.copy()
            h_pert[i] += eps
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert) 
            _, corr_pert = smooth_max(h_pert, tau)
            val_pert, _ = smooth_max(h_pert, tau)
            val_curr, _ = smooth_max(h, tau)
            
            grad[i] = (val_pert - val_curr) / eps
            
        # Normalize gradient so that sum is 0 (to respect constraint)  
        grad = grad - np.mean(grad)
        return grad
    
    # Optimization loop
    last_improvement = 0
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient via finite differences
        _, corr = smooth_max(h_current, tau)
        val_current, _ = smooth_max(h_current, tau)
        
        # Incrementally compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h_current)
        
        for j in range(n_points):
            h_pert = h_current.copy()
            h_pert[j] += eps
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            val_pert, _ = smooth_max(h_pert, tau)
            grad[j] = (val_pert - val_current) / eps
        
        grad = grad - np.mean(grad)  # Adjust to respect the constraint on gradient
        
        # Use gradient descent step
        h_new = h_current - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new objective
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Check if improved
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            last_improvement = i
            
        # Update current point
        h_current = h_new.copy()
        
        # Adjust parameters based on progress
        if i % decay_every == 0:
            tau *= tau_decay
            
        if i - last_improvement > 100:
            tau *= 1.1
            
        # Early termination condition based on tolerance
        if np.abs(val_current - new_c5) < tolerance:
            break
            
    # Final refinement with exact objective using SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        h_final = best_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
        
        def obj_func(h):
            return evaluate_erdos_solution(h, dx)
            
        try:
            res = minimize(obj_func, h_final, method='SLSQP', 
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = obj_func(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
