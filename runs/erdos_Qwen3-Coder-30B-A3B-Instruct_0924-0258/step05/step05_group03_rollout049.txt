import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from cvxpy import ECOS

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the given construction, if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Initialize tracking variables
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    best_h = h.copy()
    
    stagnation_count = 0
    max_stagnation = 50
    shift_sampling = 100
    k_values = np.linspace(0, 2, shift_sampling, endpoint=False)
    
    # Optimization loop
    for iteration in range(500):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Step 1: Find worst shift k for current h
        overlaps = []
        for k in k_values:
            # For continuous shift, we need to compute the overlap with a circular shift
            # Since h is discrete, we compute correlation with shifted version
            shifted_h = np.roll(h, int(k * n_points / 2))
            overlap = np.sum(h * (1 - shifted_h)) * (2.0 / n_points)
            overlaps.append(overlap)
        
        # Find worst shift k
        k_star_idx = np.argmax(overlaps)
        k_star = k_values[k_star_idx]
        
        # Create shifted version for convex optimization
        shifted_h = np.roll(h, int(k_star * n_points / 2))
        
        # Step 2: Solve convex subproblem for h (linearized around current h)
        try:
            # Use convex programming
            h_var = cp.Variable(n_points)
            objective = cp.Minimize(cp.sum(cp.multiply(h_var, 1 - shifted_h)))
            constraints = [
                cp.sum(h_var) == n_points / 2,
                h_var >= 0,
                h_var <= 1
            ]
            
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=ECOS, verbose=False, max_iters=100)
            
            if prob.status == cp.OPTIMAL:
                new_h = h_var.value
                # Ensure it's within bounds and correct sum
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                # Update h
                h = new_h
                new_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                # Fallback: stochastic perturbation if optimization fails
                h = h + np.random.normal(0, 0.001, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                stagnation_count += 1
        except Exception as e:
            # Fallback: small random perturbations if any error in CVXPY
            h = h + np.random.normal(0, 0.001, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count += 1
            
        # If stagnation count exceeds limit, add noise
        if stagnation_count > max_stagnation:
            h = h + np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0
            
        # Early termination if target met
        if best_c5 < 0.3807:
            break
    
    # Final refinement with scipy
    def obj(x):
        return evaluate_erdos_solution(x, 2.0 / n_points)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
