import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Initialize
    h = initial_h.copy()
    dx = 2.0 / n_points
    
    # Project to valid range and normalize
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex-concave procedure
    max_iter = 500
    tol = 1e-6
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate correlations to find worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        k_indices = np.arange(len(corr))
        # Map indices to actual shifts in [0, 2]
        shifts = k_indices * dx - 1  # shift ranges from -1 to 1
        # We want the shift that maximizes the correlation
        max_idx = np.argmax(corr)
        k_opt = shifts[max_idx]
        
        # Solve convex subproblem
        # Minimize sum h_new[i] * (1 - h[i + k_opt]) * dx
        # subject to: sum(h_new) = n_points/2, 0 <= h_new[i] <= 1
        
        # Use a simple approximation: use a shifted version of the current h
        # to construct a better candidate for h 
        # Build the shifted version of 1-h
        h_shifted = np.roll(1 - h, int(k_opt / dx))
        # Modify h to reduce overlap
        h_new = h_shifted.copy()
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        if abs(c5_new - best_c5) < tol:
            break
            
        h = h_new.copy()
        
        # Optional: local Newton-like refinement
        if iteration % 5 == 0:
            try:
                def obj(x):
                    return evaluate_erdos_solution(x, dx)
                def constraint_sum(x):
                    return np.sum(x) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(obj, h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    h = res.x
                    c5_new = obj(h)
                    if c5_new < best_c5:
                        best_h = h.copy()
                        best_c5 = c5_new
            except:
                pass
        
        if best_c5 < 0.3807:
            break
    
    # Final polishing
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-10})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
