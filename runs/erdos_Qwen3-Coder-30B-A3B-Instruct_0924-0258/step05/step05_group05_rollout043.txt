import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Block-based approach
    M = int(np.sqrt(n_points))
    if M < 2:
        M = 2
    block_size = n_points // M
    if block_size * M != n_points:
        block_size = n_points // M
        M = n_points // block_size
    
    # Initialize blocks
    blocks = [0] * M
    num_on_blocks = n_points // (2 * block_size)
    if num_on_blocks * 2 * block_size != n_points:
        num_on_blocks = (n_points + block_size - 1) // (2 * block_size)
        # Ensure we don't exceed the required sum
        if num_on_blocks * 2 * block_size > n_points:
            num_on_blocks = n_points // (2 * block_size)

    # Initialize with random configuration
    on_indices = random.sample(range(M), min(num_on_blocks, M))
    for i in on_indices:
        blocks[i] = 1
    
    # Create h from blocks
    h = np.zeros(n_points)
    for i in range(M):
        start = i * block_size
        end = start + block_size
        h[start:end] = blocks[i]
    
    # Scale to sum = n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Block optimization with iterations
    max_iters = 3000
    last_improvement = 0
    
    for iteration in range(max_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convert to blocks to make changes
        current_blocks = []
        for i in range(M):
            start = i * block_size
            end = start + block_size
            block_avg = np.mean(h[start:end])
            current_blocks.append(1 if block_avg > 0.5 else 0)
        
        # Check if we're about to get stuck
        if iteration - last_improvement > 200:
            # Try small mutations
            for _ in range(3):
                try:
                    # Randomly flip one block
                    i = random.randrange(M)
                    current_blocks[i] = 1 - current_blocks[i]
                    
                    # Construct new h
                    new_h = np.zeros(n_points)
                    for j in range(M):
                        start = j * block_size
                        end = start + block_size
                        new_h[start:end] = current_blocks[j]
                    
                    # Normalize to correct integral
                    new_sum = np.sum(new_h)
                    if new_sum > 0:
                        new_h = new_h * (n_points / 2) / new_sum
                    
                    # Evaluate
                    new_c5 = evaluate_erdos_solution(new_h, dx)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = new_h.copy()
                        last_improvement = iteration
                        
                except Exception:
                    continue
    
        # Try to improve global structure with random restarts
        if iteration % 150 == 0 and iteration > 0:
            # Try completely random configuration
            num_on = n_points // (2 * block_size)
            if num_on * 2 * block_size != n_points:
                # Adjust for small changes
                num_on = (n_points + block_size - 1) // (2 * block_size)
                num_on = min(num_on, M)
            new_blocks = [0] * M
            on_indices = random.sample(range(M), num_on)
            for idx in on_indices:
                new_blocks[idx] = 1
                
            # Create new h
            new_h = np.zeros(n_points)
            for i in range(M):
                start = i * block_size
                end = start + block_size
                new_h[start:end] = new_blocks[i]
                
            # Normalize
            new_sum = np.sum(new_h)
            if new_sum > 0:
                new_h = new_h * (n_points / 2) / new_sum
                
            new_c5 = evaluate_erdos_solution(new_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                last_improvement = iteration
                
        # Try twist optimization: shuffle some blocks
        if iteration % 250 == 0 and iteration > 0:
            # Try swapping two adjacent blocks or small modifications
            try:
                # Select two random positions to possibly swap
                if M > 1:
                    i = random.randrange(M-1) if M > 1 else 0
                    j = i + 1
                    
                    # Make a swap
                    new_blocks = current_blocks[:]
                    new_blocks[i], new_blocks[j] = new_blocks[j], new_blocks[i]
                    
                    # Construct new h
                    new_h = np.zeros(n_points)
                    for k in range(M):
                        start = k * block_size
                        end = start + block_size
                        new_h[start:end] = new_blocks[k]
                    
                    # Normalize
                    new_sum = np.sum(new_h)
                    if new_sum > 0:
                        new_h = new_h * (n_points / 2) / new_sum
                        
                    new_c5 = evaluate_erdos_solution(new_h, dx)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = new_h.copy()
                        last_improvement = iteration
                        
            except Exception:
                continue

        # Minor random variations for fine-tuning
        if iteration % 10 == 0:
            # Randomly perturb a small amount
            perturb_indices = random.sample(range(M), min(2, M))
            new_blocks = current_blocks[:]
            for i in perturb_indices:
                if random.random() < 0.5 and np.sum(new_blocks) > 1:
                    # Turn off a block if it's on
                    if new_blocks[i] == 1:
                        new_blocks[i] = 0
                else:
                    # Turn on a block if it's off
                    if new_blocks[i] == 0:
                        new_blocks[i] = 1
                        
            # Ensure correct integral
            count = np.sum(new_blocks)
            target_count = n_points // (2 * block_size)
            if count < target_count and len(new_blocks) > 0:
                # Add some blocks (possibly not balanced but better than stale)
                pass   # Simplify to prevent running out of bounds when num_on_blocks too high
            elif count > target_count:
                # Remove some blocks (randomly)
                off_blocks = [i for i in range(len(new_blocks)) if new_blocks[i] == 0]
                if len(off_blocks) > 0:
                    to_turn_off = count - target_count
                    for _ in range(to_turn_off):
                        if off_blocks:
                            idx = random.choice(off_blocks)
                            new_blocks[idx] = 1  # For now, do not entirely turn off (a bit conservative)
            
            # Construct new h
            new_h = np.zeros(n_points)
            for i in range(M):
                start = i * block_size
                end = start + block_size
                new_h[start:end] = new_blocks[i]
                
            # Normalize
            new_sum = np.sum(new_h)
            if new_sum > 0:
                new_h = new_h * (n_points / 2) / new_sum
                
            new_c5 = evaluate_erdos_solution(new_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                last_improvement = iteration

        # Early stopping if good enough
        if best_c5 < 0.3807:
            break
            
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 25:
        try:
            # Local refinement on best solution
            h_init = best_h
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            from scipy.optimize import minimize
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    
        except:
            pass

    return (list(best_h), best_c5, n_points)
