import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    # Initialize
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    max_iter = 1000
    tol = 1e-5
    stagnation_count = 0
    max_stagnation = 50
    
    # Convex-concave procedure (CCP) approach
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find worst shift k*
        shifts = np.linspace(0, 2, 100)
        max_correlations = []
        for k in shifts:
            shift_idx = int(k / dx)
            if shift_idx >= n_points:
                shift_idx = n_points - 1
            corr = np.sum(h * np.roll(1 - h, shift_idx))
            max_correlations.append(corr)
            
        worst_k_idx = np.argmax(max_correlations)
        k_star = shifts[worst_k_idx]
        k_star_idx = int(k_star / dx)
        if k_star_idx >= n_points:
            k_star_idx = n_points - 1
        
        # Formulate and solve the convex subproblem
        h_new = cp.Variable(n_points)
        constraints = [
            h_new >= 0,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2
        ]
        
        # Objective: minimize the worst-case correlation
        # Build the correlation term for the specific k*
        if k_star_idx == 0:
            corr_term = h_new @ (1 - h)
        else:
            rolled_h = np.roll(h, k_star_idx)
            corr_term = h_new @ (1 - rolled_h)
            
        objective = cp.Minimize(corr_term)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new_vals = h_new.value
                h_new_vals = np.maximum(0, np.minimum(1, h_new_vals))
                h_new_vals = h_new_vals * (n_points / 2) / np.sum(h_new_vals)
                if np.any(np.isnan(h_new_vals)):
                    h_new_vals = h.copy()
                h = h_new_vals.copy()
        except:
            pass
        
        # Evaluate new solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        if stagnation_count > max_stagnation or abs(best_c5 - current_c5) < tol:
            break
            
        if best_c5 < 0.3807:
            break
    
    # Final refinement using local optimization
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
