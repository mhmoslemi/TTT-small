import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize to sum to 1
    dx = 2.0 / n_points
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # CCP optimization
    h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Convex-concave procedure iterations
    max_iter = 1000
    tolerance = 1e-6
    last_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Identify worst shift k*
        # Sample shift values to find maximum overlap
        shifts = np.linspace(0, 2, 100)
        max_overlap = -1
        best_k = 0
        
        for k in shifts:
            # Compute convolution manually for this shift
            k_idx = int(k / dx)
            if k_idx >= n_points:
                continue
            # Compute h[i] * (1 - h[i+k])
            overlap = 0
            for i in range(n_points - k_idx):
                overlap += h[i] * (1 - h[i + k_idx]) * dx
            if overlap > max_overlap:
                max_overlap = overlap
                best_k = k
                
        # Convex subproblem: minimize sum over best_k
        # We'll reframe to optimize a linear objective
        # Define variables
        h_var = cp.Variable(n_points)
        
        # Objective: minimize sum of h[i] * (1 - h[i + k]) where k is best_k
        # Let's compute the linear part of the objective based on best_k
        k_idx = int(best_k / dx)
        if k_idx < n_points:
            objective = cp.sum(cp.multiply(h_var, 1 - h_var[k_idx:]))
            # Projected on the rest if needed
            objective = cp.sum(cp.multiply(h_var, np.ones_like(h_var) - h_var))
        else:
            objective = cp.sum(cp.multiply(h_var, np.ones_like(h_var) - h_var))
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Solve the convex subproblem
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                new_h = h_var.value
                # Ensure it's within bounds for numerical precision
                new_h = np.clip(new_h, 0, 1)
                # Renormalize to meet constraint
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                c5_new = evaluate_erdos_solution(new_h, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = new_h.copy()
                
                # Check convergence
                if abs(last_c5 - c5_new) < tolerance:
                    break
                last_c5 = c5_new
                h = new_h
        except:
            # If failed to optimize, stick with current
            pass
            
        # Early termination
        if best_c5 < 0.3807:
            break
    
    # Final polishing with SLSQP
    def obj(x):
        return evaluate_erdos_solution(x, dx)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            final_h = res.x
            final_c5 = obj(final_h)
            
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
