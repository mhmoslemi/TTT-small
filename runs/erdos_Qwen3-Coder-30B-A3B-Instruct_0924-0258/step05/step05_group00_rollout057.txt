import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Early stopping condition tracking
    last_improvement = 0
    stagnation_count = 0
    max_stagnation = 100
    
    def initialize_population(pop_size, n_points):
        if 'initial_h_values' in globals():
            population = [h_start.copy()]
            for _ in range(pop_size - 1):
                h = h_start + np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                population.append(h)
        else:
            population = []
            for _ in range(pop_size):
                h = np.random.rand(n_points)
                h = h * (n_points/2) / np.sum(h)
                population.append(h)
        return population
    
    def tournament_selection(population, fitnesses, tournament_size=5):
        selected = []
        for _ in range(len(population)):
            indices = np.random.choice(len(population), tournament_size)
            winner = indices[np.argmin([fitnesses[i] for i in indices])]
            selected.append(population[winner])
        return selected
    
    def crossover(parent1, parent2, crossover_rate=0.8):
        if np.random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = np.random.randint(1, len(parent1) - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(individual, mutation_rate=0.1, noise=0.02):
        mutated = individual.copy()
        indices = np.random.choice(len(mutated), int(len(mutated) * mutation_rate), replace=False)
        mutated[indices] += np.random.normal(0, noise, len(indices))
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points/2) / np.sum(mutated)
        return mutated
    
    def evaluate_population(population):
        return [evaluate_erdos_solution(ind, dx) for ind in population]
    
    # Main GA loop
    max_generations = 1000
    pop_size = 50
    
    population = initialize_population(pop_size, n_points)
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        fitnesses = evaluate_population(population)
        
        # Track best individual
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = population[best_idx].copy()
            last_improvement = generation
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                # Increase mutation rate if stagnation occurs
                mutation_rate = 0.2
            else:
                mutation_rate = 0.1
        
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Crossover and Mutation
        new_population = []
        for i in range(0, len(selected), 2):
            if i + 1 < len(selected):
                child1, child2 = crossover(selected[i], selected[i+1])
                new_population.append(mutate(child1, mutation_rate))
                new_population.append(mutate(child2, mutation_rate))
            else:
                new_population.append(mutate(selected[i], mutation_rate))
        
        # Replace population
        population = new_population[:pop_size]
        
        # Adaptive resolution change
        if generation > max_generations * 0.8 and best_c5 > 0.38085:
            if n_points == 35:
                n_points = 40
                dx = 2.0 / n_points
                h_interp = np.interp(np.linspace(0, 2, n_points),

                best_h = h_interp
                population = initialize_population(20, n_points)
                max_generations = 300
                last_improvement = 0
                stagnation_count = 0
    
    # Local refinement using SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
