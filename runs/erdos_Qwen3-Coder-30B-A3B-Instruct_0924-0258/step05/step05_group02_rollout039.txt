import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    # Initialize with existing construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)

    # Gradient-based local search
    def objective(h):
        return evaluate_erdos_solution(h, dx)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    # Constraint handling for optimization
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    max_restarts = 100
    restart_count = 0
    early_stop = 0
    
    # Use direct optimization first
    try:
        res = minimize(
            lambda h: evaluate_erdos_solution(h, dx), 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons_sum,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if res.success:
            h_refined = res.x
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    # Main optimization loop
    while time() - start_time < budget_s * 0.95:
        # Local search
        h_perturbed = best_h.copy() + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Optimize with SLSQP
        try:
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-7}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
                    early_stop = 0
                    continue
        except Exception:
            pass

        # Try random restarts to avoid local minima
        h_restart = np.random.rand(n_points)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        h_restart = np.clip(h_restart, 0, 1)
        
        restart_count += 1
        if restart_count > max_restarts:
            # Use different starting point
            h_roll = np.roll(best_h, 1)
            h_restart = h_roll + np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            restart_count = 0
        
        # Constrained optimization for the restart
        try:
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_restart, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-7}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            continue

    # Final normalization of best solution
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    c5_final = evaluate_erdos_solution(h_final, dx)

    return (list(h_final), c5_final, n_points)
