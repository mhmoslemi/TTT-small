import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    num_particles = 40
    max_iter = 500
    inertia_weight = 0.7
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.2
    warm_restart_after = 200
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    for i in range(num_particles):
        if i < len(h_start):
            h = h_start.copy()
        else:
            h = np.random.random(n_points)
            h = h * (n_points / 2) / np.sum(h)
        
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_fitness.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    best_idx = np.argmin(pbest_fitness)
    gbest = pbest[best_idx].copy()
    gbest_fitness = pbest_fitness[best_idx]
    
    # ADAM-like warm restart tracking
    last_improvement_iter = 0
    stagnation_count = 0
    
    for iter_count in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # PSO update
        for i in range(num_particles):
            h = particles[i]
            v = velocities[i]
            p = pbest[i]
            
            # Update velocity
            r1, r2 = np.random.random(2)
            v = (inertia_weight * v + 
                 c1 * r1 * (p - h) + 
                 c2 * r2 * (gbest - h))
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            
            # Update position
            h = h + v
            
            # Enforce constraints
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 1e-10:
                h = h * (n_points / 2) / sum_h
            else:
                h = np.ones(n_points) * (n_points / 2)
            
            particles[i] = h
            velocities[i] = v
            
            # Evaluate fitness
            fitness = evaluate_erdos_solution(h, dx)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = h.copy()
                pbest_fitness[i] = fitness
                
                # Update global best if needed
                if fitness < gbest_fitness:
                    gbest = h.copy()
                    gbest_fitness = fitness
                    last_improvement_iter = iter_count
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
                
        # Warm restart if stagnation
        if stagnation_count > warm_restart_after:
            for i in range(num_particles // 2):
                idx = random.randint(0, num_particles - 1)
                particles[idx] = np.random.random(n_points)
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                velocities[idx] = np.zeros(n_points)
                # Recompute fitness
                h = particles[idx]
                fitness = evaluate_erdos_solution(h, dx)
                pbest[idx] = h.copy()
                pbest_fitness[idx] = fitness
            stagnation_count = 0
            last_improvement_iter = iter_count
            
    # Final local refinement of best solution
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_init = gbest.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < gbest_fitness:
                    gbest = h_final
                    gbest_fitness = c5_final
        except:
            pass
    
    return (list(gbest), gbest_fitness, n_points)
