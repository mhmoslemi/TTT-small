import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    num_particles = 40
    max_iter = 500
    inertia_weight = 0.7
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.2
    warm_restart_after = 200
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    for i in range(num_particles):
        if i < len(h_start):
            h = h_start.copy()
        else:
            h = np.random.random(n_points)
            h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_fitness.append(evaluate_erdos_solution(h, dx))
    
    # Global best initialization
    best_idx = np.argmin(pbest_fitness)
    gbest = pbest[best_idx].copy()
    gbest_fitness = pbest_fitness[best_idx]
    
    stagnation_count = 0
    restart_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Check for stagnation and warm restart
        if iteration > 0 and pbest_fitness[best_idx] >= gbest_fitness * 0.99999:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        if stagnation_count >= warm_restart_after:
            # Warm restart: reset velocities and slightly perturb half the particles
            for i in range(num_particles):
                velocities[i] = np.zeros(n_points)
                if np.random.random() < 0.5:
                    particles[i] = particles[i] + np.random.normal(0, 0.01, n_points)
                    # Repair constraints
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                    pbest[i] = particles[i].copy()
                    pbest_fitness[i] = evaluate_erdos_solution(particles[i], dx)
            stagnation_count = 0
            restart_count += 1
        
        # Update all particles
        for i in range(num_particles):
            h = particles[i]
            v = velocities[i]
            p = pbest[i]
            
            # Update velocity
            r1, r2 = np.random.random(2)
            v = (inertia_weight * v 
                 + c1 * r1 * (p - h) 
                 + c2 * r2 * (gbest - h))
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            
            # Update position
            h = h + v
            
            # Enforce constraints
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 1e-10:
                h = h * (n_points / 2) / h_sum
            else:
                h = np.ones(n_points) * (n_points / 2)
            
            # Store updated values
            particles[i] = h
            velocities[i] = v
            
            # Evaluate fitness
            new_fitness = evaluate_erdos_solution(h, dx)
            
            # Update personal best
            if new_fitness < pbest_fitness[i]:
                pbest[i] = h.copy()
                pbest_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < gbest_fitness:
                    gbest = h.copy()
                    gbest_fitness = new_fitness
                    best_h = h.copy()
                    best_c5 = new_fitness

    # Local refinement of top particles using SLSQP
    top_k = min(5, len(particles))
    top_indices = np.argsort(pbest_fitness)[:top_k]
    
    for i in top_indices:
        h_refined = pbest[top_indices[i]].copy()
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con(h):
            return np.sum(h) - n_points / 2.0
            
        cons = {'type': 'eq', 'fun': con}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, h_refined, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_new = res.x
                f_new = obj(h_new)
                if f_new < pbest_fitness[i]:
                    pbest[i] = h_new
                    pbest_fitness[i] = f_new
                    if f_new < gbest_fitness:
                        gbest = h_new.copy()
                        gbest_fitness = f_new
                        best_h = h_new.copy()
                        best_c5 = f_new
        except:
            pass
            
        if time() - start_time > max_time * 0.95:
            break

    return (list(best_h), best_c5, n_points)
