import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try convex relaxation approach
    try:
        # SDP relaxation
        X = cp.Variable((n_points, n_points), symmetric=True)
        h_vars = cp.Variable(n_points)
        
        # Objective: minimize max correlation
        # For simplicity, we'll solve a modified version
        # Create a complete positive semidefinite matrix variable
        objective = cp.Minimize(cp.trace(X))
        
        constraints = [
            X >> 0,  # Positive semidefinite
            cp.sum(X, axis=0) == 1.0,  # Column sums
            cp.sum(X, axis=1) == 1.0,  # Row sums
        ]
        
        # Add constraints on diagonal elements
        for i in range(n_points):
            constraints.append(X[i, i] <= 1.0)
        
        # Solve SDP
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
        
        if prob.status == cp.OPTIMAL:
            # Extract feasible solution
            X_sol = X.value
            if X_sol is not None:
                # Reconstruct h from the relaxation
                eigenvals, eigenvecs = np.linalg.eigh(X_sol)
                h_recovered = np.abs(eigenvecs[:, np.argmax(eigenvals)])
                h_recovered = h_recovered / np.sum(h_recovered) * (n_points / 2)
                h_recovered = np.clip(h_recovered, 0, 1)
                
                # Local refinement
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(objective, h_recovered.copy(), method='L-BFGS-B', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-8})
                
                if res.success:
                    h_refined = res.x
                    h_refined = np.clip(h_refined, 0, 1)
                    h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
    except:
        pass

    if time() - start_time > budget_s * 0.9:
        return (list(best_h), best_c5, n_points)
    
    # Local search starting from current best

                    c5_var = objective(h_opt)

                        best_c5 = c5_var
                        
        except:
            pass

    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        try:
            h_final = best_h.copy()
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0

            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_final, method='L-BFGS-B',

            if res.success:

                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
