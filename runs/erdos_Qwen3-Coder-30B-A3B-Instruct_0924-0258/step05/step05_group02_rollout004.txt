import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach
    n_low = 20  # Coarse resolution
    dx_low = 2.0 / n_low
    h_low = np.ones(n_low) * 0.5
    
    # Coarse grid optimization
    if n_low < n_points:
        try:
            # Initial coarse solution from best found so far
            h_low = np.interp(np.linspace(0, 2, n_low), np.linspace(0, 2, n_points), best_h)
            h_low = h_low * (n_low / 2) / np.sum(h_low)
            h_low = np.clip(h_low, 0, 1)
        except Exception:
            pass
            
        # Convex optimization for coarse grid
        try:
            h_relaxed = cp.Variable(n_low)
            constraints = [
                cp.sum(h_relaxed) == n_low / 2.0,
                h_relaxed >= 0,
                h_relaxed <= 1
            ]
            
            # Add limited overlap constraints (subset of shifts)
            max_shift = min(5, n_low - 1)
            for k in range(1, max_shift + 1):
                if k < n_low // 2:
                    corr_sum = cp.sum(cp.multiply(h_relaxed[:n_low-k], 1 - h_relaxed[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_low-k]))
                constraints.append(corr_sum <= best_c5 * dx_low + 1e-6)
            
            prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-7)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_relaxed.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx_low)
                if c5_feasible < best_c5:
                    h_low = h_feasible
                    if n_points == 35:  # Only update best if we'll use the result
                        best_c5 = c5_feasible
        except Exception:
            pass

    # Interpolate to target resolution
    if n_low < n_points:
        h_interp = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_low), h_low)
        h_interp = h_interp * (n_points / 2) / np.sum(h_interp)
        h_interp = np.clip(h_interp, 0, 1)
        if evaluate_erdos_solution(h_interp, dx) < best_c5:
            best_h = h_interp
            best_c5 = evaluate_erdos_solution(h_interp, dx)

    # Refinement
    max_restarts = 50
    restart_count = 0
    
    # Local refinement with SLSQP
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    while time() - start_time < budget_s * 0.95:
        # SLSQP refinement step
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            res = minimize(
                objective,
                best_h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 20, 'ftol': 1e-7}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
                else:
                    restart_count += 1
            else:
                restart_count += 1
        except Exception:
            restart_count += 1
        
        # Random restart when stuck
        if restart_count > max_restarts:
            h_restart = np.random.rand(n_points)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            h_restart = np.clip(h_restart, 0, 1)
            c5_restart = evaluate_erdos_solution(h_restart, dx)
            if c5_restart < best_c5:
                best_h = h_restart
                best_c5 = c5_restart
            restart_count = 0

    # Final normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
