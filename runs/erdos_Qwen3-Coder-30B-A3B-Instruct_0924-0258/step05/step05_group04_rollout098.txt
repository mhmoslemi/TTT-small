import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define the convex optimization approach
    max_iter = 500
    rho = 1e3
    lambda_init = 0.1
    mu_init = 1.0
    
    # Build correlation matrices A_k for shifts k = 0, ..., 2*n_points-2
    def build_correlation_matrices(n):
        A = []
        for k in range(2*n-1):
            # Create matrix that computes h^T A_k (1-h) for shift k
            # This simulates the overlap integral with shift k
            a = np.zeros((n, n))
            for i in range(n):
                for j in range(n):
                    if 0 <= j-i <= k and j-i < n:
                        a[i, j] = 1
            A.append(a)
        return A
    
    A_matrices = build_correlation_matrices(n_points)
    
    # Lagrangian optimization approach
    h = cp.Variable(n_points)
    lambda_mult = cp.Variable(2*n_points-1)
    mu = cp.Variable()
    
    # Define the constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == n_points/2.0]
    
    # Objective: minimize max overlap
    # We'll approximate by minimizing the first element of the correlation vector,
    # since that's max overlap over all shifts
    # Not ideal but sufficient for this approach
    
    for it in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Solve the convex relaxation
        try:
            # Create the Lagrangian components more accurately
            # For each shift k, we want the term with h^T A_k (1-h)
            # But we want the maximum of such terms, not sum
            
            # Instead try a direct relaxation: we can solve a constrained optimization
            # let's consider C5 = max_k h^T A_k (1-h)
            # Use cvxpy to minimize an upper bound
            
            # Use the fact that max_k f_k(x) <= t <=> f_k(x) <= t for all k
            # So we impose: h^T A_k (1-h) <= t for all k (or the max of them)
            
            # Create dummy variable for max overlap
            t = cp.Variable()
            # At each step, we solve a QP that approximates the touches of the problem
            
            # Limitation: In practice, KKT conditions of this relaxed feasibility 
            # are hard to compute here, so we simplify by using a heuristic solution
            
            # Criteria for improving: 
            h_current = best_h.copy()
            
            # Apply local optimization: start from current best h
            def obj(hv):
                return evaluate_erdos_solution(hv, dx)
            
            def con_sum(hv):
                return np.sum(hv) - n_points / 2.0
            
            constraints_local = {'type': 'eq', 'fun': con_sum}
            bounds_local = [(0, 1) for _ in range(n_points)] 
            
            try:
                res = minimize(obj, h_current, method='SLSQP', bounds=bounds_local,
                               constraints=constraints_local, options={'ftol': 1e-8})
                if res.success:
                    h_new = res.x
                    f_new = obj(h_new)
                    if f_new < best_c5:
                        best_c5 = f_new
                        best_h = h_new.copy()
            except:
                pass
                
            # Periodic local search refinement
            if it % 20 == 0:
                # Try random perturbations with small jumps to escape local opt
                h_refined = best_h + np.random.normal(0, 0.01, n_points)
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)

                    best_c5 = c5_refined

            if best_c5 < 0.3807:

        except Exception as e:

    remaining_time = max_time - (time() - start_time)

        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass

    return (list(best_h), best_c5, n_points)
