import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try SDP relaxation approach first
    try:
        # Try to solve using a simpler and faster approach:
        # This is a convex relaxation of the original problem.
        # Use a local optimization approach instead
        
        # Set up the optimization problem
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        # Box constraints
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Start optimization from different initial points
        best_c5 = float('inf')
        best_h = None
        
        # Try to use multiple random restarts
        n_restarts = 5
        for restart in range(n_restarts):
            if time() - start_time > budget_s * 0.9:
                break
                
            # Random restart
            if restart == 0:
                h_init = h_start.copy()
            else:
                h_init = np.random.uniform(0, 1, n_points)
                h_init = h_init * (n_points/2) / np.sum(h_init)
            
            # Local optimization
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, h_init, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = c5_refined

        if best_h is None or best_c5 >= 0.3807:
            # Use the initial h_start and try local optimization with high iterations
            h_start = h_start.copy()
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, h_start, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 500, 'ftol': 1e-10})
                if res.success:
                    h_final = res.x
                    c5_final = objective(h_final)
                    if c5_final < best_c5:
                        best_h = h_final.copy()
                        best_c5 = c5_final
            except:
                pass
        
    except Exception as e:
        # Fallback to simple approach
        best_h = h_start.copy()
        best_c5 = evaluate_erdos_solution(best_h, dx)

    return (list(best_h), best_c5, n_points)
