import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create symmetric triangular initial function
        mid = n_points // 2
        h_start = np.minimum(np.arange(n_points), 2*mid - np.arange(n_points))
        h_start = h_start / (mid)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization parameters
    max_iter = 1000
    patience = 100
    stagnation_count = 0
    best_c5_history = []
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Start with initial solution
    current_h = h_start.copy()
    current_c5 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Local optimization using SLSQP
        try:
            res = minimize(objective, current_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                
                # Re-scale to maintain constraint
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = objective(h_refined)
                
                if c5_refined < current_c5:
                    current_h = h_refined
                    current_c5 = c5_refined
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
                        stagnation_count = 0
                        best_c5_history = []
                        if best_c5 < 0.3807:
                            break
                    else:
                        stagnation_count += 1
                else:
                    stagnation_count += 1
                    
            else:
                # If local minimization fails, try perturbation
                perturbation = np.random.normal(0, 0.01, n_points)
                h_new = current_h + perturbation
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                c5_new = objective(h_new)
                if c5_new < current_c5:
                    current_h = h_new
                    current_c5 = c5_new
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
                        stagnation_count = 0
                    else:
                        stagnation_count += 1
                else:
                    stagnation_count += 1
                    
        except Exception as e:
            # Fallback on random perturbation
            perturbation = np.random.normal(0, 0.01, n_points)
            h_new = current_h + perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = objective(h_new)
            if c5_new < current_c5:
                current_h = h_new
                current_c5 = c5_new
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
        
        # Early stopping
        if stagnation_count >= patience:
            break
            
        # Store history for possible rollback
        best_c5_history.append(current_c5)
        if len(best_c5_history) > 10:
            best_c5_history.pop(0)
    
    # Final refinement using more aggressive optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                h_final = h_final * (n_points/2) / np.sum(h_final)
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
