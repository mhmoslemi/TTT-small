import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialization
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize initial solution
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    h = initial_h.copy()
    
    # Parameters
    max_iter = 500
    patience = 50
    shift_samples = 50
    learning_rate = 0.01
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    stagnation_count = 0
    last_best_c5 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample shifts
        shifts = np.random.randint(0, n_points, shift_samples)
        max_overlap = -1
        best_shift = 0
        
        # Find worst shift
        for k in shifts:
            # Compute overlap for shift k
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlap *= 2.0 / n_points
            
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = k
        
        # Compute gradient for the worst shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_shift) % n_points
            grad[i] += (1 - h[j]) * 2.0 / n_points
            grad[j] -= h[i] * 2.0 / n_points
        
        # Update with gradient descent
        h_new = h - learning_rate * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, 2.0 / n_points)
        
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            stagnation_count = 0
            last_best_c5 = best_c5
        else:
            stagnation_count += 1
            
        # Early termination
        if best_c5 < 0.3807:
            break
            
        # Restart if stuck
        if stagnation_count > patience:
            # Random perturbation
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0
    
    # Final refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            h = res.x
            best_c5 = obj(h)
    except:
        pass
    
    return (list(h), best_c5, n_points)
