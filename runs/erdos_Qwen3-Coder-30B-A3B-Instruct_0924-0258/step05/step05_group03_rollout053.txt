import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Parameters
    max_iterations = 100
    shift_samples = 100
    min_improvement = 1e-5
    
    # Generate shift samples
    shifts = np.linspace(0, 2, shift_samples, endpoint=False)
    
    # Initialize variables
    h_opt = initial_h.copy()
    best_h = h_opt.copy()
    best_c5 = evaluate_erdos_solution(h_opt, 2.0 / n_points)
    
    # Cutting plane loop
    last_c5 = best_c5 + 1.0
    iteration = 0
    
    while iteration < max_iterations and time() - start_time < budget_s * 0.9:
        # Sample overlaps for shifts
        overlaps = []
        for k in shifts:
            # Calculate correlation for shift k
            if k == 0:
                corr = np.correlate(h_opt, 1 - h_opt, mode='full')
                overlap = np.max(corr) * (2.0 / n_points)
            else:
                # For non-zero shifts
                corr = np.correlate(h_opt, 1 - h_opt, mode='full')
                # Need to extract the appropriate slice
                idx = int(k * n_points / 2)
                overlap = corr[len(corr)//2 + idx] * (2.0 / n_points)
            overlaps.append(overlap)
        
        max_overlap_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_overlap_idx]
        
        # If improvement is minimal, break
        if abs(max_overlap - last_c5) < min_improvement:
            break
            
        # Add constraint to cutting plane model
        last_c5 = max_overlap
        
        # Refine best solution if this is a significant improvement
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h_opt.copy()
            
        iteration += 1
        
        if best_c5 < 0.3807:
            break
    
    # Final local refinement with SLSQP
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            final_c5 = obj(best_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
