import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Sequential Convex Programming approach
    max_iter = 500
    mu = 1e-3
    c5_bound = 0.4
    tolerance = 1e-6
    max_time = budget_s * 0.95
    
    # Build shift matrices for overlap constraints
    def build_shift_matrices(n):
        shifts = []
        for k in range(n):
            shift_mat = np.zeros((n, n))
            for i in range(n):
                shift_mat[i, (i+k) % n] = 1
            shifts.append(shift_mat)
        return shifts

    shift_matrices = build_shift_matrices(n_points)
    
    # Precompute correlation indices
    full_corr_len = 2 * n_points - 1
    full_indices = np.arange(full_corr_len)
    
    for iter_num in range(max_iter):
        if time() - start_time > max_time:
            break
            
        current_time = time() - start_time
        if current_time > max_time * 0.8 and best_c5 < 0.3807:
            break
            
        # Formulate QCQP approximation
        # Minimize: C5
        # Subject to: sum(h * (1 - h_shifted)) * dx <= C5 for all shifts
        #           : sum(h) * dx = 1
        #           : 0 <= h <= 1
        
        # Linearize the non-convex constraints around current iterate
        h_current = best_h.copy()
        
        # Compute all overlaps at current point
        overlaps = []
        for k in range(n_points):
            # Circular shift of h_current to left by k positions
            h_shifted = np.roll(h_current, k)
            overlap = np.dot(h_current, 1 - h_shifted)
            overlaps.append(overlap)
            
        # Identify worst overlap
        max_overlap_idx = np.argmax(overlaps)
        max_overlap_val = overlaps[max_overlap_idx]
        
        # If constraints are satisfied, exit
        if max_overlap_val <= c5_bound + tolerance:
            break
            
        # Use convex approximation approach
        # Minimize C5 s.t. sum(h * (1-h_k)) * dx <= C5 for all shifts k
        # We need to build this as a QP.
        
        # Constraint: sum(h) = n_points/2 (since dx=2/n_points)
        # Constraint for each shift k: sum_j h[j]*(1-h_shifted[j]) <= C5/dx
        # This is non-convex; we linearize it using current iterate:
        # sum_j h[j] * (1 - h_shifted_current[j]) <= C5/dx
        # We want to estimate:
        # \sum h_j * (1 - h_shifted_current[j])
        # = \sum h_j - h_j * h_shifted_current[j]
        # The h_shifted_current part is fixed.
        
        # Define optimization variables
        h_vars = cp.Variable(n_points)
        c5_var = cp.Variable()
        
        # Original objective
        objective = cp.Minimize(c5_var)
        
        constraints = []
        
        # Equality constraint: sum(h) = n_points/2
        constraints.append(cp.sum(h_vars) == n_points / 2.0)
        
        # Upper bound constraint using linearization around h_current
        # For each shift k, we linearize h_j * (1 - h_shifted[j]) at h_current
        # Linearized term around h_current = h_j*(1-h_shifted_current[j])
        # Parallel to the "linearization" that is
        # sum h_j (1 - h_shifted_current[j]) := a_j where h_shifted_current[j] is fixed
        # This is just a linear expression.
        
        # But doing that ad hoc with no repairing might be unstable.
        # Let's stabilize with a different approach using a feasible first-order DC approximation.
        # At h_current, sum(h_j * (1 - h_shifted_current[j])) 
        # = sum(h_j) - sum(h_j * h_shifted_current[j])
        # = 1 - sum(h_j * h_shifted_current[j])
        # However, the shifted view: that shifts each element; 
        # I should be doing this properly.
        # 
        # Let compute for shift k:
        # sum h[j] * (1 - h[(j-k) mod n])
        # Computed with matrix operation as in overlap calculation.
        
        # Now we use the linearization: 
        # For each k = 0,...,n-1 we want to satisfy:
        # sum h[j] * (1 - h_shifted[j]) <= C5/dx up to linearization
        # At h_current, we approximate:
        #   sum h[j] * (1 - h_shifted_current[j]) <= C5/dx
        # i.e. sum h[j] - sum h[j] * h_shifted_current[j] <= C5/dx
        
        # But that doesn't make the optimization well-crafted.
        # Approach: Instead, we use Augmented Lagrangian,
        # but it's easier to go direct like:
        
        # Consistent view:
        # Each shift k constraint:
        # sum h[i] * (1 - h[i-k]) <= C5*dx
        # Where h[i-k] is done cyclically.
        # At h_current, this becomes:
        # sum h_current[i] * (1 - h_shifted_current[i]) <= C5*dx
        # That simplifies to:
        # sum h_current[i] - sum h_current[i] * h_shifted_current[i] <= C5*dx
        
        # The idea: approximate this as:
        # sum h[i] * (1 - h[i-k]) <= C5*dx, i.e. sum(h * (1 - h_k)) <= C5*dx
        
        # Use h_current as linearization.
        # So, we'll linearize around h_current:
        # sum h[i] * (1 - h[i-k]) <= sum h_current[i] * (1 - h_current[i-k]) 
        #        = sum(h_current) - sum(h_current * shifted(h_current)) 
        # = 1 - sum(h_current * shifted(h_current))
        
        # Rough linearized constraint is:
        # sum(h) - sum(h * shifted_h) <= C5/dx
        # But we do this for all shifts.
        # And we want to minimize C5. Recast as:
        # max sum(h * (1 - shifted_h_current)) <= Optimize C5
        # But constraint needs to keep all shifts <= C5*dx
        
        # Simpler and more direct approach:
        # Use one convex master problem and then refine.
        # Let's just linearize the worst constraint around h_current
        
        worst_k = max_overlap_idx
        # Exploit the simplest linearization:
        #  At h_current, we do:
        #  sum h[i] * (1 - h[i - k]) <= sum h_current[i] * (1 - h_current[i-k]) = const
        #  But this is not simply a linear constraint in h because of shifting: that's what 
        #  the correlation handle is. 
        # So just build:
        # sum(conf[i] * h[i]) <= rhs, where conf[i] = 1 - h_current[(i-k)%n]
        # We freeze k = k_opt and use that.
        
        # Step 1: Build shifted iterate
        h_shifted_current = np.roll(h_current, worst_k)
        # Linearize sum(h * (1 - h_shifted_current)) 
        # at h_current.
        # sum h[j] * (1 - h_shifted_current[j]) = 
        # sum h_current[j] * (1 - h_shifted_current[j]) = 
        # sum h_current[j] - sum h_current[j]*h_shifted_current[j] = 
        # 1.0 - sum(h_current * h_shifted_current)
        # But here again: this constraint is not in the right form for LP/QP solvers.
        # Especially for non-linear problems that have highly non-convex combinations.
        # Sad corner of the dual space where by shifting, we cannot easily express
        # the constraint in [h_i] terms simply.
        # So instead, simplify: we keep a "reference" file structure and 
        # for constraint error tracking, we can solve a "fixed shift" scenario or 
        # if the shifts are exact shifts, then technically we just do the correlation.
        
        # Frame as: the last constraint that was violated is the one most important
        # So,
        # use the last constraint which violated h_current, across that shift k:
        # sum(h_current[i] * (1 - h_current[i-k])) <= C5 * dx is the bound we're 
        # linearizing around.
        # Now we'll create a conservatism disallowing this by bounding with a convex constraint.
            
        # Because we're doing that deterministic over-shift side-efficiency, 
        # we only require a vector of the right size to approximate it.
        
        # Just optimize h such that we improve the bound.
        # We can build constraints like:
        # H*k  <= C
        # But with correlation constraint, using constraint過ぎ을 수 있다.
        # This is:
        # We know k_opt = argmax_k sum h_current[i] * (1 - h_shifted[i]), so that'
        # sum h[i] * (1 - h[i-k_opt]) <= sum h_current[i] * (1 - h_current[i-k_opt])
        # As a constraint on h 
        # Linearization: sum h[i] - sum h[i]*h[i-k_opt] <= const
        # Rearranging:
        # sum h[i] - const <= sum h[i] * h[i-k_opt]
        # Which wouldn't make this convex in the dual.
        
        # Better we linearize at h_current in terms of mutual optimization:
        # Let's simplify then: 
        # We build sensitivity to the key overlap constraint that causes max_c5.
        # Take the shift that leads to worst overlap.
        # For all shifts: compute overlap(h_current) -> get key k.
        # Do Jacobian modularization with that key shift.
        # Build one QP:
        # Minimize c5
        # s.t. 
        #   for shift k_star
        #     sum h[i] * (1 - h[i - k_star]) <= c5 * dx
        #   sum h = n_points/2
        #   0 <= h <= 1
        #   The constraint involves classically h * shifted_h, which is quadratic.
        # In fact, let h_i be point i. Consider h[i-k]: 
        # Sum of h[i] * h[i-k] over i for fixed k (circularly).
        # That means we cannot write this as just h_i column matrix multiply with
        # such a part of h, that is, it's more complex. 
        # 
        # So let's take a better formed variant:
        
        h_shifted_k_star = np.roll(h_current, worst_k)
        # The bad overlap: sum h_current * (1 - h_shifted_k_star) = 1 - sum(h_current * h_shifted_k_star) = const_val
        
        # It means: 
        # Sum h[i] * ((1 - h_shifted_k_star[i]))
        # is constraint
        # sum(h[i] * (1 - h_shifted_k_star[i])) <= C5 * dx
        # We can write the constraint:
        # sum h[i] - sum h[i] * h_shifted_k_star[i] <= C5 * dx
        # Which is a quadratic:
        # sum h[i] - sum h[i] * h_shifted_k_star[i]
        # That's about n variables with 2nd order terms, what we need to linearize or approximate.
        # Solution approach:
        # Add the worst constraint.linearization as:
        # sum h[i] - sum h_current[i] * h_shifted_k_star[i] <= C5 * dx
        # (Simplifying incorrectly by absorbing)
        # Actually:
        # sum h[i]*(1-h_shifted_k_star[i]) = sum h[i] - sum h[i]*h_shifted_k_star[i] <= C5*dx
        # That means: sum h[i]*h_shifted_k_star[i] >= sum h[i] - C5*dx
        # So similar to other QCQP structure where that result is same quadratic itself.
        # But it's most informative.
        # So what least conservative is:
        # Constraint: h^T (1 - h_k*) > approx_constraint_val
        # Least conservative would be to:
        # For constraint at permutation corresponding to worst k:
        # Approx using first order at current h -> sum h[i] - sum h_current[i] * h_shifted_k_star[i] <= c5*dx
        # But to keep this purely quadratic, use:
        # h^T (1 - h_k_star_track) = const
        # For current h_c, this becomes Inn_prod(h_c, 1 - h_c_k_star) = const.
        # So, let's create that undefeated constraint, and form the very first quadratic,
        # for this interval just to improve the value closely.
        
        # So, 'ideal function' is:
        # Constrain all k shifts, normalize to dx.
        # sum over h[i] * (1 - h[(i+k)%n]) <= max_c5 * dx
        # Linearization around h_current:
        # sum h[i] * (1 - h_current[(i+k)%n]) <= max_c5 * dx  # This is not quite right,
        # if we approximate that sum h[i] - sum h_current[i] * h[(i+k)%n] <= C5 * dx
        # Which is still quite complicated since h[i] with h_shifted depend.
        
        # Approach: 
        # Build one step QP that resembled 
        # sum h[i] * (1 - h[i-k]) <= c5*dx
        # for k = argmax_k (sum(h[i] * (1 - h[i-k]))).
        # We do this with linearization trick around the current h.
        # Cx + d <= [sum of h] - sum(h_current[i] * h_shifted_current[i])
        # Let's approach it differently:
        # Linearization around h_current:
        # For shift k:
        # Let x = position (i) 
        # sum_{i,k} (h[i] * (1 - h[i+k])) 
        # = sum h[i] - sum h[i] * h[i+k] <= C5 * dx
        # Linearized at h_current:
        # sum h[i] - sum h_current[i] * h[i+k] <= C5 * dx
        # = sum h[i] - sum h_current[i] * h[i+k] <= C5 * dx
        # Best way to do this with one constraint  
        # This is totally non-admissible form, don't do it! Always 
        # Strongly recommend block-QP terms.
        # More reliable:
        # Run simple trick that finds good shift by saturation, and do 
        # Only update the part, which directly movement is most likely.
        
        # Do QP with focused focus:
        
        # For shifts, again compute:
        h_k_star = np.roll(h_current, worst_k)
        # We want to minimize 
        # max sum h[i] * (1 - h_shifted_current[i]) for all k
        # >= for that particular k that maximizes.
        # idea:
        # build matrix P (for such correlations as h^T H h) 
        # which enforce huge part of hard overlap.
        # Use quarterly parametrizations of that.
        
        h_vars = cp.Variable(n_points)
        c5_var = cp.Variable()
        
        objective = cp.Minimize(c5_var)
        
        constraints = []
        # Feasibility: sum(h) = 1/dx * dx = n_points/2
        constraints.append(cp.sum(h_vars) == n_points / 2.0)
        
        # Energy-max overlap constraint around
        # Somewhat inefficient but keeps it safe:
        # We approximate the constraint:
        # sum(h[i] * (1 - h[current_shifted[i]])) <= c5 * dx
        # We linearize around h_current:
        # sum h[i] - sum h[i] * (current_shifted[i]) <= c5 * dx
        # So,
        # Let 
        # a = h_current
        # b = shift_h_current
        # Then sum h[i] - sum h[i] * b[i] <= c5 * dx  
        # i.e. 1 - sum h[i] * b[i] <= c5 * dx  
        # i.e. -sum h[i] * b[i] <= c5 * dx - 1
        # i.e. sum h[i] * b[i] >= 1 - c5 * dx
        
        b_k_star = h_shifted_k_star
        # Linearization is more simple.
        # sum(h[i] * b_k_star[i]) <= 1 - c5_var * dx
        #   sum h[i] - sum h[i]*b_k_star[i] <= c5_var * dx   ===>
        #   1 - sum h[i]*b_k_star[i] <= c5_var * dx
        #   sum h[i]*b_k_star[i] >= 1 - c5_var * dx 
        #   sum h[i]*b_k_star[i] >= (1 - c5_var * dx) 
        # But due to nature, shift interaction, particularly for sum(i,j) 
        # h_i h_j that corresponds to h[i] * h[i-k].
        # Let's preserve it by 이렇게 Jenkins 정리:
        # sum h[i] * h_shifted_i[i] =< val  
        # This needs
        # Freeze the potential for violation
        # Linearisation around h_current:
        # => 1 - sum (h_current[i] * h_shifted_current[i]) <= c5 * dx
        # => sum h_current[i] * h_shifted_current[i] >= (1 - c5 * dx)  
        
        # Slightly safely reinterpret:
        # h_current^T (1 - h_shifted_current) = 1 - h_current^T h_shifted_current
        # Then linearization with h_current:
        # 1 - h_current^T h_shifted_current <= C5*dx
        # => h_current^T h_shifted_current >= (1 - C5*dx)
        
        # So we want:
        #  sum(h[i] * h_shifted_current[i]) >= (1 - c5_var * dx) 
        # But not false LHS  
        # For each k=worst_k:
        
        # Cap constraints.
        # sum h[i] - sum h[i]*b_k_star[i] <= c5_var * dx  
        # 1 - sum h[i]*b_k_star[i] <= c5_var * dx
        # => sum h[i]*b_k_star[i] >= 1 - c5_var * dx
        # => We use inequality (in CvxPy).
        # Construct the scalar part:
        rhs = 1 - c5_var * dx
        # and form constraint:
        constraints.append(cp.sum(cp.multiply(h_vars, b_k_star)) >= rhs)
        
        # Bound on h: 0 <= h <= 1
        for i in range(n_points):
            constraints.append(h_vars[i] >= 0)
            constraints.append(h_vars[i] <= 1)
        
        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.OSQP, verbose=False, max_iter=5000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                c5_new = c5_var.value
                # Normalize properly
                h_new = h_new * (n_points / 2.0) / np.sum(h_new)
                if np.isnan(h_new).any():
                    continue
                c5_eval = evaluate_erdos_solution(h_new, dx)
                if c5_eval < best_c5:
                    best_c5 = c5_eval
                    best_h = h_new.copy()
        except Exception as e:
            pass
        
        # Update mu
        if c5_new > best_c5:
            mu *= 2
        
        # Check convergence:
        if abs(best_c5 - c5_new) < 1e-6 and current_time > 10:
            break
        
        # Early termination
        if best_c5 < 0.3807:
            break
    
    # Final refinement with trust-region ...
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                           constraints=cons_sum, options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
                
    return (list(best_h), best_c5, n_points)
