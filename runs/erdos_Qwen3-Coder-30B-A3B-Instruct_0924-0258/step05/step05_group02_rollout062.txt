import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a hybrid scheme: gradient-based refinement with sparse search
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    # Local refinement with SLSQP
    max_restarts = 20
    restart_count = 0
    local_improvements = 0
    max_local_improvements = 50
    
    while time() - start_time < budget_s * 0.95:
        # Local search with perturbations
        h_perturbed = best_h.copy()
        h_perturbed += np.random.normal(0, 0.03, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    local_improvements = 0
                else:
                    local_improvements += 1
            else:
                local_improvements += 1
        except Exception:
            local_improvements += 1
            
        if local_improvements > max_local_improvements:
            # Do a heavier random restart
            h_restart = np.random.rand(n_points)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            h_restart = np.clip(h_restart, 0, 1)
            c5_restart = evaluate_erdos_solution(h_restart, dx)
            if c5_restart < best_c5:
                best_h = h_restart
                best_c5 = c5_restart
            local_improvements = 0
            
        # Periodic perturbation and clustering
        if restart_count % 5 == 0:
            h_cluster = np.copy(best_h)
            # Try clustering approach to generate sparse structure
            cluster_means = []
            for i in range(0, n_points, 5):
                if i + 5 <= n_points:
                    avg_val = np.mean(h_cluster[i:i+5])
                    cluster_means.extend([avg_val] * 5)
                else:
                    cluster_means.extend([np.mean(h_cluster[i:])] * (n_points - i))
            
            h_cluster = np.array(cluster_means)
            h_cluster = h_cluster * (n_points/2) / np.sum(h_cluster)
            c5_cluster = evaluate_erdos_solution(h_cluster, dx)
            if c5_cluster < best_c5:
                best_h = h_cluster
                best_c5 = c5_cluster
        
        restart_count += 1
        
        # Global solution restart
        if time() - start_time > budget_s * 0.9:
            break
            
    # Final refinement and normalization
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    # Final check
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    
    return (list(final_h), best_c5, n_points)
