import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Convert to binary with exact sum constraint
    def to_binary(h):
        h_bin = (h > 0.5).astype(int)
        current_sum = np.sum(h_bin)
        target_sum = n_points // 2
        if current_sum > target_sum:
            # Turn some 1s to 0s
            indices = np.where(h_bin == 1)[0]
            to_flip = current_sum - target_sum
            flip_indices = np.random.choice(indices, to_flip, replace=False)
            h_bin[flip_indices] = 0
        elif current_sum < target_sum:
            # Turn some 0s to 1s
            indices = np.where(h_bin == 0)[0]
            to_flip = target_sum - current_sum
            flip_indices = np.random.choice(indices, to_flip, replace=False)
            h_bin[flip_indices] = 1
        return h_bin
    
    best_h = to_binary(h_start)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Tabu search with binary variables
    tabu_list = []
    tabu_tenure = 10
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Initialize with parent or create random binary solution
    if 'initial_h_values' in globals():
        h_current = to_binary(h_start)
    else:
        h_current = np.zeros(n_points, dtype=int)
        ones_indices = np.random.choice(n_points, n_points // 2, replace=False)
        h_current[ones_indices] = 1
        h_current = h_current.astype(int)
    
    current_c5 = evaluate_erdos_solution(h_current, dx)
    if current_c5 < best_c5:
        best_h = h_current.copy()
        best_c5 = current_c5
    
    # Tabu search loop
    while time() - start_time < budget_s * 0.95:
        no_improvement_count += 1
        if no_improvement_count > max_no_improvement:
            # Restart with best solution
            h_current = best_h.copy()
            current_c5 = best_c5
            no_improvement_count = 0
        
        # Generate neighbors by flipping one 1 to 0 and one 0 to 1
        ones = np.where(h_current == 1)[0]
        zeros = np.where(h_current == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            # Minimal change - random restart
            h_current = np.zeros(n_points, dtype=int)
            ones_indices = np.random.choice(n_points, n_points // 2, replace=False)
            h_current[ones_indices] = 1
            current_c5 = evaluate_erdos_solution(h_current, dx)
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
            continue
        
        # Choose a random 1 and 0 to flip
        flip_1_idx = np.random.choice(ones)
        flip_0_idx = np.random.choice(zeros)
        
        # Create neighbor
        h_neighbor = h_current.copy()
        h_neighbor[flip_1_idx] = 0
        h_neighbor[flip_0_idx] = 1
        
        # Check tabu list
        move = (flip_1_idx, flip_0_idx)
        is_tabu = move in tabu_list or (flip_0_idx, flip_1_idx) in tabu_list
        
        # Evaluate neighbor
        neighbor_c5 = evaluate_erdos_solution(h_neighbor, dx)
        
        # Apply aspiration criteria: if better than best, accept even if tabu
        aspiration = neighbor_c5 < best_c5
        
        # Accept if not tabu or if aspiration
        if not is_tabu or aspiration:
            h_current = h_neighbor
            current_c5 = neighbor_c5
            no_improvement_count = 0
            
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
                
        # Update tabu list
        tabu_list.append(move)
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
        
        # Shake every 50 iterations without improvement
        if no_improvement_count % 50 == 0 and no_improvement_count > 0:
            # Flip a few random bits to escape local minima
            to_flip_count = max(1, n_points // 20)
            all_indices = np.arange(n_points)
            flip_idx = np.random.choice(all_indices, to_flip_count, replace=False)
            for i in flip_idx:
                h_current[i] = 1 - h_current[i]
                
            current_c5 = evaluate_erdos_solution(h_current, dx)
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
            no_improvement_count = 0
    
    # Ensure final solution is valid and recompute
    best_h = to_binary(best_h)
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        # Adjust to maintain sum constraint
        current_sum = np.sum(best_h)
        target_sum = n_points // 2
        if current_sum > target_sum:
            # Turn some 1s to 0s
            indices = np.where(best_h == 1)[0]
            to_flip = current_sum - target_sum
            flip_indices = np.random.choice(indices, to_flip, replace=False)
            best_h[flip_indices] = 0
        elif current_sum < target_sum:
            # Turn some 0s to 1s
            indices = np.where(best_h == 0)[0]
            to_flip = target_sum - current_sum
            flip_indices = np.random.choice(indices, to_flip, replace=False)
            best_h[flip_indices] = 1
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
