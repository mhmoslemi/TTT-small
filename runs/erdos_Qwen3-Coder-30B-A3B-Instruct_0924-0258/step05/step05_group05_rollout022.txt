import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try relaxation-based initialization
    if n_points <= 40:
        try:
            # SDP relaxation approach
            h_relax = h_start.copy()
            
            # Create the SDP problem for relaxation
            X = cp.Variable((n_points, n_points), symmetric=True)
            h_var = cp.Variable(n_points)
            
            # Objective: max of correlations (we'll use the largest element in X)
            objective = cp.Maximize(cp.trace(X))
            
            # Constraints
            constraints = [
                cp.trace(X) == (n_points/2)**2 * dx,  # integral constraint
                X >> 0,  # positive semidefinite
                X[i,i] == h_var[i] for i in range(n_points),  # diagonal
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points/2 * dx,
            ]
            
            # Try SDP first to get a good starting point
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS, verbose=False, eps=1e-6)
            
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                # Extract the solution
                X_sol = X.value
                if X_sol is not None:
                    # Get the eigenvalues and eigenvectors to recover h approximation
                    evals, evecs = np.linalg.eigh(X_sol)
                    h_relax = np.abs(evecs[:, np.argmax(evals)])
                    h_relax = h_relax / np.sum(h_relax) * (n_points / 2)
                    h_relax = np.clip(h_relax, 0, 1)
            
            # Local refine with L-BFGS-B
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            constraints = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_relax, 
                           method='L-BFGS-B', 
                           bounds=bounds, 
                           constraints=constraints, 
                           options={'maxiter': 200, 'ftol': 1e-8})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
                    
        except Exception:
            pass
    
    # Local optimization with heavy refinement
    max_iterations = 300
    stagnation_counter = 0
    max_stagnation = 30
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Gradient-based optimization with L-BFGS-B
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        constraints = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, 
                           method='L-BFGS-B',
                           bounds=bounds, 
                           constraints=constraints,
                           options={'maxiter': 50, 'ftol': 1e-8})
            
            if res.success:
                h_refined = res.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    stagnation_counter = 0
                else:
                    stagnation_counter += 1
            else:
                stagnation_counter += 1
                
        except:
            stagnation_counter += 1
            
        if stagnation_counter > max_stagnation or best_c5 < 0.3807:
            break
        
        # Sometimes perturb to escape local minima
        if i % 50 == 0 and i > 0:
            h_perturb = best_h + np.random.normal(0, 0.01, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
            c5_perturb = evaluate_erdos_solution(h_perturb, dx)
            if c5_perturb < best_c5:
                best_h = h_perturb
                best_c5 = c5_perturb
    
    # Final refined optimization if needed
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 50:  # Ensure we have time for final refinement
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        constraints = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, 
                           method='L-BFGS-B',
                           bounds=bounds, 
                           constraints=constraints,
                           options={'maxiter': 100, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                h_final = np.clip(h_final, 0, 1)
                h_final = h_final * (n_points / 2) / np.sum(h_final)
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
