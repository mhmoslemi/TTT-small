import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.linspace(0, 1, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # DE parameters
    population_size = 30
    mutation_factor = 0.8
    crossover_prob = 0.8
    max_generations = 500
    
    # Initialize population
    population = []
    fitness = []
    
    # Start with initial solution and perturbed versions
    for i in range(population_size // 2):
        h = h_start + np.random.normal(0, 0.03, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Add random solutions
    for i in range(population_size // 2):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Differential evolution loop
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        if sorted_fit[0] < best_c5:
            best_h = sorted_pop[0].copy()
            best_c5 = sorted_fit[0]
            if best_c5 < 0.3807:
                break
        
        # Create new population
        new_pop = [sorted_pop[0], sorted_pop[1]]  # Elitism
        
        # DE mutation and crossover
        for i in range(population_size - 2):
            # Select three different individuals
            idxs = random.sample(range(len(sorted_pop)), 3)
            a, b, c = sorted_pop[idxs[0]], sorted_pop[idxs[1]], sorted_pop[idxs[2]]
            
            # Mutation
            mutant = a + mutation_factor * (b - c)
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            trial = np.where(np.random.random(n_points) < crossover_prob, mutant, sorted_pop[i])
            
            # Repair
            if np.sum(trial) == 0:
                trial = np.ones_like(trial) * 0.5
            else:
                trial = trial * (n_points/2) / np.sum(trial)
            
            new_pop.append(trial)
        
        population = new_pop
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Final local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Try multiple initial points for better convergence
            best_local = best_h.copy()
            best_local_c5 = best_c5
            
            for _ in range(3):
                h_init = best_h + np.random.normal(0, 0.01, n_points)
                h_init = np.clip(h_init, 0, 1)
                h_init = h_init * (n_points/2) / np.sum(h_init)
                
                res = minimize(objective, h_init, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-9})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_local_c5:
                        best_local = h_refined
                        best_local_c5 = c5_refined
            
            if best_local_c5 < best_c5:
                best_h = best_local
                best_c5 = best_local_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
