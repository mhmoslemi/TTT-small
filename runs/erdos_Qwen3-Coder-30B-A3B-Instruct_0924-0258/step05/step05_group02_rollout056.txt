import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for GA
    pop_size = 50
    elitism = 15
    mutation_rate = 0.1
    max_generations = 1000
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    if 'initial_h_values' in globals():
        population = [h_start]
        for _ in range(pop_size - 1):
            population.append(create_individual())
    else:
        population = [create_individual() for _ in range(pop_size)]
    
    generation = 0
    stagnation_count = 0
    prev_best = best_c5
    
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate fitness
        fitness_scores = [fitness(ind) for ind in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best
        current_best = fitness_scores[0]
        if current_best < best_c5:
            best_h = population[0].copy()
            best_c5 = current_best
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > 20:
            # Perturb population to avoid local minima
            for i in range(1, len(population)):
                if random.random() < 0.3:
                    population[i] = create_individual()
            stagnation_count = 0
            
        # Selection: keep top elitism
        elite = population[:elitism]
        
        # Create new generation
        new_population = elite[:]
        
        # Crossover and mutation for rest
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = elite[random.randint(0, len(elite)-1)]
            parent2 = elite[random.randint(0, len(elite)-1)]
            
            # Crossover: weighted average
            alpha = random.uniform(0.4, 0.6)
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if random.random() < mutation_rate:
                for i in range(n_points):
                    if random.random() < 0.1:
                        child[i] += random.gauss(0, 0.02)
            
            # Normalize and clip
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        generation += 1
    
    # Final refinement with optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    for _ in range(10):
        try:
            # Refine best solution
            res = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Final normalization
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
