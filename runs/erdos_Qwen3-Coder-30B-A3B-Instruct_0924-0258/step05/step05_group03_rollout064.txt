import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.full(n_points, 0.5)
    
    # Ensure constraints are satisfied
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    h = initial_h.copy()
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Gradient-based optimization with shift sampling
    n_shifts = 50
    max_iter = 500
    stagnation_limit = 50
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample shifts
        shifts = np.random.choice(n_points, size=n_shifts, replace=False)
        
        # Evaluate overlaps for all sampled shifts
        max_overlap = -1
        best_shift = 0
        for k in shifts:
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlap *= dx
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = k
        
        # Compute gradient for the worst shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_shift) % n_points
            grad[i] += (1 - h[j]) * dx
            grad[j] -= h[i] * dx
        
        # Line search
        alpha = 1.0
        step_size = 0.01
        while alpha > 1e-10:
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                stagnation_count = 0
                if best_c5 < 0.3807:
                    break
                break
            alpha *= 0.5
        
        # Restart if no improvement
        if best_c5 >= best_c5 - 1e-12:
            stagnation_count += 1
            if stagnation_count > stagnation_limit:
                # Add perturbation
                h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                stagnation_count = 0
        else:
            stagnation_count = 0
    
    # Final refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            h = res.x
            best_c5 = obj(h)
    except:
        pass
    
    return (list(h), best_c5, n_points)
