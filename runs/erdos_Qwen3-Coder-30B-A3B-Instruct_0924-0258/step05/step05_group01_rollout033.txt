import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create triangular function as starting point
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start / np.sum(h_start) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution refinement with Simulated Annealing
    scales = [35, 50, 75, 100]
    
    for scale in scales:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to target scale if needed
        if scale > n_points:
            h_current = np.interp(np.linspace(0, 2, scale), np.linspace(0, 2, n_points), best_h)
            n_points = scale
            dx = 2.0 / n_points
        else:
            h_current = best_h.copy()
            n_points = scale
            dx = 2.0 / n_points
            
        # Simulated Annealing
        T = 0.1
        alpha = 0.99
        max_iter = int(1000 * (scale / 35))
        iter_count = 0
        
        while iter_count < max_iter and time() - start_time < budget_s * 0.95:
            # Generate neighbor by perturbing some elements
            h_new = h_current.copy()
            num_changes = max(1, n_points // 10)
            indices = np.random.choice(n_points, num_changes, replace=False)
            for i in indices:
                # Perturb by adding small random value
                delta = np.random.normal(0, 0.02)
                h_new[i] += delta
                
            # Apply constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            c5_old = evaluate_erdos_solution(h_current, dx)
            
            # Accept or reject
            if c5_new < c5_old or np.random.rand() < np.exp(-(c5_new - c5_old) / T):
                h_current = h_new
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
            T *= alpha
            iter_count += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
