import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM Parameters
    max_iter = 5000
    mu = 1e-3
    tol = 1e-6
    rho = 1.0
    
    # Initialize variables
    h = h_start.copy()
    h_prev = h.copy()
    z = np.zeros(2 * n_points - 1)
    lambda_vec = np.zeros(2 * n_points - 1)
    c5_approx = best_c5
    
    # For convergence tracking
    primal_residuals = []
    dual_residuals = []
    
    def soft_threshold(x, threshold):
        return np.sign(x) * np.maximum(np.abs(x) - threshold, 0)
    
    # Precompute shifts
    shifts = []
    for k in range(1 - n_points, n_points):
        shift = np.zeros(2 * n_points - 1)
        if k >= 0:
            shift[k:k+n_points] = 1
        else:
            shift[-k: -k+n_points] = 1
        shifts.append(shift)
    
    shifts = np.array(shifts)
    
    # ADMM iterations
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Update h: minimize ||h - h_prev||^2 + mu * ||z - (h * (1-h))||^2
        # subject to 0 <= h <= 1, sum(h) = n_points / 2
        
        # Simple quadratic approximation: h = argmin 0.5 * ||h - b||^2 + mu * ||z - (h * (1-h))||^2
        # Use a kind of fixed-point iteration or quadratic solver
        def obj(h):
            return np.sum((h - h_prev)**2) + mu * np.sum((z - h * (1 - h))**2)
        
        def constraint(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_new = res.x
            else:
                h_new = h
        except:
            h_new = h
        
        # Bound h
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize sum
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update z using soft-thresholding
        # z = argmin ||z - (h_new * (1 - h_new) + lambda_vec/mu)||^2 + mu * ||z - z_prev||^2
        # z_k = (x_k + lambda_k/mu) / (1 + mu) 
        x = h_new * (1 - h_new) + lambda_vec / mu
        z = soft_threshold(x, 1.0 / (1.0 + mu))
        
        # Update dual variables
        lambda_vec += mu * (h_new * (1 - h_new) - z)
        
        # Update approximate C5 value from current overlaps
        c5_approx = evaluate_erdos_solution(h_new, dx)
        
        # Update h_prev
        h = h_new
        
        # Convergence checking
        if iteration % 50 == 0:
            # Compute primal and dual residuals
            primal_res = np.linalg.norm(h - h_prev) / np.linalg.norm(h_prev + 1e-12)

                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h.copy(), method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass

    return (list(best_h), best_c5, n_points)
