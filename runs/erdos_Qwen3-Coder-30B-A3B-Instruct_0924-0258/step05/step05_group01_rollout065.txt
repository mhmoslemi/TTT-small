import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Convert to binary
    h_binary = np.round(h_start).astype(int)
    target_sum = n_points // 2
    
    # Adjust to ensure sum equals target_sum
    current_sum = np.sum(h_binary)
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        indices_1 = np.where(h_binary == 1)[0]
        flip_count = current_sum - target_sum
        flip_indices = np.random.choice(indices_1, flip_count, replace=False)
        h_binary[flip_indices] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        indices_0 = np.where(h_binary == 0)[0]
        flip_count = target_sum - current_sum
        flip_indices = np.random.choice(indices_0, flip_count, replace=False)
        h_binary[flip_indices] = 1
    
    best_h = h_binary.copy()
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Tabu search parameters
    tabu_tenure = 10
    tabu_list = []
    max_no_improvement = 100
    no_improvement_count = 0
    shaking_frequency = 50
    
    # Current solution
    current_h = best_h.copy()
    current_c5 = best_c5
    
    iteration = 0
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Generate neighborhood by flipping one 1 to 0 and one 0 to 1
        # Find 1s (to flip to 0)
        ones_indices = np.where(current_h == 1)[0]
        zeros_indices = np.where(current_h == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            break
            
        # Pick one 1 to flip to 0 and one 0 to flip to 1
        i1 = np.random.choice(ones_indices)
        i0 = np.random.choice(zeros_indices)
        
        # Generate neighbor
        neighbor_h = current_h.copy()
        neighbor_h[i1] = 0
        neighbor_h[i0] = 1
        
        # Check tabu
        move = (i1, i0)
        if move in tabu_list:
            # Check aspiration - if better than best, accept it
            aspiration = False
            if current_c5 > best_c5:
                aspiration = True
                
            if not aspiration:
                # Skip this move
                continue
        
        # Evaluate neighbor
        neighbor_c5 = evaluate_erdos_solution(neighbor_h, dx)
        
        # Update tabu list
        tabu_list.append(move)
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
        
        # Accept neighbor if it's better or 
        if neighbor_c5 < current_c5:
            current_h = neighbor_h
            current_c5 = neighbor_c5
            no_improvement_count = 0
            
            # Update best
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            no_improvement_count += 1
            
        # Shaking mechanism
        if no_improvement_count >= shaking_frequency:
            # Random restart - slightly perturb current solution by flipping more bits
            current_h = best_h.copy()
            current_c5 = best_c5
            
            # Flip about 5% of the bits to create a shake
            n_flip = max(1, n_points // 20)
            ones_indices = np.where(current_h == 1)[0]
            zeros_indices = np.where(current_h == 0)[0]
            
            if len(ones_indices) > 0 and len(zeros_indices) > 0:
                flip1 = np.random.choice(ones_indices, min(n_flip, len(ones_indices)), replace=False)
                flip0 = np.random.choice(zeros_indices, min(n_flip, len(zeros_indices)), replace=False)
                
                for i in flip1:
                    current_h[i] = 0
                for i in flip0:
                    current_h[i] = 1
                    
            no_improvement_count = 0
            
        # Early termination on no improvement
        if no_improvement_count >= max_no_improvement:
            break
            
    # Validate and return
    if not np.all((best_h == 0) | (best_h == 1)):
        raise ValueError("Solution not binary")
    if np.sum(best_h) != n_points // 2:
        raise ValueError("Sum constraint violated")
        
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
