import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h from initial_h_values if available

        n_points = len(h)

    # Normalize h to ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Parameters
    max_iter = 1000
    stagnation_limit = 50
    shift_samples = 100
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    # Iterative refinement
    stagnation_count = 0
    last_best = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample shifts to find maximum overlap
        shift_range = np.linspace(0, 2, shift_samples)
        max_overlap_val = -1
        best_shift = 0
        
        for shift in shift_range:
            # Compute overlap for this shift
            shift_int = int(shift * n_points / 2)
            if shift_int >= n_points:
                continue
            # For positive shifts
            if shift_int > 0:
                overlap = np.sum(h[:-shift_int] * (1 - h[shift_int:]))
            else:
                overlap = np.sum(h * (1 - h))
            
            if overlap > max_overlap_val:
                max_overlap_val = overlap
                best_shift = shift
                
        # Use the best shift for constraint
        shift_int = int(best_shift * n_points / 2)
        
        # Optimization using CVXPY for convex relaxation
        # Variable h
        h_var = cp.Variable(n_points)
        
        # Objective: minimize max overlap
        # For linearization, we try to bound the maximum inner product
        # Simply use the computed shift to add constraint
        # But actually, we want to minimize the maximum overlap across all possible k
        # This is complex to model directly;
        # Let's try a simpler approach: iterate between refining shift and refining h
        
        # Relax the problem
        # We solve a convex problem to minimize the overlap for the current shift
        try:
            # Make a line of constraints for that max overlap
            # Just use a reasonable constraint
            constraints = [cp.sum(h_var) == n_points/2, h_var >= 0, h_var <= 1]
            
            # In practice, we optimize h to try to reduce max overlap 
            # This iterative strategy still focuses on the structure
            
            # We approximate using an SDP relaxation or solve the convex subproblem
            # For now, directly improving h using gradient-based method
            def obj_fn(x):
                return evaluate_erdos_solution(x, 2.0 / n_points)
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
            
            # Local optimization on h
            res = minimize(obj_fn, h.copy(), method='SLSQP', 
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 50, 'ftol': 1e-8})
            
            if res.success:
                h_new = res.x
                c5_new = obj_fn(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                h = h_new

                h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                stagnation_count += 1
                
        except Exception:
            # Fallback to simple perturbation
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count += 1
            
        # Early termination if we've reached our target
        if best_c5 < 0.3807:

        # Stagnation check
        if stagnation_count > stagnation_limit:

            stagnation_count = 0

            stagnation_count += 1
        else:

            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-10})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
            
    return (list(best_h), best_c5, n_points)
