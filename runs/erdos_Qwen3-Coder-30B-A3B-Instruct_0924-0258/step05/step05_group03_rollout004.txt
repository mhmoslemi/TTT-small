import numpy as np
from cvxpy import *
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 35
        h0 = np.ones(n_points) * 0.5
    
    # Set up variables
    h = Variable(n_points)
    C5 = Variable()
    
    # Constraints 
    constraints = [h >= 0, h <= 1]
    constraints.append(sum(h) == n_points / 2)
    
    # Sampling of shifts
    shifts = np.linspace(0, 2, 100)
    
    # Cutting plane approach
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Precompute correlation for initial solution
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h0, dx)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h0.copy()

    # Main cutting plane loop
    max_iters = 200
    for iter_num in range(max_iters):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate correlations at current h
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx] * dx
        
        # Add constraint from the maximum overlap
        constraints.append(C5 >= max_corr)
        
        # Re-optimize with new constraint
        objective = Minimize(C5)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                new_c5 = C5.value
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = np.array(h.value)
                    
                # Update constraints to avoid returning to same solution
                constraints.remove(constraints[-1])
            else:
                break
        except:
            break
            
        if best_c5 < 0.3807:
            break
            
    # Final local refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
        
    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
