import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
        initial_h[0] = 0.4
        initial_h[1] = 0.6
    
    # Ensure the integral constraint
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Define shifts for constraint generation
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    dx = 2.0 / n_points
    
    # Initialize variables for CVXPY
    h_vars = cp.Variable(n_points)
    c5_var = cp.Variable()
    
    # CVXPY problem setup
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points / 2,
        c5_var >= 0
    ]
    
    # Remaining variables and constraints are updated in the cutting-plane loop
    prob = cp.Problem(cp.Minimize(c5_var), constraints)
    
    # Cutting plane optimization loop
    best_c5 = float('inf')
    best_h = initial_h.copy()
    max_iterations = 50
    iteration = 0
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Solve current relaxation
        try:
            prob.solve(solver=cp.ECOS, feastol=1e-6, abstol=1e-6, reltol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_vars.value
                current_c5 = c5_var.value
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_sol.copy()
        except:
            pass
        
        # Compute overlap for all shifts
        overlaps = []
        for shift in shifts:
            shift_idx = int(shift / dx)
            if shift_idx < n_points:
                corr = np.correlate(h_sol, 1 - np.roll(h_sol, shift_idx), mode='valid')
                # Note: This is approximate due to discrete nature
                overlaps.append(np.max(corr) * dx)
            else:
                overlaps.append(0)
        
        # Add the constraint from maximum overlap shift
        max_overlap_idx = np.argmax(overlaps)
        max_shift = shifts[max_overlap_idx]
        max_overlap_val = overlaps[max_overlap_idx]
        
        # Add this constraint to problem
        shift_idx = int(max_shift / dx)
        if shift_idx < n_points:
            constraint_expr = cp.sum(cp.multiply(
                h_vars[:n_points-shift_idx], 
                1 - h_vars[shift_idx:]
            )) * dx <= c5_var
            prob.constraints.append(constraint_expr)
        
        # Early termination if target achieved
        if best_c5 < 0.3807:
            break
            
        iteration += 1
    
    # Local refinement with SLSQP
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
