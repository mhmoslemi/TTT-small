import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        # Start with a coarse grid of 20 points
        n_points = 20
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Refinement loop
    max_iterations = 100
    iteration = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Interpolate to 35 points if needed
        if n_points < 35:
            h_fine = np.interp(np.linspace(0, 2, 35), np.linspace(0, 2, n_points), best_h)
            h_fine = h_fine * (35/2) / np.sum(h_fine)
            h_fine = np.clip(h_fine, 0, 1)
            h_current = h_fine
        else:
            h_current = best_h.copy()
        
        # Compute correlations to find the constraint with max overlap
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        k_offset = max_k - (len(h_current) - 1)
        
        # Constraint for max overlap (add small epsilon to enforce strict reduction)
        constraint_max = max_corr * dx - 1e-6
        
        # Set up optimization problem
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - 17.5  # normalized sum
        
        def constraint_overlap(h):
            # Get correlation at the problematic shift
            corr_current = np.correlate(h, 1 - h, mode='full')
            return constraint_max - corr_current[max_k] * dx
        
        # Bounds and constraints
        bounds = [(0, 1) for _ in range(len(h_current))]
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Try to solve with constraint for max overlap
        try:
            res = minimize(
                objective,
                h_current,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-7}
            )
            if res.success:
                h_opt = res.x
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
        except Exception:
            no_improvement_count += 1

        # Perturb solution if no improvement for too many iterations
        if no_improvement_count > 50:
            h_perturbed = best_h + np.random.normal(0, 0.01, len(best_h))
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (17.5 / np.sum(h_perturbed))

                res = minimize(

    return (list(final_h), final_c5, len(final_h))
