import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize initial h to satisfy constraint
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Initialize best solution
    h = initial_h.copy()
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Parameters
    max_iter = 500
    stagnation_limit = 50
    shift_samples = 50  # Number of shifts to sample for gradient computation
    
    # Main optimization loop
    iter_count = 0
    stagnation_count = 0
    last_best = best_c5
    
    while iter_count < max_iter and time() - start_time < budget_s * 0.95:
        # Sample shifts
        sampled_shifts = np.random.choice(n_points, shift_samples, replace=False)
        
        # Compute overlaps and find the worst shift
        max_overlap = -1
        worst_k = 0
        for k in sampled_shifts:
            # Compute overlap for shift k
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlap *= dx
            
            if overlap > max_overlap:
                max_overlap = overlap
                worst_k = k
        
        # Compute gradient for the worst shift
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + worst_k) % n_points
            gradient[i] += (1 - h[j]) * dx
            gradient[j] -= h[i] * dx
        
        # Line search for step size
        alpha = 1.0
        step_size = 0.01
        max_steps = 100
        
        # Backtracking line search
        for _ in range(max_steps):
            h_trial = h - alpha * gradient
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            trial_c5 = evaluate_erdos_solution(h_trial, dx)
            
            if trial_c5 < best_c5:
                h = h_trial
                best_c5 = trial_c5
                stagnation_count = 0
                last_best = best_c5
                break
            else:
                alpha *= 0.5
        
        # If no improvement, use small random perturbation
        if stagnation_count > 0 and stagnation_count < 10:
            # Add small random noise
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        if best_c5 < last_best:
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        iter_count += 1
        
        # Early termination if target achieved
        if best_c5 < 0.3807:
            break
    
    # Final refinement with local optimization
    def obj(x):
        return evaluate_erdos_solution(x, dx)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
