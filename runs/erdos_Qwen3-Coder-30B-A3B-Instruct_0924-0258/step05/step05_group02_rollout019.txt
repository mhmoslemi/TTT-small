import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Cooperative coarse-to-fine refinement
    n_low = 20
    low_indices = np.linspace(0, n_points-1, n_low, dtype=int)
    
    # Keep track of convergence
    no_improvement_count = 0
    max_no_improvement = 100
    
    def upsample_h(h_low):
        # Linearly interpolate to higher resolution
        h_upsampled = np.interp(np.linspace(0, n_points-1, n_points), 
                               low_indices, h_low)
        # Normalize to sum to n_points/2
        h_upsampled = h_upsampled * (n_points/2) / np.sum(h_upsampled)
        return np.clip(h_upsampled, 0, 1)
    
    # Coarse grid approach with CVXPY convex relaxation
    def solve_coarse_grid():
        h_low = cp.Variable(n_low)
        constraints = [
            cp.sum(h_low) == n_low / 2,
            h_low >= 0,
            h_low <= 1
        ]
        # Add a limited set of constraints based on shift indices
        for k in range(1, min(10, n_low)):
            if k < n_low // 2:
                corr_sum = cp.sum(cp.multiply(h_low[:n_low-k], 1 - h_low[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_low[k:], 1 - h_low[:n_low-k]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_low)), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                return h_low.value
        except:
            pass
        return None
    
    # Adaptive refinement loop
    while time() - start_time < budget_s * 0.95:
        # Step 1: Coarse grid optimization
        h_coarse = solve_coarse_grid()
        if h_coarse is not None:
            h_upsampled = upsample_h(h_coarse)
            h_refined = h_upsampled
        else:
            h_refined = best_h.copy()
        
        # Step 2: Fine grid refinement with SLSQP and gradient-based adjustments
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try to improve using gradient descent with adaptive mutations
        h_current = h_refined.copy()
        max_iter = 100
        for _ in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Evaluate current value and gradient estimate
            try:
                res = minimize(
                    lambda h: objective(h),
                    h_current,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons_sum,
                    options={'maxiter': 10, 'ftol': 1e-6}
                )
                if res.success:
                    h_new = res.x
                    c5_new = objective(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new
                        no_improvement_count = 0
                    else:
                        no_improvement_count += 1
                    h_current = h_new
                else:
                    no_improvement_count += 1
            except Exception:
                no_improvement_count += 1
            
            # Mutation to escape local optima
            if no_improvement_count > 50 and time() - start_time < budget_s * 0.9:
                # Apply small random perturbations
                pert = np.random.normal(0, 0.01, n_points)
                h_mutated = h_current + pert
                h_mutated = np.clip(h_mutated, 0, 1)
                h_mutated = h_mutated * (n_points/2) / np.sum(h_mutated)
                c5_mutated = objective(h_mutated)
                if c5_mutated < best_c5:
                    best_c5 = c5_mutated
                    best_h = h_mutated
                    no_improvement_count = 0
                h_current = h_mutated
        
        if no_improvement_count > max_no_improvement:
            break
    
    # Final validation and refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        final_c5 = evaluate_erdos_solution(h_final, dx)
        if final_c5 < best_c5:
            best_h = h_final
            best_c5 = final_c5
    except Exception:
        pass
    
    # Return the best solution found
    return (list(best_h), best_c5, n_points)
