import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a triangular function as starting point
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start / np.sum(h_start) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution SA refinement
    scales = [35, 50, 75, 100]
    
    for scale in scales:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to target scale if needed
        if scale > n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            h_interp = np.interp(x_new, x_orig, best_h)
            h_current = h_interp
            n_points = scale
        else:
            h_current = best_h
            n_points = scale
            
        dx = 2.0 / n_points
        
        # Simulated Annealing parameters
        T = 0.1
        alpha = 0.99
        max_iter = int(1000 * (scale / 35))  # Scale with resolution
        
        h_current = h_current.copy()
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate neighbor by perturbing some elements
            h_new = h_current.copy()
            indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            for idx in indices:
                # Perturb by small random amount
                delta = np.random.normal(0, 0.05)
                h_new[idx] += delta
            # Clip to bounds
            h_new = np.clip(h_new, 0, 1)
            # Rescale to maintain sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            # Metropolis criterion
            if new_c5 < current_c5 or np.random.rand() < math.exp(-(new_c5 - current_c5) / T):
                h_current = h_new
                current_c5 = new_c5
                
            # Cool down
            T = alpha * T
            
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
                
        # Final check
        final_c5 = evaluate_erdos_solution(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
        
    return (list(best_h), best_c5, n_points)
