import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution convex optimization approach
    resolutions = [35, 40, 50, 60]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Try to improve using convex relaxation
        try:
            # Create variables
            h_opt = cp.Variable(n_points)
            
            # Constraints
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add overlap constraints for significant shifts
            max_shift = min(n_points//2, 20)  # Limit shifts to avoid computational cost
            for k in range(1, max_shift):
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= (best_c5 + 1e-6) * dx)
            
            # Solve the convex relaxation
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
                    h_current = h_feasible
        except Exception:
            pass
        
        # Local refinement using SLSQP
        max_restarts = 10
        restart_count = 0
        h_refined = h_current.copy()
        
        while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                # Add small random perturbation
                h_perturbed = h_refined + np.random.normal(0, 0.005, n_points)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
                
                res = minimize(objective, h_perturbed, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-8})
                
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        restart_count = 0  # Reset restart counter
                    else:
                        restart_count += 1
                else:
                    restart_count += 1
            
            except Exception:
                restart_count += 1
        
        h_start = best_h.copy()
    
    return (list(best_h), best_c5, n_points)
