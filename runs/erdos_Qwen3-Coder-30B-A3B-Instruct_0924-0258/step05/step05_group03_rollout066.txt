import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Maximum iterations to prevent infinite loops
    max_iter = 1000
    iteration = 0
    improvement_threshold = 1e-6
    
    # CCP loop
    while time() - start_time < budget_s * 0.95 and iteration < max_iter:
        # Identify worst-case shift
        shifts = np.linspace(0, 2, 100, endpoint=False)
        worst_shift = 0
        max_correlation = -1
        
        for shift in shifts:
            # Compute correlation at this shift
            k = int(shift / dx)
            # Shift h to the right by k positions
            shifted_h = np.roll(h, k)
            correlation = np.sum(h * (1 - shifted_h)) * dx
            if correlation > max_correlation:
                max_correlation = correlation
                worst_shift = shift

        # We want to minimize sum(h_new[i] * (1 - h[i + k*]))

        # minimize sum(h_new[i] * (1 - h[i + k])) for fixed k
        # subject to 0 <= h_new[i] <= 1 and sum(h_new) = 1
        
        k_offset = int(worst_shift / dx)

        def obj(h_new):
            shifted = np.roll(h_new, k_offset)
