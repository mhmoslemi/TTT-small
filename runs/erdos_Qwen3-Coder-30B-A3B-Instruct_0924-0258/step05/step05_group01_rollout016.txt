import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        # Convert to binary
        h_start = (h_start > 0.5).astype(int)
        # Adjust to exact sum constraint
        current_sum = np.sum(h_start)
        target_sum = n_points // 2
        if current_sum > target_sum:
            # Need to flip some 1s to 0s
            indices_1 = np.where(h_start == 1)[0]
            flip_indices = np.random.choice(indices_1, size=current_sum - target_sum, replace=False)
            h_start[flip_indices] = 0
        elif current_sum < target_sum:
            # Need to flip some 0s to 1s
            indices_0 = np.where(h_start == 0)[0]
            flip_indices = np.random.choice(indices_0, size=target_sum - current_sum, replace=False)
            h_start[flip_indices] = 1
    else:
        n_points = 35
        # Initialize with exactly half ones
        h_start = np.zeros(n_points, dtype=int)
        h_start[:n_points//2] = 1
        np.random.shuffle(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_tenure = 10
    tabu_list = []
    no_improvement_count = 0
    max_no_improvement = 100
    shaking_iterations = 50
    
    # Keep track of the best solution found
    best_h_global = best_h.copy()
    best_c5_global = best_c5
    
    # Begin tabu search
    while time() - start_time < budget_s * 0.95:
        # Generate neighbors
        current_sum = np.sum(best_h)
        neighbors = []
        
        # Generate all neighbors by flipping one 1 to 0 and one 0 to 1
        indices_1 = np.where(best_h == 1)[0]
        indices_0 = np.where(best_h == 0)[0]
        
        if len(indices_1) > 0 and len(indices_0) > 0:
            for i in indices_1:
                for j in indices_0:
                    neighbor = best_h.copy()
                    neighbor[i] = 0
                    neighbor[j] = 1
                    neighbors.append((neighbor, (i, j)))
        
        if not neighbors:
            break
            
        # Evaluate neighbors
        neighbor_evals = []
        for neighbor, move in neighbors:
            c5 = evaluate_erdos_solution(neighbor, dx)
            neighbor_evals.append((neighbor, c5, move))
            
        # Find best neighbor
        best_neighbor, best_c5_neighbor, best_move = min(neighbor_evals, key=lambda x: x[1])
        
        # Check if it's better than current best
        if best_c5_neighbor < best_c5:
            best_h = best_neighbor
            best_c5 = best_c5_neighbor
            # Update global best
            if best_c5 < best_c5_global:
                best_h_global = best_h.copy()
                best_c5_global = best_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update tabu list
        if best_move in tabu_list:
            # Allow aspiration if it's a significant improvement
            if best_c5_neighbor < best_c5_global * 0.99:
                # Still add to tabu list as it's a recent move
                pass
        else:
            tabu_list.append(best_move)
            
        # Remove oldest items if tabu list is too long
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
            
        # Shaking mechanism
        if no_improvement_count >= shaking_iterations:
            # Random restart - flip a few bits
            flip_count = max(1, n_points // 20)
            indices_to_flip = np.random.choice(n_points, size=flip_count, replace=False)
            new_h = best_h.copy()
            for i in indices_to_flip:
                new_h[i] = 1 - new_h[i]  # Flip bit
            # Maintain sum constraint
            current_sum = np.sum(new_h)
            target_sum = n_points // 2
            if current_sum > target_sum:
                indices_1 = np.where(new_h == 1)[0]
                flip_indices = np.random.choice(indices_1, size=current_sum - target_sum, replace=False)
                new_h[flip_indices] = 0
            elif current_sum < target_sum:
                indices_0 = np.where(new_h == 0)[0]
                flip_indices = np.random.choice(indices_0, size=target_sum - current_sum, replace=False)
                new_h[flip_indices] = 1
            best_h = new_h
            best_c5 = evaluate_erdos_solution(best_h, dx)
            no_improvement_count = 0
            tabu_list.clear()
            
        if no_improvement_count >= max_no_improvement:
            break
            
    # Final adjustment to ensure feasibility
    try:
        current_sum = np.sum(best_h_global)
        target_sum = n_points // 2
        if current_sum > target_sum:
            indices_1 = np.where(best_h_global == 1)[0]
            flip_indices = np.random.choice(indices_1, size=current_sum - target_sum, replace=False)
            best_h_global[flip_indices] = 0
        elif current_sum < target_sum:
            indices_0 = np.where(best_h_global == 0)[0]
            flip_indices = np.random.choice(indices_0, size=target_sum - current_sum, replace=False)
            best_h_global[flip_indices] = 1
        best_c5_global = evaluate_erdos_solution(best_h_global, dx)
    except:
        pass
        
    return (list(best_h_global), best_c5_global, n_points)
