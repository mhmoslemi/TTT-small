import numpy as np
from cvxpy import *
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Get initial solution
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Sample shifts for overlap constraints
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    
    # Initialize variables
    h = Variable(n_points)
    C5 = Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2,
        C5 >= 0
    ]
    
    # Add initial constraints
    for shift in shifts:
        k = int(shift * n_points / 2)
        if k < n_points:
            # Compute correlation for this shift
            corr_val = 0.0
            for i in range(n_points - k):
                corr_val += h[i] * (1 - h[i + k])
            constraints.append(corr_val <= C5)
    
    # Objective
    objective = Minimize(C5)
    
    # Preliminary solve to get a baseline
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False, max_iters=100)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            best_h = np.array(h.value).flatten()
            best_c5 = C5.value
        else:
            best_h = initial_h.copy()
            best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    except:
        best_h = initial_h.copy()
        best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    # Iterative cutting plane approach
    max_iterations = 50
    iteration = 0
    last_c5 = best_c5 + 1
    patience = 0
    max_patience = 10
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Check improvement
        if abs(last_c5 - best_c5) < 1e-5:
            patience += 1
        else:
            patience = 0
        if patience >= max_patience:
            break
        last_c5 = best_c5
        
        # Compute current overlaps
        corr_vals = []
        for shift in shifts:
            k = int(shift * n_points / 2)
            if k < n_points:
                corr_val = 0.0
                for i in range(n_points - k):
                    corr_val += best_h[i] * (1 - best_h[i + k])
                corr_vals.append((shift, corr_val))
        
        # Find worst overlap
        worst_shift, cur_max_overlap = max(corr_vals, key=lambda x: x[1])
        
        # Add constraint for worst overlap
        k = int(worst_shift * n_points / 2)
        if k < n_points:
            corr_constraint = 0.0
            for i in range(n_points - k):
                corr_constraint += h[i] * (1 - h[i + k])
            constraints.append(corr_constraint <= C5)
        
        # Solve with added constraint
        try:
            prob = Problem(objective, constraints)
            prob.solve(solver=ECOS, verbose=False, max_iters=100)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                new_h = np.array(h.value).flatten()
                new_c5 = C5.value
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
            else:
                # Fallback if optimization fails
                break
        except:
            break
        
        iteration += 1
    
    # Local refinement using SLSQP
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    return (list(best_h), final_c5, n_points)
