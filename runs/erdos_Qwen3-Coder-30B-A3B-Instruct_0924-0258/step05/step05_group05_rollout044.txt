import numpy as np
from scipy.signal import correlate
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    # Use blocks of size sqrt(n_points) for simplicity
    M = int(np.sqrt(n_points))
    if M < 2:
        M = 2
    block_size = n_points // M
    
    # Ensure block size divides n_points evenly
    if n_points % M != 0:
        M = n_points // block_size
        if n_points % M != 0:
            M = n_points // (block_size + 1)
            if n_points % M != 0:

    if M < 2:
        M = 2
    block_size = n_points // M
    
    # Initialize block configuration (0s and 1s)
    h = np.zeros(n_points)
    on_blocks = (n_points / 2) // block_size  # Number of blocks to turn on
    
    # Determine positions of on_blocks blocks
    on_block_indices = sorted(random.sample(range(M), int(on_blocks)))
    
    # Flip the blocks to 1
    for i in on_block_indices:
        start = i * block_size
        end = start + block_size
        h[start:end] = 1
    
    # Normalize to sum to n_points/2
    h = h * (n_points/2) / np.sum(h)
    
    # Main optimization loop
    max_iterations = 500
    stagnation_counter = 0
    last_improvement = 0
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:

        current_c5 = evaluate_erdos_solution(h, dx)
