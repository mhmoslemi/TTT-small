from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

        h_sdp = cp.Variable(n_points, name="h_sdp")
        X = cp.Variable((n_points, n_points), name="X")

        constraints = [
            cp.trace(X) == n_points / 2,  # trace constraint
            X >> 0,  # positive semidefinite
        ]

        for i in range(n_points):
            for j in range(n_points):
                constraints.append(X[i, j] <= min(h_sdp[i], h_sdp[j]))

        # We'll use a simpler approach here instead of complex SDP

        def objective(h):
            return evaluate_erdos_solution(h, dx)
