import numpy as np
from scipy.signal import correlate
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Block-based optimization setup
    # Try to find a good block structure
    M = int(np.sqrt(n_points))  # Number of blocks
    if n_points % M != 0:
        # Find a good divisor
        for i in range(int(np.sqrt(n_points)), 0, -1):
            if n_points % i == 0:
                M = i
                break
    block_size = n_points // M
    
    if block_size * M != n_points:
        # Adjust M to get integer block size
        M = n_points // block_size
        block_size = n_points // M
    
    # Initialize with binary blocks
    h = np.zeros(n_points)
    on_blocks = n_points // (2 * block_size)  # Should be around half the blocks active
    
    # Create a simple structure first
    h[:on_blocks * block_size] = 1
    h = np.clip(h, 0, 1)
    h = h * (n_points/2) / np.sum(h)
    
    # Simple iterative improvement
    max_iter = 1000
    tolerance = 1e-6
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Store old
        h_old = h.copy()
        old_c5 = evaluate_erdos_solution(h, dx)
        
        # Try random perturbations to avoid local minima
        if iteration % 30 == 0 and iteration > 0:
            # Randomly change one block
            block_idx = random.randrange(M)
            if block_idx * block_size < n_points:
                # Toggle the block if can toggle
                start_idx = block_idx * block_size
                end_idx = start_idx + block_size
                current_val = np.mean(h[start_idx:end_idx])
                if current_val > 0.5:
                    h[start_idx:end_idx] = 0
                else:
                    h[start_idx:end_idx] = 1
                
                # Ensure sum constraint
                h = h * (n_points/2) / np.sum(h)
        
        # Local search with small perturbations
        if iteration % 5 == 0:
            # Try to slightly adjust values within block bounds
            h_new = h.copy()
            perturbed = False
            for block_idx in range(M):
                start_idx = block_idx * block_size
                end_idx = start_idx + block_size
                if block_idx < M - 1:
                    # Swap neighbors with small probability
                    if random.random() < 0.1:
                        # Swap some values within the block
                        swap_idx = random.randrange(block_size)
                        h_new[start_idx + swap_idx] = 1 - h_new[start_idx + swap_idx]
                        perturbed = True
            
            if perturbed:
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                new_c5 = evaluate_erdos_solution(h_new, dx)
                if new_c5 < old_c5:
                    h = h_new
                    old_c5 = new_c5
        
        # Optimization search in the valid domain
        if iteration % 20 == 0:
            # Random restart with different block structure
            new_h = np.zeros(n_points)
            skip_pattern = random.random() > 0.5
            on_count = 0
            for block_idx in range(M):
                if not skip_pattern or (block_idx % 2 == 0):
                    if on_count < n_points // (2 * block_size):
                        new_h[block_idx * block_size:(block_idx + 1) * block_size] = 1
                        on_count += 1
                
            # Apply for a few more iterations if helpful
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points/2) / np.sum(new_h)
            new_c5 = evaluate_erdos_solution(new_h, dx)
            if new_c5 < old_c5:
                h = new_h
                old_c5 = new_c5
        
        if old_c5 < best_c5:
            best_h = h.copy()
            best_c5 = old_c5
        
        # Early stopping condition
        if best_c5 < 0.3807:
            break
            
        # Progress checking
        if iteration > 0 and iteration % 50 == 0:
            if abs(old_c5 - evaluate_erdos_solution(h_old, dx)) < tolerance:
                # Possibly stuck; try to diversify
                for i in range(n_points):
                    if random.random() < 0.05:
                        h[i] = max(0, min(1, h[i] + random.uniform(-0.1, 0.1)))
                h = h * (n_points/2) / np.sum(h)
    
    # Final optimization using local descent
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        h_final = best_h.copy()
        c5_final = best_c5
        
        # Gradual refinement
        for _ in range(50):
            # Attempt small perturbations
            h_test = h_final.copy()
            for i in range(n_points):
                if random.random() < 0.1:
                    h_test[i] = max(0, min(1, h_test[i] + random.uniform(-0.02, 0.02)))
            
            # Adjust weights to maintain constraint
            h_test = h_test * (n_points/2) / np.sum(h_test)
            
            c5_test = evaluate_erdos_solution(h_test, dx)
            if c5_test < c5_final:
                h_final = h_test
                c5_final = c5_test
                
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
