import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Ensure initial solution is feasible
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Main optimization loop
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    stagnation_count = 0
    max_stagnation = 50
    iteration = 0
    
    # Gradient-based optimization with sampling
    while time() - start_time < budget_s * 0.95 and stagnation_count < max_stagnation:
        # Sample integer shifts for gradient estimation
        num_shifts = min(50, n_points)
        shifts = np.random.choice(n_points, num_shifts, replace=False)
        
        max_overlap = -1
        best_shift = 0
        
        # Find shift with maximum overlap
        for k in shifts:
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += best_h[i] * (1 - best_h[j])
            overlap *= (2.0 / n_points)
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = k
        
        # Compute gradient for the worst shift
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_shift) % n_points
            gradient[i] += (1 - best_h[j])
            gradient[j] -= best_h[i]
        
        gradient *= (2.0 / n_points)
        
        # Adaptive step size line search
        alpha = 1.0
        max_alpha = 10.0
        min_alpha = 1e-6
        step_size_reduction = 0.5
        
        while alpha > min_alpha:
            # Try update
            trial_h = best_h - alpha * gradient
            trial_h = np.clip(trial_h, 0, 1)
            trial_h = trial_h * (n_points / 2) / np.sum(trial_h)
            
            trial_c5 = evaluate_erdos_solution(trial_h, 2.0 / n_points)
            
            if trial_c5 < best_c5:
                best_h = trial_h
                best_c5 = trial_c5
                stagnation_count = 0
                break
            else:
                alpha *= step_size_reduction
                if alpha < min_alpha:
                    # If no improvement with small steps, reduce gradients
                    gradient *= 0.1
                    alpha = 1.0
                    break
        
        if alpha <= min_alpha:
            stagnation_count += 1
        
        iteration += 1
        
        # Early termination
        if best_c5 < 0.3807:
            break
        
        # Occasionally restart with perturbed solution
        if iteration % 10 == 0 and stagnation_count > 20:
            perturbation = np.random.normal(0, 0.01, n_points)
            best_h = np.clip(best_h + perturbation, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            stagnation_count = 0
    
    # Final refinement with local optimization
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
