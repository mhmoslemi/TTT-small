import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a symmetric triangular function as initial guess
        mid = n_points // 2
        h_start = np.minimum(np.arange(n_points), 2*mid - np.arange(n_points))
        h_start = h_start / (mid)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse optimization parameters
    max_iterations = 200
    patience = 50
    stagnation_counter = 0
    current_best = best_c5
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Gradient-based local optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    stagnation_counter = 0
                    current_best = best_c5
            else:
                # If optimization fails, try a simple gradient descent approach
                h_new = best_h.copy()
                grad = np.zeros_like(h_new)
                eps = 1e-6
                for i in range(n_points):
                    h_plus = h_new.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    h_minus = h_new.copy()
                    h_minus[i] -= eps
                    h_minus = np.clip(h_minus, 0, 1)
                    h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                    grad[i] = (evaluate_erdos_solution(h_plus, dx) - 
                              evaluate_erdos_solution(h_minus, dx)) / (2 * eps)
                
                # Simple gradient step with adaptive learning rate
                lr = 1e-3
                h_new = h_new - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    stagnation_counter = 0
                    current_best = best_c5
                else:
                    stagnation_counter += 1
        except Exception as e:
            stagnation_counter += 1
        
        # Check if we're stagnating
        if stagnation_counter >= patience:
            # Perturb solution and reset patience
            perturbation = np.random.normal(0, 0.02, n_points)
            best_h = best_h + perturbation
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            stagnation_counter = 0
            if best_c5 < 0.3807:
                break
    
    # Final optimization of best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
