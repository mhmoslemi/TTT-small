import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm with gradient refinement
    pop_size = 50
    elitism = 15
    mutation_rate = 0.1
    max_generations = 1000
    generation = 0
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        return h
    
    population = [create_individual() for _ in range(pop_size)]
    if 'initial_h_values' in globals():
        population[0] = h_start.copy()
    
    # Fitness evaluation
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    def selection(pop):
        fitness_scores = [fitness(ind) for ind in pop]
        indices = np.argsort(fitness_scores)
        return [pop[i] for i in indices[:elitism]]
    
    # Crossover
    def crossover(parent1, parent2):
        alpha = random.uniform(0.4, 0.6)
        child = alpha * parent1 + (1 - alpha) * parent2
        child = child * (n_points/2) / np.sum(child)
        child = np.clip(child, 0, 1)
        return child
    
    # Mutation
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                mutated[i] += np.random.normal(0, 0.02)
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points/2) / np.sum(mutated)
        return mutated
    
    # Gradient refinement using SLSQP
    def refine_individual(h):
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        try:
            res = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-6}
            )
            if res.success:
                h_refined = res.x
                return h_refined
        except:
            pass
        return h
    
    # Main GA loop
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate fitness
        fitness_scores = [fitness(ind) for ind in population]
        
        # Track best
        min_fitness = min(fitness_scores)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitness_scores.index(min_fitness)].copy()
        
        # Selection
        elites = selection(population)
        
        # Create new population
        new_population = elites[:]
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1, parent2 = random.sample(elites, 2)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        # Refine top individuals
        top_indices = np.argsort([fitness(ind) for ind in population])[:10]
        for idx in top_indices:
            refined = refine_individual(population[idx])
            new_population.append(refined)
        
        # Trim to population size
        population = new_population[:pop_size]
        
        generation += 1
    
    # Final refinement
    try:
        best_h_refined = best_h.copy()
        best_h_refined = np.clip(best_h_refined, 0, 1)
        best_h_refined = best_h_refined * (n_points/2) / np.sum(best_h_refined)
        final_c5 = evaluate_erdos_solution(best_h_refined, dx)
        if final_c5 < best_c5:
            best_h = best_h_refined
            best_c5 = final_c5
    except:
        pass

    return (list(best_h), best_c5, n_points)
