def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def smooth_max(x, tau=1.0):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(tau * x))) / tau

def smooth_max_grad(h, dx, tau=1.0):
    """Gradient of smooth max w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    # For each lag k, compute the gradient contribution
    for k in range(n):
        # h * (1 - h_{k}) 
        temp = h * (1 - np.roll(h, k))
        # For small k, need to use circular shift properly
        # Compute the correction needed for flipped indices
        temp_conv = np.zeros(n)
        for i in range(n):
            temp_conv[i] = temp[(i+k) % n]
        # Each element contributes to gradient
        temp_grad = np.zeros(n)
        if k == 0:
            temp_grad = 1 - h
        else:
            # Shift h backwards by k to get h_{i-k}
            h_shifted = np.roll(h, k)
            temp_grad = 1 - h_shifted
        # Normalize by exp(tau * value) / sum(exp(tau * all_values))
        exp_vals = np.exp(tau * temp)
        sum_exp = np.sum(exp_vals)
        # For each i, grad[i] += 1/sum_exp * exp(tau * temp[i]) * temp_grad[i] 
        for i in range(n):
            temp_grad_i = temp_grad[i]
            exp_val_i = exp_vals[i]
            grad[i] += (exp_val_i / sum_exp) * temp_grad_i
    return grad * dx
