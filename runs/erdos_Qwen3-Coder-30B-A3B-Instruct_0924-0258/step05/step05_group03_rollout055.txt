import numpy as np
from cvxpy import *
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial solution
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize to correct integral
    dx = 2.0 / n_points
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Sample shifts
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    
    # Variables
    h = Variable(n_points)
    C5 = Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, sum(h) == n_points / 2]
    
    # Convex relaxation loop
    max_iter = 50
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Formulate and solve relaxed problem
        prob = Problem(Minimize(C5), constraints)
        prob.solve(solver=ECOS, feastol=1e-8, abstol=1e-8)
        if prob.status != 'optimal':
            break
            
        h_vals = np.array(h.value).flatten()
        C5_val = C5.value
        
        # Find largest overlap constraint
        max_overlap = 0
        max_shift = 0
        for k in shifts:
            # Compute overlap for shift k
            h_shifted = np.concatenate([h_vals, np.zeros(int(k/dx))])
            h_shifted = h_shifted[:n_points]
            overlap = np.sum(h_vals * (1 - h_shifted)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
                max_shift = k
        
        # Add new constraint
        if max_overlap > C5_val + 1e-8:  # Add constraint if violated
            h_shifted = np.concatenate([h_vals, np.zeros(int(max_shift/dx))])
            h_shifted = h_shifted[:n_points]
            overlap_constraint = sum(h_vals * (1 - h_shifted)) <= C5 * n_points / 2
            constraints.append(overlap_constraint)
        else:
            break
            
        # Early termination if target achieved
        if C5_val < 0.3807:
            break
    
    # Final polishing with local optimizer
    best_h = h_vals.copy()
    best_c5 = C5_val
    
    def obj(x):
        return evaluate_erdos_solution(x, dx)
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
