import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def smooth_max(h, dx, tau=1.0):
    """Compute smooth approximation to max of correlations"""
    n = len(h)
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Apply smoothing
    smoothed = np.log(np.sum(np.exp(tau * corr))) / tau
    return smoothed * dx

    # Gradient-based optimization
    for i in range(max_iter):

            break

            # Internal to make closure work
            h = h_vec.copy()
            # Re-normalize to satisfy constraint sum = n_points / 2
            h = h * (n_points / 2) / np.sum(h)
            return evaluate_erdos_solution(h, dx)
        
        def grad(h_vec):
            # Close to exact gradient for smooth approximation 
            h = h_vec.copy()
            h = h * (n_points / 2) / np.sum(h)
            
            # Compute shifted correlations
            corr = np.correlate(h, 1 - h, mode='full')
            # Find maximum correlation index
            max_idx = np.argmax(corr) - (len(h) - 1)  # offset shift
            max_corr = corr[np.argmax(corr)]
            
            # Approximate gradient
            grad_arr = np.zeros_like(h)
            # For each position, compute gradient of correlation effect
            for j in range(len(h)):
                # This is a rough gradient computation
                # We compute the effect of small changes in h[j]
                h_copy = h.copy()
                h_copy[j] += 1e-6
                
                # Renormalize
                h_copy = h_copy * (n_points / 2) / np.sum(h_copy)
                
                corr_new = np.correlate(h_copy, 1 - h_copy, mode='full')
                grad_arr[j] = (np.max(corr_new) - max_corr) / 1e-6
                
            return grad_arr * dx

            h = h_vec.copy()

            for k in range(len(h_vec)):

                h_test = h_test * (n_points / 2) / np.sum(h_test)

                grad_val[k] = (val_test - val) / 1e-4
            return val, grad_val

            if res.success:
                h_final = res.x
                h_final = np.clip(h_final, 0, 1)
                h_final = h_final * (n_points / 2) / np.sum(h_final)
                c5_final = objective_final(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
