import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Smoothing parameters
    tau = 1.0
    tau_decay = 0.99
    tau_min = 1e-3
    step_size = 1.0
    max_iter = 3000
    
    # Initialize variables
    h = h_start.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    def smooth_max_overlap(h_vals, tau_val):
        # Compute all correlations
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        # Compute smooth max using log-sum-exp
        exp_vals = np.exp(tau_val * corr)
        smooth_max = np.log(np.sum(exp_vals)) / tau_val
        return smooth_max * dx
    
    def smooth_max_overlap_grad(h_vals, tau_val):
        # Compute all correlations
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        exp_vals = np.exp(tau_val * corr)
        sum_exp = np.sum(exp_vals)
        # Gradient of log-sum-exp
        grad = np.zeros_like(h_vals)
        # Compute gradient contributions
        for i in range(len(h_vals)):
            # Compute gradient components for the i-th entry due to shifting
            temp_corr = np.zeros(len(h_vals))
            for j in range(len(h_vals)):
                if i-j >= 0:
                    temp_corr[j] = 1 - h_vals[i - j]
                else:
                    temp_corr[j] = 0
            
            # Now compute the correlation for this shift
            corr_shifted = np.correlate(h_vals, temp_corr, mode='full')
            grad += exp_vals * (tau_val * corr_shifted) / sum_exp
        
        grad = grad * dx
        return grad
    
    # Main optimization loop
    for iter_num in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Optimize with smoothed objective
        def obj(h_vals):
            return smooth_max_overlap(h_vals, tau)
        
        def grad(h_vals):
            return smooth_max_overlap_grad(h_vals, tau)
            
        # L-BFGS-B optimization with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=bounds, 
                           options={'maxiter': 20, 'ftol': 1e-8, 'gtol': 1e-6})
            if res.success:
                h_new = res.x
                h = h_new.copy()
                
                # Enforce sum constraint
                h = h * (n_points / 2) / np.sum(h)
                
                # Evaluate actual C5
                c5_current = evaluate_erdos_solution(h, dx)
                
                if c5_current < c5_best:
                    c5_best = c5_current
                    h_best = h.copy()
                    
                # Update smoothing parameter
                tau = max(tau_min, tau * tau_decay)
                
            else:
                # If optimization fails, break
                break
                
        except Exception as e:
            # On error, break
            break
    
    # Final refinement with exact objective
    if time() - start_time < max_time * 0.9:
        def obj_exact(h_vals):
            return evaluate_erdos_solution(h_vals, dx)
        
        # Restart optimization from best solution
        h_refine = h_best.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Attempt refinement
        try:
            res = minimize(obj_exact, h_refine, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                h_final = h_final * (n_points / 2) / np.sum(h_final)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < c5_best:
                    h_best = h_final
                    c5_best = c5_final
        except:
            pass
    
    return (list(h_best), c5_best, n_points)
