import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid for relaxation (20 points)
    n_low = 20
    dx_low = 2.0 / n_low
    
    # Convex relaxation approach
    def solve_coarse_relaxation():
        h_low = cp.Variable(n_low)
        constraints = [
            cp.sum(h_low) == n_low / 2.0,
            h_low >= 0,
            h_low <= 1
        ]
        
        # Add constraints for first few shifts to manage complexity
        for k in range(1, min(8, n_low)):
            if k < n_low // 2:
                corr_sum = cp.sum(cp.multiply(h_low[:n_low-k], 1 - h_low[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_low[k:], 1 - h_low[:n_low-k]))
            constraints.append(corr_sum <= best_c5 * dx_low + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_low)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            return h_low.value
        return None
    
    # Initialize coarse solution
    h_coarse = solve_coarse_relaxation()
    if h_coarse is None:
        h_coarse = np.ones(n_low) * 0.5
        h_coarse = h_coarse * (n_low/2) / np.sum(h_coarse)
    
    # Upsample to high resolution
    x_low = np.linspace(0, 2, n_low, endpoint=False)
    x_high = np.linspace(0, 2, n_points, endpoint=False)
    h_upsampled = np.interp(x_high, x_low, h_coarse)
    h_upsampled = h_upsampled * (n_points/2) / np.sum(h_upsampled)
    h_upsampled = np.clip(h_upsampled, 0, 1)
    
    # Further refinement with adaptive gradient descent
    def obj_func(h):
        return evaluate_erdos_solution(h, dx)
    
    def grad_func(h):
        # Simple finite difference gradient approximation
        eps = 1e-5
        grad = np.zeros_like(h)
        base_val = obj_func(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            grad[i] = (obj_func(h_plus) - base_val) / eps
        return grad
    
    # Adaptive gradient descent with mutation
    h_current = h_upsampled.copy()
    learning_rate = 0.01
    patience = 0
    mutation_prob = 0.1
    mutation_strength = 0.05
    
    max_patience = 100
    mutation_interval = 50
    
    while time() - start_time < budget_s * 0.95:
        c5_current = obj_func(h_current)
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            patience = 0
        else:
            patience += 1
        
        # Adaptive learning rate
        if patience > 50:
            learning_rate *= 0.95
        
        # Compute gradient
        try:
            grad = grad_func(h_current)
        except Exception:
            grad = np.random.normal(0, 0.01, n_points)
        
        # Update with gradient
        h_new = h_current - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Mutation
        if patience % mutation_interval == 0 and patience > 0:
            mutate_mask = np.random.random(n_points) < mutation_prob
            mutation = np.random.normal(0, mutation_strength, n_points)
            mutation[mutate_mask] = mutation[mutate_mask] * np.sign(grad[mutate_mask])
            h_new = h_new + mutation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
        
        h_current = h_new
        
        if patience > max_patience:
            break
    
    # Final refinement
    try:
        cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            obj_func, 
            h_current, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons_sum,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if res.success:
            h_refined = res.x
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final constraint check
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), c5_final, n_points)
