import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Precompute correlation matrices
    def compute_correlation_matrices(n):
        A_matrices = []
        for k in range(1, 2*n - 1):
            A = np.zeros((n, n))
            for i in range(n):
                j = i + k - (n - 1)
                if 0 <= j < n:
                    A[i, j] = 1
            A_matrices.append(A)
        return A_matrices
    
    A_matrices = compute_correlation_matrices(n_points)
    
    # Lagrangian relaxation approach
    max_iter = 1000
    rho = 1e3
    lambda_vals = np.ones(2*n_points - 1) * 0.1
    mu = 1.0
    
    for iter_count in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Solve convex subproblem
        h = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(lambda_vals, 
                          [cp.sum(cp.multiply(h, A @ (1 - h))) 
                           for A in A_matrices])))
        
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
        
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.OSQP, verbose=False, 
                   eps_abs=1e-6, eps_rel=1e-6)
        
        if prob.status != cp.OPTIMAL:
            break
            
        h_sol = h.value
        h_sol = np.clip(h_sol, 0, 1)
        h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
        
        # Compute current overlap
        current_c5 = evaluate_erdos_solution(h_sol, dx)
        
        if current_c5 < best_c5:
            best_h = h_sol.copy()
            best_c5 = current_c5

        # Update multipliers
        for k, A in enumerate(A_matrices):
            overlap = np.dot(h_sol, A @ (1 - h_sol))
            lambda_vals[k] = max(0, lambda_vals[k] + rho * (overlap - current_c5 * dx))
            
        # Update penalty parameter
        sum_h = np.sum(h_sol)
        if abs(sum_h - n_points / 2) > 1e-4:
            rho = min(1e6, rho * 2)
        else:
            rho = max(1e-3, rho * 0.5)
            
        if best_c5 < 0.3807:
            break
            
    # Final local refinement
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_init = best_h.copy()
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
