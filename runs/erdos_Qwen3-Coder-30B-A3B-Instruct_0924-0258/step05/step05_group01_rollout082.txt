import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Initialize with triangular shape
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 1 - np.abs(x - 1)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach with simulated annealing
    scales = [35, 50, 75, 100]
    current_scale_idx = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale_idx < len(scales):
        scale = scales[current_scale_idx]
        # Interpolate if needed
        if scale > n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            h_interp = np.interp(x_new, x_orig, best_h)
            h_current = h_interp
            n_points = scale
        else:
            h_current = best_h
            n_points = scale
            
        dx = 2.0 / n_points
        
        # Simulated Annealing
        T0 = 0.1
        alpha = 0.99
        max_iter = int(5000 * (scale / 35))  # Scale iterations with resolution
        T = T0
        h_best = h_current.copy()
        c5_best = evaluate_erdos_solution(h_current, dx)
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate neighbor by perturbing some values
            h_new = h_current.copy()
            idxs = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
            for idx in idxs:
                # Add small random perturbation
                pert = np.random.normal(0, 0.05)
                h_new[idx] += pert
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate new solution
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            if c5_new < c5_best:
                h_current = h_new
                c5_best = c5_new
                h_best = h_new.copy()
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            else:
                # Accept with probability
                delta = c5_new - c5_best
                if np.random.rand() < np.exp(-delta / T):
                    h_current = h_new
                    c5_best = c5_new
                    
            T = alpha * T
            
        current_scale_idx += 1
    
    # Final refinement using local optimization
    if time() - start_time < budget_s * 0.9:
        try:
            # Gradient-based refinement if possible
            from scipy.optimize import minimize
            
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            # Ensure constraints
            h_refined = best_h.copy()
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            
            # Local optimization
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2}
            res = minimize(obj, h_refined, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
                c5_final = obj(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
