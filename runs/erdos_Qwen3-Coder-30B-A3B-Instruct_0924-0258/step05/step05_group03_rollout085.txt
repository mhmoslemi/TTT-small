import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Initialize
    h = initial_h.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    # Gradient-based optimization parameters
    max_iter = 500
    stagnation_limit = 50
    patience = 0
    min_improvement = 1e-8
    
    # Sample shifts for gradient computation
    shift_samples = min(50, n_points)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample some integer shifts
        shifts = np.random.choice(n_points, shift_samples, replace=False)
        max_overlap = 0
        best_shift = 0
        
        # Find the shift that gives maximum overlap
        for k in shifts:
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlap *= 2.0 / n_points
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = k
        
        # Compute gradient for this shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_shift) % n_points
            grad[i] += (1 - h[j]) * 2.0 / n_points
            grad[j] -= h[i] * 2.0 / n_points
        
        # Backtracking line search
        alpha = 1.0
        step_size = 0.01
        while alpha > 1e-10:
            h_new = h - alpha * grad
            # Project to feasible region
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_c5 = evaluate_erdos_solution(h_new, 2.0 / n_points)
            
            if new_c5 < best_c5 - min_improvement:
                h = h_new
                best_c5 = new_c5
                patience = 0
                break
            alpha *= 0.5
        
        # Check for stagnation
        if best_c5 < best_c5 - min_improvement:
            patience = 0
        else:
            patience += 1
            
        # If stagnation, perturb solution
        if patience > stagnation_limit:
            h = h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            patience = 0
            
        # Early termination
        if best_c5 < 0.3807:
            break
            
        # Update best solution
        if best_c5 < evaluate_erdos_solution(best_h, 2.0 / n_points):
            best_h = h.copy()
    
    # Final refinement with scipy
    def obj(x):
        return evaluate_erdos_solution(x, 2.0 / n_points)
        
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
