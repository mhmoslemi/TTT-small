import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()

    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    def create_individual():
        h = np.random.random(n_points)
        h = h * (n_points/2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    if 'initial_h_values' in globals():
        population = [h_start.copy()]
        for _ in range(pop_size - 1):
            population.append(create_individual())
    else:
        population = [create_individual() for _ in range(pop_size)]
    
    # GA iteration
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for ind in population:
            c5 = evaluate_erdos_solution(ind, dx)
            fitness.append(c5)
        
        # Selection with tournament
        def tournament_select():
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            tournament.sort(key=lambda x: x[1])
            return tournament[0][0].copy()
        
        # Crossover
        def crossover(parent1, parent2):
            if random.random() > crossover_rate:
                return parent1.copy(), parent2.copy()
            point = random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:point], parent2[point:]])
            child2 = np.concatenate([parent2[:point], parent1[point:]])
            
            # Repair children
            child1_sum = np.sum(child1)
            child1 = child1 * (n_points/2) / child1_sum
            child1 = np.clip(child1, 0, 1)
            
            child2_sum = np.sum(child2)
            child2 = child2 * (n_points/2) / child2_sum
            child2 = np.clip(child2, 0, 1)
            
            return child1, child2
        
        # Mutation
        def mutate(individual):
            mutated = individual.copy()
            for i in range(n_points):
                if random.random() < mutation_rate:
                    mutated[i] += np.random.normal(0, 0.02)
            mutated = mutated * (n_points/2) / np.sum(mutated)
            mutated = np.clip(mutated, 0, 1)
            return mutated
        
        # Create new population
        new_population = []
        
        # Elite preservation
        elite_indices = sorted(range(len(fitness)), key=lambda i: fitness[i])[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1 = tournament_select()
            parent2 = tournament_select()
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:pop_size]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
    
    # Resample at higher resolution if needed and time permits
    if time() - start_time < budget_s * 0.9 and best_c5 > 0.3808:
        n_points = 40
        dx = 2.0 / n_points
        h_interp = np.interp(np.linspace(0, 2, n_points),

        best_h = h_interp
        best_h = best_h * (n_points/2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
        # Local SLSQP refinement
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                best_h = res.x.copy()
                best_c5 = objective(best_h)
        except Exception:
            pass
    
    # Final refinement
    try:
        h_final = np.clip(best_h, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
