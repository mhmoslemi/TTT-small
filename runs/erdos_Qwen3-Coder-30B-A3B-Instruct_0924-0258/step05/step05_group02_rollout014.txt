import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use existing initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
        
    # Hybrid optimization: local search + global search
    max_iterations = 1000
    iteration = 0
    patience = 0
    max_patience = 50
    
    while iteration < max_iterations and time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Random perturbation
        h_perturbed = best_h.copy()
        noise = np.random.normal(0, 0.03, n_points)
        h_perturbed += noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Local optimization
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-7, 'eps': 1e-4}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    patience = 0
                else:
                    patience += 1
            else:
                patience += 1
        except Exception:
            patience += 1

        if iteration % 20 == 0:

            h_sparse = np.zeros(n_points)

            h_sparse[active_indices] = 1.0
            h_sparse = h_sparse * (n_points/2) / np.sum(h_sparse)
            try:
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(
                    lambda h: evaluate_erdos_solution(h, dx), 
                    h_sparse, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=cons_sum,
                    options={'maxiter': 30, 'ftol': 1e-6}
                )
                if res.success:
                    h_refined = res.x
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        patience = 0
            except Exception:
                pass
                
        # Early stopping check
        if patience >= max_patience:
            # Try to generate better random samples
            for _ in range(10):
                h_trial = np.random.rand(n_points)
                h_trial = h_trial * (n_points/2) / np.sum(h_trial)

                    best_c5 = c5_trial
                    patience = 0
                    
        # Check time budget
        if time() - start_time > budget_s * 0.9:

    # Final refinement
    try:

        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    # Ensure exact constraint satisfaction
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    
    return (list(h_final), best_c5, n_points)
