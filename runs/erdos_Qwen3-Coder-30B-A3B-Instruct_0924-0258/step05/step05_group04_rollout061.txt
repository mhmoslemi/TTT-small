import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    num_particles = 40
    max_iter = 500
    inertia_weight = 0.7
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.2
    warm_restart_after = 200
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    for _ in range(num_particles):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.02, n_points)
        else:
            h = np.random.random(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 1e-10:
            h = h * (n_points / 2) / h_sum
        else:
            h = np.ones(n_points) * (n_points / 2)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        fitness = evaluate_erdos_solution(h, dx)
        pbest.append(h.copy())
        pbest_fitness.append(fitness)
    
    # Find initial global best
    best_idx = np.argmin(pbest_fitness)
    gbest = pbest[best_idx].copy()
    gbest_fitness = pbest_fitness[best_idx]
    
    # PSO iterations
    stagnation_count = 0
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.98:
            break
            
        if stagnation_count >= warm_restart_after:
            # Warm restart: reset velocities and perturb some particles
            velocities = [np.zeros(n_points) for _ in range(num_particles)]
            for i in range(num_particles // 2):
                idx = random.randint(0, num_particles - 1)
                particles[idx] = np.random.random(n_points)
                h_sum = np.sum(particles[idx])
                if h_sum > 1e-10:
                    particles[idx] = particles[idx] * (n_points / 2) / h_sum
                pbest[idx] = particles[idx].copy()
                pbest_fitness[idx] = evaluate_erdos_solution(particles[idx], dx)
            stagnation_count = 0
            
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            h_sum = np.sum(particles[i])
            if h_sum > 1e-10:
                particles[i] = particles[i] * (n_points / 2) / h_sum
            
            # Evaluate fitness
            fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = fitness
                    if gbest_fitness < best_c5:
                        best_c5 = gbest_fitness
                        best_h = gbest.copy()
                    stagnation_count = 0
                    
        stagnation_count += 1
    
    # Local refinement of best solution
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_refined = best_h.copy()
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        def con(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': con}
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(obj, h_refined, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = obj(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
