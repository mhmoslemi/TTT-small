import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    elapsed = lambda: time() - start_time
    
    # Check if initial construction is available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    n_low = 20  # Coarse grid for relaxation
    dx_low = 2.0 / n_low
    
    # Initialize coarse grid solution
    h_low = np.ones(n_low) * 0.5
    h_low = h_low * (n_low/2) / np.sum(h_low)
    
    # Solve coarse problem with convex relaxation
    if n_low <= 30:  # Reasonable for ECOS solver
        try:
            h_cvx = cp.Variable(n_low)
            constraints = [
                cp.sum(h_cvx) == n_low / 2.0,
                h_cvx >= 0,
                h_cvx <= 1
            ]
            
            # Add key constraints from first few shifts
            num_constraints = min(5, n_low)
            for k in range(1, num_constraints):
                shift_constraints = []
                if k < n_low:
                    shift_constraints.append(cp.sum(cp.multiply(
                        h_cvx[:n_low-k], 1 - h_cvx[k:])))
                else:
                    shift_constraints.append(cp.sum(cp.multiply(
                        h_cvx[k:], 1 - h_cvx[:n_low-k])))
                constraints.append(shift_constraints[0] <= best_c5 * dx_low + 1e-6)
            
            prob = cp.Problem(cp.Minimize(cp.sum(h_cvx)), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-7)
            if prob.status == cp.OPTIMAL:
                h_low = h_cvx.value
                c5_low = evaluate_erdos_solution(h_low, dx_low)
                if c5_low < best_c5:
                    best_c5 = c5_low
                    best_h = h_low.copy()
        except Exception:
            pass
    
    # Interpolate coarse solution to fine grid
    h_interpolated = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, n_low), 
                              best_h[:n_low])
    h_interpolated = h_interpolated * (n_points/2) / np.sum(h_interpolated)
    h_interpolated = np.clip(h_interpolated, 0, 1)
    
    # Fine-grid optimization with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    max_restarts = 100
    iterations_since_improvement = 0
    
    while elapsed() < budget_s * 0.9:
        # Local refinement step
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                objective,
                h_interpolated,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    iterations_since_improvement = 0
                else:
                    iterations_since_improvement += 1
            else:
                iterations_since_improvement += 1
        except Exception:
            iterations_since_improvement += 1
        
        # Restart with perturbation if needed
        if iterations_since_improvement > 30:
            h_perturbed = h_interpolated + np.random.normal(0, 0.05, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            h_interpolated = h_perturbed.copy()
            iterations_since_improvement = 0
        
        # Stop condition
        if iterations_since_improvement > max_restarts:
            break
        
        # Update h_interpolated for next iteration
        h_interpolated = best_h.copy()
        h_interpolated = np.clip(h_interpolated, 0, 1)
        h_interpolated = h_interpolated * (n_points/2) / np.sum(h_interpolated)
    
    # Final cleanup
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
