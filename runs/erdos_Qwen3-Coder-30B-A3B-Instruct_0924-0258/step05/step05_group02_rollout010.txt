import numpy as np
from scipy.optimize import minimize
from time import time
from itertools import combinations

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simple grid search approach with local refinement
    max_restarts = 100
    
    # Start with a smooth, sparse solution
    def initialize_sparse():
        h = np.zeros(n_points)
        # Place 1s in roughly half the points, with spacing
        active_positions = np.arange(0, n_points, 2)
        h[active_positions[:n_points//4]] = 1.0
        h = h * (n_points/2) / np.sum(h)
        return h
    
    h_current = initialize_sparse()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    c5_current = evaluate_erdos_solution(h_current, dx)
    if c5_current < best_c5:
        best_h = h_current.copy()
        best_c5 = c5_current
    
    restart_count = 0
    local_search_count = 0
    
    while time() - start_time < budget_s * 0.95:
        # Local search with perturbation
        h_perturbed = h_current.copy()
        # Add small noise
        noise = np.random.normal(0, 0.03, n_points)
        h_perturbed += noise
        
        # Ensure bounds
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Renormalize
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Optimize using scipy
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(
                objective,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-6}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
                    local_search_count = 0
                else:
                    restart_count += 1
                    local_search_count += 1
            else:
                restart_count += 1
                local_search_count += 1
        except Exception:
            restart_count += 1
            local_search_count += 1
        
        # Global restart if needed
        if restart_count > max_restarts:
            # Try a completely different approach
            h_random = np.random.rand(n_points)
            h_random = h_random * (n_points/2) / np.sum(h_random)
            h_random = np.clip(h_random, 0, 1)
            c5_random = evaluate_erdos_solution(h_random, dx)
            if c5_random < best_c5:
                best_h = h_random
                best_c5 = c5_random
            restart_count = 0
        
        # Grid-based refinement for sparsity
        if local_search_count > 20:
            # Try a sparse pattern
            h_sparse = np.zeros(n_points)
            positions = np.random.choice(n_points, size=n_points//3, replace=False)
            h_sparse[positions] = 1.0
            
            # Scale to desired integral
            h_sparse = h_sparse * (n_points/2) / np.sum(h_sparse)
            c5_sparse = evaluate_erdos_solution(h_sparse, dx)
            if c5_sparse < best_c5:
                best_h = h_sparse
                best_c5 = c5_sparse
            local_search_count = 0
        
        h_current = best_h.copy()
        
        if time() - start_time > budget_s * 0.9:
            break
    
    # Final check
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
