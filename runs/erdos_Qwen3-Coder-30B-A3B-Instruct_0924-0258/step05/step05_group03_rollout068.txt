import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Time tracking
    time_limit = budget_s * 0.95
    
    # CCP loop
    prev_c5 = float('inf')
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    iteration = 0
    
    while time() - start_time < time_limit:
        iteration += 1
        
        # Find worst shift k*
        shifts = np.linspace(0, 2, 100)
        max_correlations = []
        
        for k in shifts:
            # Compute correlation for this shift
            if k == 0:
                corr = np.correlate(h, 1 - h, mode='full')
                corr_val = np.max(corr)
            else:
                # Shift h by k, compute correlation
                k_idx = int(k / (2.0 / n_points))
                if k_idx >= n_points:
                    k_idx = n_points - 1
                    
                # Use circular shift to implement the shift k properly
                # We need to simulate shifting by k
                h_shifted = np.roll(h, -k_idx)
                h_shifted = np.concatenate([h_shifted, np.zeros(k_idx)])
                corr = np.correlate(h, 1 - h_shifted[:n_points], mode='valid')
                corr_val = np.max(corr)
                
            max_correlations.append(corr_val)
        
        k_star = shifts[np.argmax(max_correlations)]
        k_star_idx = int(k_star / (2.0 / n_points))
        
        # Prepare convex optimization for h
        h_new = cp.Variable(n_points)
        dx = 2.0 / n_points
        
        # Objective: Minimize the worst case correlation term
        # We want to minimize max_k integral h(x)(1 - h(x + k)) for this fixed h
        # We approximate it by leveraging one particular bad shift k_star
        # Create the linear form for the fixed-shift version
        # h_new[k_star:] * (1 - h[:n_points-k_star]) at that shift
        # The effect is a quadratic form, but for optimization, we fix h and linearize
        
        # Using a dual formulation approach:
        # Linearize the contribution to the overall integral based on fixed shift
        
        # Build constraint matrix for this linearized version
        # For simplicity, since we've fixed shift k_star (and computed worst case),
        # and want to minimize the contribution due to that k_star
        objective = cp.Minimize(cp.sum(cp.multiply(h_new, 1 - h)))
        
        constraints = [
            cp.sum(h_new) == n_points / 2,
            h_new >= 0,
            h_new <= 1
        ]
        
        # Convex optimization using ECOS solver
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            new_h = np.array(h_new.value).reshape(-1)
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
        except:
            new_h = h.copy()
            
        # Evaluate the current solution
        h = new_h.copy()
        c5 = evaluate_erdos_solution(h, 2.0 / n_points)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Termination criteria
        if best_c5 < 0.3807:
            break
            
        if abs(c5 - prev_c5) < 1e-5:
            break
            
        prev_c5 = c5
    
    # Final polishing with SLSQP
    def obj(x):
        return evaluate_erdos_solution(x, 2.0 / n_points)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0

    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            final_h = res.x
            final_c5 = obj(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
