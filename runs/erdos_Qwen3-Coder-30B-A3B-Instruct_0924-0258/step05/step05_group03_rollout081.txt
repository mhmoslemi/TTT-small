import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Scale to sum to n_points/2
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Generate shift samples
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    
    # Optimization parameters
    max_iter = 50
    tolerance = 1e-6
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    # Convex optimization approach with cutting planes
    h_vars = cp.Variable(n_points)
    c5_var = cp.Variable()
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points / 2,
        c5_var >= 0
    ]
    
    # Penalty for overlap constraints
    penalization = 0.001
    
    iteration = 0
    while iteration < max_iter and (time() - start_time) < budget_s * 0.95:
        # Evaluate shifts
        overlaps = []
        for shift in shifts:
            # Compute overlap for current shift
            # This uses discrete convolution-like approach
            overlap = 0.0
            dx = 2.0 / n_points
            for i in range(n_points):
                j = i + int(shift / dx)
                if j < n_points:
                    overlap += h_vars[i] * (1.0 - h_vars[j])
            overlaps.append(overlap * dx)
        
        max_overlap = max(overlaps)
        
        # Add this as a constraint
        if abs(max_overlap - best_c5) > tolerance:
            # Add constraint that max overlap should be less than or equal to c5_var
            constraints.append(c5_var >= max_overlap)
        else:
            # If no change, break early
            break

        # Convex optimization step
        try:
            objective = cp.Minimize(c5_var)
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_sol = h_vars.value
                c5_sol = c5_var.value
                best_c5 = min(best_c5, c5_sol)
                
                if np.all(0 <= h_sol) and np.all(h_sol <= 1):
                    # Adjust sum to match constraint
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    best_h = h_sol.copy()
                else:
                    # Revert to last valid solution
                    pass
            else:
                break
                
        except Exception:
            break
            
        # Apply local refinement with SLSQP
        def obj_local(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        
        def cons_local(x):
            return np.sum(x) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': cons_local}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj_local, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons_sum,
                          options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_local = res.x
                c5_local = obj_local(h_local)
                if c5_local < best_c5:
                    best_c5 = c5_local
                    best_h = h_local.copy()
        except:
            pass
            
        # Early termination
        if best_c5 < 0.3807:
            break
            
        iteration += 1
        
    # Final cleanup and polish
    def obj_final(x):
        return evaluate_erdos_solution(x, 2.0 / n_points)
    
    def cons_final(x):
        return np.sum(x) - n_points / 2.0
        
    cons_sum = {'type': 'eq', 'fun': cons_final}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj_final, best_h, method='SLSQP',
                      bounds=bounds, constraints=cons_sum,
                      options={'maxiter': 200, 'ftol': 1e-12})
        if res.success:
            best_h = res.x
            best_c5 = obj_final(best_h)
    except:
        pass
    
    # Ensure final solution meets constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)

    return (list(best_h), best_c5, n_points)
