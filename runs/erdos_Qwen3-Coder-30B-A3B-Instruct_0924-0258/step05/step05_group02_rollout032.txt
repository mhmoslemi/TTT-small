import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm with gradient refinement
    pop_size = 50
    max_generations = 1000
    elitism_rate = 0.3
    mutation_rate = 0.1
    crossover_alpha_range = (0.4, 0.6)
    
    # Initialize population
    population = []
    fitness_scores = []
    
    # Include initial solution
    population.append(best_h.copy())
    fitness_scores.append(best_c5)
    
    # Generate random population
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
        fitness_scores.append(evaluate_erdos_solution(h, dx))
    
    # Main evolution loop
    generations = 0
    stagnation_counter = 0
    max_stagnation = 50
    
    while time() - start_time < budget_s * 0.95 and generations < max_generations:
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        if fitness_scores[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness_scores[0]
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        if stagnation_counter > max_stagnation:
            break
            
        # Select elite
        num_elite = max(1, int(elitism_rate * pop_size))
        elite = population[:num_elite]
        elite_fitness = fitness_scores[:num_elite]
        
        # Create new population
        new_population = elite[:]
        new_fitness = elite_fitness[:]
        
        # Generate offspring through crossover and mutation
        while len(new_population) < pop_size:
            # Selection: tournament
            parent1_idx = random.randint(0, num_elite - 1)
            parent2_idx = random.randint(0, num_elite - 1)
            while parent2_idx == parent1_idx:
                parent2_idx = random.randint(0, num_elite - 1)
            
            parent1 = elite[parent1_idx]
            parent2 = elite[parent2_idx]
            
            # Crossover with sparsity bias
            alpha = random.uniform(*crossover_alpha_range)
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Normalize sum constraint
            child = child * (n_points / 2) / np.sum(child)
            child = np.clip(child, 0, 1)
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += np.random.normal(0, 0.02)
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            new_fitness.append(evaluate_erdos_solution(child, dx))
            
        population = new_population
        fitness_scores = new_fitness
        generations += 1
    
    # Final refinement with gradient descent
    refined_h = best_h.copy()
    refined_h = np.clip(refined_h, 0, 1)
    refined_h = refined_h * (n_points/2) / np.sum(refined_h)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    optimized_h = minimize(
        objective,
        refined_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    if optimized_h.success:
        final_h = optimized_h.x.copy()
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    # Ensure final constraints
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
