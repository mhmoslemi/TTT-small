import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 35
        h0 = np.ones(n_points) * 0.5
    
    # Normalize initial solution to sum to n_points / 2
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Prepare shift points
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    
    # Variables for optimization
    h = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
        C5 >= 0
    ]
    
    # Add constraints from sample shifts
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    iteration = 0
    stagnation = 0
    max_stagnation = 50
    
    while time() - start_time < budget_s * 0.95 and stagnation < max_stagnation:
        # Evaluate overlaps
        overlaps = []
        for shift in shifts:
            # Compute overlap at shift
            if shift == 0:
                overlap = np.sum(h0 * (1 - h0))
            else:
                # This is to evaluate the discrete correlation
                # Shift by integer equivalent in samples (modify as needed)
                # For now using a simplified approach with an adapted shift
                shift_idx = int(shift * n_points / 2)
                if shift_idx >= n_points:
                    continue
                h_shifted = np.roll(h0, shift_idx)
                overlap = np.sum(h0 * (1 - h_shifted))
            overlaps.append(overlap)
        
        max_overlap_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_overlap_idx]
        best_shift = shifts[max_overlap_idx]
        
        # Add constraint based on the worst case
        # From overlap check at this shift
        constraint = C5 >= max_overlap
        
        # Solve the SDP relaxation
        obj = cp.Minimize(C5)
        prob = cp.Problem(obj, constraints + [constraint])
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if prob.status == cp.OPTIMAL:
                h_val = h.value
                new_c5 = C5.value
                # Check if the new solution is better
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_val.copy()
                    stagnation = 0
                else:
                    stagnation += 1
            else:
                stagnation += 1
        except:
            stagnation += 1
        
        # Local refinement
        try:
            def obj(x):
                return evaluate_erdos_solution(x, 2.0 / n_points)
            
            def constraint_sum(x):
                return np.sum(x) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(obj, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-7})
            if res.success:
                best_h = res.x
                best_c5 = obj(best_h)
        except:
            pass
        
        iteration += 1
        
        if best_c5 < 0.3807:
            break
    
    # Final polishing with local optimization
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
