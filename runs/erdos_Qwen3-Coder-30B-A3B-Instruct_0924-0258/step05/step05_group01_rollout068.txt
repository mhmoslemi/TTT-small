import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search for binary solutions
    # Convert to binary representation
    h_binary = (h_start > 0.5).astype(int)
    # Adjust to have exactly n_points/2 ones
    ones_count = np.sum(h_binary)
    if ones_count > n_points // 2:
        # Turn some 1s to 0s
        zeros_indices = np.where(h_binary == 0)[0]
        if len(zeros_indices) >= ones_count - n_points // 2:
            flip_indices = np.random.choice(zeros_indices, ones_count - n_points // 2, replace=False)
            h_binary[flip_indices] = 1
        else:
            # If we can't flip enough zeros, flip some ones to zeros
            ones_indices = np.where(h_binary == 1)[0]
            flip_indices = np.random.choice(ones_indices, ones_count - n_points // 2, replace=False)
            h_binary[flip_indices] = 0
    elif ones_count < n_points // 2:
        # Turn some 0s to 1s
        ones_indices = np.where(h_binary == 1)[0]
        if len(ones_indices) >= n_points // 2 - ones_count:
            flip_indices = np.random.choice(ones_indices, n_points // 2 - ones_count, replace=False)
            h_binary[flip_indices] = 0
        else:
            # If we can't flip enough ones, flip some zeros to ones
            zeros_indices = np.where(h_binary == 0)[0]
            flip_indices = np.random.choice(zeros_indices, n_points // 2 - ones_count, replace=False)
            h_binary[flip_indices] = 1
    
    current_h = h_binary.copy()
    current_c5 = evaluate_erdos_solution(current_h, dx)
    if current_c5 < best_c5:
        best_h = current_h.copy()
        best_c5 = current_c5
    
    # Tabu list for moves (index pairs that flipped)
    tabu_list = []
    tabu_tenure = 10
    max_no_improvement = 100
    no_improvement_count = 0
    shaking_interval = 50
    iteration = 0
    best_global = best_c5
    
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        if no_improvement_count >= max_no_improvement:
            # Reset if no improvement for a while
            current_h = h_binary.copy()
            current_c5 = evaluate_erdos_solution(current_h, dx)
            no_improvement_count = 0
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        elif iteration % shaking_interval == 0:
            # Perform shaking: flip a few bits randomly
            idxs_to_flip = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
            for idx in idxs_to_flip:
                current_h[idx] = 1 - current_h[idx]
            # Ensure exactly n_points/2 ones
            ones_count = np.sum(current_h)
            if ones_count > n_points // 2:
                ones_indices = np.where(current_h == 1)[0]
                flip_indices = np.random.choice(ones_indices, int(ones_count - n_points // 2), replace=False)
                current_h[flip_indices] = 0
            elif ones_count < n_points // 2:
                zeros_indices = np.where(current_h == 0)[0]
                flip_indices = np.random.choice(zeros_indices, int(n_points // 2 - ones_count), replace=False)
                current_h[flip_indices] = 1
            current_c5 = evaluate_erdos_solution(current_h, dx)
            no_improvement_count += 1
            
        # Generate neighbor: flip one 1 to 0 and one 0 to 1
        ones_indices = np.where(current_h == 1)[0]
        zeros_indices = np.where(current_h == 0)[0]
        if len(ones_indices) > 0 and len(zeros_indices) > 0:
            # Pick a random one and zero to flip
            i1 = np.random.choice(ones_indices)
            i0 = np.random.choice(zeros_indices)
            # Create candidate
            candidate_h = current_h.copy()
            candidate_h[i1] = 0
            candidate_h[i0] = 1
            # Check if this is a valid candidate (exactly n_points/2 ones)
            candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
            # Check if move is tabu
            move = (i1, i0)
            tabu = False
            for m in tabu_list:
                if m == move or m == (i0, i1):
                    tabu = True
                    break
            # Evaluate if better or aspiration criterion met
            if candidate_c5 < current_c5 or not tabu:
                current_h = candidate_h
                current_c5 = candidate_c5
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    best_global = best_c5
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
                # Add to tabu list
                tabu_list.append(move)
                if len(tabu_list) > tabu_tenure:
                    tabu_list.pop(0)
            else:
                no_improvement_count += 1
        else:
            # Fallback: ensure sum constraint is met and re-evaluate
            ones_count = np.sum(current_h)
            if ones_count > n_points // 2:
                ones_indices = np.where(current_h == 1)[0]
                flip_indices = np.random.choice(ones_indices, int(ones_count - n_points // 2), replace=False)
                current_h[flip_indices] = 0
                current_c5 = evaluate_erdos_solution(current_h, dx)
                no_improvement_count += 1
            elif ones_count < n_points // 2:
                zeros_indices = np.where(current_h == 0)[0]
                flip_indices = np.random.choice(zeros_indices, int(n_points // 2 - ones_count), replace=False)
                current_h[flip_indices] = 1
                current_c5 = evaluate_erdos_solution(current_h, dx)
                no_improvement_count += 1
            else:
                no_improvement_count += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
