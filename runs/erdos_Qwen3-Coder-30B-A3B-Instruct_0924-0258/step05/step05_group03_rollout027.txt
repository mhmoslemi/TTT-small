import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 35
        h_init = np.ones(n_points) * 0.5
    
    # Normalization
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Shift sampling for constraint tightening
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    dx = 2.0 / n_points
    
    # Cutting-plane approach
    h_sol = h_init.copy()
    c5_bound = evaluate_erdos_solution(h_sol, dx)
    
    max_iter = 50
    improvement_threshold = 1e-5
    last_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all overlaps
        overlaps = []
        for k in shifts:
            # Shift index for correlation
            k_idx = int(k / dx)
            if k_idx >= n_points:
                k_idx = n_points - 1
            cross_corr = np.correlate(h_sol, 1 - h_sol, mode='full')
            overlap_val = cross_corr[n_points - 1 + k_idx] * dx
            overlaps.append(overlap_val)
        
        max_overlap_idx = np.argmax(overlaps)
        max_overlap_shift = shifts[max_overlap_idx]
        max_overlap_val = overlaps[max_overlap_idx]
        
        # Add constraint for this shift (cutting plane)
        if abs(max_overlap_val - c5_bound) < 1e-10:
            break  # No improvement, done
        
        c5_bound = max_overlap_val
        
        # Stop early if target reached
        if c5_bound < 0.3807:
            break
            
        # Local optimization refinement
        def obj_local(x):
            return evaluate_erdos_solution(x, dx)
            
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj_local, h_sol, method='SLSQP', 
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 50, 'ftol': 1e-7})
            if res.success:
                h_sol = res.x
                c5_bound = obj_local(h_sol)
        except:
            pass
        
        # Re-check and potentially tighten more
        if abs(last_c5 - c5_bound) < improvement_threshold:
            break
        last_c5 = c5_bound
    
    # Final polishing
    def obj_final(x):
        return evaluate_erdos_solution(x, dx)
        
    def constraint_sum_final(x):
        return np.sum(x) - n_points / 2.0
    
    cons_final = {'type': 'eq', 'fun': constraint_sum_final}
    bounds_final = [(0, 1) for _ in range(n_points)]
    
    try:
        res_final = minimize(obj_final, h_sol, method='SLSQP',
                            bounds=bounds_final, constraints=cons_final,
                            options={'maxiter': 100, 'ftol': 1e-9})
        if res_final.success:
            h_sol = res_final.x
            c5_bound = obj_final(h_sol)
    except:
        pass
    
    # Final validation
    c5_bound = evaluate_erdos_solution(h_sol, dx)
    
    return (list(h_sol), c5_bound, n_points)
