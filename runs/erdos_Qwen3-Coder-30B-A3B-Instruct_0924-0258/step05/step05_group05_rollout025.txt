import numpy as np
from scipy.optimize import minimize
from time import time
import hashlib

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def hash_solution(h):
    h_sorted = np.sort(h)
    return hashlib.md5(h_sorted.tobytes()).hexdigest()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize tabu search parameters
    tabu_list = []
    tabu_tenure = min(50, max(5, n_points // 4))
    max_stagnation = 100
    mutation_scale = 0.04
    max_iter = 1000
    refinement_interval = 10
    mutation_budget = max(1, int(0.3 * n_points))
    
    # Create initial tabu list with the starting solution
    tabu_list.append(hash_solution(h_start))
    
    current_h = h_start.copy()
    current_c5 = best_c5
    stagnation_counter = 0
    improvement_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate new candidate by perturbing some elements
        candidate_h = current_h.copy()
        
        # Perturb a subset of points
        indices_to_perturb = np.random.choice(n_points, mutation_budget, replace=False)
        for i in indices_to_perturb:
            candidate_h[i] += np.random.normal(0, mutation_scale)
        
        # Clip and normalize
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        
        # Compute overlap for candidate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Check tabu list for this candidate
        candidate_hash = hash_solution(candidate_h)
        is_tabu = candidate_hash in tabu_list
        
        # Accept better solution or non-tabu worse solution
        if candidate_c5 < current_c5 or (not is_tabu and candidate_c5 <= current_c5 + 1e-8):
            current_h = candidate_h
            current_c5 = candidate_c5
            
            # Update best if improved
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
                # Check if we have a target improvement
                if best_c5 < 0.3807:
                    break
                    
                improvement_count += 1
                
                # Apply local refinement periodically
                if improvement_count % refinement_interval == 0:
                    def objective(h):
                        return evaluate_erdos_solution(h, dx)
                    
                    def constraint_sum(h):
                        return np.sum(h) - n_points / 2.0
                    
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    bounds = [(0, 1) for _ in range(n_points)]
                    
                    try:
                        res = minimize(objective, current_h, method='SLSQP', 
                                       bounds=bounds, constraints=cons_sum, 
                                       options={'maxiter': 100, 'ftol': 1e-8})
                        if res.success:
                            h_refined = res.x
                            c5_refined = objective(h_refined)
                            if c5_refined < current_c5:
                                current_h = h_refined
                                current_c5 = c5_refined
                                if c5_refined < best_c5:
                                    best_h = h_refined
                                    best_c5 = c5_refined
                    except:
                        pass
                        
            # Update tabu list
            tabu_list.append(candidate_hash)
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
                
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Handle stagnation
        if stagnation_counter > max_stagnation:
            mutation_scale = min(0.3, mutation_scale * 1.2)  # Increase mutation
            stagnation_counter = 0

    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
