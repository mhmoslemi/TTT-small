import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        # Convert to binary (0 or 1) and adjust to sum to n/2
        h_start = np.round(h_start)
        current_sum = np.sum(h_start)
        target_sum = n_points / 2
        if current_sum != target_sum:
            # Adjust by flipping bits
            diff = int(target_sum - current_sum)
            if diff > 0:
                # Need to set more 0s to 1s
                zero_indices = np.where(h_start == 0)[0]
                if len(zero_indices) >= diff:
                    for i in range(diff):
                        h_start[zero_indices[i]] = 1
            else:
                # Need to set more 1s to 0s
                one_indices = np.where(h_start == 1)[0]
                if len(one_indices) >= -diff:
                    for i in range(-diff):
                        h_start[one_indices[i]] = 0
    else:
        n_points = 35
        # Start with a more uniform binary distribution
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 10
    max_no_improvement = 100
    no_improvement_count = 0
    
    # Initial tabu list
    tabu_list = []
    
    def get_neighbors(h):
        """Generate neighbors by flipping one 1 and one 0"""
        neighbors = []
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            for i in ones[:20]:  # Limit for efficiency
                for j in zeros[:20]:
                    h_new = h.copy()
                    h_new[i] = 0
                    h_new[j] = 1
                    neighbors.append((h_new, (i, j)))
        return neighbors
    
    # Start tabu search
    current_h = best_h.copy()
    iterations = 0
    max_iter = int(budget_s * 50)  # Rough estimate
    
    while time() - start_time < budget_s * 0.95 and iterations < max_iter:
        # Get neighbors
        neighbors = get_neighbors(current_h)
        
        if not neighbors:
            break
            
        best_neighbor = None
        best_c5_neighbor = np.inf
        best_move = None
        
        for h_neighbor, move in neighbors:
            # Check tabu
            if move in tabu_list:
                # Consider aspiration: if better than best
                c5_val = evaluate_erdos_solution(h_neighbor, dx)
                if c5_val < best_c5_neighbor:
                    best_neighbor = h_neighbor
                    best_c5_neighbor = c5_val
                    best_move = move
            else:
                c5_val = evaluate_erdos_solution(h_neighbor, dx)
                if c5_val < best_c5_neighbor:
                    best_neighbor = h_neighbor
                    best_c5_neighbor = c5_val
                    best_move = move
        
        # If no neighbor found, break
        if best_neighbor is None:
            break
            
        # Update tabu list
        tabu_list.append(best_move)
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
            
        if best_c5_neighbor < best_c5:
            current_h = best_neighbor
            best_c5 = best_c5_neighbor
            best_h = current_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Shaking - random restart if no improvement for a while
        if no_improvement_count > 50:
            # Flip some bits to escape local minimum
            h_new = current_h.copy()
            num_flips = max(1, n_points // 20)
            flip_indices = np.random.choice(n_points, size=num_flips, replace=False)
            for idx in flip_indices:
                h_new[idx] = 1 - h_new[idx]
            
            # Maintain sum constraint
            ones_new = np.sum(h_new)
            if ones_new != n_points / 2:
                # Adjust to restore sum
                diff = int((n_points / 2) - ones_new)
                if diff > 0:
                    # Need to make more zeros to ones
                    zero_indices = np.where(h_new == 0)[0]
                    for i in range(diff):
                        if i < len(zero_indices):
                            h_new[zero_indices[i]] = 1
                else:
                    # Need to make more ones to zeros
                    one_indices = np.where(h_new == 1)[0]
                    for i in range(-diff):
                        if i < len(one_indices):
                            h_new[one_indices[i]] = 0
            
            current_h = h_new
            no_improvement_count = 0
            
        iterations += 1
        
        # Occasionally check for time limit
        if (iterations % 500) == 0 and time() - start_time > budget_s * 0.9:
            break

    # Final validation
    try:
        final_h = np.clip(best_h, 0, 1)
        final_h = final_h * (n_points/2) / np.sum(final_h)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h
    except:
        pass

    # Ensure binary result
    try:
        h_binary = np.round(best_h)
        h_binary = h_binary * (n_points/2) / np.sum(h_binary)
        final_c5 = evaluate_erdos_solution(h_binary, dx)
        if final_c5 < best_c5:
            best_h = h_binary
            best_c5 = final_c5
    except:
        pass

    return (list(best_h), best_c5, n_points)
