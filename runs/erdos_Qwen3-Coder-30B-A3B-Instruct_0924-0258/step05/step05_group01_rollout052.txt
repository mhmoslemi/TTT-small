import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def gaussian_smooth(x, sigma=0.5):
    """Apply Gaussian smoothing to reduce noise in gradients"""
    kernel = np.exp(-0.5 * (np.arange(-3*sigma, 3*sigma+1))**2 / sigma**2)
    kernel = kernel / np.sum(kernel)
    return np.convolve(x, kernel, mode='same')

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Triangular function as a good starting point
        x = np.linspace(0, 2, 35)
        h_start = 1 - np.abs(x - 1)
        h_start = np.clip(h_start, 0, 1)
        n_points = 35
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Scale_levels: increasing resolution
    scale_levels = [35, 50, 75, 100]
    current_scale_index = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale_index < len(scale_levels):
        scale = scale_levels[current_scale_index]
        
        # Interpolate to the target resolution
        if scale > n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, scale)
            h_interpolated = np.interp(x_new, x_orig, best_h)
            h_current = h_interpolated
            n_points = scale
        else:
            h_current = best_h.copy()
            n_points = scale
        
        dx = 2.0 / n_points
        
        # Gradient-based optimization
        max_iter = 100
        for iter in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute current C5
            c5_current = evaluate_erdos_solution(h_current, dx)
            
            # Find shift k that achieves max overlap (for gradient computation)
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_k = np.argmax(corr)
            max_overlap = corr[max_k] * dx
            
            # Compute gradient w.r.t. h at k such that max overlap is achieved
            # Since corr[i] = sum(h[j] * (1 - h[j + i - n+1])), we need to compute
            # d/dh_j of sum(h[j] * (1 - h[j+k])) = 1 - 2*h[j+k] if j+k < n
            # Ignore edge effects, may use finite differences for robustness
            
            # For k=0: d/dh_j(sum(h_j * (1 - h_j))) = 1 - 2*h_j
            # For k>=1, except edge effects:
            # d/dh_j(sum(h_j * (1 - h_{j+k}))) = -h_{j+k}
            
            grad = np.zeros(n_points)
            # The max overlap is at position max_k
            # When max_k < n, the correlation corresponds to a shift k = max_k - (n-1)
            if max_k >= n_points-1:
                actual_k = max_k - (n_points - 1)
            else:
                actual_k = max_k  # For small shifts
            # Estimate gradient based on "if" decompositions of correlation
            # Applying chain rule: we want gradient of correlation w.r.t. h
            # Let's use finite differences of correlation over h
            
            # Use perturbation around current solution
            epsilon = 1e-5
            grad_approx = np.zeros(n_points)
            
            # Simple finite difference approximation for gradient direction
            for i in range(n_points):
                h_pert = h_current.copy()
                h_pert[i] += epsilon
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points/2) / np.sum(h_pert)
                c5_pert = evaluate_erdos_solution(h_pert, dx)
                grad_approx[i] = (c5_pert - c5_current) / epsilon
            # Smooth gradient
            grad = gaussian_smooth(grad_approx, sigma=0.5)
            
            # Gradient descent step
            lr = 0.01
            h_new = h_current - lr * grad
            
            # Apply constraint check
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                h_current = h_new.copy()
            else:
                # Try a small random perturbation to escape local minima
                pert = np.random.normal(0, 0.01, n_points)
                h_pert = h_current + pert
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points/2) / np.sum(h_pert)
                c5_pert = evaluate_erdos_solution(h_pert, dx)
                if c5_pert < best_c5:
                    best_h = h_pert.copy()
                    best_c5 = c5_pert
                    h_current = h_pert.copy()
        
        current_scale_index += 1
    
    # Final validation
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_final.copy()
    
    return (list(best_h), best_c5, n_points)
