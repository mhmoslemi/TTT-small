import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy.spatial.distance import cdist

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Multi-start local optimization
    max_restarts = 50
    for restart in range(max_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation
        h_restart = h_start + np.random.normal(0, 0.05, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        # Local optimization
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res_opt = minimize(objective, h_restart, method='SLSQP', 
                            bounds=bounds, constraints=cons_sum, 
                            options={'maxiter': 300, 'ftol': 1e-9})
            if res_opt.success:
                h_refined = res_opt.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            continue
    
    # Additional refinement at higher resolution
    if time() - start_time < budget_s * 0.8:
        n_high = min(100, n_points + 20)
        h_interp = np.interp(np.linspace(0, 2, n_high), np.linspace(0, 2, n_points), best_h)
        h_interp = np.clip(h_interp, 0, 1)
        h_interp = h_interp * (n_high/2) / np.sum(h_interp)
        
        dx_high = 2.0 / n_high
        c5_high = evaluate_erdos_solution(h_interp, dx_high)
        if c5_high < best_c5:
            best_h = h_interp
            best_c5 = c5_high
            n_points = n_high
    
    # Final local refinement with trusted approach
    if time() - start_time < budget_s * 0.9:
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            res_opt = minimize(objective, best_h, method='SLSQP', 
                            bounds=bounds, constraints=cons_sum, 
                            options={'maxiter': 200, 'ftol': 1e-10})
            if res_opt.success:
                final_h = res_opt.x
                final_c5 = objective(final_h)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
