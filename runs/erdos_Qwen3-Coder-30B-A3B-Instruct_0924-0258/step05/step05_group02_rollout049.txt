import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with existing construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Normalize to sum to 17.5
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    h_start = np.clip(h_start, 0, 1)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse grid optimization
    if n_points > 20:
        h_coarse = np.interp(np.linspace(0, 1, 20), np.linspace(0, 1, n_points), best_h)
        h_coarse = h_coarse * 10 / np.sum(h_coarse)
        h_coarse = np.clip(h_coarse, 0, 1)
        # Optimize coarse grid
        def objective_coarse(h_coarse_flat):
            h_coarse_full = h_coarse_flat
            h_coarse_full = h_coarse_full * 10 / np.sum(h_coarse_full)
            h_coarse_full = np.clip(h_coarse_full, 0, 1)
            return evaluate_erdos_solution(h_coarse_full, 2.0/20)
        
        bounds_coarse = [(0, 1) for _ in range(20)]
        res_coarse = minimize(objective_coarse, h_coarse, method='L-BFGS-B', bounds=bounds_coarse, options={'maxiter': 100})
        if res_coarse.success:
            h_coarse = res_coarse.x
            h_coarse = h_coarse * 10 / np.sum(h_coarse)
            h_coarse = np.clip(h_coarse, 0, 1)
            # Interpolate back to fine grid
            h_fine = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, 20), h_coarse)
            h_fine = h_fine * (n_points/2) / np.sum(h_fine)
            h_fine = np.clip(h_fine, 0, 1)
            best_h = h_fine.copy()
            best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Iterative refinement with constraint tightening
    max_iter = 100
    iter_count = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Get current maximum overlap
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        max_k = max_idx - (n_points - 1)  # shift index
        max_overlap = corr[max_idx] * dx
        
        # If no improvement based on recent iterations, perturb
        if no_improvement_count > 50:
            h_perturbed = best_h + np.random.normal(0, 0.05, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            # Try to refine using optimization with tightened constraint
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            def constraint_overlap(h, shift_k):
                # For the specific shift k, check if overlap constraint holds
                if shift_k >= 0:
                    corr_sum = np.sum(h[:n_points-shift_k] * (1 - h[shift_k:]))
                else:
                    corr_sum = np.sum(h[-shift_k:] * (1 - h[:n_points+shift_k]))
                return corr_sum * dx - best_c5 + 1e-6
                
            # Check entire range of shifts to tighten
            best_shift = None
            max_corr_val = 0
            for k in range(1, n_points):
                corr_val = 0
                if k < n_points:
                    corr_val = np.sum(best_h[:n_points-k] * (1 - best_h[k:]))
                elif k == n_points:
                    corr_val = 0
                else:
                    corr_val = np.sum(best_h[k-n_points:] * (1 - best_h[:n_points-(k-n_points)]))
                if corr_val > max_corr_val:
                    max_corr_val = corr_val
                    best_shift = k
                    
            if best_shift is not None:
                # Add tight constraint for the worst shift
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                # Try optimization with tightened constraint
                try:
                    res = minimize(objective, best_h.copy(), method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 50, 'ftol': 1e-7})
                    if res.success:
                        h_new = res.x
                        h_new = h_new * (n_points/2) / np.sum(h_new)
                        h_new = np.clip(h_new, 0, 1)
                        c5_new = evaluate_erdos_solution(h_new, dx)
                        if c5_new < best_c5:
                            best_h = h_new
                            best_c5 = c5_new
                            no_improvement_count = 0
                        else:
                            no_improvement_count += 1
                    else:
                        no_improvement_count += 1
                except Exception:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
        iter_count += 1
    
    # Final normalization
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), c5_final, n_points)
