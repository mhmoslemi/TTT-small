import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid optimization
    n_low = 20
    dx_low = 2.0 / n_low
    
    # First, try coarse grid solution
    try:
        h_low = cp.Variable(n_low)
        constraints = [
            cp.sum(h_low) == n_low / 2.0,
            h_low >= 0,
            h_low <= 1
        ]
        
        # Focus on first few shifts for tractability
        for k in range(1, min(8, n_low)):  
            if k < n_low // 2:
                corr_sum = cp.sum(cp.multiply(h_low[:n_low-k], 1 - h_low[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_low[k:], 1 - h_low[:n_low-k]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_low)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_low.value
            h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                                   np.linspace(0, 2, n_low), h_feasible)
            h_upsampled = np.clip(h_upsampled, 0, 1)
            h_upsampled = h_upsampled * (n_points/2) / np.sum(h_upsampled)
            c5_upsampled = evaluate_erdos_solution(h_upsampled, dx)
            if c5_upsampled < best_c5:
                best_h = h_upsampled
                best_c5 = c5_upsampled
    except Exception:
        pass
    
    # Fine tune with local search
    c5_prev = best_c5 + 1
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95:
        # Perturb
        h_perturbed = best_h.copy()
        h_perturbed += np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Local optimization
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                objective,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                options={'maxiter': 50, 'ftol': 1e-7}
            )
            
            if res.success:
                h_opt = res.x
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
        except:
            no_improvement_count += 1

        # Periodic mutation to escape local optima
        if no_improvement_count > 50:
            # Adaptive mutation based on gradient magnitude
            h_mutated = best_h.copy()
            mutation_strength = 0.05
            for i in range(n_points):
                if np.random.rand() < 0.1:  # 10% chance to mutate
                    h_mutated[i] = np.clip(h_mutated[i] + np.random.normal(0, mutation_strength), 0, 1)
            
            # Normalize
            h_mutated = h_mutated * (n_points/2) / np.sum(h_mutated)
            
            c5_mutated = evaluate_erdos_solution(h_mutated, dx)
            if c5_mutated < best_c5:
                best_h = h_mutated
                best_c5 = c5_mutated
            no_improvement_count = 0

        if abs(c5_prev - best_c5) < 1e-6:
            if no_improvement_count > 100:
                break
        c5_prev = best_c5
    
    # Final fine-tune
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda h: evaluate_erdos_solution(h, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if res.success:
            best_h = res.x
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Return best solution found
    return (list(best_h), final_c5, n_points)
