import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Get initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm with Gradient Descent refinement
    pop_size = 50
    generations = 1000
    elitism = int(0.3 * pop_size)
    mutation_rate = 0.1
    max_eval = 1000
    
    # Initial population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
    
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    eval_count = len(population)
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append((c5, h))
        
        fitness.sort(key=lambda x: x[0])
        
        # Update best solution
        if fitness[0][0] < best_c5:
            best_c5 = fitness[0][0]
            best_h = fitness[0][1].copy()
        
        # Selection - top elitism
        elite = [h for _, h in fitness[:elitism]]
        
        # Create new population
        new_pop = elite.copy()
        
        # Crossover and mutation
        while len(new_pop) < pop_size:
            if len(elite) >= 2:
                parent1, parent2 = random.sample(elite, 2)
                # Blend crossover
                alpha = random.uniform(0.4, 0.6)
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = np.random.rand(n_points)
                child = child * (n_points/2) / np.sum(child)
                child = np.clip(child, 0, 1)
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += random.gauss(0, 0.02)
            
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            new_pop.append(child)
        
        population = new_pop
        eval_count += len(new_pop)
        
        # Gradient descent refinement on best individuals
        if eval_count > max_eval:
            for i in range(min(10, len(population))):
                h_refined = population[i].copy()
                
                def obj_func(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                try:
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    bounds = [(0, 1) for _ in range(n_points)]
                    res = minimize(
                        obj_func, 
                        h_refined, 
                        method='SLSQP', 
                        bounds=bounds, 
                        constraints=cons_sum,
                        options={'maxiter': 50, 'ftol': 1e-7}
                    )
                    if res.success:
                        h_fine = res.x
                        c5_fine = evaluate_erdos_solution(h_fine, dx)
                        if c5_fine < best_c5:
                            best_c5 = c5_fine
                            best_h = h_fine
                except Exception:
                    pass
                
                eval_count += 1
                if time() - start_time > budget_s * 0.95:
                    break
            
            if time() - start_time > budget_s * 0.95:
                break
    
    # Final check
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final
    
    return (list(best_h), best_c5, n_points)
