import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use CVXPY for convex subproblems
    max_iter = 500
    rho = 1e3
    lambda_init = 0.1
    mu = 1.0
    
    # Precompute correlation matrices for all shifts k
    # For a signal of length n, full correlation produces 2n-1 values
    # The shift indices for the full correlation are from -(n-1) to (n-1)
    shift_indices = np.arange(-(n_points - 1), n_points)
    A_matrices = {}
    
    for k in shift_indices:
        # Create shift matrix A_k such that (A_k * h)_i = h[i-k] for valid k
        A_k = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = i - k
            if 0 <= j < n_points:
                A_k[i, j] = 1
        A_matrices[k] = A_k
    
    # Iterate using Lagrangian relaxation
    for iteration in range(max_iter):
        if time() - start_time >= max_time * 0.95:
            break
            
        # Create CVXOPT problem for convex relaxation
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum(cp.multiply(h_var, 1 - h_var)))
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add Lagrangian terms
        # For simplicity, we consider only one constraint at a time
        # Implement a basic adaptive method using convex relaxation
        
        # Fixed iterations with small number of steps
        try:
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.OSQP, verbose=False)
            h_new = h_var.value
            
            # Ensure feasibility
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
        except Exception as e:
            pass
            
        # Local refinement using SLSQP
        if iteration % 20 == 0:
            h_init = best_h.copy()
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            def con(h):
                return np.sum(h) - n_points / 2.0
                
            cons = {'type': 'eq', 'fun': con}
            bounds = [(0, 1) for _ in range(n_points)]
            try:
                res = minimize(obj, h_init, method='SLSQP', bounds=bounds, 
                               constraints=cons, options={'maxiter': 50, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = obj(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    # Final local refinement with tighter tolerance
    remaining_time = max_time - (time() - start_time)
    
    if remaining_time > 10:
        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
