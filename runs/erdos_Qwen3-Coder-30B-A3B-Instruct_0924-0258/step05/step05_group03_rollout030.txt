import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 35
        h0 = np.ones(n_points) * 0.5
    
    # Scale to correct integral
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Shifts to sample
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    
    # Convex relaxation approach
    h = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
        C5 >= 0
    ]
    
    # Add constraints for each shift
    max_iter = 50
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    iteration = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current h and get worst shift constraint
        corr = np.correlate(best_h, 1 - best_h, mode='full') * (2.0 / n_points)
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        
        # Add constraint
        h_shifted = np.roll(1 - best_h, -max_idx)
        constraint_expr = cp.sum(cp.multiply(h, h_shifted)) <= C5 * (2.0 / n_points)
        constraints.append(constraint_expr)
        
        # Solve convex relaxation
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_val = h.value
                c5_val = C5.value
                if c5_val < best_c5:
                    best_h = h_val.copy()
                    best_c5 = c5_val
            else:
                # Fallback to local search on current point
                pass
        except Exception:
            pass
            
        # Final polish with local method
        try:
            def obj(x):
                return evaluate_erdos_solution(x, 2.0 / n_points)
            def constraint_sum(x):
                return np.sum(x) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(obj, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                best_h = res.x
                best_c5 = obj(best_h)
        except:
            pass

        # Early termination check
        if best_c5 < 0.3807:
            break
            
        # Remove first constraint to avoid degeneracy
        if len(constraints) > 4:
            constraints.pop(0)
    
    # Final local refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
