import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Initial evaluation
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # CCP Iteration
    max_iterations = 50
    stagnation_count = 0
    last_best = current_c5
    
    # Sample shifts for inner loop search
    shift_samples = np.linspace(0, 2, 100)
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Inner loop: Find the worst k for current h
        best_k = 0
        max_overlap = -1
        for k in shift_samples:
            # Compute overlap for shift k
            k_index = int(k / 2 * n_points)
            overlap = 0
            for i in range(n_points):
                j = (i + k_index) % n_points
                overlap += h[i] * (1 - h[j])
            overlap *= dx
            
            if overlap > max_overlap:
                max_overlap = overlap
                best_k = k
        
        # Outer loop: Use convex optimization to improve h
        # Construct and solve convex subproblem via CVXPY
        try:
            h_var = Variable(n_points)
            # Objective: maximize the overlap for current best_k
            k_index = int(best_k / 2 * n_points)
            objective = 0
            for i in range(n_points):
                j = (i + k_index) % n_points
                objective += h_var[i] * (1 - h_var[j])
            
            objective *= dx
            
            constraints = [
                h_var >= 0,
                h_var <= 1,
                sum(h_var) == n_points / 2
            ]
            
            prob = cp.Problem(cp.Minimize(objective), constraints)
            prob.solve(solver=ECOS, verbose=False)
            
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                new_h = np.array(h_var.value).flatten()
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                new_c5 = evaluate_erdos_solution(new_h, dx)
                
                if new_c5 < current_c5:
                    h = new_h
                    current_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
                
        except Exception as e:
            stagnation_count += 1
        
        # Perturbation on stagnation
        if stagnation_count >= 5:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        if best_c5 < 0.3807:
            break
    
    # Final refinement with scipy
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
