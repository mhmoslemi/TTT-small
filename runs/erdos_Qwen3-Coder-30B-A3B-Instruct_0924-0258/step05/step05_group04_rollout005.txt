import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Smooth approximation parameters
    tau = 1.0
    max_iter = 3000
    tol = 1e-7
    history_size = 10
    patience = 100
    patience_count = 0
    
    # Function to compute smooth max
    def smooth_max(values, tau):
        return np.log(np.sum(np.exp(tau * values))) / tau
    
    # Function to compute gradient of smooth max
    def smooth_max_grad(values, tau):
        exp_vals = np.exp(tau * values)
        sum_exp = np.sum(exp_vals)
        return exp_vals / (sum_exp * tau)
    
    # Optimize using L-BFGS-B
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Compute all correlations efficiently
        h_flip = 1 - h_current
        corr_full = np.correlate(h_current, h_flip, mode='full')
        # Take the relevant part with max overlap
        corr_vals = corr_full[len(corr_full)//2:]
        
        # Compute smooth maximum and gradient
        smooth_c5 = smooth_max(corr_vals, tau)
        grad_smooth = smooth_max_grad(corr_vals, tau)
        
        # Density of one shift is 1/dx
        # We need to map the gradient back to h space
        # This approach is a simplification that assumes the shifts are uniform 
        # and simply reshapes the gradient contributions back to the step function
        grad_h_full = np.zeros_like(h_current)
        
        for i in range(len(h_current)):
            for k in range(len(h_current)):
                if k < len(grad_smooth):
                    # Simulate how many times shifted value contributes
                    if 0 <= i + k < len(h_current):
                        grad_h_full[i] += grad_smooth[k]
        
        # Use this gradient to update
        try:
            # Apply optimization with correct bounds
            bounds = [(0, 1) for _ in range(len(h_current))]
            res = minimize(lambda x: smooth_max(np.correlate(x, 1-x, mode='full')[len(h_current)//2:], tau),
                           h_current, method='L-BFGS-B',
                           bounds=bounds, jac=lambda x: grad_h_full,
                           options={'maxiter': 5, 'ftol': tol})
            if res.success:
                h_new = res.x
                # Renormalize to sum constraint
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h_current = np.clip(h_new, 0, 1)
            else:
                h_current = np.clip(h_current - 1e-4 * grad_h_full, 0, 1)
                h_current = h_current * (n_points / 2) / np.sum(h_current)
        except:
            h_current = np.clip(h_current - 1e-4 * grad_h_full, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            
        # Evaluate new solution and update best
        current_c5 = evaluate_erdos_solution(h_current, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            patience_count = 0
        else:
            patience_count += 1
            
        # Adaptive adjustments
        if patience_count > 100:
            tau *= 1.1
            patience_count = 0
            
        if iteration % 100 == 0:
            tau *= 0.99
            
        if current_c5 < 0.3807:
            break
            
    # Final refinement with more precise method
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
                
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h.copy(), method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass

    return (list(best_h), best_c5, n_points)
