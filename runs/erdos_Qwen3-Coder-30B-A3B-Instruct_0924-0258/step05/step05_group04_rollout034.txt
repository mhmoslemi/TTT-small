import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def smooth_max(h, dx, tau=1.0):
    n = len(h)
    total = 0.0
    for k in range(1, 2*n - 1):
        if k < n:
            corr = np.sum(h[:k] * (1 - h[n-k:]))
        else:
            corr = np.sum(h[k-n+1:] * (1 - h[:2*n-k-1]))
        total += np.exp(tau * corr * dx)
    return np.log(total) / tau

def gradient_smooth_max(h, dx, tau=1.0):
    n = len(h)
    grad = np.zeros(n)
    exp_vals = np.zeros(2*n-1)
    total = 0.0
    
    for k in range(1, 2*n - 1):
        if k < n:
            corr = np.sum(h[:k] * (1 - h[n-k:]))
        else:
            corr = np.sum(h[k-n+1:] * (1 - h[:2*n-k-1]))
        exp_val = np.exp(tau * corr * dx)
        exp_vals[k] = exp_val
        total += exp_val
    
    for i in range(n):
        term = 0.0
        for k in range(1, 2*n - 1):
            if k < n:
                if i < k:
                    term += exp_vals[k] * (-dx * tau * (1 - h[n-k+i]))
                if i >= n-k:
                    term += exp_vals[k] * (dx * tau * h[i-(n-k)])
            else:
                if i >= k-n+1:
                    term += exp_vals[k] * (dx * tau * h[n-k+i])
                if i < 2*n-k-1:
                    term += exp_vals[k] * (-dx * tau * (1 - h[i]))
        grad[i] = term / (total * tau)
    
    return grad

    h = h_start.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    # Coarse optimization first
    tau = 1.0
    max_iter = 3000
    tol = 1e-7
    step_size = 1.0
    
    for i in range(max_iter):
        if time() - start_time >= max_time * 0.95:
            break
            
        # Compute smooth max and gradient
        smooth_c5 = smooth_max(h, dx, tau)
        grad = gradient_smooth_max(h, dx, tau)
        
        # Optimization step
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        # Update h
        h = h_new
        
        # Adaptive adjustments
        if i % 100 == 0:

        if i % 500 == 0 and i > 0:
