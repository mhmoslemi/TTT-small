import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid optimization to get a good structure
    n_coarse = 20
    h_coarse = np.ones(n_coarse) * 0.5
    dx_coarse = 2.0 / n_coarse
    
    def coarse_objective(h_coarse):
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, n_coarse), h_coarse)
        h_interp = h_interp * (n_points/2) / np.sum(h_interp)
        return evaluate_erdos_solution(h_interp, dx)
    
    # We use a simpler approach with less complex constraints
    max_restarts = 20
    restart_count = 0
    no_improvement_count = 0
    
    # Smooth initial solution
    h_current = best_h.copy()
    
    while time() - start_time < budget_s * 0.9:
        # Simple noise perturbation with random walk
        perturbation = np.random.normal(0, 0.03, n_points)
        h_new = h_current + perturbation
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Local refinement using a more focused optimization
        if no_improvement_count > 5:
            # Use trust-constr with simpler constraints for refinement
            h_refined = h_current.copy()
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            
            cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(
                    lambda h: evaluate_erdos_solution(h, dx), 
                    h_refined, 
                    method='trust-constr', 
                    bounds=bounds, 
                    constraints=cons_sum,
                    options={'maxiter': 50, 'xtol': 1e-6}
                )
                if res.success:
                    h_refined = res.x
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        no_improvement_count = 0
            except Exception:
                pass
            
            no_improvement_count = 0
            
        h_current = best_h.copy()
        
        if no_improvement_count > max_restarts:
            # Global restart
            h_current = np.random.rand(n_points)
            h_current = h_current * (n_points/2) / np.sum(h_current)
            h_current = np.clip(h_current, 0, 1)
            no_improvement_count = 0
            
        if time() - start_time > budget_s * 0.95:
            break
    
    # Final refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
