```python
import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
import warnings
warnings.filterwarnings("ignore")

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initial candidates for GP
    initial_candidates = []
    initial_values = []
    
    # Add parent if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points/2) / np.sum(h_parent)
        initial_candidates.append(h_parent)
        initial_values.append(evaluate_erdos_solution(h_parent, dx))
    
    # Add random candidates
    n_random = 20
    for _ in range(n_random):
        h_random = np.random.rand(n_points)
        h_random = np.clip(h_random, 0, 1)
        h_random = h_random * (n_points/2) / np.sum(h_random)
        initial_candidates.append(h_random)
        initial_values.append(evaluate_erdos_solution(h_random, dx))
    
    # Training data for GP
    X_train = np.array(initial_candidates)
    y_train = np.array(initial_values)
    
    # Define GP kernel
    kernel = C(1.0, (1e-3, 1e3)) * RBF(1.0, (1e-2, 1e2))
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-6, n_restarts_optimizer=5, 
                                  normalize_y=True, random_state=seed)
    gp.fit(X_train, y_train)
    
    n_evaluations = len(initial_candidates)
    
    # Optimization loop
    max_iterations = 1000
    improvement_threshold = 1e-5
    improvement_count = 0
    
    def project_h(h):
        h_clipped = np.clip(h, 0, 1)
        h_scaled = h_clipped * (n_points/2) / np.sum(h_clipped)
        return h_scaled
    
    while time() - start_time < budget_s * 0.95 and n_evaluations < max_iterations:
        # Generate candidates using GP
        n_suggested = 50
        candidates = []
        
        for _ in range(n_suggested):
            h_suggested = np.random.rand(n_points)
            h_suggested = project_h(h_suggested)
            candidates.append(h_suggested)
        
        # Use GP to predict and sort
        X_suggest = np.array(candidates)
        mean_pred, std_pred = gp.predict(X_suggest, return_std=True)
        # Sort by predicted C5 and uncertainty (lower C5 + higher uncertainty)
        indices = np.argsort(mean_pred + 0.1 * std_pred)
        
        # Evaluate top candidates
        top_candidates = []
        top_values = []
        
        for i in range(min(10, len(indices))):
            idx = indices[i]
            h_candidate = candidates[idx]
            h_projected = project_h(h_candidate)
            c5_candidate = evaluate_erdos_solution(h_projected, dx)
            top_candidates.append(h_projected)
            top_values.append(c5_candidate)
        
        # Update best solution
        for h_test, c5_test in zip(top_candidates, top_values):
            if c5_test < best_c5:
                best_h = h_test.copy()
                best_c5 = c5_test
                improvement_count = 0
            else:
                improvement_count += 1
        
        # Add new data to training set
        for h_new, c5_new in zip(top_candidates, top_values):
            if len(X_train) < 200:  # Prevent data explosion
                X_train = np.vstack([X_train, h_new])
                y_train = np.append(y_train, c5_new)
        
        # Retrain GP
        try:
            gp.fit(X_train, y_train)
        except:
            pass
            
        n_evaluations += len(top_candidates)
        
        # Resolution upgrade
        if n_evaluations % 50 == 0 and n_points < 100:
            n_points += 5
            dx = 2.0 / n_points
            # Interpolate best solution
            if len(best_h) < n_points:
                old_x = np.linspace(0, 2, len(best_h))
                new_x = np.linspace(0, 2, n_points)
                best_h = np.interp(new_x, old_x, best_h)
            else:
                best_h = best_h[:n_points]
            # Rescale
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
```
