import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use SDP relaxation approach for better starting point
    def sdp_initialization(h0, n_points, dx):
        # SDP relaxation approach
        try:
            # Create decision variables for the lifted matrix
            X = cp.Variable((n_points, n_points), symmetric=True)
            
            # Constraints
            constraints = [
                cp.trace(X) == (n_points/2)**2 * dx,
                X >> 0,  # Positive semidefinite
            ]
            
            # Add bound constraints to X
            for i in range(n_points):
                for j in range(n_points):
                    if i == j:
                        constraints.append(X[i, j] <= h0[i])
                    else:
                        constraints.append(X[i, j] <= min(h0[i], h0[j]))
            
            # Objective: minimize the maximum overlap
            # This requires enumerating all shifts, which is complex.
            # Instead we solve a simplified version or use heuristic.
            # For this implementation, we'll use a local refinement approach.
            return h0  # fallback to initial if SDP not working
        except:
            return h0  # Fallback
    
    # Keep initial for now - we'll refine globally
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Global search with local refinement
    max_iter = 1000
    tol = 1e-9
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Local optimization step using L-BFGS-B
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_current, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons,

            if res.success:
                h_new = res.x
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                c5_new = objective(h_new)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    if best_c5 < 0.3807:
                        break
                h_current = h_new
            else:
                # If local optimization fails, try perturbation
                h_perturb = h_current + np.random.normal(0, 0.01, n_points)
                h_perturb = np.clip(h_perturb, 0, 1)
                h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
                c5_perturb = objective(h_perturb)
                if c5_perturb < best_c5:

                    best_c5 = c5_perturb
                    if best_c5 < 0.3807:

            h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
            c5_perturb = objective(h_perturb)

            return evaluate_erdos_solution(h, dx)

                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
