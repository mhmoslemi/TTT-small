import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use provided initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Start with a lower resolution grid to avoid local optima
    n_low = 20
    dx_low = 2.0 / n_low
    
    # Initialize low-resolution step function
    h_low = np.ones(n_low) * 0.5
    h_low = h_low * (n_low/2) / np.sum(h_low)
    
    # Try to solve with convex relaxation for coarse grid
    try:
        h_relaxed = cp.Variable(n_low)
        constraints = [
            cp.sum(h_relaxed) == n_low / 2.0,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        
        # Add a manageable subset of overlap constraints
        max_shift = min(10, n_low // 2)
        for k in range(1, max_shift):
            if k < n_low // 2:
                corr_sum = cp.sum(cp.multiply(h_relaxed[:n_low-k], 1 - h_relaxed[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_low-k]))
            constraints.append(corr_sum <= best_c5 * dx_low + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        
        if prob.status == cp.OPTIMAL:
            h_low = h_relaxed.value
    except Exception:
        pass
    
    # Interpolate to target resolution
    h_target = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_low), h_low)
    h_target = h_target * (n_points/2) / np.sum(h_target)
    h_target = np.clip(h_target, 0, 1)
    
    best_h = h_target.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Fine grained refinement using SLSQP
    max_restarts = 50
    restart_count = 0
    
    while time() - start_time < budget_s * 0.95:
        # Local optimization with SLSQP
        try:
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                best_h.copy(), 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-7}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
                else:
                    restart_count += 1
            else:
                restart_count += 1
        except Exception:
            restart_count += 1
        
        # Restart with perturbation if needed
        if restart_count > max_restarts:
            h_perturb = best_h + np.random.normal(0, 0.05, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
            best_h = h_perturb
            restart_count = 0
            
        if time() - start_time > budget_s * 0.9:
            break
            
    # Ensure final solution is properly constrained
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
