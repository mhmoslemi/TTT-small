def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale gradient-based optimization
    scale_levels = [35, 50, 75, 100]  # Resolutions to try
    max_iter_per_level = 100  # Iterations per scale
    
    # Sample initial points for gradient estimation
    def estimate_gradient(h, k, dx):
        # Compute correlation at shift k to get the gradient w.r.t. elements
        # We'll use finite differences in a simplified manner
        # The gradient of C5 with respect to h at position i involves terms
        # from the convolution. We'll compute a finite difference approximation
        h_plus = h.copy()
        h_minus = h.copy()
        eps = 1e-6
        h_plus[k] += eps
        h_minus[k] -= eps
        c_plus = evaluate_erdos_solution(h_plus, dx)
        c_minus = evaluate_erdos_solution(h_minus, dx)
        return (c_plus - c_minus) / (2 * eps)
    
    for scale in scale_levels:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to target scale
        if scale > n_points:
            h_current = np.interp(np.linspace(0, 2, scale), np.linspace(0, 2, n_points), best_h)
            n_points = scale
        else:
            h_current = best_h.copy()
            n_points = scale
            
        dx = 2.0 / n_points
        
        # Gradient descent optimization at current level
        lr = 0.01
        for iteration in range(max_iter_per_level):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute the full correlation to find which shift is maximized
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            max_shift = max_idx - (len(h_current) - 1)
            
            # Build the gradient analytically for the respective shift
            # We approximate by finite differences at key points
            gradient = np.zeros_like(h_current)
            eps = 1e-6
            
            # For each element, we evaluate how changing it affects max correlation
            for i in range(n_points):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[i] = min(1.0, h_plus[i] + eps)
                h_minus[i] = max(0.0, h_minus[i] - eps)
                
                # Recompute correlation
                corr_plus = np.correlate(h_plus, 1-h_plus, mode='full')
                corr_minus = np.correlate(h_minus, 1-h_minus, mode='full')
                
                grad_at_i = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
                gradient[i] = grad_at_i
                
            # Stochastic gradient update
            h_new = h_current - lr * gradient
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Test new point
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                lr = min(0.1, lr * 1.05)  # Adaptive learning rate
            else:
                lr = max(1e-6, lr * 0.99)  # Reduce LR if no improvement
            
            h_current = h_new
            
    # Final refinement
    if time() - start_time < budget_s * 0.1:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
