import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Read initial values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Sequential Convex Programming approach
    max_iter = 500
    mu = 1e-3
    C5_nih = 0.4
    
    # List to store violations
    violated_indices = []
    
    for it in range(max_iter):
        current_time = time()
        if current_time - start_time > max_time * 0.95:
            break
            
        # Calculate all overlaps
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        overlaps = corr * dx
        
        # Identify maximum overlap
        max_overlap = np.max(overlaps)
        
        # Update best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = best_h.copy()
            
        # Early stopping if target met
        if best_c5 < 0.3807:
            break
            
        # Check which constraints are violated
        violated_indices = np.where(overlaps > C5_nih)[0]
        
        # Sequential Convex Programming step
        # We approximate the non-convex constraints with linear ones
        h = cp.Variable(n_points)
        objective = cp.Minimize(C5_nih)
        
        constraints = []
        # Add bounds
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        constraints.append(cp.sum(h) == n_points / 2.0)
        
        # Add quadratic constraints from overlap
        for i in range(len(overlaps)):
            if overlaps[i] > C5_nih:
                # Linearize at the current h
                shift = i - (len(overlaps) // 2)
                # Mask for convolution with shifted h
                mask = np.zeros(n_points)
                if shift >= 0:
                    mask[shift:] = best_h[:n_points-shift]
                else:
                    mask[:shift] = best_h[-shift:]
                
                # Linearized constraint 
                linear_term = np.sum(mask * (1 - best_h)) * dx
                # Inequality: (linear_term) <= C5_nih
                constraints.append(cp.sum(h * (1 - best_h)) * dx <= C5_nih + mu * (overlaps[i] - C5_nih))
        
        # Create and solve QP
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.OSQP, max_iter=5000, eps_abs=1e-6, eps_rel=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h.value
                if h_new is not None:
                    best_h = np.clip(h_new, 0, 1)
                    # Normalize to sum to n_points/2
                    best_h = best_h * (n_points / 2) / np.sum(best_h)
        except Exception:
            pass
            
        # Update penalty parameter if needed
        if len(violated_indices) > 0:
            mu *= 2
        else:
            mu = max(1e-6, mu * 0.8)
            
        # Update C5_nih
        C5_nih = min(C5_nih * 0.999, best_c5)
        
    # Final local refinement
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
