import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization
    resolutions = [35, 40, 50, 60]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Iterative improvement using cutting-plane method
        max_iterations = 200
        for i in range(max_iterations):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute current overlap
            corr = np.correlate(h_current, 1 - h_current, mode='full') * dx
            max_idx = np.argmax(corr)
            max_k = max_idx - (n_points - 1)
            
            # If we already have a better solution, stop
            current_c5 = corr[max_idx]
            if current_c5 >= best_c5 * 0.999:
                break
            
            # Add a constraint for the worst overlap
            try:
                # Create a relaxed convex problem with the constraint
                h_var = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_var) == n_points / 2.0,
                    h_var >= 0,
                    h_var <= 1
                ]
                
                # Add the overlap constraint for max_k
                if max_k >= 0:
                    overlap_sum = cp.sum(cp.multiply(h_var[:n_points-max_k], 1 - h_var[max_k:]))
                else:
                    overlap_sum = cp.sum(cp.multiply(h_var[-max_k:], 1 - h_var[:n_points+max_k]))
                constraints.append(overlap_sum <= current_c5 * dx + 1e-8)
                
                # Solve convex relaxation
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6)
                
                if prob.status == cp.OPTIMAL:
                    h_new = h_var.value
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points/2) / np.sum(h_new)
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
                        h_current = h_new.copy()
            except:
                pass
            
            # Local SLSQP optimization with restarts
            try:
                restart_count = 0
                max_restarts = 5
                while restart_count < max_restarts:
                    h_restart = np.clip(h_current + np.random.normal(0, 0.01, n_points), 0, 1)
                    h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                    
                    def objective(h):
                        return evaluate_erdos_solution(h, dx)
                    
                    def constraint_sum(h):
                        return np.sum(h) - n_points / 2.0
                    
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    bounds = [(0, 1) for _ in range(n_points)]
                    res = minimize(objective, h_restart, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-8})
                    
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                            h_current = h_refined.copy()
                            restart_count = 0
                        else:
                            restart_count += 1
                    else:
                        restart_count += 1
            except:
                restart_count += 1
    
    # Final refinement
    try:
        h_final = np.clip(best_h, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
