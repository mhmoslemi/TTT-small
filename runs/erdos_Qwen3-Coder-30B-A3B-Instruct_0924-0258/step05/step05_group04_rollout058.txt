import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Smooth approximation parameters
    tau = 1.0
    max_iter = 3000
    tol = 1e-7
    step_size = 1.0
    
    def smooth_max_overlap(h, tau):
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        # Smooth maximum approximation
        smooth_max = np.log(np.sum(np.exp(tau * corr))) / tau
        return smooth_max * dx
    
    def smooth_max_overlap_grad(h, tau):
        # Compute correlation and its gradient
        n = len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        exp_tau_corr = np.exp(tau * corr)
        sum_exp = np.sum(exp_tau_corr)
        
        # Gradient of smooth maximum
        grad = np.zeros(n)
        for i in range(n):
            # Compute d/dh_i of sum_k exp(tau * h_k * (1-h_{k+i}))
            gi = 0.0
            for j in range(n):
                k = j - i
                if 0 <= k < n:
                    gi += exp_tau_corr[n + k - 1] * (1 - h[j]) * tau
            grad[i] = gi / sum_exp
            
        return grad * dx
    
    # Optimizer loop
    prev_c5 = float('inf')
    stagnation_count = 0
    
    for i in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        # Smooth objective
        def obj(h):
            return smooth_max_overlap(h, tau)
            
        # Gradient
        def grad(h):
            return smooth_max_overlap_grad(h, tau)
        
        # Update h
        try:
            # Use L-BFGS-B with bounds
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, best_h, method='L-BFGS-B', jac=grad, bounds=bounds, 
                           options={'maxiter': 5, 'ftol': tol, 'gtol': 1e-6})
            if res.success:
                h_new = res.x
            else:
                # Fall back to projected gradient descent
                h_new = best_h - step_size * grad(best_h)
                # Project to bounds
                h_new = np.clip(h_new, 0, 1)
        except:
            # Fallback handling
            h_new = best_h - step_size * grad(best_h)
            h_new = np.clip(h_new, 0, 1)
            
        # Enforce the sum constraint
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
            
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1

        # Adaptive adjustments
        if i % 100 == 0 and i > 0:
            tau *= 0.99
            if stagnation_count > 50:
                tau *= 1.1
                stagnation_count = 0
                
        if stagnation_count > 100:
            step_size *= 0.8
            stagnation_count = 0
            
        if np.abs(prev_c5 - c5_new) < 1e-8:
            break
        prev_c5 = c5_new 
    
    # Final polishing
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 50:
        def obj_final(h):
            return evaluate_erdos_solution(h, dx)
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(obj_final, best_h, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = obj_final(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass

    return (list(best_h), best_c5, n_points)
