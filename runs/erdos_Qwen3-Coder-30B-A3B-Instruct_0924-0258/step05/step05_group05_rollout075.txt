import numpy as np
from scipy.optimize import minimize
from time import time
import hashlib

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    max_iter = 1000
    max_stagnation = 100
    mutation_scale = 0.04
    tabu_tenure = 50
    tabu_list = []
    tabu_keys = set()
    stagnation_counter = 0
    
    # Initialize
    current_h = h_start.copy()
    current_c5 = best_c5
    
    def hash_h(h_array):
        # Normalize and sort
        sorted_h = np.sort(h_array)
        return hashlib.md5(sorted_h.tobytes()).hexdigest()

    def perturb_h(h, scale, n_perturb=0):
        if n_perturb == 0:
            n_perturb = max(1, int(0.3 * len(h)))
        h_new = h.copy()
        indices = np.random.choice(len(h), n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, scale)
        h_new = np.clip(h_new, 0, 1)
        # Rescale to desired sum
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by perturbation
        candidate_h = perturb_h(current_h, mutation_scale)
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Check tabu list
        candidate_key = hash_h(candidate_h)
        is_tabu = candidate_key in tabu_keys
        
        # Accept if better than current or not tabu
        if candidate_c5 < current_c5 or not is_tabu:
            current_h = candidate_h
            current_c5 = candidate_c5
            
            # Update tabu
            tabu_list.append(candidate_key)
            tabu_keys.add(candidate_key)
            if len(tabu_list) > tabu_tenure:
                old_key = tabu_list.pop(0)
                tabu_keys.discard(old_key)
                
            # Update best if improved
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                stagnation_counter = 0
                if best_c5 < 0.3807:
                    break
            else:
                stagnation_counter += 1
        else:
            stagnation_counter += 1
        
        # Adaptive mutation
        if stagnation_counter > max_stagnation:
            mutation_scale = min(0.5, mutation_scale * 1.2)
            stagnation_counter = 0
        else:
            mutation_scale = max(0.01, mutation_scale * 0.98)
    
    # Local refinement on best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
