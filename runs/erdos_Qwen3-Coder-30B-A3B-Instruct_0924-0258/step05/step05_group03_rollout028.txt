import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Scale by integral constraint
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Parameters
    max_iterations = 100
    shift_samples = 100
    dx = 2.0 / n_points
    
    # Generate shifts for overlap computation
    shifts = np.linspace(0, 2, shift_samples, endpoint=False)
    
    # Cutting plane approach
    h_var = Variable(n_points)
    C5_var = Variable()
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        sum(h_var) == n_points / 2,
        C5_var >= 0
    ]
    
    # Initial solution
    h_current = initial_h.copy()
    c5_current = evaluate_erdos_solution(h_current, dx)
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Cut plane iteration
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all shifts
        max_overlap = 0
        worst_shift = 0
        for shift in shifts:
            # Compute discrete convolution for this shift
            shift_int = int(shift / dx)
            if shift_int >= 0:
                corr_val = np.sum(h_current[:n_points-shift_int] * (1 - h_current[shift_int:]))
            else:
                corr_val = np.sum(h_current[-shift_int:] * (1 - h_current[:n_points+shift_int]))
            corr_val *= dx
            if corr_val > max_overlap:
                max_overlap = corr_val
                worst_shift = shift
        
        # Convert shift to discrete index
        shift_idx = int(worst_shift / dx)
        
        # Add the constraint corresponding to the worst overlap
        if shift_idx <= n_points - 1:
            sum_term = 0
            for i in range(n_points - shift_idx):
                sum_term += h_var[i] * (1 - h_var[i + shift_idx])
            sum_term *= dx
            constraints.append(sum_term <= C5_var)
        else:
            # Handle boundary case with aligned conv
            # Use a different approach for shift that causes incomplete convolution
            # For now, ignore this case to simplify
            continue
            
        # Re-optimize with new constraint
        objective = Minimize(C5_var)
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_new = np.array(h_var.value).flatten()
                c5_val = C5_var.value
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_new.copy()
        except:
            break
            
        # Early termination
        if best_c5 < 0.3807:
            break
    
    # Final local refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
