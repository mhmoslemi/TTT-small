```python
import numpy as np
from scipy.optimize import minimize
from time import time
import hashlib

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    max_iter = 1000
    max_stagnation = 100
    tabu_tenure = min(50, n_points // 2)
    mutation_scale = 0.04
    perturb_fraction = 0.3
    refinement_interval = 10
    
    # Tabu list as set of hashes
    tabu_list = set()
    stagnation_counter = 0
    iteration = 0
    improvements = 0
    
    # Start with initial solution
    current_h = h_start.copy()
    current_c5 = best_c5
    
    while iteration < max_iter and (time() - start_time) < budget_s * 0.95:
        # Generate candidate by perturbing 30% of values
        candidate_h = current_h.copy()
        indices_to_perturb = np.random.choice(n_points, 
                                            size=int(n_points * perturb_fraction), 
                                            replace=False)
        
        for i in indices_to_perturb:
            # Add Gaussian noise
            candidate_h[i] += np.random.normal(0, mutation_scale)
        
        # Clip to [0,1] and rescale to sum to n_points/2
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        
        # Compute overlap metric
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Generate hash key to check tabu list
        h_sorted = tuple(sorted(candidate_h))
        h_key = hashlib.md5(str(h_sorted).encode()).hexdigest()
        
        # Check if this is a new improving solution
        if candidate_c5 < current_c5:
            if h_key not in tabu_list:
                current_h = candidate_h.copy()
                current_c5 = candidate_c5
                improvements += 1
                
                # update best if better
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    if best_c5 < 0.3807:
                        break
                    
                stagnation_counter = 0
            else:
                # force accept if better than current
                current_h = candidate_h.copy()
                current_c5 = candidate_c5
                stagnation_counter = 0
        else:
            stagnation_counter += 1
            if stagnation_counter > max_stagnation:
                mutation_scale = min(0.5, mutation_scale * 1.2)
                stagnation_counter = 0
        
        # Manage tabu list
        if h_key in tabu_list:
            tabu_list.remove(h_key)
        if len(tabu_list) >= tabu_tenure:
            # Remove oldest element
            # This is an approximation since sets don't preserve insertion order
            pass
        tabu_list.add(h_key)
        
        # Local refinement every few improvements
        if improvements % refinement_interval == 0 and improvements > 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, current_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    refined_h = res.x
                    refined_c5 = objective(refined_h)
                    if refined_c5 < current_c5:
                        current_h = refined_h
                        current_c5 = refined_c5
                        if refined_c5 < best_c5:
                            best_h = refined_h
                            best_c5 = refined_c5
            except:
                pass
        
        iteration += 1
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30 and best_c5 > 0.3807:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                final_h = res.x
                final_c5 = objective(final_h)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
```
