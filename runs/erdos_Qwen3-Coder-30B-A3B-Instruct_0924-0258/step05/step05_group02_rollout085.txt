import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def generate_sparse_configurations(n_points, sparsity):
    """Generate all binary configurations with given sparsity"""
    if sparsity == 0:
        return [np.zeros(n_points)]
    if sparsity == n_points:
        return [np.ones(n_points)]
    
    # For small sparsities, we can enumerate
    from itertools import combinations
    configs = []
    for indices in combinations(range(n_points), sparsity):
        h = np.zeros(n_points)
        h[list(indices)] = 1
        configs.append(h)
    return configs

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different sparsity levels and configurations
    max_sparsity = min(n_points // 2, 15)
    
    # For first few seconds, try integer-based search
    for s in range(1, max_sparsity + 1):
        if time() - start_time > budget_s * 0.8:
            break
            
        configs = generate_sparse_configurations(n_points, s)
        for i, h_binary in enumerate(configs):
            if time() - start_time > budget_s * 0.8:
                break
                
            # Normalize to have sum = n_points/2
            if np.sum(h_binary) > 0:
                h_scaled = h_binary * (n_points/2) / np.sum(h_binary)
            else:
                h_scaled = np.ones(n_points) * 0.5
                
            # Refine with optimization if not all ones or zeros
            if not np.all(h_binary == 1) and not np.all(h_binary == 0):
                try:
                    def objective(h_ref):
                        return evaluate_erdos_solution(h_ref, dx)
                    
                    bounds = [(0, 1) for _ in range(n_points)]
                    cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2}
                    
                    res = minimize(
                        objective,
                        h_scaled,
                        method='SLSQP',
                        bounds=bounds,
                        constraints=cons_sum,
                        options={'maxiter': 50, 'ftol': 1e-6}
                    )
                    if res.success:
                        h_refined = res.x
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
    
    # Local search refinement with current best
    h_refined = best_h.copy()
    h_refined = np.clip(h_refined, 0, 1)
    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
    
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2}
        
        res = minimize(
            lambda h: evaluate_erdos_solution(h, dx),
            h_refined,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'maxiter': 100, 'ftol': 1e-7}
        )
        
        if res.success:
            h_final = res.x
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    # Final validation and normalization
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
