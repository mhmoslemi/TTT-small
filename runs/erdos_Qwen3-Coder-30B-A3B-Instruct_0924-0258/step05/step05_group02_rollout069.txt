import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse-to-fine optimization with adaptive mutations
    n_low = 20
    n_high = n_points
    
    # Create coarse grid
    h_low = np.ones(n_low) * 0.5
    if 'initial_h_values' in globals():
        h_temp = np.array(initial_h_values)
        # Interpolate from coarse to fine
        factor = n_points / n_low
        indices = np.arange(n_low) * factor
        h_low = np.interp(np.arange(n_points), indices, h_temp[:n_low])
        h_low = np.clip(h_low, 0, 1)
        h_low = h_low * (n_points/2) / np.sum(h_low)
    
    # Convex relaxation on coarse grid (subset of constraints)
    try:
        h_relaxed = cp.Variable(n_low)
        constraints = [
            cp.sum(h_relaxed) == n_low / 2.0,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        # Limit constraints to avoid solver blowup
        for k in range(1, min(10, n_low)):  
            if k < n_low // 2:
                corr_sum = cp.sum(cp.multiply(h_relaxed[:n_low-k], 1 - h_relaxed[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_low-k]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_relaxed.value
            h_coarse = np.clip(h_feasible, 0, 1)
            h_coarse = h_coarse * (n_low/2) / np.sum(h_coarse)
        else:
            h_coarse = h_low
    except Exception:
        h_coarse = h_low

    # Upsample to high resolution
    h_upsampled = np.interp(np.linspace(0, n_points-1, n_points), 
                          np.linspace(0, n_low-1, n_low), 
                          h_coarse)
    h_upsampled = np.clip(h_upsampled, 0, 1)
    h_upsampled = h_upsampled * (n_points/2) / np.sum(h_upsampled)

    # Refinement using gradient descent with adaptive mutations
    max_iter = 5000
    patience = 0
    patience_limit = 100
    min_improvement = 1e-6
    lr = 0.01
    mutation_freq = 50
    mutation_strength = 0.05

    h_current = h_upsampled.copy()
    c5_current = evaluate_erdos_solution(h_current, dx)
    
    if c5_current < best_c5:
        best_h = h_current
        best_c5 = c5_current

    def compute_gradient(h, dx):
        # Approximate gradient of C5 w.r.t h
        n = len(h)
        grad = np.zeros(n)
        eps = 1e-6
        base_c5 = evaluate_erdos_solution(h, dx)
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n/2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    iteration = 0

        try:
            # Compute gradient
            grad = compute_gradient(h_current, dx)
            # Update with gradient descent
            h_new = h_current - lr * grad
            # Ensure bounds and constraint
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)

                best_c5 = c5_new
                patience = 0

                patience += 1

            if iteration % mutation_freq == 0 and patience > 0:
                # Mutate based on gradient sign
                mutated = h_current + mutation_strength * np.sign(grad) * np.random.randn(n_points)
                mutated = np.clip(mutated, 0, 1)
                mutated = mutated * (n_points/2) / np.sum(mutated)
                c5_mutated = evaluate_erdos_solution(mutated, dx)
                if c5_mutated < best_c5:
                    best_h = mutated
                    best_c5 = c5_mutated
                    patience = 0
                else:
                    h_new = h_current  # revert to previous if no improvement
                    
            # Update current step
            h_current = h_new
            # Adjust learning rate
            if patience < 20:
                lr = min(0.05, lr * 1.01)
            else:
                lr = max(1e-6, lr * 0.99)
                
            # Early stopping
            if patience > patience_limit:
                break
                
        except Exception:
            patience += 1
            if patience > patience_limit:
                break
    
    # Final check and normalization
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    # Ensure constraints are met
    assert abs(np.sum(final_h) - n_points/2) < 1e-6
    assert np.all(final_h >= 0) and np.all(final_h <= 1)
    
    return (list(final_h), final_c5, n_points)
