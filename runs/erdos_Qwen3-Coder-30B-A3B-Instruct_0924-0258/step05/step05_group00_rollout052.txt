import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive Stochastic Descent (ASD)
    max_iters = 500
    step_size = 0.1
    min_step_size = 0.01
    decay_rate = 0.995
    
    for i in range(max_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb
        noise = np.random.normal(0, step_size, n_points)
        h_new = best_h + noise
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct sum
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Decay step size
        step_size = max(min_step_size, step_size * decay_rate)
    
    # Sequential QCP refinement
    max_qcp_iters = 100
    for i in range(max_qcp_iters):
        if time() - start_time > budget_s * 0.99:
            break
            
        try:
            # Formulate QCP
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add overlap constraints
            # Only add a subset of constraints to reduce complexity
            for k in range(1, n_points // 2 + 1, 2):
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
            
            if prob.status == cp.OPTIMAL:
                h_refined = h_opt.value
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except Exception:
            continue
    
    # Final cleanup and validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
