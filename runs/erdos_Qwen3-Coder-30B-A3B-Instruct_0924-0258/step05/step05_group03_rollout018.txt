import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Time budget for main algorithm
    remaining_time = budget_s * 0.95
    
    # CCP approach: Alternate between optimizing h and k
    h = initial_h.copy()
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Initial worst case shift detection
    shifts = np.linspace(0, 2, 100)
    worst_k = 0.0
    for k in shifts:
        corr = np.correlate(h, 1 - np.roll(h, int(k/dx)), mode='full')
        max_corr = np.max(corr) * dx
        if max_corr > best_c5:
            best_c5 = max_corr
            worst_k = k
    
    # Iteration counter
    iteration = 0
    max_iterations = 500
    
    while time() - start_time < remaining_time and iteration < max_iterations:
        # Find worst shift k
        k_candidates = np.linspace(0, 2, 100)
        max_val = -np.inf
        best_k = 0.0
        
        for k in k_candidates:
            # Compute correlation
            roll_idx = int(k / dx)
            h_shifted = np.roll(h, roll_idx)
            corr = np.sum(h * (1 - h_shifted)) * dx
            if corr > max_val:
                max_val = corr
                best_k = k
        
        # Convex relaxation to solve for h
        h_new = h.copy()
        
        # Use CVXPY for convex optimization of h given k
        try:
            h_var = cp.Variable(n_points)
            # Objective: minimize sum of h[i] * (1 - h[i + k]) for all i
            objective = cp.Minimize(cp.sum(cp.multiply(h_var, 1 - cp.roll(h_var, int(best_k/dx)))))
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points / 2
            ]
            
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
        except:
            h_new = h.copy()
        
        h = h_new
        c5 = evaluate_erdos_solution(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        iteration += 1
        
        if best_c5 < 0.3807:
            break
    
    # Final polishing with scipy
    try:
        def obj(x):
            return evaluate_erdos_solution(x, dx)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            final_h = res.x
            final_c5 = obj(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h
    except:
        pass
    
    # Ensure it satisfies constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
