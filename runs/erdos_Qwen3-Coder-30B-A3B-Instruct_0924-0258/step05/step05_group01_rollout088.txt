import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary algorithm parameters
    pop_size = 50
    max_generations = 1000
    mutation_rate = 0.15
    crossover_rate = 0.8
    tournament_size = 3
    
    # Create initial population
    population = []
    for _ in range(pop_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Multi-resolution approach
    resolutions = [35, 40, 50]
    current_res = 0
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def tournament_selection(pop, k):
        candidates = random.sample(pop, k)
        return min(candidates, key=objective)
    
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        crossover_point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                h[i] += np.random.normal(0, 0.03)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Main evolutionary loop
    generation = 0
    stagnation_count = 0
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate population
        population_fitness = [(h, objective(h)) for h in population]
        population_fitness.sort(key=lambda x: x[1])
        
        # Update best solution
        current_best_h, current_best_c5 = population_fitness[0]
        if current_best_c5 < best_c5:
            best_h = current_best_h.copy()
            best_c5 = current_best_c5
            last_improvement = generation
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Check for stagnation
        if stagnation_count > 50:
            # Reset with new random population near best solution
            population = []
            for _ in range(pop_size):
                h = best_h + np.random.normal(0, 0.03, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                population.append(h)
            stagnation_count = 0
            last_improvement = generation
        
        # Select parents and generate new population
        new_population = [current_best_h.copy()]  # Elitism
        
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, tournament_size)
            parent2 = tournament_selection(population, tournament_size)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
        
        # Resolution upgrade
        if (time() - start_time > (budget_s * 0.7) and 
            current_res < len(resolutions) - 1):
            new_res = resolutions[current_res + 1]
            if new_res > n_points:
                # Interpolate to higher resolution
                h_new = np.interp(np.linspace(0, 2, new_res), 
                                np.linspace(0, 2, n_points), best_h)
                n_points = new_res
                dx = 2.0 / n_points
                best_h = h_new
                best_c5 = evaluate_erdos_solution(best_h, dx)
                # Update population with new resolution
                new_pop = []
                for h in population:
                    h_new = np.interp(np.linspace(0, 2, new_res), 
                                    np.linspace(0, 2, n_points), h)
                    new_pop.append(h_new)
                population = new_pop
                current_res += 1
                last_improvement = generation
        
        generation += 1
    
    # Final refinement with local optimization
    if time() - start_time < budget_s * 0.05:
        try:
            # Try local optimization on best solution
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            h_refined = best_h.copy()
            # SLSQP local optimization
            res_opt = minimize(objective, h_refined, method='SLSQP', 
                            bounds=bounds, constraints=cons_sum, 
                            options={'maxiter': 200, 'ftol': 1e-8})
            if res_opt.success:
                h_refined = res_opt.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
