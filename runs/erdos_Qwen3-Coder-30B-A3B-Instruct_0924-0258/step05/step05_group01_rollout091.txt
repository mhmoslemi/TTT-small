import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Convert to binary if needed and ensure sum constraint
    h_start = np.round(h_start).astype(int)
    target_sum = n_points // 2
    current_sum = np.sum(h_start)
    
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        flip_indices = np.where(h_start == 1)[0]
        np.random.shuffle(flip_indices)
        for i in range(current_sum - target_sum):
            h_start[flip_indices[i]] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        flip_indices = np.where(h_start == 0)[0]
        np.random.shuffle(flip_indices)
        for i in range(target_sum - current_sum):
            h_start[flip_indices[i]] = 1
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 10
    max_no_improvement = 100
    no_improvement_count = 0
    shaking_frequency = 50
    
    # Shaking function
    def shaking(h, fraction=0.05):
        h_new = h.copy()
        n_flip = max(1, int(n_points * fraction))
        flip_indices = np.random.choice(n_points, size=n_flip, replace=False)
        for idx in flip_indices:
            h_new[idx] = 1 - h_new[idx]
        return h_new
    
    # Tabu search loop
    while time() - start_time < budget_s * 0.95:
        current_h = best_h.copy()
        improved = False
        
        # Try all single flips (neighborhood search)
        for i in range(n_points):
            h_trial = current_h.copy()
            h_trial[i] = 1 - h_trial[i]  # Flip bit
            
            # Ensure sum constraint (count 1s)
            if np.sum(h_trial) != n_points // 2:
                continue
                
            # Check if this move is tabu
            if (i, 1 - current_h[i], current_h[i]) in tabu_list:
                if evaluate_erdos_solution(h_trial, dx) < best_c5 * 0.999:
                    # Aspiration: allow even if tabu
                    pass
                else:
                    continue  # Skip tabu move
            
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
                improved = True
                no_improvement_count = 0
                # Update tabu list
                tabu_list.append((i, current_h[i], 1 - current_h[i]))
                if len(tabu_list) > tabu_tenure:
                    tabu_list.pop(0)
                break  # Take the first improving neighbor
        
        # If no improvement, shake the solution
        if not improved:
            no_improvement_count += 1
            if no_improvement_count >= shaking_frequency:
                h_restart = shaking(best_h)
                h_restart = np.clip(h_restart, 0, 1)
                c5_restart = evaluate_erdos_solution(h_restart, dx)
                if c5_restart < best_c5:
                    best_h = h_restart
                    best_c5 = c5_restart
                no_improvement_count = 0
        
        # Early termination condition
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
