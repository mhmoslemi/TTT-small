import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization parameters
    max_iter = 1000
    patience = 50
    
    # Initialize variables for tracking progress
    iteration = 0
    patience_counter = 0
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Local search with gradient-based optimization
    while time() - start_time < budget_s * 0.95 and iteration < max_iter:
        # Generate candidate solution
        h_candidate = best_h.copy()
        
        # Add small random perturbation
        noise = np.random.normal(0, 0.01, n_points)
        h_candidate += noise
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Normalize to satisfy integral constraint
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        
        # Optimize using scipy minimize
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_candidate, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            
            if res.success:
                h_opt = res.x
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
                    patience_counter = 0
                else:
                    patience_counter += 1
            else:
                patience_counter += 1
                
        except Exception:
            patience_counter += 1
        
        # Periodic restarts to avoid local minima
        if patience_counter >= patience:
            h_restart = np.random.rand(n_points)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            h_restart = np.clip(h_restart, 0, 1)
            best_h = h_restart
            patience_counter = 0
            c5_restart = evaluate_erdos_solution(h_restart, dx)
            if c5_restart < best_c5:
                best_c5 = c5_restart
        
        iteration += 1
        
        # Early termination if not improving
        if patience_counter >= patience:
            break
    
    # Final refinement step
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    # Return final result
    return (list(best_h), best_c5, n_points)
