import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization with cutting plane method
    resolutions = [35, 40, 50, 60]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Cutting plane optimization for this resolution
        max_iter = 100
        iteration = 0
        
        while iteration < max_iter and time() - start_time < budget_s * 0.95:
            # Compute overlaps for current h
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_k = np.argmax(corr)
            max_overlap = corr[max_k] * dx
            
            if max_overlap < best_c5:
                best_h = h_current.copy()
                best_c5 = max_overlap
            
            # Add constraint for current maximum overlap shift
            left_idx = max_k - (len(h_current) - 1)
            if left_idx < 0:
                # overlap at shift k is sum of h[i]* (1-h[i+k]) for valid indices
                corr_constraint = np.sum(h_current[:len(h_current)+left_idx] * 
                                       (1 - h_current[-left_idx:]))
            else:
                corr_constraint = np.sum(h_current[left_idx:] * 
                                       (1 - h_current[:len(h_current)-left_idx]))
            
            # Solve a convex relaxation with improved constraint
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                # Add a constraint for the most problematic overlap shift
                if left_idx < 0:
                    overlap_expr = cp.sum(cp.multiply(h_opt[:len(h_current)+left_idx], 
                                                    1 - h_opt[-left_idx:]))
                else:
                    overlap_expr = cp.sum(cp.multiply(h_opt[left_idx:], 
                                                    1 - h_opt[:len(h_current)-left_idx]))
                
                # Add constraint only if it's not already satisfied enough
                if overlap_expr.value is not None and overlap_expr.value > best_c5 * dx * 1.001:
                    constraints.append(overlap_expr <= best_c5 * dx)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
                
                if prob.status == cp.OPTIMAL:
                    new_h = h_opt.value
                    new_c5 = evaluate_erdos_solution(new_h, dx)
                    if new_c5 < best_c5:
                        best_h = new_h
                        best_c5 = new_c5
                    h_current = new_h
                else:
                    h_current = h_current + np.random.normal(0, 0.005, n_points)
                    h_current = np.clip(h_current, 0, 1)
                    h_current = h_current * (n_points/2) / np.sum(h_current)
            
            except Exception:
                # Fallback to constrained local optimization
                h_current = h_current + np.random.normal(0, 0.005, n_points)
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points/2) / np.sum(h_current)
            
            iteration += 1
        
        h_start = best_h.copy()
    
    # Final polishing with SLSQP
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
