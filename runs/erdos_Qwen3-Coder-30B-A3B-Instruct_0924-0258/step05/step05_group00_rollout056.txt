import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    generations = 1000
    tournament_size = 5
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    def create_individual():
        ind = np.random.random(n_points)
        ind = ind * (n_points/2) / np.sum(ind)
        return np.clip(ind, 0, 1)
    
    if 'initial_h_values' in globals():
        population = [h_start.copy()]
        population.extend([create_individual() for _ in range(population_size - 1)])
    else:
        population = [create_individual() for _ in range(population_size)]
    
    # Evolution loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
        
        # Selection
        def tournament_select():
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            tournament.sort(key=lambda x: x[1])
            return tournament[0][0]
        
        new_population = []
        
        # Elitism
        elite_idx = np.argsort(fitness)[:2]
        new_population.extend([population[i].copy() for i in elite_idx])
        
        # Generate offspring
        while len(new_population) < population_size:
            if random.random() < crossover_rate:
                parent1 = tournament_select()
                parent2 = tournament_select()
                
                # Single-point crossover
                crossover_point = random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutation
                for child in [child1, child2]:
                    for i in range(n_points):
                        if random.random() < mutation_rate:
                            child[i] += np.random.normal(0, 0.02)
                    
                    # Repair
                    child = np.clip(child, 0, 1)
                    child = child * (n_points/2) / np.sum(child)
                    
                    new_population.append(child)
                    if len(new_population) >= population_size:
                        break
            else:
                # Just mutate random parent
                parent = tournament_select()
                child = parent.copy()
                for i in range(n_points):
                    if random.random() < mutation_rate:
                        child[i] += np.random.normal(0, 0.02)
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                new_population.append(child)
        
        population = new_population[:population_size]
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
