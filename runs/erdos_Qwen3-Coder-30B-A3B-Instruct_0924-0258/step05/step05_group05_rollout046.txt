import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Determine initialization
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize with the best known construction if available
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # SDP relaxation approach for initial guess
    # This provides a high-quality starting point
    
    # Do not spend too much time on SDP for this version
    # Instead, try local refinement with simulated annealing approach
    
    # Simulated Annealing parameters
    max_iters = 10000
    temp_start = 1.0
    temp_end = 0.001
    alpha = 0.995
    stagnation_limit = 1000
    
    current_h = h_current.copy()
    current_c5 = evaluate_erdos_solution(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Track for cooling schedule
    temp = temp_start
    stagnation_count = 0
    
    for i in range(max_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor by perturbing randomly
        neighbor_h = current_h.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.05)
        neighbor_h[idx] += delta
        
        # Repair to keep in [0,1] and maintain integral
        neighbor_h = np.clip(neighbor_h, 0, 1)
        neighbor_h = neighbor_h * (n_points/2) / np.sum(neighbor_h)
        
        neighbor_c5 = evaluate_erdos_solution(neighbor_h, dx)
        
        # Metropolis acceptance criterion
        if neighbor_c5 < current_c5:
            current_h = neighbor_h.copy()
            current_c5 = neighbor_c5
            if neighbor_c5 < best_c5:
                best_h = neighbor_h.copy()
                best_c5 = neighbor_c5
                if best_c5 < 0.3807:
                    break
            stagnation_count = 0
        else:
            # Accept with probability based on temperature
            delta_e = neighbor_c5 - current_c5
            prob = np.exp(-delta_e / temp)
            if np.random.rand() < prob:
                current_h = neighbor_h.copy()
                current_c5 = neighbor_c5
            stagnation_count += 1
            
        # Cooling schedule
        if stagnation_count > stagnation_limit:
            temp *= alpha
            stagnation_count = 0
            
        if temp < temp_end:
            break
    
    # Final optimization with local method
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Use L-BFGS-B for local refinement
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
