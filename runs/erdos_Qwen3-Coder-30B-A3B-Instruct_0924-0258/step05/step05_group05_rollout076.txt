import numpy as np
from scipy.optimize import minimize
from time import time
import hashlib

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_tenure = max(10, n_points // 5)
    max_stagnation = 100
    max_iter = 1000
    mutation_scale = 0.04
    perturb_fraction = 0.3
    
    # Tabu list: store hashes of recent solutions
    tabu_list = []
    tabu_hashes = set()
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    stagnation_counter = 0
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create new candidate by perturbing 30% of elements
        perturb_count = max(1, int(n_points * perturb_fraction))
        indices = np.random.choice(n_points, size=perturb_count, replace=False)
        
        new_h = current_h.copy()
        for i in indices:
            new_h[i] += np.random.normal(0, mutation_scale)
        
        # Clip and rescale
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Compute overlap
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Create hash for tabu check
        sorted_h = tuple(np.sort(new_h))
        h_hash = hashlib.md5(str(sorted_h).encode()).hexdigest()
        
        # Check if this is in tabu list (and not a good move)
        if h_hash in tabu_hashes and new_c5 >= current_c5:
            # Skip this move
            continue
            
        # Update tabu list
        if len(tabu_list) >= tabu_tenure:
            old_hash = tabu_list.pop(0)
            tabu_hashes.discard(old_hash)
            
        tabu_list.append(h_hash)
        tabu_hashes.add(h_hash)
        
        # Update current solution
        current_h = new_h
        current_c5 = new_c5
        
        # Update best if improved
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_counter = 0
            
            # Local refinement every 10 improvements
            if len(tabu_list) % 10 == 0:
                try:
                    def objective(h):
                        return evaluate_erdos_solution(h, dx)
                    
                    def constraint_sum(h):
                        return np.sum(h) - n_points / 2.0
                    
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    bounds = [(0, 1) for _ in range(n_points)]
                    
                    res = minimize(objective, best_h, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-8})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
        else:
            stagnation_counter += 1
            
        # Adaptive mutation scale
        if stagnation_counter > max_stagnation:
            mutation_scale = min(0.5, mutation_scale * 1.5)
            stagnation_counter = 0
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
