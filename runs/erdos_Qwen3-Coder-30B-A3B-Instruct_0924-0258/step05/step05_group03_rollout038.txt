import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Ensure initial solution is feasible
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Parameters
    max_iter = 500
    stagnation_limit = 50
    shift_samples = 50
    
    # Initialize
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    # History tracking
    last_improvement = 0
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample shifts to approximate gradient
        shifts = np.random.choice(n_points, size=shift_samples, replace=True)
        
        # Find maximum overlapping shift
        max_overlap = -1
        best_shift = 0
        for k in shifts:
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlap *= (2.0 / n_points)
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = k
        
        # Compute gradient for this shift
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + best_shift) % n_points
            gradient[i] += (1 - h[j]) * (2.0 / n_points)
            gradient[j] -= h[i] * (2.0 / n_points)
        
        # Line search for step size
        alpha = 1.0
        step_size = 0.1
        while alpha > 1e-8:
            new_h = h - alpha * gradient
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_c5 = evaluate_erdos_solution(new_h, 2.0 / n_points)
            if new_c5 < best_c5:
                h = new_h
                c5 = new_c5
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    last_improvement = iteration
                break
            alpha *= step_size
        
        # If no improvement, perturb slightly
        if iteration - last_improvement > stagnation_limit // 2:
            h = h + np.random.normal(0, 0.001, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        stagnation_count = iteration - last_improvement
        
        # Early exit if target achieved
        if best_c5 < 0.3807:
            break
    
    # Final refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
