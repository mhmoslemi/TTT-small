import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    # Normalize h to have sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Track best solution
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    best_h = h.copy()

    # Bilevel optimization approach
    max_iters = 100
    stagnation_count = 0
    prev_best = best_c5
    
    for iteration in range(max_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Inner loop: find optimal k for current h
        shifts = np.linspace(0, 2, 100)
        overlaps = []
        for k in shifts:
            # Compute overlap for shift k
            corr = np.correlate(h, 1 - np.roll(h, int(k * n_points / 2)), mode='full')
            overlap = np.max(corr) * (2.0 / n_points)
            overlaps.append(overlap)
        
        # Find the worst shift k_star
        k_star_idx = np.argmax(overlaps)
        k_star = shifts[k_star_idx]
        
        # Outer loop: solve convex subproblem to find better h
        # We solve the problem: minimize overlap_k_star(h)
        # which is equivalent to: minimize sum h[i] * (1 - h[i + k_star])
        # under box constraints and sum(h) = n_points / 2
        
        # Create convex optimization problem
        h_var = cp.Variable(n_points)
        
        # Objective: maximize overlap for k_star
        # We use a relaxation and minimize the negative overlap
        # To compute overlap at k_star
        shift_idx = int(k_star * n_points / 2)
        if shift_idx == 0:
            overlap_expr = cp.sum(cp.multiply(h_var, 1 - h_var))
        else:
            # h[i] * (1 - h[i + shift_idx])
            expr = cp.multiply(h_var, 1 - cp.roll(h_var, shift_idx))
            # Take the sum but be careful about periodicity
            # This is tricky in convex programming, so we sample and approximate
            overlap_expr = cp.sum(expr)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2.0,
            h_var >= 0,
            h_var <= 1
        ]
        
        # To exactly solve this in one go we need to work smarter.
        # We proceed with a simpler approach using projection gradient descent
        # for now, to avoid CVXPY overhead
        
        # Approximate the overlap for k_star in this case.
        # Let's just do proper gradient descent with projected updates.
        # We have all h, and we can compute gradient.
        
        # Instead, we'll take an alternate approach.
        # Re-evaluate we should solve for max k for a fixed h, then minimize for h.
        # This is a bilinear problem.
        
        # For small scale problems, just aggregate overlap over approximate k values
        # Manually extract max overlap for current h and iterate
        
        # Re-evaluate handling of shift and correlation
        dx = 2.0 / n_points
        
        # Project gradient descent
        step_size = 0.01  # adaptively valid
        
        def calculate_overlap(h_vec):
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return np.max(corr) * dx
        
        def project_h(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            return h_vec
        
        # Simple iterative descent, project between steps
        # We repeat the subproblem optimization cycle
        # This was suboptimal in last version. Let's take a different path:
        # let's try a simpler approach using coordinate descent
        
        # Try coordinate descent approach with projected updates for h
        
        h_prev = h.copy()
        for i in range(10):  # Multiple steps
            # Compute gradient approx. - use local search approach
            h_current = project_h(h)
            
            # Try small perturbations to get grad estimate
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            
            for i_idx in range(n_points):
                # Different strategies:
                # Use simple line search or heuristic direction
                h_plus[i_idx] += 0.01
                h_minus[i_idx] -= 0.01
                h_plus = project_h(h_plus)
                h_minus = project_h(h_minus)
                
                # Approximate gradient
                overlap_plus = calculate_overlap(h_plus)
                overlap_minus = calculate_overlap(h_minus)
                
                # Central difference gradient
                grad_h = 0.5 * (overlap_plus - overlap_minus) / 0.01
                
                # Update h (Backwards in gradient)
                h_current[i_idx] -= step_size * grad_h

            h_current = project_h(h_current)
            c5_current = calculate_overlap(h_current)
            
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                stagnation_count = 0
            elif c5_current == best_c5:
                stagnation_count += 1
            else:
                stagnation_count += 1
                
            h = h_current 
            
            if time() - start_time > budget_s * 0.95:
                break
                
        if abs(calculate_overlap(h) - prev_best) < 5e-6:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        prev_best = calculate_overlap(h)
        
        if stagnation_count > 50:
            # Perturb solution slightly
            h = h + np.random.normal(0, 0.001, n_points)
            h = project_h(h)
            stagnation_count = 0
            
        if best_c5 < 0.3807:
            break
            
    # Final polishing with scipy
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
        
    return (list(best_h), best_c5, n_points)
