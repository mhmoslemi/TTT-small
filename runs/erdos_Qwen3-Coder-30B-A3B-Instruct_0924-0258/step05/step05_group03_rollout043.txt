import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Get initial solution
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Initialize
    h = initial_h.copy()
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    # Convex-concave procedure
    max_iter = 1000
    tol = 1e-6
    step_size = 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find worst shift k*
        # Discretize shifts
        shifts = np.linspace(0, 2, 100)
        worst_k = 0
        max_corr_val = -1
        
        for k in shifts:
            # Compute correlation at shift k
            # We compute shift by moving h and computing correlation
            # For numerical efficiency: use discrete convolution
            # But since we are just trying to identify worst shift:
            idx_k = int(k * n_points / 2)
            if idx_k >= n_points:
                idx_k = n_points - 1
            h_shifted = np.roll(h, -idx_k)
            corr_val = np.sum(h * (1 - h_shifted)) * (2.0 / n_points)
            if corr_val > max_corr_val:
                max_corr_val = corr_val
                worst_k = k
                
        # Solve convex subproblem with fixed k
        # Minimize: sum over i of h_new[i] * (1 - h[i + k]) * dx
        # Subject to: 
        #   0 <= h_new[i] <= 1
        #   sum(h_new) * dx = 1
        #   h_new = h_new.vector
        
        # Simplified:
        # Maybe consider solving a convex relaxation
        # We'll use a projected gradient approach  
        h_new = h.copy()
        # For this case k*, solve:
        # min_xyz sum(h_new[i] * (1 - h[i + k])) dx
        # s.t. h_new[i] in [0, 1], sum(h_new) = n_points/2
        
        # Just use projected gradient descent 
        # Compute gradient at h
        if worst_k == 0:
            grad = 1 - h
        else:
            idx_k = int(worst_k * n_points / 2)
            if idx_k >= n_points:
                idx_k = n_points - 1
            h_shifted = np.roll(h, -idx_k)
            grad = 1 - h_shifted
        
        # Gradient step
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Keep h_new as starting point for next iteration
        h = h_new
        
        c5 = evaluate_erdos_solution(h, 2.0 / n_points)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
        if iteration % 10 == 0:
            if best_c5 < 0.3807:
                break
                
        # Early termination if improvement is small
        if iteration > 1 and abs(best_c5 - evaluate_erdos_solution(h, 2.0 / n_points)) < tol:
            break
    
    # Final polishing with SLSQP
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
        
    return (list(best_h), best_c5, n_points)
