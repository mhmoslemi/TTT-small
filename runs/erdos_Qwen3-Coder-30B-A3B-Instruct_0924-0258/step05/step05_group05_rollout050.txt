import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    max_iter = 1000
    max_stagnation = 100
    mutation_scale = 0.04
    tabu_tenure = min(50, n_points // 2)
    tabu_list = []
    
    stagnation_counter = 0
    current_best = best_c5
    
    # Initialize with the starting solution
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    current_c5 = evaluate_erdos_solution(h_current, dx)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate new candidate solution
        h_new = h_current.copy()
        
        # Perturb about 30% of the entries
        indices_to_mutate = random.sample(range(n_points), max(1, int(0.3 * n_points)))
        for i in indices_to_mutate:
            h_new[i] += np.random.normal(0, mutation_scale)
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Compute new overlap
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Check tabu list
        h_key = tuple(sorted(h_new))
        is_tabu = h_key in tabu_list
        
        # Accept if better, or not tabu and at least as good
        if (c5_new < current_c5) or (not is_tabu and c5_new <= current_c5):
            h_current = h_new
            current_c5 = c5_new
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
                
                # Early stopping if target achieved
                if best_c5 < 0.3807:
                    break
                    
                stagnation_counter = 0
            else:
                stagnation_counter += 1
        else:
            stagnation_counter += 1
            
        # Update tabu list
        if len(tabu_list) >= tabu_tenure:
            tabu_list.pop(0)
        tabu_list.append(h_key)
        
        # Adjust mutation rate if stagnating
        if stagnation_counter > max_stagnation:
            mutation_scale = min(0.3, mutation_scale * 1.2)
            stagnation_counter = 0
            
        # Local refinement every 10 iterations
        if iteration % 10 == 0 and current_c5 < best_c5 * 1.01:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
                
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < current_c5:
                        h_current = h_refined
                        current_c5 = c5_refined
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
            except:
                pass
    
    # Final optimization on best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
                
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
