import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with parent if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        n_points = len(h_current)
        # Round to binary and adjust sum
        h_current = np.round(h_current).astype(int)
        current_sum = np.sum(h_current)
        target_sum = n_points // 2
        if current_sum != target_sum:
            # Adjust by flipping bits
            flips_needed = target_sum - current_sum
            if flips_needed > 0:
                # Flip zeros to ones
                zero_indices = np.where(h_current == 0)[0]
                flip_indices = np.random.choice(zero_indices, flips_needed, replace=False)
                h_current[flip_indices] = 1
            elif flips_needed < 0:
                # Flip ones to zeros
                one_indices = np.where(h_current == 1)[0]
                flip_indices = np.random.choice(one_indices, -flips_needed, replace=False)
                h_current[flip_indices] = 0
    else:
        n_points = 35
        h_current = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, n_points // 2, replace=False)
        h_current[ones_indices] = 1
    
    dx = 2.0 / n_points
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Tabu search parameters
    tabu_tenure = 10
    tabu_list = []
    max_no_improvement = 100
    no_improvement_count = 0
    shaking_freq = 50
    
    # Main tabu search loop
    while time() - start_time < budget_s * 0.95:
        # Generate neighborhood by flipping exactly one 1 to 0 and one 0 to 1
        neighbors = []
        ones_indices = np.where(h_current == 1)[0]
        zeros_indices = np.where(h_current == 0)[0]
        
        if len(ones_indices) > 0 and len(zeros_indices) > 0:
            # Try up to 50 random neighbor flips
            for _ in range(min(50, len(ones_indices) * len(zeros_indices))):
                i1 = np.random.choice(ones_indices)
                i0 = np.random.choice(zeros_indices)
                h_neighbor = h_current.copy()
                h_neighbor[i1] = 0
                h_neighbor[i0] = 1
                neighbors.append((h_neighbor, i1, i0))
        
        # Evaluate neighbors
        if neighbors:
            best_neighbor = None
            best_neighbor_c5 = float('inf')
            
            for h_n, i1, i0 in neighbors:
                c5 = evaluate_erdos_solution(h_n, dx)
                # Check if the move is tabu
                move = (i1, i0) if i1 < i0 else (i0, i1)
                is_tabu = move in tabu_list
                
                # Use aspiration criteria to allow tabu moves if they are significantly better
                if c5 < best_neighbor_c5:
                    if not is_tabu or c5 < best_c5 * 0.99:  # Aspiration
                        best_neighbor = h_n
                        best_neighbor_c5 = c5
                        tabu_move = move
                    elif is_tabu and c5 < best_neighbor_c5:
                        # Allow tabu move if it's best so far
                        best_neighbor = h_n
                        best_neighbor_c5 = c5
                        tabu_move = move
            
            # Apply best neighbor if found
            if best_neighbor is not None:
                h_current = best_neighbor
                if best_neighbor_c5 < best_c5:
                    best_h = h_current
                    best_c5 = best_neighbor_c5
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
                
                # Update tabu list
                if len(tabu_list) >= tabu_tenure:
                    tabu_list.pop(0)
                tabu_list.append(tabu_move)
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Shaking mechanism
        if no_improvement_count >= shaking_freq:
            # Randomly flip some bits
            h_current = best_h.copy()
            flips = max(1, n_points // 20)  # Flip ~5% bits
            ones_indices = np.where(h_current == 1)[0]
            zeros_indices = np.where(h_current == 0)[0]
            
            if len(ones_indices) > 0 and len(zeros_indices) > 0:
                for _ in range(flips):
                    i1 = np.random.choice(ones_indices)
                    i0 = np.random.choice(zeros_indices)
                    h_current[i1] = 0
                    h_current[i0] = 1
                    
            no_improvement_count = 0
        
        # Early termination if no improvement for a while
        if no_improvement_count >= max_no_improvement:
            break
            
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
