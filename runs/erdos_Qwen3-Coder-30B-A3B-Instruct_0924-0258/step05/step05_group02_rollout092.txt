import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Coarse-to-fine approach
    n_low = 20
    n_high = 35
    
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        h_coarse = h_start.copy()
    else:
        n_points = n_high
        h_coarse = np.ones(n_low) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_coarse.copy()
    best_c5 = evaluate_erdos_solution(h_coarse, dx)
    
    # Maintain consistent scaling
    sum_target = n_points / 2.0
    if np.sum(h_coarse) != sum_target:
        h_coarse = h_coarse * sum_target / np.sum(h_coarse)
    
    # 1. Coarse Convex Relaxation
    if time() - start_time > budget_s * 0.8:
        return (list(best_h), best_c5, n_points)
    
    try:
        # Create a convex relaxation using CVXPY with fewer constraints
        h_relaxed = cp.Variable(n_low)
        constraints = [
            cp.sum(h_relaxed) == sum_target,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        
        # Add subset of constraints
        k_max = min(5, n_low)
        for k in range(1, k_max):
            if k < n_low - k:
                corr_val = cp.sum(cp.multiply(h_relaxed[:n_low-k], 1 - h_relaxed[k:]))
            else:
                corr_val = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_low-k]))
            constraints.append(corr_val <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_relaxed.value
            h_feasible = np.clip(h_feasible, 0, 1)
            if np.sum(h_feasible) != sum_target:
                h_feasible = h_feasible * sum_target / np.sum(h_feasible)
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # 2. Upsample coarse values to high resolution
    if time() - start_time > budget_s * 0.85:
        return (list(best_h), best_c5, n_points)
    
    # Simple linear interpolation and normalization
    h_upsampled = np.interp(np.linspace(0, 1, n_high), 
                           np.linspace(0, 1, n_low), 
                           best_h)
    h_upsampled = np.clip(h_upsampled, 0, 1)
    h_upsampled = h_upsampled * sum_target / np.sum(h_upsampled)
    
    # 3. Fine-grid local optimization with adaptive mutation
    h_fine = h_upsampled.copy()
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Gradient estimator (simple finite difference for smoother)
    def grad_estimate(h, eps=1e-5):
        g = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * sum_target / np.sum(h_plus)
            f_plus = objective(h_plus)
            g[i] = (f_plus - objective(h)) / eps
        return g
    
    # Initialize optimization loop
    last_improvement = 0
    mutation_count = 0
    learning_rate = 0.01
    max_no_improvement = 100
    mutation_threshold = 50
    
    while time() - start_time < budget_s * 0.95:
        # Compute gradients
        g = grad_estimate(h_fine)
        
        # Update with gradient descent
        h_new = h_fine - learning_rate * g
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * sum_target / np.sum(h_new)
        
        # Evaluate new point
        c5_new = objective(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h_fine = h_new.copy()
            last_improvement = 0
            learning_rate = 0.01  # Reset learning rate
        else:
            last_improvement += 1
            h_fine = h_new.copy()
            
            # Adaptive mutation to escape local optima
            if last_improvement > mutation_threshold or np.random.rand() < 0.1:
                # Apply mutation
                h_mutated = h_fine.copy()
                mutation_strength = 0.05 + 0.01 * np.random.rand()
                noise = np.random.normal(0, mutation_strength, n_points)
                mask = np.random.rand(n_points) < 0.1
                h_mutated += noise * mask
                h_mutated = np.clip(h_mutated, 0, 1)
                h_mutated = h_mutated * sum_target / np.sum(h_mutated)
                c5_mut = objective(h_mutated)
                if c5_mut < best_c5:
                    best_h = h_mutated.copy()
                    best_c5 = c5_mut
                    h_fine = h_mutated.copy()
                    last_improvement = 0
                    mutation_count += 1
        
        # Early stopping
        if last_improvement > max_no_improvement:
            break
            
        # Reduce learning rate if slow progress
        if last_improvement > 20:
            learning_rate *= 0.95
    
    # Final evaluation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * sum_target / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
