import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Grid size strategy
    n_low = 20
    dx_low = 2.0 / n_low
    h_low = np.ones(n_low) * 0.5
    normalization_factor = n_low / 2.0
    h_low = h_low * normalization_factor / np.sum(h_low)
    
    # Coarse grid optimization using convex relaxation
    try:
        h_relaxed = cp.Variable(n_low)
        constraints = [
            cp.sum(h_relaxed) == normalization_factor,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        
        # Add a subset of overlap constraints for Convex relaxation
        max_k = min(10, n_low // 2)
        for k in range(1, max_k):
            if k < n_low // 2:
                corr_sum = cp.sum(cp.multiply(h_relaxed[:n_low-k], 1 - h_relaxed[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_low-k]))
            constraints.append(corr_sum <= best_c5 * dx_low + 1e-6)

        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            h_low = h_relaxed.value
    except Exception:
        pass

    # Interpolate to target resolution
    h_interp = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_low), h_low)
    h_interp = np.clip(h_interp, 0, 1)
    h_interp = h_interp * (n_points/2) / np.sum(h_interp)
    
    # Local refinement with SLSQP
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    try:
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            objective, 
            h_interp, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons_sum,
            options={'maxiter': 100, 'ftol': 1e-7}
        )
        if res.success:
            h_refined = res.x
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Adaptive restart loop with exponential backoff
    max_restarts = 100
    restart_count = 0
    patience = 0
    
    while time() - start_time < budget_s * 0.95:
        h_perturbed = best_h.copy()
        perturbation = 0.05 + 0.1 * np.random.rand(n_points)  # Variable perturbation
        h_perturbed += perturbation * (np.random.rand(n_points) - 0.5)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        try:
            res = minimize(
                objective, 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-7}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    patience = 0
                else:
                    patience += 1
            else:
                patience += 1
        except Exception:
            patience += 1

        if patience > 10:
            patience = 0
            # Generate new starting point
            h_perturbed = np.random.rand(n_points)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            
        restart_count += 1
        if restart_count > max_restarts:
            break

    # Final validation
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
