import numpy as np
from scipy.optimize import minimize
from time import time
import hashlib

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    max_iter = 1000
    max_stagnation = 100
    mutation_scale = 0.04
    tabu_tenure = min(50, n_points // 2)
    tabu_list = []
    stagnation_counter = 0
    
    # Refinement parameters
    refinement_interval = 10
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    def hash_state(h):
        # Create hashable representation of the state
        sorted_h = tuple(sorted(h))
        return hashlib.md5(str(sorted_h).encode()).hexdigest()
    
    def perturb_solution(h, scale):
        h_new = h.copy()
        # Perturb about 30% of elements
        to_perturb = np.random.choice(n_points, size=int(0.3 * n_points), replace=False)
        for i in to_perturb:
            h_new[i] += np.random.normal(0, scale)
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        return h_new
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate new candidate
        h_candidate = perturb_solution(current_h, mutation_scale)
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Check if this solution is tabu
        candidate_hash = hash_state(h_candidate)
        is_tabu = candidate_hash in tabu_list
        
        # Accept if better or not tabu
        if c5_candidate < current_c5 or not is_tabu:
            current_h = h_candidate
            current_c5 = c5_candidate
            
            # Add to tabu list
            if len(tabu_list) >= tabu_tenure:
                tabu_list.pop(0)
            tabu_list.append(candidate_hash)
            
            # Update global best
            if c5_candidate < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_candidate
                stagnation_counter = 0
                
                # Local refinement
                if i % refinement_interval == 0:
                    def objective(h):
                        return evaluate_erdos_solution(h, dx)
                    
                    def constraint_sum(h):
                        return np.sum(h) - n_points / 2.0
                    
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    bounds = [(0, 1) for _ in range(n_points)]
                    
                    try:
                        res = minimize(objective, current_h, method='SLSQP', 
                                       bounds=bounds, constraints=cons_sum, 
                                       options={'maxiter': 50, 'ftol': 1e-8})
                        if res.success:
                            h_refined = res.x
                            c5_refined = objective(h_refined)
                            if c5_refined < current_c5:
                                current_h = h_refined
                                current_c5 = c5_refined
                                if c5_refined < best_c5:
                                    best_h = current_h.copy()
                                    best_c5 = c5_refined
                    except:
                        pass
            else:
                stagnation_counter += 1
        else:
            stagnation_counter += 1
        
        # Adjust mutation rate if stuck
        if stagnation_counter > max_stagnation:
            mutation_scale = min(0.2, mutation_scale * 1.2)
            stagnation_counter = 0
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
