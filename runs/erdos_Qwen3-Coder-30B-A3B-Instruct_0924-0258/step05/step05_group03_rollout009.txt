import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Normalize initial solution
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Sample shifts for constraint generation
    n_shifts = 100
    shifts = np.linspace(0, 2, n_shifts, endpoint=False)
    
    # Convex relaxation approach with cutting planes
    h_vars = cp.Variable(n_points)
    C5_var = cp.Variable()
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points / 2,
        C5_var >= 0
    ]
    
    # Add initial constraints from shifts
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    # Cutting plane loop
    iteration = 0
    max_iterations = 50
    stagnation_count = 0
    last_c5 = best_c5
    
    while (time() - start_time < budget_s * 0.95 and 
           iteration < max_iterations and 
           best_c5 > 0.3807):
        
        # Compute overlaps for current solution at sampled shifts
        overlaps = []
        for shift in shifts:
            # Implement shift operation
            shifted = np.zeros(n_points)
            if shift == 0:
                shifted = best_h
            elif shift > 0:
                # Shift to the right by shift*dx
                dx = 2.0 / n_points
                # This is simplified representation of correlation
                shift_samples = int(shift / dx)
                if shift_samples >= n_points:
                    overlap = 0
                else:
                    overlap = np.sum(best_h[:n_points-shift_samples] * (1 - best_h[shift_samples:]))
                overlaps.append(overlap * dx)
            else:
                # For negative shifts -> not used in standard definition,
                # and in full correlation we just go to end
                overlap = 0
                overlaps.append(overlap)
        
        # Compute the worst case overlap
        samples = []
        for k in range(1, n_points):
            if k <= n_points - 1:
                # Compute overlap with shift k
                shift_val = k * 2.0 / n_points
                # Correlation between h and 1 - h (right aligned) 
                overlap = 0
                for i in range(n_points):
                    if i + k < n_points:
                        overlap += best_h[i] * (1 - best_h[i+k])
                samples.append(overlap * (2.0 / n_points))
            else:
                break
                
        # Equivalently, full correlation
        full_corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_overlap = np.max(full_corr) * (2.0 / n_points)
        
        # Add constraint
        new_constraint = C5_var >= max_overlap
        constraints.append(new_constraint)
        
        # Solve convex problem
        objective = cp.Minimize(C5_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                new_h = h_vars.value
                new_c5 = C5_var.value
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                break
        except:
            break
            
        iteration += 1
        
        if abs(last_c5 - best_c5) < 1e-6:
            stagnation_count += 1
        last_c5 = best_c5
        
        if stagnation_count > 10:
            break
    
    # Final refinement with SLSQP
    def obj(x):
        return evaluate_erdos_solution(x, 2.0 / n_points)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 300, 'ftol': 1e-9})
        if res.success:
            final_h = res.x
            final_c5 = obj(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    # Ensure the solution is within bounds
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
