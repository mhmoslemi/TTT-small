import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
        
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    
    # Initialize
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    best_c5_global = best_c5
    
    # Parameters
    iter_limit = 50
    min_improvement = 1e-6
    max_shift_samples = 100
    tolerance = 1e-5
    
    for iteration in range(iter_limit):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Step 1: Find the worst shift k for current h
        shifts = np.linspace(0, 2, max_shift_samples, endpoint=False)
        overlaps = []
        for k in shifts:
            # Compute overlap for this shift
            k_int = int(k * n_points / 2)
            k_frac = k * n_points / 2 - k_int
            if k_int >= n_points or k_int < 0:
                overlap = 0
            else:
                # Linear interpolation-like handling
                if k_frac == 0:
                    overlap = np.sum(h * np.roll(1 - h, k_int))
                else:
                    h1 = np.roll(h, k_int)
                    h2 = np.roll(h, k_int + 1)
                    h_interp = h1 * (1 - k_frac) + h2 * k_frac
                    overlap = np.sum(h * (1 - h_interp))
            overlaps.append(overlap)
        
        max_overlap_idx = np.argmax(overlaps)
        k_star = shifts[max_overlap_idx]
        
        # Step 2: Optimization with respect to h for fixed k*
        # We'll solve a simplified version with CVXPY for the inner optimization
        def solve_optimization():
            # Objective is to minimize the overlap for fixed k_star
            # Since overlap is bilinear, and we are minimizing aggregate,
            # we can try to constrain the worst case using quadratic constraints
            h_var = cp.Variable(n_points)
            
            # Constraints
            constraints = [h_var[i] >= 0 for i in range(n_points)]
            constraints += [h_var[i] <= 1 for i in range(n_points)]
            constraints += [cp.sum(h_var) == n_points / 2]
            
            # We approximate the max overlap constraint by using a quadratic trick
            # Create overlap approximation
            eps = 1e-8
            obj = cp.Minimize(cp.sum(h_var) - eps)  # dummy objective

            prob = cp.Problem(obj, constraints)
            try:
                prob.solve(solver=ECOS, verbose=False)
                return h_var.value
            except:
                return None
        
        # For a fast search, we use simple gradient steps
        # Approximate gradient for overlap with respect to h for fixed shift
        def compute_overlap_for_shift(h, k_shift):
            # k_shift in [0,2], convert to index
            k_int = int(k_shift * n_points / 2)
            k_frac = k_shift * n_points / 2 - k_int
            if k_int >= n_points or k_int < 0:
                return 0
                
            h1 = np.roll(h, k_int)
            h2 = np.roll(h, k_int + 1)
            h_interp = h1 * (1 - k_frac) + h2 * k_frac
            return np.sum(h * (1 - h_interp)) * (2.0 / n_points)
        
        def compute_gradient(h, k_shift):
            k_int = int(k_shift * n_points / 2)
            k_frac = k_shift * n_points / 2 - k_int
            
            grad = np.zeros_like(h)
            # Approximate gradient of h[i] * (1 - h[i + k_shift]) term
            for i in range(n_points):
                idx1 = (i + k_int) % n_points
                idx2 = (i + k_int + 1) % n_points
                grad[i] = 1 - (1 - k_frac) * h[idx1] - k_frac * h[idx2]
            return grad
            
        # Small optimization step with stochastic gradient descent
        h_new = h.copy()
        lr = 0.01
        steps = 50
        for step in range(steps):
            if time() - start_time > budget_s * 0.95:
                break
            grad = compute_gradient(h_new, k_star)
            # projected gradient step
            h_new += lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
        h = h_new.copy()
        c5_current = evaluate_erdos_solution(h, 2.0 / n_points)
        
        if c5_current < best_c5_global:
            best_c5_global = c5_current
            best_h = h.copy()
            
        # Early exit if target reached
        if best_c5_global < 0.3807:
            break
            
    # Final polishing using scipy minimize
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5_global = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5_global, n_points)
