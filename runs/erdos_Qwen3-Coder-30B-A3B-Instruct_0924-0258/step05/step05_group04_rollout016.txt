import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Sequential Convex Programming approach
    max_iter = 500
    mu = 1e-3  # Augmented Lagrangian parameter
    c5_upper = 0.4  # Initial upper bound
    
    # Initialize with a good starting point
    h_current = h_start.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Define constraints matrix for overlap constraints
    # Constraint indexes
    for iteration in range(max_iter):
        if time() - start_time > max_time * 0.95:
            break
            
        current_c5 = evaluate_erdos_solution(h_current, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            
        if best_c5 < 0.3807:
            break
            
        # Formulate QP approximating QCQP
        # The key insight: overlap integral is quadratic in h
        # We treat the QCQP as approximate with linearized constraints
        
        # Get overlaps for current h
        overlaps = np.correlate(h_current, 1 - h_current, mode='full')
        
        # Find worst-case overlap (index)
        max_overlap_idx = np.argmax(overlaps)
        max_overlap_val = overlaps[max_overlap_idx]
        
        # Finish optimization if constraint satisfied
        if max_overlap_val * dx <= c5_upper:
            continue
            
        # Approximate problem with QP:
        # min  C5
        # s.t., sum(h) = 1, 0 <= h <= 1,
        #       sum(h_i * (1 - h_{i+k})) <= C5 for all k
        
        # Let's use a simpler optimization: we optimize h using L-BFGS
        # Re-formulate with a quadratic penalty
        
        # Create a robust coarse optimization attempt:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        # Constraints for optimization
        def sum_constraint(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': sum_constraint}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Local refine with scipy
        try:
            res = minimize(obj, h_current.copy(), method='SLSQP', 
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success and res.fun < current_c5:
                h_current = res.x
        except:
            pass
            
        # Better: try a simple projected gradient method for a few steps
        h_proj = h_current.copy()
        for step in range(10):
            # Compute gradient (approximately by finite differences)
            eps = 1e-6
            grad = np.zeros_like(h_proj)
            
            for i in range(n_points):
                h_plus = h_proj.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_proj.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                grad[i] = (evaluate_erdos_solution(h_plus, dx) - 
                           evaluate_erdos_solution(h_minus, dx)) / (2 * eps)
            
            # Gradient step
            step_size = 1e-3
            h_proj = h_proj - step_size * grad
            # Project to feasible set
            h_proj = np.clip(h_proj, 0, 1)
            h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
            
        if evaluate_erdos_solution(h_proj, dx) < current_c5:
            h_current = h_proj
            
    # Final refinement
    remaining_time = max_time - (time() - start_time)
    if remaining_time > 30:
        h_final = best_h.copy()
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        def con_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': con_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, h_final, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = obj(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass

    return (list(best_h), best_c5, n_points)
