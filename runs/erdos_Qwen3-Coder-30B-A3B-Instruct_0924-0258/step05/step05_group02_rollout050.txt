import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Get initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid optimization (n=20) to get initial feasible solution
    n_coarse = 20
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.ones(n_coarse) * 0.5
    h_coarse = h_coarse * (n_coarse/2) / np.sum(h_coarse)
    
    # If we can solve convex relaxation and get good starting point
    try:
        h_relaxed = cp.Variable(n_coarse)
        constraints = [
            cp.sum(h_relaxed) == n_coarse / 2.0,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        # Add constraints for a few shifts to keep size reasonable
        for k in range(1, min(8, n_coarse)):  
            corr_sum = cp.sum(cp.multiply(h_relaxed[:n_coarse-k], 1 - h_relaxed[k:]))
            constraints.append(corr_sum <= best_c5 * dx_coarse + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7)
        if prob.status == cp.OPTIMAL:
            h_coarse = h_relaxed.value
    except Exception:
        pass
    
    # Interpolate to full resolution
    h_interp = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
    h_interp = h_interp * (n_points/2) / np.sum(h_interp)
    h_interp = np.clip(h_interp, 0, 1)
    
    best_h = h_interp.copy()
    best_c5 = evaluate_erdos_solution(h_interp, dx)
    
    # Local refinement with SLSQP 
    max_iter = 150
    iter_count = 0
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Perturb current solution
        h_perturbed = best_h.copy()
        h_perturbed += np.random.normal(0, 0.03, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Local optimization step
        try:
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-7}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    last_improvement = 0
                else:
                    last_improvement += 1
            else:
                last_improvement += 1
        except Exception:
            last_improvement += 1
            
        # Restart with new random point if stuck
        if last_improvement > 20:
            h_restart = np.random.rand(n_points)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            h_restart = np.clip(h_restart, 0, 1)
            best_h = h_restart
            best_c5 = evaluate_erdos_solution(h_restart, dx)
            last_improvement = 0
            
        iter_count += 1
    
    # Final cleanup to enforce constraints exactly
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    
    return (list(h_final), best_c5, n_points)
