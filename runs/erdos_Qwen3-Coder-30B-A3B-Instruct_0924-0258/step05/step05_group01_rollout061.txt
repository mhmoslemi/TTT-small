import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based evolutionary approach
    pop_size = 50
    max_gen = 1000
    mutation_rate = 0.15
    elite_size = 5
    
    # Initialize population
    def create_individual():
        h = np.random.random(n_points) * 0.5 + 0.25  # Random in [0.25, 0.75]
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    population = [create_individual() for _ in range(pop_size)]
    population[0] = best_h.copy()  # Ensure we have at least one from parent
    
    # Ensure sum constraint and bounds
    def normalize(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def evaluate_individual(h):
        return evaluate_erdos_solution(h, dx)
    
    # Evolution loop
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time() - start_time < budget_s * 0.95 and generation < max_gen:
        # Evaluate fitness
        fitness_scores = [evaluate_individual(ind) for ind in population]
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        if fitness_scores[0] < best_c5:
            best_c5 = fitness_scores[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            # Restart with new random population but keep best
            population = [best_h.copy()] + [create_individual() for _ in range(pop_size - 1)]
            no_improvement_count = 0
        
        # Create new population
        new_population = population[:elite_size]  # Elitism
        
        # Tournament selection + crossover + mutation
        while len(new_population) < pop_size:
            # Tournament selection (size 3)
            tournament_indices = random.sample(range(elite_size, pop_size), 3)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Select another parent
            tournament_indices = random.sample(range(elite_size, pop_size), 3)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            parent2_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[winner_idx]
            parent2 = population[parent2_idx]
            
            # Crossover (uniform)
            mask = np.random.random(n_points) > 0.5
            child = np.where(mask, parent1, parent2)
            
            # Mutation
            if random.random() < mutation_rate:
                # Add Gaussian noise
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = normalize(child)
                
            new_population.append(child)
        
        population = new_population
        generation += 1
    
    # Final refinement using local optimization
    if time() - start_time < budget_s * 0.05:
        try:
            h_refined = best_h.copy()
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
