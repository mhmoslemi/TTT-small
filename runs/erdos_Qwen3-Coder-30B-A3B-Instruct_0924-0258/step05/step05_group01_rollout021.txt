import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Multi-scale optimization approach
    scales = [35, 40, 50, 60, 70]
    current_scale = 0
    
    while time() - start_time < budget_s * 0.95 and current_scale < len(scales):
        res = scales[current_scale]
        if res > n_points:
            # Interpolate to higher resolution
            h_new = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), best_h)
            n_points = res
            dx = 2.0 / n_points
        else:
            h_new = best_h
            n_points = res
            dx = 2.0 / n_points
            
        # Random restarts with different strategies
        max_restarts = 30
        h_current = h_new.copy()
        improvements = 0
        
        for i in range(max_restarts):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Try different initialization strategies
            strategy = i % 4
            if strategy == 0:
                # Random perturbation with clipping
                h_restart = h_current + np.random.normal(0, 0.05, n_points)
                h_restart = np.clip(h_restart, 0, 1)
            elif strategy == 1:
                # Start with best, then add noise in interesting regions
                h_restart = h_current.copy()
                for j in range(n_points):
                    if np.random.random() < 0.3:
                        h_restart[j] += np.random.normal(0, 0.03)
                h_restart = np.clip(h_restart, 0, 1)
            elif strategy == 2:
                # Initialize with pattern (interleaved high/low)
                h_restart = np.zeros(n_points)
                for j in range(n_points):
                    if j % 2 == 0:
                        h_restart[j] = np.random.uniform(0.1, 0.9)
                    else:
                        h_restart[j] = np.random.uniform(0.1, 0.9)
                h_restart = np.clip(h_restart, 0, 1)
            else:
                # Use a small random component
                h_restart = h_current + np.random.uniform(-0.1, 0.1, n_points) * 0.5
                h_restart = np.clip(h_restart, 0, 1)
            
            # Rescale to satisfy constraint
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            # SLSQP local optimization
            try:
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res_opt = minimize(objective, h_restart, method='SLSQP', 
                                bounds=bounds, constraints=cons_sum, 
                                options={'maxiter': 300, 'ftol': 1e-9})
                if res_opt.success:
                    h_refined = res_opt.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        improvements += 1
                        h_current = h_refined.copy()
                else:
                    h_current = h_restart.copy()
            except Exception:
                h_current = h_restart.copy()
        
        # If no improvement in many restarts, increase resolution or stop
        if improvements == 0 and current_scale < len(scales)-1:
            current_scale += 1
        else:
            current_scale += 1
    
    # Final refinement
    if time() - start_time < budget_s * 0.1:
        try:
            h_final = best_h.copy()
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except:
            pass
    
    # Final validation
    try:
        h_valid = best_h.copy()
        h_valid = np.clip(h_valid, 0, 1)
        h_valid = h_valid * (n_points/2) / np.sum(h_valid)
        c5_valid = evaluate_erdos_solution(h_valid, dx)
        if c5_valid < best_c5:
            best_h = h_valid
            best_c5 = c5_valid
    except:
        pass
        
    return (list(best_h), best_c5, n_points)
