```python
import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h from parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    best_h = h.copy()
    
    # CCP loop
    max_iter = 100
    stagnation_count = 0
    max_stagnation = 50
    
    # Shift sampling grid
    shift_grid = np.linspace(0, 2, 100)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Inner optimization: find k that maximizes overlap
        max_overlap = -np.inf
        best_k = 0
        
        # We'll compute overlaps over a finite set of shifts
        # and choose the maximum
        for k in shift_grid:
            # Compute overlap for given shift
            k_idx = int(k / 2 * n_points)
            if k_idx >= n_points:
                k_idx = n_points - 1
                
            # Use convolution for overlap computation
            h_shifted = np.roll(h, -k_idx)
            overlap = np.sum(h * (1 - h_shifted)) * (2.0 / n_points)
            
            if overlap > max_overlap:
                max_overlap = overlap
                best_k = k
                
        # Outer optimization using convex relaxation
        # Create variable for h
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # For convex relaxation, we'll approximate by fixing best_k
        # and using shifted version of h for computation
        
        # Select a representative small number of shifts
        test_shifts = np.linspace(0, 2, 20)
        total_overlap = 0
        
        # Approximate sum of overlaps for various shifts
        for shift in test_shifts:
            shift_idx = int(shift / 2 * n_points)
            if shift_idx >= n_points:
                shift_idx = n_points - 1
                
            if shift_idx >= 0:
                h_shifted = np.roll(h, -shift_idx)
                total_overlap += np.sum(h_var * (1 - h_shifted)) * (2.0 / n_points)
        
        # Minimize total overlap (we're doing dual optimization)
        # This is still a heuristic
        objective = cp.Minimize(total_overlap)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_new = np.array(h_var.value).flatten()
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, 2.0 / n_points)
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                    if stagnation_count > 10:
                        # Apply small random noise to escape local minima
                        noise = np.random.normal(0, 0.005, n_points)
                        h = np.clip(h + noise, 0, 1)
                        h = h * (n_points / 2) / np.sum(h)
                        stagnation_count = 0
                h = h_new.copy()
            else:
                # Fall back to simple update
                noise = np.random.normal(0, 0.005, n_points)
                h = np.clip(h + noise, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # If SDP fails, just perturb
            noise = np.random.normal(0, 0.005, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Early termination if target achieved
        if best_c5 < 0.3807:
            break
    
    # Final refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
```
