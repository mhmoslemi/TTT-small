import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 35
        h0 = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Define shift sampling
    shift_samples = np.linspace(0, 2, 100, endpoint=False)
    
    # Cutting-plane approach
    max_iter = 50
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    # Use CVXPY for convex relaxation
    h_var = cp.Variable(n_points)
    C5_var = cp.Variable()
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2,
        C5_var >= 0
    ]
    
    # Convex relaxation approach
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find worst shift from current h
        max_shift_idx = -1
        max_overlap_val = -1
        
        for i, shift in enumerate(shift_samples):
            # Compute overlap manually for this shift
            shift_int = int(shift * n_points / 2)
            shift_frac = shift * n_points / 2 - shift_int
            
            if shift_int >= n_points:
                continue
            if shift_int < 0:
                continue
                
            overlap = 0
            for j in range(n_points):
                k = (j + shift_int) % n_points
                overlap += best_h[j] * (1 - best_h[k])
            overlap += best_h[j] * (1 - best_h[(j + shift_int + 1) % n_points]) * shift_frac
            
            overlap *= 2.0 / n_points
            
            if overlap > max_overlap_val:
                max_overlap_val = overlap
                max_shift_idx = i
        
        # Add constraint for the worst shift
        if max_shift_idx != -1:
            shift_val = shift_samples[max_shift_idx]
            shift_int = int(shift_val * n_points / 2)
            shift_frac = shift_val * n_points / 2 - shift_int
            
            # Sum constraint
            sum_constraint = 0
            for j in range(n_points):
                k = (j + shift_int) % n_points
                sum_constraint += h_var[j] * (1 - h_var[k])
            if shift_frac > 0:
                k = (j + shift_int + 1) % n_points
                sum_constraint += h_var[j] * (1 - h_var[k]) * shift_frac
            
            # Scale factor
            sum_constraint *= 2.0 / n_points
            constraints.append(sum_constraint <= C5_var)
        
        try:
            # Solve the convex problem
            objective = cp.Minimize(C5_var)
            prob = cp.Problem(objective, constraints)
            
            # Try different solvers
            solver_options = {'maxit': 100}
            prob.solve(solver=cp.ECOS, **solver_options)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                # Extract the solution
                h_new = h_var.value
                c5_new = C5_var.value
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            else:
                # If convex optimization fails, fall back to SLSQP
                break
                
        except Exception as e:
            # On failure, use regular optimization
            break
            
        # Early termination
        if best_c5 < 0.3807:
            break
    
    # Do local refinement
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
