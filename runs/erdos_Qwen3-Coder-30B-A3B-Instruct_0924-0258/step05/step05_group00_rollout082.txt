import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 50
    generations = 1000
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    def initialize_population(size, n):
        population = []
        if 'initial_h_values' in globals():
            population.append(h_start.copy())
            size -= 1
        for _ in range(size):
            h = np.random.rand(n)
            h = h * (n/2) / np.sum(h)
            population.append(h)
        return population
    
    # Selection using tournament
    def tournament_select(population, fitnesses, k):
        selected = []
        for _ in range(len(population)):
            tournament = random.sample(list(zip(population, fitnesses)), k)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0].copy())
        return selected
    
    # Crossover
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = random.randint(1, len(parent1) - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    # Mutation
    def mutate(individual, mutation_rate):
        mutated = individual.copy()
        for i in range(len(mutated)):
            if random.random() < mutation_rate:
                mutated[i] += np.random.normal(0, 0.02)
        # Clip and normalize
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points/2) / np.sum(mutated)
        return mutated
    
    # Main GA loop
    population = initialize_population(population_size, n_points)
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(ind, dx) for ind in population]
        
        # Track best
        for ind, fit in zip(population, fitnesses):
            if fit < best_c5:
                best_h = ind.copy()
                best_c5 = fit
        
        # Selection
        selected = tournament_select(population, fitnesses, tournament_size)
        
        # Crossover and mutation
        new_population = []
        
        # Elitism: carry over best individual
        best_idx = np.argmin(fitnesses)
        new_population.append(population[best_idx].copy())
        
        # Generate rest
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected, 2)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
        
        # Trim to exact size
        population = new_population[:population_size]
        
    # Adaptive resolution if needed
    if best_c5 > 0.3808 and time() - start_time < budget_s * 0.8:
        n_points = 40
        dx = 2.0 / n_points
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, len(best_h)), 
                            best_h)
        best_h = h_interp
        
        # Re-optimize with fewer points
        population = initialize_population(20, n_points)
        generations = 300
        
        for gen in range(generations):
            if time() - start_time > budget_s * 0.95:
                break
                
            fitnesses = [evaluate_erdos_solution(ind, dx) for ind in population]
            
            for ind, fit in zip(population, fitnesses):
                if fit < best_c5:
                    best_h = ind.copy()
                    best_c5 = fit
            
            selected = tournament_select(population, fitnesses, tournament_size)
            new_population = []
            best_idx = np.argmin(fitnesses)
            new_population.append(population[best_idx].copy())
            
            while len(new_population) < 20:
                parent1, parent2 = random.sample(selected, 2)
                child1, child2 = crossover(parent1, parent2)
                child1 = mutate(child1, mutation_rate)
                child2 = mutate(child2, mutation_rate)
                new_population.extend([child1, child2])
            
            population = new_population[:20]
    
    # Final SLSQP refinement
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
