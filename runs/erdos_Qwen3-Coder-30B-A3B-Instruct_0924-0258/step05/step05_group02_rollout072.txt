import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    else:
        # Start with a simple construction
        h_start = np.ones(n_points) * 0.5
        
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Helper to ensure sum constraint
    def ensure_sum_constraint(h):
        return h * (n_points / 2) / np.sum(h)
    
    # Gradient-based local optimization with constraints
    def optimize_local(h_init, max_iter=50):
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        for _ in range(max_iter):
            try:
                result = minimize(
                    obj, h_init, method='SLSQP',
                    bounds=bounds, constraints=cons,
                    options={'maxiter': 50, 'ftol': 1e-8}
                )
                if result.success:
                    h_init = result.x
                    h_init = np.clip(h_init, 0, 1)
                    h_init = ensure_sum_constraint(h_init)
                else:
                    break
            except Exception:
                break
        return h_init
    
    # Refinement loop
    iteration = 0
    max_iter = 100
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iter:
        # Local optimization
        h_refined = optimize_local(best_h.copy())
        
        # Evaluate
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Add some perturbation if stuck
        if no_improvement_count > 20:
            h_perturb = best_h + np.random.normal(0, 0.02, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = ensure_sum_constraint(h_perturb)
            c5_perturb = evaluate_erdos_solution(h_perturb, dx)
            
            if c5_perturb < best_c5:
                best_h = h_perturb
                best_c5 = c5_perturb
                no_improvement_count = 0
        
        iteration += 1
            
        # Occasionally sample to keep solution accurate
        if iteration % 10 == 0:
            h_adjusted = best_h.copy()
            h_adjusted = np.clip(h_adjusted, 0, 1)
            h_adjusted = ensure_sum_constraint(h_adjusted)
            c5_adjusted = evaluate_erdos_solution(h_adjusted, dx)
            if c5_adjusted < best_c5:
                best_h = h_adjusted
                best_c5 = c5_adjusted
        
        if best_c5 < 0.3807:
            break
    
    # Final sanitization
    final_h = np.clip(best_h, 0, 1)
    final_h = ensure_sum_constraint(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
