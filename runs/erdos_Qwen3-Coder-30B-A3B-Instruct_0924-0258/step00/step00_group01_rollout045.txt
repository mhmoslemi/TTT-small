import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    # Compute overlap integrals for all lags
    corr = signal.correlate(h, 1-h, mode='full')
    # The maximum overlap integral
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points, target_sum):
    # Ensure sum equals target_sum and values are in [0, 1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if np.isclose(current_sum, target_sum):
        return h
    # Adjust by scaling and clipping
    scale_factor = target_sum / current_sum
    h = h * scale_factor
    h = np.clip(h, 0, 1)
    # If still not exactly correct due to clipping, adjust one element
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Add to last element (should be safe due to clipping)
        h[-1] = h[-1] + diff
        h = np.clip(h, 0, 1)
    return h

def initialize_population(pop_size, n_points, initial_h, target_sum):
    population = []
    # Add initial solution
    population.append(initial_h.copy())
    # Add random solutions
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points, target_sum)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, tournament_size=3):
    selected_indices = []
    for _ in range(len(population)):
        # Draw randomly
        Tournament = np.random.choice(len(population), tournament_size)
        best_idx = Tournament[np.argmin([fitnesses[i] for i in Tournament])]
        selected_indices.append(best_idx)
    return selected_indices

def crossover(parent1, parent2, n_points):
    # Average crossover
    child1 = (parent1 + parent2) / 2.0
    child2 = (parent1 + parent2) / 2.0
    # Project to feasible domain
    child1 = project_to_feasible(child1, n_points, n_points/2.0)
    child2 = project_to_feasible(child2, n_points, n_points/2.0)
    return child1, child2

def mutate(h, mutation_rate, std_dev, n_points, target_sum):
    # Make a copy
    h_new = h.copy()
    for i in range(n_points):
        if np.random.rand() < mutation_rate:
            h_new[i] += np.random.normal(0, std_dev)
    # Project to feasible
    h_new = project_to_feasible(h_new, n_points, target_sum)
    return h_new
