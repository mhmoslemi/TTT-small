import numpy as np
from scipy import signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except:
        # Fallback: create initial simple construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.2
        h[1] = 0.8
        h[2] = 0.5
        h[-1] = 0.3
        h[-2] = 0.7
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
    
    def evaluate(h_vec):
        # Compute C5 using cross-correlation
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / len(h_vec)
        return max_corr * dx
    
    def project(h_vec):
        # Ensure sum(h) = n_points / 2
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        target_sum = len(h_vec) / 2.0
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    def perturb(h_vec, T):
        # Add Gaussian noise
        noise = np.random.normal(0, T * 0.1, len(h_vec))
        h_new = h_vec + noise
        return project(h_new)
    
    def adaptive_cooling(iteration, init_T, factor=0.99):
        return init_T * (factor ** iteration)
    
    C5_best = evaluate(h)
    best_h = h.copy()
    
    # Multi-resolution search with simulated annealing
    n_levels = 5  # 70 samples is our high resolution
    n_low = 10
    n_current = n_low
    
    # Original resolution levels
    resolutions = [n_low, 20, 40, 70]
    
    for level in range(len(resolutions)):
        n_current = resolutions[level]
        if level == 0:
            h = np.resize(h, n_current)
            h = project(h)
        else:
            # Interpolate from previous level
            old_h = h.copy()
            h = np.interp(np.linspace(0, n_current-1, n_current), 
                         np.linspace(0, len(old_h)-1, len(old_h)), old_h)
            h = project(h)
        
        # Simulated Annealing parameters
        T = 0.1
        max_iter = 500
        iter_no_improve = 0
        max_no_improve = 100
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_new = perturb(h, T)
            C5_new = evaluate(h_new)
            
            delta = C5_new - C5_best
            
            if delta < 0 or random.random() < np.exp(-delta / T):
                h = h_new
                if C5_new < C5_best:
                    C5_best = C5_new
                    best_h = h.copy()
                    iter_no_improve = 0
                else:
                    iter_no_improve += 1
            else:
                iter_no_improve += 1
            
            T = adaptive_cooling(iteration, 0.1, 0.99)
            if iter_no_improve > max_no_improve:
                break
    
    # Final validation
    C5_best = evaluate(best_h)
    return (best_h.tolist(), C5_best, len(best_h))
