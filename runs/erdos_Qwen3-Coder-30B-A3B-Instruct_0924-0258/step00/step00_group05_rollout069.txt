import numpy as np
import cvxpy as cp
from time import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except:
        # Default initialization
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    low, high = 0.0, 0.38085
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    
    start_time = time()
    
    # Binary search on C5
    while high - low > 1e-4 and time() - start_time < budget_s - 5:
        C_mid = (low + high) / 2.0
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = []
        
        # Sum constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Bound constraints
        for i in range(n_points):
            constraints.append(h_vars[i] >= 0)
            constraints.append(h_vars[i] <= 1)
        
        # Overlap constraints (relaxed)
        # Using linear relaxation: h[i]*(1-h[i+k]) <= (h[i] + 1 - h[i+k])/2
        for k in range(1, n_points):
            # Compute correlations
            corr = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                # Relaxed constraint: h[i]*(1-h[j]) <= (h[i] + 1 - h[j]) / 2
                corr += h_vars[i] * (1 - h_vars[j])
            
            if corr > 0:
                constraints.append(corr <= C_mid / dx)
        
        # Objective
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_vars.value
                if h_sol is not None:
                    # Ensure bounds
                    h_sol = np.clip(h_sol, 0, 1)
                    # Renormalize
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    new_c5 = evaluate_erdos_solution(h_sol)
                    if new_c5 < best_c5:
                        best_h = h_sol.copy()
                        best_c5 = new_c5
                    # Update binary search bounds
                    high = C_mid
                else:
                    low = C_mid
            else:
                low = C_mid
        except:
            low = C_mid
    
    # Refine with local optimization
    h_refined = best_h.copy()
    try:
        # Try using scipy optimization to refine
        from scipy.optimize import minimize
        def obj(h_vec):
            return evaluate_erdos_solution(h_vec)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        def constraint_bounds(h_vec):
            return np.concatenate([h_vec, 1 - h_vec])
        
        # Simplified local search
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        res = minimize(obj, h_refined, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if res.success:
            h_refined = res.x
            
        # Final evaluation
        final_c5 = evaluate_erdos_solution(h_refined)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_refined.copy()
    
    except:
        pass
    
    # Ensure final bounds
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
