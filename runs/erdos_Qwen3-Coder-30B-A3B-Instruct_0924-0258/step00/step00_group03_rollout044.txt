import numpy as np
import time
from scipy import signal

def evaluate_C5(h):
    """Evaluate the maximum overlap integral for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = signal.correlate(h, 1 - h, mode='full')
    # Take only the relevant part
    half = n - 1
    corr = corr[half:]
    # Maximum value times dx
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to ensure sum equals n_points/2 and values in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

    # Evaluate initial
    best_C5 = evaluate_C5(h)
    best_h = h.copy()
    
    # Parameters
    T0 = 1.0
    alpha = 0.999
    sigma0 = 0.1
    t = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        t += 1

        # Generate candidate by adding noise
        noise = np.random.normal(0, sigma, n_points)

        # Project to feasible region

        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):

            if C5_prime < best_C5:
                best_C5 = C5_prime
                best_h = h.copy()
    
    return (best_h.tolist(), best_C5, n_points)
