import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, n_points):
    """Evaluate C5 for given h"""
    dx = 2.0 / n_points
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # Take the maximum value and scale
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to satisfy integral constraint"""
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust values to make sum correct
    diff = target_sum - current_sum
    # Add diff/n_points to each element
    h = h + diff / n_points
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    alpha = 0.1
    tol = 1e-4
    best_C5 = float('inf')
    best_h = None

    # Initialize h
    if 'initial_h_values' in globals():

    else:
        # We'll just create a uniform distribution
        h = np.full(n_points, 0.5)
        h = project_h(h, n_points)
    
    start_time = time.time()
    iter_count = 0

    while iter_count < max_iter and (time.time() - start_time) < budget_s:
        current_time = time.time()
        if current_time - start_time > budget_s:
            break
            
        # Evaluate current solution
        current_C5 = evaluate_erdos_solution(h, n_points)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        # Find all indices where maximum occurs
        max_indices = np.where(np.abs(corr - max_corr) < 1e-10)[0]
        
        # Compute gradient (involves two-step process)
        # g[i] = sum over k where max is achieved of 
        # [ (1 - h[(i+k)%n_points]) - (1 - h[i]) ] 
        # = sum over k where max is achieved of 
        # [ h[i] - h[(i+k)%n_points] ]
        
        # Start with zero gradient
        grad = np.zeros(n_points)
        for k in max_indices:
            if k > n_points-1: 
                k = k - (n_points - 1)  # This may be wrong, but keep it simple
            # We need to compute contribution for all valid shifts
            for i in range(n_points):
                if k == 0:
                    # Not a valid lag; skip
                    continue
                j = (i + k) % n_points
                grad[i] += (h[i] - h[j]) * dx
        # Normalize gradient similar to how the objective was scaled  
        # Maybe better approach for constraint handling:
        # Early exit if we think we're already close:
        if abs(current_C5 - best_C5) < 1e-5:
            break
        
        # Update h
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_h(h_new, n_points)
        
        # Assess improvement
        new_C5 = evaluate_erdos_solution(h_new, n_points)
        improvement = current_C5 - new_C5
        if improvement > 0:
            h = h_new.copy()
            if improvement > tol:
                alpha *= 1.05  # increase a bit more
        else:
            # heuristic: if no improvement, perturb h slightly and retry
            h = np.clip(h + np.random.normal(0, 0.001, n_points), 0.0, 1.0)
            h = project_h(h, n_points)
            # possibly decrease alpha here as well? 
            alpha *= 0.95
        
        # Adaptive alpha
        if abs(improvement) < 1e-6:
            alpha *= 0.9
        
        iter_count += 1
        if iter_count % 50 == 0:
            elapsed = time.time() - start_time
            if elapsed > 0.9 * budget_s:
                break

    # Final validation
    if best_h is None:
        best_h = h.copy()
        best_C5 = evaluate_erdos_solution(best_h, n_points)
        
    # Final evaluation
    final_C5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), float(final_C5), n_points)
