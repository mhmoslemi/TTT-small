import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Fallback: create a uniform distribution
        n_points = 47
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)  # normalize
    
    dx = 2.0 / n_points
    max_iter = 5000
    start_time = time.time()
    
    def compute_C5(h):
        # Compute the overlap integral
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * dx
    
    def projection(h):
        # Project onto feasible set: 0 <= h[i] <= 1
        h = np.clip(h, 0, 1)
        # Rescale to satisfy sum condition
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Gradient descent with projected updates
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute approximate gradient via finite differences
        eps = 1e-3
        grad = np.zeros(n_points)
        
        # Compute C5 for current h
        base_val = compute_C5(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = projection(h_plus)
            val_plus = compute_C5(h_plus)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = projection(h_minus)
            val_minus = compute_C5(h_minus)
            
            grad[j] = (val_plus - val_minus) / (2 * eps)
        
        # Determine step size using line search
        alpha = 0.1
        step_size = alpha
        best_update = h.copy()
        best_update_C5 = best_C5
        
        for _ in range(10):  # At most 10 line search steps
            h_new = h - step_size * grad
            h_new = projection(h_new)
            new_C5 = compute_C5(h_new)
            
            if new_C5 < best_update_C5:
                best_update = h_new.copy()
                best_update_C5 = new_C5
                step_size *= 1.5
            else:
                step_size *= 0.5
                
        h = best_update.copy()
        if best_update_C5 < best_C5:
            best_C5 = best_update_C5
            best_h = h.copy()
    
    return (best_h.tolist(), best_C5, n_points)
