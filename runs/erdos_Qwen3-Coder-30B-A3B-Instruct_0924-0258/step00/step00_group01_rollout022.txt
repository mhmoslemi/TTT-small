import numpy as np
from scipy import signal
import time
import random

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    # Compute correlations for all lags
    corr = signal.correlate(h, 1 - h, mode='full')
    # The valid lags are from -len(h)+1 to len(h)-1
    # But we want to match the indexing of the problem
    # Let's extract the meaningful part
    n = len(h)
    # The correlation has length 2*n - 1
    # The indices correspond to lags from - (n-1) to n-1
    # The overlaps we care about are for lags 0 to n-1
    # But we have to be careful about the indexing (as in problem statement)
    # Actually, we're using corr[0:2*n-1] with lag indices from -(n-1) to n-1
    # We need max for lags 0 to n-1 (which corresponds to indices n-1 to 2*n-2 in corr)
    
    # So we only consider the slice from n-1 to 2*n-2 since that's the lags from 0 to n-1
    # But looking at the problem carefully, it seems we want:
    # max_k int h(x)(1 - h(x+k)) dx
    # which for discrete h is max_k sum_i h[i] * (1 - h[(i+k)%n])
    # The standard correlate gives us efficient computation of all such sums
    # The valid overlapping index range is where we have both h[i] and h[i+k]
    # In the context of 'full' mode this means we should extract the middle part
    # of length n, specifically indices from n-1 to 2*n-2 of the full correlation
    
    max_corr = np.max(corr[n-1:2*n-1])
    return max_corr * dx

def project_feasible(h, n_points):
    """Ensure h is feasible: (1) all values in [0,1], (2) sum = n_points/2."""
    # Clip h to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale so that sum is exactly n_points / 2
    current_sum = np.sum(h)
    desired_sum = n_points / 2
    if current_sum > 0:
        h = h * desired_sum / current_sum
    return h

def crossover(parent1, parent2):
    """Simple average crossover."""
    child1 = (parent1 + parent2) / 2.0
    child2 = (parent1 + parent2) / 2.0
    return child1, child2

def mutate(h, mutation_rate=0.05, sigma=0.1):
    """Apply mutation to a candidate h."""
    n_points = len(h)
    # Create copy
    h_new = np.copy(h)
    # Flip some bits (Gaussian noise)
    noise = np.random.normal(0, sigma, size=n_points)
    h_new += noise
    # Apply mutation rate
    mask = np.random.rand(n_points) < mutation_rate
    h_new[mask] = h_new[mask] + np.random.normal(0, sigma, size=np.sum(mask))
    # Project back to feasible
    h_new = project_feasible(h_new, n_points)
    return h_new

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select two parents via tournament selection."""
    # Select two candidates from population by tournament
    selected = []
    for _ in range(2):
        # Pick k candidates at random
        tournament_indices = np.random.choice(len(population), size=tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(winner_idx)
    return selected[0], selected[1]

    mutation_rate = 0.05
    sigma = 0.1

    # Initialize population
    population = [h0.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)

        population.append(h)
    
    best_h = h0.copy()
    best_C5 = evaluate_h(best_h, dx)
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    
    # Keep track of best individual so far
    best_individual = best_h.copy()
    best_C5 = evaluate_h(best_individual, dx)
    
    while time.time() - start_time < budget_s and no_improvement_count < min_improvement_generations:

        # Evaluate all individuals in population
        fitnesses = []
        for h in population:
            C5 = evaluate_h(h, dx)
            fitnesses.append(C5)

        current_best = min(fitnesses)

            no_improvement_count += 1
        
        # Create next generation
        new_population = []
        # Elitism: keep best individual
        new_population.append(best_individual.copy())

        while len(new_population) < pop_size:
            parent1_idx, parent2_idx = tournament_selection(population, fitnesses, tournament_size=3)

            parent2 = population[parent2_idx]
            
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate, sigma)
            child2 = mutate(child2, mutation_rate, sigma)
            
            new_population.extend([child1, child2])
        
        # Trim population to exact size
        population = new_population[:pop_size]
        
    # Final evaluation
    final_C5 = evaluate_h(best_individual, dx)
    
    return (best_individual.tolist(), final_C5, len(best_individual))
