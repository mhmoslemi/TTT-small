import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction or create one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)
        h[0] = 0.4
        h[1] = 0.6
        # Normalize to exact sum
        h *= (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, n_points)
    
    # Gradient descent parameters
    max_iter = 500
    alpha = 0.1
    convergence_threshold = 1e-4
    alpha_decay = 0.95
    perturb_freq = 10
    perturb_scale = 0.05
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap integrals
        cross_correlation = np.correlate(h, 1 - h, mode='full')
        # The full correlation includes indices from -(n-1) to n-1
        # We're interested in lags 0 to n-1 (the positive lags)
        # max_k corresponds to forward shifts
        correlations = cross_correlation[n_points-1:] * dx
        
        # Find the maximum overlap and all lags that achieve it
        max_C5 = np.max(correlations)
        max_indices = np.where(np.abs(correlations - max_C5) < 1e-10)[0]
        
        # Compute gradient using only the maximizing lags
        grad = np.zeros(n_points)
        for k in max_indices:
            # For lag k, derivative with respect to h[i] is:
            # (1 - h[i+k]) + h[i+k-1] - (1 - h[i]) - h[i-1]
            # The full gradient contribution from overlap term:
            # 1 - h[i+k] - (1 - h[i]) = h[i] - h[i+k]
            # But k=0 case is special - when we shift by k=0, i+k=i
            # contribution is 1-h[i] - (1-h[i]) = 0
            # So shift by more than 0
            for i in range(n_points):
                # We will represent hang as circular at edges
                i_plus_k = (i + k) % n_points
                grad[i] += (h[i] - h[i_plus_k])
        
        grad *= dx
        
        # Gradient descent update
        h_new = h - alpha * grad
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new *= (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_C5 = evaluate_erdos_solution(h_new, n_points)
        
        # Accept improvement
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
            alpha *= alpha_decay
        else:
            # If no improvement, perturb and try again
            if iteration % perturb_freq == 0:
                # Random local perturbation
                h_perturbed = h + np.random.normal(0, perturb_scale, n_points)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed *= (n_points / 2) / np.sum(h_perturbed)
                new_C5_try = evaluate_erdos_solution(h_perturbed, n_points)
                if new_C5_try < best_C5:
                    h_new = h_perturbed
                    best_C5 = new_C5_try
                    best_h = h_new.copy()
                    
        h = h_new
        # Convergence check
        if abs(best_C5 - new_C5) < convergence_threshold:
            alpha *= alpha_decay
    
    return (best_h.tolist(), best_C5, n_points)

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    cross_correlation = np.correlate(h, 1 - h, mode='full')
    correlations = cross_correlation[n_points-1:] * dx
    return np.max(correlations)
