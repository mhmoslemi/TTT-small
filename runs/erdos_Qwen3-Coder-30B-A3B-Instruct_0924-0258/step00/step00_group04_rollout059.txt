import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_c5(h):
    """Evaluate C5 for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all k
    # Using numpy's correlate with mode="full"
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value over all lags
    c5 = np.max(correlation) * dx
    return c5

def project_and_rescale(h, n_points):
    """Project h to [0,1] and rescale to make sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2, n_points):
    """Perform crossover between two parents"""
    # Choose 2 random crossover points
    crossover_points = sorted(random.sample(range(1, n_points), 2))
    # Create offspring by combining segments
    child = np.zeros(n_points)
    child[:crossover_points[0]] = parent1[:crossover_points[0]]
    child[crossover_points[0]:crossover_points[1]] = parent2[crossover_points[0]:crossover_points[1]]
    child[crossover_points[1]:] = parent1[crossover_points[1]:]
    return project_and_rescale(child, n_points)

def mutate(h, n_points, mutation_rate=0.05):
    """Mutate h with given rate"""
    h = h.copy()
    indices_to_mutate = random.sample(range(n_points), int(n_points * mutation_rate))
    for i in indices_to_mutate:
        # Add Gaussian noise
        h[i] += np.random.normal(0, 0.1)
    return project_and_rescale(h, n_points)

def greedy_optimize(h, n_points):
    """Apply local optimization to h"""
    def objective(h_flat):
        h_vec = h_flat.reshape(-1)
        return evaluate_c5(h_vec)
    
    def constraint_func(h_flat):
        return np.sum(h_flat.reshape(-1)) - n_points / 2
    
    # Initialize optimization problem
    h_flat = h.reshape(-1)
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Use L-BFGS-B for local optimization
    try:
        result = minimize(objective, h_flat, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 200})
        h_opt = result.x.reshape(-1)
    except Exception:
        h_opt = h_flat.reshape(-1)
    
    # Ensure feasibility after optimization
    h_opt = project_and_rescale(h_opt, n_points)
    return h_opt
