import numpy as np
import time
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up discretization
    n_points = 47  # From initial_h_values
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if abs(np.sum(h) * dx - 1.0) > 1e-6:
            h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure constraints
    assert abs(np.sum(h) * dx - 1.0) < 1e-6
    assert all(0 <= x <= 1 for x in h)
    
    alpha = 0.1
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    no_improvement_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute overlaps using FWHT
        overlaps = compute_overlaps_fwht(h, n_points)
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        if max_overlap < best_C5:
            best_C5 = max_overlap
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Update step
        grad = compute_gradient(h, max_k, n_points)
        h += alpha * grad
        h = np.clip(h, 0, 1)
        
        # Maintain sum constraint
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
        # Adaptive learning rate
        if no_improvement_count % 50 == 0 and no_improvement_count > 0:
            alpha *= 0.95
            
        # Random perturbation to escape local minima
        if np.random.rand() < 0.05:
            perturbation = np.random.normal(0, 0.01, n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            
        # Stop if no improvement for too long
        if no_improvement_count > 2000:
            break
            
    # Final evaluation
    final_C5 = evaluate_C5(best_h, dx)
    
    return (best_h.tolist(), final_C5, n_points)

def compute_overlaps_fwht(h, n_points):
    # This uses FFT-based fast Walsh-Hadamard transform approach
    # We compute correlations via convolution in frequency domain
    one_minus_h = 1 - h
    # Padding for full convolution
    h_padded = np.pad(h, (0, n_points-1), mode='constant', constant_values=0)
    one_minus_h_padded = np.pad(one_minus_h, (0, n_points-1), mode='constant', constant_values=0)
    
    # FFT method for convolution
    H_fft = fft(h_padded)
    OneMinusH_fft = fft(one_minus_h_padded)
    convolution_fft = H_fft * np.conj(OneMinusH_fft)
    conv_result = ifft(convolution_fft).real
    # Extract full correlation (first n_points elements)
    overlaps = conv_result[:n_points]
    return overlaps

def compute_gradient(h, k_max, n_points):
    # Calculate gradient of overlap at k_max
    # For each i, gradient contribution from term h[i] * (1 - h[(i+k) % n_points])
    # Gradient w.r.t. h[i]: (1 - h[(i+k) % n_points]) - h[i] * 0 = 1 - h[(i+k) % n_points]
    # Gradient w.r.t. h[j] (where j != i): -h[i] for term h[i] * (1 - h[(i+k) % n_points])
    
    # Since these are specific to k_max
    grad = np.zeros(n_points)
    for i in range(n_points):
        # For correlation with lag k_max: 
        # i-th component appears in sum and changes with h[i]
        # The contribution to the cross correlation sum involving h[i] is:
        # 1 * (1 - h[(i+k) % n_points]) = (1 - h[(i+k) % n_points])
        j = (i + k_max) % n_points
        grad[i] = 1 - h[j]
        
    # The gradient estimation gives more accuracy
    # but for simplicity in program: we use just that direction.
    # However, we should also compute the full gradient w.r.t. h
    
    # Our original approach was more complete. Here's corrected one:
    # Define the indices that contribute to overlap at lag k
    # For a fixed k, we have h[i] * (1 - h[i+k]) 
    # which contributes to overlap correlation indexed by k
    
    # Correct way: 
    # For i-th coordinate, its impact on overlap at lag k is:
    # dynamic derivative of sum over all i of h[i] * (1 - h[i+k]) for fixed k
    # This involves: 
    # d/dh[i] of sum h[i] * (1 - h[i+k]) = (1 - h[i+k]) - h[i] * 0 = 1 - h[i+k]
    # But in full convolution it's more nuanced.
    
    # Our simpler gradient will be derived directly by correlation:
    # The contribution for overlap at lag k to the gradient of h[i]:
    # Here we work with one iteration efficiently for each element
    # Let's recompute properly:
    
    # counts for each position along with corresponding weights
    grad_sparse = np.zeros(n_points)
    # Confusing part: the gradient w.r.t. overlap w.r.t h[i]
    # h[i] contributes to sum only when used with 1-h[i+k] for some k 
    # When we are optimizing max_k, we only care about the contribution to that max_k
    
    for i in range(n_points):
        j = (i + k_max) % n_points
        grad_sparse[i] = 1 - h[j]  # This is correct when maximizing the overlap

    # When we take a step in this direction, and want to scale it,
    # we need to respect h[i] being in [0,1]
    return grad_sparse

def evaluate_C5(h, dx):
    # Fast calculation of maximum overlap
    # Use numpy correlate directly as per original formulation
    h_rev = np.flip(h)
    corr = np.correlate(h, h_rev, mode='full')
    corr = corr[len(corr)//2:]  # Full correlation shifted
    # Now we want: integral of h(x) * (1 - h(x+k))dx
    # But in discrete form, it is sum_h over i, of h[i] * (1 - h[i+k])
    # This is essentially corr_computed with h and (1-h)
    
    # Correct way:
    one_minus_h = 1 - h
    # Cross-correlation: h * (1-h)
    cross_corr = np.correlate(h, one_minus_h, mode='full')
    cross_corr = cross_corr[len(cross_corr)//2:]  # Just positive lags
    max_overlap = np.max(cross_corr) * dx
    return max_overlap
