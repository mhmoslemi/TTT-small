import time
import numpy as np
from scipy.signal import correlate
import math

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    # This computes the full correlation with 1-h
    corr = correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum of the correlation values
    # We need to be careful about indexing; the valid range [0, n_points) for the first array
    # and [0, n_points) for the second gives correlation for lags from -(n_points-1) to (n_points-1)
    # The maximum should include all relevant lags
    max_val = np.max(corr)
    return max_val * dx

def project_h(h, n_points):
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    # Correct the sum to be exactly n_points / 2
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-10:
        # Adjust by redistributing the difference
        adjustment = (target_sum - current_sum) / n_points
        h += adjustment
        # Clamp again
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 47  # Using the provided initial construction size
    max_iter = 10000
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.resize(h, n_points)
    else:
        h = 0.5 * np.ones(n_points)
    h = project_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    # Gradient descent parameters
    learning_rate = 0.1
    momentum = 0.9
    velocity = np.zeros_like(h)
    
    # Control parameters for learning rate decay and randomness
    iter_since_improvement = 0
    lr_decay_factor = 0.95
    random_restart_prob = 0.05
    random_restart_freq = 50
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        # Find the shift with the maximum overlap. The shift is k = i - (n_points-1)
        # So for each correlation index, the corresponding shift is (index - (n_points-1))
        # The correlation array has 2*n_points-1 elements; indices [0, n_points-1] correspond
        # to shifts [-n_points+1, 0], indices [n_points, 2*n_points-2] correspond to shifts [1, n_points-1]
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n_points - 1)
        
        # Compute gradient only for this worst-case shift
        # The gradient wrt h[i] is (1 - h[i + shift]) if i + shift < n_points
        # Or -h[i - shift] if i >= shift 
        gradient = np.zeros_like(h)
        if max_shift >= 0:
            # Gradient wrt h[i]: 1 - h[i + shift] for i in [0, n_points - shift)
            # and -h[i - shift] for i in [shift, n_points)
            # We compute it by turning the correlation relationship into a gradient
            # That is: if we add 1 to h[i], and check the effect on the overlap
            # It's equal to the contribution of that to the overlap
            for j in range(n_points - max_shift):
                gradient[j] += 1 - h[j + max_shift]
            for j in range(max_shift, n_points):
                gradient[j] -= h[j - max_shift]
        else:
            # Negative shift: max_shift is negative, i in [0, n_points + max_shift)
            abs_shift = -max_shift
            for j in range(n_points + max_shift):
                gradient[j] += 1 - h[j + abs_shift]
            for j in range(abs_shift, n_points):
                gradient[j] -= h[j - abs_shift]
                
        # Update with momentum and learning rate
        velocity = momentum * velocity - learning_rate * gradient
        h_new = h + velocity
        h_new = project_h(h_new, n_points)  # This maintains sum exactly
        
        # Check if we should have a random restart
        should_restart = (iter_since_improvement >= random_restart_freq and
                          np.random.rand() < random_restart_prob)
        if should_restart:
            h_new = np.random.rand(n_points) * 2
            h_new = project_h(h_new, n_points)
            iter_since_improvement = 0
            velocity = np.zeros_like(h)
        
        # Evaluate the new value
        new_c5 = evaluate_erdos_solution(h_new, n_points)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        # Update h
        h = h_new.copy()
        
        # Decay learning rate if no improvement for some time
        if iter_since_improvement % 20 == 0 and iter_since_improvement > 0:
            learning_rate *= lr_decay_factor
            
    # Return the best result found
    return (best_h.tolist(), best_c5, n_points)
