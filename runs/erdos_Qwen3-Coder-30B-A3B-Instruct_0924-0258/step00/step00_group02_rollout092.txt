import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for the given step function h."""
    dx = 2.0 / len(h)
    # Correlate h with (1 - h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap integral C_5 is the maximum of the correlation
    # times dx
    c5 = np.max(corr) * dx
    return c5

def project_to_feasible(h, n_points):
    """Project h to satisfy the constraints: h[i] in [0,1] and sum(h) == n_points/2."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Fallback for when no initial construction
        n_points = 45
        h_init = np.ones(n_points) * 0.5  # Simple uniform distribution
        
    # Sets the discretization
    dx = 2.0 / n_points
    
    # Initial parameters for PSO
    n_particles = 30
    w_init = 0.9
    c1 = 2.0
    c2 = 2.0
    v_max = 0.5
    
    # Initialize particles
    particles = []
    velocities = []
    
    # First particle is the initial solution
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Other particles are perturbed versions
    for _ in range(n_particles - 1):
        h_perturbed = h_init + np.random.uniform(-0.1, 0.1, n_points)
        h_perturbed = project_to_feasible(h_perturbed, n_points)
        particles.append(h_perturbed)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track best solutions
    pBest = [evaluate_erdos_solution(h) for h in particles]
    pBest_h = [h.copy() for h in particles]
    gBest = min(pBest)
    gBest_h = particles[pBest.index(gBest)].copy()
    
    # PSO main loop
    start_time = time.time()
    
    # Set up the inertia weight decay factor
    w_decay = 0.99
    
    for t in range(100000):  # Limit iterations but check time regularly
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate each particle
        for i in range(n_particles):
            h = particles[i]
            c5 = evaluate_erdos_solution(h)
            
            if c5 < pBest[i]:
                pBest[i] = c5
                pBest_h[i] = h.copy()
                
                if c5 < gBest:
                    gBest = c5
                    gBest_h = h.copy()
        
        # Update velocities and positions
        w = w_init * (w_decay ** t)
        
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            v = w * velocities[i] + c1 * r1 * (pBest_h[i] - particles[i]) + c2 * r2 * (gBest_h - particles[i])
            
            # Clamp velocity
            v = np.clip(v, -v_max, v_max)
            
            velocities[i] = v
            particles[i] = particles[i] + v
            
            # Apply feasibility projection
            particles[i] = project_to_feasible(particles[i], n_points)
    
    # Final check of the best solution found
    final_c5 = evaluate_erdos_solution(gBest_h)
    
    return (gBest_h.tolist(), final_c5, n_points)
