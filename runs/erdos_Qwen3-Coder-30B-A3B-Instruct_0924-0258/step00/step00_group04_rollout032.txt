import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import random

def evaluate_c5(h):
    """Compute C5 for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation for all lags
    corr = correlate(h, 1 - h, mode="full")
    # Extract the relevant part and scale
    c5 = np.max(corr) * dx
    return c5

def project_and_rescale(h, n_points):
    """Project h to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def crossover(h1, h2):
    """Perform crossover between two step functions."""
    n = len(h1)
    # Random number of crossover points (2 to 3)
    n_crossover = random.randint(2, 3)
    points = sorted(random.sample(range(1, n), n_crossover))
    points = [0] + points + [n]
    
    # Create new solution
    h_new = np.zeros(n)
    for i in range(len(points) - 1):
        start, end = points[i], points[i+1]
        if i % 2 == 0:
            h_new[start:end] = h1[start:end]
        else:
            h_new[start:end] = h2[start:end]
    
    return h_new

def mutate(h, mutation_rate=0.05):
    """Mutate a step function."""
    n = len(h)
    n_mutate = max(1, int(n * mutation_rate))
    indices = random.sample(range(n), n_mutate)
    for i in indices:
        # Add Gaussian noise
        h[i] += np.random.normal(0, 0.1)
    return project_and_rescale(h, n)

def greedy_optimize(h, n_points):
    """Apply L-BFGS-B to refine the solution."""
    def objective(h_flat):
        h_vec = h_flat.reshape(-1)
        return evaluate_c5(h_vec)
    
    def gradient(h_flat):
        # Simple gradient approximation
        eps = 1e-6
        grad = np.zeros_like(h_flat)
        for i in range(len(h_flat)):
            h_plus = h_flat.copy()
            h_plus[i] += eps
            h_minus = h_flat.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad

    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=gradient, options={'maxiter': 20})
    return project_and_rescale(result.x.reshape(-1), n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Start with reasonable size
    pop_size = 50
    max_generations = 200
    mutation_rate = 0.05
    stagnation_threshold = 50
    stagnant_generations = 0
    best_c5 = float('inf')
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_and_rescale(h0, n_points)
        population.append(h0)
    else:
        # Generate random starting solutions
        h0 = np.random.rand(n_points)
        h0 = project_and_rescale(h0, n_points)
        population.append(h0)
    
    # Generate remaining population
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_and_rescale(h, n_points)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h)
    
    generation = 0
    while time.time() - start_time < budget_s - 2:
        if generation >= max_generations:
            break
            
        # Evaluate population
        fitness = [evaluate_c5(h) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnant_generations = 0
        else:
            stagnant_generations += 1
        
        # Adjust mutation rate if stagnating
        if stagnant_generations > stagnation_threshold:
            mutation_rate = 0.1
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        parents = [population[i] for i in sorted_indices[:pop_size // 5]]
        
        # Generate offspring
        offspring = []
        for _ in range(pop_size - len(parents)):
            if len(parents) < 2:
                parent1 = parents[0]
                parent2 = parents[0]
            else:
                parent1, parent2 = random.sample(parents, 2)
            child = crossover(parent1, parent2)
            child = mutate(child, mutation_rate)
            offspring.append(child)
        
        # Add offspring to population
        population = parents + offspring
        
        # Local optimization on top 10%
        top_indices = sorted_indices[:pop_size // 10]
        for i in top_indices:
            if random.random() < 0.5:
                population[i] = greedy_optimize(population[i], n_points)
        
        generation += 1
    
    # Final evaluation of best solution
    best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
