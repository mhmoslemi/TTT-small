import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution h * (1 - h) with full mode
    conv = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum of convolution values scaled by dx
    c5 = np.max(conv) * dx
    return c5

def project_onto_feasible(h, n):
    """Project h onto the feasible region: 0 <= h[i] <= 1 and sum(h) = n/2."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n / 2) / current_sum
    return h

def convex_surrogate(h):
    """Compute a differentiable convex surrogate for the max overlap."""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full")
    # Quadratic loss on the maximum value
    max_val = np.max(conv)
    # Use squared difference from max to favor reducing the maximum
    # This encourages making the largest value smaller
    # Hessian approximates the quadratic increase away from the maximum
    loss = np.sum((conv - max_val)**2) * dx**2
    return loss

def optimize_at_resolution(h_start, n_points):
    """Optimize h at a given resolution n_points."""
    dx = 2.0 / n_points
    h = h_start.copy()
    
    # Define objective and constraints for scipy.optimize
    def objective(h_vec):
        h = h_vec.copy()
        # Rescale to ensure sum = n_points / 2
        h = project_onto_feasible(h, n_points)
        # Use surrogate objective that approximates the max overlap loss
        # Approximate the true objective with squared loss on the maximum overlap
        return convex_surrogate(h)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    def constraint_bounds(i):
        def c(h_vec):
            return h_vec[i] - 1  # h[i] <= 1
        return c
    
    # Construct constraints
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    for i in range(n_points):
        cons.append({'type': 'ineq', 'fun': constraint_bounds(i)})
    
    # Bounded variable limits
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimize using SLSQP
    res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-6}, 
                   tol=1e-8)
    
    if res.success:
        h_final = res.x
    else:
        # Fall back to feasible projection
        h_final = project_onto_feasible(h_start, n_points)
    
    # Verify and project
    h_final = project_onto_feasible(h_final, n_points)
    return h_final

def initialize_h(n_points):
    """Initialize h using initial_h_values or a default construction."""
    if 'initial_h_values' in globals():
        orig_h = np.array(initial_h_values)
        # Interpolate to new resolution
        x_orig = np.linspace(0, 2, len(orig_h))
        x_new = np.linspace(0, 2, n_points)
        h_init = np.interp(x_new, x_orig, orig_h)
    else:
        # Default starting construction
        h_init = np.ones(n_points) * 0.5
    return h_init

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution levels
    resolutions = [45]
    best_c5 = float('inf')
    best_h = None
    best_n = None

    # Strategy: Multi-resolution optimization
    for n_points in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Initialize at current resolution
        h = initialize_h(n_points)
        
        # Optimize this resolution
        h_opt = optimize_at_resolution(h, n_points)
        
        # Evaluate the refinement
        c5 = evaluate_erdos_solution(h_opt)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            best_n = n_points
            
        # Early termination if we're below threshold
        if best_c5 <= 0.3807:
            break
    
    # Final check with known constraint satisfaction
    if best_h is not None:
        final_c5 = evaluate_erdos_solution(best_h)
        best_h = project_onto_feasible(best_h, best_n)  # Extra safety
        return (best_h.tolist(), final_c5, best_n)
    else:
        # Fallback if no result was found
        h_default = initialize_h(45)
        h_default = project_onto_feasible(h_default, 45)
        c5_default = evaluate_erdos_solution(h_default)
        return (h_default.tolist(), c5_default, 45)
