import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    # Compute the overlap integral using convolution
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def convex_surrogate(h, n_points, dx):
    """Compute a convex surrogate of the maximum overlap"""
    corr = np.correlate(h, 1 - h, mode='full')
    # Use a quadratic penalty for the maximum term to approximate max
    max_val = np.max(corr)
    # Sum of squares to smooth: penalize large values in all correlation terms
    return np.sum(corr**2) * dx**2 + 0.1 * (max_val - 0.3)**2

def gradient_of_surrogate(h, n_points, dx):
    """Compute numerical gradient of the convex surrogate"""
    eps = 1e-8
    grad = np.zeros_like(h)
    base_obj = convex_surrogate(h, n_points, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        obj_plus = convex_surrogate(h_plus, n_points, dx)
        grad[i] = (obj_plus - base_obj) / eps
    return grad

def optimize_resolution(h_start, n_points, dx, time_limit):
    """Optimize a resolution level with time constraint"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use SLSQP with gradient of surrogate
    def obj(h):
        return convex_surrogate(h, n_points, dx)
    
    def grad(h):
        return gradient_of_surrogate(h, n_points, dx)
    
    # Constraints
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Perturb the starting point slightly before optimization
    h_perturbed = project_to_feasible(h_start + np.random.normal(0, 0.01, n_points), n_points)
    
    for _ in range(100):  # arbitrary limit to prevent runs forever
        if time() - start_time > time_limit:
            break
        
        try:
            res = minimize(obj, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons, jac=grad, 
                           options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_res = project_to_feasible(res.x, n_points)
                c5 = evaluate_erdos_solution(h_res, dx)
                if c5 < best_c5:
                    best_h = h_res.copy()
                    best_c5 = c5
        except:
            pass
        
        # Perturb for next round
        h_perturbed = project_to_feasible(best_h + np.random.normal(0, 0.02, n_points), n_points)
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        n0 = 45
        h_start = np.ones(n0) / 2
    
    # Resolution levels to try
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if necessary
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Time limit allocation per resolution
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 50)
        
        # Optimize at this resolution with several restarts
        best_h_level = h_start.copy()
        best_c5_level = evaluate_erdos_solution(h_start, dx)
        
        for _ in range(3):  # Multiple restarts to avoid local minima
            h_restart = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            h_opt, c5_opt = optimize_resolution(h_restart, n_points, dx, time_per_res / 3)
            if c5_opt < best_c5_level:
                best_c5_level = c5_opt
                best_h_level = h_opt.copy()
        
        # If improvement, store
        if best_c5_level < best_c5:
            best_h = best_h_level.copy()
            best_c5 = best_c5_level
            best_n_points = n_points
            
        h_start = best_h_level.copy()
    
    return (list(best_h), best_c5, best_n_points)
