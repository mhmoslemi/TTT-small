import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    corr = correlate(h, 1 - h, mode='full')
    # Only consider the valid range of lags
    # We want max of corr[center - (n-1): center + n]
    n = len(h)
    center = len(corr) // 2
    valid_corr = corr[center - n + 1:center + n]
    return np.max(valid_corr) * dx

def rescale(h, target_sum):
    """Rescale h to have target_sum while keeping bounds"""
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.full_like(h, target_sum / len(h))
    scaled = h * (target_sum / current_sum)
    return np.clip(scaled, 0, 1)

def crossover(parent1, parent2):
    """Single-point crossover"""
    n = len(parent1)
    if n <= 1:
        return parent1.copy(), parent2.copy()
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, delta=0.1):
    """Gaussian mutation with rescaling"""
    mutated = h.copy()
    n = len(h)
    for i in range(n):
        if np.random.random() < mutation_rate:
            mutated[i] += np.random.normal(0, delta)
    return rescale(mutated, np.sum(h))

def tournament_selection(population, fitnesses, k=3):
    """Select an individual using tournament selection"""
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values)
        n_points = len(h_parent)
    else:
        # Fallback construction
        n_points = 45
        h_parent = np.ones(n_points) * 0.5
        h_parent[0] = 1.0
        h_parent[-1] = 1.0
        h_parent = rescale(h_parent, n_points / 2)
    
    dx = 2.0 / n_points
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    max_generations = 200
    
    # Initialize population
    population = [h_parent.copy()]
    for _ in range(population_size - 1):
        if np.random.random() < 0.5:
            # Perturb the parent
            h_mut = h_parent + np.random.uniform(-0.2, 0.2, n_points)
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = rescale(h_mut, n_points / 2)
            population.append(h_mut)
        else:
            # Random construction
            h_rand = np.random.random(n_points)
            h_rand = rescale(h_rand, n_points / 2)
            population.append(h_rand)
    
    # Evaluation loop
    best_h = h_parent.copy()
    best_c5 = evaluate(h_parent, dx)
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        fitnesses = [evaluate(h, dx) for h in population]
        current_best_index = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_index]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_index].copy()
        
        # New generation
        new_population = []
        
        # Elitism: keep best
        new_population.append(best_h.copy())
        
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.random() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
                # Mutation
                child1 = mutate(child1, mutation_rate)
                child2 = mutate(child2, mutation_rate)
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Just mutate parents
                child1 = mutate(parent1, mutation_rate)
                child2 = mutate(parent2, mutation_rate)
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
        
        # Replace population
        population = new_population
    
    return (best_h.tolist(), best_c5, n_points)
