import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    # Evaluate the overlap integral C5
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for lags k = 0..n-1
    corr = correlate(h, 1 - h, mode="full")
    # Extract correlations for lags 0 to n-1
    corr_vals = corr[n - 1:n + n - 1] * dx
    return np.max(corr_vals)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42  # Use the initial construction size
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Because integral should be 1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        assert np.all(h >= 0) and np.all(h <= 1)
        assert abs(np.sum(h) - target_sum) < 1e-10
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)  # normalize

    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)

    iteration = 0
    # Time and convergence tracking
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1

        # Get all overlaps for all lags and find the worst one
        corr = correlate(h, 1 - h, mode="full")
        corr_vals = corr[n_points - 1:n_points + n_points - 1] * dx
        k_star = np.argmax(corr_vals)
        C5_now = corr_vals[k_star]

        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_star) % n_points
            grad[i] = 1 - 2 * h[j]

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-12:
            grad = grad / grad_norm
        else:
            # If gradient is near zero, break to avoid stalling
            break

        # Line search for step size
        alpha = 0.1
        backtrack_factor = 0.8
        max_backtracks = 10
        for _ in range(max_backtracks):
            h_new_unproj = h - alpha * grad
            h_new = np.clip(h_new_unproj, 0, 1)
            # Rescale to target sum
            h_new = h_new * target_sum / np.sum(h_new)
            C5_new = evaluate_erdos_solution(h_new)
            if C5_new < C5_now:
                h = h_new
                if C5_new < best_C5:
                    best_C5 = C5_new
                    best_h = h.copy()
                break
            alpha *= backtrack_factor
            if alpha < 1e-6:
                break
        else:
            # If line search failed, reduce alpha and try again
            h = h - 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            C5_new = evaluate_erdos_solution(h)
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()

    # Final validation
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_C5, n_points)
