import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Computes the maximum overlap integral for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Extract the valid range (shifts from -n+1 to n-1)
    # Correlation indices: 0 to 2*n-2
    # We want indices corresponding to k from -n+1 to n-1
    # The shift k corresponds to index k + n - 1
    valid_corr = corr[n-1:2*(n-1)+1]  # indices from n-1 to 2n-3 
    # Normalize by dx
    c5 = np.max(valid_corr) * dx
    return c5

def compute_fitness(h):
    """Return fitness which is the maximum overlap C5"""
    return evaluate_erdos_solution(h)

def project(h):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n/2"""
    n = len(h)
    h = np.clip(h, 0, 1)
    # Normalize sum to exactly n/2
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (n / 2) / h_sum
    return h

def mutate(h, sigma):
    """Add Gaussian noise and project to feasible region"""
    h_new = h + np.random.normal(0, sigma, len(h))
    return project(h_new)

def crossover(h1, h2, alpha=0.6):
    """Blend crossover with parameter alpha in [0.5, 1.0]"""
    beta = np.random.uniform(alpha, 1 - alpha)
    h_new = beta * h1 + (1 - beta) * h2
    return project(h_new)

def compute_gradient(h):
    """Compute gradient of the maximum overlap with respect to h"""
    n = len(h)
    # For computational efficiency, we'll use finite differences
    # This is a heuristic, but sufficient for this application
    eps = 1e-6
    grad = np.zeros_like(h)
    base_fitness = compute_fitness(h)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project(h_plus)
        fitness_plus = compute_fitness(h_plus)
        grad[i] = (fitness_plus - base_fitness) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Default construction: simplicity
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[10:20] = 1.0
        h[50:60] = 0.0
        h = project(h)
    
    best_h = h.copy()
    best_c5 = compute_fitness(best_h)
    
    # Population optimization
    pop_size = 100
    population = [project(h + np.random.normal(0, 0.05, n_points)) for _ in range(pop_size)]
    population[0] = h.copy()  # Ensure initial solution is in population
    
    generation = 0
    no_improvement = 0
    max_no_improvement = 200
    
    while time.time() - start_time < budget_s - 1 and no_improvement < max_no_improvement:
        generation += 1
        # Evaluate population
        fitnesses = [compute_fitness(individual) for individual in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Selection: tournament selection
        selected = []
        for _ in range(pop_size // 2):
            tournament_ids = np.random.choice(pop_size, 5, replace=False)
            tournament_fitness = [fitnesses[i] for i in tournament_ids]
            winner = tournament_ids[np.argmin(tournament_fitness)]
            selected.append(population[winner])
        
        # Crossover and mutation
        new_population = []
        for i in range(0, len(selected), 2):
            if i+1 < len(selected):
                child1 = crossover(selected[i], selected[i+1], alpha=0.6)
                child2 = crossover(selected[i+1], selected[i], alpha=0.6)
                new_population.extend([child1, child2])
            else:
                new_population.append(selected[i])
                
        # Apply mutation to new population
        for i in range(len(new_population)):
            new_population[i] = mutate(new_population[i], 0.05 * np.exp(-generation/50))
        
        # Local refinement (gradient step for best few)
        local_candidates = selected[:len(selected)//5]
        for candidate in local_candidates:
            if time.time() - start_time > budget_s - 1:
                break
            try:
                # Simple gradient step as a form of local search
                lr = 0.01 * np.exp(-generation/100)
                grad = compute_gradient(candidate)
                updated = candidate - lr * grad
                updated = project(updated)
                new_fitness = compute_fitness(updated)
                if new_fitness < compute_fitness(candidate):
                    new_population.append(updated)
                else:
                    new_population.append(candidate)
            except:
                new_population.append(candidate)
                
        # Keep best from parent and offspring
        combined = population + new_population
        combined_fitness = [compute_fitness(ind) for ind in combined]
        top_indices = np.argsort(combined_fitness)[:pop_size]
        population = [combined[i] for i in top_indices]
            
    return (best_h.tolist(), best_c5, n_points)
