import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is maximized over all shifts
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: clip to [0,1] and rescale to sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def convex_surrogate(h, n_points):
    """Compute a convex surrogate for the maximum overlap."""
    dx = 2.0 / n_points
    correlation = np.correlate(h, 1 - h, mode="full")
    # Quadratic penalty on the maximum value
    max_val = np.max(correlation)
    return np.sum(correlation**2) * dx**2 - max_val**2

def optimize_at_resolution(h0, n_points, time_limit):
    """Optimize at a specific resolution using convex surrogate."""
    dx = 2.0 / n_points
    start_time = time()
    
    # Define constraints for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
    
    def objective(h):
        # Use convex surrogate
        correlation = np.correlate(h, 1 - h, mode="full")
        # Include a quadratic penalty to emphasize maximizing the peak
        max_val = np.max(correlation)
        return np.sum(correlation**2) * dx**2 - max_val**2
    
    def gradient(h):
        # Approximate gradient for the surrogate
        correlation = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(correlation)
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Gradient of quadratic part
            grad[i] = 2 * dx**2 * np.sum((1 - h) * (h[i] - h[max_idx]))
        return grad
    
    # Start with the projected solution
    h = project_to_feasible(h0.copy(), n_points)
    
    # Use a custom projected gradient method
    best_h = h.copy()
    best_val = evaluate_erdos_solution(h)
    tolerance = 1e-6
    iter_count = 0
    
    while time() - start_time < time_limit and iter_count < 200:
        iter_count += 1
        # Evaluate at current h
        val = evaluate_erdos_solution(h)
        if val < best_val:
            best_h = h.copy()
            best_val = val
            
        # Perturb slightly and optimize
        perturb = np.random.normal(0, 0.01, n_points)
        h_pert = h + perturb
        h_pert = project_to_feasible(h_pert, n_points)
        
        # Quick local optimization with SLSQP
        try:
            res = minimize(objective, h_pert, method='SLSQP', bounds=bounds, 
                           constraints=constraint, options={'maxiter': 10})
            h_new = res.x
            h_new = project_to_feasible(h_new, n_points)
            
            # Accept if improving
            if evaluate_erdos_solution(h_new) < evaluate_erdos_solution(h):
                h = h_new
        except:
            pass
    
    return best_h, best_val
