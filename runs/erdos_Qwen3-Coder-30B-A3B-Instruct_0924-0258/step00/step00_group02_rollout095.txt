import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute overlap integral for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The overlap integral for shift k is conv[n-1+k]*dx
    # We want max_k conv[n-1+k] * dx  = max_k conv[n-1+k] * dx
    # The +1 is due to 0-indexing and padding
    max_overlap = np.max(conv[n-1:])
    return max_overlap * dx

def project_to_feasible(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    if np.sum(h) > 0:
        h = h * (n_points / 2) / np.sum(h)
    return h

def surrogate_objective(h, n_points, dx):
    # Surrogate for max_k integral of h(x)(1-h(x+k))
    # Use quadratic penalty: sum of all overlaps squared
    conv = np.correlate(h, 1 - h, mode="full")
    # Sum of squared overlaps with appropriate scaling
    return np.sum(conv**2) * dx**2

def surrogate_gradient(h, n_points, dx):
    # Simple gradient approximation for surrogate
    # d/dh_i of sum_k (h * (1-h))[k]^2
    conv = np.correlate(h, 1 - h, mode="full")
    # This is complex to compute analytically, so we'll just return zeros for now
    # A more refined method might compute this properly
    return np.zeros_like(h)

def optimize_level(h0, n_points, dx, max_iter=100):
    def obj(h):
        return surrogate_objective(h, n_points, dx)
    
    def jac(h):
        return surrogate_gradient(h, n_points, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(obj, h0, method='SLSQP', bounds=bounds, 
                      constraints=constraints, jac=jac, 
                      options={'maxiter': max_iter, 'disp': False})
    
    if result.success:
        return project_to_feasible(result.x, n_points)
    else:
        print(f"Optimization failed at level {n_points}")
        return project_to_feasible(h0, n_points)
