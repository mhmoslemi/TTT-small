import numpy as np
import time
from scipy import signal

def evaluate_C5(h):
    """Evaluate the maximum overlap integral for given h."""
    # Compute full convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Extract the relevant part and scale by dx = 2 / len(h)
    dx = 2.0 / len(h)
    # The maximum overlap occurs at the center of the convolution
    # but we want the maximum over all possible shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

def project_h(h):
    """Project h to satisfy sum(h) = n_points / 2 and bounds [0,1]."""
    n_points = len(h)
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a simple starting configuration
        n_points = 70
        h = np.ones(n_points) * 0.5  # Simple uniform distribution

    # Ensure feasibility of initial solution
    h = project_h(h)
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    # SA parameters
    T0 = 1.0
    alpha = 0.999
    sigma0 = 0.1
    t = 0
    start_time = time.time()
    
    # For logging best values
    log_interval = 100
    last_logged = 0
    
    while time.time() - start_time < budget_s:
        t += 1
        T = T0 * (alpha ** t)
        sigma = sigma0 * (alpha ** (t / 200))
        
        # Generate candidate by adding Gaussian noise and projecting
        h_candidate = h + np.random.normal(0, sigma, len(h))
        h_candidate = project_h(h_candidate)
        
        # Evaluate candidate
        C5_candidate = evaluate_C5(h_candidate)
        
        # Accept with probability based on temperature and energy difference
        if C5_candidate < best_C5 or np.random.rand() < np.exp(-(C5_candidate - best_C5) / T):
            h = h_candidate
            if C5_candidate < best_C5:
                best_h = h.copy()
                best_C5 = C5_candidate
        
        # Log best value occasionally
        if t - last_logged >= log_interval:
            last_logged = t
            
    return (best_h.tolist(), best_C5, len(best_h))
