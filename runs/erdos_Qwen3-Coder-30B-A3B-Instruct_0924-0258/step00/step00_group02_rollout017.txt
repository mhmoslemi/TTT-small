import numpy as np
from scipy import signal
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1 - h)
    corr = signal.correlate(h, 1 - h, mode="full")
    # The maximum value corresponds to the maximum overlap
    max_overlap = np.max(corr)
    # Scale by dx to convert to continuous integral
    c5 = max_overlap * dx
    return c5

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

    # Initialize swarm of particles
    n_particles = 30
    particles = []
    velocities = []
    pBest = []
    pBest_vals = []

    # First particle is the initial solution
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    pBest.append(h_init.copy())
    pBest_vals.append(evaluate_erdos_solution(h_init))

    # Generate other particles with small random perturbations
    for i in range(1, n_particles):
        h_perturb = h_init + np.random.normal(0, 0.1, n_points)
        h_perturb = project_to_feasible(h_perturb, n_points)
        particles.append(h_perturb)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        pBest.append(h_perturb.copy())
        pBest_vals.append(evaluate_erdos_solution(h_perturb))

    # Initialize global best
    gBest_idx = np.argmin(pBest_vals)
    gBest = pBest[gBest_idx]
    gBest_val = pBest_vals[gBest_idx]

    # PSO parameters
    omega = 0.9
    c1 = 2.0
    c2 = 2.0
    v_max = 0.5

    # Timing
    start_time = time.time()

    # Main PSO loop
    iteration = 0
    while time.time() - start_time < budget_s:
        # Update for each particle
        for i in range(n_particles):
            h = particles[i].copy()
            current_val = evaluate_erdos_solution(h)
            if current_val < pBest_vals[i]:
                pBest[i] = h.copy()
                pBest_vals[i] = current_val
            if current_val < gBest_val:
                gBest = h.copy()
                gBest_val = current_val

        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            # Update velocity
            v = omega * velocities[i] + c1 * r1 * (pBest[i] - particles[i]) + c2 * r2 * (gBest - particles[i])
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            velocities[i] = v
            # Update position
            new_h = particles[i] + v
            new_h = project_to_feasible(new_h, n_points)
            particles[i] = new_h

        # Update inertia weight (time-dependent decay)
        omega *= 0.99
        iteration += 1

    # Return the best solution found
    return (gBest, gBest_val, n_points)
