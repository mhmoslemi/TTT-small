import time
from scipy.optimize import minimize

    h = h * n_points / (2 * np.sum(h))
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    def compute_C5(h):
        dx = 2.0 / n_points
        correlation = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(correlation)
        return max_corr * dx
    
    def project_h(h):
        # Project onto [0,1]
        h = np.clip(h, 0, 1)
        # Adjust to maintain sum = n_points / 2
        current_sum = np.sum(h)
        if abs(current_sum - n_points / 2) > 1e-6:
            scale_factor = (n_points / 2) / current_sum
            h = h * scale_factor
        return h
    
    # Record best solution
    best_C5 = compute_C5(h)
    best_h = h.copy()
    
    # Gradient descent with momentum
    momentum = 0.9
    alpha = 0.1
    alpha_decay = 0.95
    no_improvement_count = 0
    max_no_improvement = 50
    
    iters = 0
    while time.time() - start_time < budget_s:
        iters += 1
        # Compute all overlaps
        h1 = 1 - h
        correlation = np.correlate(h, h1, mode="full")
        kmax = np.argmax(correlation)
        max_corr = correlation[kmax]
        
        # Find gradient for the worst case
        grad = np.zeros_like(h)
        
        if kmax <= n_points - 1:
            # Left-shifted overlap term
            for i in range(n_points - kmax):
                grad[i] += 1 - h[i + kmax]
        else:
            # Right-shifted overlap term
            k0 = kmax - (n_points - 1)
            for i in range(k0, n_points):
                grad[i] -= h[i - k0]
        
        # Update with momentum and step size
        if iters == 1:
            velocity = -alpha * grad
        else:
            velocity = momentum * velocity - alpha * grad
        
        h_new = h + velocity
        
        h_new = project_h(h_new)
        
        new_C5 = compute_C5(h_new)
        
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        h = h_new
        
        # Adaptive learning rate
        if no_improvement_count > 20:
            alpha *= alpha_decay
            no_improvement_count = 0
            
        if no_improvement_count > max_no_improvement:
            break
            
    # Final normalization
    best_h = project_h(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), float(final_C5), n_points)
