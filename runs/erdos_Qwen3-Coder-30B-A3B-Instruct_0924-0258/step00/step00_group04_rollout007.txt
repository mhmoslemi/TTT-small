import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_c5(h_values):
    n = len(h_values)
    dx = 2.0 / n
    # Compute the full convolution of h with (1-h)
    conv = np.correlate(h_values, 1 - np.array(h_values), mode='full')
    # The maximum value is the largest element of the convolution
    max_val = np.max(conv)
    return max_val * dx

def project_and_rescale(h, n_points):
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def crossover(parent1, parent2, n_points):
    # Crossover at 2 or 3 random points
    n_cuts = random.randint(2, 3)
    cuts = sorted(random.sample(range(1, n_points), n_cuts))
    children = [parent1.copy(), parent2.copy()]
    start = 0
    for i, cut in enumerate(cuts):
        if i % 2 == 0:
            children[0][start:cut] = parent2[start:cut]
            children[1][start:cut] = parent1[start:cut]
        start = cut
    if start < n_points:
        if len(cuts) % 2 == 0:
            children[0][start:] = parent2[start:]
            children[1][start:] = parent1[start:]
    return project_and_rescale(children[0], n_points), project_and_rescale(children[1], n_points)

def mutate(h, n_points):
    # Mutate 5-10% of elements
    num_mutate = random.randint(max(1, n_points//20), max(1, n_points//10))
    indices = random.sample(range(n_points), num_mutate)
    for i in indices:
        noise = random.gauss(0, 0.1)
        h[i] += noise
    return project_and_rescale(h, n_points)

def local_optimize(h, n_points):
    def obj(h_flat):
        # Flatten to 1D for optimization
        h_array = h_flat.reshape(-1)
        return evaluate_c5(h_array)
    
    def constraint_sum(h_flat):
        h_array = h_flat.reshape(-1)
        return np.sum(h_array) - n_points / 2
    
    def constraint_bounds(h_flat):
        h_array = h_flat.reshape(-1)
        return np.concatenate([h_array, 1 - h_array])  # 0 <= h <= 1
    
    h_original = h.copy()
    h_flat = h_original.flatten()
    
    # Use L-BFGS-B with bounds for multiple restarts
    bounds = [(0, 1)] * n_points
    opt_result = minimize(obj, h_flat, method='L-BFGS-B', bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-10})
    
    if opt_result.success:
        h_opt = opt_result.x.reshape(-1)
        return project_and_rescale(h_opt, n_points)
    else:
        return h_original

def run(seed=42, budget_s=600):
    random.seed(seed)
    np.random.seed(seed)
    
    # Parameters
    n_points = 42  # Start with known good size
    pop_size = 50
    max_gens = 200
    time_start = time.time()
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = project_and_rescale(h_base, n_points)
        population.append(h_base.copy())
        for _ in range(pop_size - 1):
            h = np.random.random(n_points)
            h = project_and_rescale(h, n_points)
            population.append(h)
    else:
        for _ in range(pop_size):
            h = np.random.random(n_points)
            h = project_and_rescale(h, n_points)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    stagnant = 0
    mutation_rate = 0.05
    
    while time.time() - time_start < budget_s - 1 and generation < max_gens:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h)
            fitness.append(c5)
        
        # Find best
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_c5 = fitness[min_idx]
            best_h = population[min_idx].copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Adjust mutation rate if stagnation
        if stagnant > 50:
            mutation_rate = 0.1
        
        # Sort and select top 20%
        sorted_ids = np.argsort(fitness)
        parents = [population[i] for i in sorted_ids[:pop_size//5]]
        
        # Generate offspring
        offspring = []
        while len(offspring) < pop_size // 2:
            # Crossover pairs
            p1 = random.choice(parents)
            p2 = random.choice(parents)
            c1, c2 = crossover(p1, p2, n_points)
            
            # Mutate
            if random.random() < 0.5:
                c1 = mutate(c1, n_points)
            if random.random() < 0.5:
                c2 = mutate(c2, n_points)
            
            offspring.append(c1)
            if len(offspring) < pop_size // 2:
                offspring.append(c2)
        
        # Local optimization
        top_for_opt = parents[:pop_size//10]
        for h in top_for_opt:
            if random.random() < 0.8:
                h_local = local_optimize(h, n_points)
                offspring.append(h_local)
        
        # Replace population
        population = parents + offspring
        population = population[:pop_size]
        
        generation += 1
    
    # Final best check
    final_fitness = [evaluate_c5(h) for h in population]
    min_idx = np.argmin(final_fitness)
    if final_fitness[min_idx] < best_c5:
        best_c5 = final_fitness[min_idx]
        best_h = population[min_idx]
    
    return (best_h.tolist(), best_c5, n_points)
