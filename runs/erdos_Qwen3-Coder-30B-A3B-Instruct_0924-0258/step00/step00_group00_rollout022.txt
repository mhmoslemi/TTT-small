import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral for h."""
    # Compute full correlation between h and (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # The valid range of lags is from -(n-1) to (n-1) but we're interested
    # in the maximum among the valid lags. There are 2*n-1 lags.
    # The kernel is of length n, so for each lag k, we compute the sum
    # of h[i] * (1 - h[i+k]) over valid i.
    # In the full correlation, the middle index is at n-1. 
    # For k in range(n-1), we have lag k and -k.
    # The valid lag range we're actually looking at is [0, n-1] for shifts, and
    # we reflect it appropriately.
    # A better approach: compute only the maximum cross-correlation term,
    # normalized by dx. The maximum of these values is C5.
    max_corr_val = np.max(corr) * dx
    return max_corr_val

def ensure_sum_constraint(h, target_sum):
    """Adjust h to make sum equal to target_sum while keeping bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Rescale to target sum while preserving order/relative spacing
    # Simple rescale: h = h * target_sum / current_sum
    h = h * (target_sum / current_sum)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Again rescale to target sum
    h = h * (target_sum / np.sum(h))
    return h

def mutate(h, mutation_rate=0.1, delta=0.1):
    """Mutate h by small random changes."""
    h_new = h.copy()
    mask = np.random.rand(len(h)) < mutation_rate
    h_new[mask] += np.random.uniform(-delta, delta, size=np.sum(mask))
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Rescale to maintain sum
    h_new = ensure_sum_constraint(h_new, np.sum(h))
    return h_new

def crossover(p1, p2):
    """Perform one-point crossover."""
    n = len(p1)
    if n < 2:
        return p1.copy(), p2.copy()
    point = np.random.randint(1, n)
    c1 = np.concatenate([p1[:point], p2[point:]])
    c2 = np.concatenate([p2[:point], p1[point:]])
    # Ensure sum constraint
    c1 = ensure_sum_constraint(c1, np.sum(p1))
    c2 = ensure_sum_constraint(c2, np.sum(p1))
    return c1, c2

def tournament_selection(population, fitnesses, k=3):
    """Select a parent using tournament selection."""
    idxs = np.random.choice(len(population), size=k, replace=False)
    winners = [i for i in idxs if fitnesses[i] == np.min([fitnesses[j] for j in idxs])]
    return population[np.random.choice(winners)]

            fitnesses.append(fit)

            # Selection
            p1 = tournament_selection(population, fitnesses, k=tournament_k)
            p2 = tournament_selection(population, fitnesses, k=tournament_k)

                if len(new_pop) < pop_size:
