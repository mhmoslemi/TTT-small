import numpy as np
from scipy import signal
import time

def evaluate_h(h, dx):
    # Compute the maximum overlap integral
    corr = signal.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a default starting point
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2 and 0 <= h[i] <= 1
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
        
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        # Compute correlations and find maximum
        corr = signal.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k_offset = max_idx - (n_points - 1)
        
        # Gradient for the maximum correlation
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k_offset
            if 0 <= j < n_points:
                grad[i] = -1.0  # Simplified gradient
        
        # Allocate points for descent direction
        s_indices = np.argsort(grad)[-n_points//2:]
        s = np.zeros_like(h)
        s[s_indices] = 1.0
        
        # Line search to find the best step size
        gamma_vals = [0.05, 0.1, 0.25, 0.5, 0.75, 1.0]
        best_gamma = 0.0
        best_w = h.copy()
        best_w_c5 = best_c5
        
        for gamma in gamma_vals:
            w = h + gamma * (s - h)
            w = np.clip(w, 0, 1)
            w_sum = np.sum(w)
            if w_sum > 0:
                w = w * (n_points / 2) / w_sum
            w_c5 = evaluate_h(w, dx)
            if w_c5 < best_w_c5:
                best_w_c5 = w_c5
                best_w = w.copy()
                best_gamma = gamma
                
        # Update h
        h = best_w
        if best_w_c5 < best_c5:
            best_c5 = best_w_c5
            best_h = h.copy()
            
        # Occasionally restart with a random perturbation
        if iteration % 10 == 0:
            h = np.clip(h + np.random.normal(0, 0.02, n_points), 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points / 2) / h_sum
        
    return (best_h.tolist(), best_c5, n_points)
