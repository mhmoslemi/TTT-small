import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        h = np.full(n_points, 0.5)
    
    # Ensure sum equals n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Validate constraints
    assert np.all(h >= 0) and np.all(h <= 1)
    assert abs(np.sum(h) - n_points / 2) < 1e-10
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    # Main optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute overlaps for all lags k
        h_rev = np.flip(h)
        corrs = correlate(h, h_rev, mode='full')  # This gives sum_h[i] * h_rev[j] for all i,j
        # For overlap integral we want: h[i] * (1 - h[(i + k) % n])  
        # So each correlation includes: sum(h[i] * h[(i+j)%n])
        # We actually need: sum(h[i] * (1 - h[(i+k)%n]))
        # = sum(h[i]) - sum(h[i] * h[(i+k)%n]) = sum(h) - corr[k]
        
        # Compute overlap integrals
        overlap_vals = []
        for k in range(n_points):
            # Overlap for lag k: int h(x)(1-h(x+k))dx
            corr_val = np.sum(h * np.roll(h, k))  # h[i] * h[i-k]
            overlap = np.sum(h) - corr_val  # sum(h) - sum(h[i]*h[(i+k)%n])
            overlap_vals.append(overlap)
        
        # Scale by dx = 2/n_points
        overlap_vals = np.array(overlap_vals) * (2.0 / n_points)
        
        # Find the worst overlap (largest value)
        max_k = np.argmax(overlap_vals)
        max_val = overlap_vals[max_k]
        
        # Compute gradient for this worst lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            # grad[i] = d/dh[i] sum_j h[j] * (1 - h[(j+k)%n]) 
            # = sum_j d/dh[i] h[j] * (1 - h[(j+k)%n])
            # = (1 - h[(i+k)%n] - h[(i-k)%n]) if i is involved in corr else 0
            # Actually we're looking at corr[i] = sum_j h[j]*h[(j+i)%n] 
            # So d/dh[i] = sum_j d/dh[i] h[j]*h[(j+i)%n] 
            
            # For fixed k, d/dh[i] { h[i] * (1 - h[i+k]) } = 1 - h[(i+k)%n]  if j = i
            # But it's more complex since h[i] appears in several terms
            
            # Let's recompute correctly:
            # In sum_j h[j] * (1 - h[(j+k)%n])
            # d/dh[i] = (1 - h[(i+k)%n]) - h[(i-k)%n]
            
            k_shift = (i + max_k) % n_points
            k_neg_shift = (i - max_k) % n_points
            grad[i] = 1 - h[k_shift] - h[k_neg_shift]
        
        # Normalize gradient
        grad = grad * (2.0 / n_points)
        # Update with step size
        step_size = 0.1
        h_new = h - step_size * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        # Rescale to satisfy sum(h) = n_points / 2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if this improved the minimum overlap
        c5_new = evaluate_erdos_solution(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new.copy()
        
        # Occasionally add small random noise to escape local minima
        if iter_count % 50 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return best_h, best_c5, n_points

def evaluate_erdos_solution(h):
    # Compute the overlap integral C5 = max_k int h(x) (1 - h(x+k)) dx 
    n_points = len(h)
    dx = 2.0 / n_points
    h_arr = np.array(h)
    # At this point if we have: h[i] (1 - h[(i+shift)%n])
    # which is actually sum_i h[i] * (1 - h[i+shift]) 
    
    h_rev = np.flip(h_arr)
    corr = correlate(h_arr, np.flip(h_arr), mode='full')  # match scipy.signal.correlate behavior
    
    # Compute overlaps for all shifts
    overlaps = []
    for k in range(n_points):
        # Correlation k is: sum_i h[i] * h[i-k]
        # So overlap = sum_i h[i] * (1 - h[i-k]) = sum(h) - corr[k]
        overlap = 2 - corr[n_points-1+k]  # Adjust for full correlation
        overlaps.append(overlap)
        
    # Direct calculation using numpy arrays for robustness
    overlaps = []
    for k in range(n_points):
        # Shift h by k
        shifted = np.roll(h_arr, k)
        # Compute int h(x) * (1 - h(x+k)) dx = sum(h) - sum(h[i] * h[i+k])
        integral = np.sum(h_arr) - np.sum(h_arr * shifted)
        overlaps.append(integral)
    
    # Multiply by dx = 2/n_points to make it consistent with the integral scaling
    overlaps = np.array(overlaps) * (2.0 / n_points)
    return np.max(overlaps)
