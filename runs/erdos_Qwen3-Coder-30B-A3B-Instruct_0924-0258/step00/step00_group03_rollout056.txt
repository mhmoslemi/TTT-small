import numpy as np
import time
from scipy import signal

def evaluate(h):
    # Evaluate the overlap integral for a given step function h
    # h is a numpy array of values in [0,1]
    # We want to maximize over all k: ∫ h(x)(1 - h(x+k)) dx
    # Which is equivalent to maximizing over all lags: np.correlate(h, 1-h, mode="full")
    dx = 2.0 / len(h)
    conv = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(conv) * dx
    return max_val

def feasible(h):
    # Check if h satisfies constraints: 0 <= h[i] <= 1 and sum(h) = n/2
    n = len(h)
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - n/2) < 1e-10)

def normalize(h):
    # Normalize h to satisfy ∫ h(x) dx = 1 (i.e. sum(h) = n/2)
    n = len(h)
    s = np.sum(h)
    if s > 0:
        h = h * (n/2) / s
    return h

def mutate(h, rate=0.1):
    # Mutate h by adding noise and normalizing
    # First create a copy to avoid modifying input
    h = h.copy()
    # Add noise only to some elements
    mask = np.random.rand(len(h)) < rate
    noise = np.random.normal(0, 0.05, len(h)) * mask
    h += noise
    # Project back to [0,1]
    h = np.clip(h, 0, 1)
    # Ensure constraint sum(h) = n/2
    h = normalize(h)
    return h

def crossover(h1, h2):
    # Simple uniform crossover
    mask = np.random.rand(len(h1)) > 0.5
    child = np.where(mask, h1, h2)
    # Ensure constraint sum(h) = n/2
    child = normalize(child)
    return child

def tournament_selection(pop, fitnesses, t_size=5):
    # Tournament selection: pick t_size individuals and return the best
    indices = np.random.choice(len(pop), size=t_size, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return pop[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Setup
    n_points = 70
    start_time = time.time()
    max_generations = 1000
    
    # Initialize population
    pop_size = 50
    population = []
    if 'initial_h_values' in globals():
        # Use the predefined initial solution
        h0 = np.array(initial_h_values)
        h0 = normalize(h0)
        population.append(h0)
    else:
        # Create a simple initial construction if none provided
        h0 = np.ones(n_points) * 0.5
        h0 = normalize(h0)
        population.append(h0)
    
    # Generate additional random feasible solutions to fill population
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = normalize(h)
        if feasible(h):
            population.append(h)
    
    best_fitness = float('inf')
    best_h = None
    
    # Evolution is based on maximizing fitness, so we minimize overlap integral
    generation = 0
    while time.time() - start_time < budget_s and generation < max_generations:
        # Evaluate fitness for entire population
        fitnesses = [evaluate(h) for h in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Create new population through selection, crossover, mutation
        new_population = [population[current_best_idx]]  # Elitism
        
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.rand() < 0.8:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < 0.1:
                child = mutate(child, rate=0.1)
            
            # Ensure feasibility of child
            if not feasible(child):
                child = normalize(child)
                
            new_population.append(child)
        
        population = new_population
        generation += 1
    
    # Final check on best solution
    if best_h is not None and not feasible(best_h):
        best_h = normalize(best_h)
    final_c5 = evaluate(best_h) if best_h is not None else float('inf')
    
    return (best_h.tolist(), final_c5, len(best_h))
