import numpy as np
import time
from scipy import optimize

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for the given h function."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation for all shifts 1 to n-1
    corr = np.correlate(h, 1 - h, mode="full")
    # Only consider valid overlaps
    max_overlap = 0.0
    for k in range(1, n):
        overlap = corr[n-1+k] * dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def compute_fitness(h):
    return evaluate_erdos_solution(h)

def project(h):
    """Project h to satisfy constraints"""
    h = np.clip(h, 0, 1)
    return h * (len(h) / 2) / np.sum(h)

def mutate(h, sigma):
    """Apply Gaussian mutation and reproject"""
    h_new = h + np.random.normal(0, sigma, len(h))
    return project(h_new)

def crossover(h1, h2, alpha=0.6):
    """Blend crossover"""
    beta = np.random.uniform(alpha, 1 - alpha)
    h_new = beta * h1 + (1 - beta) * h2
    return project(h_new)

def compute_gradient(h):
    """Analytical gradient for local refinement [approximate]"""
    n = len(h)
    grad = np.zeros(n)
    dx = 2.0 / n
    # Approximate gradient by finite differences
    eps = 1e-4
    base_fitness = compute_fitness(h)
    for i in range(n):
        h_eps = h.copy()
        h_eps[i] += eps
        h_eps = project(h_eps)
        eps_fitness = compute_fitness(h_eps)
        grad[i] = (eps_fitness - base_fitness) / eps
    return grad

    # Use population-based approach
    n_points = len(h_start)
    pop_size = 100
    max_generations = 5000
    start_time = time.time()
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(pop_size - 1):
        population.append(mutate(h_start, 0.1))
    
    # Compute fitness
    fitnesses = [compute_fitness(h) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # Optimization loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Selection (tournament)
        tournament_size = 5
        selected_indices = []
        for _ in range(pop_size):
            idxs = np.random.choice(pop_size, tournament_size, replace=False)
            winner = idxs[np.argmin([fitnesses[i] for i in idxs])]
            selected_indices.append(winner)
        
        # Create offspring
        offspring = []
        for _ in range(pop_size // 3):
            i1, i2 = np.random.choice(selected_indices, 2, replace=False)
            child = crossover(population[i1], population[i2])
            offspring.append(child)
            
        # Add mutations
        for _ in range(pop_size - len(offspring)):
            parent_idx = np.random.choice(selected_indices)
            mut_h = mutate(population[parent_idx], 0.1 * np.exp(-gen / 100))
            offspring.append(mut_h)
        
        # Local refinement for top candidates
        top_indices = np.argsort(fitnesses)[:pop_size // 10]
        for idx in top_indices:
            h = population[idx]
            grad = compute_gradient(h)
            lr = 0.02 * np.exp(-gen / 100)
            h_new = h - lr * grad
            h_new = project(h_new)
            offspring.append(h_new)
        
        # Evaluate offspring
        offspring_fitnesses = [compute_fitness(h) for h in offspring]
        
        # Replace worst with best offspring
        worst_indices = np.argsort(fitnesses)[-pop_size // 2:]
        for i, (worst_idx, off_idx) in enumerate(zip(worst_indices, 
                                                       np.argsort(offspring_fitnesses)[:pop_size // 2])):
            population[worst_idx] = offspring[off_idx].copy()
            fitnesses[worst_idx] = offspring_fitnesses[off_idx]
            
        # Update best if improved
        new_best_idx = np.argmin(fitnesses)
        if fitnesses[new_best_idx] < best_fitness:
            best_fitness = fitnesses[new_best_idx]
            best_h = population[new_best_idx].copy()
    
    # Final evaluation
    final_c5 = compute_fitness(best_h)
    return (best_h, final_c5, n_points)
