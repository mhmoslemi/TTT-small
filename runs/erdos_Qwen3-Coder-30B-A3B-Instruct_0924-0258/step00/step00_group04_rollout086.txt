import numpy as np
import time
from scipy.optimize import minimize

def evaluate_overlap(h):
    """Evaluate the maximum overlap for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation: h(x) * (1 - h(x+k))
    corr = np.correlate(h, 1 - h, mode='full')
    # Extract the relevant part (shifts from -n+1 to n-1)
    max_corr = np.max(corr[n-1:])  # This corresponds to C5
    return max_corr * dx

def project_and_rescale(h, n_points):
    """Project h to [0,1] and rescale to meet sum constraint."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def gradient_step(h, step_size=0.1):
    """Perform a gradient-guided step."""
    n = len(h)
    dx = 2.0 / n
    
    # Evaluate at current h
    current_c5 = evaluate_overlap(h)
    
    # Compute finite difference gradient for each component
    grad = np.zeros(n)
    epsilon = 1e-4
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = project_and_rescale(h_plus, n)
        
        h_minus = h.copy()
        h_minus[i] -= epsilon
        h_minus = project_and_rescale(h_minus, n)
        
        c5_plus = evaluate_overlap(h_plus)
        c5_minus = evaluate_overlap(h_minus)
        grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
    
    # Apply gradient step
    h_new = h - step_size * grad
    h_new = project_and_rescale(h_new, n)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 42  # From initial_h_values
    max_iter = 600
    initial_temp = 1.0
    cooling_rate = 0.99
    min_temp = 0.01
    patience = 50
    noise_std = 0.05
    gradient_scale = 0.1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_and_rescale(h, n_points)
    else:
        h = np.ones(n_points) * 0.5
        h = project_and_rescale(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    
    temp = initial_temp
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Mutation step with gradient guidance
        h_candidate = h.copy()
        
        # Add gradient-guided noise (simplified gradient direction)
        h_candidate += np.random.normal(0, noise_std, n_points)
        h_candidate = project_and_rescale(h_candidate, n_points)
        
        # Optionally perform a gradient step
        h_candidate = gradient_step(h_candidate, gradient_scale)
        h_candidate = project_and_rescale(h_candidate, n_points)
        
        # Evaluate candidate
        c5_candidate = evaluate_overlap(h_candidate)
        
        # Accept or reject
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            last_improvement = iteration
        elif c5_candidate < evaluate_overlap(h):
            # Accept with probability
            prob = np.exp((evaluate_overlap(h) - c5_candidate) / temp)
            if np.random.rand() < prob:
                h = h_candidate.copy()
        
        # Annealing
        temp = max(temp * cooling_rate, min_temp)
        
        # Warm restart if no improvement
        if iteration - last_improvement > patience and temp > min_temp * 2:
            temp = initial_temp
            last_improvement = iteration
    
    # Final evaluation
    final_c5 = evaluate_overlap(best_h)
    
    # Ensure final solution is valid
    best_h = project_and_rescale(best_h, n_points)
    final_c5 = evaluate_overlap(best_h)
