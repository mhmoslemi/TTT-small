import numpy as np
import time
import random
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute the overlap integral C5
    corr = correlate(h, 1 - h, mode='full')
    # The full correlation has length 2*n - 1
    # We want the maximum over all lags k = -n+1, ..., n-1
    # But the correlation is centered at index n-1
    # So we look at indices 0 to n-2 and n to 2*n-2
    # They correspond to k = -n+1 to -1 and 1 to n-1
    # k=0 is at index n-1, which we don't need for maximum
    max_corr = np.max(corr[:n-1])  # first half (negative lags)
    max_corr = max(max_corr, np.max(corr[n+1:]))  # second half (positive lags)
    return max_corr * dx

def rescale(h, target_sum):
    # Rescale h such that sum(h) = target_sum
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.full_like(h, target_sum / len(h))
    return h * target_sum / current_sum

def evolve_candidate(h, mutation_rate=0.1, delta=0.1):
    # Mutate a candidate h
    h_new = h.copy()
    for i in range(len(h_new)):
        if random.random() < mutation_rate:
            h_new[i] += random.uniform(-delta, delta)
            h_new[i] = max(0, min(1, h_new[i]))
    return rescale(h_new, np.sum(h))

def crossover(parent1, parent2):
    # One-point crossover
    point = random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return rescale(child1, np.sum(parent1)), rescale(child2, np.sum(parent2))

def tournament_selection(population, fitnesses, k=3):
    # Select one individual using tournament selection
    selected_indices = random.sample(range(len(population)), k)
    selected_fitnesses = [fitnesses[i] for i in selected_indices]
    best_index = selected_indices[np.argmin(selected_fitnesses)]
    return population[best_index]

    population_size = 50
    population = [initial_h]
    for _ in range(population_size - 1):
        h = evolve_candidate(initial_h, 0.2, 0.2)
        h = np.clip(h, 0, 1)
        h = rescale(h, n_points / 2)
        population.append(h)
    
    # Best solution so far
    best_h = initial_h.copy()
    best_C5 = evaluate_h(best_h, dx)
    
    # Time tracking
    start_time = time.time()
    
    # GA loop
    for generation in range(200):
        if time.time() - start_time > budget_s:

        # Evaluate population
        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Track best solution
        current_best_idx = np.argmin(fitnesses)
        current_best_C5 = fitnesses[current_best_idx]
        if current_best_C5 < best_C5:
            best_C5 = current_best_C5
            best_h = population[current_best_idx].copy()
        
        # Select parents using tournament selection
        new_pop = [best_h]  # Elitism: keep best

        while len(new_pop) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            if random.random() < 0.7:  # crossover rate
                child1, child2 = crossover(parent1, parent2)
                new_pop.append(child1)

                    new_pop.append(mut2)
        
        # Truncate to population size
        population = new_pop[:population_size]
    
    # Final evaluation of best solution
    final_C5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_C5, n_points)
