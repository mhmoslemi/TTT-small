import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum value corresponds to the maximum overlap
    C5 = np.max(corr) * dx
    return C5

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    if 'initial_h_values' in globals():

        n_points = len(h_start)

        n_points = 45
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.6
        h_start[1] = 0.7
        h_start[-1] = 0.4
        h_start[-2] = 0.3

    # Initialize swarm
    n_particles = 30
    particles = []
    velocities = []
    
    # Initialize first particle as the starting point
    particles.append(h_start.copy())
    velocities.append(np.zeros(n_points))
    
    # Generate other particles with small random perturbations
    for i in range(1, n_particles):
        pert = np.random.uniform(-0.1, 0.1, n_points)
        h_pert = h_start + pert
        h_pert = project_to_feasible(h_pert, n_points)
        particles.append(h_pert)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    
    # Compute initial fitness
    fitness = [evaluate_erdos_solution(p) for p in particles]
    pBest = particles[:]
    pBest_fitness = fitness[:]
    gBest_idx = np.argmin(fitness)
    gBest = particles[gBest_idx]
    gBest_fitness = fitness[gBest_idx]
    
    # PSO parameters
    w_start = 0.9
    c1, c2 = 2.0, 2.0
    max_vel = 0.5
    
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1.0:
        # Update inertia weight
        w = w_start * (0.99 ** iteration)
        
        # For each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            v_new = (w * velocities[i] + 
                     c1 * r1 * (pBest[i] - particles[i]) + 
                     c2 * r2 * (gBest - particles[i]))
            # Clip velocity
            v_new = np.clip(v_new, -max_vel, max_vel)
            velocities[i] = v_new
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply feasibility projection
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate fitness
            new_fit = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if new_fit < pBest_fitness[i]:
                pBest[i] = particles[i][:]
                pBest_fitness[i] = new_fit
            
            # Update global best
            if new_fit < gBest_fitness:
                gBest = particles[i][:]
                gBest_fitness = new_fit
                
        iteration += 1
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(gBest)
    
    return (gBest, final_c5, n_points)
