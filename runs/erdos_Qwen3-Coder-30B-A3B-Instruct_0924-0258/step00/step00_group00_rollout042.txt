import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h *= (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    # Gradient-based optimization
    max_iter_without_improvement = 0
    max_no_improve = 20
    step_sizes = [0.1, 0.25, 0.5]
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 0.1 and max_iter_without_improvement < max_no_improve:
        iteration += 1
        if iteration % 100 == 0:
            print(f"Iteration {iteration}, C5: {best_C5:.6f}")
            
        # Compute all correlations
        full_corr = np.correlate(h, 1 - h, mode='full')
        # Extract valid correlations for k from 0 to 2*n_points -2
        correlations = []
        for k in range(2 * n_points - 1):
            corr = full_corr[k] * (2.0 / n_points)
            correlations.append(corr)
        
        # Find maximum correlation
        max_corr = max(correlations)
        
        # Identify indices that contribute to max correlation
        max_indices = [k for k, c in enumerate(correlations) if abs(c - max_corr) < 1e-10]
        
        # Compute gradient for maximum contributing lags
        gradient = np.zeros(n_points)
        for k in max_indices:
            # Shift k to proper index
            for i in range(n_points - k):
                if i + k < n_points:
                    gradient[i] += (1 - h[i + k])
                    gradient[i + k] -= h[i]
        
        # Try different step sizes
        best_step = None
        best_new_h = None
        best_new_C5 = best_C5
        
        for alpha in step_sizes:
            # Try step
            h_candidate = h - alpha * gradient
            # Clip to [0,1]
            h_candidate = np.clip(h_candidate, 0, 1)
            # Rescale
            h_candidate *= (n_points / 2) / np.sum(h_candidate)
            
            # Evaluate
            new_C5 = evaluate_erdos_solution(h_candidate)
            
            if new_C5 < best_new_C5:
                best_new_C5 = new_C5
                best_new_h = h_candidate.copy()
                best_step = alpha
                
        if best_new_C5 < best_C5:
            h = best_new_h
            best_C5 = best_new_C5
            max_iter_without_improvement = 0
        else:
            max_iter_without_improvement += 1
            
        # Random perturbation if stuck
        if max_iter_without_improvement >= 10:
            h += 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h *= (n_points / 2) / np.sum(h)
            
    # Final evaluation
    final_C5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_C5), len(best_h))

def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    # Compute correlations the same way as specified
    full_corr = np.correlate(h, 1 - h, mode='full')
    n_points = len(h)
    dx = 2.0 / n_points
    correlation_values = []
    for k in range(2 * n_points - 1):
        correlation_values.append(full_corr[k] * dx)
    return max(correlation_values)
