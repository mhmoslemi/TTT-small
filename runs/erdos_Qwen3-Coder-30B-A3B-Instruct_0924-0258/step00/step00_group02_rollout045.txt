import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h_vals, n_points):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h)
    conv = np.correlate(h_vals, 1 - h_vals, mode='full')
    # Normalize by dx
    conv *= dx
    # Return maximum value
    return np.max(conv)

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] with sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.ones_like(h) * (n_points / 2) / n_points
    return h

def surrogate_objective(h, n_points):
    """Convex surrogate to approximate max overlap"""
    dx = 2.0 / n_points
    conv = np.correlate(h, 1 - h, mode='full')
    # Use quadratic surrogate to penalize large overlaps
    return np.sum(conv**2) * dx**2

def surrogate_gradient(h, n_points):
    """Gradient of surrogate objective"""
    dx = 2.0 / n_points
    # Compute convolution h * (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Gradient w.r.t. h: use chain rule
    # d/dh (sum_i (conv[i])^2) = 2 * sum_i conv[i] * dconv[i]/dh
    # dconv[i]/dh = (d/dh) sum_j h[j] * (1-h[i+j]) = (1-h[i+j]) - h[i+j] = 1 - 2*h[i+j] when i+j <= n_points-1  
    # For simplicity: compute numerical gradient
    grad = np.zeros_like(h)
    for i in range(len(h)):
        for j in range(len(h)):
            k = i + j - (len(h) - 1)
            if 0 <= k < len(conv) and k < len(h):
                grad[i] += 2 * conv[k] * (1 - 2 * h[j]) * dx
    return grad

def optimize_at_resolution(h_start, n_points, max_time):
    """Optimize at one resolution level"""
    if time() > max_time:
        return h_start, evaluate_erdos_solution(h_start, n_points)
    
    # Ensure h_start is feasible
    h_start = project_to_feasible(h_start, n_points)
    
    # Prepare constraints
    bounds = [(0, 1) for _ in range(n_points)]
    # Sum constraint
    def sum_constraint(x):
        return np.sum(x) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': sum_constraint}
    
    # Optimize using SLSQP
    result = minimize(
        surrogate_objective, 
        h_start, 
        args=(n_points,),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-8}
    )
    
    # Project result to feasible region
    h_opt = project_to_feasible(result.x, n_points)
    c5_bound = evaluate_erdos_solution(h_opt, n_points)
    return h_opt, c5_bound

    max_time = time() + budget_s
    
    # Start with the initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_best)
    else:
        # fallback to basic construction
        n_points = 45
        h_best = np.ones(n_points) * 0.5
    
    c5_best = evaluate_erdos_solution(h_best, n_points)
    best_h = h_best.copy()
    
    # Optimization levels (resolution)
    resolution_levels = [15, 30, 45, 60, 75]
    
    for n_points in resolution_levels:
        if time() > max_time - 2:
            break
            
        h_opt, c5_opt = optimize_at_resolution(best_h, n_points, max_time)
        
        if c5_opt < c5_best:
            c5_best = c5_opt
            best_h = h_opt.copy()
            
        # Perturb and restart for better exploration
        for p in range(3):
            if time() > max_time - 2:
                break
            h_pert = h_opt + np.random.normal(0, 0.05, n_points)
            h_pert = project_to_feasible(h_pert, n_points)
            h_opt_pert, c5_opt_pert = optimize_at_resolution(h_pert, n_points, max_time)
            if c5_opt_pert < c5_best:
                c5_best = c5_opt_pert
                best_h = h_opt_pert.copy()
                
        # Interpolate to next resolution level if possible
        if len(best_h) < n_points:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_to_feasible(h_interp, n_points)
            best_h = h_interp
            
    return (best_h, c5_best, len(best_h))
