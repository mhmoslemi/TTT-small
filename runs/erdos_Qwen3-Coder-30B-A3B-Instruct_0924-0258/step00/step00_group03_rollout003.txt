import numpy as np
from scipy import signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    lr = 0.01
    lr_decay = 0.95

    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a basic initialization
        h = np.ones(n_points) * 0.5

    best_h = h.copy()
    best_c5 = np.inf

    # Precompute full convolution for performance: we will compute over full mode
    # and only use the relevant values for maximum
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        conv = signal.correlate(h_vec, 1 - h_vec, mode="full")
        # The valid range of k corresponds to the first n_points entries
        # from full convolution
        k_max = len(conv) - 1
        # So we take indices in [n_points-1, 2*n_points-2] (inclusive)
        valid_conv = conv[n_points-1:2*n_points-1]
        return np.max(valid_conv) * dx

    def project_h(h_vec):
        # Use CVXPY to solve the projection: minimize ||x - h||_2^2 s.t.
        # 0 <= h[i] <= 1, sum(h) = n_points / 2
        h_proj = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(h_proj - h_vec))
        constraints = [
            h_proj >= 0,
            h_proj <= 1,
            cp.sum(h_proj) == n_points / 2
        ]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return h_proj.value
        except:
            # Fallback: clip and renormalize
            h_proj = np.clip(h_vec, 0, 1)
            h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
            return h_proj

    iteration = 0
    while iteration < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Compute convolution for gradient estimation
        h_vec = np.array(h)
        conv = signal.correlate(h_vec, 1 - h_vec, mode="full")
        # Compute maximum lag index
        k_max = np.argmax(conv[n_points-1:2*n_points-1]) + (n_points-1)
        # We need to compute the gradient of this maximum 
        # The contribution to the gradient of term h[i] in the convolution 
        # at lag k_max comes from the fact that 
        # conv[i + k_max] = sum_{j} h[j] * (1 - h[j + i + k_max]) is being maximized
        # For the term at position k_max:
        # This is h[0] * (1 - h[k_max]) + h[1] * (1-h[k_max+1]) + ...
        # So d/dh[i] = (1 - h[k_max + i]) + (if i+k_max < n_points, ...)

        # Let's compute gradient accurately
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Derivative of the convolution term w.r.t. h[i]
            if i + k_max < n_points:
                grad[i] += 1 - h[i + k_max]
            if i - k_max >= 0:
                grad[i] += 1 - h[i - k_max]

        # Step
        h = h - lr * grad
        h = project_h(h)

        # Update learning rate
        if iteration % 100 == 0:
            lr *= lr_decay
            
        iteration += 1

    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
