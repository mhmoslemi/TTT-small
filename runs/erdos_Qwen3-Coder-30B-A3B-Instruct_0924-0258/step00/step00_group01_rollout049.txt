import numpy as np
import time
import random
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral C5 for given step function h."""
    # Compute correlation for all lags
    corr = signal.correlate(h, 1 - h, mode='full')
    # Take the maximum value and scale by dx
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to ensure sum(h) = n_points / 2 and h[i] in [0, 1]."""
    h = np.array(h)
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust sum to exactly n_points / 2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-10:
        # Simple adjustment: scale and clip
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        # If still not exact, adjust one element
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-10:
            h[-1] += diff
            h = np.clip(h, 0, 1)
    return h

def crossover(parent1, parent2):
    """Perform simulated crossover (averaging)."""
    child1 = (parent1 + parent2) / 2
    child2 = (parent1 + parent2) / 2
    # Project to feasible region
    child1 = project_to_feasible(child1, len(parent1))
    child2 = project_to_feasible(child2, len(parent2))
    return child1, child2

def mutate(h, mutation_rate=0.05, sigma=0.1):
    """Mutate h with Gaussian noise."""
    h = np.array(h)

    # For each element, possibly apply Gaussian noise
    for i in range(n_points):
        if random.random() < mutation_rate:
            h[i] += np.random.normal(0, sigma)
    return project_to_feasible(h, n_points)
