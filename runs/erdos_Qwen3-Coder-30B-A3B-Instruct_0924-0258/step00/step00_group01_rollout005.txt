import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral C_5"""
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integral corresponds to the maximum value of the cross-correlation
    # scaled by dx (since we're integrating over the domain [0,2])
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to ensure sum(h) = n_points / 2 and 0 <= h[i] <= 1"""
    # Ensure h is within bounds
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return np.clip(h, 0, 1)

def compute_gradient(h, dx, n_points):
    """Compute the gradient of the maximum overlap with respect to h"""
    # Compute cross-correlation of h with 1-h
    corr = np.correlate(h, 1 - h, mode="full")
    # Find the lag where maximum occurs
    max_lag_idx = np.argmax(corr)
    # The gradient corresponds to the derivative of the cross-correlation
    # with respect to h at the maximum lag
    grad = np.zeros_like(h)
    lag = max_lag_idx - (n_points - 1)
    # How changing h[i] affects the cross-correlation at maximum lag
    for i in range(n_points):
        if 0 <= i - lag < n_points:
            # For simplicity, approximating gradient as 1 for involved indices
            # Marginal - For a small exact gradient, we compute changes directly
            # But a better method would examine finite differences at the critical value
            grad[i] = 1.0 - 2.0 * h[i - lag] if 0 <= i - lag < n_points else 0.0
        else:
            grad[i] = 0.0
    return -grad

def line_search_step(h, grad, step_size, n_points, dx):
    """Try a line search step"""

        if new_c5 < current_c5 - tolerance:
