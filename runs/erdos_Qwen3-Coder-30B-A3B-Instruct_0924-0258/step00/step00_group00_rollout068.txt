import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up parameters
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    max_iter = 10000
    patience = 20
    perturb_mag = 0.01
    alphas = [0.1, 0.25, 0.5]
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h += 0.1 * np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    start_time = time.time()
    iter_count = 0
    no_improvement_iter = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Compute all cross-correlations
        full_corr = correlate(h, 1-h, mode='full')
        cross_correlations = full_corr * dx
        
        # Find maximum value
        C5 = np.max(cross_correlations)
        max_indices = np.where(np.abs(cross_correlations - C5) < 1e-10)[0]
        
        # If no improvement for a while, perturb
        if no_improvement_iter > patience:
            h += perturb_mag * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            no_improvement_iter = 0
        
        # Compute gradient for max cross-correlations
        gradient = np.zeros(n_points)
        for k in max_indices:
            k_offset = k - (n_points - 1)
            for i in range(n_points - abs(k_offset)):
                if k_offset >= 0:
                    gradient[i] += (1 - h[i + k_offset])
                    if i + k_offset < n_points:
                        gradient[i + k_offset] -= h[i]
                else:  # k_offset < 0
                    gradient[i - k_offset] += (1 - h[i])
                    if i < n_points + k_offset:
                        gradient[i] -= h[i - k_offset]
        
        # Line search
        best_h_candidate = h.copy()
        best_C5_candidate = C5
        
        for alpha in alphas:
            h_candidate = h - alpha * gradient
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (target_sum / np.sum(h_candidate))
            
            C5_candidate = evaluate_C5(h_candidate, dx)
            
            if C5_candidate < best_C5_candidate:
                best_C5_candidate = C5_candidate
                best_h_candidate = h_candidate.copy()
        
        # Update if improved
        if best_C5_candidate < best_C5:
            h = best_h_candidate
            best_C5 = best_C5_candidate
            best_h = h.copy()
            no_improvement_iter = 0
        else:
            no_improvement_iter += 1
            
    # Final validation
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (target_sum / np.sum(h_final))
    final_C5 = evaluate_C5(h_final, dx)
    
    return (h_final.tolist(), float(final_C5), n_points)

def evaluate_C5(h, dx):
    full_corr = correlate(h, 1-np.array(h), mode='full')
    return np.max(full_corr) * dx
