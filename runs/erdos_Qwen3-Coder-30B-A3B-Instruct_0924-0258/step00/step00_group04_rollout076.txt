import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract the valid lags (from -n+1 to n-1)
    c5_vals = corr[n-1:] * dx
    return np.max(c5_vals)

def project_to_feasible(h):
    """Project h to feasible region: clip to [0,1] and rescale to meet sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = len(h) / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def gradient_estimate(h, eps=1e-4):
    """Estimate gradient of C5 w.r.t. h using finite differences."""
    n = len(h)
    grad = np.zeros_like(h)
    base_c5 = evaluate_c5(h)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus)
        c5_plus = evaluate_c5(h_plus)
        grad[i] = (c5_plus - base_c5) / eps
    return grad
