import numpy as np
import time
from scipy import correlate

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral for given h values."""
    # Compute full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # The max correlation is calculated for lags from -n+1 to n-1
    # We only want the positive lags (0 to n-1) for valid overlaps
    # The center corresponds to lag 0, so we take elements from n-1 to 2*n-2
    max_corr = np.max(corr[len(h)-1:])
    return max_corr * dx

def rescale_to_unit_sum(h, target_sum):
    """Ensure h sums to target_sum by rescaling."""
    current_sum = np.sum(h)
    if current_sum == 0:
        h.fill(target_sum / len(h))
        return h
    h = h * (target_sum / current_sum)
    return h

def generate_initial_population(parent_h, n_points, pop_size=50):
    """Generate initial population."""
    population = [parent_h.copy()]
    for _ in range(pop_size - 1):
        # Randomly perturb parent
        h = parent_h.copy()
        for i in range(n_points):
            if np.random.random() < 0.1:
                h[i] += np.random.uniform(-0.2, 0.2)
        h = np.clip(h, 0, 1)
        h = rescale_to_unit_sum(h, n_points / 2)
        population.append(h)
    return population

def mutate(h, mutation_rate=0.1, delta=0.1):
    """Apply mutation to h vector."""
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h[i] += np.random.uniform(-delta, delta)
    h = np.clip(h, 0, 1)
    return rescale_to_unit_sum(h, len(h) / 2)

def crossover(h1, h2):
    """One-point crossover between h1 and h2."""
    n = len(h1)
    split = np.random.randint(1, n - 1)
    child1 = np.concatenate((h1[:split], h2[split:]))
    child2 = np.concatenate((h2[:split], h1[split:]))
    return rescale_to_unit_sum(child1, n / 2), rescale_to_unit_sum(child2, n / 2)

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select one parent using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    if 'initial_h_values' in globals():

    else:

        n_points = 45
        parent_h = np.ones(n_points) * 0.5
        parent_h[10:35] = 1.0
        parent_h = rescale_to_unit_sum(parent_h, n_points / 2)
        
    # Setup for optimization
    n_points = len(parent_h)
    dx = 2.0 / n_points
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    max_generations = 200
    
    # Generate initial population
    population = generate_initial_population(parent_h, n_points, pop_size)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for gen in range(max_generations):
        # Evaluate all individuals
        fitnesses = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitnesses.append(c5)
            
        # Track best individual
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Early termination if time is nearly up
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        new_population = [population[i] for i in sorted_indices[:pop_size//2]]
        
        # Fill the rest with offspring
        while len(new_population) < pop_size:
            # Select parents
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.random() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
                new_population.append(child1)
                if len(new_population) < pop_size:
                    new_population.append(child2)
            else:
                # Select one parent and mutate
                if np.random.random() < 0.5:
                    new_h = mutate(parent1.copy(), mutation_rate)
                else:
                    new_h = mutate(parent2.copy(), mutation_rate)
                new_population.append(new_h)
        
        population = new_population[:pop_size] 
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
