import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    # Normalize to correct integral
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization parameters
    max_iter = 10000
    perturb_steps = 20
    alpha_candidates = [0.1, 0.5, 1.0]
    
    start_time = time.time()
    iter_count = 0
    stagnant = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        # Compute all cross-correlations
        corr = correlate(h, 1 - h, mode='full')
        # The correlation starts at lag = -n+1 to n-1
        # Find the maximum lag where correlation is maximum
        k_indices = np.arange(-n_points + 1, n_points)
        max_corr_idx = np.argmax(corr)
        max_k = k_indices[max_corr_idx]
        
        # Compute gradient at that max correlation lag
        grad = np.zeros(n_points)
        if max_k >= 0:
            # For positive lag k, the gradient is 1 - h[i+k] for i = 0..n-k-1
            for i in range(n_points - max_k):
                grad[i] += -(1 - h[i + max_k])
        else:
            # For negative lag k, the gradient is 1 - h[i] for i = -k .. n-1
            for i in range(-max_k, n_points):
                grad[i] += -(1 - h[i + max_k])
                
        # Try different step sizes to find a good alpha
        best_alpha = 0.0
        best_h_new = h.copy()
        best_c5_new = best_c5
        
        for alpha in alpha_candidates:
            h_new = h - alpha * grad
            # Clip to feasible range
            h_new = np.clip(h_new, 0, 1)
            # Re-normalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5_new:
                best_c5_new = c5_new
                best_h_new = h_new.copy()
                best_alpha = alpha
                
        # If nothing improved, try perturbation
        if best_c5_new >= best_c5:
            stagnant += 1
            if stagnant > perturb_steps:
                # Add small random perturbation
                h = h + 0.01 * np.random.randn(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                stagnant = 0
        else:
            # Accept improvement
            h = best_h_new
            best_c5 = best_c5_new
            best_h = h.copy()
            stagnant = 0
        if best_c5_new < best_c5:
            best_c5 = best_c5_new
            best_h = h.copy()
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def compute_c5(h, dx):
    """Compute C5 = max_k int h(x)(1 - h(x+k)) dx"""
    corr = correlate(h, 1 - h, mode='full')
    # Since correlate runs from lag -n+1 to n-1,
    # we need to take the maximum over the full range
    max_corr = np.max(corr)
    return max_corr * dx
