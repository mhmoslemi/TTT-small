import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    
    # Start with initial construction if available
    try:
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    except:
        h = np.full(n_points, 0.5)
    
    # Binary search bounds
    low = 0.0
    high = 0.38085
    
    best_h = None
    best_c5 = high
    
    while high - low > 1e-4 and (time.time() - start_time) < budget_s - 1:
        mid = (low + high) / 2.0
        
        # Create variables
        h_vars = cp.Variable(n_points)
        constraints = []
        
        # Add constraints
        for k in range(2 * n_points - 1):
            # Correlation term for shift k
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_vars[i] * (1 - h_vars[j])
            constraints.append(expr <= mid / dx)
        
        # Add sum constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Add bounds
        for i in range(n_points):
            constraints.append(h_vars[i] >= 0)
            constraints.append(h_vars[i] <= 1)
        
        # Objective (minimize mid directly)
        # We just need to check feasibility
        
        # Create and solve problem using GLOP
        try:
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.GLPK, verbose=False)
            
            if prob.status == cp.OPTIMAL or prob.status == cp.UNBOUNDED:
                new_h = h_vars.value.copy()
                # Check constraints
                if (np.all(new_h >= 0) and np.all(new_h <= 1) and
                    abs(np.sum(new_h) - n_points/2) < 1e-3):
                    # Test again with full evaluation
                    corr_vals = np.correlate(new_h, 1 - new_h, mode='full')
                    max_corr = np.max(corr_vals) * dx
                    if max_corr <= mid:
                        # Valid solution
                        high = mid
                        best_h = new_h.copy()
                        best_c5 = mid
                    else:
                        low = mid
                else:
                    low = mid
            else:
                low = mid
        except:
            low = mid

    # Final validation and return
    if best_h is not None:
        h = best_h
    else:
        # Fall back to initial vector, or create a uniform one
        h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(n_points, 0.5)
        h = h / np.sum(h) * (n_points / 2)
    
    # Final evaluation
    corr_vals = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(corr_vals) * dx
    
    return (h.tolist(), final_c5, n_points)
