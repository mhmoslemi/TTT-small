import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = 47  # Starting with the given construction
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if abs(np.sum(h) - target_sum) > 1e-6:
            # Rescale to match sum constraint
            h = h * target_sum / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * target_sum / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    if abs(np.sum(h) - target_sum) > 1e-6:
        h = h * target_sum / np.sum(h)
    
    # Optimization parameters
    alpha = 0.1
    beta = 0.95
    momentum = 0.9
    max_iters = 10000
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    prev_c5 = float('inf')
    no_improvement = 0
    start_time = time.time()
    
    # Gradient descent loop
    v = np.zeros_like(h)
    for iter in range(max_iters):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlaps for all shifts
        correlation = correlate(h, 1 - h, mode='full')
        # Extract upper triangular part (shifts k = 0 to n_points-1)
        c_values = correlation[n_points-1:]
        k0 = np.argmax(c_values)
        c5 = c_values[k0]
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient w.r.t. k0
        grad = np.zeros_like(h)
        if k0 < n_points:
            # gradient[i] += (1 - h[i + k0]) if i + k0 < n_points
            # gradient[i] -= h[i - k0] if i >= k0
            for i in range(n_points):
                if i + k0 < n_points:
                    grad[i] += 1 - h[i + k0]
                if i >= k0:
                    grad[i] -= h[i - k0]
        
        # Update with momentum
        v = momentum * v - alpha * grad
        h_new = h + v
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        
        h = h_new
        
        # Adaptive stepsize
        if abs(c5 - prev_c5) < 1e-8:
            no_improvement += 1
            if no_improvement > patience:
                alpha *= beta
                no_improvement = 0
        else:
            no_improvement = 0
        prev_c5 = c5
    
    # Final evaluation of best solution
    correlation = correlate(best_h, 1 - best_h, mode='full')
    c_values = correlation[n_points-1:]
    c5_final = np.max(c_values)
    c5_final *= dx  # Convert to actual integral scale

    return (best_h.tolist(), c5_final, n_points)
