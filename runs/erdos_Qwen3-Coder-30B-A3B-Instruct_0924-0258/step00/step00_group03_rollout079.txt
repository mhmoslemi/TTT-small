import numpy as np
from scipy import signal
import cvxpy as cp
import time

def evaluate(h_values):
    """Evaluate C5 for given h_values."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute the convolution of h with (1-h) for all lags
    conv = signal.correlate(h_values, 1 - h_values, mode='full')
    # Take maximum over all lags
    max_val = np.max(conv)
    return max_val * dx

def project_h(h, n_points):
    """Project h to feasible set with sum = n_points / 2 and bounds [0,1]."""
    # Use CVXPY for projection onto convex constraints
    x = cp.Variable(n_points)
    # Objective: minimize ||x - h||^2
    obj = cp.Minimize(cp.sum_squares(x - h))
    # Constraints: 0 <= x <= 1, sum(x) = n_points/2
    constraints = [x >= 0, x <= 1, cp.sum(x) == n_points / 2]
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        return np.array(x.value).flatten()
    except:
        # Fallback: clip to bounds and rescale
        h_proj = np.clip(h, 0, 1)
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        return h_proj
