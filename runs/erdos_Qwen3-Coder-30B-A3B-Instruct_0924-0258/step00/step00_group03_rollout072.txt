    dx = 2.0 / n_points

    # Using correlation via FFT for efficiency

    # Let's compute this directly
    conv = np.correlate(h, 1 - h, mode="full")
    
    # The overlap integral is from full to only the middle window 
    s = len(conv)
    mid = s // 2
    overlap = conv[mid:]
    max_over = np.max(overlap)
    
    return max_over * dx
