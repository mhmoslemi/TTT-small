import numpy as np
import time
from scipy import optimize

def evaluate_c5(h):
    """Evaluate C5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations at once
    corr = np.correlate(h, 1 - h, mode="full")
    # Only consider valid overlaps (shifts from -n+1 to n-1)
    # Maximum overlap occurs at shift 0 (no shift)
    max_overlap = np.max(corr[n-1:])
    return max_overlap * dx

def project_feasible(h):
    """Ensure h is feasible: sum = n/2 and 0 <= h[i] <= 1"""
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2
    
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Adjust to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.full_like(h, 1.0 / n)
    
    return h

def random_feasible(n):
    """Generate a random feasible step function"""
    h = np.random.uniform(0, 1, n)
    h = project_feasible(h)
    return h

def crossover(parent1, parent2):
    """Uniform crossover"""
    n = len(parent1)
    mask = np.random.random(n) > 0.5
    offspring1 = np.where(mask, parent1, parent2)
    offspring2 = np.where(mask, parent2, parent1)
    offspring1 = project_feasible(offspring1)
    offspring2 = project_feasible(offspring2)
    return offspring1, offspring2

def mutate(h, mutation_rate=0.05, sigma=0.1):
    """Mutation with Gaussian noise"""
    n = len(h)
    mask = np.random.random(n) < mutation_rate
    h = h.copy()
    h[mask] += np.random.normal(0, sigma, np.sum(mask))
    h = project_feasible(h)
    return h

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select two parents using tournament selection"""
    n = len(population)
    idx1 = np.random.choice(n, tournament_size)
    idx2 = np.random.choice(n, tournament_size)
    
    parent1 = population[np.argmin(fitnesses[idx1])]
    parent2 = population[np.argmin(fitnesses[idx2])]
    
    return parent1, parent2

    best_c5 = evaluate_c5(best_h)
    stagnant = 0
