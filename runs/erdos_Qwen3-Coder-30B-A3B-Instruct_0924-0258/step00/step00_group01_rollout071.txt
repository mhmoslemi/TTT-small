import numpy as np
import time
from scipy.signal import correlate
import random

def evaluate_h(h, dx):
    # Compute the overlap integral for all lags and return max
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_feasible(h, target_sum):
    # Ensure h sums to target_sum and stays in [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    # Adjust final value if needed
    if np.sum(h) != target_sum:
        diff = target_sum - np.sum(h)
        # Add difference to last element
        h[-1] += diff
        h = np.clip(h, 0, 1)
    return h

def crossover(parent1, parent2):
    # Uniform crossover with partial sum correction
    child1 = (parent1 + parent2) / 2.0
    child2 = (parent1 + parent2) / 2.0
    # Ensure sum constraint
    child1 = project_feasible(child1, np.sum(parent1))
    child2 = project_feasible(child2, np.sum(parent2))
    return child1, child2

def mutate(h, mutation_rate, sigma):
    # Gaussian mutation with clipping
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, sigma, len(h))
    h = h + noise * mask
    h = np.clip(h, 0, 1)
    # Project back to correct sum
    h = project_feasible(h, np.sum(h))
    return h

def tournament_select(population, fitnesses, tournament_size=3):
    # Tournament selection
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]
