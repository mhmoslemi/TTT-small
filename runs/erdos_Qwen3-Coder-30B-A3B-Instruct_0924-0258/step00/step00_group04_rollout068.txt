import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 42
    dx = 2.0 / n_points
    max_iter = 1200
    initial_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Optimizer parameters
    alpha = 0.1
    decay = 0.995
    tolerance = 1e-5
    
    # Gradient-based optimization
    for iteration in range(max_iter):
        elapsed = time.time() - initial_time
        if elapsed >= budget_s * 0.98:
            break
            
        # Compute all overlaps
        overlaps = []
        for k in range(n_points):
            # Reflected version of h
            h_reflected = np.concatenate([h[n_points-k:], h[:n_points-k]])
            z = np.sum(h * (1 - h_reflected)) * dx
            overlaps.append(z)
        
        # Find maximum overlap
        max_idx = np.argmax(overlaps)
        max_z = overlaps[max_idx]
        
        # Gradient computation
        g = np.zeros(n_points)
        for i in range(n_points):
            j1 = (i - max_idx) % n_points
            j2 = (i + max_idx) % n_points
            g[i] = 1 - h[j1] - h[j2]
        g *= 2 * dx / n_points
        
        # Line search for step size
        new_alpha = alpha
        while new_alpha > 1e-8:
            h_new = h - new_alpha * g
            # Project to feasible region [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to match constraint
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            
            new_C5 = evaluate_C5(h_new, dx)
            if new_C5 < best_C5:
                h = h_new
                best_C5 = new_C5
                break
            new_alpha *= 0.5
        
        # Early stopping
        if abs(max_z - best_C5) < tolerance:
            break
    
    # Final validation
    final_C5 = evaluate_C5(best_h, dx)
    
    return (best_h.tolist(), final_C5, n_points)

def evaluate_C5(h, dx):
    # compute all overlaps
    n = len(h)
    max_value = 0.0
    h_padded = np.concatenate([h, h])
    
    for k in range(n):
        overlap = 0.0
        for i in range(n):
            overlap += h[i] * (1 - h_padded[i + k])
        overlap *= dx
        max_value = max(max_value, overlap)
    return max_value
