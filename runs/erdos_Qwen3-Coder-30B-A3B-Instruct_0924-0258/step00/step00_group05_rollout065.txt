import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Binary search on C5
    low = 0.0
    high = 0.38085
    tol = 1e-4
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    while high - low > tol and time.time() - start_time < budget_s - 1:
        mid = (low + high) / 2.0
        C5 = mid
        
        # Formulate LP relaxation for this C5
        # Variables
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = []
        
        # Sum constraint
        constraints.append(cp.sum(h_vars) == target_sum)
        
        # Bound constraints
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        
        # Approximate quadratic constraints
        # h[i] * (1 - h[i + k]) <= (h[i] + (1 - h[i + k])) / 2
        for k in range(n_points):
            # For each shift k, compute the correlation
            # We linearize h[i] * (1 - h[i + k]) <= (h[i] + 1 - h[i + k]) / 2
            lhs = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                lhs += h_vars[i] * (1 - h_vars[j])
            constraints.append(lhs <= C5 / dx)
        
        # Objective: minimize sum of all such constraints (just one)
        # Minimize a dummy variable to allow the problem to be feasible
        # Actually minimize 0 (just feasibility)
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.SOLVED]:
                # Check resulting solution
                h_sol = h_vars.value
                if h_sol is not None:
                    # Project back to valid range
                    h_sol = np.clip(h_sol, 0.0, 1.0)
                    h_sol = h_sol / np.sum(h_sol) * target_sum
                    
                    # Compute actual C5
                    c5_val = evaluate_erdos_solution(h_sol, dx)
                    if c5_val < best_c5:
                        best_h = h_sol.copy()
                        best_c5 = c5_val
                        
                    if c5_val <= C5:
                        high = C5
                    else:
                        low = C5
                else:
                    low = C5
            else:
                low = C5
        except:
            low = C5
    
    # Final run with best candidate
    try:
        final_c5 = evaluate_erdos_solution(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    except:
        pass
    
    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    """Evaluate the C5 value using numpy correlation."""
    # Reflect h to compute (1 - h)
    h_reflect = 1 - h
    # Compute full convolution
    corr = np.correlate(h, h_reflect, mode='full')
    # Maximum over all lags
    max_corr = np.max(corr) * dx
    return max_corr
