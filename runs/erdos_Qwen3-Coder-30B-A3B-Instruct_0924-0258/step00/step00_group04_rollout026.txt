import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_c5(h):
    """Evaluate C5 for the given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for different lags
    corr = np.correlate(h, 1 - h, mode="full")
    # Take only the relevant part (positive lags)
    c5 = np.max(corr[n-1:]) * dx
    return c5

def smooth_and_normalize(h, sigma=0.1):
    """Apply Gaussian smoothing and normalize to maintain constraint"""
    h_smooth = gaussian_filter1d(h, sigma=sigma)
    # Clip to [0,1]
    h_smooth = np.clip(h_smooth, 0, 1)
    # Rescale to meet integral constraint
    current_sum = np.sum(h_smooth)
    target_sum = len(h) / 2.0
    if current_sum > 0:
        h_smooth = h_smooth * target_sum / current_sum
    return h_smooth

def compute_gradient(h, eps=1e-5):
    """Compute approximate gradient using finite differences"""
    n = len(h)
    grad = np.zeros_like(h)
    base_c5 = evaluate_c5(h)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        # Rescale to maintain constraint
        current_sum = np.sum(h_plus)
        target_sum = n / 2.0
        if current_sum > 0:
            h_plus = h_plus * target_sum / current_sum
        c5_plus = evaluate_c5(h_plus)
        grad[i] = (c5_plus - base_c5) / eps
    return grad
