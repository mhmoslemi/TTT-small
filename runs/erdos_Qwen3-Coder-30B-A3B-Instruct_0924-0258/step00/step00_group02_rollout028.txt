import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = signal.correlate(h, 1 - h, mode='full')
    # Extract the relevant part (shifts from -n+1 to n-1)
    c5 = np.max(corr) * dx
    return c5

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1, sum(h) = n_points/2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

    # Initialize from available construction if present
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Construct an initial valid h
        h = np.ones(n_points) * 0.5

    h = project_h(h, n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()
    
    # Gradient descent parameters
    lr = 0.01
    decay = 0.9995
    perturb_mag = 0.05
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Compute all correlations
        corr = signal.correlate(h, 1 - h, mode='full')
        # Find the lag where max overlap occurs
        max_lag_idx = np.argmax(corr)
        # For `full` correlation, the range is -n+1 to n-1
        # So we want to get the corresponding shift
        shift = max_lag_idx - (n_points - 1)
        
        # Compute gradient in terms of the correlation
        # This correlation is h * (1-h) with shift k
        # Gradient w.r.t. h[i] = (1-h[i+k]) for some k, depending on shift
        # For simplicity in finite differences or for a step, we simulate this:
        # Compute gradient for shift `shift`
        # This is a derivative of the integral h(x) * (1 - h(x + shift))
        # So for grid positions, gradient[i] ~ (1 - h[i+shift])
        # We compute this derivative efficiently using correlation
        # gradient = np.zeros_like(h)
        # If correlate(h, 1-h, shift) for any shift, then derivative of overlap at that k
        # For (h * (1-h)) with shift k, the gradient is: 1 - h_{i+k}
        # So for lag k, gradient of overlap at that lag:
        # gradient[i] = (1 - h[i+k]) for valid i+k
        # To get gradient at current h with respect to overlap at max_lag:
        # 1. w.r.t. h[i]: contribution from term h[i] * (1 - h[i + shift])
        # 2. But this overlaps with our indexing
                
        # Simpler: we actually do:
        # We compute the reverse convolution for gradient:
        # A = h * (1 - h - shift) which means we put gradient for each i from the derivative of integral
        # Derivative of h[i] with overlap at shift k is 1 - h[i + k] (we maximize consumption)
        if shift >= 0:
            gradient = np.concatenate([np.zeros(shift), (1 - h[:-shift])])
        else:
            gradient = np.concatenate([(1 - h[-shift:]), np.zeros(-shift)])
        # Make it valid for indices:
        gradient = gradient[:n_points]
        gradient = gradient * dx  # Scale by dx
        
        # Update
        h = h - lr * gradient
        # Project
        h = project_h(h, n_points)
        # Perturbation occasionally to escape local optima
        if iter_count % 50 == 0:
            h += np.random.uniform(-perturb_mag, perturb_mag, n_points)
            h = project_h(h, n_points)
        # Adjust learning rate
        lr *= decay
        
        # Compute new C5 only if not too expensive - keeping it simple
        # But do it regularly to check progress
        # Update best if improved
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
