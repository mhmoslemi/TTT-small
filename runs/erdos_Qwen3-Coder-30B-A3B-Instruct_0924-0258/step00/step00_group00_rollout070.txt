import numpy as np
import time
from scipy import correlate

def evaluate_h(h, dx):
    """Compute C_5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    # Take only the valid correlations, ignore padding
    n = len(h)
    valid_corr = corr[n-1:2*n-1]
    return np.max(valid_corr) * dx

def rescale_h(h, target_sum):
    """Rescale h to have sum = target_sum, keep in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def init_population(n_points, pop_size, parent=None):
    """Initialize population with parent and random variants"""
    pop = []
    if parent is not None:
        pop.append(parent)
    else:
        # Create a basic initial guess
        h = np.ones(n_points) * 0.5
        h = rescale_h(h, n_points / 2.0)
        pop.append(h)
    
    target_sum = n_points / 2.0
    for _ in range(pop_size - 1):
        if np.random.random() < 0.5:
            # Perturb parent
            h = pop[0].copy()
            for i in range(n_points):
                if np.random.random() < 0.2:
                    h[i] += np.random.uniform(-0.2, 0.2)
            h = rescale_h(h, target_sum)
        else:
            # Random construction
            h = np.random.random(n_points)
            h = rescale_h(h, target_sum)
        pop.append(h)
    return pop

def tournament_selection(pop, fitnesses, k=3):
    """Select one individual using tournament selection"""
    indices = np.random.choice(len(pop), k)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return pop[best_idx]

def crossover(parent1, parent2):
    """One-point crossover"""
    n = len(parent1)
    split = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:split], parent2[split:]])
    child2 = np.concatenate([parent2[:split], parent1[split:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, delta=0.1):
    """Mutate h by small random changes"""
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h[i] += np.random.uniform(-delta, delta)
    return rescale_h(h, np.sum(h))
