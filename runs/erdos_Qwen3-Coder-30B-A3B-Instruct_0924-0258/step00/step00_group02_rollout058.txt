def evaluate_erdos_solution(h_vals):
    """Evaluate the C5 constant for given h values."""
    n = len(h_vals)
    dx = 2.0 / n
    h = np.array(h_vals)
    # Compute all overlaps
    conv = np.correlate(h, 1 - h, mode="full")
    # The full convolution has 2*n - 1 elements
    # The valid overlaps are from index n-1 to 2*n-2 (corresponding to k=0 to n-1)
    overlap_values = conv[n-1:]
    max_overlap = np.max(overlap_values) * dx
    return max_overlap
