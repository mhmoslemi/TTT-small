import numpy as np
from scipy import signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    # Ensure the constraint is met
    h = h * (n_points / 2) / np.sum(h)
    
    # Target: minimize max_k int h(x)(1-h(x+k)) dx = max_k correlate(h, 1-h, mode='full')
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Begin optimization loop
    start_time = time.time()
    iter_count = 0
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1:
        # Compute correlations
        corr = np.correlate(h, 1 - h, mode='full')
        # The lag for which we have the max overlap
        max_k = np.argmax(corr)
        # Evaluate the value at max_k
        c5 = corr[max_k]
        
        # If better than current best, update
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Compute gradient w.r.t. h for the max_k lag
        # For lag k, the h[i] contributes to corr[k + i] if 1-h[i] exists
        # Let this be approximate by finite difference
        grad = np.zeros_like(h)
        
        # Gradient at i from lag k0
        # corr[k0] = sum_i h[i] * (1-h[i+k0]) = sum_i h[i] - h[i]*h[i+k0]
        # For h[i], gradient is (1 - h[i+k0]) - h[i] * (1 when i+k0 inside valid range and 0 otherwise)
        # But lag can go from -n+1 to n-1 (so for positive offset, corresponding index is k0 + i)
        # So the derivative of the i-th term is:
        # d/dh[i]( h[i]* (1-h[i+k0]) ) if i+k0 is in [0, n-1] and i in [0,n-1]
        # = 1 - h[i+k0] if (i+k0) >= 0 and (i+k0) < n
        # = 1 - h[i+k0] if 0 <= i+k0 <= n-1 => i in [-k0, n-1-k0]
        # To handle indices correctly, let k_shift = k0 - (n - 1) so 0 -> k_shift + n-1 = k0
        
        k_shift = max_k - (n_points - 1)
        for i in range(n_points):
            j = i + k_shift
            if 0 <= j < n_points:
                grad[i] += 1.0 - h[j]
            else:
                grad[i] += 0.0
        
        # Update h in direction of negative gradient
        h = h - lr * grad / np.linalg.norm(grad) * dx  # Normalize per dx
        
        # Project onto feasible set
        # minimize ||h - h_old||^2 s.t. 0 <= h[i] <= 1, sum(h) = n_points/2
        h = project_onto_feasible_set(h, n_points, dx)
        
        # Decay learning rate
        iter_count += 1
        if iter_count % 100 == 0:
            lr *= 0.95

    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)

def project_onto_feasible_set(h, n_points, dx):
    """Project h onto [0,1]^n and sum(h) = n_points/2"""
    # Linear constraints
    # h >= 0
    # h <= 1
    # sum(h) = n_points/2
    
    # Use CVXPY for projection
    h_var = cp.Variable(n_points)
    objective = cp.Minimize(cp.sum_squares(h_var - h))
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        # Ensure projection is within bounds slightly
        proj_h = np.maximum(0, np.minimum(1, h_var.value))
        # Ensure sum is exactly n_points/2 by rescaling
        proj_h = proj_h * (n_points / 2) / np.sum(proj_h)
        return proj_h
    except:
        # Fallback to manual projection
        new_h = np.clip(h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        return new_h

def compute_c5(h, dx):
    # Evaluate with convolution
    # Should align with what `evaluate_erdos_solution` expects, which computes the max of the correlation
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
