import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a reasonable initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[10:35] = 1.0  # Some intervals at max value
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all overlaps
        unfolded = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Only keep the relevant part, we want k from 0 to n_points-1
        overlaps = unfolded[n_points-1:]
        return np.max(overlaps) * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale so that sum equals target_sum
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Optimizer parameters
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Inner optimization loop with random restarts
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps and identify top k contributors to max
        overlaps = np.correlate(h, 1 - h, mode='full')
        overlaps = overlaps[n_points-1:]
        top_k = np.argsort(overlaps)[-5:]  # Top 5 contributing overlaps
        
        # Compute gradient of current max overlaps
        grad = np.zeros(n_points)
        for k in top_k:
            # Gradient of the overlap term
            for i in range(n_points):
                grad[i] += (1 - h[(i+k) % n_points]) * dx
            
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h = h + velocity
        
        # Project back to feasible region
        h = project_h(h)
        
        # Evaluate current C5
        current_C5 = compute_C5(h)
        
        # Update best if improved
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            
        # Adaptive learning rate
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.95
            if learning_rate < 1e-6:
                learning_rate = 1e-6
        
        # Occasionally restart with new random perturbation
        if iteration % 500 == 0 and iteration > 0:
            # Add small random perturbations
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = project_h(h)
    
    return (list(best_h), best_C5, n_points)
