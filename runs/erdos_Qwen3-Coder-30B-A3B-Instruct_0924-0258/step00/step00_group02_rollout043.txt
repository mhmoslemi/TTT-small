import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for given step function h."""
    dx = 2.0 / len(h)
    # Compute correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Maximize over all lags
    c5 = np.max(corr) * dx
    return c5

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

    dx = 2.0 / n_points
    
    # PSO Parameters
    n_particles = 30
    max_iter = 1200
    c1, c2 = 2.0, 2.0
    inertia = 0.9
    v_max = 0.5
    decay_rate = 0.99
    
    # Initialize particles
    particles = []
    velocities = []
    pBest = []
    pBest_vals = []
    gBest_idx = 0
    gBest_val = float('inf')
    
    # Add initial solution
    particles.append(h0.copy())
    velocities.append(np.zeros(n_points))
    val = evaluate_erdos_solution(h0)
    pBest.append(h0.copy())
    pBest_vals.append(val)
    if val < gBest_val:
        gBest_val = val
        gBest_idx = 0

    # Add other particles with small random perturbations
    for _ in range(n_particles - 1):
        pert = np.random.normal(0, 0.1, n_points)
        h = h0 + pert
        h = project_feasible(h, n_points)
        particles.append(h)
        velocities.append(np.random.uniform(-0.2, 0.2, n_points))

        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] 
                            + c1 * r1 * (pBest[i] - particles[i]) 
                            + c2 * r2 * (pBest[gBest_idx] - particles[i]))
            # clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply feasibility projection
            particles[i] = project_feasible(particles[i], n_points)
