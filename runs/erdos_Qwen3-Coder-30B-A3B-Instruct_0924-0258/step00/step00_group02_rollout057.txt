import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create an initial reasonable guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[1] = 0.6
        h[-1] = 0.4
        h[-2] = 0.6
        h = h * (n_points / 2) / np.sum(h)
        
    # Parameters
    n_points = len(h)
    dx = 2.0 / n_points
    max_iter = 600
    lr = 0.01
    momentum = 0.9
    v = np.zeros_like(h)
    
    def evaluate_C5(h_vec):
        # Compute the overlap integral
        h = h_vec.copy()
        # For each k from 0 to n_points-1, compute overlap_k
        # This is equivalent to computing correlation of h with (1-h) with wraparound
        # Using numpy's correlate is faster
        h_padded = np.concatenate([h, h])
        inv_h = 1 - h
        inv_h_padded = np.concatenate([inv_h, inv_h])
        # Full correlation
        correlation = np.correlate(h_padded, inv_h_padded, mode='full')
        # Extract the relevant part (shifted indices)
        # Corresponds to compute: int h(x)(1-h(x+k)) dx for k = 0 to n_points-1
        # In discrete: sum h[i] * (1 - h[(i+k) % n_points])
        overlaps = correlation[n_points-1:2*n_points-1] * dx
        return np.max(overlaps)
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute all overlaps
        h_padded = np.concatenate([h, h])
        inv_h = 1 - h
        inv_h_padded = np.concatenate([inv_h, inv_h])
        correlation = np.correlate(h_padded, inv_h_padded, mode='full')
        overlaps = correlation[n_points-1:2*n_points-1] * dx
        
        dominate_k = np.argsort(-overlaps)[:5]  # Top 5 k that dominate max
        
        # Compute gradient approximation based only on most dominant overlaps
        # Simple forward difference approximation for gradient of sum of top overlaps
        # This is a simplification to make optimization more stable
        grad = np.zeros_like(h)
        for k in dominate_k:
            # In each overlap_k computation, derivative of sum h[i] (1 - h[(i+k)%n_points])
            # is (1 - h[(i+k)%n_points]) - h[i] (when i -> i+k)
            for i in range(n_points):
                grad[i] += (1 - h[(i + k) % n_points]) * dx
                grad[(i + k) % n_points] -= h[i] * dx

        # Apply momentum
        v = momentum * v + lr * grad
        h = h - v
        
        # Project to feasible set: [0,1] and sum = n_points/2
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total > 0:
            h = h * (n_points / 2) / total

        # Evaluate new solution
        C5 = evaluate_C5(h)
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()

        # Adaptive learning rate
        if iteration % 20 == 0:
            lr *= 0.95

    return (best_h.tolist(), best_C5, n_points)
