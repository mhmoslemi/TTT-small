import numpy as np
from scipy import signal
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction from parent
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Compute initial C5
    def compute_c5(h_vals):
        correlation = signal.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Optimization parameters
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)
    start_time = time.time()
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute all overlaps efficiently
        # The overlap for shift k is sum(h[i] * (1 - h[(i+k) % n])) * dx
        # This corresponds to correlation of h with (1-h) with mode='full'
        corr = signal.correlate(h, 1 - h, mode='full')
        all_k = np.arange(-n_points + 1, n_points)
        max_k_indices = np.argsort(corr)[-5:]  # Top 5 contributing shifts
        dominant_k = all_k[max_k_indices]
        
        # Compute gradient vector for dominant shifts
        grad = np.zeros_like(h)
        for k in dominant_k:
            # Compute the contribution of overlap for shift k
            if k >= 0:
                # h[i] * (1 - h[i+k]) has gradient h[i+k] - 1
                grad += np.concatenate([np.zeros(k), h[:-k]]) if k > 0 else h
            else:
                # h[i] * (1 - h[i+k]) has gradient (1 - h[i+k]) - 0 = 1 - h[i+k]
                grad += np.concatenate([h[-k:], np.zeros(-k)]) if k < 0 else (1 - h)
                
        # Apply weighting and scale gradient
        weights = np.exp(-0.2 * np.abs(dominant_k))
        weights /= np.sum(weights)  # normalize
        weighted_grad = np.zeros_like(h)
        for i, k in enumerate(dominant_k):
            k = int(k)  # convert to int to avoid numpy's dynamic indexing confusion
            if k >= 0:
                temp_grad = np.concatenate([np.zeros(k), h[:-k]]) if k > 0 else h
            else:
                temp_grad = np.concatenate([h[-k:], np.zeros(-k)]) if k < 0 else (1 - h)
            weighted_grad += weights[i] * temp_grad
        
        # Update velocity and parameters
        velocity = momentum * velocity - learning_rate * weighted_grad
        h = h + velocity
        
        # Project h onto feasible region [0,1] and preserve mean
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum  # Rescale to meet integral constraint
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Adaptive learning rate
        if iteration % 10 == 0:
            # If improvement is small, reduce learning rate
            progress = abs(current_c5 - best_c5) if iteration > 10 else 1.0
            if progress < 1e-6:
                learning_rate *= 0.9
                if learning_rate < 1e-6:
                    learning_rate = 1e-6
    
    return (best_h.tolist(), best_c5, n_points)
