def evaluate_C5(h, dx):
    # Correlation with 1-h
    corr = correlate(h, 1-h, mode='full')
    # Extract the middle part which corresponds to valid overlaps
    mid_idx = len(corr) // 2
    # The overlap integral is the max of each lag's correlation
    return np.max(corr[mid_idx - (len(h) - 1): mid_idx + len(h)])
