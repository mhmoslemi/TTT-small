from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from initial_h_values if available, otherwise create a uniform one
    if 'initial_h_values' in globals():

    else:

        h = h * (n_points / 2) / np.sum(h)  # normalize to sum to n_points / 2

    dx = 2.0 / len(h)
    max_iter = 500
    alpha = 0.1
    tol = 1e-4
    start_time = time.time()
    
    # Project h to feasible set
    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (len(h) / 2) / h_sum
        return h
    
    # Evaluate C5
    def evaluate_C5(h):
        # Compute overlap integral
        corr = correlate(h, 1 - h, mode='full')
        # Take maximum over all lags, scaled by dx
        return np.max(corr) * dx
    
    # Initialize
    current_C5 = evaluate_C5(h)
    best_C5 = current_C5
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the correlation
        corr = correlate(h, 1 - h, mode='full')
        
        # Find maximum correlation and corresponding lags
        max_corr = np.max(corr)
        max_lags = np.where(np.abs(corr - max_corr) < 1e-10)[0]
        
        # Compute gradient w.r.t. h based on max_lags
        # This is a simplified gradient approximation
        grad = np.zeros_like(h)
        for k in max_lags:
            # Determine which elements contribute to the gradient at lag k
            # corr[i] = \sum_j h[j] * (1 - h[j+k])
            # We add -1 for each j where k is valid
            # The gradient w.r.t. h[i] counting this correlation is:
            if k < len(h):
                grad[k] += -1
            if k >= len(h):
                k_shift = k - len(h) + 1
                grad[k_shift] += -1
                
        # Compute updated h
        h_new = h - alpha * grad
        
        # Project
        h_new = project_h(h_new)
        
        # Evaluate new C5
        new_C5 = evaluate_C5(h_new)
        
        # Update if improvement
        if new_C5 < current_C5:
            h = h_new
            current_C5 = new_C5
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
        else:
            # Line search or just accept

    # Final evaluation
    final_C5 = evaluate_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
