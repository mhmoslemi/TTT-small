def evaluate_erdos_solution(h_values, n_points):
    """Evaluate the overlap integral for given h_values."""
    dx = 2.0 / n_points

    correlations = np.correlate(h_values, 1 - h_values, mode='full')
    # Select the maximum overlap
    max_overlap = np.max(correlations) * dx
    return max_overlap
