import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize n_points
    n_points = 42  # Use the provided initial construction size
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        assert np.all(h >= 0) and np.all(h <= 1), "Initial h must be in [0,1]"
        assert abs(np.sum(h) * dx - 1.0) < 1e-10, "Initial h must integrate to 1"
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1e-3:
        # Compute full correlation
        corr_full = correlate(h, 1 - h, mode='full')
        # Extract the relevant part (n_points elements starting from the middle)
        lags = corr_full[len(corr_full)//2 : len(corr_full)//2 + n_points]
        c5_val = np.max(lags) * dx
        
        # Find worst lag
        k_star = np.argmax(lags)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_star) % n_points
            grad[i] = 1 - 2 * h[j]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-12:
            grad = grad / grad_norm
        
        # Line search
        alpha = 0.1
        min_alpha = 1e-6
        step_size = 0.8
        max_steps = 20
        
        for _ in range(max_steps):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_c5 = evaluate_erdos_solution(h_new)
            
            if new_c5 < c5_val:
                h = h_new
                c5_val = new_c5
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h.copy()
                break
            else:
                alpha *= step_size
                if alpha < min_alpha:
                    break
        
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for lags 0 to n-1
    corr_full = correlate(h, 1 - h, mode='full')
    # Extract the relevant part (n_elements - n_points to n_elements + n_points - 1)
    # corresponding to lags 0 to n-1
    start_idx = len(corr_full) // 2
    correlations = corr_full[start_idx : start_idx + n]
    return np.max(correlations) * dx
