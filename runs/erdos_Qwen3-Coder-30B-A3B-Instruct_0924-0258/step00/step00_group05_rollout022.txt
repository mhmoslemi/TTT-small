import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    # Compute all correlations with 1-h
    corr = correlate(h, 1 - h, mode='full')
    # Get only the relevant part (positive lags)
    max_val = np.max(corr[n_points-1:])
    return max_val * dx

def project_h(h, n_points):
    """Project h to ensure sum(h) = n_points / 2 and h[i] in [0,1]"""
    dx = 2.0 / n_points
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-10:
        # Adjust to hit target sum
        scale = target_sum / current_sum
        h *= scale
        h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, n_points):
    """Compute gradient w.r.t. the objective"""
    dx = 2.0 / n_points
    # Compute correlations
    corr = correlate(h, 1 - h, mode='full')
    # Find max lag
    max_lag = np.argmax(corr[n_points-1:]) + (n_points-1)
    # Compute gradient based on max lag
    g = np.zeros(n_points)
    for i in range(n_points):
        # Contribution from lag = max_lag - (n_points-1)
        lag = max_lag - (n_points-1)
        # where lag is k, so shift by k
        j = (i + lag) % n_points
        g[i] = 1 - 2*h[i] - (1 - 2*h[j])
    
    return g * dx
