def try_relaxed_lp(h0, n_points, dx, C5_target):
    """Try solving a relaxed linear program"""
    try:
        # Variables
        h = cp.Variable(n_points, bounds=(0, 1))
        
        # Sum constraint
        constraints = [cp.sum(h) == n_points / 2]
        
        # Overlap constraints relaxed as linear
        # For each k, we have sum h[i] * (1 - h[(i+k)%n]) <= C5_target / dx
        for k in range(n_points):
            # This is approximate: use linearization of quadratic terms
            # h[i] * (1 - h[i+k]) <= (h[i] + 1 - h[i+k]) / 2  
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h[i] * (1 - h[j])
            constraints.append(expr <= C5_target / dx)
            
        # Objective: just feasibility
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.GLPK, verbose=False)
        
        if prob.status == 'optimal':
            return True
        else:
            return False
    except:
        return False

def solve_lp_refined(h0, n_points, dx, C5_target):
    """Solve with cutting-plane approach"""
    try:
        # Variables
        h = cp.Variable(n_points, bounds=(0, 1))
        
        # Sum constraint
        constraints = [cp.sum(h) == n_points / 2]
        
        # Iteratively add constraints from the current solution
        h_current = h0.copy()
        for k in range(n_points):
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h[i] * (1 - h[j])
            constraints.append(expr <= C5_target / dx)
        
        # Solve the refined LP
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.GLPK, verbose=False)
        
        if prob.status == 'optimal':
            solution = h.value
            # Project to ensure feasibility
            solution = np.clip(solution, 0, 1)
            solution = solution * (n_points / 2) / np.sum(solution)
            return solution
        else:
            return None
    except:
        return None

def evaluate_erdos_solution(h, n_points, dx):
    """Compute C5 for given h"""
    h_arr = np.array(h)
    correlations = correlate(h_arr, 1 - h_arr, mode="full")
    max_corr = np.max(correlations) * dx
    return max_corr
