import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_c5(h, dx):
    """Evaluate C5 for a given h vector"""
    # Compute overlap integral for all lags
    overlap = np.correlate(h, 1 - h, mode="full")
    # The overlap array has length 2*n_points - 1
    # We want the maximum over all non-trivial lags
    # Lags 0 to n_points-1 correspond to k = 0 to n_points-1
    # But we exclude k=0 (no shift) and k=n_points-1 (full shift)
    max_overlap = np.max(overlap[1:-1]) * dx
    return max_overlap

def project_to_feasible(h):
    """Project h to [0,1] and rescale to meet sum constraint"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (len(h) / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute finite difference gradient of C5 with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    base_c5 = evaluate_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus)
        c5_plus = evaluate_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_feasible(h_minus)
        c5_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Validate constraints
        if np.any(h < 0) or np.any(h > 1) or abs(np.sum(h) - n_points / 2) > 1e-10:
            h = np.ones(n_points) * 0.5
    else:
        h = np.ones(n_points) * 0.5
    
    h = project_to_feasible(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    
    # Optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Local optimization step
        grad = compute_gradient(h, dx)
        # Simple gradient descent step
        step_size = 0.01
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new)
        
        # Low-pass filter step to smooth
        sigma = 0.1 * n_points
        h_filtered = gaussian_filter1d(h_new, sigma=sigma)
        h_filtered = project_to_feasible(h_filtered)
        
        c5_new = evaluate_c5(h_filtered, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_filtered.copy()
            
        h = h_filtered.copy()
        
        # Occasionally check for early termination or reset
        if iter_count % 50 == 0 and best_c5 < 0.38085:
            break
            
        # Safety break - prevent excessive iterations
        if iter_count > 1000:
            break
            
    # Final validation and return
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
