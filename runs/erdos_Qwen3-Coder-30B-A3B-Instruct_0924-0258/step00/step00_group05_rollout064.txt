            # Cool down

def compute_C5(h):
    # Compute the overlap integral C5
    dx = 2.0 / len(h)
    correlation = signal.correlate(h, 1-h, mode="full")
    # Find maximum value among all lags
    max_val = np.max(correlation)
    return max_val * dx
