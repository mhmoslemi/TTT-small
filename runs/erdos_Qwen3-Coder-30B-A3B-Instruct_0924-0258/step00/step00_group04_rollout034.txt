def evaluate_h(h):
    """Evaluate the C5 constant using the provided h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the full correlation with (1-h)
    corr = correlate(h, 1-h, mode="full")
    # Extract the relevant part and normalize
    c5 = np.max(corr) * dx
    return c5

def project_h(h, n_points):
    """Project h to ensure it sums to n_points/2 and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, n_points, mutation_rate=0.05):
    """Mutate h with given rate, then project."""
    h = h.copy()
    indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
    for i in indices:
        h[i] += np.random.normal(0, 0.1)
    return project_h(h, n_points)

def crossover_h(h1, h2, n_points):
    """Perform crossover between h1 and h2."""
    h = h1.copy()

    cuts = [0] + cuts + [n_points]
    for i in range(len(cuts)-1):
        if i % 2 == 1:
            start, end = cuts[i], cuts[i+1]
            h[start:end] = h2[start:end]
    return project_h(h, n_points)

            current_sum = np.sum(h)
            if current_sum > 0:

        pass
    return h
