import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 42  # Use the provided initial construction size
    dx = 2.0 / n_points
    max_iter = 6000
    alpha_init = 0.1
    beta = 0.8
    tol = 1e-6
    
    # Get initial h or create a reasonable starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        assert np.all(h >= 0) and np.all(h <= 1), "Initial h must be in [0,1]"
        assert abs(np.sum(h) * dx - 1.0) < 1e-6, "Initial h must integrate to 1"
    else:
        # Create uniform random with constraint
        h = np.random.rand(n_points)
        h = h * (n_points / 2.0) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = float('inf')
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all lags
        corr = correlate(h, 1 - h, mode="full")
        lag_offset = n_points - 1
        z_k = corr[lag_offset:lag_offset + n_points] * dx
        C5 = np.max(z_k)
        k_star = np.argmax(z_k)
        
        # Update best if improved
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
        
        # Compute gradient for worst lag
        g = np.zeros(n_points)
        for j in range(n_points):
            g[j] = 1 - 2 * h[(j + k_star) % n_points]
        
        # Normalize gradient
        norm_g = np.linalg.norm(g)
        if norm_g < tol:
            continue
        g = g / norm_g
        
        # Backtracking line search
        alpha = alpha_init
        while alpha > 1e-6:
            h_new = h - alpha * g
            # Project to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to satisfy integral constraint
            h_new = h_new * (n_points / 2.0) / np.sum(h_new)
            
            # Evaluate new C5
            corr_new = correlate(h_new, 1 - h_new, mode="full")
            z_k_new = corr_new[lag_offset:lag_offset + n_points] * dx
            C5_new = np.max(z_k_new)
            
            if C5_new < C5:
                h = h_new
                break
            alpha *= beta
    
    # Final validation
    corr_final = correlate(best_h, 1 - best_h, mode="full")
    z_k_final = corr_final[lag_offset:lag_offset + n_points] * dx
    final_C5 = np.max(z_k_final)
    
    return (best_h.tolist(), final_C5, n_points)
