import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h):
    """Evaluate the maximum overlap integral for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx
