import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h):
    """Compute C5 for the given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    overlap = signal.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points / 2 and 0 <= h[i] <= 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.full_like(h, 1.0 / n_points)
    return h

def mutate_h(h, n_points, mutation_rate=0.1):
    """Mutate h by perturbing a fraction of its entries."""
    h_new = h.copy()
    n_mutate = int(mutation_rate * n_points)
    indices = np.random.choice(n_points, n_mutate, replace=False)
    for i in indices:
        h_new[i] += 0.1 * np.random.randn()
    return project_to_feasible(h_new, n_points)

def crossover_h(parent1, parent2):
    """Perform single-point crossover."""
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    n_population = 50
    n_generations = 12
    n_parents = n_population // 2
    
    # Read or create initial solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        # Create initial uniform distribution
        h0 = np.ones(n_points) * 0.5
    
    # Initialize population
    population = [h0]
    for _ in range(n_population - 1):
        h = h0 + np.random.uniform(-0.1, 0.1, n_points)
        population.append(project_to_feasible(h, n_points))
    
    # Evaluate initial population
    fitness = []
    for h in population:
        c5 = evaluate_erdos_solution(h)
        fitness.append(c5)
    
    # Track best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolution loop
    for generation in range(n_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top parents
        parents = population[:n_parents]
        
        # Crossover and mutation
        offspring = []
        while len(offspring) < n_population - n_parents:
            i1, i2 = np.random.choice(n_parents, 2, replace=False)
            child1, child2 = crossover_h(parents[i1], parents[i2])
            child1 = mutate_h(child1, n_points)
            child2 = mutate_h(child2, n_points)
            offspring.append(child1)
            offspring.append(child2)
        
        # Create new population
        population = parents + offspring[:n_population - n_parents]
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness.append(c5)
        
        # Update best
        curr_best_idx = np.argmin(fitness)
        if fitness[curr_best_idx] < best_c5:
            best_h = population[curr_best_idx].copy()
            best_c5 = fitness[curr_best_idx]
    
    return (list(best_h), best_c5, n_points)
