    # Gradient-based optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        if iter_count % 100 == 0:
            # Periodic check of best solution
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            # Adjust learning rate periodically
            lr *= 0.999
        
        # Compute all overlaps and identify top contributing
        dx = 2.0 / n_points
        overlaps = np.correlate(h, 1-h, mode="full") * dx
        k_indices = np.argsort(overlaps)[-5:]  # Top 5 contributing indices
        
        # Compute gradient w.r.t. overlaps for these indices
        grad = np.zeros_like(h)
        for k in k_indices:
            # Convolution derivative for overlap term
            # We compute contribution of h[i] to overlap_k term
            # overlap_k = sum_i h[i] * (1 - h[(i+k) % n_points])
            # d/dh[i] (overlap_k) = 1 - h[(i+k) % n_points] - h[i] * (1 - h[(i+k) % n_points]) if k >= 0 else 0
            for i in range(n_points):
                if i + k < n_points:
                    grad[i] += 1 - h[i+k] - h[i] * (1 - h[i+k])
                else:
                    # wrap around
                    j = i + k - n_points
                    grad[i] += 1 - h[j] - h[i] * (1 - h[j])
        
        # Apply momentum and update
        velocity = momentum * velocity - lr * grad
        h = h + velocity
        
        # Project onto feasible set: [0,1] and sum = n_points/2
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
        # Check if we're getting closer to the target
        if time.time() - start_time > budget_s - 10:
            break
    
    # Final check
    final_c5 = evaluate_erdos_solution(h)
    if final_c5 < best_c5:
        best_h = h.copy()
        best_c5 = final_c5
    
    return (best_h.tolist(), float(best_c5), int(n_points))
