import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluates the overlap integral for the given step function h."""
    # Compute all correlations
    correlations = correlate(h, 1 - h, mode="full")
    # Scale by dx
    dx = 2.0 / len(h)
    correlations *= dx
    # Return maximum overlap
    return np.max(correlations)

def project_h(h, n_points):
    """Projects h onto feasible region: [0,1] and sum = n_points/2."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize sum to exactly n_points/2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, max_lags, n_points):
    """Computes gradient with respect to the maximum contribution lags."""
    # Initialize gradient
    grad = np.zeros_like(h)
    dx = 2.0 / n_points
    
    # Compute gradient more efficiently
    for k in max_lags:
        # For lag k, the derivative w.r.t. h[i] is:
        # -1 (from h[i] term in 1-h[i+k]) if i+k < n_points
        # +1 (from h[i+k] term in 1-h[i+k]) if i < n_points
        for i in range(n_points):
            if i + k < n_points:
                # Contribution to grad[i] from position i+k
                grad[i] -= dx
            if i < n_points:
                # Contribution to grad[i] from position i (if it's part of 1-h[i])  
                grad[i] += dx
    
    return grad

    c5_prev = float('inf')
    current_time = time.time()
    elapsed = current_time - start_time
    
    # Gradient descent loop

        if elapsed > 0.95 * budget_s:

        # Compute overlaps for all lags
        correlations = correlate(h, 1 - h, mode="full")
        dx = 2.0 / n_points
        correlations *= dx
        
        # Find max value and all lags achieving it
        max_val = np.max(correlations)
        max_lags = np.where(np.abs(correlations - max_val) < 1e-10)[0]
        
        # Compute gradient based on maximum lags
        grad = compute_gradient(h, max_lags, n_points)
        
        # Update with step size
        h_new = h - step_size * grad
        h_new = project_h(h_new, n_points)
        
        # Check convergence
        c5_current = max_val
        if abs(c5_prev - c5_current) < tol:
            break
            
        h = h_new
        c5_prev = c5_current
        
        current_time = time.time()
        elapsed = current_time - start_time

    c5_bound = evaluate_erdos_solution(h)
