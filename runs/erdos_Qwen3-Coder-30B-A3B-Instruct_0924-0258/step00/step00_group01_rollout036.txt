import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 47
        h = 0.5 * np.ones(n_points)
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = 1e10
    start_time = time.time()
    
    # Parameters
    alpha = 0.1
    momentum = 0.9
    last_improvement = 0
    iter_count = 0
    
    # Gradient descent with momentum
    v = np.zeros_like(h)
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        
        # Find the maximum correlation
        max_k = np.argmax(corr)
        C5 = corr[max_k] * dx
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            last_improvement = iter_count
        
        if iter_count - last_improvement > 50:
            break
            
        # Compute gradient for the active shift
        grad = np.zeros_like(h)
        k0 = max_k - (n_points - 1)  # convert full correlation index to shift
        
        # Compute partial derivatives for correlation at k0
        for i in range(n_points):
            if i + k0 >= 0 and i + k0 < n_points:
                grad[i] += 1 - h[i + k0]
            if i - k0 >= 0 and i - k0 < n_points:
                grad[i] -= h[i - k0]
        
        # Update with momentum
        v = momentum * v - alpha * grad
        h_new = h + v
        
        # Project back onto feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust sum to be exactly n_points / 2
        actual_sum = np.sum(h_new)
        if abs(actual_sum - n_points / 2) > 1e-6:
            diff = (n_points / 2 - actual_sum) / n_points
            h_new += diff
        
        h = h_new
        
        # Occasionally reduce learning rate
        if iter_count % 10 == 0 and iter_count > 100:
            alpha *= 0.95
            
        # Occasionally restart
        if iter_count % 50 == 0 and np.random.random() < 0.05:
            h = np.random.rand(n_points) * 0.5
            h = h * (n_points / 2) / np.sum(h)
    
    # Final verification
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_C5 = np.max(corr)
    
    return (best_h.tolist(), final_C5, n_points)
