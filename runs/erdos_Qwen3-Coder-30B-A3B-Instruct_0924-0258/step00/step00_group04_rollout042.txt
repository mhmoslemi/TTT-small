import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except:
        # Default construction
        n_points = 42
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure constraints
    assert np.all(h >= 0) and np.all(h <= 1)
    assert np.isclose(np.sum(h), n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Gradient descent
    max_iter = 10000
    lr = 0.1
    decay = 0.9995
    eps = 1e-8
    perturb_freq = 50
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute all overlaps
        conv = correlate(h, 1-h, mode="full")
        overlaps = conv * (2.0 / n_points)
        
        # Find max overlap index
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Compute gradient w.r.t. h for this k
        grad = np.zeros(n_points)
        for j in range(n_points):
            prev_idx = (j - max_k) % n_points
            next_idx = (j + max_k) % n_points
            grad[j] = (2.0 / n_points) * (1 - h[prev_idx] - h[next_idx])
        
        # Gradient step
        h_new = h - lr * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        scale = (n_points / 2) / np.sum(h_new)
        h_new = h_new * scale
        
        # Update
        h = h_new
        
        # Ad hoc perturbations
        if i % perturb_freq == 0:
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h = h * scale
            
        # Evaluate
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        lr *= decay
        
    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h):
    n_points = len(h)
    # Ensure h is valid (0 <= h[i] <= 1, sum = n_points/2)
    assert np.all(h >= 0) and np.all(h <= 1)
    assert np.isclose(np.sum(h), n_points / 2)
    
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode="full")
    result = np.max(corr) * dx
    return result
