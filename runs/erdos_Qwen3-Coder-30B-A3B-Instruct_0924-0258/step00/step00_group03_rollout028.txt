import numpy as np
from scipy import signal
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the overlap integral C5 for the given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution
    conv = signal.correlate(h, 1 - h, mode='full')
    # Find maximum value
    max_val = np.max(conv)
    return max_val * dx

def project_h(h, n_points):
    """Project h onto feasible set: 0 ≤ h[i] ≤ 1 and sum(h) = n_points/2."""
    # First make sure values are within [0,1]
    h = np.clip(h, 0, 1)
    # Use cvxpy to enforce sum constraint
    x = cp.Variable(n_points)
    constraints = [0 <= x, x <= 1, cp.sum(x) == n_points / 2]
    prob = cp.Problem(cp.Minimize(0), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    h_proj = x.value
    return h_proj

def gradient_step(h, k0, n_points):
    """Compute gradient of convolution at lag k0 and update h."""
    dx = 2.0 / n_points
    # Compute reverse convolution gradient
    grad = np.zeros(n_points)
    # For lag k0, the gradient has contributions from h[i] and h[i+k0] (if valid)
    # We consider the contribution of (1-h[k0+j]) to the convolution at k0
    # from h[j]
    for j in range(n_points):
        k = k0 + j
        if 0 <= k < n_points:
            grad[j] += (1 - h[k]) * dx
        # And from h[j] to 1-h[k] where k = j-k0 here (for the other way)
        k = j - k0
        if 0 <= k < n_points:
            grad[j] += h[k] * dx
    return grad
