import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    
    # Ensure constraint sum(h) * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_C5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        # Maintain sum(h) = n_points / 2
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2) / current_sum
        return h
    
    def gradient(h, k_max):
        # Compute gradient for active lags
        grad = np.zeros_like(h)
        n = len(h)
        for k in k_max:
            for i in range(n):
                j = (i + k) % n
                grad[i] += (1 - h[j] - (1 - h[i])) * dx
        return grad
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    start_time = time.time()
    
    # Use simple gradient descent with adaptive step size
    alpha = 0.1
    last_improvement = 0
    stagnation_count = 0
    max_stagnation = 10
    
    for iteration in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_C5 = evaluate_C5(h)
        
        # Compute all correlations and find k_max
        corr = np.correlate(h, 1 - h, mode="full")
        max_val = np.max(corr)
        k_max = np.where(np.abs(corr - max_val) < 1e-10)[0]
        
        # Adjust for Python's convention of k starting from 0
        if len(k_max) == 0:
            k_max = [0]
        k_max = [k - n_points for k in k_max if k >= n_points] + [k for k in k_max if k < n_points]
        if not k_max:
            k_max = [0]
        
        # Compute gradient, do gradient descent
        g = gradient(h, k_max)
        h_new = h - alpha * g
        
        # Project onto feasible region
        h_new = project_h(h_new)
        
        # Evaluate
        new_C5 = evaluate_C5(h_new)
        
        # Accept if better or occasionally accept worse
        if new_C5 <= current_C5 or np.random.random() < 0.01:
            h = h_new
            if new_C5 < best_C5:
                best_h = h.copy()
                best_C5 = new_C5
                last_improvement = 0
                stagnation_count = 0
            else:
                last_improvement += 1
                stagnation_count += 1
        else:
            last_improvement += 1
            stagnation_count += 1
            
        # Adapt step size
        if last_improvement > 5:
            alpha *= 0.95
            last_improvement = 0
        if stagnation_count > max_stagnation:
            h = project_h(h + np.random.normal(0, 0.01, n_points))
            stagnation_count = 0
            alpha *= 0.9
    
    # Final evaluation
    final_C5 = evaluate_C5(best_h)
    
    return (list(best_h), final_C5, n_points)
