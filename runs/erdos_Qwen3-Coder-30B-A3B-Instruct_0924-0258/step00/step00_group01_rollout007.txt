import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        # Fallback: uniform distribution
        n_points = 47
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2.0) / np.sum(h)
        
    dx = 2.0 / n_points
    
    def compute_C5(h_vals):
        # Compute correlation between h and 1-h
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Extract the maximum overlap (excluding edge effects)
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    def objective(h_vals):
        return compute_C5(h_vals)
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2.0
    
    def constraint_bounds(h_vals):
        # Return list of bounds violations
        violations = []
        for i in range(len(h_vals)):
            if h_vals[i] < 0:
                violations.append(h_vals[i])
            if h_vals[i] > 1:
                violations.append(1 - h_vals[i])
        return np.array(violations)
    
    # Optimization parameters
    max_iter = 10000
    current_time = time.time()
    best_h = h.copy()
    best_C5 = objective(h)
    
    # Gradient approximation using finite differences
    epsilon = 1e-4
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        grad = np.zeros_like(h)
        current_obj = objective(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_minus = h.copy()
            h_minus[i] -= epsilon
            
            # Ensure feasibility
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            # Rescale to maintain sum constraint
            sum_plus = np.sum(h_plus)
            sum_minus = np.sum(h_minus)
            h_plus = h_plus * (n_points / 2.0) / sum_plus
            h_minus = h_minus * (n_points / 2.0) / sum_minus
            
            obj_plus = objective(h_plus)
            obj_minus = objective(h_minus)
            grad[i] = (obj_plus - obj_minus) / (2 * epsilon)
            
        # Update h
        step_size = 0.01
        h_new = h - step_size * grad
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        # Evaluate new objective
        new_obj = objective(h_new)
        
        # Accept if improvement
        if new_obj < current_obj:
            h = h_new.copy()
            current_obj = new_obj
            
            if new_obj < best_C5:
                best_C5 = new_obj
                best_h = h.copy()
    
    return (best_h.tolist(), best_C5, n_points)
