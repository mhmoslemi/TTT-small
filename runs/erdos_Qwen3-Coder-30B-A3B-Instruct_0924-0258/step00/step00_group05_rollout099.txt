import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, n_points):
    """Evaluate the Erdős constant C5 for given step function h."""
    dx = 2.0 / n_points
    # Compute all correlations for all lags
    full_corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(full_corr)
    return max_corr * dx

def normalize_h(h, n_points):
    """Normalize h to have sum equal to n_points / 2."""
    dx = 2.0 / n_points
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum != target_sum:
        h *= target_sum / current_sum
    return h

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction or initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Fallback initialization
        n_points = 70
        h = np.full(n_points, 0.5)
        h = normalize_h(h, n_points)
    
    dx = 2.0 / n_points
    
    # Initialization
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, n_points)
    
    # Optimization parameters
    max_iter = 500
    alpha = 0.1
    tolerance = 1e-4
    time_limit = time.time() + budget_s
    last_improvement = 0
    no_improvement_count = 0
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() >= time_limit:
            break
            
        # Compute overlaps and find max lags
        full_corr = correlate(h, 1 - h, mode="full")
        max_val = np.max(full_corr)
        max_lags = np.where(full_corr == max_val)[0]
        
        # Get indices of maximum lags
        # Since mode="full", indices go from 0 to 2*n_points-2
        # Convert to actual lag values [-n_points+1, n_points-1]
        lag_indices = max_lags - (n_points - 1)
        
        # Compute gradient over max lag contributions
        gradient = np.zeros(n_points)
        for lag in lag_indices:
            # lag can be negative (for leading elements)
            for i in range(n_points):
                if 0 <= i + lag < n_points:
                    gradient[i] += (1 - h[i + lag] - (1 - h[i])) * dx
                elif i + lag < 0:
                    # circular wrap-around
                    wrapped_idx = i + lag + n_points
                    gradient[i] += (1 - h[wrapped_idx] - (1 - h[i])) * dx
                else:
                    # wrap-around from higher end
                    wrapped_idx = i + lag - n_points
                    gradient[i] += (1 - h[wrapped_idx] - (1 - h[i])) * dx
        
        # Apply gradient descent
        h_new = h - alpha * gradient
        
        # Project onto feasible set
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        new_C5 = evaluate_erdos_solution(h_new, n_points)
        
        # Check if improved
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
            last_improvement = iteration
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Update h
        h = h_new.copy()
        
        # Adaptive step size
        improvement = best_C5 - new_C5
        if improvement > 0:
            # Slightly increase alpha
            alpha = min(1.0, alpha * 1.05)
        else:
            # Reduce alpha if no improvement
            alpha *= 0.95
        
        # Perturbation if stagnating
        if no_improvement_count > 10:
            noise = np.random.normal(0, 0.01, n_points)
            h += noise
            h = project_h(h, n_points)
            no_improvement_count = 0

    # Final check
    final_C5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), final_C5, n_points)
