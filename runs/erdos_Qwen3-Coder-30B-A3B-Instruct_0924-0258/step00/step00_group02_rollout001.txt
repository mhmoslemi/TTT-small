import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create initial guess: 45 points, 1/2 area
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    # Normalize to have sum = n_points / 2
    h *= (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization parameters
    lr = 0.01
    decay = 0.999
    max_iter = int(budget_s)
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all cross-correlations
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Compute gradient for the worst case lag
        # Since correlation is symmetric, we compute it manually
        grad = np.zeros_like(h)
        for j in range(len(h)):
            if k_max < len(h):
                if j + k_max < len(h):
                    grad[j] += -1.0
            else:
                # Handle wraparound; we're interested in "full" correlation
                shift = k_max - len(h)
                if j - shift >= 0:
                    grad[j] += -1.0
        
        # Apply gradient
        h = h - lr * grad
        lr *= decay

        # Projection step
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
                
    return (best_h.tolist(), best_c5, len(best_h))
