import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h):
    # Compute the overlap integral for all shifts
    n = len(h)
    dx = 2.0 / n
    # Use correlate with flipped h to compute the overlap
    # 1-h is the flipped function
    corr = correlate(h, 1 - h, mode='full')
    # max_k of int h(x)(1-h(x+k))dx
    # corr[k] represents sum of h[i]*(1-h[i+k]) for all valid i
    # The maximum is over all k shifts, which corresponds to 
    # max of corr from index n-1 to 2*n-2
    max_corr = np.max(corr[n-1:2*n-1])
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # sum(h) should be this
    T0 = 1.0
    alpha = 0.999
    sigma0 = 0.1
    max_iter = int(budget_s * 1000)  # Roughly 1000 iterations/sec
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        assert np.abs(np.sum(h) - target_sum) < 1e-10, "Initial h doesn't sum to target"
    else:
        # Construct initial solution
        h = np.ones(n_points) * 0.5  # Start with uniform distribution
        h = h * target_sum / np.sum(h)  # Normalize to target sum
    
    # Use initial feasible h
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    start_time = time.time()
    
    # Annealing
    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Temperature
        T = T0 * (alpha ** t)
        
        # Mutation: Add noise
        # decrease sigma as annealing progresses
        sigma = sigma0 * (alpha ** (t / 200))
        noise = np.random.normal(0, sigma, n_points)
        h_new = h + noise
        
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize so sum is target_sum
        h_new = h_new * target_sum / np.sum(h_new)
        
        # Evaluate
        C5_new = evaluate_C5(h_new)
        
        # Acceptance probability
        delta_E = C5_new - best_C5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_new.copy()
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
    
    return (best_h.tolist(), best_C5, n_points)
