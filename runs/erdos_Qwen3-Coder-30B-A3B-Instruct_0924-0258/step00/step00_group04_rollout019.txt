def evaluate_erdos_solution(h):
    """Evaluate C5 = max_k int h(x)(1 - h(x+k)) dx"""
    n_points = len(h)
    dx = 2.0 / n_points
    # This function uses the correlation approach from scipy.signal
    
    # We can compute all integrals via full correlation.
    # Or compute sum in a vectorized way for each k.
    correlations = []
    for k in range(n_points):
        # This is sum_i h[i] (1 - h[(i+k)%n])  
        # = sum_i h[i] - sum_i h[i] h[(i+k)%n]
        # = 1 - sum h[i] h[(i+k)%n]   
        integral = 0.0
        for i in range(n_points):
            j = (i + k) % n_points
            integral += h[i] * (1 - h[j]) 
        integral *= dx
        correlations.append(integral)
        
    return max(correlations)
