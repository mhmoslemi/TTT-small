import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given step function h."""
    # Compute correlation for all lags (full mode)
    correlation = signal.correlate(h, 1 - h, mode="full")
    # Take maximum over all lags and scale by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_to_feasible(h):
    """Project h to the feasible region: sum=h_sum, 0<=h<=1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    target_sum = 1.0  # because sum(h) * dx = 1 => sum(h) = 1/dx = n_points / 2, but we scale dx = 2/n_points,
                      # so sum(h) must be 1 for the constraint to hold
    if h_sum > 0:
        h = h * (target_sum / h_sum)
    return h

def initialize_population(n_points, pop_size, initial_h):
    """Initialize genetic algorithm population."""
    pop = []
    # Add the initial solution
    pop.append(initial_h.copy())
    # Generate others randomly
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h)
        pop.append(h)
    return pop

def crossover(parent1, parent2):
    """Generate two children from two parents."""
    child1 = (parent1 + parent2) / 2.0
    child2 = (parent1 + parent2) / 2.0
    # Project back to feasible set
    child1 = project_to_feasible(child1)
    child2 = project_to_feasible(child2)
    return child1, child2

def mutate(h, mutation_rate=0.05, sigma=0.05):
    """Mutate a solution by adding Gaussian noise to a fraction of elements."""
    h_new = h.copy()
    indices = np.random.choice(len(h), size=int(len(h) * mutation_rate), replace=False)
    for idx in indices:
        h_new[idx] += np.random.normal(0, sigma)
    return project_to_feasible(h_new)

def tournament_selection(pop, fits, k=2):
    """Select one individual from the population using tournament selection."""
    idxs = np.random.choice(len(pop), size=k, replace=False)
    best_idx = idxs[np.argmin([fits[i] for i in idxs])]
    return pop[best_idx]
