import numpy as np
import time
from scipy import signal

def evaluate_C5(h):
    """Evaluate the maximum overlap integral for given h."""
    # Compute all correlations
    corr = signal.correlate(h, 1 - h, mode='full')
    # Extract the relevant part (the valid overlaps)
    # The full correlation has length 2*n - 1
    n = len(h)
    # The overlap integrals correspond to lags from -(n-1) to (n-1)
    # We're interested in the max value among all those overlaps
    # Which are the middle 2*n - 1 values in corr (middle)
    mid_idx = len(corr) // 2
    # First valid overlap corresponds to lag 0 in the middle
    # Last valid overlap corresponds to lag 2*n-2 in corr
    # But also, for lag 0, h[0] * (1 - h[0]) and so on
    # So we need to look carefully at how the overlap is defined
    # After checking the signal.correlate, the index for lag k
    # is at corr[mid_idx + k]
    # Let's find all the overlaps:
    # the actual overlap integral for lag k, where k from -n+1 to n-1
    # is the sum from i = 0 to n-1-k of h[i] * (1 - h[i+k])
    # corresponds to sum of correlation from i = 0 to n-1 of h[i] * h[i+k] 
    # except that we want h[i]*(1 - h[i+k])
    # which equals h[i] - h[i]*h[i+k]
    overlaps = []
    for k in range(-(n-1), n):  # from -n+1 to n-1
        start = max(0, -k)
        end = min(n, n - k) 
        if start < end:
            overlap = np.sum(h[start:end] * (1 - h[start+k:end+k]))
            overlaps.append(overlap)
    # Return max over the overlaps
    return max(overlaps)


def run(seed=42, budget_s=600, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)

    # Config
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # sum(h) = 1 in continuous case => sum(h)*dx = 1 => sum(h) = 1/dx = 35

    # Initialize from the global initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a reasonable initial configuration
        # Start with a piecewise constant function
        h = np.ones(n_points) * 0.5

    # Ensure h sums to target_sum
    h = h * (target_sum / np.sum(h))

    # Check initial feasibility
    assert np.all(h >= 0) and np.all(h <= 1), "Initial h should be in [0,1]"
    assert abs(np.sum(h) - target_sum) < 1e-6, "Initial h should sum to target_sum"

    # Optimization parameters
    T0 = 1.0
    alpha = 0.999
    sigma0 = 0.1
    iteration = 0
    start_time = time.time()
    
    # Store best
    best_h = h.copy()
    best_C5 = evaluate_C5(h)

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Update temperature and sigma
        T = T0 * (alpha ** iteration)
        sigma = sigma0 * (alpha ** (iteration / 200))
        
        # Generate candidate solution
        h_new = h + np.random.normal(0, sigma, n_points)
        h_new = np.clip(h_new, 0, 1)
        
        # Project back to sum = target_sum
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Evaluate new solution
        C5_new = evaluate_C5(h_new)
        
        # Accept or reject
        if C5_new < best_C5:
            # Always accept better
            h = h_new.copy()
            best_C5 = C5_new
            best_h = h_new.copy()
        else:
            # Accept with Metropolis probability
            delta = C5_new - best_C5
            if np.random.random() < np.exp(-delta / T):
                # Accept
                h = h_new.copy()
        
        iteration += 1
        
        if iteration % 100 == 0:
            # Periodically update best if still holding
            if evaluate_C5(h) < best_C5:
                best_C5 = evaluate_C5(h)
                best_h = h.copy()

    return (best_h.tolist(), best_C5, n_points)
