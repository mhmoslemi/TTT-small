import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    population_size = 50
    generations = 12
    elite_size = population_size // 2
    mutation_rate = 0.1
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        population.append(h0)
    else:
        # Create a basic initial solution
        h0 = np.ones(n_points) * 0.5
        population.append(h0)
    
    # Generate rest of population by adding noise
    for _ in range(population_size - 1):
        noise = np.random.normal(0, 0.1, n_points)
        h = h0 + noise
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Track the best solution
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Evolution loop
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection
        parents = population[:elite_size]
        
        # Crossover and mutation
        new_population = parents.copy()
        
        # Generate offspring
        while len(new_population) < population_size:
            # Select two parents
            p1, p2 = np.random.choice(elite_size, 2, replace=False)
            parent1, parent2 = parents[p1], parents[p2]
            
            # Single-point crossover
            crossover_point = np.random.randint(1, n_points)
            offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutate
            for offspring in [offspring1, offspring2]:
                # Mutate some genes
                num_mutations = int(mutation_rate * n_points)
                mutation_indices = np.random.choice(n_points, num_mutations, replace=False)
                for i in mutation_indices:
                    offspring[i] += 0.1 * np.random.randn()
                
                # Clip
                offspring = np.clip(offspring, 0, 1)
                
                # Rescale to maintain constraint
                offspring = offspring * (n_points / 2) / np.sum(offspring)
                
                new_population.append(offspring)
        
        # Trim population
        population = new_population[:population_size]
    
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
