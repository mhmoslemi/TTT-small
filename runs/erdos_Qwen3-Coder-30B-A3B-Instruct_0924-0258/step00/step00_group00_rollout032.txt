import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h[0] = 0.2
        h[1] = 0.8
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Frank-Wolfe algorithm
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute cross-correlations
        h_rev = np.flip(1 - h)
        cross_correlations = correlate(h, h_rev, mode='full')
        cross_correlations = cross_correlations * dx
        
        # Find maximum
        max_idx = np.argmax(cross_correlations)
        max_lag = max_idx - (len(h) - 1)
        max_val = cross_correlations[max_idx]
        
        # Update best if applicable
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
        
        # Gradient calculation for the maximizing lag
        # Gradient of h[i] with respect to cross_correlation at lag k
        gradients = np.zeros(n_points)
        h_rev = np.flip(1 - h)
        grad_h = np.zeros(n_points)
        
        # Compute gradient directly using convolution of h with flipped (1-h)
        for i in range(n_points):
            if max_lag >= 0:
                grad_h[i] = h_rev[max_lag + i] if max_lag + i < len(h_rev) else 0
            else:
                grad_h[i] = h_rev[i] if i + max_lag >= 0 else 0
                
        # Frank-Wolfe step: find the direction
        for i in range(n_points):
            gradients[i] = grad_h[i]
            
        # Sort indices based on gradients to pick S elements
        sorted_indices = np.argsort(-gradients)
        
        # Simple line search for step size
        step_sizes = [0.1, 0.25, 0.5, 0.75, 1.0]
        best_h_new = best_h.copy()
        best_new_c5 = best_c5
        
        for gamma in step_sizes:
            # Simple modification update using the best gradient direction
            h_new = h.copy()
            for i in sorted_indices[:n_points//2]:
                h_new[i] = np.clip(h_new[i] + gamma * gradients[i], 0, 1)
            
            # Rescale to maintain sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Recompute C5 for this step
            h_rev = np.flip(1 - h_new)
            cross_correlations_new = correlate(h_new, h_rev, mode='full')
            cross_correlations_new = cross_correlations_new * dx
            new_c5 = np.max(cross_correlations_new)
            
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_h_new = h_new.copy()
                
        h = best_h_new.copy()
        if np.abs(best_c5 - best_new_c5) < 1e-6:
            break
        best_c5 = best_new_c5
        
    return (list(best_h), float(best_c5), int(n_points))
