import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except:
        # Generate initial construction
        n_points = 45
        h0 = np.random.rand(n_points)
        h0 = h0 / np.sum(h0) * (n_points / 2)
    
    # Parameters for PSO
    n_particles = 30
    inertia_weight = 0.7
    cognitive_weight = 2.0
    social_weight = 2.0
    max_iterations = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    global_best = h0.copy()
    global_best_val = float('inf')
    
    # Initialize particles
    for i in range(n_particles):
        if i == 0:
            h = h0.copy()
        else:
            h = h0 + np.random.normal(0, 0.05, n_points)
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        val = evaluate_overlap(h)
        personal_best.append(h.copy())
        personal_best_vals.append(val)
        if val < global_best_val:
            global_best_val = val
            global_best = h.copy()
    
    start_time = time.time()
    iteration = 0
    
    # PSO main loop
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                             cognitive_weight * r1 * (personal_best[i] - particles[i]) + 
                             social_weight * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
            
            # Evaluate
            val = evaluate_overlap(particles[i])
            
            # Update personal best
            if val < personal_best_vals[i]:
                personal_best_vals[i] = val
                personal_best[i] = particles[i].copy()
            
            # Update global best
            if val < global_best_val:
                global_best_val = val
                global_best = particles[i].copy()
        
        iteration += 1
    
    return (global_best, global_best_val, n_points)

def evaluate_overlap(h):
    dx = 2.0 / len(h)
    # Compute convolution of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Take maximum and multiply
    max_val = np.max(corr) * dx
    return max_val
