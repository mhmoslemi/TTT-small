import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations for all lags
    corr = correlate(h, 1 - h, mode='full')
    # The overlap integral is the max of these correlations scaled by dx
    c5 = np.max(corr) * dx
    return c5

def project_h(h, n_points):
    """Project h onto feasible region: clip to [0,1] and normalize sum"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, max_lags, n_points, dx):
    """Compute gradient w.r.t. h for the maximum overlap lags"""
    g = np.zeros(n_points)
    # For each lag k contributing to max overlap
    for k in max_lags:
        # Gradient contribution from correlation term
        # The gradient w.r.t. h[i] is sum of (1-h[i+k]) for valid i,k
        # and sum of -h[i-k] for valid i,k
        if k == 0:
            # This case is handled correctly by the full correlation
            pass
        # Use the fact that correlate computes weighted sums that
        # are essentially dot products. This simplifies to contributions
        # of entities at different time lags.
        # Perform correlation derivative analytically:
        # For 1-h at lag k, terms are (1-h[i+k]) contributions
        # For h at lag k, terms are -h[i] contributions in (1-h[i+k]) context
        # Total gradient terms at each i:
        # At position i, (i+k) is the time_index for the (1-h) term
        # which impacts d/dh[i] contribution as 1 if i+k is valid and 0 otherwise
        # But since we're maximizing the max, we do this in steps for gradient.
        for i in range(n_points):
            if i+k < n_points:
                g[i] += 1.0
            if i-k >= 0:
                g[i] -= 1.0
    return g * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h with either provided initial_h_values or a uniform construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)
    dx = 2.0 / n_points
    
    # Ensure sum constraint is met
    h = project_h(h, n_points)
    
    max_iter = 500
    alpha = 0.1
    tol = 1e-4
    
    last_c5 = evaluate_erdos_solution(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        c5_values = corr * dx
        # Find maximum and lags contributing to it (there could be multiple)
        max_val = np.max(c5_values)
        max_lags = np.where(np.abs(c5_values - max_val) < 1e-10)[0]
        # Adjust indices from full correlation (center_offset to [0, 2*npoints-1])
        center_offset = n_points - 1
        effective_lags = max_lags - center_offset
        
        # Compute gradient for contributing lags
        g = np.zeros(n_points)
        for k in effective_lags:
            for i in range(n_points):
                # Gradient term involves influence on the maximum correlation
                if 0 <= i+k < n_points:
                    g[i] += 1.0
                if 0 <= i-k < n_points and k != 0:
                    g[i] -= 1.0
                    
        # Update step
        h_new = h - alpha * g
        
        # Projection
        h_new = project_h(h_new, n_points)
        
        # Evaluate new C5
        new_c5 = evaluate_erdos_solution(h_new)
        
        if new_c5 < last_c5:
            h = h_new
            last_c5 = new_c5
        else:
            # Accept improvement or stop if not making progress
            if abs(new_c5 - last_c5) < tol:
                break
            elif new_c5 > last_c5 and iteration < 20: # Allow some initial noise
                pass # Skip update, continue
    
    return (h, last_c5, n_points)
