import numpy as np
from scipy import signal
import time
import random

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute overlap integrals for all shifts
    overlap = signal.correlate(h, 1 - h, mode='full') * dx
    # The maximum overlap is the maximum value in overlap array
    max_overlap = np.max(overlap)
    return max_overlap

def project_to_feasible(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def crossover(parent1, parent2):
    """Single-point crossover"""
    n = len(parent1)
    crossover_point = random.randint(1, n - 1)
    offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return offspring1, offspring2

def mutate(h, mutation_rate):
    """Mutate h by adding Gaussian noise and reprojecting"""
    h = h.copy()
    n = len(h)
    n_mutate = int(n * mutation_rate)
    indices = random.sample(range(n), min(n_mutate, n))
    for i in indices:
        h[i] += 0.1 * np.random.randn()
    return project_to_feasible(h, len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        # Create an initial construction manually
        n_points = 45
        initial_h = np.full(n_points, 0.5)
        initial_h[10:20] = 1.0
        initial_h[25:35] = 1.0
        initial_h = project_to_feasible(initial_h, n_points)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    population_size = 50
    generations = 12
    mutation_rate = 0.1  # 10% of genes mutated
    tournament_size = 5

    # Initial population
    population = [initial_h.copy()]
    for i in range(population_size - 1):
        perturbed = initial_h + np.random.normal(0, 0.05, n_points)
        population.append(project_to_feasible(perturbed, n_points))

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate population
        evals = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            evals.append((h, c5))
        
        # Sort by fitness (ascending)
        evals.sort(key=lambda x: x[1])
        
        # Keep best individual
        if evals[0][1] < best_c5:
            best_h = evals[0][0].copy()
            best_c5 = evals[0][1]
        
        # Select parents (top half)
        parents = [h for h, _ in evals[:population_size//2]]
        
        # Create new population
        new_population = parents.copy()
        
        # Crossover
        while len(new_population) < population_size:
            p1, p2 = random.sample(parents, 2)
            offspring1, offspring2 = crossover(p1, p2)
            new_population.append(offspring1)
            if len(new_population) < population_size:
                new_population.append(offspring2)
        
        # Mutation
        for i in range(len(new_population)):
            if random.random() < 0.7:  # 70% mutation rate
                new_population[i] = mutate(new_population[i], mutation_rate)
        
        population = new_population

    if evaluate_erdos_solution(best_h) < best_c5:
        best_c5 = evaluate_erdos_solution(best_h)
        
    return (best_h.tolist(), best_c5, n_points)
