import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    # Compute overlap integral for all lags
    full_corr = correlate(h, 1 - h, mode="full")
    # Extract the relevant part (corresponding to actual overlaps)
    n = len(h)
    start_idx = n - 1
    overlaps = full_corr[start_idx:start_idx + n] * dx
    return np.max(overlaps)

def project_h(h, target_sum):
    # Clip to [0,1], then rescale to target sum
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, k_star):
    # Gradient w.r.t. h: d/dh[i] of max_k of overlap
    # For the worst lag k*, we have: gradient[i] = 1 - 2*h[(i + k_star) % n]
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        j = (i + k_star) % n
        grad[i] = 1 - 2 * h[j]
    return grad

def line_search(h, grad, alpha_init=0.1, backtrack=0.8, max_iter=100):
    alpha = alpha_init
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_proj = project_h(h_new, np.sum(h))
        if np.all(h_proj >= 0) and np.all(h_proj <= 1):
            return alpha, h_proj
        alpha *= backtrack
        if alpha < 1e-6:
            return alpha, h_proj
    return alpha, h_proj

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert np.all(h >= 0) and np.all(h <= 1)
        assert abs(np.sum(h) - target_sum) < 1e-6
    else:
        # Initialize randomly and rescale
        h = np.random.rand(n_points)
        h = project_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    start_time = time.time()
    
    # Iteration loop with time limit
    max_iter = 10000
    iter_counter = 0
    
    while iter_counter < max_iter and (time.time() - start_time) < budget_s - 1:
        # Compute overlap for all lags and find worst
        full_corr = correlate(h, 1 - h, mode="full")
        start_idx = n_points - 1
        overlaps = full_corr[start_idx:start_idx + n_points] * dx
        k_star = np.argmax(overlaps)
        
        # Compute gradient
        grad = compute_gradient(h, k_star)
        
        # Line search
        alpha, h_new = line_search(h, grad)
        
        # Update
        h = h_new
        c5 = evaluate_overlap(h, dx)
        
        # Track best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        iter_counter += 1
        
    return (best_h.tolist(), best_c5, n_points)
