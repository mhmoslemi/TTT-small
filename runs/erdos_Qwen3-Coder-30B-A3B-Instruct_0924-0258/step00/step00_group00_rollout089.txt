import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 2400  # rough estimate for 600s budget
    perturb_steps = 20
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Normalize to sum to 1
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    iter_count = 0
    last_improvement = 0
    
    # Gradient descent loop
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 0.1:
        # Compute all cross-correlations
        corr = np.correlate(h, 1 - h, mode='full')
        k_indices = np.arange(-n_points + 1, n_points)
        max_k = k_indices[np.argmax(corr)]
        max_corr = np.max(corr)
        
        # Compute gradient w.r.t. h for the maximum correlation
        grad = np.zeros(n_points)
        if max_k >= 0:
            for i in range(n_points - max_k):
                grad[i] += (1 - h[i + max_k])
        else:
            for i in range(-max_k, n_points):
                grad[i] += (1 - h[i + max_k])
        
        # Line search with different step sizes
        alpha_candidates = [0.1, 0.5, 1.0]
        best_alpha = 0.0
        best_new_h = h.copy()
        best_new_c5 = best_c5
        
        for alpha in alpha_candidates:
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            
            # Rescale to maintain constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_new_h = h_new.copy()
                best_alpha = alpha
                
        # If no improvement, perturb
        if best_new_c5 >= best_c5:
            last_improvement += 1
            if last_improvement > perturb_steps:
                h_pert = h + 0.01 * np.random.randn(n_points)
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
                h = h_pert
                last_improvement = 0
        else:
            h = best_new_h
            best_c5 = best_new_c5
            last_improvement = 0
            
        if best_c5 < evaluate_c5(best_h, dx):
            best_h = h.copy()
            
        iter_count += 1
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
