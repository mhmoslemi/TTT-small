import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute overlap for all shifts
    overlap = correlate(h, 1 - h, mode="full") * dx
    # The maximum overlap is the maximum value
    return np.max(overlap)

def initialize_population(initial_h_values, n_points, pop_size=50):
    """Initialize population around the initial solution with perturbations"""
    population = [initial_h_values]
    for _ in range(pop_size - 1):
        # Create a perturbed version of initial_h_values
        h_perturbed = initial_h_values + 0.1 * (np.random.rand(n_points) - 0.5)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Rescale to maintain the integral constraint
        h_sum = np.sum(h_perturbed)
        h_perturbed *= (n_points / 2) / h_sum
        population.append(h_perturbed)
    return population

def crossover(parent1, parent2):
    """Single-point crossover of two parent solutions"""
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return offspring1, offspring2

def mutate(h, mutation_rate=0.1):
    """Mutate h by adding random noise to a fraction of the entries"""
    n_points = len(h)
    indices_to_mutate = np.random.choice(n_points, size=int(mutation_rate * n_points), replace=False)
    new_h = h.copy()
    new_h[indices_to_mutate] += 0.1 * np.random.randn(len(indices_to_mutate))
    new_h = np.clip(new_h, 0, 1)
    # Adjust the sum to be exactly n_points / 2
    h_sum = np.sum(new_h)
    new_h *= (n_points / 2) / h_sum
    return new_h

def run(seed=42, budget_s=600, **kwargs):
    """Run the genetic algorithm to find an improved C5 bound"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Early exit if no initial construction
    try:
        initial_h_values
    except NameError:
        initial_h_values = np.ones(45) * 0.5  # fallback
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    pop_size = 50
    generations = 12
    mutation_rate = 0.1
    
    population = initialize_population(initial_h_values, n_points, pop_size)
    
    best_h = initial_h_values
    best_c5 = evaluate_erdos_solution(best_h)
    
    for gen in range(generations):
        # Check if time allows the next generation
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break

        # Evaluate all individuals in population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness.append((c5, h))
        
        # Sort by fitness (minimize C5)
        fitness.sort(key=lambda x: x[0])
        
        # Update best solution
        if fitness[0][0] < best_c5:
            best_c5 = fitness[0][0]
            best_h = fitness[0][1].copy()
        
        # Select top 50% (25 individuals)
        parents = [h for _, h in fitness[:pop_size//2]]
        
        # Generate offspring
        offspring = []
        for _ in range(pop_size // 2):
            # Randomly select two parents
            parent1, parent2 = np.random.choice(parents, size=2, replace=False)
            off1, off2 = crossover(parent1, parent2)
            offspring.append(mutate(off1, mutation_rate))
            offspring.append(mutate(off2, mutation_rate))
        
        # Replace population with top individuals and offspring
        # Keep top 25 parents and 25 mutated offspring
        population = parents[:pop_size//2] + offspring[:pop_size//2]
        
    return (best_h, best_c5, n_points)
