import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate C5 for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # The max value corresponds to the middle of the correlation
    c5 = np.max(corr) * dx
    return c5

def project_and_rescale(h, n_points):
    """Project h to [0,1] and rescale to sum to n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2, n_points):
    """Perform crossover between two parents"""
    # Select random crossover points
    n_crossover = np.random.randint(2, 4)
    points = sorted(np.random.choice(n_points, n_crossover, replace=False))
    points = [0] + points + [n_points]
    
    child = np.zeros(n_points)
    for i in range(len(points)-1):
        start, end = points[i], points[i+1]
        if i % 2 == 0:
            child[start:end] = parent1[start:end]
        else:
            child[start:end] = parent2[start:end]
    
    return project_and_rescale(child, n_points)

def mutate(h, n_points, mutation_rate=0.05):
    """Mutate a solution with some probability"""
    h = h.copy()
    n_mutate = max(1, int(n_points * mutation_rate))
    indices = np.random.choice(n_points, n_mutate, replace=False)
    for i in indices:
        h[i] += np.random.normal(0, 0.1)
    return project_and_rescale(h, n_points)

def local_optimize(h, n_points):
    """Apply local optimization to improve h"""
    def objective(h_flat):
        h = h_flat.copy()
        h = project_and_rescale(h, n_points)
        return evaluate_c5(h)
    
    # Use L-BFGS-B with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 50, 'maxfun': 100})
    if result.success:
        h = project_and_rescale(result.x, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 42  # Use the provided initial construction if available
    pop_size = 50
    max_generations = 200
    stagnation_limit = 50
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h0 = project_and_rescale(np.array(initial_h_values), n_points)
    else:
        h0 = np.random.rand(n_points)
        h0 = project_and_rescale(h0, n_points)
    
    # Add the initial solution and random ones to population
    population.append(h0)
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_and_rescale(h, n_points)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    stagnation_count = 0
    current_mutation_rate = 0.05
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Evaluate all individuals
        fitness = []
        for i, h in enumerate(population):
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        
        # Keep top individuals
        elite_size = max(1, pop_size // 5)
        top_individuals = population[:elite_size]
        
        # Create new generation
        new_population = top_individuals[:]
        while len(new_population) < pop_size:
            # Crossover
            parent1, parent2 = np.random.choice(top_individuals, 2, replace=False)
            child = crossover(parent1, parent2, n_points)
            
            # Mutation
            if np.random.rand() < 0.5:
                child = mutate(child, n_points, current_mutation_rate)
            
            # Local optimization
            if np.random.rand() < 0.3:
                child = local_optimize(child, n_points)
            
            new_population.append(child)
        
        population = new_population
        
        # Update stagnation and mutation rate
        if gen > 0 and gen % 10 == 0:
            new_best_c5 = min(fitness)
            if abs(new_best_c5 - best_c5) < 1e-6 or new_best_c5 >= best_c5:
                stagnation_count += 1
                if stagnation_count > stagnation_limit:
                    current_mutation_rate = 0.1  # Increase mutation rate
            else:
                stagnation_count = 0
                current_mutation_rate = 0.05
    
    # Final evaluation of the best solution
    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
