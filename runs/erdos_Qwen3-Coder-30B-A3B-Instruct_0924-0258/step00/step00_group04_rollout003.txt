import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # Take only the relevant part (k from 0 to n-1)
    c5_val = np.max(corr[n-1:]) * dx
    return c5_val

def project_h(h, n_points):
    # Ensure values are in [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to satisfy integral constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def obj_func(h, n_points):
    # For gradient estimation
    return evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert len(h) == n_points
        assert np.all(h >= 0) and np.all(h <= 1)
        assert abs(np.sum(h) * dx - 1.0) < 1e-10
    else:
        h = np.ones(n_points) * 0.5
        h = project_h(h, n_points)
        
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 0.1:
        # --- Local Optimization Step ---
        def fun(h_vec):
            return evaluate_c5(h_vec)
        
        def jac(h_vec):
            # Using finite differences
            eps = 1e-6
            grad = np.zeros_like(h_vec)
            for i in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_minus = h_vec.copy()
                h_minus[i] -= eps
                grad[i] = (fun(h_plus) - fun(h_minus)) / (2 * eps)
            return grad
        
        # Optimize using L-BFGS-B
        try:
            # Make a copy and ensure valid constraints
            h_trial = h.copy()
            h_trial = project_h(h_trial, n_points)
            res = minimize(fun, h_trial, method='L-BFGS-B', jac=jac, bounds=[(0,1)]*n_points, options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                h = res.x.copy()
                h = project_h(h, n_points)
        except:
            pass
        
        # --- Low-Pass Filtering ---
        # Smooth with Gaussian filter
        sigma = 0.1 * n_points
        h_filtered = gaussian_filter1d(h, sigma=sigma)
        h_filtered = project_h(h_filtered, n_points)
        h = h_filtered
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
