import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate C5 for given h"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1, sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.ones(n_points) * 0.5
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial solution if available, otherwise create one
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 45
        h0 = np.ones(n_points) * 0.5
    
    # PSO parameters
    n_particles = 30
    max_iter = 120  # Adjusted for budget
    omega = 0.9
    c1 = 2.0
    c2 = 2.0
    v_max = 0.5
    perturb_scale = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pBest = []
    pBest_vals = []
    
    # First particle is the initial solution
    particles.append(h0.copy())
    velocities.append(np.zeros(n_points))
    pBest.append(h0.copy())
    pBest_vals.append(evaluate_erdos_solution(h0))
    
    # Generate other particles
    for i in range(1, n_particles):
        perturb = np.random.uniform(-perturb_scale, perturb_scale, n_points) * h0
        h = h0 + perturb
        h = project_to_feasible(h, n_points)
        particles.append(h)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        pBest.append(h.copy())
        pBest_vals.append(evaluate_erdos_solution(h))
    
    # Global best
    gBest_idx = np.argmin(pBest_vals)
    gBest = pBest[gBest_idx].copy()
    gBest_val = pBest_vals[gBest_idx]
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        current_vals = []
        for i in range(n_particles):
            h = particles[i]
            c5 = evaluate_erdos_solution(h)
            current_vals.append(c5)
            
            # Update personal best
            if c5 < pBest_vals[i]:
                pBest[i] = h.copy()
                pBest_vals[i] = c5
                
                # Update global best
                if c5 < gBest_val:
                    gBest = h.copy()
                    gBest_val = c5
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            v = omega * velocities[i] + c1 * r1 * (pBest[i] - particles[i]) + c2 * r2 * (gBest - particles[i])
            
            # Apply velocity bounds
            v = np.clip(v, -v_max, v_max)
            velocities[i] = v
            
            # Update position
            new_h = particles[i] + v
            new_h = project_to_feasible(new_h, n_points)
            
            particles[i] = new_h
            
        # Update inertia weight
        omega = max(0.4, omega * 0.99)
    
    return (list(gBest), gBest_val, n_points)
