import numpy as np
import time
from scipy import signal

def evaluate(h, dx):
    # compute correlation of h with (1-h) for all lags
    corr = signal.correlate(h, 1 - h, mode='full')
    # max over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, target_sum):
    # normalize sum to target, keep in [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.ones_like(h) * (target_sum / len(h))
    else:
        h = h * target_sum / current_sum
        return h

def create_initial_population(initial_h, n_points, pop_size):
    # Start with the initial solution
    pop = [initial_h.copy()]
    # Add random feasible solutions
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project(h, n_points / 2)
        pop.append(h)
    return pop

def crossover(parent1, parent2):
    # Uniform crossover
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    # Project to maintain feasibility
    child1 = project(child1, len(child1) / 2)
    child2 = project(child2, len(child2) / 2)
    return child1, child2

def mutate(h, mutation_rate=0.05, sigma=0.1):
    # Mutate selected elements
    mask = np.random.rand(len(h)) < mutation_rate
    if np.any(mask):
        h = h.copy()
        h[mask] += np.random.normal(0, sigma, np.sum(mask))
        h = project(h, len(h) / 2)
    return h

def tournament_selection(pop, fit_vals, tournament_size=2):
    idxs = np.random.choice(len(pop), tournament_size)
    best_idx = idxs[np.argmin([fit_vals[i] for i in idxs])]
    return pop[best_idx]

    np.random.seed(seed)
    n_points = 47  # Use the provided initial solution size
    dx = 2.0 / n_points

        initial_h = np.random.rand(n_points)
        initial_h = project(initial_h, n_points / 2)

    pop_size = 20
    max_generations = 1000
    generation = 0
    start_time = time.time()
    
    # Initialize population
    pop = create_initial_population(initial_h, n_points, pop_size)
    
    best_h = None
    best_C5 = float('inf')
    
    no_improvement_count = 0
    best_stale_generations = 50

    while generation < max_generations and (time.time() - start_time) < budget_s:
        # Evaluate fitness
        fit_vals = [evaluate(h, dx) for h in pop]
        
        # Track best solution
        min_fit = np.min(fit_vals)
        min_idx = np.argmin(fit_vals)
        if min_fit < best_C5:
            best_C5 = min_fit
            best_h = pop[min_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Early termination if no improvement for too many generations
        if no_improvement_count > best_stale_generations:
            break

        # Selection, crossover, mutation
        new_pop = []
        # Keep top 2 individuals
        top_indices = np.argsort(fit_vals)[:2]
        new_pop.extend([pop[i] for i in top_indices])

        # Generate rest of the population
        while len(new_pop) < pop_size:
            parent1 = tournament_selection(pop, fit_vals)
            parent2 = tournament_selection(pop, fit_vals)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_pop.extend([child1, child2])
        new_pop = new_pop[:pop_size]
        pop = new_pop

        generation += 1

    # Final evaluation of best solution
    if best_h is not None:
        final_C5 = evaluate(best_h, dx)
        return (best_h, final_C5, n_points)
    else:
        # If no good solution, return initial one
        initial_C5 = evaluate(initial_h, dx)
        return (initial_h, initial_C5, n_points)
