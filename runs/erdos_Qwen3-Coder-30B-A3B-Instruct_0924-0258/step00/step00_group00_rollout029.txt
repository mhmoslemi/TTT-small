import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.random.rand(n_points)
        h0 = h0 / np.sum(h0) * (n_points / 2)
    
    def evaluate(h):
        if time.time() - start_time > budget_s - 0.1:
            return float('inf')
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # Take the maximum over all lags
        max_corr = np.max(corr) * dx
        return max_corr
    
    # PSO parameters
    n_particles = 30
    max_iters = 2000
    inertia = 0.7
    cogn_rate = 2.0
    soc_rate = 2.0
    max_vel = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    for i in range(n_particles):
        if i == 0:
            h = h0.copy()
        else:
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        score = evaluate(h)
        pbest.append(h.copy())
        pbest_scores.append(score)
    
    gbest_idx = np.argmin(pbest_scores)
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO iterations
    for iteration in range(max_iters):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                inertia * velocities[i] +
                cogn_rate * r1 * (pbest[i] - particles[i]) +
                soc_rate * r2 * (pbest[gbest_idx] - particles[i])
            )
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            
            # Evaluate
            score = evaluate(particles[i])
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                
                # Update global best
                if score < gbest_score:
                    gbest_score = score
                    gbest_idx = i
                    # print(f"New best C5 = {gbest_score:.6f} at iter {iteration}")
    
    # Find final best
    final_h = pbest[gbest_idx].copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h / np.sum(final_h) * (n_points / 2)
    
    # Double check the final result
    c5_bound = evaluate(final_h)
    
    return (final_h.tolist(), c5_bound, n_points)
