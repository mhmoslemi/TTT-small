def evaluate_c5(h):
    """Evaluate the C5 bound for given h"""
    # Discretized overlap
    n = len(h)
    dx = 2.0 / n
    # Compute the correlation manually
    correlation = np.correlate(h, 1 - h, mode="full")
    # Extract the relevant part for maximum overlap
    max_over = np.max(correlation)
    # Scale by dx for continuous interpretation
    return max_over * dx
