import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all shifts
    overlap = correlate(h, 1 - h, mode="full")
    # Scale by dx and find maximum
    max_overlap = np.max(overlap) * dx
    return max_overlap

def initial_population(initial_h_values, n_points, pop_size=50):
    population = []
    # Add the initial solution
    population.append(initial_h_values.copy())
    # Generate random perturbations
    for i in range(pop_size - 1):
        perturbed = initial_h_values + 0.1 * np.random.randn(n_points)
        perturbed = np.clip(perturbed, 0, 1)
        # Rescale to meet constraint
        scale = (n_points / 2) / np.sum(perturbed)
        perturbed *= scale
        population.append(perturbed)
    return population

def crossover(parent1, parent2):
    n_points = len(parent1)
    point = np.random.randint(1, n_points - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(child, mutation_rate=0.1):
    n_points = len(child)
    num_mutation = int(n_points * mutation_rate)
    indices = np.random.choice(n_points, num_mutation, replace=False)
    for i in indices:
        child[i] += 0.1 * np.random.randn()
    # Clip to [0,1]
    child = np.clip(child, 0, 1)
    # Rescale to meet constraint
    scale = (n_points / 2) / np.sum(child)
    child *= scale
    return child

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial solution if available
    try:
        initial_h_values
    except NameError:
        # Fallback: create a simple step function
        n_points = 45
        initial_h_values = np.ones(n_points) * 0.5  # Uniform distribution
        # Then we'll adjust the sum
        initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
        # A simple feasible construct
    else:
        n_points = len(initial_h_values)
    
    # Use initial_h_values if it exists and is in the right format
    if not 'initial_h_values' in locals():
        n_points = 45
        initial_h_values = np.ones(n_points) * 0.5
        initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    
    pop_size = 50
    generations = 12
    
    # Initialize population
    population = initial_population(initial_h_values, n_points, pop_size)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        c5 = evaluate_erdos_solution(h)
        fitness.append(c5)
    
    # Track best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 25 as parents
        parents = population[:pop_size // 2]
        parents_fitness = fitness[:pop_size // 2]
        
        # Create offspring through crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                # Crossover
                child1, child2 = crossover(parents[i], parents[i+1])
                # Mutate
                child1 = mutate(child1)
                child2 = mutate(child2)
                offspring.extend([child1, child2])
        
        # Evaluate offspring
        offspring_fitness = []
        for child in offspring:
            c5 = evaluate_erdos_solution(child)
            offspring_fitness.append(c5)
        
        # Combine population and parents
        all_h = parents + offspring
        all_fitness = parents_fitness + offspring_fitness
        # Sort and keep top pop_size individuals
        sorted_indices = np.argsort(all_fitness)
        population = [all_h[i] for i in sorted_indices[:pop_size]]
        fitness = [all_fitness[i] for i in sorted_indices[:pop_size]]
        
        # Update global best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
    
    # Final evaluation
    optimized_c5 = evaluate_erdos_solution(best_h)
    
    # Return final values
    return (best_h, optimized_c5, len(best_h))
