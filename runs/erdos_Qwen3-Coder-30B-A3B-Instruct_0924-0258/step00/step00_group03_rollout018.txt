import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h):
    """Evaluate the C5 constant for given h"""
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # The maximum index is n_points - 1, we only consider relevant lags
    n = len(h)
    # Get max over all valid lags (1 to n-1) - for zero lag, we get 0
    max_val = -1
    for k in range(1, n):
        val = corr[n-1+k]  # offset due to full correlation
        if val > max_val:
            max_val = val
    return max_val * 2.0 / n

def project_to_feasible(h):
    """Project h to feasible region: sum to n/2, all values in [0,1]"""
    n = len(h)
    h = np.clip(h, 0, 1)
    scale = (n / 2) / np.sum(h)
    h = h * scale
    return h
