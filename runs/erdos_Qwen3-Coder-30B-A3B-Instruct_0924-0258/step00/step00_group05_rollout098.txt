def evaluate_erdos_solution(h, n_points):
    """Evaluate the C5 value for a given step function h"""
    dx = 2.0 / n_points
    h_extended = np.concatenate([h, h])
    correlations = []
    for k in range(n_points):
        corr = np.sum(h * (1 - h_extended[k:k+n_points]))
        correlations.append(corr)
    return np.max(correlations) * dx
