import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup discretization
    n_points = 42
    dx = 2.0 / n_points
    lag_offset = n_points - 1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert np.all(h >= 0) and np.all(h <= 1), "Initial h not in [0,1]"
        assert abs(np.sum(h) * dx - 1.0) < 1e-6, "Initial h does not integrate to 1"
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx, lag_offset)
    
    # Projective gradient descent with line search
    max_iter = 10000
    alpha0 = 0.1
    beta = 0.8
    min_alpha = 1e-6
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps and find worst lag
        z = correlate(h, 1 - h, mode='full') * dx
        z_k = z[lag_offset:lag_offset + n_points]
        k_star = np.argmax(z_k)
        C5_current = z_k[k_star]
        
        if C5_current < best_C5:
            best_C5 = C5_current
            best_h = h.copy()
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_star) % n_points
            grad[i] = 1 - 2 * h[j]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            continue
        grad /= grad_norm
        
        # Line search
        alpha = alpha0
        while alpha >= min_alpha:
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            
            C5_new = evaluate_C5(h_new, dx, lag_offset)
            if C5_new < C5_current:
                h = h_new
                break
            alpha *= beta
            
    return (best_h.tolist(), best_C5, n_points)

def evaluate_C5(h, dx, lag_offset):
    z = correlate(h, 1 - h, mode='full') * dx
    return np.max(z[lag_offset:lag_offset + len(h)])
