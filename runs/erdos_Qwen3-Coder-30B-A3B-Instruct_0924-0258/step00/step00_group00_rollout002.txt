import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate C5 for given h"""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum value over all lags
    max_corr = np.max(corr)
    # Convert to actual C5 value using dx
    return max_corr * dx

def rescale_h(h, target_sum):
    """Rescale h to target sum while maintaining constraints"""
    current_sum = np.sum(h)
    if current_sum == 0:
        return h
    scaled = h * (target_sum / current_sum)
    # Clip to [0,1]
    scaled = np.clip(scaled, 0, 1)
    # Rescale again to maintain target sum
    final_sum = np.sum(scaled)
    if final_sum > 0:
        scaled = scaled * (target_sum / final_sum)
    return scaled

def initialize_h(n_points):
    """Initialize h with target sum"""
    # Start with truncated normal or some other distribution
    h = np.random.rand(n_points)
    # Rescale to sum equal to target_sum = n_points / 2
    h = rescale_h(h, n_points / 2)
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # If we have initial_h_values, use it; otherwise create decent starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Default size for reasonable performance and optimization scope
        n_points = 64
        # Initialize a simple staircase or binary-like pattern
        h = initialize_h(n_points)
    
    # dx for step size
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initializations for PSO
    num_particles = 30
    max_iter = 5000
    inertia = 0.7
    phi_p = 2.0
    phi_g = 2.0
    
    # Store history
    particles = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    global_best = None
    global_best_val = float('inf')
    
    # Initialize particles with feasible values
    for _ in range(num_particles):
        # Start with h
        h_current = h.copy()
        # Add small random noise
        h_current += np.random.normal(0, 0.05, n_points)
        # Clip to [0,1]
        h_current = np.clip(h_current, 0, 1)
        # Rescale
        h_current = rescale_h(h_current, target_sum)
        particles.append(h_current)
        velocities.append(np.zeros(n_points))
        val = evaluate_c5(h_current, dx)
        personal_best.append(h_current.copy())
        personal_best_vals.append(val)
        if val < global_best_val:
            global_best = h_current.copy()
            global_best_val = val
    
    last_improvement = 0
    iter_count = 0
    
    # Main PSO loop
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
            
        for i in range(num_particles):
            h_current = particles[i]
            
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            cognitive = phi_p * r1 * (personal_best[i] - h_current)
            social = phi_g * r2 * (global_best - h_current)
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Update position
            particles[i] = h_current + velocities[i]
            
            # Enforce boundary constraints and sum constraint
            particles[i] = np.clip(particles[i], 0, 1)
            # Rescale to sum = target_sum
            particles[i] = rescale_h(particles[i], target_sum)
            
            # Evaluate
            current_val = evaluate_c5(particles[i], dx)
            
            # Update personal best if better
            if current_val < personal_best_vals[i]:
                personal_best[i] = particles[i].copy()
                personal_best_vals[i] = current_val
            
            # Update global best if better
            if current_val < global_best_val:
                global_best = particles[i].copy()
                global_best_val = current_val
                last_improvement = iter_count
        
        iter_count += 1
        
        # Early stopping if no improvement for some time
        if iter_count - last_improvement > 1000:
            break
    
    # Final projection of global best to ensure constraints
    final_h = np.clip(global_best, 0, 1)
    final_h = rescale_h(final_h, target_sum)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, len(final_h))
