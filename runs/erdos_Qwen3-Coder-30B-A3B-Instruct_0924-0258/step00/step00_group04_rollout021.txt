import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h_values, 1 - h_values, mode='full')
    # Extract the overlapping parts (k from 0 to n-1)
    overlaps = corr[n-1:n-1+n] * dx
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial if available, otherwise create a uniform start
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute all overlaps for all lags
        corr = correlate(h, 1 - h, mode='full')
        overlaps = corr[n_points-1:n_points-1+n_points] * dx
        
        # Find worst lag
        k_star = np.argmax(overlaps)
        
        # Compute gradient: dC5/dh[j] = (1 - 2*h[(j + k_star) % n])
        gradient = np.zeros(n_points)
        for j in range(n_points):
            idx = (j + k_star) % n_points
            gradient[j] = 1 - 2 * h[idx]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm < 1e-12:
            break
        gradient /= grad_norm
        
        # Line search
        alpha = 0.1
        step_size_min = 1e-6
        
        while alpha >= step_size_min:
            h_new_unproj = h - alpha * gradient
            h_new = np.clip(h_new_unproj, 0, 1)
            
            # Rescale to maintain integral constraint
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            
            new_c5 = evaluate_erdos_solution(h_new)
            if new_c5 < best_c5:
                h = h_new
                best_c5 = new_c5
                best_h = h.copy()
                break
            alpha *= 0.8
        
        if alpha < step_size_min:
            # No progress, try another strategy
            h = np.clip(h + 0.01 * (np.random.rand(n_points) - 0.5), 0, 1)
            h = h / np.sum(h) * (n_points / 2)
    
    return (best_h.tolist(), best_c5, n_points)
