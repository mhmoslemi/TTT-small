import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode="full")
    # Maximize over all possible shifts
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points
    max_generations = 12
    population_size = 50
    elite_size = 25
    mutation_rate = 0.1

    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        # Create a simple initial guess
        initial_h = np.ones(n_points) * 0.5
        initial_h[0] = 0.0
        initial_h[-1] = 1.0
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2)

    population.append(initial_h)
    for i in range(population_size - 1):
        # Add perturbed versions
        perturbed = initial_h + np.random.randn(n_points) * 0.1
        perturbed = np.clip(perturbed, 0, 1)
        # Rescale to meet constraint
        scale_factor = (n_points / 2) / np.sum(perturbed)
        perturbed = perturbed * scale_factor
        population.append(perturbed)

    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)

    # Evolution loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s * 0.99:
            break
        
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0]
            best_c5 = fitness[0]
        
        # Select elite
        elite = population[:elite_size]
        
        # Create offspring
        offspring = []
        while len(offspring) < population_size - elite_size:
            # Crossover
            parent1_idx, parent2_idx = np.random.choice(elite_size, 2, replace=False)
            parent1 = elite[parent1_idx]
            parent2 = elite[parent2_idx]
            
            # Single point crossover
            crossover_point = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            for child in [child1, child2]:
                if len(offspring) >= population_size - elite_size:
                    break
                # Mutate some genes
                mut_indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
                for i in mut_indices:
                    child[i] += np.random.randn() * 0.1
                # Clip
                child = np.clip(child, 0, 1)
                # Rescale
                scale_factor = (n_points / 2) / np.sum(child)
                child = child * scale_factor
                offspring.append(child)
        
        # Create new population
        population = elite + offspring

    return (best_h.tolist(), best_c5, n_points)
