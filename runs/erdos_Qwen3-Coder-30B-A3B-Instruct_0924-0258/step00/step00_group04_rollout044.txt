import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        h = np.array(initial_h_values)

        # Initialize uniform distribution
        n_points = 42
        h = np.full(n_points, 0.5)
    
    # Ensure sum equals n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Validate constraints
    assert np.all(h >= 0) and np.all(h <= 1), "Initial h not in [0,1]"
    assert abs(np.sum(h) - n_points / 2) < 1e-10, "Initial h sum constraint violated"
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Parameters
    alpha = 0.1
    max_iter = 10000
    perturb_freq = 50
    conv_thresh = 1e-5
    last_improvement = 0
    iter_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute all correlations
        corr_full = correlate(h, 1 - h, mode='full')
        c5_vals = corr_full * (2.0 / n_points)
        k_max = np.argmax(c5_vals)
        max_c5 = c5_vals[k_max]
        
        # Compute gradient for the worst case lag
        g = np.zeros(n_points)
        for j in range(n_points):
            # Two terms: h[j] * (1 - h[(j-k_max) % n_points]) and h[j] * (-h[(j+k_max) % n_points])  
            # Gradient wrt h[j]
            term1 = 1 - h[(j - k_max) % n_points]
            term2 = -h[(j + k_max) % n_points]  
            g[j] = (2.0 / n_points) * (term1 + term2)
            
        # Line search to find good step size
        old_h = h.copy()
        step_size = alpha
        for _ in range(10):
            h_new = h - step_size * g
            # Project to feasible region
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_c5 = evaluate_erdos_solution(h_new)
            if new_c5 < max_c5 - 1e-10:  # Improvement
                h = h_new
                break
            step_size *= 0.5
            
        # Check if we improved
        if max_c5 < best_c5 - 1e-10:
            best_h = h.copy()
            best_c5 = max_c5
            last_improvement = i
            
        # Add noise occasionally to escape local minima
        if i % perturb_freq == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        iter_count = i + 1
        
        # Early stopping if no improvement in a while
        if i - last_improvement > 200:
            break
            
    # Validate final solution
    final_c5 = evaluate_erdos_solution(best_h)
    return (list(best_h), final_c5, n_points)

def evaluate_erdos_solution(h):
    """Use the provided evaluation function"""
    h = np.array(h)
    n_points = len(h)
    result = np.max(np.correlate(h, 1-h, mode="full")) * (2.0 / n_points)
    return result
