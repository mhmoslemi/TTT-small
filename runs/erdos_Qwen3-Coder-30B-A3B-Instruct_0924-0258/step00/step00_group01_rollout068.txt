import numpy as np
import time
from scipy.fft import fft, ifft

def evaluate_c5(h):
    """Evaluate the maximum overlap integral using correlation."""
    n = len(h)
    dx = 2.0 / n
    # Compute np.correlate(h, 1-h, mode="full") * dx
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def fwht_correlate(h, h_conj):
    """Fast Walsh-Hadamard transform for cross-correlation."""
    n = len(h)
    # Forward transform of both
    F_h = fft(h)
    F_h_conj = fft(h_conj)
    # Pointwise multiplication in frequency domain
    F_correlation = F_h * np.conj(F_h_conj)
    # Inverse transform
    correlation = ifft(F_correlation).real
    return correlation

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47  # default as per task
    
    # Initialize h using initial_h_values if available, else uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if abs(np.sum(h) - n_points / 2.0) > 1e-6:
            h = project_h(h, n_points)
    else:
        h = np.full(n_points, 0.5)
        h = project_h(h, n_points)

    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    alpha = 0.1
    
    last_improvement_iter = 0
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Direct correlation evaluation (more reliable)
        corr = np.correlate(h, 1-h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        if max_overlap * (2.0 / n_points) < best_c5:
            best_c5 = max_overlap * (2.0 / n_points)
            best_h = h.copy()
            last_improvement_iter = iter_count
        
        # Compute gradient efficiently with circular shifts
        # For using a gradient-like update method
        # direction of descent is in h direction based on shift patterns
        
        # Compute v = h conved with (1-h) in a fixed pattern
        # Simple heuristic: for the max overlap lag, update h using differences
        
        # This uses full correlation to identify the worst overlap
        max_k = k_max - (n_points - 1)   # shift corresponding to max overlap
        
        # Simple gradient: update based on differences of h values at a few key lags
        # Direct heuristic: increase h where it contributes to overlap
        
        new_h = h.copy()
        
        # For the current max_k that causes biggest overlap,
        # we adjust h to decrease overlap: check the gradient
        
        # Fix one pattern: for lags, increase h in region (where diff = 1-h)
        # and increase 1-h to decrease impact at k_max
        
        # Let's do it for the most dominant k
        for i in range(n_points):
            # Simulate by looking at contribution to overlap at k_max
            new_h[i] += alpha * (1.0 - h[(i - max_k) % n_points])
        
        # Keep feasible and preserve normalization
        new_h = project_h(new_h, n_points)
        h = new_h

        # Adaptively lower the learning rate
        if iter_count - last_improvement_iter > 50:
            alpha *= 0.95
            last_improvement_iter = iter_count

        # Occasionally perturb to escape local minima
        if iter_count % 200 == 0 and np.random.rand() < 0.05:
            # Add small random noise to h
            noise = np.random.normal(0, 0.001, n_points)
            h += noise
            h = project_h(h, n_points)
            
    # Final check of best
    final_c5 = evaluate_c5(best_h)
    
    return (list(best_h), final_c5, n_points)
