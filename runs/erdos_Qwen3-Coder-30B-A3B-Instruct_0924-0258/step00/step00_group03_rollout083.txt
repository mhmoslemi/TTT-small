import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import differential_evolution

def evaluate_overlap(h, dx):
    # Compute the overlap integral for the given step function h
    h = np.array(h)
    # Compute correlation of h with (1-h)
    correlation = correlate(h, 1 - h, mode="full")
    # The maximum overlap is over all shifts
    max_overlap = np.max(correlation) * dx
    return max_overlap

def validate_solution(h, dx, n_points):
    # Ensure all values are in [0,1] and sum equals n_points/2
    h = np.array(h)
    if np.any(h < 0) or np.any(h > 1):
        return False
    if abs(np.sum(h) * dx - 1.0) > 1e-10:
        return False
    return True

def normalize_h(h, dx, target_sum=1.0):
    h = np.array(h)
    current_sum = np.sum(h) * dx
    if current_sum == 0:
        return h
    scale = target_sum / current_sum
    h = h * scale
    return np.clip(h, 0, 1)

def mutate_h(h, dx, mutation_rate=0.1):
    # Modify values with noise, then normalize
    h = np.array(h) + np.random.normal(0, mutation_rate, len(h))
    h = normalize_h(h, dx)
    return h

def crossover_h(h1, h2, dx, crossover_rate=0.8):
    # Simple uniform crossover
    if np.random.rand() > crossover_rate:
        return h1.copy()
    mask = np.random.rand(len(h1)) > 0.5
    h = np.where(mask, h1, h2)
    return normalize_h(h, dx)

def tournament_selection(population, fitnesses, tournament_size=5):
    selected = []
    for _ in range(len(population)):
        idxs = np.random.choice(len(population), tournament_size)
        winner = idxs[np.argmin([fitnesses[i] for i in idxs])]
        selected.append(population[winner])
    return selected

def evolve_population(population, fitnesses, dx, generation, population_size=50,
                      mutation_rate=0.1, crossover_rate=0.8):
    # Sort by fitness (lower is better)
    sorted_indices = np.argsort(fitnesses)
    population = [population[i] for i in sorted_indices]
    fitnesses = [fitnesses[i] for i in sorted_indices]
    
    # Select top half for reproduction
    selected = tournament_selection(population[:population_size//2], fitnesses[:population_size//2])
    
    # Generate offspring
    offspring = []
    for i in range(0, len(selected), 2):
        parent1 = selected[i]
        parent2 = selected[i+1] if i+1 < len(selected) else selected[0]
        child1 = crossover_h(parent1, parent2, dx, crossover_rate)
        child2 = crossover_h(parent2, parent1, dx, crossover_rate)
        offspring.append(mutate_h(child1, dx, mutation_rate))
        offspring.append(mutate_h(child2, dx, mutation_rate))
    
    # Keep best individuals
    new_population = population[:population_size//2] + offspring[:population_size//2]
    return new_population

        h0 = normalize_h(h0, 2.0/n_points, 1.0)
        
    n_points = len(h0)
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_overlap(best_h, dx)
    start_time = time.time()
    
    # Configuration
    population_size = 50
    generations = 2000  # Will check elapsed time regularly
    
    # Initial population
    population = [h0.copy()]
    for _ in range(population_size - 1):
        noise = np.random.normal(0, 0.05, n_points)
        h = h0 + noise
        h = normalize_h(h, dx)
        population.append(h)
    
    # Evolve
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Compute fitness for all individuals
        fitnesses = [evaluate_overlap(h, dx) for h in population]
        
        # Track best
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[np.argmin(fitnesses)].copy()
        
        # Evolve
        population = evolve_population(population, fitnesses, dx, gen, population_size,
                                     mutation_rate=0.1, crossover_rate=0.8)

        if gen % 50 == 0 and gen > 0:
