import numpy as np
from scipy import signal
import cvxpy as cp
import time

def evaluate_h(h, n_points):
    """Evaluate the C5 bound for given h."""
    dx = 2.0 / n_points
    # Compute the full convolution: h * (1 - h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # The correlation values are scaled by dx
    conv *= dx
    # Return maximum value as the C5 bound
    return np.max(conv)

def project_h(h, n_points):
    """Project h to feasible set: 0 <= h[i] <= 1 and sum(h) = n_points/2."""
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint using CVXPY
    h_cvx = cp.Variable(n_points)
    constraints = [h_cvx >= 0, h_cvx <= 1, cp.sum(h_cvx) == n_points / 2]
    prob = cp.Problem(cp.Minimize(0), constraints)
    prob.solve()
    return h_cvx.value

        grad = np.zeros(n_points)

        current_h *= (n_points / 2) / np.sum(current_h)

        if i % 100 == 0 and i > 0:
