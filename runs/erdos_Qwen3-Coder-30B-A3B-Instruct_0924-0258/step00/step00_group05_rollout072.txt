import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    # Compute all overlaps
    overlaps = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(overlaps) * dx
    return max_overlap

def project_h(h, n_points):
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to meet exact constraint sum(h) = n_points / 2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum != target_sum:
        # Simple projection: adjust values to meet sum
        diff = target_sum - current_sum
        if n_points > 0:
            h += diff / n_points
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    n_points = 70  # Use the same number as initial construction
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        h = np.full(n_points, 0.5)
        h = project_h(h, n_points)
    
    start_time = time.time()
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, n_points)
    
    # Parameters
    alpha = 0.1
    max_iter = 500
    conv_threshold = 1e-4
    last_improvement = 0
    patience = 0
    
    # Gradient descent
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Compute current overlaps for all lags
        h_rev = np.flip(h)
        overlaps = correlate(h, h_rev, mode='full')
        # The overlap integral is defined as max_k ∫ h(x)(1-h(x+k)) dx
        # We compute correlation of h(x) and (1 - h(x+k))
        # Need to compute both h(x) * (1 - h(x+k)) and (1 - h(x)) * h(x+k)  
        # The first term corresponds to the same integral formula
        # We compute h * (1 - h(x+k))
        overlaps = correlate(h, 1 - h, mode='full')
        # Max overlap, scaled with dx
        current_max = np.max(overlaps) * dx
        
        # Extract max lags
        max_indices = np.where(overlaps == np.max(overlaps))[0]
        if len(max_indices) > 0:
            max_k_indices = max_indices
        else:
            max_k_indices = [0]
        
        # Gradient computation
        # For each k in max_k_indices, compute gradient contribution
        g = np.zeros(n_points)
        for k in max_k_indices:
            # h[i] * (1 - h[i+k]) for correlation indexing
            # With indices, h[i] * (1 - h[(i+k) % n_points])
            for idx in range(n_points):
                k_idx = (idx + k) % n_points
                # J = h[i] * (1 - h[i+k]) 
                # dJ/dh[i] = 1 - h[i+k]
                # dJ/dh[i+k] = - h[i]
                # So gradient contribution at index i is (1 - h[i+k]) 
                # at position k_idx, but it is (1 - h[i+k]) * delta_i == (1-h[i+k]) 
                g[idx] -= (1.0 - h[k_idx]) * dx
        
        # Update step
        h_new = h - alpha * g
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        new_C5 = evaluate_erdos_solution(h_new, n_points)
        
        # Accept if better
        if new_C5 < best_C5:
            h = h_new
            best_h = h.copy()
            best_C5 = new_C5
            last_improvement = i
            patience = 0
            alpha *= 1.1  # Increase step size slightly
        else:
            patience += 1
            # If no improvement, reduce step size and try perturbation
            alpha *= 0.95  # Reduce step size
            if patience > 5:
                # Add stochastic perturbation to escape local minima
                noise = np.random.normal(0, 0.01, n_points)
                h = h + noise
                h = project_h(h, n_points)
                patience = 0
                alpha *= 0.9
    
    return (best_h.tolist(), best_C5, n_points)
