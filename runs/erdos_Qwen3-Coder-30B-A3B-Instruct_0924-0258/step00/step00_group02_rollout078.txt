import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate the overlap integral for given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum correlation scaled by dx
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h onto feasible region: [0,1] and sum = n_points/2."""
    # Clamp values to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

        grad = np.zeros_like(h)

            learning_rate = min_lr
        
        iter_count += 1

    return (best_h.tolist(), best_c5, n_points)
