import numpy as np
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for the given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution h * (1 - h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is at most the maximum of the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def surrogate_objective(h, n_points, dx):
    """Compute the surrogate objective using squared convolution values."""
    # Compute convolution h * (1 - h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Use a convex surrogate: sum of squares of convolution elements
    surrogate = np.sum(conv * conv) * dx * dx
    return surrogate

def surrogate_gradient(h, n_points, dx):
    """Compute gradient of surrogate objective (simple approximation)."""
    # This is a rough approximation using finite differences
    eps = 1e-6
    grad = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        obj_plus = surrogate_objective(h_plus, n_points, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_feasible(h_minus, n_points)
        obj_minus = surrogate_objective(h_minus, n_points, dx)
        
        grad[i] = (obj_plus - obj_minus) / (2 * eps)
    return grad

            # For performance, we implement a simple gradient descent
            # We compute step with simplified gradient
            grad = surrogate_gradient(best_h, n_points, dx)
            step_size = 0.001
            
            # Update with gradient step
            new_h = best_h - step_size * grad
            
            # Project to feasible set
            new_h = project_to_feasible(new_h, n_points)
            
            # Use a simple backtracking line search
            trial_obj = surrogate_objective(new_h, n_points, dx)
            current_obj = surrogate_objective(best_h, n_points, dx)
            
            # If no improvement, give up and move on
            if trial_obj >= current_obj:
                iter_count += 1
                if iter_count > 10:
                    break
                continue
            
            # Accept step if it improves surrogate
            best_h = new_h
            
            # Evaluate true objective
            current_c5 = evaluate_erdos_solution(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                
        except Exception:
            break
            
        iter_count += 1
        if iter_count > 100:
            break
            
    return best_h, best_c5
