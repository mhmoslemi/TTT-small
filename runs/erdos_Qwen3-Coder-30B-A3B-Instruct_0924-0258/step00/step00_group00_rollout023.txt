import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode='full')
    # Only consider the valid range
    mid = len(corr) // 2
    valid_corr = corr[mid - len(h) + 1:mid + len(h) - 1]
    return np.max(valid_corr * dx)

def rescale(h, target_sum):
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.full_like(h, target_sum / len(h))
    return h * (target_sum / current_sum)

def mutate(h, mutation_rate=0.1, delta=0.1):
    h = h.copy()
    mask = np.random.random(len(h)) < mutation_rate
    h[mask] += np.random.uniform(-delta, delta, np.sum(mask))
    h = np.clip(h, 0, 1)
    return rescale(h, len(h) / 2)

def crossover(parent1, parent2, crossover_rate=0.7):
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    split_point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:split_point], parent2[split_point:]])
    child2 = np.concatenate([parent2[:split_point], parent1[split_point:]])
    return rescale(child1, len(child1) / 2), rescale(child2, len(child2) / 2)

def tournament_selection(population, fitnesses, k=3):
    idxs = np.random.choice(len(population), k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize paramters
    n_points = 45  # Use the initial construction size if available
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Load or create initial solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = rescale(h0, target_sum)
    else:
        h0 = np.full(n_points, 0.5)  # Fallback to uniform
    
    # GA parameters
    population_size = 50
    generations = 200
    mutation_rate = 0.1
    crossover_rate = 0.7
    tournament_k = 3
    
    # Initial population
    population = [h0.copy()]
    for _ in range(population_size - 1):
        # Start with random perturbed version
        h = h0.copy()
        mask = np.random.random(n_points) < 0.2
        h[mask] += np.random.uniform(-0.2, 0.2, np.sum(mask))
        h = np.clip(h, 0, 1)
        h = rescale(h, target_sum)
        population.append(h)
    
    # GA Loop
    best_h = h0.copy()
    best_c5 = evaluate(best_h, dx)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate(h, dx) for h in population]
        
        # Update best
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
        
        # Selection
        new_pop = []
        for _ in range(population_size):
            parent1 = tournament_selection(population, fitnesses, tournament_k)
            parent2 = tournament_selection(population, fitnesses, tournament_k)
            
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            # Mutate
            if np.random.random() < 0.2:
                child1 = mutate(child1, mutation_rate, 0.1)
            if np.random.random() < 0.2:
                child2 = mutate(child2, mutation_rate, 0.1)
            
            new_pop.extend([child1, child2])
            
        # Keep only population_size individuals
        population = new_pop[:population_size]
        
        # Occasionally add a random individual to keep diversity
        if gen % 20 == 0:
            for i in range(2):
                h = np.random.random(n_points)
                h = rescale(h, target_sum)
                population[np.random.randint(0, population_size)] = h
    
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
