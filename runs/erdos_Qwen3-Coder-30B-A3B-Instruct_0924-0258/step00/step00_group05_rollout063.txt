import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate C5 for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation with 1-h
    corr = correlate(h, 1 - h, mode="full")
    # Max over all lags
    maxcorr = np.max(corr)
    return maxcorr * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum(h) = target_sum."""
    h = np.array(h)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Linear scaling
    scale = target_sum / current_sum
    h = h * scale
    return np.clip(h, 0, 1)

def perturb_step(h, temperature):
    """Perturb h with Gaussian noise, then project."""
    n = len(h)
    # Add Gaussian noise
    noise = np.random.normal(0, temperature, n)
    h_new = h + noise
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Project to correct sum
    h_new = project_to_constraint(h_new, n/2)
    return h_new

def simulated_annealing_step(h, temperature, best_h, best_c5):
    """Do one SA iteration."""
    h_new = perturb_step(h, temperature)
    c5_new = evaluate_c5(h_new)
    delta_c5 = c5_new - best_c5
    if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temperature):
        return h_new, c5_new
    return h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_low = 10
    n_high = 70
    n_levels = int(np.log2(n_high // n_low)) + 1  # levels including base
    
    # Initial construction
    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() and len(initial_h_values) > 0 else None
    if initial_h is None:
        # Create random starting point
        h = np.random.rand(n_low)
        h = project_to_constraint(h, n_low/2)
    else:
        h = np.array(initial_h)
        h = project_to_constraint(h, n_low/2)
        
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Cooling schedule & parameters
    T_start = 0.1
    T_min = 1e-6
    alpha = 0.99
    iterations_per_level = 500
    
    # Multi-resolution optimization
    n_points = n_low
    for level in range(n_levels):
        T = T_start
        # Resize h (interpolate or copy if needed)
        if level > 0:
            # interpolate from previous
            n_points = min(n_points * 2, n_high)
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
            h = project_to_constraint(h, n_points/2)
            # Update best value if needed
            c5_current = evaluate_c5(h)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()
        
        # Optimize on this level
        for i in range(iterations_per_level):
            if time.time() - start_time > budget_s:
                break
            h, c5_new = simulated_annealing_step(h, T, best_h, best_c5)
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
            # Cool down
            T = max(T_min, T * alpha)
    
    # Final check
    final_c5 = evaluate_c5(best_h)
    
    return (list(best_h), final_c5, len(best_h))
