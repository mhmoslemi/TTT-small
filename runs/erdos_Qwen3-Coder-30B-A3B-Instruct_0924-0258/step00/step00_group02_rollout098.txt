import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx=2.0 / len(h)):
    """Compute C5 using the full convolution."""
    conv = np.correlate(h, 1 - h, mode="full")
    return np.max(conv) * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) == n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def build_surrogate(h, n_points):
    """Build convex surrogate of the maximum overlap."""
    conv = np.correlate(h, 1 - h, mode="full")
    return np.sum(conv**2) * (2.0 / n_points)**2

def optimize_at_resolution(h0, n_points, max_time):
    """Optimize at a given resolution level."""
    dx = 2.0 / n_points
    start_time = time()
    
    def objective(h):
        if time() - start_time > max_time:
            return np.inf
        return build_surrogate(h, n_points)
    
    def gradient(h):
        # Approximate gradient for surrogate
        conv = np.correlate(h, 1 - h, mode="full")
        # Crude finite-difference gradient for surrogate
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad[i] = (build_surrogate(h_plus, n_points) - build_surrogate(h_minus, n_points)) / (2 * eps)
        return grad
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Setup constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Perturb initial
    h_init = h0 + np.random.normal(0, 0.01, n_points)
    h_init = project_to_feasible(h_init, n_points)
    
    # Optimize
    try:
        result = minimize(objective, h_init, method='SLSQP', bounds=bounds, 
                          constraints=constraints, jac=gradient, 
                          options={'maxiter': 500, 'ftol': 1e-6})
        if result.success:
            h_opt = result.x
        else:
            h_opt = h_init
    except:
        h_opt = h_init
    
    h_opt = project_to_feasible(h_opt, n_points)
    c5 = evaluate_erdos_solution(h_opt, dx)
    
    return h_opt, c5

        h_base = np.ones(n_points_base) * 0.5
        h_base[10:20] = 1.0  # Step-like
        h_base = project_to_feasible(h_base, n_points_base)
    
    best_h = h_base.copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points_base)
    
    # Resolution levels
    res_levels = [15, 30, 45, 60, 75]
    
    for n_points in res_levels:
        if time() - start_time > budget_s:
            break
            
        # Interpolate from previous level
        if n_points > n_points_base:
            x_orig = np.linspace(0, 2, n_points_base)
            x_new = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_new, x_orig, best_h)
            h_interp = project_to_feasible(h_interp, n_points)
        else:
            # Downsample
            h_interp = best_h[::n_points_base // n_points]
            h_interp = project_to_feasible(h_interp, n_points)
        
        # Perturb multiple times to avoid local optima
        for _ in range(3):
            if time() - start_time > budget_s:
                break
            h_opt, c5 = optimize_at_resolution(h_interp, n_points, budget_s - (time() - start_time))
            if c5 < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5

    dx = 2.0 / len(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (best_h.tolist(), final_c5, len(best_h))
