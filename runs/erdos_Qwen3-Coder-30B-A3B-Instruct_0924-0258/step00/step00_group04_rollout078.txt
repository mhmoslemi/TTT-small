import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_h(h, dx):
    # Computes C5 = max_k integral of h(x)(1-h(x+k))dx
    # Using full convolution with 1-h
    h_rev = 1 - h
    correlation = np.correlate(h, h_rev, mode='full')
    # The correlation is of length 2*n - 1, we want the middle part
    # corresponding to k from 0 to n-1, which is the second half
    # But we need all lag values, so we take the full result
    # Normalization by dx is required
    return np.max(correlation) * dx

def project_to_feasible(h, dx):
    # Project h to satisfy constraints: 0 <= h <= 1, sum(h)*dx = 1
    h = np.clip(h, 0, 1)
    # Rescale to make sum(h) * dx = 1
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * (1.0 / (sum_h * dx))
    return h

def compute_gradient(h, dx):
    # Use finite differences
    n = len(h)
    h1 = h.copy()
    grad = np.zeros_like(h)
    eps = 1e-6
    c0 = evaluate_h(h, dx)
    for i in range(n):
        h1[i] += eps
        c1 = evaluate_h(h1, dx)
        grad[i] = (c1 - c0) / eps
        h1[i] = h[i]  # restore
    return grad
