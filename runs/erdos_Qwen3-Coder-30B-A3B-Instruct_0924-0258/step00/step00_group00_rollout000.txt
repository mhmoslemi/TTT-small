import numpy as np
from scipy import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode='full')
    # Take the maximum correlation value (excluding trivial zeros)
    max_corr = np.max(corr[len(h)-1:])
    return max_corr * dx

def rescale_h(h, n_points, target_sum):
    """Ensure sum(h) == target_sum, keep values in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        # Create a reasonable starting construction
        n_points = 45
        h_current = np.ones(n_points) * 0.5
        h_current[0] = 0.7
        h_current[-1] = 0.3
        h_current = rescale_h(h_current, n_points, n_points / 2)
    
    dx = 2.0 / len(h_current)
    target_sum = len(h_current) / 2
    
    # PSO parameters
    n_particles = 30
    max_iter = 2000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    inertia_factor = 0.9
    v_max = 0.1
    
    # Initialize particles
    particles_h = []
    particles_v = []
    particles_best_h = []
    particles_best_c5 = []
    for i in range(n_particles):
        if i == 0:
            h = h_current.copy()
        else:
            h = h_current + np.random.normal(0, 0.1, len(h_current))
            h = rescale_h(h, len(h_current), target_sum)
        particles_h.append(h.copy())
        particles_v.append(np.zeros(len(h_current)))
        c5 = evaluate_c5(h, dx)
        particles_best_h.append(h.copy())
        particles_best_c5.append(c5)
    
    # Global best
    global_best_idx = np.argmin(particles_best_c5)
    global_best_c5 = particles_best_c5[global_best_idx]
    global_best_h = particles_best_h[global_best_idx].copy()
    
    iter_count = 0
    last_improvement = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 0.1:
        iter_count += 1
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            particles_v[i] = (w * particles_v[i] + 
                              c1 * r1 * (particles_best_h[i] - particles_h[i]) +
                              c2 * r2 * (global_best_h - particles_h[i]))
            
            # Keep velocity within bounds
            particles_v[i] = np.clip(particles_v[i], -v_max, v_max)
            
            # Update position
            particles_h[i] += particles_v[i]
            
            # Rescale to keep constraints
            particles_h[i] = rescale_h(particles_h[i], len(h_current), target_sum)
            
            # Evaluate the new position
            c5 = evaluate_c5(particles_h[i], dx)
            
            # Update particle's personal best
            if c5 < particles_best_c5[i]:
                particles_best_c5[i] = c5
                particles_best_h[i] = particles_h[i].copy()
            
            # Update global best
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best_h = particles_h[i].copy()
                last_improvement = iter_count
                
        # Adjust inertia weight
        if iter_count - last_improvement > 100:
            w = max(0.4, w * inertia_factor)
    
    return (global_best_h, global_best_c5, len(global_best_h))
