def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The full convolution has 2*n - 1 elements
    # The maximum overlap is over the non-zero lags
    max_val = np.max(conv)
    return max_val * dx
