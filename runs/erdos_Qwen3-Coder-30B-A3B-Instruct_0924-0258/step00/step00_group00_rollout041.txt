import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Binary-like construction
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    # Used to track consecutive non-improvements
    no_improvement_count = 0
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    def compute_correlations(h_vec):
        # Compute full correlation: h * (1 - h)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Extract the part corresponding to positive lags
        # The actual lags range from 0 to 2*n_points-2
        # So for lag k = 0, it's corr[n_points-1]
        # For lag k = 1, it's corr[n_points]
        # etc.
        # For backwards lag: corr[0] ... corr[n_points-2]
        max_corr = 0.0
        max_k = 0
        # We need to compute for lags 0 to 2*n_points-2
        # But we only care about the valid correlations for all k in [0, n_points-1]
        # Because for lag k > n_points-1, we don't have enough data
        correlations = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            if k < n_points:
                # Forward correlation from lag k
                corr_k = 0.0
                for i in range(n_points - k):
                    corr_k += h_vec[i] * (1 - h_vec[i + k])
                correlations[k] = corr_k * dx
            else:
                # Backward correlation, but only for valid entries
                corr_k = 0.0
                for i in range(k - n_points + 1, n_points):
                    if (i + (2 * n_points - 2 - k)) < n_points:
                        corr_k += h_vec[i] * (1 - h_vec[i + (2 * n_points - 2 - k)])
                correlations[k] = corr_k * dx
            if correlations[k] > max_corr:
                max_corr = correlations[k]
                max_k = k
        return correlations, max_corr, max_k
    
    def compute_gradients(h_vec, max_k_indices):
        gradient = np.zeros(n_points)
        for k in max_k_indices:
            # Compute contribution to gradient for this lag
            # Only the terms in correlation make a difference
            if k < n_points:
                # Forward correlation: h[i] * (1 - h[i+k])
                for i in range(n_points - k):
                    gradient[i] += (1 - h_vec[i + k]) * dx
                    if i + k < n_points:
                        gradient[i + k] -= h_vec[i] * dx
            else:
                # Backward: h[i] * (1 - h[i+k]) with appropriate bounds
                # But implementation becomes complex. We'll simple approximate it.
                # This term is not used in this simplified version.
                pass
        return gradient
    
    iter_count = 0
    
    while time.time() - start_time < budget_s - 0.1 and no_improvement_count < 20:
        iter_count += 1
        correlations, max_corr, max_k = compute_correlations(h)
        if max_corr < best_C5:
            best_C5 = max_corr
            best_h = h.copy()
            
        # Find all lags that achieve the maximum
        max_indices = [i for i in range(2 * n_points - 1) if abs(correlations[i] - max_corr) < 1e-10]
        
        # Compute gradient
        gradient = compute_gradients(h, max_indices)
        
        # Line search over step sizes
        step_sizes = [0.1, 0.25, 0.5]
        best_new_h = None
        best_new_C5 = max_corr
        
        for alpha in step_sizes:
            candidate_h = h - alpha * gradient
            candidate_h = project_h(candidate_h)
            new_C5 = evaluate_erdos_solution(candidate_h)
            if new_C5 < best_new_C5:
                best_new_C5 = new_C5
                best_new_h = candidate_h.copy()
                
        if best_new_h is not None:
            h = best_new_h
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # If stuck, perturb
            if no_improvement_count % 5 == 0:
                perturbation = 0.01 * np.random.randn(n_points)
                h = h + perturbation
                h = project_h(h)
    
    # Final validation
    final_C5 = evaluate_erdos_solution(best_h)
    return (list(best_h), final_C5, n_points)

def evaluate_erdos_solution(h_vec):
    n_points = len(h_vec)
    dx = 2.0 / n_points
    corr = np.correlate(h_vec, 1 - h_vec, mode="full")
    max_val = 0.0
    for k in range(2 * n_points - 1):
        val = corr[k] * dx
        if val > max_val:
            max_val = val
    return max_val
