from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Helper to compute overlaps
    def compute_overlaps(h_vec):
        h_vec = np.array(h_vec)
        # Create 1-h vector
        one_minus_h = 1 - h_vec
        # Compute cross-correlation for all lags
        corr = np.correlate(h_vec, one_minus_h, mode='full')
        # Adjust indices for the proper alignment
        corr_full = corr[len(h_vec)-1:]
        # Normalize
        overlaps = corr_full * dx
        return overlaps
    
    # Gradient computation for a given k
    def compute_gradient_for_k(h_vec, k):
        h_vec = np.array(h_vec)
        grad = np.zeros_like(h_vec)
        n = len(h_vec)
        for i in range(n):
            i_minus_k = (i - k) % n
            i_plus_k = (i + k) % n
            grad[i] = 1 - h_vec[i_minus_k] - h_vec[i_plus_k]
        return grad * dx

    # Adagrad-style updates
    learning_rate = 0.1
    accumulated_grad = np.zeros_like(h)
    momentum = 0.9
    velocity = np.zeros_like(h)
    eps = 1e-8
    
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        if iteration % 50 == 0:
            # Add noise to escape local minima
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        overlaps = compute_overlaps(h)
        k_star = np.argmax(overlaps)
        max_overlap = overlaps[k_star]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        grad = compute_gradient_for_k(h, k_star)
        
        # Adaptive learning rate
        velocity = momentum * velocity + grad
        accumulated_grad += grad ** 2
        adaptive_lr = learning_rate / (np.sqrt(accumulated_grad) + eps)
        h_new = h - adaptive_lr * velocity
        
        # Project to feasible space
        h_new = np.clip(h_new, 0, 1)
        # Rescale to sum to n_points / 2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        iteration += 1
        if iteration - last_improvement > 100 and best_c5 < 0.38085:
            break
            
    return (best_h.tolist(), best_c5, n_points)
