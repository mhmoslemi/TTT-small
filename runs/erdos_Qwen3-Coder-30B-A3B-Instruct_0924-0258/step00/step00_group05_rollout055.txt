import numpy as np

from scipy.signal import correlate

def project_h(h, dx):
    """Project h onto the feasible region."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    target_sum = 1.0 / dx  # n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def compute_c5(h, dx):
    """Compute C5 from the step function h."""
    # Compute correlation of h with 1-h
    corr = correlate(h, 1 - h, mode='full')
    # The max overlap is the maximum of the correlation
    max_overlap = np.max(corr) * dx
    return max_overlap

def compute_gradient(h, dx, max_lags):

    n = len(h)
    grad = np.zeros_like(h)
    # We compute approx gradient for the max lag pattern.
    # This is a simplified approach which should work well
    for k in max_lags:
        # Add contribution from lag k
        # Note: for correlation, when we're looking at h[i] vs 1-h[i+k], 
        # we get contribution from i to i+k, and from i+k to i.
        # We're partial derivative of sum over i of h[i] * (1-h[i+k]) 
        # = sum over i of (1 - h[i+k] - h[i])
        # So gradient[i] += 1 - h[i+k] for each lag k we're using
        # And we also consider -h[i] term, which gives us -1 on that term
        if k < n:
            # For the maximum lags, we're updating the corresponding locations
            # This is a simplification over full derivative calculation
            grad += 1
            # We also include the effect of -h[i+k] terms coming from the -1 multiplier
            # But split into symmetric small term
            if k < n:
                grad[:n-k] -= 1
            if k > 0:
                grad[k:] -= 1
    return grad * dx
