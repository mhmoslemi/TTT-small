import numpy as np
import scipy.optimize as opt
from time import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 42
    dx = 2.0 / n_points
    max_k = n_points - 1
    
    # Get initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure initial h satisfies constraints
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx, max_k)
    
    # Annealing parameters
    temp = 1.0
    min_temp = 0.01
    cooling_factor = 0.99
    patience = 50
    patience_counter = 0
    
    # Gradient parameters
    gradient_scale = 0.1
    noise_std = 0.05
    
    start_time = time()
    
    while time() - start_time < budget_s:
        # Compute current C5 value
        current_C5 = evaluate_C5(h, dx, max_k)
        
        # Random mutation with gradient guidance
        h_candidate = h.copy()
        
        # Add noise based on gradient
        grad = compute_gradient(h, dx, max_k)
        noise = np.random.normal(0, noise_std, n_points)
        h_candidate += gradient_scale * grad + noise
        
        # Project back to [0,1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Rescale to meet constraint
        h_sum = np.sum(h_candidate)
        if h_sum > 0:
            h_candidate = h_candidate * (n_points / 2) / h_sum
        
        # Evaluate new C5
        new_C5 = evaluate_C5(h_candidate, dx, max_k)
        
        # Accept or reject
        if new_C5 < current_C5 or np.random.random() < np.exp((current_C5 - new_C5) / temp):
            h = h_candidate
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
                patience_counter = 0
            else:
                patience_counter += 1
        else:
            patience_counter += 1
        
        # Adapt temperature
        temp = max(temp * cooling_factor, min_temp)
        
        # Warm restart if needed
        if patience_counter >= patience:
            temp = 1.0
            patience_counter = 0
    
    return (best_h.tolist(), best_C5, n_points)

def evaluate_C5(h, dx, max_k):
    # This computes max over all k of the overlap integral
    # Using correlation-style computation
    n = len(h)
    
    # Compute the correlation-like dot products
    # This is a bit inefficient but avoids explicit loops
    corr = np.correlate(h, 1 - h, mode='full')
    # Only use the right half (k = 0 to max_k)
    if len(corr) > max_k + 1:
        corr = corr[n-1:]
    max_val = np.max(corr) * dx
    return max_val

def compute_gradient(h, dx, max_k):
    n = len(h)
    
    # Finite difference approximation of gradient
    eps = 1e-5
    grad = np.zeros(n)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_sum = np.sum(h_plus)
        if h_sum > 0:
            h_plus = h_plus * (n / 2) / h_sum
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = np.clip(h_minus, 0, 1)
        h_sum = np.sum(h_minus)
        if h_sum > 0:
            h_minus = h_minus * (n / 2) / h_sum
        
        C5_plus = evaluate_C5(h_plus, dx, max_k)
        C5_minus = evaluate_C5(h_minus, dx, max_k)
        grad[i] = (C5_plus - C5_minus) / (2 * eps)
    
    return grad
