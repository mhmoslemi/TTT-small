import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Problem parameters
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * target_sum / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * target_sum / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_h(h, n_points, dx)
    
    # Gradient descent parameters
    max_iterations = int(budget_s * 2)  # Rough estimate: ~0.5s per iteration
    alpha = 0.1
    tolerance = 1e-6
    last_improvement = 0
    
    start_time = time.time()
    
    for i in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.98:  # Leave some buffer
            break
            
        # Compute all overlaps
        overlaps = evaluate_overlaps(h, n_points, dx)
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # If improvement, update best
        if max_overlap < best_c5:
            best_h = h.copy()
            best_c5 = max_overlap
            
        # Compute gradient for max overlap
        grad = np.zeros(n_points)
        for j in range(n_points):
            k = max_k
            # Compute partial derivative d/dh[j] of z_k
            if (j - k) % n_points < n_points:
                left = h[(j - k) % n_points]
            else:
                left = 0
            
            if (j + k) % n_points < n_points:
                right = h[(j + k) % n_points]
            else:
                right = 0
                
            grad[j] = (2.0 / n_points) * (1 - left - right)
            
        # Update h
        h_new = h - alpha * grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain constraint
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * target_sum / current_sum
            
        # Ensure we maintain the exact sum constraint
        h_new -= np.mean(h_new) - target_sum / n_points
        
        # Perturb occasionally to escape local optima
        if i % 50 == 0 and i > 0:
            h_new += np.random.normal(0, 0.005, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            
        h = h_new
        
        # Simple stopping criterion
        if i > 10 and abs(max_overlap - best_c5) < tolerance:
            last_improvement += 1
            if last_improvement > 10:
                break
        else:
            last_improvement = 0
                    
    # Final evaluation
    final_c5 = evaluate_h(best_h, n_points, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_h(h, n_points, dx):
    """Compute C5 for given h"""
    # This uses fft-based correlation for efficiency
    # Wrap the forward and backward correlations
    h_vec = h.copy()
    h_rev = h_vec[::-1]  # reversed for auto-correlation
    
    # Do full convolution with 1-h
    corr = correlate(h_vec, 1-h_vec, mode='full') * dx
    
    # Extract valid overlaps
    start_idx = n_points - 1
    end_idx = 2 * n_points - 1
    overlaps = corr[start_idx:end_idx]
    
    return np.max(overlaps)

def evaluate_overlaps(h, n_points, dx):
    """Compute all overlaps for given h"""
    overl = np.zeros(n_points)  # We only compute up to n_points/2 shifts
    h_vec = h.copy()
    h_adj = 1 - h_vec
    
    # Use direct method for overlap computation
    for k in range(n_points):
        # Compute convolution at lag k
        if k == 0:
            overlap = np.sum(h_vec * h_adj) * dx
        else:
            h_shifted = np.roll(h_vec, k)
            overlap = np.sum(h_vec * h_shifted) * dx
        overl[k] = overlap
        
    return overl
