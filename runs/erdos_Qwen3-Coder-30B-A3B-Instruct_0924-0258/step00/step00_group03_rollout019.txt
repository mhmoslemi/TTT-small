import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for a given step function h."""
    # Compute the overlap integral for all shifts
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum of the correlation values
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Since integral should be 1
    
    # Initialize from the provided construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        # Ensure proper normalization
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
    else:
        # Create a reasonable starting point (simplified initial construction)
        h = np.ones(n_points) * 0.5
        # Adjust to get sum = target_sum
        h = h * target_sum / np.sum(h)
    
    # Check constraints
    assert np.all(h >= 0) and np.all(h <= 1), "Initial h must be in [0,1]"
    assert abs(np.sum(h) - target_sum) < 1e-10, "Initial h must sum to target_sum"
    
    # Optimization parameters
    T0 = 1.0
    alpha = 0.999
    sigma0 = 0.1
    max_iter = int(budget_s * 1000)  # Rough estimate: 1000 iterations/sec
    
    # Current best
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(best_h)
    
    # Annealing schedule and iteration tracking
    t = 0
    T = T0
    start_time = time.time()
    
    while t < max_iter:
        t += 1
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Update temperature
        T = T0 * (alpha ** t)
        
        # Mutation: Add Gaussian noise
        sigma = sigma0 * (alpha ** (t / 200))
        h_new = h + np.random.normal(0, sigma, n_points)
        
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to sum to target_sum
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * target_sum / h_new_sum
        
        # Evaluate new solution
        C5_new = evaluate_erdos_solution(h_new)
        
        # Accept or reject
        delta_E = C5_new - best_C5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_new
            if C5_new < best_C5:
                best_h = h_new.copy()
                best_C5 = C5_new
                
    # Final check
    final_C5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_C5, n_points)
