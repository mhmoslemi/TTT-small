import numpy as np
from scipy import optimize
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, norm, square

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute convolution
    conv = np.correlate(h, 1 - h, mode="full")
    # Maximum value among all shifts
    max_val = np.max(conv)
    return max_val * dx

def project_h(h, n_points):
    # Use CVXPY to project onto feasible set
    h_var = Variable(n_points)
    # Constraints
    constraints = [0 <= h_var, h_var <= 1, cp_sum(h_var) == n_points / 2]
    # Objective: minimize distance to h
    objective = Minimize(square(cp_sum((h_var - h)**2)))
    prob = Problem(objective, constraints)
    prob.solve(solver='ECOS')
    return np.array(h_var.value).flatten()

            # How much h[j] affects the term at lag k0
            # i: position in h
            # i+k0: corresponding position in (1-h) if mode full
            # Since 1-h is flipped, for lag k0, we are looking at j in h, j+k0 in (1-h)
            # So h[j] contributes to the k0-th shift with weight (1-h[j+k0])
            if j + k0 < n_points:
                grad[j] = -1 * (1 - h[j + k0])
            # Same for the reverse position
            if k0 <= j:
                grad[j] += (1 - h[j - k0])
                
        # Update and project
        h_new = h - learning_rate * grad

            learning_rate *= decay_rate
    
    return (best_h, best_c5, n_points)
