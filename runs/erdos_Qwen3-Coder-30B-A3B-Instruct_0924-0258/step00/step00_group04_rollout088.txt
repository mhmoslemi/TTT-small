import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate C5 for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # The maximum correlation corresponds to indices from n-1 to 2*n-2
    # But we only need from 0 to n-1 (first half)
    max_corr = np.max(corr[n-1:])
    return max_corr * dx

def project_h(h, n_points):
    """Project h to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 42  # Use the initial construction size if available
    
    # Initialize h from existing construction or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        h = np.ones(n_points) * 0.5
        h = project_h(h, n_points)
        
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Annealing parameters
    temperature = 1.0
    cooling_factor = 0.99
    min_temp = 0.01
    patience = 50
    patience_counter = 0
    start_time = time.time()
    
    # Optimization loop
    while time.time() - start_time < budget_s:
        # Perturb h with gradient guidance and random noise
        c5_current = evaluate_c5(h)
        
        # Compute gradient via finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            grad[i] = (c5_plus - c5_current) / eps
        
        # Generate candidate
        noise = np.random.normal(0, 0.05, n_points)
        h_candidate = h - 0.1 * grad + noise
        h_candidate = project_h(h_candidate, n_points)
        
        # Evaluate candidate
        c5_new = evaluate_c5(h_candidate)
        
        # Accept or reject
        if c5_new < c5_current or np.random.random() < np.exp((c5_current - c5_new) / temperature):
            h = h_candidate.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                patience_counter = 0
            else:
                patience_counter += 1
        else:
            patience_counter += 1
            
        # Update temperature
        temperature = max(temperature * cooling_factor, min_temp)
        
        # Reset temperature if no improvement for some steps
        if patience_counter >= patience:
            temperature = 1.0
            patience_counter = 0
    
    return (best_h.tolist(), best_c5, n_points)
