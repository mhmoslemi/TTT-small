import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the overlap integral for a given step function h."""
    # Compute the full correlation (equivalent to the overlap integral)
    corr = correlate(h, 1 - h, mode="full")
    # The maximum correlation (which corresponds to the worst-case overlap)
    max_overlap = np.max(corr)
    return max_overlap * dx

def normalize(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.full_like(h, target_sum / len(h))
    return h * (target_sum / current_sum)

def feasible_h(h, target_sum):
    """Make h feasible with bounds and target sum."""
    h = np.clip(h, 0, 1)
    return normalize(h, target_sum)

def mutate(h, mutation_rate=0.1):
    """Mutate h with bounded noise."""
    # Add noise with bounded magnitude
    noise = np.random.normal(0, mutation_rate * 0.5, len(h))
    h_new = h + noise
    # Keep within bounds
    h_new = np.clip(h_new, 0, 1)
    # Renormalize to satisfy constraint
    return normalize(h_new, np.sum(h))

def crossover(h1, h2):
    """Perform crossover between two step functions."""
    # Simple uniform crossover
    mask = np.random.rand(len(h1)) > 0.5
    h_new = np.where(mask, h1, h2)
    return h_new

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection."""
    idx = np.random.choice(len(population), tournament_size)
    best_idx = idx[np.argmin([fitnesses[i] for i in idx])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 1000
    tournament_size = max(1, pop_size // 10)
    
    # Initial population

        h = np.random.rand(n_points)
        h = feasible_h(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            f = evaluate(h, dx)
            fitnesses.append(f)
            
        # Track best
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = population[best_idx].copy()
            
        # Tournament selection
        new_population = []
        for _ in range(pop_size):
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            if np.random.rand() < crossover_rate:
                parent2 = tournament_selection(population, fitnesses, tournament_size)
                child = crossover(parent1, parent2)
            else:
