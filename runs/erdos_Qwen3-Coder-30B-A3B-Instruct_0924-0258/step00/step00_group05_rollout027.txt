import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def evaluate_erdos_solution(h, dx):
    # Compute the overlap integral for the given step function h
    n = len(h)
    # Create the flipped version of h
    h_flip = 1 - h
    # Compute the full convolution with flipped h
    correlation = np.correlate(h, h_flip, mode="full")
    # Take the maximum value (excluding the zero padding)
    max_overlap = np.max(correlation[n-1:])
    return max_overlap * dx

        constraints = [

            # h[i] * (1 - h[i + k]) with proper wrapping

            if k == 0:
