import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for the given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the full convolution of h with (1-h)
    conv = np.correlate(h, 1-h, mode="full")
    # Return the maximum value scaled by dx
    return np.max(conv) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration
    if 'initial_h_values' in globals():

        n_points = len(h_base)

        n_points = 45

        h_base[0] = 0.0
        h_base[n_points-1] = 0.0
        h_base = h_base / np.sum(h_base) * n_points / 2.0
    
    # Optimization parameters
    resolutions = [15, 30, 45, 60, 75]
    max_perturbations = 3
    time_start = time()
    
    best_h = h_base.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Sequential resolution optimization
    for n in resolutions:
        if time() - time_start > budget_s - 5:
            break
        
        # Interpolate to current resolution
        h_current = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_h)), best_h)
        n_points = n
        
        for _ in range(max_perturbations):
            if time() - time_start > budget_s - 5:
                break
                
            # Initial guess
            h0 = h_current.copy()
            
            # Rescale to meet constraint
            h0 = h0 / np.sum(h0) * n_points / 2.0
            
            # Define the convex surrogate objective (sum of squares of overlaps)
            def surrogate_objective(h_vec):
                h_vec = np.clip(h_vec, 0, 1)
                # Convolution
                conv = np.correlate(h_vec, 1 - h_vec, mode="full")
                # Return sum of squares
                return np.sum(conv**2) * (2.0 / n)**2
            
            # Constraints
            def constraint_sum(h_vec):
                return np.sum(h_vec) - n_points / 2.0
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            # Bounds
            bounds = [(0, 1) for _ in range(n)]
            
            # Optimize
            result = minimize(surrogate_objective, h0, 
                             method='SLSQP', bounds=bounds, constraints=cons, 
                             options={'maxiter': 200})
            
            if result.success:
                h_opt = np.clip(result.x, 0, 1)
                # Correct the sum constraint
                h_opt = h_opt / np.sum(h_opt) * n_points / 2.0
                c5_new = evaluate_erdos_solution(h_opt)
                
                if c5_new < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_new
            
            # Perturb current solution for next iteration
            perturbation = np.random.normal(0, 0.02, n)
            h_current = h_current + perturbation
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current / np.sum(h_current) * n_points / 2.0
    
    return (best_h.tolist(), best_c5, n_points)
