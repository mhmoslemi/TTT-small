import numpy as np
from scipy import optimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    except:
        h = np.full(n_points, 0.5)
    
    # Ensure sum(h) = n_points / 2 exactly
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_C5(h):
        # Compute all correlations 
        corr = np.correlate(h, 1 - h, mode='full')
        # Max over all lags (including negative)
        max_corr = np.max(corr)
        return max_corr * dx
    
    def project_h(h):
        # Clip to [0,1] and renormalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def compute_gradient(h, max_lags):
        # Initialize gradient
        grad = np.zeros_like(h)
        # For each lag contributing to max
        for k in max_lags:
            # This is an approximation to the gradient
            # dy/dh_i = sum_k (1 - h[i+k] - 1 + h[i]) delta_k
            # = sum_k (h[i] - h[i+k]) delta_k
            for i in range(n_points):
                # In periodic boundary conditions:
                j = (i + k) % n_points
                grad[i] += (h[i] - h[j])
        return grad * dx
    
    # Initialization
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    # Line search parameters
    alpha_max = 0.1
    alpha_min = 1e-6
    alpha = alpha_max
    no_improvement_count = 0
    max_no_improvement = 50
    
    t = 0
    while time() - start_time < budget_s - 1 and t < 10000:
        t += 1
        
        # Compute all overlaps and find max lags
        overlaps = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(overlaps)
        max_lags = np.where(np.abs(overlaps - max_val) < 1e-10)[0]
        # Adjust indices to be within [-n+1, n-1]
        max_lags = max_lags - (n_points - 1)
        
        # Compute gradient
        grad = compute_gradient(h, max_lags)
        
        # Line search for alpha
        alpha_trial = alpha
        h_trial = h - alpha_trial * grad
        h_trial = project_h(h_trial)
        C5_trial = evaluate_C5(h_trial)
        
        # Backtrack if needed
        while C5_trial >= best_C5 and alpha_trial > alpha_min:
            alpha_trial *= 0.5
            h_trial = h - alpha_trial * grad
            h_trial = project_h(h_trial)
            C5_trial = evaluate_C5(h_trial)
        
        if C5_trial < best_C5:
            h = h_trial
            best_C5 = C5_trial
            alpha = min(alpha * 1.2, alpha_max)
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Perturb
            pert = np.random.normal(0, 0.01, n_points)
            h += pert
            h = project_h(h)
            C5_new = evaluate_C5(h)
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
            
        if no_improvement_count > max_no_improvement:
            # Random restart
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
            alpha = alpha_max
    
    # Final projection and evaluation
    best_h = project_h(best_h)
    final_C5 = evaluate_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
