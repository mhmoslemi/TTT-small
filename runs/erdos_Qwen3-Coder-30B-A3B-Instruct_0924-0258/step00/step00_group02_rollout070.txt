import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution h * (1 - h)
    conv = np.correlate(h, 1 - h, mode="full")
    # Take maximum over all shifts
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_to_feasible(h, n_points):
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def convex_surrogate(h, n_points):
    dx = 2.0 / n_points
    # Use square of overlap as surrogate
    conv = np.correlate(h, 1 - h, mode="full")
    return np.sum(conv**2) * dx**2

def gradient_convex_surrogate(h, n_points):
    dx = 2.0 / n_points
    # Approximate gradient of surrogate
    conv = np.correlate(h, 1 - h, mode="full")
    # Gradient w.r.t h is 2 * conv * (1 - h) * dx^2
    # But this is approximate; actual would need to compute full Jacobian
    grad = np.zeros_like(h)
    for i in range(len(h)):
        for k in range(len(conv)):
            # The derivative of the convolution at i with respect to h[i]
            # Involves the shifted terms of (1-h)
            if 0 <= k - i < len(h):
                grad[i] += 2 * conv[k] * (-1) * dx**2
    return grad

def optimize_at_resolution(h0, n_points, time_limit):
    start_time = time.time()
    dx = 2.0 / n_points
    
    def objective(h):
        return convex_surrogate(h, n_points)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial feasible h
    h = project_to_feasible(h0, n_points)
    
    # Optimize
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds, 
                       constraints=constraints, 
                       options={'maxiter': 1000, 'ftol': 1e-8}, 
                       callback=lambda x: None if time.time() - start_time < time_limit else None)
        if res.success:
            h = res.x
    except:
        pass
        
    h = project_to_feasible(h, n_points)
    return h

        if time.time() - start_time > budget_s - 10:
            break
        # Interpolate to required resolution
        if res > n_points:
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            h = np.interp(x_new, x_old, h)
