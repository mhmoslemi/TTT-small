import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
    except:
        # Fallback: create simple piecewise constant construction
        h = np.ones(45)
        h[:22] = 0.5
        h[22:] = 1.5
        
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h *= target_sum / np.sum(h)  # Normalize to sum = target_sum
    
    # Optimization parameters
    lr = 0.01
    momentum = 0.9
    v = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    iterations = 0
    while time.time() - start_time < budget_s - 0.1:
        iterations += 1
        
        # Compute all overlaps
        overlap_vals = np.correlate(h, 1 - h, mode='full') * dx
        # We only care about overlaps from k=0 to k=n_points-1 
        # (the full correlation includes negative lags which are not needed)
        overlaps = overlap_vals[n_points-1:]  # This excludes k=0, but we want max_k
        
        # Should include k=0 to k=n_points-1 
        # actual indices and val indices: 
        # idx=0 corresponds to k=n_points-1, 
        # idx=n_points-1 corresponds to k=0,
        # idx=n_points corresponds to k=1 etc.
        # So, overlaps = overlap_vals[n_points-1:] indexed by k from 0 to n_points-1
        # This assumes we want the maximal overlap_k for k ranging from 0 to n_points-1
        k_max = np.argmax(overlaps)
        if k_max < 0:
            k_max = 0
            
        # Use top contributing overlaps to build gradient
        top_k = [k_max]
        # Include 4 more nearby k values for gradient
        for i in range(1, 5):
            if k_max + i < n_points:
                top_k.append(k_max + i)
            if k_max - i >= 0:
                top_k.append(k_max - i)
        top_k = list(set(top_k))  # remove duplicates
        
        # Compute hessian and gradient for each k in top_k
        grad = np.zeros_like(h)
        for k in top_k:
            # For k=0, k=1, ..., n_points-1 we need to consider:
            # overlap_k = sum_i h[i] * (1 - h[(i+k)%n_points]) * dx
            # Grad = sum_i (1 - h[(i+k)%n_points]) * dx
            
            # For this k, compute derivative w.r.t. h[i]: 
            # d/dh[j] sum_i h[i] * (1 - h[(i+k)%n_points]) = 
            # = sum_i 1_{i=j} * (1 - h[(i+k)%n_points]) = (1 - h[(j+k)%n_points])
            # the gradient is [1 - h[(0+k)%n], 1 - h[(1+k)%n], ...]
            # Wait no: h[i] is the m-th element in h
            # gradient w.r.t. h_i = sum_{all j} d/dh_i [h_j * (1 - h_{(j+k)%n})] = 
            # 1_{i=j} * (1 - h_{(j+k)%n}) = (1 - h_{(i+k)%n})
            # so d/dh[i] sum_i h[i] * (1 - h[(i+k)%n])
            #   = 1 - h[(i+k)%n]
            # But that's the component of the gradient:
            # grad_h[i] = 1 - h[(i+k)%n]
            # We sum over all k of these contributions. 
            # More accurately: since overlap  = sum_i h[i] * (1 - h[(i+k)%n]) 
            # The contribution to grad for k is: (1 - h[(i+k)%n] ) 
            # So total gradient for i = sum_k (1 - h[(i+k)%n] ) 
            # But that is for over all the k involved in top_k.
            # This is approximate. We compute for a select few k.
            # Actual gradient: for k = idx_arr[0], ..., until we fill top_k:
            # And mod indices properly
            k_shift = k
            h_shifted = np.roll(h, k_shift)  # This is h[(i+k)%n_points] 
            grad += (1 - h_shifted)
        
        # Apply gradient with momentum
        v = momentum * v + lr * grad
        h_new = h - v
        
        # Project h onto constraints
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        h_new = h_new * target_sum / h_sum  # Rescale
        
        # Evaluate new solution
        # Overlaps now computed on loop
        h = h_new
        c5_new = evaluate_erdos_solution(h, dx)
        
        # Update best
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Step size adaptation - we reduce learning rate if no improvement
        # Simple strategy: every 50 iterations or if small step
        if iterations % 50 == 0:
            lr *= 0.95
            
    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_erdos_solution(h_values, dx):
    h = np.array(h_values)
    # Compute the overlaps for k=0 to k=n_points-1
    crosscorr = np.correlate(h, 1-h, mode='full') 
    n_points = len(h)
    overlaps = crosscorr[n_points-1:]  # Overlaps from k=0 to n_points-1  
    c5 = np.max(overlaps) * dx
    return c5
