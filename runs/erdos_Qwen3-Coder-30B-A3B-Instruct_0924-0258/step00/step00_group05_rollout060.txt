import numpy as np
from scipy import signal
import time
import random

def evaluate_C5(h):
    """Evaluate C5 using the correlation approach"""
    n = len(h)
    dx = 2.0 / n
    # Compute full cross-correlation of h and (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Max value of correlation scaled by dx
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_constraint(h, target_sum):
    """Project h to satisfy integral constraint"""
    current_sum = np.sum(h)
    if current_sum == target_sum:
        return h
    # Rescale to target sum
    h = h * target_sum / current_sum
    return h

def perturb_h(h, T, sigma_factor=1.0):
    """Perturb h with Gaussian noise, then project"""
    n = len(h)
    # Add noise
    noise = np.random.normal(0, T * sigma_factor, n)
    h_new = h + noise
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Project to constraint
    h_new = project_to_constraint(h_new, n / 2)
    return h_new

def adaptive_cooling(T, iteration, max_iter):
    """Adaptive cooling schedule"""
    # Start at T=0.1, reduce with cooling factor
    cooling_factor = 0.99
    return T * (cooling_factor ** iteration)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction if available

        n_points = len(h)
    else:

        n_points = 10
        h = np.random.rand(n_points)
        h = project_to_constraint(h, n_points / 2)
    
    # Store best solution
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    # Multi-resolution optimization
    resolution_levels = [10, 14, 20, 28, 40, 56, 70]
    current_n = n_points
    
    for level, target_n in enumerate(resolution_levels):
        if target_n < current_n:
            continue  # Skip if target smaller than current
            
        # Interpolate to target resolution
        if target_n > current_n:
            # Simple upsampling strategy
            h_upsampled = np.interp(np.linspace(0, 1, target_n), 
                                  np.linspace(0, 1, current_n), h)
            # Clip to [0,1]
            h = np.clip(h_upsampled, 0, 1)
            # Project to constraint
            h = project_to_constraint(h, target_n / 2)
            current_n = target_n
        
        # Simulated Annealing at this resolution
        T = 0.1
        T_min = 1e-6
        max_iter = 500
        iter_count = 0
        
        while iter_count < max_iter and (time.time() - start_time) < budget_s * 0.95:
            # Perturb h
            h_new = perturb_h(h, T)
            
            # Evaluate
            C5_new = evaluate_C5(h_new)
            
            # Metropolis criterion
            delta = C5_new - best_C5
            
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_new
                if C5_new < best_C5:
                    best_C5 = C5_new
                    best_h = h.copy()
            
            # Cooling
            T = adaptive_cooling(T, iter_count, max_iter)
            iter_count += 1
            
            # Early termination if T gets too small
            if T < T_min:
                break
                
            if (time.time() - start_time) > budget_s * 0.95:
                break
    
    # Final validation
    final_C5 = evaluate_C5(best_h)
    return (best_h.tolist(), final_C5, len(best_h))
