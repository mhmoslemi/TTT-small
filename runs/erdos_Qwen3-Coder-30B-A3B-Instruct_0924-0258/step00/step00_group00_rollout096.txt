import numpy as np
import time
import random
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute the correlation with 1-h
    corr = correlate(h, 1 - h, mode='full')
    # Extract the relevant part (the maximum overlap)
    # The maximum lag is 2/dx, so corr has length 2*n_points - 1
    # We want from lag -n_points+1 to n_points-1, inclusive
    # This maps to indices n_points-1 to 2*n_points-2
    max_corr = np.max(corr[n_points-1:2*n_points-1])
    return max_corr * dx

def normalize_h(h, dx):
    # Rescale h to ensure sum(h) * dx = 1
    total = np.sum(h)
    target = n_points / 2.0
    if total > 0:
        h = h * target / total
    return h

def mutate_individual(h, mutation_rate=0.1, delta=0.1):
    h_mut = h.copy()
    for i in range(n_points):
        if random.random() < mutation_rate:
            h_mut[i] += random.uniform(-delta, delta)
    h_mut = np.clip(h_mut, 0, 1)
    h_mut = normalize_h(h_mut, dx)
    return h_mut

def crossover(parent1, parent2):
    # One-point crossover
    point = random.randint(1, n_points - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    child1 = normalize_h(child1, dx)
    child2 = normalize_h(child2, dx)
    return child1, child2

def tournament_selection(population, fitnesses, k=3):
    idxs = random.sample(range(len(population)), k)
    best = min(idxs, key=lambda i: fitnesses[i])
    return population[best]

def run(seed=42, budget_s=600, **kwargs):
    global n_points, dx
    random.seed(seed)
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    pop_size = 50
    generations = 1000
    crossover_rate = 0.7
    mutation_rate = 0.2
    
    # Initialize population
    if 'initial_h_values' in globals():

        parent = normalize_h(parent, dx)
    else:
        # Create initial random solution
        parent = np.random.rand(n_points)
        parent = normalize_h(parent, dx)
    
    population = [parent]
    # Add diverse candidates
    for _ in range(pop_size - 1):
        # Perturb the parent slightly
        h = mutate_individual(parent, delta=0.2)
        population.append(h)
    
    # Evaluate fitness of initial population
    start_time = time.time()
    
    best_h = parent.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute fitness for all individuals
        fitnesses = []
        for h in population:
            fitness = evaluate_h(h, dx)
            fitnesses.append(fitness)
            
        # Track best
        for i in range(len(population)):
            if fitnesses[i] < best_c5:
                best_c5 = fitnesses[i]
                best_h = population[i].copy()
                
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep top 20% as elites
        elites = population[:int(0.2 * pop_size)]
        
        # Generate new population
        new_population = elites[:]
        
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if random.random() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
                new_population.extend([child1, child2])
            else:
                # Clone
                new_population.extend([parent1, parent2])
                
        # Keep only pop_size
        new_population = new_population[:pop_size]
        
        # Mutation
        for i in range(int(mutation_rate * pop_size)):
            idx = int(random.random() * len(new_population))
            new_population[idx] = mutate_individual(new_population[idx])
            
        population = new_population
        
    return (best_h.tolist(), best_c5, n_points)
