import numpy as np
import time
from scipy import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for a given step function h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def rescale_h(h, target_sum):
    """Rescale h to have the target sum while keeping values in [0,1]"""
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.ones_like(h) * (target_sum / len(h))
    h_scaled = h * (target_sum / current_sum)
    # Clamp to [0,1] and rescale again
    h_clamped = np.clip(h_scaled, 0, 1)
    h_clamped = h_clamped * (target_sum / np.sum(h_clamped))
    return h_clamped

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Adjusted for speed and performance
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a reasonable starting solution
        h = np.random.rand(n_points)
        h = rescale_h(h, target_sum)
    
    # Initialize PSO variables
    n_particles = 30
    inertia = 0.7
    c1 = c2 = 2.0
    max_iter = 1000
    no_improve_count = 0
    no_improve_limit = 200
    
    # Initialize particles
    particles = [h.copy() for _ in range(n_particles)]
    velocities = [np.zeros_like(h) for _ in range(n_particles)]
    
    # Initialize personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [evaluate_c5(p, dx) for p in particles]
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 0.1 and iteration < max_iter:
        iteration += 1
        no_improve_count += 1
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] +
                             c1 * r1 * (personal_best[i] - particles[i]) +
                             c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Boundary conditions and rescaling
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = rescale_h(particles[i], target_sum)
            
            # Evaluate current particle
            current_score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_best[i] = particles[i].copy()
                no_improve_count = 0
            
            # Update global best
            if current_score < global_best_score:
                global_best_score = current_score
                global_best = particles[i].copy()
                no_improve_count = 0
                
        if no_improve_count > no_improve_limit:
            break
    
    # Final evaluation of global best
    final_c5 = evaluate_c5(global_best, dx)
    
    return (global_best.tolist(), final_c5, n_points)
