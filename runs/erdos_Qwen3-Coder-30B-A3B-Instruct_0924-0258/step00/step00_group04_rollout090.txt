import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    n_points = 42
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure sum is correct
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    # Define step size and parameters
    step_size = 0.1
    max_iter = int(budget_s * 2)  # Rough estimate of iterations per second
    iter_count = 0
    
    # Gradient descent with line search
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute all correlations for all lags
        correlations = np.correlate(h, 1 - h, mode='full')
        correlations = correlations[len(h) - 1:]  # Only the valid parts
        correlations *= 2.0 / n_points  # Scale properly
        
        # Find worst case lag
        k_star = np.argmax(correlations)
        
        # Compute gradient for this lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            prev_idx = (i - k_star) % n_points
            next_idx = (i + k_star) % n_points
            grad[i] = 2.0 / n_points * (1 - h[prev_idx] - h[next_idx])
            
        # Line search to find best step size
        alpha = step_size
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        new_C5 = evaluate_erdos_solution(h_new)
        
        if new_C5 < best_C5:
            h = h_new
            best_C5 = new_C5
            best_h = h.copy()
            step_size *= 1.1  # Increase step size if improvement happened
        else:
            step_size *= 0.5  # Reduce step size
            
        iter_count += 1
        
        # Occasionally add noise to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_C5, n_points)

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    corr = corr[len(h) - 1:] * dx
    return np.max(corr)
