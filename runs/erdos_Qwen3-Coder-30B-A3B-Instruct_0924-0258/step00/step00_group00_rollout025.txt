import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial solution or fallback
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
    
    # Rescale to ensure sum constraint
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    
    def evaluate(h):
        # Ensure sum constraint
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        # Compute overlap integrals
        h_corr = correlate(h, 1 - h, mode="full")
        # Scale by dx to get integral
        max_overlap = np.max(h_corr) * dx
        return max_overlap
    
    # Initialize swarm
    n_particles = 30
    particles = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    
    for i in range(n_particles):
        pos = h.copy()
        if i > 0:  # Perturb previous solution
            pos += np.random.normal(0, 0.05, n_points)
        pos = np.clip(pos, 0, 1)
        pos = pos * target_sum / np.sum(pos)
        particles.append(pos)
        velocities.append(np.zeros(n_points))
        val = evaluate(pos)
        personal_best.append(pos.copy())
        personal_best_vals.append(val)
    
    global_best = min(personal_best_vals)
    global_best_pos = personal_best[np.argmin(personal_best_vals)].copy()
    
    # PSO parameters
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    stagnation_count = 0
    stagnation_limit = 100
    
    iteration = 0
    while time.time() - start_time < budget_s - 1.0 and iteration < max_iter:
        iteration += 1
        
        improved = False
        for i in range(n_particles):
            pos = particles[i]
            vel = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            vel = (inertia * vel +
                   c1 * r1 * (personal_best[i] - pos) +
                   c2 * r2 * (global_best_pos - pos))
            velocities[i] = vel
            
            # Update position
            pos += vel
            pos = np.clip(pos, 0, 1)
            pos = pos * target_sum / np.sum(pos)
            particles[i] = pos
            
            # Evaluate
            val = evaluate(pos)
            if val < personal_best_vals[i]:
                personal_best_vals[i] = val
                personal_best[i] = pos.copy()
                if val < global_best:
                    global_best = val
                    global_best_pos = pos.copy()
                    improved = True
        
        if not improved:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        if stagnation_count > stagnation_limit:
            # Reinitialize some particles
            for i in range(n_particles // 3):
                idx = np.random.randint(0, n_particles)
                particles[idx] = np.random.rand(n_points)
                particles[idx] = particles[idx] / np.sum(particles[idx]) * target_sum
                velocities[idx] = np.zeros(n_points)
                val = evaluate(particles[idx])
                personal_best_vals[idx] = val
                personal_best[idx] = particles[idx].copy()
                if val < global_best:
                    global_best = val
                    global_best_pos = particles[idx].copy()
            stagnation_count = 0

    # Final evaluation
    final_h = np.clip(global_best_pos, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h)
    
    return (final_h.tolist(), global_best, n_points)
