import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_low = 10
    n_high = 70
    max_iter_per_level = 500
    initial_T = 0.1
    cooling_factor = 0.99
    max_levels = int(np.log2(n_high // n_low)) + 1
    
    # Initialize starting solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_low / 2)
    else:
        h = np.random.rand(n_low)
        h = h / np.sum(h) * (n_low / 2)
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Multi-resolution optimization
    current_n = n_low
    T = initial_T
    
    level = 0
    while current_n <= n_high and time.time() - start_time < budget_s * 0.95:
        if level > 0:
            # Interpolate to higher resolution
            h = np.interp(np.linspace(0, 1, current_n), 
                         np.linspace(0, 1, current_n // 2), 
                         h)
            h = h / np.sum(h) * (current_n / 2)
            
        # Local optimization at this resolution
        for iter_num in range(max_iter_per_level):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Perturb solution
            h_new = h + np.random.normal(0, T, len(h))
            h_new = np.clip(h_new, 0, 1)
            # Adjust to sum constraint
            h_new = h_new / np.sum(h_new) * (current_n / 2)
            
            # Compute new C5
            C5_new = compute_C5(h_new)
            C5_old = compute_C5(h)
            
            # Accept or reject
            delta = C5_new - C5_old
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_new
                if C5_new < best_C5:
                    best_h = h.copy()
                    best_C5 = C5_new
            
            # Cool down
            T *= cooling_factor
            
        current_n *= 2
        level += 1
    
    # Final evaluation
    final_h = best_h
    final_C5 = compute_C5(final_h)
    
    # Ensure bounds
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h / np.sum(final_h) * (len(final_h) / 2)
    
    assert np.isclose(np.sum(final_h), len(final_h)/2), "Sum constraint violated"
    assert np.all((final_h >= 0) & (final_h <= 1)), "Bounds violated"
    
    return (final_h.tolist(), final_C5, len(final_h))


def compute_C5(h):
    """Compute C5 from step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation with 1 - h
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value scaled by dx
    return np.max(corr) * dx
