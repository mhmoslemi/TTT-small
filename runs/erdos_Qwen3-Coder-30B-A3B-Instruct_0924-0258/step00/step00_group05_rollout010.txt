import numpy as np
from scipy import signal
import time
import random

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound for a given h vector."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    conv = signal.convolve(h_values, 1 - np.array(h_values), mode='full')
    # Scale by dx and return max
    return np.max(conv) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_low = 10
    n_high = 100
    max_iter_per_level = 500
    
    # Get initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n = len(h)
        # Initialize with current solution if proper size
        if n_low <= n <= n_high:
            h = h.copy()
        else:
            h = np.ones(n_low) * 0.5
            n = n_low
    else:
        h = np.ones(n_low) * 0.5
        n = n_low

    # Normalize to sum to n/2
    h = h * (n / (2 * np.sum(h)))
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    # Multi-resolution approach
    levels = []
    current_n = n_low
    while current_n <= n_high:
        levels.append(current_n)
        current_n *= 2
    
    current_n = n_low
    
    # Initialize residual time
    remaining_time = budget_s
    
    for level_n in levels:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Interpolate to next resolution if needed
        if level_n > current_n:
            old_h = h.copy()
            h = np.interp(np.linspace(0, 2, level_n), 
                         np.linspace(0, 2, current_n), old_h)
            h = h * (level_n / (2 * np.sum(h)))  # Normalize again
        
        current_n = level_n
        
        # Simulated Annealing at this resolution
        T = 0.1
        cooling_rate = 0.99
        current_C5 = evaluate_erdos_solution(h)
        accepted = 0
        iterations = 0
        
        # Local optimizer loop
        while iterations < min(max_iter_per_level, int(remaining_time / 0.1)) and \
              time.time() - start_time < budget_s - 1:
            iterations += 1
            
            # Save current solution
            h_old = h.copy()
            C5_old = current_C5
            
            # Perturb by Gaussian noise
            noise = np.random.normal(0, T, len(h))
            h_new = h + noise
            
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Adjust to satisfy sum constraint
            h_new = h_new * (current_n / (2 * np.sum(h_new)))
            
            # Evaluate new solution
            C5_new = evaluate_erdos_solution(h_new)
            
            # Accept or reject
            delta = C5_new - C5_old
            if delta < 0 or np.random.random() < np.exp(-delta / T):
                h = h_new.copy()
                current_C5 = C5_new
                accepted += 1
                
                # Update best if improved
                if current_C5 < best_C5:
                    best_h = h.copy()
                    best_C5 = current_C5
                    
            # Decrease temperature
            T *= cooling_rate
            
            # Adaptive cooling adjustment
            if iterations % 50 == 0 and T < 0.001:
                T = 0.001
                
        remaining_time = budget_s - (time.time() - start_time)
                
    # Final validation
    final_C5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
