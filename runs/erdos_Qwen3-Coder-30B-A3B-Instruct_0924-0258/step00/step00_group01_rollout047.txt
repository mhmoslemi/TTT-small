import numpy as np
from scipy import signal
import time
import random

def evaluate(h, dx):
    # Compute the overlap integral for all lags
    corr = signal.correlate(h, 1 - h, mode='full')
    # We want the maximum of the first 2*n_points - 1 values
    max_overlap = np.max(corr[:2*len(h) - 1])
    return max_overlap * dx

def project(h, n_points):
    # Ensure the sum is exactly n_points / 2 and all elements are in [0,1]
    s = np.sum(h)
    target = n_points / 2.0
    if abs(s - target) < 1e-10:
        return np.clip(h, 0, 1)
    # Adjust values to meet the sum exactly
    diff = target - s
    if abs(diff) < 1e-10:
        return np.clip(h, 0, 1)
    # Distribute difference evenly
    h = h + diff / len(h)
    return np.clip(h, 0, 1)

def crossover(h1, h2):
    # Average two solutions and project to feasible region
    offspring = (h1 + h2) / 2.0
    return project(offspring, len(h1))

def mutate(h, mutation_rate=0.05, std=0.1):
    # Mutate each element with probability mutation_rate
    mask = np.random.random(len(h)) < mutation_rate
    if np.any(mask):
        noise = np.random.normal(0, std, len(h))
        h = h + noise * mask
    return project(h, len(h))

def tournament_selection(pop, fitnesses, tournament_size=3):
    # Select 2 parents using tournament selection
    indices = np.random.choice(len(pop), tournament_size, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return pop[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial construction if available
    global initial_h_values
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except NameError:
        # Fallback if no initial construction
        n_points = 47
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.6
        h0[1] = 0.4
        h0[2] = 0.7
        h0[3] = 0.3
        h0 = project(h0, n_points)
    
    dx = 2.0 / n_points
    population_size = 20

    # Initialize population
    population = [h0.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project(h, n_points)
        population.append(h)

    # Track best solution and time
    start_time = time.time()
    best_h = h0.copy()
    best_c5 = evaluate(h0, dx)

    generation = 0
    no_improvement = 0
    max_no_improvement = 100

    while time.time() - start_time < budget_s - 1.0 and no_improvement < max_no_improvement:
        # Evaluate population
        fitnesses = []
        for h in population:
            c5 = evaluate(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement = 0
        
        # Selection, crossover, mutation
        new_population = [best_h.copy()]
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            # Crossover
            offspring = crossover(parent1, parent2)
            # Mutation
            offspring = mutate(offspring)
            new_population.append(offspring)

        population = new_population
        generation += 1
        no_improvement += 1

    return (best_h, best_c5, n_points)
