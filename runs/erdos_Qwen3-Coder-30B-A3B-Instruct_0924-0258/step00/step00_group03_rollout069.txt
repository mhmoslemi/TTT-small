def evaluate_erdos_solution(h):
    """Evaluates the Erdős constant C_5 for a given step function h."""
    # Normalize h so that sum(h)*dx = 1, where dx = 2/n_points
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.array(h)
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute the correlation integral
    correlation = signal.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5
