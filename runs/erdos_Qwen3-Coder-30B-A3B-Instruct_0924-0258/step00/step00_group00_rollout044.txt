import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Setup
    n_points = 100  # Start with a reasonable size
    dx = 2.0 / n_points
    max_iter = 20000
    time_budget = budget_s
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Ensure sum is correct
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Timing
    start_time = time.time()
    last_improvement = 0
    
    for i in range(max_iter):
        current_time = time.time()
        if current_time - start_time > time_budget * 0.95:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h)
        
        # Check if better
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = i
            
        # Compute cross correlation
        full_correlation = correlate(h, 1 - h, mode='full')
        cross_correlations = []
        for k in range(2 * n_points - 1):
            cor = full_correlation[k] * dx
            cross_correlations.append(cor)
        
        # Find indices that maximize C5
        max_val = max(cross_correlations)
        max_indices = [k for k in range(2 * n_points - 1) if abs(cross_correlations[k] - max_val) < 1e-8]
        
        # If multiple maxima, bias towards smaller k to avoid oscillation
        max_indices = sorted(max_indices, key=lambda k: k if k < n_points else 2 * n_points - 2 - k)
        
        # Gradient computation
        gradient = np.zeros(n_points)
        for k in max_indices[:5]:  # Use first few for efficiency
            for i in range(n_points - k):
                if i < n_points:
                    gradient[i] += (1 - h[i + k])
                if i + k < n_points:
                    gradient[i + k] -= h[i]
        
        # Line search
        step_sizes = [0.1, 0.25, 0.5]
        best_h_candidate = h.copy()
        best_candidate_c5 = current_c5
        
        for alpha in step_sizes:
            h_candidate = h - alpha * gradient
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            
            if candidate_c5 < best_candidate_c5:
                best_candidate_c5 = candidate_c5
                best_h_candidate = h_candidate.copy()
        
        # Update h to best found
        h = best_h_candidate
        
        # Escape mechanism
        if i - last_improvement > 20:
            noise = 0.01 * np.random.randn(n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = i
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
