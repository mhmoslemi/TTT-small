import numpy as np
import time
from scipy import optimize

def evaluate_c5(h, dx):
    """Evaluate C5 using correlation."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, dx):
    """Project h to satisfy constraints."""
    h = np.clip(h, 0, 1)
    target_sum = 1.0 / dx
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 10000
    n_particles = 30
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    max_stagnation = 100
    
    # Initialize particles
    if 'initial_h_values' in globals():
        particles = np.array([initial_h_values] * n_particles)
    else:
        # Start from random construction respecting sum constraint
        particles = np.random.rand(n_particles, n_points)
        for i in range(n_particles):
            particles[i] = project_h(particles[i], dx)
    
    velocities = np.zeros_like(particles)
    particle_best = np.copy(particles)
    particle_best_c5 = np.array([evaluate_c5(p, dx) for p in particles])
    global_best_idx = np.argmin(particle_best_c5)
    global_best = np.copy(particle_best[global_best_idx])
    global_best_c5 = particle_best_c5[global_best_idx]
    
    iteration = 0
    stagnation_count = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        for i in range(n_particles):
            # Update position
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * np.random.rand() * (particle_best[i] - particles[i]) +
                            social * np.random.rand() * (global_best - particles[i]))
            
            particles[i] += velocities[i]
            particles[i] = project_h(particles[i], dx)
            
            # Evaluate
            current_c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < particle_best_c5[i]:
                particle_best_c5[i] = current_c5
                particle_best[i] = np.copy(particles[i])
                
                # Update global best
                if current_c5 < global_best_c5:
                    global_best = np.copy(particles[i])
                    global_best_c5 = current_c5
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
                
        iteration += 1
        
        # Early exit if converged
        if stagnation_count > max_stagnation:
            break
    
    return (global_best, global_best_c5, n_points)
