import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):

    # Initialize h

    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    # Optimization parameters
    max_iter_without_improvement = 20
    iter_without_improvement = 0
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    start_time = time.time()
    
    # Iteration loop
    while time.time() - start_time < budget_s - 0.1:
        # Compute cross correlations
        full_correlation = np.correlate(h, 1 - h, mode='full')
        cross_correlations = np.array([full_correlation[n_points - 1 + k] * dx for k in range(2 * n_points - 1)])
        
        # Find the maximum correlation
        max_correlation = np.max(cross_correlations)
        max_indices = np.where(np.abs(cross_correlations - max_correlation) < 1e-10)[0]
        
        if len(max_indices) == 0:
            break
        
        # Compute gradient for the maximum correlation contribution
        gradient = np.zeros(n_points)
        for k in max_indices:
            if k < n_points:
                for i in range(n_points - k):
                    gradient[i] += (1 - h[i + k])
                    if i + k < n_points:
                        gradient[i + k] -= h[i]
            else:
                k_rev = 2 * n_points - 2 - k
                for i in range(n_points - k_rev):
                    gradient[i] += (1 - h[i + k_rev])
                    if i + k_rev < n_points:
                        gradient[i + k_rev] -= h[i]
        
        # Line search with step sizes
        step_sizes = [0.1, 0.25, 0.5]
        best_alpha = 0.0
        best_new_c5 = max_correlation
        new_h = h.copy()
        
        for alpha in step_sizes:
            # Try update
            h_candidate = h - alpha * gradient
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            
            # Test new C5 value
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            if candidate_c5 < best_new_c5:
                best_new_c5 = candidate_c5
                best_alpha = alpha
                new_h = h_candidate.copy()
        
        if best_alpha > 0:
            h = new_h.copy()
            if best_new_c5 < best_c5 - 1e-10:
                best_c5 = best_new_c5
                best_h = h.copy()
                iter_without_improvement = 0
            else:
                iter_without_improvement += 1
        else:
            iter_without_improvement += 1
            
        # Perturbation escape
        if iter_without_improvement >= max_iter_without_improvement:
            # Random perturbation
            noise = 0.01 * np.random.randn(n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            iter_without_improvement = 0
            
    return (best_h.tolist(), best_c5, n_points)
