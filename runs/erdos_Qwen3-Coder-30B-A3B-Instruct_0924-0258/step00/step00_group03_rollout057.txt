import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the required sum while keeping values in [0,1]."""
    h = np.array(h, dtype=float)
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.ones_like(h) * (target_sum / len(h))
    # Scale to target sum
    h = h * target_sum / current_sum
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Re-normalize if still outside
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Adjust carefully
        diff = target_sum - current_sum
        # Distribute the difference among elements
        idx = np.argsort(h)
        update = min(diff, 1.0 - h[idx[0]])  # max can increase by
        h[idx[0]] += update
        diff -= update
        for i in range(len(idx)-1, -1, -1):
            if diff == 0:
                break
            j = idx[i]
            if h[j] < 1.0 and diff > 0:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
    return h

def mutate_h(h, rate=0.1):
    """Mutate h by adding noise and normalizing."""
    h = np.array(h)
    n = len(h)
    # Add gaussian noise
    noise = np.random.normal(0, rate * 0.1, n)
    h = h + noise
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize
    h = normalize_h(h, n / 2.0)
    return h

def crossover_h(h1, h2):
    """Perform single-point crossover between two h vectors."""
    n = len(h1)
    point = np.random.randint(1, n)
    h_new = np.concatenate([h1[:point], h2[point:]])
    return h_new

def tournament_select(pop, fitnesses, t_size=5):
    """Select an individual via tournament selection."""
    idxs = np.random.choice(len(pop), t_size)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return pop[best_idx]

    for _ in range(30):

        h = normalize_h(h, n_points / 2)
        population.append(h)
    
    # Set dx
    dx = 2.0 / n_points
    
    # Main loop
    best_h = None
    best_c5 = float('inf')
    generation = 0

            fitnesses.append(f)
        
        # Update best
        min_fit = min(fitnesses)

            best_h = population[np.argmin(fitnesses)].copy()
        
        # Next generation
