import numpy as np
from scipy import signal
import time
import random

def evaluate_h(h, dx):
    # Compute overlap integral for all lags
    correlation = signal.correlate(h, 1 - h, mode='full')
    # Select the relevant part (maximum lag is n_points-1)
    max_overlap = np.max(correlation[len(h)-1:])
    return max_overlap * dx

def project_feasible(h, n_points):
    # Ensure h sums to n_points / 2 and stays in [0, 1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-10:
        # Adjust by scaling and clipping
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        # Final adjustment to make sure we hit the target
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-10:
            # Add the difference evenly by increasing some elements
            # Find the smallest elements to increase
            indices = np.argsort(h)
            for i in range(int(np.ceil(abs(diff)))):
                idx = indices[i % len(indices)]
                if h[idx] < 1:
                    h[idx] += min(1 - h[idx], abs(diff))
                    diff -= min(1 - h[idx], abs(diff))
                    if abs(diff) < 1e-10:
                        break
    return h

def initialize_population(n_points, pop_size, initial_h):
    population = []
    population.append(initial_h.copy())
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, tournament_size=3):
    selected_indices = []
    for _ in range(len(population)):
        # Tournaments
        tournament_indices = random.sample(range(len(population)), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner = tournament_indices[np.argmin(tournament_fitnesses)]
        selected_indices.append(winner)
    return [population[i] for i in selected_indices]

def crossover(parent1, parent2):
    # Single point crossover with averaging, projection to maintain constraints
    n_points = len(parent1)
    point = random.randint(1, n_points-1)
    child1 = (parent1[:point] + parent2[:point]) / 2.0
    child2 = (parent1[point:] + parent2[point:]) / 2.0
    child1 = np.concatenate([child1, child2])
    # Project to feasible
    child1 = project_feasible(child1, n_points)
    return child1

def mutate(h, mutation_rate=0.05, sigma=0.1):
    # Add Gaussian noise
    h = h + np.random.normal(0, sigma, len(h)) * (np.random.rand(len(h)) < mutation_rate)
    h = project_feasible(h, len(h))
    return h
