def evaluate_erdos_solution(h):
    # This is a placeholder - in the actual environment this function exists
    # and computes the maximum correlation as described.
    # For now we'll simulate what it would compute based on problem description.
    n_points = len(h)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to have correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.full_like(h, 1.0 / n_points)
    return h

        h_init = project_to_feasible(h_init, n_points)
    
    # PSO parameters
    n_particles = 30
    omega = 0.9
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.5

    particles = []
    velocities = []
    pbest = []
    pbest_values = []

    particles.append(h_init.copy())

    for _ in range(n_particles - 1):
        perturb = np.random.normal(0, 0.05, n_points)

        particles.append(h.copy())

        pbest_values.append(evaluate_erdos_solution(h))
    
    # Initialize velocities
    for _ in range(n_particles):
        velocities.append(np.random.uniform(-0.3, 0.3, n_points))
    
    # Global best
    best_idx = np.argmin(pbest_values)

            # Get current particle
            h = particles[i]
            
            # Evaluate current solution
            current_value = evaluate_erdos_solution(h)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = h.copy()
                pbest_values[i] = current_value
            
            # Update global best
            if current_value < gbest_value:
                gbest = h.copy()
                gbest_value = current_value

        for i in range(n_particles):

            r1 = np.random.random(n_points)
