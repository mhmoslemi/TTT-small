import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode="full")
    # Maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_constraint(h, target_sum=1.0):
    """Adjust h to satisfy the sum constraint"""
    n = len(h)
    current_sum = np.sum(h)
    if current_sum == 0:
        h = np.ones_like(h) * target_sum / n
    else:
        h = h * target_sum / current_sum
    return h

def perturb_h(h, temperature, sigma_factor=1.0):
    """Perturb the function h with Gaussian noise"""
    n = len(h)
    sigma = temperature * sigma_factor
    noise = np.random.normal(0, sigma, n)
    h_new = h + noise
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Project back to meet constraint
    h_new = project_to_constraint(h_new)
    return h_new
