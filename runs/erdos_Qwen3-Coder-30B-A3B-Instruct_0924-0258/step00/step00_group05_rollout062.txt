import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h):
    h = np.array(h)
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for different lags
    correlation = correlate(h, 1-h, mode="full")
    # Extract the maximum value (over all lags)
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def project_to_constraint(h, n_points):
    """Project h to satisfy sum(h) = n_points/2"""
    h = np.array(h)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-10:
        # Scale to target sum
        scale = target_sum / current_sum
        h = h * scale
    return h

def perturb_h(h, temperature):
    """Perturb h with Gaussian noise scaled by temperature"""
    h = np.array(h)
    n_points = len(h)
    sigma = temperature
    noise = np.random.normal(0, sigma, n_points)
    h_new = h + noise
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Renormalize
    h_new = project_to_constraint(h_new, n_points)
    return h_new

def cooling_schedule(iteration, initial_temp, cooling_rate):
    """Adaptive cooling schedule"""
    return initial_temp * (cooling_rate ** iteration)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points_levels = [10, 20, 40, 70]
    # Initialize solution
    if 'initial_h_values' in globals():
        best_h = np.array(initial_h_values)
        n_points = len(best_h)
    else:
        n_points = 10
        best_h = np.random.rand(n_points)
        best_h = project_to_constraint(best_h, n_points)
    best_c5 = evaluate_c5(best_h)
    
    for i_level, n_points in enumerate(n_points_levels):
        if i_level > 0:
            # Interpolate from previous level
            old_h = best_h
            old_n = len(old_h)
            new_h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, old_n), old_h)
            best_h = new_h
            # Project to sum constraint
            best_h = project_to_constraint(best_h, n_points)
            
        # Simulated Annealing
        temp = 0.1
        cooling_rate = 0.99
        iterations = 500
        for iteration in range(iterations):
            current_time = time.time()
            if current_time - start_time > budget_s * 0.95:
                break
            new_h = perturb_h(best_h, temp)
            new_c5 = evaluate_c5(new_h)
            delta = new_c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                best_h = new_h
                best_c5 = new_c5
            temp = cooling_schedule(iteration, 0.1, cooling_rate)
            
    # Final validation
    best_h = project_to_constraint(best_h, len(best_h))
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
