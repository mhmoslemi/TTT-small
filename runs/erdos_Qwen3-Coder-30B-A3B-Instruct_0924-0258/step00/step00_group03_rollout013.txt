import numpy as np
import time
from scipy import optimize

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute correlation for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract the valid overlaps (excluding boundary effects)
    valid_corr = corr[n-1:2*n-1]
    # Maximize over all shifts
    c5 = np.max(valid_corr) * dx
    return c5

def project(h, n_points):
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize sum to n_points / 2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def mutate(h, sigma):
    h_new = h + np.random.normal(0, sigma, len(h))
    return project(h_new, len(h))

def crossover(h1, h2, alpha=0.6):
    beta = np.random.uniform(alpha, 1-alpha)
    h_new = beta * h1 + (1-beta) * h2
    return project(h_new, len(h1))

def local_refinement(h, max_iter=50, lr=0.01):
    h = h.copy()
    prev_f = evaluate_erdos_solution(h)
    for _ in range(max_iter):
        # Simple gradient estimation via finite difference
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project(h_plus, len(h))
            f_plus = evaluate_erdos_solution(h_plus)
            grad[i] = (f_plus - prev_f) / eps
        h -= lr * grad
        h = project(h, len(h))
        new_f = evaluate_erdos_solution(h)
        if new_f >= prev_f:
            break
        prev_f = new_f
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        n_points = len(h_original)
    else:
        # Create a simple initial vector
        n_points = 70
        h_original = np.ones(n_points) * 0.5
        h_original = project(h_original, n_points)
    
    # Population of solutions
    population_size = 100
    population = []
    population.append(h_original)
    for i in range(population_size - 1):
        h_m = mutate(h_original, 0.1)
        population.append(h_m)
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(h) for h in population]
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Main search loop
    max_gens = 1000
    for gen in range(max_gens):
        if time.time() - start_time > budget_s - 0.5:
            break
            
        # Selection: Top 40% + 10% random
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:int(0.4 * population_size)]
        rand_indices = np.random.choice(population_size, int(0.1 * population_size), replace=False)
        selected_indices = np.concatenate([top_indices, rand_indices])
        
        # Generate offspring via crossover and mutation
        offspring = []
        for _ in range(30):  # Number of offspring
            idx1, idx2 = np.random.choice(selected_indices, 2, replace=False)
            child = crossover(population[idx1], population[idx2], 0.6)
            # Mutate with adaptive sigma
            sigma = 0.1 * np.exp(-gen / 100)
            child = mutate(child, sigma)
            offspring.append(child)
        
        # Local refinement on top few
        top_refine = sorted_indices[:int(0.05 * population_size)]
        for idx in top_refine:
            h_refined = local_refinement(population[idx].copy(), max_iter=5, lr=0.01)
            offspring.append(h_refined)
        
        # Evaluate offspring
        offspring_fitness = [evaluate_erdos_solution(h) for h in offspring]
        
        # Replace worst half of population
        worst_indices = sorted_indices[int(0.5 * population_size):]
        for i, idx in enumerate(worst_indices):
            if i < len(offspring):
                population[idx] = offspring[i]
                fitness[idx] = offspring_fitness[i]
        
        # Update best
        new_best_idx = np.argmin(fitness)
        if fitness[new_best_idx] < best_c5:
            best_c5 = fitness[new_best_idx]
            best_h = population[new_best_idx].copy()
    
    return (best_h.tolist(), best_c5, n_points)
