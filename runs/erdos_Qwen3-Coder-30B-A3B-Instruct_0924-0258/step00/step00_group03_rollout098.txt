import numpy as np
from scipy import signal
import time
from math import sqrt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h / np.sum(h) * target_sum
    else:
        # Initialize with a feasible random vector
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # Keep track of the best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    max_iter = 10000
    alpha = 0.05
    decay = 0.9995
    min_alpha = 1e-6
    
    # Precompute convolution indices
    indices = np.arange(n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts
        overlap = np.correlate(h, 1 - h, mode='full')
        overlap = overlap * dx  # Scale by dx
        
        # Find the maximum overlap
        max_idx = np.argmax(overlap)
        max_val = overlap[max_idx]
        
        # Compute gradient at max overlap position
        # For convolution h * (1 - h), gradient w.r.t. h[k] at max_idx is:
        # - This is effectively the derivative of the overlap with respect to h
        # - At the max overlap, the gradient of the maximum extreme is used
        # - Simpler numerical approach: finite difference
        h_grad = np.zeros_like(h)
        
        # FDM approach for gradient of the maximum
        eps = 1e-6
        for i in range(n_points):
            h_eps = h.copy()
            h_eps[i] += eps
            h_eps = np.clip(h_eps, 0, 1)
            overlap_eps = np.correlate(h_eps, 1 - h_eps, mode='full')
            overlap_eps = overlap_eps * dx
            grad_est = (overlap_eps[max_idx] - max_val) / eps
            h_grad[i] = grad_est
        
        # Update step
        h_new = h - alpha * h_grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * target_sum
        
        h = h_new
        
        # Evaluate current solution
        c5 = evaluate_c5(h, dx)
        
        # Update best if better
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Adaptive learning rate
        alpha = max(min_alpha, alpha * decay)
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    """Evaluate the C5 bound for given h values"""
    # Compute the overlap integral
    overlap = np.correlate(h, 1 - h, mode='full')
    # Return the maximum overlap
    return np.max(overlap) * dx
