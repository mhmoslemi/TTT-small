import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant for a given step function."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all cross-correlations
    corr = signal.correlate(h, 1 - h, mode='full')
    # The valid range of lags is [-n+1, n-1], so we extract the middle part
    mid = len(corr) // 2
    # Extract cross-correlation for lags 0 to n-1 (positive lags)
    corr_pos = corr[mid:mid + n_points]
    max_corr = np.max(corr_pos)
    return max_corr * dx
