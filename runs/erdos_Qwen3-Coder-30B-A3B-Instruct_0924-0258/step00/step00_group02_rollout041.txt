import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for a given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral for all shifts
    conv = signal.correlate(h, 1 - h, mode='full')
    # The maximum value in the full convolution
    max_value = np.max(conv)
    return max_value * dx

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones_like(h) * (n_points / 2) / n_points
    return h

    # Initialize swarm
    particles = []
    velocities = []
    
    # First particle is initial solution
    particles.append(initial_h_values.copy())
    velocities.append(np.zeros(n_points))
    
    # Other particles are perturbed versions
    for _ in range(n_particles - 1):
        perturbed = initial_h_values + np.random.uniform(-perturbation_scale, perturbation_scale, n_points) * initial_h_values
        particles.append(project_feasible(perturbed, n_points))
        velocities.append(np.random.uniform(-velocity_bounds, velocity_bounds, n_points))
    
    # Evaluate particles
    particle_fitness = [evaluate_erdos_solution(p) for p in particles]
    pbest_positions = [p.copy() for p in particles]
    pbest_fitness = particle_fitness.copy()
    gbest_idx = np.argmin(pbest_fitness)
    gbest_fitness = pbest_fitness[gbest_idx]
    gbest_position = pbest_positions[gbest_idx].copy()
    
    # Timing
    start_time = time.time()
    
    # Main PSO loop
    omega = inertia_start
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity (with bounds)
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                            c1 * r1 * (pbest_positions[i] - particles[i]) +
                            c2 * r2 * (gbest_position - particles[i]))
            velocities[i] = np.clip(velocities[i], -velocity_bounds, velocity_bounds)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project
            particles[i] = project_feasible(particles[i], n_points)
            
            # Evaluate
            fitness = evaluate_erdos_solution(particles[i])
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest_positions[i] = particles[i].copy()
                
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_position = particles[i].copy()
        
        # Update inertia weight
        omega *= 0.99
    
    return (gbest_position, gbest_fitness, n_points)
