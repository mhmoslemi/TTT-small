import numpy as np

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts k
    correlation = np.correlate(h, 1 - h, mode='full')
    # Shift the correlation to focus on positive lags (k >= 0)
    # The overlap integral for shift k is correlation[n-1+k]
    # We compute the maximum over all possible shifts
    overlaps = []
    for k in range(-n+1, n):
        # Normalize the correlation by dx
        overlap_k = correlation[n-1+k] * dx
        overlaps.append(overlap_k)
    # Return the maximum overlap (the upper bound C_5)
    return max(overlaps)

def project(h, n_points):
    """Project h to satisfy integral constraint and bounds."""
    # Clamp values to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust to satisfy sum(h) * dx = 1 => sum(h) = n_points / 2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    # Simple proportional adjustment - keeping the relative ratios
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def mutate(h, sigma):
    """Add Gaussian noise to h and project."""
    h_new = h + np.random.normal(0, sigma, len(h))
    return project(h_new, len(h))

def crossover(h1, h2, alpha=0.6):
    """Blend crossover between two solutions."""
    # alpha controls the blend range
    beta = np.random.uniform(alpha, 1 - alpha)
    h_child = beta * h1 + (1 - beta) * h2
    return project(h_child, len(h1))

def compute_gradient(h):
    """Compute approximate gradient of the maximum overlap function."""
    # This is an approximation by finite differencing
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    eps = 1e-4
    base_val = evaluate_erdos_solution(h)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project(h_plus, n)
        val_plus = evaluate_erdos_solution(h_plus)
        grad[i] = (val_plus - base_val) / eps
    return grad
