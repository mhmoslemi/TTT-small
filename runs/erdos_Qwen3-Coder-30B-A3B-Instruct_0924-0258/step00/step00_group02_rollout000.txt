import numpy as np
from scipy import optimize
import time
import math

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for the given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation using full mode to get all lags
    corr = np.correlate(h, 1 - h, mode="full")
    # Corresponding lag indices
    lags = np.arange(-n_points + 1, n_points)
    # Maximum over all lags
    max_val = np.max(corr)
    return max_val * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert len(h) == n_points
    else:
        # Create default construction with even spacing
        h = np.ones(n_points) * 0.5
    # Ensure sum is correct
    h = project_to_feasible(h, n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Gradient based optimization with perturbations
    lr = 0.01
    decay_factor = 0.9995
    momentum = 0.95
    velocity = np.zeros_like(h)
    t = 0
    
    while time.time() - start_time < budget_s - 1:
        t += 1
        # Compute correlation for all lags
        corr = np.correlate(h, 1 - h, mode="full")
        lag_indices = np.arange(-n_points + 1, n_points)
        # Find index of maximum correlation
        max_idx = np.argmax(corr)
        max_lag = lag_indices[max_idx]
        
        # Compute gradient with respect to max_lag
        # This is simplified; forward differences would be more precise
        # For lag k, gradient wrt h[i] is (1 - h[i + k])
        # So we construct gradient for the current max lag
        grad_h = np.zeros_like(h)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad_h[i] += (1 - h[j])
        
        # Update with momentum and learning rate
        velocity = momentum * velocity - lr * grad_h
        h = h + velocity
        # Project to feasible region
        h = project_to_feasible(h, n_points)
        
        # Occasionally add small random perturbations
        if t % 10 == 0:
            noise = np.random.uniform(-0.05, 0.05, n_points)
            h = h + noise
            h = project_to_feasible(h, n_points)
        
        # Update best if better
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Decrease learning rate over time
        lr = max(lr * decay_factor, 1e-6)
        
    return (best_h.tolist(), best_c5, n_points)
