import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

def evaluate_overlap(h):
    """Evaluate C_5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Take maximum over all shifts
    max_corr = np.max(corr)
    return max_corr * dx

def project_and_rescale(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def crossover_h(s1, s2):
    """Crossover two sequences at 2-3 random points"""
    n = len(s1)
    if n < 4:
        return s1.copy(), s2.copy()
    num_cuts = np.random.randint(2, 4)
    cuts = sorted(np.random.choice(n - 1, num_cuts, replace=False))
    cuts = [0] + cuts + [n]
    new_s1, new_s2 = np.zeros_like(s1), np.zeros_like(s2)
    for i in range(len(cuts) - 1):
        start, end = cuts[i], cuts[i+1]
        if i % 2 == 0:
            new_s1[start:end] = s1[start:end]
            new_s2[start:end] = s2[start:end]
        else:
            new_s1[start:end] = s2[start:end]
            new_s2[start:end] = s1[start:end]
    return new_s1, new_s2

def mutate_h(h, mutation_rate=0.05):
    """Mutate h by adding small noise and reprojecting"""
    n = len(h)
    indices = np.random.choice(n, int(n * mutation_rate), replace=False)
    h = h.copy()
    h[indices] += np.random.normal(0, 0.1, len(indices))
    return project_and_rescale(h, n)

def optimize_candidate(h, n_points, maxiter=100):
    """Run gradient-based optimization on candidate"""
    # Reconstruct bounds
    bounds = [(0, 1) for _ in range(n_points)]
    def obj(h_vec):
        return evaluate_overlap(h_vec)
    
    # Convert to 1d array for minimize
    h_initial = h.copy()
    # Use L-BFGS-B
    result = minimize(obj, h_initial, method='L-BFGS-B', bounds=bounds, options={'maxiter': maxiter})
    return project_and_rescale(result.x, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialization
    n_points = 42  # Start with initial construction size
    pop_size = 50  # Population size
    mutation_rate = 0.05
    
    # Create initial population
    population = []
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = project_and_rescale(base_h, n_points)
        population.append(base_h.copy())
        pop_size -= 1
    else:
        base_h = np.random.rand(n_points)
        base_h = project_and_rescale(base_h, n_points)
        population.append(base_h.copy())
    
    # Fill population
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = project_and_rescale(h, n_points)
        population.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    gen = 0
    stagnation = 0
    
    while time.time() - start_time < budget_s - 10:
        gen += 1
        # Evaluate all candidates
        fitness = []
        for h in population:
            c5 = evaluate_overlap(h)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Check for stagnation
        if gen % 10 == 0 and gen > 10:
            # Update mutation rate if no progress
            if gen > 50 and stagnation > 10:
                mutation_rate = 0.1
            if gen > 20:
                stagnation = 0
        
        # Selection - top 20%
        top_k = max(1, int(0.2 * len(population)))
        indices = np.argsort(fitness)[:top_k]
        parents = [population[i].copy() for i in indices]
        
        # Create offspring via crossover and mutation
        new_population = parents.copy()
        
        # Crossover
        for i in range(0, len(parents) - 1, 2):
            p1, p2 = parents[i], parents[i+1]
            c1, c2 = crossover_h(p1, p2)
            new_population.extend([c1, c2])
        
        # Mutation
        num_mutate = min(len(new_population) // 2, len(new_population))
        for i in range(num_mutate):
            h_index = np.random.randint(len(new_population))
            new_population[h_index] = mutate_h(new_population[h_index], mutation_rate)
        
        # Greedy optimization
        top_candidates = min(5, len(parents))
        for i in range(top_candidates):
            candidate_idx = np.random.randint(len(new_population))
            new_population[candidate_idx] = optimize_candidate(new_population[candidate_idx], n_points)
        
        # Keep population size consistent
        if len(new_population) > pop_size:
            # Re-evaluate and keep best pop_size
            new_fitness = []
            for h in new_population:
                c5 = evaluate_overlap(h)
                new_fitness.append(c5)
            top_indices = np.argsort(new_fitness)[:pop_size]
            population = [new_population[i].copy() for i in top_indices]
        else:
            population = new_population
    
    # Final optimization of best solution
    if best_h is not None:
        best_h = optimize_candidate(best_h, n_points, maxiter=50)
        final_c5 = evaluate_overlap(best_h)
        best_c5 = min(final_c5, best_c5)
    
    return (best_h.tolist(), best_c5, n_points)
