    h = np.clip(h, 0, 1)

        # Use numpy correlate with full mode
        corrs = signal.correlate(h_vals, 1 - h_vals, mode='full')
        # Only take the part that corresponds to valid shifts
        valid_corrs = corrs[len(h_vals)-1 : 2*len(h_vals)-1]
        return np.max(valid_corrs) * dx
    
    def project_to_constraints(h_vals):
        # Project to sum = n_points / 2 and [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        current_sum = np.sum(h_vals)
        if abs(current_sum - n_points / 2) > 1e-10:
            h_vals = h_vals * (n_points / 2) / current_sum
        return h_vals
    
    best_h = h.copy()
    best_C5 = compute_C5(h)

            best_C5 = C5
            best_h = h.copy()
            
        # Identify worst lag

        valid_corrs = corrs[len(h)-1 : 2*len(h)-1]
        k0 = np.argmax(valid_corrs)

        # Update step
        h = h - alpha * grad
