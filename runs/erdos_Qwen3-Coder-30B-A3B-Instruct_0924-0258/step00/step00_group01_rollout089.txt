    decay = 0.95

    # Gradient descent loop
    for i in range(max_iter):

        correlation = np.correlate(h, 1 - h, mode="full")
        k0 = np.argmax(correlation)
        C5 = correlation[k0] * (2.0 / n_points)
        
        # Update best if improved
        if C5 < best_c5:
            best_c5 = C5
            best_h = h.copy()
            
        # Compute gradient for the worst overlap
        grad = np.zeros(n_points)
        k_offset = k0 - (n_points - 1)  # Convert full correlation index to shift
        
        # Gradient components for shift k0
        for j in range(n_points):
            if 0 <= j + k_offset < n_points:
                grad[j] += 1 - h[j + k_offset]
            if 0 <= j - k_offset < n_points and k_offset != 0:
                grad[j] -= h[j - k_offset]
        
        # Update h
        h_new = h - alpha * grad
        
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust sum to match constraint
        h_new *= (n_points / 2) / np.sum(h_new)

    final_c5 = correlation[np.argmax(correlation)] * (2.0 / n_points)
