import numpy as np
from scipy import signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Fallback initial construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
    
    n_points = len(h)
    dx = 2.0 / n_points
    max_iter = 1000
    lr = 0.01
    decay = 0.95
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Projection function using CVXPY
    def project_h(h_vec):
        # Define CVXPY variables
        x = cp.Variable(n_points)
        # Constraints
        constraints = [
            x >= 0,
            x <= 1,
            cp.sum(x) == n_points / 2.0
        ]
        # Optimization problem
        prob = cp.Problem(cp.Minimize(cp.sum_squares(x - h_vec)), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                return np.array(x.value).reshape(-1)
            else:
                return h_vec
        except:
            return h_vec
    
    for i in range(max_iter):
        # Compute C5
        conv = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Check time budget
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Find k where maximum occurs
        k_max = np.argmax(conv)
        # In full convolution, k_max corresponds to shift k = k_max - (n-1)
        shift_k = k_max - (n_points - 1)
        
        # Compute gradient (approximate by finite difference or direct derivative)
        # Gradient w.r.t. h at the k_max lag
        grad = np.zeros(n_points)
        # Convolution gradient is the shifted values of (1-h)
        # Shift by -k - fill with 0s
        padded = np.concatenate([np.zeros(n_points), h[::-1], np.zeros(n_points)])
        # For the continuous approximation, the derivative of the convolution term with resp to h_j is the shifted 1-h values
        # We compute it as a sum over valid indices
        # Let's do this carefully based on the convolution formula manually derived
        # Convolution at k is \sum_j h[j] * (1 - h[j+k]) for valid j,k pairs
        # So the derivative of this term with respect to h[i] is d/dh[i](\sum_j h[j] * (1 - h[j+k]))  
        # If i != j, derivative is 0
        # If i == j, it is (1 - h[j+k])  
        # But we also need to consider that shifts modify this
        # Here we compute approximate gradient for the exact max index k_max
        
        # Simpler approach: 
        # For each position i, compute the gradient contribution from the maximum convolution term
        # At lag k_max, we have int h(x) (1 - h(x+k_max)) dx 
        # Gradient of that w.r.t. h at position j is (1 - h[j + k_max])
        # But this is relative to the full correlation computation
        # For simplicity of the update, estimate gradient for the k_max lag  
        # Gradient: for each j, contribution to max convolution term 
        # This is not precisely the right gradient but a workable approximation
        # Reconstruct k_max lag contribution to approximate gradient
        
        # Compute convolution of h with (1-h) using full mode and find the max
        # Focus on exact gradient estimate
        # Let's take a simpler gradient approximation by considering shift contributions and no model induced errors:
        # The gradient is based on finite differences or direct derivative in continuous approximation.
        # For practicality let's consider:
        # Let J(h) = max_k int h(x) (1 - h(x+k)) dx over all k
        # Gradient = dJ/dh 
        # We compute it as derivative of the specific k_max term
        # Gradient is 1-h shifted by -k_max at each position for that k_max term
        # But interpreted as the direction where increasing h[i] will increase the k_max term
        
        # Approximator: gradient of h[i] is 1-h[i-k_max] - this is conceptually the contribution
        # Which index is relevant depends on what's the contribution of h[i] to the gradient
        # (assuming h[i] contributes to \int h(x) (1 - h(x+k)) dx)
        # Argument per point i: 
        # Contribution is (1 - h[i + k_max])
        # But this is not *entirely* correct due to discretization
        # So in discrete approximation: near k_max 
        # Gradient for element i is: 1 - h[(i - k_max) % n]
        # where it's circular as shifter isn't necessarily linear
        # A correct derivation for lag k >=0 
        # resulting term is sum_j h[j] (1 - h[j+k])
        # derivative w.r.t h[i] = -(1 - h[i+k])
        
        # So for k_max Lag:
        for j in range(n_points):
            if shift_k == 0:
                grad[j] = 0
            else:
                # h[i] makes contribution to term with lag k = shift_k  
                # Lag k means poition j receives (1 - h[j+k])
                # Gradient contributes - (1 - h[j - shift_k]) to h[j]
                idx = (j - shift_k) % n_points
                grad[j] -= (1 - h[idx])
        
        # Update h with gradient
        h = h - lr * grad
        h = project_h(h)
        
        # Decrease learning rate every 100 iterations
        if (i + 1) % 100 == 0:
            lr *= decay
            
    # Final validation
    conv_final = signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(conv_final) * dx
    
    return (best_h.tolist(), float(final_c5), n_points)
