import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_low = 10
    n_high = 70
    max_iter_per_level = 500
    initial_T = 0.1
    cooling_rate = 0.99
    max_levels = int(np.log2(n_high / n_low))
    
    # Initialize with initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h / np.sum(h) * (n_low / 2)
    else:
        h = np.random.rand(n_low)
        h = h / np.sum(h) * (n_low / 2)
    n = len(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    T = initial_T
    elapsed = time.time() - start_time
    
    # Multi-resolution approach
    for level in range(max_levels):
        if elapsed >= budget_s * 0.98:
            break
        # Increase resolution
        n_prev = n
        n = min(n * 2, n_high)
        
        # Interpolate from previous resolution
        h_new = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, n_prev), h)
        h = h_new
        
        # Normalize to sum = n/2
        h = h / np.sum(h) * (n / 2)
        
        # Optimizer loop at current resolution
        for i in range(max_iter_per_level):
            elapsed = time.time() - start_time
            if elapsed >= budget_s * 0.95:
                break
                
            # Make a perturbation
            h_proposal = h + np.random.normal(0, T, n)
            h_proposal = np.clip(h_proposal, 0, 1)
            
            # Adjust to sum constraint
            sum_ratio = (n / 2) / np.sum(h_proposal)
            h_proposal = h_proposal * sum_ratio
            
            # Compute C5
            c5_proposal = compute_c5(h_proposal)
            
            # Accept or reject
            delta_c5 = c5_proposal - current_c5
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
                h = h_proposal
                current_c5 = c5_proposal
                
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
            
            # Reduce temperature
            T *= cooling_rate
            
            if T < 1e-6:
                T = 1e-6
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))

def compute_c5(h):
    """Compute C5 using convolution"""
    # Compute correlation
    corr = signal.correlate(h, 1 - h, mode='full')
    # Convolution result is of length 2*n - 1
    # We only want the relevant portions
    dx = 2.0 / len(h)
    return np.max(corr) * dx
