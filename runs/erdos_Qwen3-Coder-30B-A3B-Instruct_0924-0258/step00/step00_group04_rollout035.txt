import numpy as np
import time
from scipy.optimize import minimize
from math import exp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42  # Use the provided initial construction size
    dx = 2.0 / n_points
    max_iter = 600
    initial_temp = 1.0
    cooling_factor = 0.99
    min_temp = 0.01
    patience = 50
    gradient_scale = 0.1
    noise_std = 0.05
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Helper functions
    def evaluate_c5(h_vec):
        # Compute correlation of h with (1-h)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value scaled by dx
        return np.max(corr) * dx
    
    def project(h_vec):
        # Clamp values to [0,1]
        h_proj = np.clip(h_vec, 0, 1)
        # Rescale to meet constraint
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        return h_proj
    
    # Initial evaluation
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    temp = initial_temp
    patience_count = 0
    current_iter = 0
    
    while time.time() - start_time < budget_s and current_iter < max_iter:
        current_iter += 1
        
        # Compute gradient approximation using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project(h_plus)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project(h_minus)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Gradient-guided mutation
        noise = np.random.normal(0, noise_std, n_points)
        h_candidate = h - temp * gradient_scale * grad + noise
        
        # Project to feasible region
        h_candidate = project(h_candidate)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_candidate)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.random() < exp((current_c5 - new_c5) / temp):
            h = h_candidate
            current_c5 = new_c5
            
            # Update best if improved
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                patience_count = 0
            else:
                patience_count += 1
        else:
            patience_count += 1
            
        # Temperature update
        temp = max(temp * cooling_factor, min_temp)
        
        # Warm restart if no improvement for patience steps
        if patience_count >= patience:
            temp = initial_temp
            patience_count = 0
            
    return (best_h.tolist(), best_c5, n_points)
