    correlation = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_h(h):

    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to have correct sum
    current_sum = np.sum(h) * dx
    target_sum = 1.0
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Add the difference uniformly to all points
        h += diff / n_points
        # Ensure bounds after adjustment
        h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, k0):
    global n_points, dx
    grad = np.zeros_like(h)
    if k0 >= 0:
        # For positive k0: h[i] contributes to overlap with h[i-k0] in 1-h
        grad[k0:] += (1 - h[:-k0]) * dx
        # And to overlap with h[i] in h itself
        grad[:n_points-k0] -= h[k0:] * dx
    else:
        # For negative k0: h[i] contributes to overlap with h[i+k0] in 1-h
        k0 = -k0
        grad[:n_points-k0] += (1 - h[k0:]) * dx
        grad[k0:] -= h[:-k0] * dx
    return grad

def compute_active_shift(h):

    correlation = np.correlate(h, 1 - h, mode='full')
    # Take maximum, along with its shift
    max_idx = np.argmax(correlation)
    k0 = max_idx - (n_points - 1)
    return k0, correlation[max_idx]
