import numpy as np
from scipy import optimize
import cvxpy as cp
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the overlap integral for given h_values."""
    h = np.array(h_values)
    dx = 2.0 / len(h)
    # Compute all overlaps using convolution
    overlaps = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is our C5
    return np.max(overlaps) * dx

def solve_feasible(h0, C5, n_points):
    """Check if there exists h with given C5 using convex optimization."""
    # Setup the CVXPY problem
    h = cp.Variable(n_points)
    
    # Constraints: 0 <= h[i] <= 1
    constraints = [h >= 0, h <= 1]
    
    # Constraint: integral = 1 
    constraints.append(cp.sum(h) == n_points / 2)
    
    # For each shift k = 0..n_points-1, we compute:
    # sum(h[i] * (1 - h[(i+k)%n_points])) <= C5 * dx
    dx = 2.0 / n_points
    for k in range(n_points):
        # This is equivalent to correlating h with (1-h)
        # Compute sum_{i} h[i] * (1 - h[i+k]) 
        contrib = 0
        for i in range(n_points):
            j = (i + k) % n_points
            contrib += h[i] * (1 - h[j])
        constraints.append(contrib <= C5 / dx)
    
    # No objective function needed (feasibility check)
    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
        return h.value is not None and not np.isnan(h.value).any()
    except:
        return False

def binary_search_C5(h0, n_points, low=0.0, high=0.38085, tolerance=1e-4):
    """Binary search for the smallest valid C5."""
    while high - low > tolerance:
        mid = (low + high) / 2.0
        if solve_feasible(h0, mid, n_points):
            high = mid
        else:
            low = mid
    return high

            # This is the overlap to minimize
