import numpy as np
from scipy import correlate
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all k using full correlation
    corr = correlate(h, 1 - h, mode='full')
    # The correlation has length 2*n-1, corresponding to k from -(n-1) to (n-1)
    # But we only care about k from 0 to n-1, so we just take the second half
    # The max is over all valid k values
    max_val = np.max(corr[n-1:]) * dx
    return max_val

def project_to_feasible(h, n_points):
    """Project h to satisfy 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to fix sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.full_like(h, 1.0 / n_points)
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

        n_points = 45
        h = np.full(n_points, 0.5)
    else:

        n_points = len(h)
    
    # Particle Swarm Optimization parameters
    n_particles = 30
    max_iter = 1000
    omega = 0.9
    c1 = 2.0
    c2 = 2.0
    v_max = 0.5
    
    # Initialize particles
    particles = []
    velocities = []
    
    # First particle is initial solution
    particles.append(h.copy())
    velocities.append(np.zeros_like(h))
    
    # Random particles
    for _ in range(n_particles - 1):
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        h_perturb = h + perturb
        h_perturb = project_to_feasible(h_perturb, n_points)
        particles.append(h_perturb.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Evaluate initial particles
    fitness = [evaluate_erdos_solution(p) for p in particles]
    
    # Track personal bests
    pBest = particles.copy()
    pBest_fitness = fitness.copy()
    
    # Find global best
    best_idx = np.argmin(fitness)
    gBest = particles[best_idx].copy()
    gBest_fitness = fitness[best_idx]
    
    best_h = gBest.copy()
    best_c5 = gBest_fitness
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time >= budget_s - 2:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            v = omega * velocities[i] + c1 * r1 * (pBest[i] - particles[i]) + \
                c2 * r2 * (gBest - particles[i])
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            velocities[i] = v
            
            # Update position
            particles[i] += v
            
            # Apply constraints
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate new fitness
            new_fitness = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if new_fitness < pBest_fitness[i]:
                pBest[i] = particles[i].copy()
                pBest_fitness[i] = new_fitness
                
                # Update global best if needed
                if new_fitness < gBest_fitness:
                    gBest = particles[i].copy()
                    gBest_fitness = new_fitness
                    
                    # Update best solution found
                    best_h = gBest.copy()
                    best_c5 = gBest_fitness
        
        # Update inertia weight
        omega *= 0.99
        
        # Re-evaluate the global best with latest dynamics
        current_gBest_fitness = evaluate_erdos_solution(gBest)
        if current_gBest_fitness < gBest_fitness:
            gBest_fitness = current_gBest_fitness
    
    # Final evaluation of best solution
    eval_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), eval_c5, n_points)
