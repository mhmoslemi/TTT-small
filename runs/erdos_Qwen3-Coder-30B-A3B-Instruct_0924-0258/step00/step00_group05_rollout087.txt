import time

    start_time = time.time()

    except NameError:
        # Initialize with random values
        h_current = np.random.rand(n_low)
        h_current = h_current / np.sum(h_current) * (n_low / 2)
        n_points = n_low
    
    # Temperature annealing
    T = 0.1
    cooling_rate = 0.99
    
    # Optimization loop
    best_c5 = float('inf')
    best_h = None

    for level in range(n_levels + 1):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        n_points = n_low * (2 ** level)
        if n_points > n_high:
            break
            
        # Initialize current solution for this level
        if level == 0:
            h = h_current.copy()
        else:
            # Interpolate previous solution
            old_points = len(h_current)
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, old_points), h_current)
        
        # Ensure sum = n_points / 2
        h = h / np.sum(h) * (n_points / 2)
        
        # SA loop for this resolution
        for iteration in range(500):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Perturb h
            h_new = h.copy()
            noise = np.random.normal(0, T, n_points)
            h_new += noise
            
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Adjust to maintain sum constraint
            sum_new = np.sum(h_new)
            if sum_new > 0:
                h_new = h_new / sum_new * (n_points / 2)
            
            # Compute C5 (overlap integral)
            c5_new = max(signal.correlate(h_new, 1 - h_new, mode='full')) * (2.0 / n_points)
            
            # Accept or reject
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            # Metropolis acceptance
            if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / T):

    else:
        c5_bound = best_c5
