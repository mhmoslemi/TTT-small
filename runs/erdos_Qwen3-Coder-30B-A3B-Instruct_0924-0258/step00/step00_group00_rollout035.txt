import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    perturb_steps = 20
    alpha_candidates = [0.1, 0.5, 1.0]
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Normalize to correct sum
    h *= target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Compute all cross-correlations
        h_rev = np.flip(h)
        correlations = correlate(h, h_rev, mode='full')
        correlations = correlations[len(correlations)//2:] * dx
        
        # Find the maximum correlation index
        max_k = np.argmax(correlations)
        max_correlation = correlations[max_k]
        
        # Compute the gradient at this lag
        grad = np.zeros(n_points)
        if max_k < n_points:
            grad[max_k:] = -h[:-max_k] if max_k > 0 else -h
        else:
            offset = max_k - n_points + 1
            grad[:n_points-offset] = -h[offset:]
        
        # Try different step sizes
        best_alpha = None
        best_h_new = None
        best_new_c5 = best_c5
        
        for alpha in alpha_candidates:
            h_new = h - alpha * grad
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale
            h_new *= target_sum / np.sum(h_new)
            
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_h_new = h_new.copy()
                best_alpha = alpha
        
        # If any alpha improved
        if best_h_new is not None:
            h = best_h_new
            if best_new_c5 < best_c5:
                best_c5 = best_new_c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Perturb if no improvement
        if no_improvement_count >= perturb_steps:
            h += 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h *= target_sum / np.sum(h)
            no_improvement_count = 0
            
    return (list(best_h), best_c5, n_points)

def compute_c5(h, dx):
    # Use correlation with (1-h)
    h_rev = np.flip(1 - h)
    correlations = correlate(h, h_rev, mode='full')
    # Get the maximum value scaled by dx
    return np.max(correlations) * dx
