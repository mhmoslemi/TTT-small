```python
import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h values"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations and find max
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, n_points):
    """Project h to satisfy constraints: clamp and normalize"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def mutate(h, sigma):
    """Apply Gaussian mutation"""
    h_new = h + np.random.normal(0, sigma, len(h))
    return project(h_new, len(h))

def crossover(h1, h2, alpha):
    """Blend crossover with alpha parameter"""
    beta = np.random.uniform(alpha, 1 - alpha)
    h_new = beta * h1 + (1 - beta) * h2
    return project(h_new, len(h1))

def compute_fitness(h):
    """Compute fitness (C5 bound) for h"""
    return evaluate_erdos_solution(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 70
    max_generations = 1000
    population_size = 100
    elite_size = int(0.4 * population_size)
    crossover_rate = 0.3
    mutation_rate = 0.7
    alpha = 0.6
    initial_sigma = 0.1
    tau = 100
    
    # Initialize population
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
    else:
        # Create initial uniform step function
        population = [np.ones(n_points) * 0.5]
    
    # Add diverse initial population
    for _ in range(population_size - 1):
        h = np.random.random(n_points) * 0.5
        h = project(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = [compute_fitness(h) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # Main evolution loop
    gen = 0
    while time.time() - start_time < budget_s - 1 and gen < max_generations:
        gen += 1
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        if fitnesses[0] < best_fitness:
            best_h = population[0].copy()
            best_fitness = fitnesses[0]
        
        # Selection: keep top elite_size and some random
        selected = population[:elite_size]
        for _ in range(population_size - elite_size):
            idx = random.randint(0, elite_size - 1)
            selected.append(population[idx])
        
        # Generate new population
        new_population = []
        
        # Crossover
        for _ in range(int(crossover_rate * population_size)):
            if len(selected) >= 2:
                i1, i2 = random.sample(range(len(selected)), 2)
                new_h = crossover(selected[i1], selected[i2], alpha)
                new_population.append(new_h)
        
        # Mutation
        for _ in range(int(mutation_rate * population_size)):
            h = random.choice(selected)
            sigma = initial_sigma * np.exp(-gen / tau)
            new_h = mutate(h, sigma)
            new_population.append(new_h)
        
        # Fill up to population size
        while len(new_population) < population_size:
            new_population.append(np.random.random(n_points) * 0.5)
            new_population[-1] = project(new_population[-1], n_points)
        
        population = new_population[:population_size]
        fitnesses = [compute_fitness(h) for h in population]
    
    # Final evaluation of best solution
    c5_bound = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), c5_bound, n_points)
```
