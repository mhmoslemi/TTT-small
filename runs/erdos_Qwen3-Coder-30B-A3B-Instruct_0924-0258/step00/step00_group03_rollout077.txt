import numpy as np
from scipy import signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization parameters
    max_iter = 2000
    lr = 0.01
    decay_rate = 0.95
    iter_decay = 100
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute C5 and find maximum lag k0
        conv = np.correlate(h, 1 - h, mode='full')
        k0 = np.argmax(conv)
        
        # Compute gradient of the convolution at k0
        # The gradient wrt h[i] of conv[k0] is the number of times h[i] appears
        # in the cross-correlation computation over all shifts
        grad = np.zeros_like(h)
        for j in range(n_points):
            if k0 - j >= 0 and k0 - j < n_points:
                grad[j] += 1 - h[k0 - j]
            if k0 + j < len(conv) and k0 + j >= len(conv) - n_points:
                grad[j] += h[k0 + j - len(conv) + n_points]
                
        # Update h
        h_new = h - lr * grad
        
        # Project onto feasible set using convex optimization
        h_proj = project_onto_feasible(h_new, n_points, dx)
        
        # Check if we improved
        c5_new = compute_c5(h_proj, dx)
        if c5_new < best_c5:
            best_h = h_proj.copy()
            best_c5 = c5_new
            
        h = h_proj
        
        # Decay learning rate
        if (i + 1) % iter_decay == 0:
            lr *= decay_rate
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    # Compute the full convolution and find its maximum
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx

def project_onto_feasible(h, n_points, dx):
    # Use convex optimization to project onto constraints:
    # 0 <= h[i] <= 1     and     sum(h) * dx = 1
    
    h_proj = cp.Variable(n_points)
    
    # Objective is not needed, just solve with constraints
    constraints = [
        h_proj >= 0,
        h_proj <= 1,
        cp.sum(h_proj) == 1.0 / dx
    ]
    
    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        h_result = h_proj.value
        return np.clip(h_result, 0, 1)
    except:
        # fallback to projection if optimization fails
        h_result = np.clip(h, 0, 1)
        h_result = h_result / (np.sum(h_result) * dx) * (1.0 / dx)
        return h_result
