```python
import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_c5(h, n_points):
    """Evaluate the maximum overlap integral C5 for given h."""
    dx = 2.0 / n_points
    # Compute the full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum value (over all lags) times dx
    max_overlap = np.max(conv) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to ensure it's in [0,1] and sum(h) = n_points / 2."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def finite_difference_gradient(h, n_points, eps=1e-5):
    """Compute finite difference gradient of C5 w.r.t. h."""
    base_c5 = evaluate_c5(h, n_points)
    grad = np.zeros_like(h)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus, n_points)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert len(h) == n_points
        assert np.all(h >= 0) and np.all(h <= 1)
        assert abs(np.sum(h) * dx - 1.0) < 1e-10
    else:
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points)
    
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Local optimization step
        def obj(h_vec):
            return evaluate_c5(h_vec, n_points)
        
        def grad(h_vec):
            return finite_difference_gradient(h_vec, n_points)
        
        # Use L-BFGS-B with projected gradients
        try:
            result = minimize(
                obj,
                h,
                method='L-BFGS-B',
                jac=grad,
                bounds=[(0, 1)] * n_points,
                options={'maxiter': 10}
            )
            if result.success:
                h = result.x
            else:
                h = project_to_feasible(h, n_points)
        except:
            h = project_to_feasible(h, n_points)
        
        # Apply low-pass filter to smooth h
        h_smoothed = gaussian_filter1d(h, sigma=0.1 * n_points)
        h = project_to_feasible(h_smoothed, n_points)
        
        # Update best solution
        current_c5 = evaluate_c5(h, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit if we're close to target
        if best_c5 < 0.3807:
            break
    
    # Final validation and return
    final_c5 = evaluate_c5(best_h, n_points)
    
    return (best_h.tolist(), float(final_c5), n_points)
```
