import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the C5 bound for given h"""
    n_points = len(h)
    # Compute all correlations
    correlation = np.correlate(h, 1 - h, mode="full")
    # Only consider valid overlaps (the middle part)
    mid = len(correlation) // 2
    valid_correlations = correlation[mid - n_points + 1:mid + n_points - 1]
    max_overlap = np.max(valid_correlations * dx)
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def convex_surrogate(h, n_points):
    """Compute a convex surrogate of the maximum overlap"""
    h = np.array(h)
    dx = 2.0 / n_points
    correlation = np.correlate(h, 1 - h, mode="full")
    mid = len(correlation) // 2
    # Use the maximum as convex surrogate
    max_val = np.max(correlation[mid - n_points + 1:mid + n_points - 1])
    # Use squared max to make it differentiable
    return max_val**2 * dx**2

def constraint_func(h, n_points):
    """Constraint for sum(h) = n_points / 2"""
    return np.sum(h) - n_points / 2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    best_h = None
    best_c5 = float('inf')
    n_points_list = [15, 30, 45, 60, 75]  # Resolution levels
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        current_h = np.array(initial_h_values)
        current_n_points = len(current_h)
        # Ensure initial is feasible
        current_h = project_to_feasible(current_h, current_n_points)
        init_c5 = evaluate_erdos_solution(current_h, 2.0 / current_n_points)
        if init_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = init_c5
    else:
        # Create a reasonable starting point
        current_n_points = 45
        current_h = np.ones(current_n_points) * 0.5
        current_h = project_to_feasible(current_h, current_n_points)
        init_c5 = evaluate_erdos_solution(current_h, 2.0 / current_n_points)
        if init_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = init_c5
    
    # Loop over resolutions
    for idx, n_points in enumerate(n_points_list):
        if time() - start_time > budget_s - 2:
            break
        if idx > 0:
            # Interpolate from previous resolution
            old_n = n_points_list[idx - 1]
            x_old = np.linspace(0, 2, old_n)
            x_new = np.linspace(0, 2, n_points)
            current_h = np.interp(x_new, x_old, current_h)
            current_h = project_to_feasible(current_h, n_points)
        
        # Perturb and optimize at current resolution
        for pert_idx in range(3):  # Try 3 random perturbations
            if time() - start_time > budget_s - 2:
                break
            # Add small random perturbation
            h_pert = current_h + np.random.normal(0, 0.05, n_points)
            h_pert = project_to_feasible(h_pert, n_points)
            
            # Optimization with constraints
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = [{'type': 'eq', 'fun': constraint_func, 'args': (n_points,)}]
            
            def objective(h):
                return convex_surrogate(h, n_points)
            
            try:
                result = minimize(objective, h_pert, method='SLSQP', bounds=bounds, 
                                constraints=constraints, tol=1e-6, 
                                options={'maxiter': 500, 'ftol': 1e-6})
                if result.success:
                    h_opt = project_to_feasible(result.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, 2.0 / n_points)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
            except:
                pass
            
            # Also check the perturbed solution directly
            c5_pert = evaluate_erdos_solution(h_pert, 2.0 / n_points)
            if c5_pert < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_pert
            
            # Update current_h for next iteration
            current_h = h_pert if c5_pert < evaluate_erdos_solution(current_h, 2.0 / n_points) else current_h

    # Final validation
    if best_h is None:
        return None, float('inf'), 0
    return best_h.tolist(), best_c5, len(best_h)
