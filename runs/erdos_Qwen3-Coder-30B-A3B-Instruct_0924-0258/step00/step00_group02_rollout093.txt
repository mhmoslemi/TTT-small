import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Compute convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode="full")
    # Scale by dx
    conv *= dx
    # Return the maximum value
    return np.max(conv)

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points / 2."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

    else:
        # Fallback: create a simple step function
        n_points = 45
        h_init = np.ones(n_points) * 0.5
        h_init[10:20] = 1.0  # Step up
        h_init[25:35] = 0.0  # Step down
        h_init = project_to_feasible(h_init, n_points)
    
    n_points = len(h_init)
    dx = 2.0 / n_points
    particle_count = 30
    max_iterations = max(100, int(budget_s / 5))  # Rough estimate
    
    # Initialize swarm
    particles = []
    velocities = []
    
    # First particle: initial solution
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Additional particles: slightly perturbed versions
    for i in range(1, particle_count):
        perturbation = np.random.normal(0, 0.1, n_points)
        h_pert = h_init + perturbation
        h_pert = project_to_feasible(h_pert, n_points)
        particles.append(h_pert)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    
    # Initialize personal bests
    pBest = [evaluate_erdos_solution(p) for p in particles]
    pBest_h = [p.copy() for p in particles]
    
    # Initialize global best
    gBest_idx = np.argmin(pBest)
    gBest = pBest[gBest_idx]
    gBest_h = particles[gBest_idx].copy()
    
    # PSO parameters
    w = 0.9  # Inertia weight
    c1 = 2.0  # Cognitive parameter
    c2 = 2.0  # Social parameter
    
    # Main loop
    for iter_num in range(max_iterations):
        if time.time() - start_time > budget_s - 0.5:
            break
            
        # Evaluate current positions
        for i, particle in enumerate(particles):
            current_c5 = evaluate_erdos_solution(particle)
            if current_c5 < pBest[i]:
                pBest[i] = current_c5
                pBest_h[i] = particle.copy()
                
                if current_c5 < gBest:
                    gBest = current_c5
                    gBest_h = particle.copy()
        
        # Update velocities and positions
        for i in range(particle_count):
            r1, r2 = np.random.rand(2)
            v = w * velocities[i] + c1 * r1 * (pBest_h[i] - particles[i]) + \
                c2 * r2 * (gBest_h - particles[i])
            
            # Bound velocity
            v = np.clip(v, -0.5, 0.5)
            velocities[i] = v
            
            # Update position
            particles[i] += v
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
        # Decay inertia weight
        w = max(0.4, w * 0.99)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(gBest_h)
    
    return (gBest_h.tolist(), final_c5, n_points)
