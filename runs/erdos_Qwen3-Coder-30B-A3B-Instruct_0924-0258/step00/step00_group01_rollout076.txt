import numpy as np
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    # Compute the maximum overlap using convolution
    conv = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Optimization iterations
    max_iter = 100
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Get current overlaps
        overlaps = np.correlate(h, 1-h, mode="full")
        max_overlap = np.max(overlaps)
        # Add small epsilon to avoid numerical issues
        max_overlap += 1e-12
        
        # CVXPY optimization
        try:
            # Create variables and constraints
            h_vars = cp.Variable(n_points)
            C = cp.Variable()
            
            # Constraints: h_vars[i] >= 0, h_vars[i] <= 1
            constraints = [
                h_vars >= 0,
                h_vars <= 1,
                cp.sum(h_vars) == target_sum
            ]
            
            # Constraints for each overlap
            for i in range(2 * n_points - 1):
                # Modify the overlap constraint based on current i
                lag = i - (n_points - 1)
                if lag < 0:
                    start_idx = -lag
                    end_idx = n_points
                else:
                    start_idx = 0
                    end_idx = n_points - lag
                    
                if start_idx < end_idx:
                    # Add constraint: sum(h_vars[j] * (1 - h_vars[j+lag])) <= C*dx
                    lag_sum = 0
                    for j in range(start_idx, end_idx):
                        lag_sum += h_vars[j] * (1 - h_vars[j + lag])
                    constraints.append(lag_sum <= C * dx)
            
            # Objective: minimize C
            objective = cp.Minimize(C)
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_vars.value
                if new_h is not None and np.all(new_h >= 0) and np.all(new_h <= 1):
                    if abs(np.sum(new_h) - target_sum) < 1e-6:
                        h = new_h
                        new_c5 = evaluate_c5(h, dx)
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
        except Exception:
            pass

    c5_bound = best_c5
    h_values = best_h.tolist()
    return (h_values, c5_bound, n_points)
