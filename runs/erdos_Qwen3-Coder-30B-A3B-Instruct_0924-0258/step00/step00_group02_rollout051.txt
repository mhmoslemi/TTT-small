import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h_values):
    # Evaluate the maximum overlap for given h
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    corr = signal.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_h(h, n_points):
    # Project h onto feasible region: [0,1] and sum(h) = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h
