import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Normalize to sum to n_points/2
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    last_improvement = 0
    iter_count = 0
    perturb_steps = 20
    alpha_candidates = [0.1, 0.5, 1.0]
    
    while time.time() - start_time < budget_s - 0.1:
        iter_count += 1
        
        # Compute cross-correlations
        corr = correlate(h, 1 - h, mode='full')
        k_indices = np.arange(-n_points + 1, n_points)
        max_k_idx = np.argmax(corr)
        max_k = k_indices[max_k_idx]
        max_corr = corr[max_k_idx]
        
        # Compute gradient for max_k
        gradient = np.zeros(n_points)
        if max_k >= 0:
            gradient[max_k:] = -(1 - h[:-max_k] if max_k > 0 else -h)
        else:
            gradient[:max_k] = -(1 - h[-max_k:] if max_k < 0 else -h)
        
        # Try different step sizes
        best_h_new = h.copy()
        best_c5_new = best_c5
        for alpha in alpha_candidates:
            h_new = h - alpha * gradient
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            h_new = h_new * (n_points / 2) / h_new_sum
            
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < best_c5_new:
                best_c5_new = c5_new
                best_h_new = h_new.copy()
        
        # Add perturbation if no improvement for too long
        if iter_count - last_improvement > perturb_steps:
            h_new = h + 0.01 * np.random.randn(n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            h_new = h_new * (n_points / 2) / h_new_sum
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < best_c5_new:
                best_c5_new = c5_new
                best_h_new = h_new.copy()
            last_improvement = iter_count
        
        # Update if we found a better solution
        if best_c5_new < best_c5:
            h = best_h_new.copy()
            best_c5 = best_c5_new
            last_improvement = iter_count
            
        # Check if we've already run long enough
        if iter_count % 100 == 0 and time.time() - start_time > 0.8 * budget_s:
            break
    
    return (best_h, best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5
