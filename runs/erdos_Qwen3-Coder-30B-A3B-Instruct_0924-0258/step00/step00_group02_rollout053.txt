import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[1] = 0.6
    
    # Normalize to correct sum
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Learning parameters
    lr = 0.01
    min_lr = 1e-6
    decay_rate = 0.999
    perturb_factor = 0.05
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        if iteration % 10 == 0 and lr > min_lr:
            lr *= decay_rate
        
        # Compute overlap for all shifts
        corr = correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_overlap = corr[max_k] * dx
        
        # Compute gradient
        # Gradient w.r.t h for the k-th shift is the shifted 1-h values
        # Corresponds to the convolution of h with 1-h, but we just take the derivative
        # We're optimizing \max_k \int h(x)(1-h(x+k)) dx
        # Gradient of that integral w.r.t. h_i is the k-shift of (1-h_{i+k})
        # Need to be careful with indices and boundary conditions
        
        # The gradient is a summation of shifted terms.
        # For each shift k, the gradient for h[i] is (1-h[i-k]), we'll compute full gradient
        # This is computationally messy, so we'll use finite difference for more robustness
        # Or use standard formula for gradient of correlation
        
        # Simpler approach: finite difference gradient estimation
        # We will compute for all possible shifts and work with what gives max
        # We'll just use the current max overlap gradient for updating
        
        # For k-th shift we want to compute gradient of h * (1-h_{k}) 
        # The actual derivative of correlation w.r.t h_i is convolve((1-h), 1) at shift k
        # Which is (1-h) at k shifted left at index i, which is 1-h[i-k] if i-k ≥ 0, else 0
        # i.e. gradient for h_i is the part of 1-h that affects the current correlation
        # create shifted copies of (1-h) based on the maximizing shift
        
        # Construct gradient
        gradient = np.zeros_like(h)
        # Which shift gave maximum correlation?
        # Find the peak by manipulating the signal
        shifted_ones = np.roll(1 - h, max_k - (len(h) - 1))
        gradient = shifted_ones
        
        # Update h
        h -= lr * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        # Evaluate new C5
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Occasionally perturb h
        if iteration % 20 == 0:
            perturb = np.random.uniform(-perturb_factor, perturb_factor, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            
        iteration += 1
        
    return (best_h.tolist(), best_c5, n_points)
