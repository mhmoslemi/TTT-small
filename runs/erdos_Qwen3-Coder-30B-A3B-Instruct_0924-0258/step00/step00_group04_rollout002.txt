import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral for given step function h."""
    # Compute correlation for all lags
    corr = np.correlate(h, 1 - h, mode="full")
    # Take only the relevant part corresponding to the actual overlap
    n = len(h)
    # The full correlation is 2*n-1 long, so we want the middle n elements
    # which correspond to lags from -n+1 to n-1
    corr_middle = corr[n-1:n-1+n]  # 0 to n-1 lag
    return np.max(corr_middle) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if not (np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - n_points / 2) < 1e-10):
            h = np.ones(n_points) * 0.5
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Local optimization step (using L-BFGS-B with finite differences)
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0, 1)
            return evaluate_c5(h, dx)
        
        # Define constraints and bounds
        bounds = [(0, 1) for _ in range(n_points)]
        # Constraint: sum(h) = n_points / 2
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
        
        con = {'type': 'eq', 'fun': constraint_sum}
        
        # We will use scipy.optimize.minimize on the discretized function
        # Use bounds and constraints
        try:
            res = minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=con,
                options={'maxiter': 20, 'eps': 1e-4},
                callback=lambda x: None  # dummy callback
            )
            if res.success:
                h_new = res.x.reshape(n_points)
                h_new = np.clip(h_new, 0, 1)
                # Rescale to meet constraint again
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
            else:
                # If optimization didn't succeed, try a simple update
                pass
        except:
            pass

        # Apply low-pass filter
        sigma = 0.1 * n_points
        h_filtered = gaussian_filter1d(h, sigma)
        h = np.clip(h_filtered, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        c5_current = evaluate_c5(h, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
