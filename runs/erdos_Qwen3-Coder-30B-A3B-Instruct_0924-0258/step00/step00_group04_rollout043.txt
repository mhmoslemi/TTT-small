import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        h = np.full(n_points, 0.5)
    
    # Ensure correct integral
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h, n_points)
    
    # Timing
    start_time = time.time()
    elapsed = 0
    
    # Gradient descent parameters
    step_size = 0.1
    tolerance = 1e-6
    perturb_every = 50
    max_iter = 100000
    
    for iter_count in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.1:
            break
            
        # Compute all overlaps
        overlaps = []
        for k in range(n_points):
            # Compute overlap integral for lag k
            # h[i] * (1 - h[(i+k) % n_points])
            lagged = np.roll(h, k)
            overlap = np.dot(h, 1 - lagged)
            overlaps.append(overlap)
        
        # Find worst case
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # If we're already good enough, we can stop early
        if max_overlap <= 0.3807:
            break
        
        # Compute gradient for the maximum overlap case
        grad = np.zeros(n_points)
        for i in range(n_points):
            prev_idx = (i - max_k) % n_points
            next_idx = (i + max_k) % n_points
            grad[i] = 1 - h[prev_idx] - h[next_idx]
        
        # Apply step
        h_new = h - step_size * grad
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to correct integral
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update h
        h = h_new
        
        # Perturb sometimes to escape local minima
        if iter_count % perturb_every == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate
        current_C5 = evaluate_C5(h, n_points)
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
    
    return (best_h.tolist(), best_C5, n_points)

def evaluate_C5(h, n_points):
    dx = 2.0 / n_points
    # Compute auto-correlation-like
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum value
    max_val = np.max(corr)
    # Multiply by dx to match scaling
    return max_val * dx
