import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_c5(h, n_points):
    """Compute the maximum overlap integral C5"""
    dx = 2.0 / n_points
    # Compute the full correlation (overlap) for all lags
    overlap = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value and scale by dx
    return np.max(overlap) * dx

def project_to_feasible(h):
    """Project h to [0,1] and rescale to meet sum constraint"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (2.0 / h_sum)  # Make sum = 2.0, which means sum(h)*dx = 1
    return h

def gradient_step(h, n_points, eps=1e-5):
    """Compute gradient using finite differences"""
    grad = np.zeros_like(h)
    c5_base = evaluate_c5(h, n_points)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus)
        c5_plus = evaluate_c5(h_plus, n_points)
        grad[i] = (c5_plus - c5_base) / eps
    
    return grad

def optimize_step(h, n_points, max_iter=10):
    """Local optimization step using L-BFGS-B"""
    # Project to feasible region before optimization
    h = project_to_feasible(h)
    
    def objective(h_vec):
        h_new = h_vec.reshape(-1)
        h_new = project_to_feasible(h_new)
        return evaluate_c5(h_new, n_points)
    
    def gradient(h_vec):
        h_new = h_vec.reshape(-1)
        g = gradient_step(h_new, n_points)
        return g.flatten()
    
    # Flatten the vector for optimization
    h_flat = h.flatten()
    
    # Perform optimization
    result = minimize(
        objective,
        h_flat,
        method='L-BFGS-B',
        jac=gradient,
        bounds=[(0, 1)] * len(h_flat),
        options={'maxiter': max_iter}
    )
    
    h_opt = result.x.reshape(-1)
    h_opt = project_to_feasible(h_opt)
    
    return h_opt

def smooth_h(h, sigma_factor=0.1):
    """Apply Gaussian smoothing to h"""
    n_points = len(h)
    sigma = sigma_factor * n_points
    h_smooth = gaussian_filter1d(h, sigma=sigma)
    return h_smooth

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42  # Reasonable for memory and speed
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
        
    # Make sure the sum constraint is satisfied
    h = project_to_feasible(h)
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points)
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:  # Keep some margin
        iteration += 1
        
        # Phase 1: Local optimization
        h_opt = optimize_step(h, n_points, max_iter=20)
        
        # Phase 2: Low-pass filtering
        h_filtered = smooth_h(h_opt, sigma_factor=0.1)
        h = project_to_feasible(h_filtered)
        
        # Evaluate new C5
        c5_new = evaluate_c5(h, n_points)
        
        # Update best solution
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
            
        # Early termination check
        if best_c5 < 0.3807:  # Target bound
            break
    
    # Final check
    final_c5 = evaluate_c5(best_h, n_points)
    
    return (best_h, final_c5, n_points)
