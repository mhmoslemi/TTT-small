import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from initial_h_values if available, else create own

        n_points = len(h)
    else:

    h = h * (n_points / 2.0) / np.sum(h)
    
    # Ensure within bounds
    h = np.clip(h, 0, 1)
    
    # Time tracking
    start_time = time.time()
    
    # Objective function
    def compute_C5(h_vec):
        # correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        # return np.max(correlation) * (2.0 / n_points)  # dx = 2/n_points
        # About 50% faster to compute manually:
        dx = 2.0 / n_points
        corr = np.zeros(2 * n_points - 1)
        for i in range(n_points):
            for j in range(n_points):
                k = i + j
                if k < len(corr):
                    corr[k] += h_vec[i] * (1 - h_vec[j])
        return np.max(corr) * dx
    
    # Gradient calculation
    def compute_gradient(h_vec, max_k):
        dx = 2.0 / n_points
        grad = np.zeros(n_points)
        for k in max_k:
            for i in range(n_points):
                if i + k < n_points:
                    grad[i] += (1 - h_vec[i + k])
                    grad[i + k] -= h_vec[i]
        return grad * dx
    
    # Line search
    def line_search(h_vec, grad_vec, max_k):
        alphas = [0.1, 0.25, 0.5]
        best_h = h_vec.copy()
        best_C5 = compute_C5(h_vec)
        best_alpha = 0.0
        
        for alpha in alphas:
            h_trial = h_vec - alpha * grad_vec
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2.0) / np.sum(h_trial)
            
            C5_new = compute_C5(h_trial)
            
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h_trial.copy()
                best_alpha = alpha
        
        return best_h, best_C5, best_alpha

    best_C5 = compute_C5(h)
    best_h = h.copy()
    no_improvement_count = 0
    perturb_factor = 0.01
    
    while time.time() - start_time < budget_s:
        # Compute correlation for all lags
        correlation = correlate(h, 1 - h, mode='full')
        # Adjust for proper indexing
        C5_vals = correlation * (2.0 / n_points)
        max_C5 = np.max(C5_vals)

        grad = np.zeros(n_points)

            # Only consider meaningful lags (within valid range)
            for i in range(n_points):
                if i + k < 2 * n_points - 1:
                    ii = i
                    jj = i + k - (n_points - 1)
                    if jj >= 0 and jj < n_points:
                        grad[ii] += (1 - h[jj])
                        grad[jj] -= h[ii]
        grad = grad * (2.0 / n_points)

        # Line search
        h_new, new_C5, alpha = line_search(h, grad, max_indices)
        
        # Update if improvement
        if new_C5 < best_C5:
            h = h_new.copy()
            best_C5 = new_C5
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        if no_improvement_count > 20:
            h = h + perturb_factor * np.random.randn(n_points)
            h = np.clip(h, 0, 1)

            no_improvement_count = 0

        # Update best solution
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h.copy()

        # Ensure time is left for final check
        if time.time() - start_time > budget_s - 0.1:
            break

    # Final validation
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points / 2.0) / np.sum(h_final)
    final_C5 = compute_C5(h_final)

    return (h_final.tolist(), float(final_C5), int(n_points))
