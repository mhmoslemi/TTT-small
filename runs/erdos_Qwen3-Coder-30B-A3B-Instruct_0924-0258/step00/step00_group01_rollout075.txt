import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral for given step function h."""
    # Compute correlations: h(x)(1 - h(x+k)) for all lags k
    corr = correlate(h, 1 - h, mode="full")
    # The correlation array has length 2*n_points - 1
    # The valid overlaps are from index n_points-1 to 2*n_points-2
    valid_corr = corr[len(h)-1:]
    max_overlap = np.max(valid_corr) * dx
    return max_overlap

def project_feasible(h, dx):
    """Project h to satisfy sum(h) * dx = 1 and bounds [0,1]."""
    h = np.clip(h, 0, 1)
    total = np.sum(h) * dx
    if total > 0:
        h = h / total * (2.0 / dx)
    h = np.clip(h, 0, 1)
    return h

def update_h_convex(h, dx, max_overlap, prev_h=None):
    """Solve a convex relaxation to improve h."""
    n = len(h)
    # Create variables
    h_new = cp.Variable(n, name='h_new')
    # Objective: minimize max_overlap
    obj = cp.Minimize(max_overlap)
    # Constraints
    constraints = []
    # h_new in [0,1]
    constraints.append(h_new >= 0)
    constraints.append(h_new <= 1)
    # Integral constraint
    constraints.append(cp.sum(h_new) == n / 2.0)
    # Add constraints for current maximum overlap
    corr = correlate(h_new, 1 - h_new, mode="full")
    valid_corr = corr[n-1:]
    constraints.append(cp.MAX(valid_corr) <= max_overlap)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            new_h = h_new.value
            # Project to feasible set
            new_h = project_feasible(new_h, dx)
            return new_h
    except Exception:
        pass
    # If optimization fails, return original h
    return h

def run(seed=42, budget_s=600, **kwargs):

        dx = 2.0 / n_points

        h = project_feasible(h, dx)

    dx = 2.0 / n_points
    
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Iteratively refine h using a simple approach:
    # Compute overlaps, try to reduce the max overlap with a convex relaxation
    max_iter = 100
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate current C5
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Update with convex relaxation
        c5 = evaluate_c5(h, dx)
        # Approximate h with a binary structure that minimizes the overlap
        h = update_h_convex(h, dx, c5, h)
        h = project_feasible(h, dx)
        
    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
