import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Parameters
    perturb_steps = 20
    step_size_candidates = [0.1, 0.5, 1.0]
    max_iter = int(budget_s * 4)  # Rough estimate: ~0.25s/iteration
    current_iter = 0
    last_improvement = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    while current_iter < max_iter and (time.time() - start_time) < budget_s - 0.1:
        current_iter += 1
        # Compute all cross-correlations
        corr = np.correlate(h, 1 - h, mode='full')
        k_indices = np.arange(-n_points + 1, n_points)
        max_k_idx = np.argmax(corr)
        max_k = k_indices[max_k_idx]
        max_corr = corr[max_k_idx]
        
        # Compute gradient for maximum correlation
        gradient = np.zeros(n_points)
        if max_k >= 0:
            gradient[max_k:] = -(1 - h[:-max_k])
        else:
            gradient[:max_k] = -(1 - h[-max_k:])
        
        # Line search
        best_alpha = 0.0
        best_h_new = h.copy()
        best_c5_new = best_c5
        
        for alpha in step_size_candidates:
            h_new = h - alpha * gradient
            h_new = np.clip(h_new, 0, 1)
            # Rescale to correct sum
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5_new:
                best_c5_new = c5_new
                best_h_new = h_new.copy()
                best_alpha = alpha
        
        # If no improvement, perturb
        if best_c5_new >= best_c5:
            last_improvement += 1
            if last_improvement > perturb_steps:
                noise = 0.01 * np.random.randn(n_points)
                h += noise
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                last_improvement = 0
                continue
        else:
            last_improvement = 0
            h = best_h_new
            best_c5 = best_c5_new
        
        # Update best solution
        if best_c5_new < compute_c5(best_h):
            best_h = h.copy()
    
    return (best_h.tolist(), compute_c5(best_h), n_points)

def compute_c5(h):
    # Correlation from scipy's correlate yields maximum at index n_points-1
    # if we want correlation from -n_points+1 to n_points-1, use full mode
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) / len(h)
