import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure bounds [0,1]
    h = np.clip(h, 0, 1)
    
    # Objective function to compute C5
    def compute_C5(h_vals):
        n = len(h_vals)
        dx = 2.0 / n
        # Compute cross-correlation for all lags
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # The maximum is over all valid lags (excluding trivial ones)
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Optimization loop with Frank-Wolfe style approach
    current_h = h.copy()
    best_h = current_h.copy()
    best_C5 = compute_C5(current_h)
    
    # Frank-Wolfe parameters
    max_iter = 500
    tolerance = 1e-4
    learning_rate = 0.9
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_C5 = compute_C5(current_h)
        if current_C5 < best_C5:
            best_h = current_h.copy()
            best_C5 = current_C5
            
        # Compute gradient at current point
        # Gradient of Correlation_{-k} w.r.t h_i
        n = len(current_h)
        dx = 2.0 / n
        
        # Precompute partial gradients
        grad = np.zeros(n)
        for i in range(n):
            for k in range(n):
                if i + k < n:
                    grad[i] += (1 - current_h[i + k])  # d/dh_i [ h[i] * (1 - h[i+k]) ]
                if i - k >= 0:
                    grad[i] += current_h[i - k]        # d/dh_i [ h[i] * (1 - h[i-k]) ]
        
        # We want to reduce the maximum correlation, so we move in direction of negative gradient
        # For Frank-Wolfe, we solve a linear approximation and pick subgradient
        # Here rather than full FW, use simple gradient descent with projection
        update_dir = -grad
        
        # Line search to find optimal step size
        step_sizes = [0.1, 0.25, 0.5, 0.75, 1.0]
        best_step = 0.0
        best_val = current_C5
        
        for step in step_sizes:
            new_h = current_h + step * update_dir
            # Clamp values to [0,1]
            new_h = np.clip(new_h, 0, 1)
            # Rescale to maintain constraint sum(h) = n/2
            new_h = new_h * (n / 2) / np.sum(new_h)
            new_C5 = compute_C5(new_h)
            if new_C5 < best_val:
                best_val = new_C5
                best_step = step
                
        # Accept the new point
        current_h = current_h + best_step * update_dir
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (n / 2) / np.sum(current_h)
        
        # Safeguard against convergence to local minima
        if iteration > 0 and abs(current_C5 - best_C5) < tolerance:
            # Small random perturbation
            noise = np.random.normal(0, 0.01, n)
            current_h += noise
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n / 2) / np.sum(current_h)
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
