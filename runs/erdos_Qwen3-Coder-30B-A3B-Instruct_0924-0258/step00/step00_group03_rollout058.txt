import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute the overlap integral for all lags
    overlap = correlate(h, 1 - h, mode="full")
    # Scale by dx and return maximum
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the given sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return np.full_like(h, target_sum / len(h))

def mutate_h(h, mutation_rate=0.1, noise_std=0.1):
    """Mutate h with bounded noise and normalize."""
    # Add random noise
    noise = np.random.normal(0, noise_std, len(h))
    h_mut = h + mutation_rate * noise
    
    # Clip values to [0, 1]
    h_mut = np.clip(h_mut, 0, 1)
    
    # Normalize to maintain sum constraint
    h_mut = normalize_h(h_mut, np.sum(h))
    
    return h_mut

def crossover_h(h1, h2, crossover_rate=0.8):
    """Perform crossover between two h vectors."""
    if np.random.rand() > crossover_rate:
        return h1.copy()
    
    # Single-point crossover
    point = np.random.randint(1, len(h1))
    h_child = np.concatenate([h1[:point], h2[point:]])
    
    # Normalize to maintain sum constraint
    h_child = normalize_h(h_child, np.sum(h1))
    
    return h_child

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = population_size // 10
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    max_generations = 5000
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = normalize_h(h_init, target_sum)
        population.append(h_init)
    else:
        # Create a simple initial guess
        h_init = np.full(n_points, 0.5)  # Uniform
        h_init = normalize_h(h_init, target_sum)
        population.append(h_init)
        
    # Fill rest of population with random solutions
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Evolutionary process
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate population
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Create next generation
        new_population = [population[0].copy()]  # Keep best
        
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            
            # Crossover
            child = crossover_h(parent1, parent2, crossover_rate)
            
            # Mutation
            child = mutate_h(child, mutation_rate)
            
            new_population.append(child)
            
        population = new_population[:population_size]
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
        return (list(best_h), final_c5, n_points)
    else:
        # Fallback if something went wrong
        h = np.full(n_points, 0.5)
        h = normalize_h(h, target_sum)
        c5 = evaluate_h(h, dx)
        return (list(h), c5, n_points)
