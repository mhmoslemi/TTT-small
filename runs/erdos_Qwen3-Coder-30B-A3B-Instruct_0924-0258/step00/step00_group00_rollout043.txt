import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    max_iter_without_improvement = 20
    iter_without_improvement = 0
    perturbation_magnitude = 0.01
    
    def evaluate_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = 0
        for k in range(len(corr)):
            corr_val = corr[k] * dx
            if corr_val > max_corr:
                max_corr = corr_val
        return max_corr
    
    def compute_gradient(h, max_indices):
        gradient = np.zeros_like(h)
        for k in max_indices:
            # i ranges from 0 to n_points - 1 - k 
            for i in range(n_points - k):
                if i < len(gradient):
                    gradient[i] += (1 - h[i + k])
                if i + k < len(gradient):
                    gradient[i + k] -= h[i]
        return gradient
    
    def line_search(h, gradient, alpha_vals=[0.1, 0.25, 0.5]):
        best_h_new = h.copy()
        best_c5_new = evaluate_c5(h)
        for alpha in alpha_vals:
            h_new = h - alpha * gradient
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5_new:
                best_c5_new = c5_new
                best_h_new = h_new
        return best_h_new, best_c5_new
    
    iteration = 0
    while time.time() - start_time < budget_s - 0.1 and iter_without_improvement < max_iter_without_improvement:
        # Compute cross correlations
        corr = correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(corr)
        
        # Get max lag indices
        max_indices = np.where(np.abs(corr - c5) < 1e-8)[0]
        
        # If no improvement found, escape by perturbing
        if c5 >= best_c5:
            iter_without_improvement += 1
            if iter_without_improvement > 10:
                h += perturbation_magnitude * np.random.randn(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            iteration += 1
            continue
            
        # Reset improvement counter
        iter_without_improvement = 0
        
        # Compute gradient
        gradient = compute_gradient(h, max_indices)
        
        # Line search
        h_new, c5_new = line_search(h, gradient)
        
        # Accept new point if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h = h_new
        iteration += 1
        
    # Final check and output
    h_final = best_h if evaluate_erdos_solution(best_h) < evaluate_erdos_solution(h) else h
    c5_final = evaluate_erdos_solution(h_final)
    
    return (h_final.tolist(), c5_final, len(h_final))
