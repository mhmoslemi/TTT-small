import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    population_size = 50
    generations = 200
    crossover_rate = 0.7
    mutation_rate = 0.2
    mutation_delta = 0.1
    tournament_k = 3
    
    # Initialize population
    def initialize_population():
        population = []
        if 'initial_h_values' in globals():
            h_parent = np.array(initial_h_values)
        else:
            h_parent = np.ones(n_points) * 0.5
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        population.append(h_parent.copy())
        
        for _ in range(population_size - 1):
            # Perturb parent slightly
            h = h_parent + np.random.uniform(-0.2, 0.2, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Add random candidates occasionally
            if np.random.rand() < 0.3:
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
            
            population.append(h.copy())
        return population
    
    def evaluate_fitness(h):
        cross_corr = correlate(h, 1 - h, mode='full')
        # Take only the relevant part for max overlap
        full_corr = cross_corr * dx
        return np.max(full_corr)
    
    def tournament_selection(population, fitnesses):
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_k)
            winner = tournament[np.argmin([fitnesses[i] for i in tournament])]
            selected.append(population[winner].copy())
        return selected
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = np.random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        # Rescale for sum constraint
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        return child1, child2
    
    def mutate(h):
        new_h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                new_h[i] += np.random.uniform(-mutation_delta, mutation_delta)
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        return new_h
    
    # Main optimization loop
    population = initialize_population()
    best_h = None
    best_c5 = float('inf')
    elapsed = 0
    
    for gen in range(generations):
        if elapsed > budget_s:
            break
        # Evaluate fitness
        fitnesses = [evaluate_fitness(h) for h in population]
        
        # Track best
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitnesses.index(min_fitness)].copy()
        
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Generate new population
        new_population = [best_h.copy()]  # Keep best
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        population = new_population
        elapsed = time.time() - start_time
    
    if best_h is None:
        best_h = population[0]
    
    # Final evaluation
    final_c5 = evaluate_fitness(best_h)
    return (best_h.tolist(), final_c5, n_points)
