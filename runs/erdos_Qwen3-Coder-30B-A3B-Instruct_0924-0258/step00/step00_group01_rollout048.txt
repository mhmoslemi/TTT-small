import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute correlation for all lags, then max
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_feasible(h, target_sum):
    # Ensure h sums to target_sum, values in [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.ones_like(h) / len(h) * target_sum
    scale = target_sum / current_sum
    h = h * scale
    return h

def initialize_population(n_points, pop_size, initial_h_values):
    population = []
    if initial_h_values is not None:
        h_init = np.array(initial_h_values)
        h_init = project_feasible(h_init, n_points / 2)
        population.append(h_init)
    else:
        # Create base solution
        h = np.ones(n_points) * 0.5
        population.append(h)
    
    # Fill population
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points / 2)
        population.append(h)
    
    return population

def tournament_selection(population, fitnesses, k=2):
    idxs = np.random.choice(len(population), size=k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def crossover(parent1, parent2):
    # Average
    child1 = (parent1 + parent2) / 2.0
    child2 = (parent1 + parent2) / 2.0
    return child1, child2

def mutate(h, mutation_rate=0.05, sigma=0.1):
    # Make copy, apply mutation
    h = h.copy()
    n_points = len(h)
    for i in range(n_points):
        if np.random.rand() < mutation_rate:
            h[i] += np.random.normal(0, sigma)
    return h

def ga_optimize(initial_h_values, n_points, budget_s):
    start_time = time.time()
    
    pop_size = 20
    max_gen = 1000
    mutation_rate = 0.05
    sigma = 0.1
    dx = 2.0 / n_points
    population = initialize_population(n_points, pop_size, initial_h_values)
    
    best_h = None
    best_c5 = np.inf
    no_improvement_count = 0
    
    for gen in range(max_gen):
        if time.time() - start_time > budget_s - 1:
            break
        # Evaluate
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Check for no improvement
        if gen > 0 and fitnesses[0] == fitnesses[1]:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            
        if no_improvement_count > 100:
            break
        
        # Selection, crossover, mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1, mutation_rate, sigma)
            child2 = mutate(child2, mutation_rate, sigma)
            
            child1 = project_feasible(child1, n_points / 2)
            child2 = project_feasible(child2, n_points / 2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Downsize for efficiency
    initial_h_values = None
    try:
        initial_h_values = __import__('sys').modules.get('__main__').initial_h_values
    except:
        pass
        
    h, c5 = ga_optimize(initial_h_values, n_points, budget_s)
    if h is not None:
        # Normalize to exactly sum = n_points / 2
        h = project_feasible(h, n_points / 2)
        return h.tolist(), c5, n_points
    else:
        # Fallback
        h = np.ones(n_points) * 0.5
        h = project_feasible(h, n_points / 2)
        return h.tolist(), 0.5, n_points
