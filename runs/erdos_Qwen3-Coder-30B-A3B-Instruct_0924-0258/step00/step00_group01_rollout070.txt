import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the overlap integral for given h"""
    dx = 2.0 / len(h)
    # Compute correlation of h with (1-h) for all lags
    conv = correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum value in conv
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_to_feasible(h):
    """Project h to feasible region: sum=h_sum, bounds in [0,1]"""
    h = np.array(h)
    h_sum = np.sum(h)
    target_sum = len(h) / 2.0
    if h_sum == 0.0:
        h = np.ones_like(h) * (target_sum / len(h))
    else:
        h = h * target_sum / h_sum
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Re-adjust sum
    h_sum = np.sum(h)
    h = h * target_sum / h_sum
    return np.clip(h, 0, 1)

def initialize_population(n_points, n_pop=20, seed=42):
    """Initialize population"""
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        pop = [np.array(initial_h_values)]
        n_pop -= 1
    else:
        pop = []
    for i in range(n_pop):
        # Random feasible solution
        h = np.random.rand(n_points)
        h = project_to_feasible(h)
        pop.append(h)
    return pop

def crossover(parent1, parent2):
    """Crossover two parents to create offspring"""
    # Blend crossover
    alpha = 0.5
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    child1 = project_to_feasible(child1)
    child2 = project_to_feasible(child2)
    return child1, child2

def mutate(h, rate=0.05, std=0.1):
    """Mutate a solution"""
    h = np.array(h)
    mask = np.random.rand(len(h)) < rate
    h[mask] += np.random.normal(0, std, np.sum(mask))
    h = project_to_feasible(h)
    return h

def tournament_selection(pop, fitnesses, k=2):
    """Select a solution by tournament"""
    indices = np.random.choice(len(pop), k)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return pop[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    n_points = 47  # use the provided parent construction size
    start_time = time.time()
    population = initialize_population(n_points, seed=seed)
    
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    
    generation = 0
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitnesses = [evaluate_h(h) for h in population]
        
        # Track best
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Termination condition
        if no_improvement_count > 100:
            break
            
        # Selection and reproduction
        new_population = []
        
        # Keep best individual
        new_population.append(population[min_idx])
        
        # Generate rest via tournament selection
        while len(new_population) < len(population):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            # Add to population
            new_population.append(child1)
            if len(new_population) < len(population):
                new_population.append(child2)
                
        population = new_population[:len(population)]
    
    if best_h is None:
        # Return last evaluated best
        fitnesses = [evaluate_h(h) for h in population]
        min_idx = np.argmin(fitnesses)
        best_h = population[min_idx].copy()
        best_c5 = fitnesses[min_idx]
        
    return (best_h.tolist(), best_c5, n_points)
