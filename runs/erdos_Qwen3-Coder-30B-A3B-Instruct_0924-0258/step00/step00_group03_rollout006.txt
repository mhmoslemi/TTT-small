import numpy as np
import time
from scipy import signal

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given step function h."""
    # Compute correlation of h with (1-h)
    corr = signal.correlate(h, 1-h, mode="full")
    # Normalize and return the maximum value
    return np.max(corr) * dx

def is_feasible(h):
    """Check if h satisfies the constraints."""
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - len(h)/2) < 1e-10)

def normalize_h(h):
    """Normalize h to satisfy ∫h dx = 1."""
    h = np.array(h)
    h_sum = np.sum(h)
    if abs(h_sum) < 1e-10:
        return np.ones_like(h) / 2  # Avoid division by zero
    return h * (len(h)/2) / h_sum

def mutate_h(h, mutation_rate=0.1):
    """Apply mutation to a step function h."""
    h = np.array(h)
    n = len(h)
    # Add normally distributed noise with small variance
    noise = np.random.normal(0, 0.05, n)
    mutation_mask = np.random.rand(n) < mutation_rate
    h[mutation_mask] += noise[mutation_mask]
    # Ensure feasibility after mutation
    h = np.clip(h, 0, 1)
    h = normalize_h(h)
    return h

def crossover_h(h1, h2):
    """Perform crossover between two step functions."""
    n = len(h1)
    crossover_point = np.random.randint(1, n)
    h = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    h = normalize_h(h)
    return h

    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), 
                                         size=tournament_size, 
                                         replace=False)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx]
