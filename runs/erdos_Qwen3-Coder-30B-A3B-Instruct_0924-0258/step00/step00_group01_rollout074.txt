import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute overlap integral for all lags
    corr = correlate(h, 1 - h, mode='full')
    # Take only the valid lags (from -n+1 to n-1)
    n = len(h)
    valid_corr = corr[n-1:2*n-1]
    max_overlap = np.max(valid_corr) * dx
    return max_overlap

def project_feasible(h, target_sum):
    # Ensure h is in [0, 1] and adjusts sum to target_sum
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple adjustment: add/subtract uniformly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0, 1)
    return h

def crossover(parent1, parent2):
    # Simple averaging crossover
    child1 = (parent1 + parent2) / 2.0
    child2 = (parent1 + parent2) / 2.0
    return child1, child2

def mutate(h, mutation_rate=0.05, std=0.1):
    # Gaussian mutation with boundary handling
    mask = np.random.random(len(h)) < mutation_rate
    if not np.any(mask):
        # At least one mutation
        idx = np.random.randint(0, len(h))
        mask[idx] = True
    h = h.copy()
    h[mask] += np.random.normal(0, std, np.sum(mask))
    h = project_feasible(h, np.sum(h))
    return h
