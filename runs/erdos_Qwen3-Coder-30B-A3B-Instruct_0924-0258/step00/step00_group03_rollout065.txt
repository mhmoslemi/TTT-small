import numpy as np
import time
from scipy import signal

def evaluate_C5(h):
    """Compute the maximum overlap integral for given h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h) for all lags
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum value over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a simple initial guess
        h = np.ones(70) * 0.5
        h[0:35] = 1.0
        h[35:] = 0.0
    
    # Ensure it sums to 35 (since dx = 2/70, sum(h)*dx = 1)
    h = h * 35.0 / np.sum(h)
    
    # Validate initial solution
    assert np.all(h >= 0) and np.all(h <= 1), "Initial h not in [0,1]"
    assert abs(np.sum(h) - 35.0) < 1e-10, "Initial h sum not 35"
    
    # Parameters
    n_points = len(h)
    T0 = 1.0
    alpha = 0.999
    sigma0 = 0.1
    t = 0
    start_time = time.time()
    
    # Initialize best solution
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    # Main loop
    while time.time() - start_time < budget_s - 0.1:
        t += 1
        
        # Annealing schedule
        T = T0 * (alpha ** t)
        
        # Mutation: add Gaussian noise, clamp, normalize
        sigma = sigma0 * (alpha ** (t / 200))
        h_new = h + np.random.normal(0, sigma, n_points)
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize to sum to 35 (the target)
        h_new = h_new * 35.0 / np.sum(h_new)
        
        # Evaluate
        C5_new = evaluate_C5(h_new)
        
        # Accept or reject
        if C5_new < best_C5 or np.random.rand() < np.exp(-(C5_new - best_C5) / T):
            h = h_new.copy()
            if C5_new < best_C5:
                best_h = h.copy()
                best_C5 = C5_new
    
    return (best_h.tolist(), best_C5, n_points)
