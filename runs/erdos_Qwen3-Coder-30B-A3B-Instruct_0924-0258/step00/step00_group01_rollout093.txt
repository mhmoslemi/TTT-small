import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    n_points = 47  # Using the given initial construction size if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if abs(np.sum(h) - n_points / 2.0) > 1e-6:
            h = h * (n_points / 2.0) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2.0) / np.sum(h)
    
    # Validation
    assert abs(np.sum(h) - n_points / 2.0) < 1e-6
    assert all(0 <= x <= 1 for x in h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    # Main optimization loop
    alpha = 0.1
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        # Compute overlaps using np.correlate, mode='full'
        # Corresponds to max_k ∫ h(x)(1 - h(x+k)) dx = max_k correlation(h, 1-h)
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # If no improvement for a while, reduce learning rate
        if no_improvement_count > max_no_improvement:
            alpha *= 0.95
            no_improvement_count = 0
            if alpha < 1e-6:
                break
        
        # Compute gradient
        # correlation of h with 1-h
        corr_h_1mh = signal.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr_h_1mh)
        
        # Compute gradient manually using convolution properties
        # Gradient w.r.t. h at index i 
        # = d/dh[i] Σ_j h[j] * (1 - h[(j+k)%n_points]) for k = max_k
        # = (1 - h[(i+k)%n_points]) - h[i] if k is fixed
        # We compute it properly via cross-correlation of (1-h) and h,
        # and cross-correlation of h and (1-h)
        # The gradient direction for deterministic improvement:
        # grad[i] = Σ_j h[j] * (1 - h[(j+k)%n_points]) - h[i] itself
        
        # More strictly, the gradient in direction of steepest descent of C₅:
        # Use Ф[h,k] = h * (1 - h_{k})
        # where h_{k}[i] = h[(i+k)%n_points]
        # ∂Ф[h,k]/∂h[i] = (1 - h[(i+k)%n_points]) - h[i] (approximation)
        # Let's try using the correlation derivative via convolution:
        # We get approximate gradient for each position i:
        
        # Using proper formulas:
        # When varying h[i], 
        # ∫ h(x)(1 - h(x+k)) dx = Σ h[j](1 - h[j+k]) 
        # For j = i, we get h[i](1 - h[i+k])
        # Gradient wrt h[i] = (1 - h[i+k]) - h[i]
        
        # So gradient[i] should sum across indices j for terms:
        # For each lag k, we compute:
        # For j = i we consider h[i] in overlap with 1-h[i+k]
        # So the total gradient is (1 - h[i+k] - h[i]) 
        # Where it's for j=0 to n_points-1, the overlap with 1-h[i]
        # But here we use the dependence of the k-th lag
        
        # Do correlation of (1-h) with h starting from different lags
        gradient = np.zeros(n_points)
        for k in range(0, n_points):
            # gradient for h[i] = (1 - h[(i+k)%n_points]) - h[i]  
            # = 1 - h[(i+k)%n_points] - h[i]
            h_shifted = np.roll(h, k)
            gradient += (1 - h_shifted) - h
            
        # Clip gradient to prevent huge steps
        gradient = np.clip(gradient, -1, 1)
        
        # Update h
        h += alpha * gradient
        
        # Project to [0,1]
        h = np.clip(h, 0, 1)
        
        # Maintain integral constraint: ∫ h(x) dx = 1
        current_sum = np.sum(h)
        target_sum = n_points / 2.0
        if abs(current_sum - target_sum) > 1e-6:
            scale_factor = target_sum / current_sum
            h = h * scale_factor
            
        # Optional escape from local minima 
        if np.random.rand() < 0.05:
            # Add small random noise
            noise = np.random.normal(0, 0.01, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            # Re-scale 
            current_sum = np.sum(h)
            if abs(current_sum - target_sum) > 1e-6:
                scale_factor = target_sum / current_sum
                h = h * scale_factor
    
    # Final validation and return
    assert abs(np.sum(best_h) - n_points / 2.0) < 1e-6
    assert all(0 <= x <= 1 for x in best_h)
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, len(best_h))

def compute_c5(h, dx):
    # Use direct computation for return accuracy
    corr = signal.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
