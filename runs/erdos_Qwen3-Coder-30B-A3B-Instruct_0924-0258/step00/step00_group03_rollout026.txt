import numpy as np
from scipy import signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create initial construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 1.0
    
    # Ensure constraints
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)  # Normalize to integral = 1
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    start_time = time.time()
    current_h = h.copy()
    iter_count = 0
    
    # Learning rate decay
    lr = 0.01
    decay_factor = 0.95
    
    while time.time() - start_time < budget_s - 1:
        # Compute convolution and find max lag
        conv = np.correlate(current_h, 1 - current_h, mode="full")
        k0 = np.argmax(conv)
        max_val = conv[k0]
        
        # Normalize indices to [-n+1, n-1]
        lags = np.arange(-n_points + 1, n_points)
        k0_shift = lags[k0]
        
        # Compute gradient with respect to h at k0
        grad = np.zeros_like(current_h)
        for i in range(n_points):
            # For each position, count contributions from all positions j where j + k0 = i
            for j in range(n_points):
                if j + k0_shift == i:
                    grad[i] += (1 - current_h[j])
                if i + k0_shift == j:
                    grad[i] -= current_h[i]
                
        # Update h
        current_h += lr * grad
        
        # Project onto feasible set using CVXPY
        # Minimize ||x - x0||^2 s.t. 0 <= x <= 1, sum(x) = n_points/2*dx
        x = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(x - current_h))
        constraints = [
            x >= 0,
            x <= 1,
            cp.sum(x) == n_points / 2.0
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        current_h = x.value
        
        # Evaluate new solution
        c5 = evaluate_overlap(current_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h.copy()
            
        iter_count += 1
        if iter_count % 100 == 0:
            lr *= decay_factor
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_overlap(h, dx):
    # Compute full convolution from scipy
    conv = np.correlate(h, 1-h, mode="full")
    # Find maximum value
    max_val = np.max(conv)
    return max_val * dx
