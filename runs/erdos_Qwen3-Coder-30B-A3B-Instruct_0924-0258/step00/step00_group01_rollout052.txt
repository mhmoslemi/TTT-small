import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction or create a uniform one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[0] = 1.0
        h[1] = 0.0

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure normalization constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h *= target_sum / current_sum
    
    # Initial evaluation
    def evaluate_C5(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    # Optimize by convex relaxation
    max_iter = 100
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 0.5:
            break
        
        current_C5 = evaluate_C5(h)
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
            
        # Identify the maximum overlap again
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_corr_val = corr[max_idx]
        
        # Set up convex optimization
        # Variables
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add a constraint for the maximum correlation
        # Create a matrix that represents the correlation constraint
        # using disjoint support approximations
        # We'll modify the correlation coefficients to be tight
        
        # Approximate the worst overlap constraint
        # This is a simplified convex approximation
        # Which sums over efficiently selected pairs
        
        # Compute all overlaps and find worst case
        worst_lag = max_idx - (n_points - 1)
        correlation_constraint = 0
        for i in range(n_points):
            j = i + worst_lag
            if 0 <= j < n_points:
                correlation_constraint += h_var[i] - h_var[i] * h_var[j]
        
        # Minimize max overlap as 1 variable
        C_var = cp.Variable()
        constraints.append(correlation_constraint <= C_var * dx)
        
        # Objective
        objective = cp.Minimize(C_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                new_h = h_var.value
                if np.all(new_h >= 0) and np.all(new_h <= 1):
                    h = new_h.copy()
                    h /= np.sum(h) * dx
                    h *= target_sum
        except Exception:
            pass
    
    # Final evaluation
    final_C5 = evaluate_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
