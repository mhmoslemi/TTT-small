from scipy import signal
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the overlap integral C5 for given h_values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Evaluate all correlations
    corr = signal.correlate(h_values, 1 - np.array(h_values), mode="full")
    # Scale by dx
    corr *= dx
    # Return maximum overlap
    return np.max(corr)
