import numpy as np
from scipy import signal
import time
import cvxpy as cp

def evaluate_erdos_solution(h_values, n_points):
    """Compute C5 bound from h_values"""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute all overlaps
    overlaps = signal.correlate(h, 1-h, mode="full")
    max_overlap = np.max(overlaps) * dx
    return max_overlap

    low = 0.0
    high = 0.38085
    threshold = 1e-4
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(h_init, n_points)
    
    while high - low > threshold and (time.time() - start_time) < budget_s * 0.95:
        mid = (low + high) / 2.0
        
        # Formulate feasibility problem
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2.0  # sum(h) * dx = 1
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            # For shift k, sum_{i} h[i] * (1 - h[i+k]) <= mid/dx
            expr = cp.sum(cp.multiply(h, 1 - cp.roll(h, k)))
            constraints.append(expr <= mid / dx)
        
        # Problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
            if prob.status == 'optimal':
                new_h = h.value
                if new_h is not None:
                    test_h = np.maximum(0, np.minimum(1, new_h))
                    test_c5 = evaluate_erdos_solution(test_h, n_points)
