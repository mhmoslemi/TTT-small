import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations for all lags
    full_corr = correlate(h, 1 - h, mode='full')
    # Extract the relevant part (the actual overlap integrals for k in [0,2])
    # For full correlation, the valid range is from -n+1 to n-1
    # We want indices corresponding to k from 0 to 2, which is from n-1 to 2*n-2
    # For k in [0,2], we have b = k/dx with k = 0..2, so b = 0 to 2/dx = 2*n/2 = n
    # So we want from index n-1 to 2*n-2 (inclusive)
    overlap_integrals = full_corr[n_points-1 : 2*n_points-1]
    C5 = np.max(overlap_integrals) * dx
    return C5

def project_h(h, n_points):
    """Project h onto feasible set: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust to ensure sum is exactly n_points/2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Simple adjustment: distribute the difference
        # Could be more sophisticated but this works for now
        h += diff / n_points
        h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, n_points, k_max):
    """Compute approximate gradient for given maximum contributing lag"""
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    # For contribution at lag k_max:
    # Contribution to h[i] is from term (1-h[i+k_max] - (1-h[j])) = -h[i+k_max] + h[j]
    # For each i, we want the derivative of C5 with respect to h[i]
    for i in range(n_points):
        # Involves lag k_max, we consider both h[i] and h[i+k_max] terms
        # If h[i] contributes to overlap when lag k is used, then it depends on h[i+k]
        # But actually, for the contribution to overlap integral via h[i](1-h[i+k_max])  
        # We have that for each position i, the gradient is:
        # - contribution from h_i = -d/dh_i [sum_over_j h_j (1-h_{j+k})] 
        # which gives - (1-h_{i+k_max}) = h_{i+k_max} -1
        # The contribution to h[i] in the overlap at lag k_max comes from (1-h[i+k_max])
        # But we should be thinking how the total overlap integral changes with h[i]
        
        # For causal interpretation of overlap integrals:
        # For lag k_max, integral = sum h[i] * (1 - h[i+k_max]) 
        # d/dh[i] = (1 - h[i+k_max])
        # It's not straightforward because the term at i appears in multiple overlap integrals
        # But we can consider derivatives as arising from the maximum contributing index
        
        if k_max >= 0:
            # When we have overlap at k_max, if i >= k_max then h[i] appears in the integral at
            # position i-k_max, contributing (1-h[i-k_max])
            # Overall, for each i : partial C5 / partial h[i] = 
            # sum over all lags k :
            #    (1-h[i-k]) if i >= k else 0  if overlap at k uses h[i]
            # This requires full careful computation of how derivative changes with h[i]  
            
            # For maximum contributing lag, we are simplifying:
            # The max overlap is sum h[i] * (1 - h[i+k_max])
            # By symmetry and chain rule approach:
            if i + k_max < n_points:
                grad[i] = - (1 - h[i + k_max]) 
            else:
                grad[i] = 0
        # Rest assumed zero or computed for all contributing lags similarly
            
    return grad

        full_corr = correlate(h, 1 - h, mode='full')
        overlap_integrals = full_corr[n_points-1 : 2*n_points-1]
        k_max = np.argmax(overlap_integrals)
        C5_new = overlap_integrals[k_max] * dx
        
        # Early terminate if no change
        if abs(C5_new - current_C5) < tol:

        # Update h using approximate gradient based on k_max
        grad = np.zeros(n_points)
