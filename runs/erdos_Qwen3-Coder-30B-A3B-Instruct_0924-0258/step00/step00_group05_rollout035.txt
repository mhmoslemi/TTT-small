import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define initial resolution and parameters
    n_low = 10
    n_high = 70
    n_points = n_low
    
    # Initialize h with initial_h_values if it exists
    try:
        h = np.array(initial_h_values)
        h = h[:n_points]
    except:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * n_points / 2
    
    # Ensure h is within bounds and sum constraint
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Initial temperature and cooling schedule
    T = 0.1
    cooling_factor = 0.99
    max_iter_per_level = 500
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h, n_points)
    
    # Multi-resolution SA
    levels = [n_low * (2**i) for i in range(int(np.log2(n_high/n_low)) + 1)]
    if levels[-1] != n_high:
        levels.append(n_high)
    
    for n in levels[1:]:  # Skip first level as it's already initialized
        if time.time() - start_time > budget_s - 1:
            break
            
        h_upsampled = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h)), h)
        h = h_upsampled.copy()
        
        for _ in range(max_iter_per_level):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_new = h.copy()
            
            # Perturb 
            noise = np.random.normal(0, T, n)
            h_new += noise
            
            # Project back to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Adjust to meet sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate
            C5_new = evaluate_C5(h_new, n)
            C5_old = evaluate_C5(h, n)
            
            # Acceptance probability
            if C5_new < C5_old or np.random.rand() < np.exp(-(C5_new - C5_old) / T):
                h = h_new.copy()
                if C5_new < best_C5:
                    best_h = h.copy()
                    best_C5 = C5_new
            
            # Cool down
            T *= cooling_factor
            
        n_points = n
    
    # Final validation
    result_C5 = evaluate_C5(best_h, len(best_h))
    return (best_h.tolist(), result_C5, len(best_h))

def evaluate_C5(h, n_points):
    dx = 2.0 / n_points
    h = np.array(h)
    # Compute full correlation with 1-h
    corr = signal.correlate(h, 1 - h, mode='full')
    # Full correlation length = 2*n_points - 1
    max_corr = np.max(corr) * dx
    return max_corr
