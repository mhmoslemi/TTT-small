import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    # This requires computing correlation of h with (1-h) and taking maximum
    corr = correlate(h, 1-h, mode='full')
    # The full correlation length is 2*n-1
    # Each entry is scaled by dx
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    # Project h to satisfy sum(h) * dx = 1
    # So sum(h) = 1/dx = n/2
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Scale to have the right sum
    h = h * target_sum / current_sum
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust again to ensure correct sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find largest element and increment it
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def perturb(h, temp, dx):
    # Perturb with Gaussian noise, then reproject
    n = len(h)
    noise = np.random.normal(0, temp, n)
    h_new = h + noise
    h_new = np.clip(h_new, 0, 1)
    h_new = project_to_constraints(h_new, dx)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with lower resolution
    n_low = 10
    n_high = 70
    n_points = n_low
    dx = 2.0 / n_points
    
    # Procedure: start from initial or random, refine, interpolate
    # Use multiple resolutions
    
    # Initialize
    if 'initial_h_values' in globals():

        h = h[:n_points]
    else:
        # Random initial
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Track best ever
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    last_c5 = best_c5
    
    # Simulated annealing parameters
    T = 0.1
    cooling = 0.99
    iter_per_temp = 500
    
    iter_count = 0
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for _ in range(iter_per_temp):
            if time.time() - start_time > budget_s:
                break
                
            h_new = perturb(h, T, dx)
            c5_new = evaluate(h_new, dx)
            
            delta = c5_new - last_c5
            if delta < 0 or np.random.uniform() < np.exp(-delta / T):
                h = h_new
                last_c5 = c5_new
                if last_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = last_c5
                    
            iter_count += 1
            
        # Cool down
        T *= cooling
        
        if T < 1e-6:
            # If low temp, try to interpolate to next resolution
            if n_points < n_high:
                # Interpolate to make h longer
                h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                                  np.linspace(0, 2, n_points), h)
                h = h_long
                n_points = 2 * n_points
                dx = 2.0 / n_points
                h = project_to_constraints(h, dx)
                last_c5 = evaluate(h, dx)
                T = 0.1
            else:
                break
    
    return best_h.tolist(), best_c5, len(best_h)
