import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    tolerance = 1e-4
    alpha = 0.1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Adjust to satisfy sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Evaluate initial solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, n_points)
    
    start_time = time.time()
    
    # Gradient descent loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        max_k = np.argmax(overlaps)
        max_val = overlaps[max_k]
        
        # Identify which lags contribute to max
        K_max = []
        for k in range(len(overlaps)):
            if abs(overlaps[k] - max_val) < 1e-8:
                K_max.append(k)
        
        # Gradient computation
        gradient = np.zeros(n_points)
        for k in K_max:
            for i in range(n_points):
                j = (i + k) % n_points
                gradient[i] += (1 - h[j] - (1 - h[i])) * dx
        
        # Update h
        h_new = h - alpha * gradient
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_C5 = evaluate_erdos_solution(h_new, n_points)
        
        # Accept improvement
        if new_C5 < best_C5:
            best_h = h_new.copy()
            best_C5 = new_C5
            h = h_new.copy()
        else:
            # Stagnation handling
            h = h_new.copy()
            
        # Adaptive step size
        if iteration > 0 and abs(new_C5 - best_C5) < tolerance:
            alpha *= 0.9
            if alpha < 1e-6:
                alpha = 1e-6
                
        # Perturb occasionally
        if iteration % 50 == 0 and iteration > 0:
            noise = np.random.normal(0, 1e-3, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (list(best_h), best_C5, n_points)
