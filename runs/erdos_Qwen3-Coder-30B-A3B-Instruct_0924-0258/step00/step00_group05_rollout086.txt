import numpy as np
from scipy import signal
import time
import math

def evaluate_c5(h):
    """Evaluate C5 bound for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h and (1-h)
    cross_corr = signal.correlate(h, 1 - h, mode='full')
    # Max value times dx
    return np.max(cross_corr) * dx

def project_h(h, n_points):
    """Project h to meet sum constraint and bounds"""
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Adjust to meet sum = n/2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    # Simple scaling approach
    if current_sum > 0:
        h *= (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def perturb_h(h, temp):
    """Apply perturbation with temperature control"""
    n = len(h)
    # Gaussian noise with variance = temp
    noise = np.random.normal(0, temp, n)
    h_new = h + noise
    return project_h(h_new, n)

def cooling_schedule(iteration, max_iter, initial_temp):
    """Adaptive cooling schedule"""
    return initial_temp * (0.99 ** iteration)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points_list = [10, 20, 40, 70]
    initial_temp = 0.1
    max_iter_per_level = 500
    
    # Start with either initial_h_values or create a random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = n_points_list[0]
        h = np.random.rand(n_points)
        h = project_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Multi-resolution optimization
    for level, n_points in enumerate(n_points_list):
        if time.time() - start_time > budget_s - 5:
            break
            
        if level > 0:
            # Interpolate from previous resolution
            old_h = h.copy()
            h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(old_h)), old_h)
            h = project_h(h, n_points)
        
        # Optimize at current resolution
        current_temp = initial_temp
        for iteration in range(max_iter_per_level):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_proposal = perturb_h(h, current_temp)
            c5_proposal = evaluate_c5(h_proposal)
            
            delta_c5 = c5_proposal - evaluate_c5(h)
            
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / current_temp):
                h = h_proposal
                if c5_proposal < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_proposal
                    
            current_temp = cooling_schedule(iteration, max_iter_per_level, initial_temp)
    
    # Final validation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
