import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 47  # Use the provided initial construction size if available
    max_iter = 10000
    alpha = 0.1
    decay = 0.95
    momentum = 0.9
    random_restart_prob = 0.05
    early_stop_iter = 50
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * n_points / 2
    else:
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * n_points / 2
    
    # Ensure bounds and sum constraint
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Initialize momentum
    v = np.zeros_like(h)
    
    start_time = time.time()
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlaps for all shifts
        corr = correlate(h, 1 - h, mode='full')
        # The overlap for each shift (k) starts from index n_points-1
        overlaps = corr[n_points-1:]
        k0 = np.argmax(overlaps)
        max_overlap = overlaps[k0]
        
        if max_overlap < best_C5:
            best_C5 = max_overlap
            best_h = h.copy()
            last_improvement = i
            
        # Compute gradient for the active shift k0
        grad = np.zeros(n_points)
        if k0 < n_points:
            for i_idx in range(n_points):
                if i_idx + k0 < n_points:
                    grad[i_idx] += 1 - h[i_idx + k0]
                if i_idx >= k0:
                    grad[i_idx] -= h[i_idx - k0]
        
        # Gradient step with momentum
        v = momentum * v - alpha * grad
        h_new = h + v
        
        # Clip and re-normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * n_points / 2
        
        # Accept step if it's better or with low probability
        h = h_new
        if np.random.random() < random_restart_prob:
            # Random restart
            h = np.ones(n_points) * 0.5
            h = h / np.sum(h) * n_points / 2
            
        # Early stopping
        if i - last_improvement > early_stop_iter:
            break
            
        # Adaptive learning rate
        if i % 10 == 0 and i > 0:
            alpha *= decay
            
    return (best_h, best_C5, n_points)
