import numpy as np

import time

def evaluate_erdos_solution(h_values, n_points):
    """Evaluate the overlap integral for a given step function h."""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute all correlations
    correlations = np.correlate(h, 1 - h, mode='full')
    # Take the maximum (over all shifts)
    max_correlation = np.max(correlations)
    return max_correlation * dx

                    corr += h_vars[i] * (1 - h_vars[j])
            constraints.append(corr <= mid / dx)

        prob = cp.Problem(cp.Minimize(0), constraints)
