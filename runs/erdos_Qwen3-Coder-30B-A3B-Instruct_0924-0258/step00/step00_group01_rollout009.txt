    start_time = time.time()

    def objective(h_vec):
        h_vec = np.maximum(0, np.minimum(1, h_vec))

        h_vec = h_vec * target_sum / np.sum(h_vec)
        # Compute overlaps
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
