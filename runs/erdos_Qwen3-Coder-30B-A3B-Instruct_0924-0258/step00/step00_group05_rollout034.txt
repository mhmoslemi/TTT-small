import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 70
        h = np.full(n_points, 0.5)
    
    # Normalize to satisfy sum constraint
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    # Parameters
    max_iter = 500
    alpha = 0.1
    tol = 1e-4
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current C5
        correlations = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlations) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient
        max_lags = np.where(correlations == np.max(correlations))[0]
        offset = len(h) - 1
        
        grad = np.zeros_like(h)
        for lag in max_lags:
            k = lag - offset  # Convert to actual lag
            if k >= 0:
                # Correlation is h[i] * (1 - h[i+k])
                # Gradient w.r.t h[i] is (1 - h[i+k]) - h[i+k] = 1 - 2*h[i+k]
                # Grad for h[i+k] is -h[i]
                for i in range(len(h)):
                    if i+k < len(h):
                        grad[i] += 1.0 * dx
                        if i+k < len(h):
                            grad[i+k] -= 1.0 * dx
            else:
                # Negative lag case
                for i in range(len(h)):
                    if i + k >= 0:
                        grad[i] += 1.0 * dx
                        if i+k >= 0:
                            grad[i+k] -= 1.0 * dx
        
        # Update step
        h_new = h - alpha * grad
        
        # Project onto feasible set
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize sum constraint
        current_sum = np.sum(h_new)
        if abs(current_sum - n_points / 2) > 1e-8:
            h_new = h_new * (n_points / 2) / current_sum
        
        # Check for convergence
        h = h_new
        if iteration > 0:
            if abs(c5 - best_c5) < tol:
                break
    
    # Final evaluation
    correlations = signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlations) * dx
    
    return (best_h.tolist(), final_c5, n_points)
