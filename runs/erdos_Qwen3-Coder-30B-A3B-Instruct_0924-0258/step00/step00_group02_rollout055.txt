import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback to a simple construction if no initial_h_values
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial h satisfies constraint
    h = h * target_sum / np.sum(h)
    
    # Helper function to compute C5
    def compute_C5(h):
        h_padded = np.concatenate([h, h])
        overlaps = []
        for k in range(n_points):
            overlap = np.sum(h * (1 - h_padded[k:k+n_points])) * dx
            overlaps.append(overlap)
        return max(overlaps)
    
    # Gradient-based optimization helper
    def compute_gradients(h, top_k_indices):
        n = len(h)
        grad = np.zeros(n)
        h_padded = np.concatenate([h, h])
        for k in top_k_indices:
            # Compute contribution to gradient for overlap k
            # d/dh_i [h_i * (1 - h_{i+k})] = 1 - h_{i+k} - h_i * delta_{i+k,j}
            # but we sum over i, so it's different for each parameter
            for i in range(n):
                grad[i] += (1 - h_padded[i+k]) * dx
        return grad
    
    # Main optimization loop
    best_C5 = compute_C5(h)
    best_h = h.copy()
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        if iteration % 20 == 0:
            # Periodically recompute dominant overlaps
            h_padded = np.concatenate([h, h])
            overlaps = []
            for k in range(n_points):
                overlap = np.sum(h * (1 - h_padded[k:k+n_points])) * dx
                overlaps.append(overlap)
            top_k_indices = np.argsort(overlaps)[-5:][::-1]
        
        # Compute gradient
        grad = np.zeros(n_points)
        h_padded = np.concatenate([h, h])
        for k in top_k_indices:
            for i in range(n_points):
                grad[i] += (1 - h_padded[i+k]) * dx
        
        # Gradient descent step with momentum
        velocity = momentum * velocity - learning_rate * grad
        h = h + velocity
        
        # Project to feasible set: [0,1] and sum to target_sum
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        
        # Compute new C5
        new_C5 = compute_C5(h)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h.copy()
            learning_rate = min(0.01, learning_rate * 1.01)
        else:
            learning_rate = max(0.0001, learning_rate * 0.999)
        
        iteration += 1
    
    return (best_h.tolist(), best_C5, n_points)
