import numpy as np
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        initial_h = initial_h_values.copy()
        n_points = len(initial_h)
    except NameError:
        # Fallback to default construction
        n_points = 70
        initial_h = np.ones(n_points) * 0.5
        initial_h[:n_points//2] = 1.0
        initial_h[n_points//2:] = 0.0

    # Normalize
    initial_h = initial_h * (n_points / 2.0) / np.sum(initial_h)
    h = initial_h.copy()
    
    # Optimizer settings
    num_iter = 3000
    alpha = 0.1
    decay = 1000.0
    lambda_proj = 0.9
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    def project(h_in):
        """Project onto feasible set"""
        h_proj = np.clip(h_in, 0, 1)
        h_proj = h_proj * (n_points / 2.0) / np.sum(h_proj)
        return h_proj
    
    # Main loop
    for it in range(num_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Fourier approach for overlap computation
        hf = fft(h)
        overlap_spectrum = np.abs(hf)**2
        overlap = ifft(overlap_spectrum).real
        
        # Find maximum overlap index
        max_idx = np.argmax(overlap)
        
        # delta vector
        delta = np.zeros(n_points)
        delta[max_idx] = 1.0
        
        # Compute gradient (simplified)
        # Since overlap is conv h * (1-h), gradient w.r.t. h at k* is:
        # (1-h) at position k* and -h at position k*+1 (circular)
        gradient = np.zeros_like(h)
        for i in range(n_points):
            k = (i - max_idx) % n_points
            gradient[i] += (1 - h[k])
            k = (i - max_idx - 1) % n_points
            gradient[i] -= h[k]
            
        # Update
        t = it + 1
        step_size = alpha * np.exp(-t / decay)
        h_new = h - step_size * gradient
        
        # Project back to feasibility
        h = project(h_new)
        
        # Maintain convexity/refinement
        h = h * lambda_proj + h * (1 - lambda_proj)
        
        # Evaluate and update best
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h):
    """Computed from correlation of h with (1-h)"""
    n = len(h)
    dx = 2.0 / n
    # correlate h with 1-h to get max overlap
    corr = np.correlate(h, 1 - h, mode='full')
    # max of all positions
    return np.max(corr) * dx
