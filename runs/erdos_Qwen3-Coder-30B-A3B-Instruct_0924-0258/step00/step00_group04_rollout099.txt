import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h_values):
    # Compute C5 using the overlap integral
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute all correlations
    corr = signal.correlate(h_values, 1 - h_values, mode='full')
    # Extract the overlapping parts
    lag_offset = n_points - 1
    overlaps = corr[lag_offset:lag_offset + n_points] * dx
    # Return maximum overlap
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42  # Use initial construction size if available
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
        assert np.all(h >= 0) and np.all(h <= 1)
        h_sum = np.sum(h)
        assert abs(h_sum - n_points / 2) < 1e-6
    else:
        # Default initialization
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Gradient descent loop
    max_iter = 50000  # High upper bound to avoid early termination by time
    iter_count = 0
    
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
            
        iter_count += 1
        
        # Compute overlaps for all lags
        corr = signal.correlate(h, 1 - h, mode='full')
        lag_offset = n_points - 1
        overlaps = corr[lag_offset:lag_offset + n_points] * dx
        k_star = np.argmax(overlaps)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_star) % n_points
            grad[i] = 1 - 2 * h[j]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            continue
        grad = grad / grad_norm
        
        # Line search
        alpha = 0.1
        max_backtracks = 10
        backtrack_factor = 0.8
        for _ in range(max_backtracks):
            h_new = h - alpha * grad
            # Project to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to satisfy constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_c5 = evaluate_erdos_solution(h_new)
            if new_c5 < best_c5:
                h = h_new
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                break
            elif alpha < 1e-6:
                break
            else:
                alpha *= backtrack_factor
                
    return (best_h.tolist(), best_c5, n_points)
