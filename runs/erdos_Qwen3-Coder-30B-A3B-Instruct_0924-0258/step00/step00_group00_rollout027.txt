import numpy as np
import time
from scipy import correlate

def evaluate_C5(h, dx):
    """Evaluate C5 using the provided method."""
    # Compute cross-correlation of h and (1-h) with full mode
    corr = correlate(h, 1-h, mode="full")
    # The maximum value is the one we want (the largest overlap)
    max_corr = np.max(corr) * dx
    return max_corr

def rescale_h(h, target_sum):
    """Rescale h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum == 0:
        h = np.ones_like(h) * (target_sum / len(h))
    else:
        h = h * (target_sum / current_sum)
    return h

def clip_and_rescale(h, target_sum):
    """Clip h to [0,1] and rescale to target sum."""
    h = np.clip(h, 0, 1)
    h = rescale_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45 if 'initial_h_values' in globals() else 60
    dx = 2.0 / n_points
    target_sum = n_points / 2  # For integral constraint
    
    # Initialize particles
    n_particles = 30
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = rescale_h(h_init, target_sum)
    else:
        # Create initial solution
        h_init = np.random.rand(n_points) * 0.5  # Start with low values
        h_init = rescale_h(h_init, target_sum)
        
    h_particles = [clip_and_rescale(h_init + np.random.normal(0, 0.05, n_points), target_sum) 
                   for _ in range(n_particles)]
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Track best solutions
    best_h = None
    best_C5 = float('inf')
    no_improvement_count = 0
    max_no_improvement = 100
    
    # PSO parameters
    w = 0.7  # Inertia weight
    c1 = c2 = 2.0  # Cognitive and social param
    max_iter = 20000
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 2:
        iter_count += 1
        for i in range(n_particles):
            # Update particle
            h = h_particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            personal_best = h_particles[i]
            global_best = best_h if best_h is not None else h_particles[0]
            
            v = w * v + c1 * r1 * (personal_best - h) + c2 * r2 * (global_best - h)
            velocities[i] = v
            
            # Update particle position
            h_new = h + v
            
            # Apply constraints
            h_new = clip_and_rescale(h_new, target_sum)
            
            # Evaluate
            C5 = evaluate_C5(h_new, dx)
            
            # Update personal best
            h_old_C5 = evaluate_C5(h, dx)
            if C5 < h_old_C5:
                h_particles[i] = h_new
            
            # Update global best
            if C5 < best_C5:
                best_C5 = C5
                best_h = h_new.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            if no_improvement_count > max_no_improvement:
                # Random perturbation
                h_particles[i] = clip_and_rescale(h_new + np.random.normal(0, 0.02, n_points), target_sum)
                no_improvement_count = 0
        
    # Final check with the best solution
    if best_h is not None:
        final_C5 = evaluate_C5(best_h, dx)
    else:
        # Fallback to the initial solution
        best_h = h_init
        final_C5 = evaluate_C5(best_h, dx)
        
    return (best_h.tolist(), final_C5, n_points)
