import numpy as np
from scipy import signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with provided initial construction or create a simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to project to feasible set
    def project_h(h_vec):
        # Optimize using CVXPY for projection onto constraints:
        # 0 <= h[i] <= 1, sum(h) = target_sum
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(h_var - h_vec))
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)
        return h_var.value
    
    def compute_C5(h_vec):
        # Compute C5 = max_k int h(x)(1-h(x+k)) dx
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # The convolution result corresponds to lags from -(n-1) to (n-1)
        # But our domain is [0,2], so we should consider all shifts between 0 and 2
        # So only consider the valid range shifts = [0, 2]; lag corresponds to x+k where k is shift
        # In the full correlation, real lags correspond to indices where i >= n-1 and i <= 2*n-2  
        # For the specific setup we have:
        # The main part we care about is the central part for shifts in [0,2]
        # Half length of h is nPoints/2, so 2-dx*1 is the shift
        # Actually, the shift k goes from 0 to 2
        # Over n_points samples, step size is 2/n_points = dx
        # So the lags we're interested in are those around 0 to 2, which means k = i*dx where i from 0 to 2/dx
        # The convolution has length 2*n_points - 1
        # It represents shifts from -n+1 to +n-1, so shift 0 to 2 corresponds to indices ~ n_points-1 to ~ n_points+1
        # Actually: correlation gives us indices from 0 to 2*n_points - 2
        # shift k == 0 corresponds to index n_points - 1
        # shift k == 2 corresponds to index n_points + 1 
        # We're interested in the part from 0 to 2 (shifts), so we want indices n_points-1 - (2/dx) to n_points-1 + (2/dx)
        # But since we're dealing with full length for shift -2 to 2 (centered), just take full to be safe.
        # Actually the only way to make sense of it is to think about simulation in discretized way:
        # h(x) x is in [0,2], discrete x values = i*dx with i=0 to n_points-1
        # We compute h(x) for an integer grid and for each x we evaluate at the correct values of h(x+k)
        # We're asked for the integral, and due to discrete time, this is a convolution scaled with dx
        # So the max over all shifts, from shift 0 to 2 with s = k*dx. We can just explore lags from d=0 to d=2/dx
        # No, we can consider evaluating integration over [0, 2] = 2*dx = 2*2/n_points = 4/n_points
        # That's still tricky. Let's just compute full convolution and get the max; the notion of shift is
        # how much we shift h backward w.r.t. itself, which means convolution with 1-h
        # The induced integral is a scalar. Want to maximize that
        # be precise: 
        # We want max_k int h(x)(1 - h(x+k)) dx = max_k called C(k)
        # So for each shift k in [0, 2], the integral is computed.
        # On a discretized mesh, h sampled as h[0],...,h[n_points-1] and h(x) is approximation on x = i*dx  
        # For a fixed k, we compute that integral as sum of h[i] * (1 - h[i + k/dx]) * dx  
        # This is just computing a discrete convolution approximation.
        # So we get a vector of values for different shift indices.
        # Full convolution puts the value at the same index as shift, but we must keep in mind.
        # Correlation via numpy gives: Convolution h * (1-h) gives scalar for each shift. 
        # Código que tiene sentido: conv = signal.correlate(h, 1-h, mode='full') 
        # Result has 2*n_points - 1 elements.
        # The origins are different: we're computing: 
        # sum over i of h[i] * (1 - h[i+k]) * dx for k in (-n+1, n-1) 
        # This is equivalent to the correlation of h with 1-h but in the FAST version 
        # It is tricky to properly select part of the convolution.
        # But it means: consider multiplying each h[i] with (1-h[i+shift]) for shift in 0:2
        # So in discrete setting, shift = 0 means we are looking at sums h[i](1-h[i]) 
        # shift = 1 means h[i](1-h[i+1]) and so on...
        # Position 0 is from shift = 0, and shift j range goes from -n+1 to n-1
        # The shift zero occurs at index n_points - 1, so we are trying to maximize from the middle
        # We want to find max_k over all possible genuine shifts in interval [0,2]
        # The range of shift we undersample is from 0 to 2 in real space.
        # On grid we have 0, dx, 2*dx, ..., dx*(n_points-1)  
        # So shifts correspond to 0 to 2, which means from 0 to 2/dx steps.
        # Let's check just the standard formula and take what matters.
        # Yes, Convolution relates to this via 
        # Integrate [0,2] h(x) (1-h(x+k)) dx = sum h[i] (1-h[i + k/dx]) * dx
        # The correlation convolution at lag k corresponds to sum h[i] (1-h[i + k]) * dx
        # Lag values are from -n+1 to n-1. The shift we care about is from 0 to 2 = 2/dx * dx
        # The matrix/vector representation is:
        # We are looking for:  
        # C(k) = integrate h(x) * (1 - h(x+k)) dx, where x in [0,2] 
        # So if h[] is on indices 0..n_points-1, and we go for shift k in [0,2]
        # h[i] is in original discrete scale, so i represents index at spatial location i*dx  
        # h[i+k/dx] corresponds to element at index [i + k/dx], but shifted in samples.
        # This is found by shifting right by k/dx samples.
        # In correlation (with indexing): h[i] (1 - h[i + shift])  
        # Shift = k/dx, but we have h[i] and h[i + shift] = h[i + k/dx]
        # So this convolution is correct  
        # Now the shift k ranges from [0,2] in continuous space, so that's from 0 to [2/dx] steps.
        # The correlation will give us value at index corresponding to certain shift.
        # Its inputs are h and (1-h) concatenated to full extent. 
        # the max will be over the first n_points (or 2*n_points-1) values that are affected by the 0-2 shift range.
        # In fact: the resulting vector C(k), after applying correlation of h and (1-h), is at indices from shift -n+1 to +n-1. 
        # But we are only interested in actual shifts in [0,2]
        # which means indices from 0 to 2/dx = 2* n_points/ 2 = n_points  
        # so index n_points - 1 = 0 shift, corresponding to maximum
        # Index n_points-1 + 2/dx = n_points-1 + n_points = 2*npoints-1  
        # But that's not right. Why?  Let's re-read.
        # Let's compute more directly:
        # shift k means integration from 0 to 2 of h(x) (1 - h(x+k))  
        # indices are from 0 to n_points-1 on the grid  
        # For shift k = 0: sum h[i] * (1-h[i])  
        # Sum over shift k = 0 to 2, computed via discretization  
        # For shift k:
        # Sum h[i] * (1 - h[i + k/dx]) 
        # Because shift k transforms x -> x+k (which is x -> x when in integral form)  
        # So if i = 0 to n_points -1
        # and h needs y = x+k = i*dx + k = (i*dx + k) 
        # which when discretized points means we look at the value h(index = (i*dx +k)/dx) = h(i + k/dx) 
        # But k ranges from [0,2], so k = 0 to 2, same as 0 to 2* n_points / 2  
        # we take steps 0 to n_points = 0 to 2/dx * 2/2 = n_points  
        # So convolution is equivalent, we just need correct analytical shift.
        # use full correlation.
        # Since for general problem we are dealing with correlation of signal with its complement shifted one
        # Index range for shift 0 to 2 spans:
        # Ck = sum h(i) * (1 - h(i+k/dx)) for all i, for shift k = 0 to 2 
        # We only care about the valid indices with small shift. Sometimes inside or offset.
        # We compute full correlation, which does h * (1-h) properly, but produces indexed relationships. 
        # For shift k = 0 the correlation at index n_points - 1
        # For shift k = 2, the correlation at index n_points + 1 
        # So we're in right range  
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # conv indexes from -n+1 to +n-1  
        # shift k = 0 is at index n_points -1  
        # shift k = 2 is at index n_points +1  
        # shift k in [0,2] -> index from n_points-1 to n_points+1
        # so actually we want indices from 0 to n_points + 1  
        # Actually integer recon version:
        # forward shifts are classified properly as positive indices in array, if right indexing.
        # Here in Python, full correlate of h, (1-h) gives discrete correlation of length 2*n-1 
        # shift k = 0 at index n-1 = n_points-1  
        # shift k = 2 located at n_points +1  
        # hence our shift k in [0,2] corresponds to indices in [n_points-1 : n_points+2] with shift between 0 to 2
        # On index len = 2*n-1  
        # Sample.shifted(i, k) gives i + k for discrete case  
        # If i = 0 to n-1, shift k from 0 to n_points -1  
        # But in real domain shift k = dx * k_sample  
        # Our constraint goes from 0 to 2  
        # In samples this corresponds to k_samples = 0 to 2/dx = 0 to n_points  
        # So start from index where shift 0=0 (which is n_points - 1)  
        # end at index where shift 2 = n_points (which is n_points + 1 - 1)  
        # So for shifts k from 0 to 2 in real space, neighbors in correlation
        # Are indices:  
        # shifts: from index n_points - 1 (shift=0) to n_points+1  (shift=2) 
        # in correlation we have index for shift = 0 to 2 * n_points - 2  
        # we want shifts 0..2
        # correlation indices 0..2*n_points-2  
        # from shift 0 to 2, positions  
        # 0 -> corr_idx = n_points -1  
        # 1 -> corr_idx = n_points  
        # 2 -> corr_idx = n_points + 1  
        # So interested range is corr index n_points - 1 to n_points + 1, but including this one
        # slice [n_points-1 : n_points+2]  
        values = conv[n_points-1 : n_points+2] # shift from 0 to 2, and have 3 values total
        c5 = np.max(values * dx)  
        return c5
    
    best_h = h.copy()
    best_c5 = compute_C5(h)
    iter = 0
    learning_rate = 0.01
    while True:
        current_time = time.time()
        if current_time - start_time > budget_s - 0.1:
            break
        
        # Compute the current C5
        c5 = compute_C5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
        # Find the k that gives the maximum integral
        conv = signal.correlate(h, 1 - h, mode='full')
        values = conv[n_points-1 : n_points+2] # shift from 0 to 2 in integer samples
        max_index = np.argmax(values * dx)  # This tells us which k gives max, where 0,1,2 for k=0,1,2
        k0 = max_index  # Since max_index is 0,1,2 for real shifts 0,1,2
        # print(f"Iter={iter} max_k={k0}")
        # But in continuous world k from 0 to 2 corresponds to shifts sampled at k0 samples  
        # so we need to identify the gradient w.r.t h for lag k0, and then step.
        # We look at max of discrete convolution
        
        # For gradient w.r.t. each h[i], 
        # the correlation h * (1-h) at position k0 (i.e. k0th lag in 0..2*n-2 indices)
        # D/dh[i] [ sum_j h[j] * (1 - h[j+k0]) ]  
        # = 1 - h[j + k0] (when j == i) 
        # And h[ j ] = 2h[ j ] (when j + k0 == i) = h[ i - k0 ]
        # Total:
        # if j = i => (1 - h[i+k0])  
        # if j = i - k0 => (h[i - k0])  
        # But the gradient is already computed by differentiate convolution w.r.t. its value:
        # That's part of the dX/dh = grad of convolution =  (1-h) * delta[shift] + h * delta[shift - k0]
        # Or take standard derivative: h*1 - h*(1-h) = h - (h-h^2) = h^2  
        # No! Use for convolution relationships. Too complicated. Use an indirect way
        # The most direct way is to use finite differences to get gradient estimations for h[i] regions  
        # Let's find pairs of variables that affect h_k0 given by mode='full', then bc correlation at index lags  
        # key idea: correlation of h with (1-h) is:
        # Evaluate G[i] = sum_j h[j] * (1 - h[j+i]) [conv] 
        # => dG[i]/dh[j] = 1 - h[j-i] if j == i  
        # This seems confusing in indexing.  
        # Gradients appear via this form: 
        # G[i] = sum_j h[j] * (1 - h[j + i]) =>  dG[i]/dh[j] = (1 - h[j + i]) if j == 0  
        # That's not correct. Let's think differently.
        # Let's reuse the convolution in a behavioral sense.
        # Problem is: we compute a * b = c, so c[i] = sum_j a[j] b[i - j]
        # so dC[i]/da[j] = b[i - j]  
        # Now for a = h  
        # b = 1 - h  
        # C[lag] = sum_{j} h[j] (1 - h[j + lag])  
        # so dC[lag]/dh[j] = (1 - h[j + lag]) if j = j 
        # Which is just incorrect approach.
        # Or: the "err" is d/dh[u]. sum_k h[k] * (1 - h[k + lag])  = 1 - h[u+lag] 
        # So dC[lag]/dh[j] = 1 - h[j + lag] 
        # This is wrong again.
        
        # A simpler approach:
        # We'll compute finite differences of C5 w.r.t. each h element.
        # This gets the direction of steepest descent
        eps = 1e-4  
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c_plus = compute_C5(h_plus)
            grad[i] = (c_plus - c5) / eps
            
        # Now update h with gradient descent step.
        h = h - learning_rate * grad
        h = project_h(h)
        
        # Adaptively decrease step size:
        iter += 1
        if iter % 100 == 0:
            learning_rate *= 0.95
            
    return (best_h, best_c5, n_points)
