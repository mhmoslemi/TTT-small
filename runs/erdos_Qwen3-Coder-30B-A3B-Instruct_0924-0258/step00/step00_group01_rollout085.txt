import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 47
        h = np.ones(n_points) * 0.5

    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Validate bounds
    h = np.clip(h, 0, 1)
    
    # Optimization parameters
    max_iter = 10000
    alpha = 0.1
    decay = 0.95
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    start_time = time.time()
    
    # Project h to meet sum constraint
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if abs(h_sum - n_points / 2) > 1e-6:
            # Adjust proportionally
            scale = (n_points / 2) / h_sum
            h_vec = h_vec * scale
        return h_vec
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlaps for all shifts
        overlap_full = np.correlate(h, 1 - h, mode='full')
        # The correlation length is 2*n_points - 1, but we only care about shifts
        # from 0 to n_points-1 (the actual shift values)
        c5 = np.max(overlap_full)
        
        # If we found a better solution, save it
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = i
            
        # Find shift that gives the max overlap
        max_k = np.argmax(overlap_full)
        # The actual shift k is in [0, n_points-1] range
        # overlap_full gives k from -n+1 to n-1, so actual shift
        # for i-th element is (max_k - (n_points - 1))
        k0 = max_k - (n_points - 1)
        
        # Compute gradient of overlap w.r.t h[i] for this k0
        grad = np.zeros(n_points)
        if k0 >= 0:
            # h[i] contributes to k0 shift when h[i+k0] term exists
            for j in range(n_points - k0):
                grad[j] += 1 - h[j + k0]
            # Additionally, term 1-h[i] contributes to shift k0
            for j in range(k0, n_points):
                grad[j] -= h[j - k0]
        else:
            k0_abs = -k0
            # h[i] contributes to k0 when h[i-k0] exists (but k0 < 0, so i-k0 is i+|k0|)
            for j in range(n_points - k0_abs):
                grad[j] -= h[j + k0_abs]
            # Add the other part
            for j in range(k0_abs, n_points):
                grad[j] += 1 - h[j - k0_abs]

        # Update h with gradient descent
        h_new = h - alpha * grad
        
        # Project to [0,1] and maintain sum
        h_new = project_h(h_new)
        
        # Apply momentum
        h = 0.9 * h + 0.1 * h_new
        
        # Random restart with small probability
        if i % 50 == 0 and np.random.random() < 0.05:
            # Random small perturbation
            pert = np.random.normal(size=n_points) * 0.01
            h = project_h(h + pert)
        
        # Reduce step size if no improvement
        if i - last_improvement > patience:
            alpha *= decay
            last_improvement = i
    
    best_h = project_h(best_h)
    # Final C5 evaluation
    final_overlap = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_overlap)
    
    # Ensure final values are within bounds
    final_h = np.clip(best_h, 0, 1)
    
    return (final_h, final_c5, n_points)
