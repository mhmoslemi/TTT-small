import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral for a step function h."""
    # Compute the full correlation integral
    corr = correlate(h, 1 - h, mode="full")
    # The maximum value among all lags
    max_val = np.max(corr) * dx
    return max_val

def is_feasible(h, dx):
    """Check if h is feasible: 0 <= h <= 1 and integral = 1"""
    if np.any(h < 0) or np.any(h > 1):
        return False
    return abs(np.sum(h) * dx - 1.0) < 1e-10

def normalize_h(h, dx):
    """Normalize h to ensure its integral equals 1"""
    total = np.sum(h) * dx
    if abs(total - 1.0) > 1e-10:
        h = h / total * (1.0 / dx)
    return np.clip(h, 0, 1)

def mutate_h(h, dx, rate=0.1):
    """Mutate h by adding bounded random noise and re-normalizing"""
    n = len(h)
    mask = np.random.rand(n) < rate
    delta = (np.random.rand(n) - 0.5) * 0.2
    h_new = h + mask * delta
    return normalize_h(h_new, dx)

def crossover_h(h1, h2):
    """Blend two h vectors"""
    n = len(h1)
    alpha = np.random.rand()
    h_new = alpha * h1 + (1 - alpha) * h2
    return h_new

def tournament_selection(population, fitnesses, k=5):
    """Select an individual via tournament selection"""
    idx = np.random.choice(len(population), size=k)
    best_idx = idx[np.argmin([fitnesses[i] for i in idx])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    max_generations = 1000
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        # Start with the parent and its neighbors
        h0 = normalize_h(np.array(initial_h_values), dx)
        population.append(h0)
        for _ in range(population_size - 1):
            h = mutate_h(h0, dx, rate=0.3)
            population.append(h)
    else:
        # Seed with a uniform distribution
        h0 = np.ones(n_points) * 0.5
        population.append(h0)
        for _ in range(population_size - 1):
            h = mutate_h(h0, dx, rate=0.2)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for entire population
        fitnesses = []
        for h in population:
            fit = evaluate_h(h, dx)
            fitnesses.append(fit)
            
        # Update best solution
        for i, fit in enumerate(fitnesses):
            if fit < best_c5:
                best_c5 = fit
                best_h = population[i].copy()
                
        # Create new population
        new_population = []
        sorted_indices = np.argsort(fitnesses)
        
        # Keep top 20% as elite
        elite_count = max(1, population_size // 5)
        for i in range(elite_count):
            new_population.append(population[sorted_indices[i]].copy())
            
        # Generate rest via selection, crossover, and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses, k=5)
            if np.random.rand() < crossover_rate:
                parent2 = tournament_selection(population, fitnesses, k=5)
                child = crossover_h(parent1, parent2)
            else:
                child = parent1.copy()
                
            # Mutate
            child = mutate_h(child, dx, rate=mutation_rate)
            new_population.append(child)
            
        population = new_population
    
    # Final evaluation of best solution
    if best_h is not None:
        best_c5 = evaluate_h(best_h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
