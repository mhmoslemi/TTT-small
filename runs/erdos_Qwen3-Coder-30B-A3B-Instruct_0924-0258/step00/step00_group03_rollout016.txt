from scipy.signal import correlate

def evaluate_C5(h):
    """Evaluate the maximum overlap integral for given h"""
    # Compute all correlations
    corr = correlate(h, 1-h, mode="full")
    # Find maximum overlap
    max_overlap = np.max(corr)
    return max_overlap
