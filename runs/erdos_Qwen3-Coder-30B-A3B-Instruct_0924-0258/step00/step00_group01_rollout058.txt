import numpy as np
from scipy.optimize import minimize
import time


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a simple initial guess
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[0] = 0.3
        h[1] = 0.7
        h[-1] = 0.3
        h[-2] = 0.7
        h = h * (n_points / 2) / np.sum(h)
        
    dx = 2.0 / n_points
    start_time = time.time()
    
    def compute_C5(h_vals):
        # Compute the full correlation of h with (1-h)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # The maximum overlap is the maximum of this correlation
        return np.max(corr) * dx
    
    def obj_func(h_vals):
        return compute_C5(h_vals)
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) * dx - 1.0
    
    def gradient(h_vals):
        # Approximate gradient by finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vals)
        base_val = obj_func(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            elif h_plus[i] < 0:
                h_plus[i] = 0
            grad[i] = (obj_func(h_plus) - base_val) / eps
        return grad
    
    # Optimization using projected gradient descent
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Initialize step size
    alpha = 0.01
    
    # Main optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1.0:
        iter_count += 1
        # Compute gradient
        grad = gradient(best_h)
        
        # Step in neg. gradient direction
        new_h = best_h - alpha * grad
        
        # Project onto feasible set [0, 1]
        new_h = np.clip(new_h, 0, 1)
                
        # Project to satisfy sum constraint
        current_sum = np.sum(new_h) * dx
        if abs(current_sum - 1.0) > 1e-6:
            # Adjust so sum is exactly 1.0, preserving relative positions
            adjustment = (1.0 - current_sum) / n_points
            new_h += adjustment
            
        # Ensure valid values
        new_h = np.clip(new_h, 0, 1)
        
        # Check new C5
        new_C5 = compute_C5(new_h)
        if new_C5 < best_C5:
            best_h = new_h.copy()
            best_C5 = new_C5
            alpha = min(alpha * 1.1, 0.1)  # Increase step size
        else:
            alpha *= 0.5  # Decrease step size
            
        # Early stopping
        if abs(best_C5 - 0.3807) < 1e-6:
            break
    
    return (best_h.tolist(), best_C5, n_points)
