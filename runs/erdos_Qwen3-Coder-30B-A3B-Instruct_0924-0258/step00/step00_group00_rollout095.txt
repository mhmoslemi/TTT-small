import numpy as np
import time
from scipy import signal

def evaluate_c5(h, dx):
    """Evaluate C5 for given h and dx"""
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # The effective range is from -len(h)+1 to len(h)-1
    # But we want max over valid lags
    max_corr = np.max(corr)
    return max_corr * dx

def rescale_to_unit_sum(h, target_sum=1.0):
    """Rescale h to have unit sum"""
    current_sum = np.sum(h)
    if np.abs(current_sum) < 1e-10:
        h.fill(target_sum / len(h))
    else:
        h *= (target_sum / current_sum)
    return h

def gen_random_h(n_points, target_sum=1.0):
    """Generate a random h with target sum"""
    h = np.random.random(n_points)
    return rescale_to_unit_sum(h, target_sum)

def mutate_h(h, mutation_delta=0.1):
    """Mutate h by adding random changes and rescaling"""
    new_h = h.copy()
    # Add small random changes
    for i in range(len(new_h)):
        if np.random.random() < 0.1:  # 10% chance per gene
            new_h[i] += np.random.uniform(-mutation_delta, mutation_delta)
    # Clip to [0, 1]
    new_h = np.clip(new_h, 0, 1)
    # Rescale to ensure unit sum
    return rescale_to_unit_sum(new_h)

def crossover_h(parent1, parent2, crossover_point=None):
    """One point crossover"""
    if crossover_point is None:
        crossover_point = np.random.randint(1, len(parent1))
    child = parent1.copy()
    child[crossover_point:] = parent2[crossover_point:]
    return child

def evaluate_population(population, dx):
    """Evaluate all candidates in population"""
    fitnesses = []
    for h in population:
        fitness = evaluate_c5(h, dx)
        fitnesses.append(fitness)
    return np.array(fitnesses)

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select parents via tournament selection"""
    selected = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size)
        tournament_fitness = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitness)]
        selected.append(population[winner_idx].copy())
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 45
    dx = 2.0 / n_points
    target_sum = 1.0
    population_size = 50
    generations = 200
    mutation_rate = 0.2
    crossover_rate = 0.7
    start_time = time.time()
    
    # Initialize population
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        parent = np.clip(parent, 0, 1)
        parent = rescale_to_unit_sum(parent)
    else:
        parent = gen_random_h(n_points, target_sum)
    
    population = [parent.copy()]
    # Generate random candidates
    while len(population) < population_size:
        # Perturb parent
        mutated = mutate_h(parent.copy())
        population.append(mutated)
        # Add completely random samples
        if len(population) < population_size:
            random_h = gen_random_h(n_points, target_sum)
            population.append(random_h)
    
    best_h = parent.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Begin GA loop
    for generation in range(generations):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Evaluate current population
        fitnesses = evaluate_population(population, dx)
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Selection
        selected = tournament_selection(population, fitnesses, tournament_size=3)
        
        # Crossover
        new_population = [best_h.copy()]
        for i in range(1, len(selected), 2):
            if i+1 < len(selected) and np.random.random() < crossover_rate:
                child1 = crossover_h(selected[i], selected[i+1])
                child2 = crossover_h(selected[i+1], selected[i])
            else:
                child1 = selected[i].copy()
                child2 = selected[i+1].copy()
            new_population.extend([child1, child2])
            
        # Mutation
        for i in range(1, len(new_population)):
            if np.random.random() < mutation_rate:
                new_population[i] = mutate_h(new_population[i])
        
        # Replace worst with new population
        for i in range(len(population)):
            if i < len(new_population):
                population[i] = new_population[i]
            else:
                population[i] = gen_random_h(n_points, target_sum)
    
    # Final validation  
    best_h = np.clip(best_h, 0, 1)
    best_h = rescale_to_unit_sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    # Make sure result is not corrupted
    if np.isnan(final_c5) or np.isinf(final_c5):
        final_c5 = 1.0
        best_h = parent.copy()
    
    return (best_h.tolist(), final_c5, n_points)
