import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load initial construction if available, otherwise create one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        # Binary-like construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Normalize so that sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iteration counter for escape mechanism
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s:
        # Compute correlations
        full_correlation = correlate(h, 1 - h, mode='full')
        # Extract cross-correlations for lags 0 to n_points-1
        cross_correlations = np.array([
            full_correlation[k + n_points - 1] * dx 
            for k in range(n_points)
        ])
        
        # Find maximum
        current_c5 = np.max(cross_correlations)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Stop if no improvement for 20 iterations
        if no_improvement_count >= 20:
            # Add perturbation
            h += 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
            continue
            
        # Gradient computation for maximum cross-correlations
        max_cross = np.max(cross_correlations)
        # Find all lags where max is achieved
        max_indices = np.where(np.abs(cross_correlations - max_cross) < 1e-10)[0]
        
        # If we had no gradient to make a better point, escape
        if len(max_indices) == 0:
            h += 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            continue
            
        gradient = np.zeros(n_points)
        
        for k in max_indices:
            if k == 0:
                # Only h[i] * (1 - h[i]) terms
                for i in range(n_points):
                    gradient[i] += (1 - h[i])
            else:
                # h[i] * (1 - h[i+k]) terms
                for i in range(n_points - k):
                    gradient[i] += (1 - h[i + k])
                    if i + k < n_points:
                        gradient[i + k] -= h[i]
        
        # Line search
        step_sizes = [0.1, 0.25, 0.5]
        best_candidate_h = h.copy()
        best_candidate_c5 = current_c5
        
        for alpha in step_sizes:
            h_candidate = h - alpha * gradient
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Evaluate new C5
            full_corr = correlate(h_candidate, 1 - h_candidate, mode='full')
            cross_corr = np.array([
                full_corr[k + n_points - 1] * dx
                for k in range(n_points)
            ])
            new_c5 = np.max(cross_corr)
            
            if new_c5 < best_candidate_c5:
                best_candidate_c5 = new_c5
                best_candidate_h = h_candidate.copy()
        
        # If improvement found, accept it
        if best_candidate_c5 < current_c5:
            h = best_candidate_h.copy()
        else:
            # If no descent, possibly evolve by small random change
            h += 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final check on best solution
    final_correlation = correlate(best_h, 1 - best_h, mode='full')
    final_cross_corr = np.array([
        final_correlation[k + n_points - 1] * dx
        for k in range(n_points)
    ])
    final_c5 = np.max(final_cross_corr)

    return (best_h.tolist(), final_c5, n_points)
