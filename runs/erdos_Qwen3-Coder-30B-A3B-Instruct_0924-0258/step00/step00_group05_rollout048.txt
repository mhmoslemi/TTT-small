import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise start with uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure sum(h) * dx = 1 (i.e., sum(h) = n_points / 2)
    h = h / np.sum(h) * (n_points / 2)
    
    def evaluate_c5(h_vals):
        # Compute overlap integrals for all lags
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        # Only take the valid correlations (same as using mode='valid' but correctly indexed)
        full_corr = corr[len(h_vals)-1:]
        return np.max(full_corr) * dx
    
    def project_h(h_vals):
        # Ensure h_vals are in [0, 1]
        h_vals = np.clip(h_vals, 0, 1)
        # Adjust to ensure sum(h_vals) = n_points / 2
        current_sum = np.sum(h_vals)
        if current_sum != 0:
            h_vals = h_vals * (n_points / 2) / current_sum
        return h_vals
    
    # Evaluate initial solution
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Gradient descent parameters
    alpha = 0.1
    max_iter = 500
    improvement_threshold = 1e-4
    stagnation_count = 0
    max_stagnation = 20
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current overlap integrals for all lags
        corr = correlate(h, 1 - h, mode='full')
        full_corr = corr[len(h)-1:]
        max_lags = np.where(full_corr == np.max(full_corr))[0]
        
        # Compute gradient
        grad = np.zeros_like(h)
        for k in max_lags:
            # The contribution of lag k to the gradient
            # [h[i] * (1 - h[i + k])] for i in [0, n_points)
            # So gradient wrt h[i] is (1 - h[i + k] - (1 - h[i])) = h[i] - h[i + k]
            # For the actual integration, we consider both h[i] and (1 - h[i + k])
            # gradient[i]: contrib from h[i] * (1 - h[i + k]) => 1 - h[i + k]
            # gradient[i]: contrib from (1 - h[i]) * h[i + k] => -h[i + k]
            # Thus: g[i] = (1 - h[i + k]) - h[i + k] - (1 - h[i]) = h[i] - 2*h[i + k] + h[i + k] - 1 = h[i] - h[i + k] - 1
            # Actually, let's carefully recompute it:
            # The integral is sum_i h[i] * (1 - h[i + k]) = sum_i h[i] - h[i] * h[i + k]
            # Gradient for h[i]: (1 - h[i + k]) - h[i + k] = 1 - 2 * h[i + k]
            # We use np.roll to compute h[i + k] with wraparound
            shifted_h = np.roll(h, k)
            grad = grad + (1 - 2 * shifted_h)
        grad = grad * dx
        
        # Update h using gradient descent
        h_new = h - alpha * grad
        
        # Project onto feasible region
        h_new = project_h(h_new)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new)
        
        # Check improvement
        if new_c5 < best_c5:
            h = h_new
            current_c5 = new_c5
            best_h = h_new.copy()
            best_c5 = current_c5
            alpha = min(alpha * 1.1, 1.0)  # Increase step size on improvement
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                # Reset with random perturbation
                h_noise = h + np.random.normal(0, 0.01, n_points)
                h_noise = project_h(h_noise)
                h_new = h_noise
                new_c5 = evaluate_c5(h_new)
                if new_c5 < best_c5:
                    h = h_new
                    current_c5 = new_c5
                    best_h = h_new.copy()
                    best_c5 = current_c5
                # Reset stagnation tracker
                stagnation_count = 0
                alpha *= 0.9  # Decrease step size
            else:
                h = h_new
                current_c5 = new_c5

    return (best_h.tolist(), best_c5, n_points)
