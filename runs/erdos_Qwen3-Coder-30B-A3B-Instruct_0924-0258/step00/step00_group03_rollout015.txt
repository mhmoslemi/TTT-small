import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h):
    """Evaluate the maximum overlap integral for the given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations for different shifts
    corr = correlate(h, 1 - h, mode='full')
    # Compute max over all shifts
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy the constraints: sum(h) * dx = 1."""
    dx = 2.0 / n_points
    h = np.clip(h, 0, 1)
    # Scale so that sum(h) * dx = 1, i.e., sum(h) = n_points / 2
    scale_factor = (n_points / 2) / np.sum(h)
    h = h * scale_factor
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    # Start with initial solution
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        # Create a reasonable initial solution, e.g., uniform + small perturbation
        h_current = np.ones(n_points) * 0.5
        h_current = project_to_feasible(h_current, n_points)
    
    # Validate initial solution
    assert np.isclose(np.sum(h_current) * dx, 1.0), "Initial h does not sum to 1"
    assert np.all((h_current >= 0) & (h_current <= 1)), "Initial h values out of bounds"
    
    best_h = h_current.copy()
    best_C5 = evaluate_C5(h_current)
    
    # Simulated annealing parameters
    T0 = 1.0
    alpha = 0.999
    sigma0 = 0.1
    t = 0
    start_time = time.time()
    budget_s = min(budget_s, 600)
    
    while time.time() - start_time < budget_s:
        t += 1
        T = T0 * (alpha ** t)
        sigma = sigma0 * (alpha ** (t / 200))
        
        # Generate candidate solution
        h_candidate = h_current + np.random.normal(0, sigma, n_points)
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        # Evaluate candidate
        C5_candidate = evaluate_C5(h_candidate)
        
        # Accept or reject
        if C5_candidate < best_C5:
            # Always accept if better
            h_current = h_candidate.copy()
            best_C5 = C5_candidate
            best_h = h_candidate.copy()
        else:
            # Accept with probability based on temperature
            delta_E = C5_candidate - best_C5
            if np.random.rand() < np.exp(-delta_E / T):
                h_current = h_candidate.copy()
    
    return (best_h.tolist(), best_C5, n_points)
