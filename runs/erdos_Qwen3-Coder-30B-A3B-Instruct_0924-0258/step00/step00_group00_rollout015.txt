import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        # Create a reasonable initial guess
        n_points = 45
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h[mid:] = 0.0
        # Add small random noise
        h += 0.01 * np.random.randn(n_points)
        h = np.clip(h, 0, 1)
    
    # Ensure the constraint sum(h) * dx = 1
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    iter_no_improve = 0
    max_iter_no_improve = 20
    
    while time.time() - start_time < budget_s - 0.1:
        # Compute cross-correlations
        full_correlation = correlate(h, 1 - h, mode='full')
        cross_correlations = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            cross_correlations[k] = full_correlation[k] * dx
            
        # Find maximum C5 (takes into account all elements)
        max_c5 = np.max(cross_correlations)
        
        # Find indices where maximum occurs
        max_indices = np.where(np.abs(cross_correlations - max_c5) < 1e-10)[0]
        
        # Compute gradients for all max_indices
        grad = np.zeros(n_points)
        for k_idx in max_indices:
            # In the full correlation output, k_idx goes from 0 to 2*n_points-2
            # The actual k value is the offset from middle point
            # We shift for the cross-correlation definition
            if k_idx < n_points - 1:
                # k is negative
                k = k_idx - (n_points - 1)
                # This case is tricky in indexing, skip with safe default  
                continue
            elif k_idx == n_points - 1:
                # k = 0
                for i in range(n_points):
                    grad[i] += (1 - h[i])
            else:
                # k = k_idx - (n_points - 1) which is positive
                k = k_idx - (n_points - 1)
                if k <= 0:
                    continue
                for i in range(n_points - k):
                    grad[i] += (1 - h[i + k])
                    if i + k < n_points:
                        grad[i + k] -= h[i]
        
        # Construct search directions and line search
        best_alpha = 0.25
        best_h_candidate = h.copy()
        best_new_c5 = max_c5
        
        for alpha in [0.1, 0.25, 0.5]:
            h_candidate = h - alpha * grad
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            new_c5 = evaluate_erdos_solution(h_candidate)
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_alpha = alpha
                best_h_candidate = h_candidate.copy()
                
        # Accept change if it improves
        if best_new_c5 < best_c5:
            h = best_h_candidate
            best_c5 = best_new_c5
            best_h = h.copy()
            iter_no_improve = 0
        else:
            iter_no_improve += 1
            
        # Escape local minima
        if iter_no_improve >= max_iter_no_improve:
            noise_scale = 0.01 * (1 + np.random.rand())
            h = h + noise_scale * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            iter_no_improve = 0

    return (best_h.tolist(), best_c5, n_points)


def evaluate_erdos_solution(h):
    """Evaluates C5 using provided method or considers existing implementation"""
    # Use the appropriate operation here to match the required problem
    # Use numpy correlate along with proper indexing
    
    if 'initial_h_values' in globals():

    max_corr = np.max(correlation) * dx
