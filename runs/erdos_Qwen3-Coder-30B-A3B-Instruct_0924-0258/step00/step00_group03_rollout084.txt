import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the overlap integral for a given h."""
    # Compute the full correlation with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # The maximum value is at the center, but we need to consider all lags
    # The maximum of correlation values (excluding trivial 0)
    max_corr = np.max(corr)
    return max_corr * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.ones_like(h) * (target_sum / len(h))
    return h * (target_sum / current_sum)

def enforce_constraints(h, target_sum):
    """Enforce h in [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    return normalize_h(h, target_sum)

def mutate_h(h, mutation_rate=0.1):
    """Mutate h by adding bounded random noise and re-enforcing constraints."""
    # Create noise
    noise = np.random.normal(0, mutation_rate * 0.1, len(h))
    h_new = h + noise
    # Enforce constraints
    h_new = enforce_constraints(h_new, np.sum(h))
    return h_new

def crossover_h(h1, h2):
    """Perform crossover between two h vectors."""
    crossover_point = np.random.randint(1, len(h1))
    h_new = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    # Enforce constraints
    h_new = enforce_constraints(h_new, np.sum(h1))
    return h_new

def evaluate_fitness(h, dx):
    """Evaluate fitness as the negative of overlap (since lower is better)."""
    return -evaluate_overlap(h, dx)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.ones(n_points) * 0.5  # fallback: uniform
    
    # Population size
    population_size = 50
    population = [initial_h.copy()]
    
    # Generate initial population
    for _ in range(population_size - 1):
        h = mutate_h(initial_h.copy(), mutation_rate=0.05)
        population.append(h)
    
    # Main optimization loop
    best_h = initial_h.copy()
    best_fitness = evaluate_fitness(initial_h, dx)
    
    generation = 0
    while time.time() - start_time < budget_s:
        generation += 1
        # Evaluate fitness for population
        fitnesses = [evaluate_fitness(h, dx) for h in population]
        
        # Track best solution
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_fitness:
            best_fitness = fitnesses[best_idx]
            best_h = population[best_idx].copy()
        
        # Selection: tournament
        new_population = []
        tournament_size = max(2, population_size // 10)
        
        for _ in range(population_size):
            # Select tournament participants randomly
            idxs = np.random.choice(population_size, tournament_size, replace=False)
            selected_fitnesses = [fitnesses[i] for i in idxs]
            winner_idx = idxs[np.argmin(selected_fitnesses)]
            new_population.append(population[winner_idx].copy())
        
        # Crossover and mutation
        offspring = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                parent1, parent2 = new_population[i], new_population[i+1]
                if np.random.rand() < 0.8:
                    child = crossover_h(parent1, parent2)
                else:
                    child = parent1.copy()
                # Mutate with low probability
                if np.random.rand() < 0.1:
                    child = mutate_h(child, mutation_rate=0.05)
                offspring.append(child)
            else:
                # Odd population: just mutate
                child = new_population[i].copy()
                if np.random.rand() < 0.1:
                    child = mutate_h(child, mutation_rate=0.05)
                offspring.append(child)
        
        # Replace worst with new offspring
        # Fitnesses of existing population
        existing_fitnesses = [evaluate_fitness(h, dx) for h in new_population]
        # Sort to find worst
        worst_indices = np.argsort(existing_fitnesses)[-len(offspring):]
        for i, (worst_idx, child) in enumerate(zip(worst_indices, offspring)):
            new_population[worst_idx] = child
            
        population = new_population
        if time.time() - start_time > budget_s - 1:  #预留一点时间处理
            break
    
    # Final evaluation
    final_c5 = evaluate_overlap(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
