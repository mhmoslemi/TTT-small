import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    n_points = 45
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h += 0.01 * np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    
    # Feasibility check
    assert abs(np.sum(h) - n_points/2) < 1e-6
    assert all(0 <= h_i <= 1 for h_i in h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    iter_without_improvement = 0
    max_iter_without_improvement = 20
    
    while time.time() - start_time < budget_s - 0.1:
        # Compute cross-correlations
        full_correlation = correlate(h, 1 - h, mode='full')
        cross_correlations = np.array([full_correlation[n_points - 1 + k] * dx 
                                    for k in range(2 * n_points - 1)])
        c5 = np.max(cross_correlations)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            iter_without_improvement = 0
        else:
            iter_without_improvement += 1
            
        if iter_without_improvement > max_iter_without_improvement:
            # Perturbation escape
            h += 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            iter_without_improvement = 0
            continue
            
        # Find max indices
        max_indices = np.where(np.abs(cross_correlations - c5) < 1e-10)[0]
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for k in max_indices:
            for i in range(n_points - k):
                if i < n_points:
                    gradient[i] += (1 - h[i + k])
                if i + k < n_points:
                    gradient[i + k] -= h[i]
        
        # Line search
        step_sizes = [0.1, 0.25, 0.5]
        best_h_candidate = h.copy()
        best_c5_candidate = c5
        
        for alpha in step_sizes:
            h_candidate = h - alpha * gradient
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
            
            # Evaluate candidate
            full_corr = correlate(h_candidate, 1 - h_candidate, mode='full')
            corr_vals = np.array([full_corr[n_points - 1 + k] * dx 
                                for k in range(2 * n_points - 1)])
            c5_candidate = np.max(corr_vals)
            
            if c5_candidate < best_c5_candidate:
                best_c5_candidate = c5_candidate
                best_h_candidate = h_candidate.copy()
        
        h = best_h_candidate
        if best_c5_candidate < best_c5:
            best_c5 = best_c5_candidate
            
    # Final validation
    full_corr = correlate(best_h, 1 - best_h, mode='full')
    corr_vals = np.array([full_corr[n_points - 1 + k] * dx 
                        for k in range(2 * n_points - 1)])
    final_c5 = np.max(corr_vals)
    
    assert abs(np.sum(best_h) - n_points/2) < 1e-6
    assert all(0 <= h_i <= 1 for h_i in best_h)
    
    return (best_h.tolist(), final_c5, n_points)
