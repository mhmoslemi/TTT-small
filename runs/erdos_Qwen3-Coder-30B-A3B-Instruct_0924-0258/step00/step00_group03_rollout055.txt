import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the overlap integral for a given h."""
    dx = 2.0 / len(h)
    # Compute all correlations and find maximum
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def is_feasible(h):
    """Check feasibility: bounds and sum constraint."""
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - len(h) / 2.0) < 1e-10)

def normalize_h(h):
    """Normalize h to satisfy sum(h) = n/2."""
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    
    # Adjust to meet sum constraint
    scale = target_sum / current_sum
    h_scaled = h * scale
    return h_scaled

def mutate_h(h, mutation_rate=0.1):
    """Mutate h with bounded noise and normalization."""
    h_new = h.copy()
    n = len(h)
    
    for i in range(n):
        if np.random.random() < mutation_rate:
            # Add bounded random noise
            delta = np.random.normal(0, 0.05)
            h_new[i] += delta
            
    # Clamp to [0,1]
    h_new = np.clip(h_new, 0, 1)
    
    # Re-normalize
    h_new = normalize_h(h_new)
    
    return h_new

def crossover_h(h1, h2):
    """Swap portions of two h vectors."""
    n = len(h1)
    cut = np.random.randint(1, n-1)
    h_new = np.concatenate([h1[:cut], h2[cut:]])
    
    # Normalize
    h_new = normalize_h(h_new)
    
    return h_new

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select parent with tournament selection."""
    selected_indices = []
    for _ in range(len(population)):
        candidates = np.random.choice(len(population), tournament_size)
        best_candidate = candidates[np.argmin([fitnesses[i] for i in candidates])]
        selected_indices.append(best_candidate)
    return [population[i] for i in selected_indices]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise create something
    global initial_h_values
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        n = 70
        # Simple construction as fallback
        h0 = np.ones(n) * 0.5  # Midpoint
    
    n_points = len(h0)
    # Initial population
    population = [h0.copy()]
    for _ in range(49):
        perturbed = mutate_h(h0, mutation_rate=0.2)
        population.append(perturbed)
    
    best_h = h0.copy()
    best_c5 = evaluate_h(best_h)
    
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        # Evaluate population
        fitnesses = [evaluate_h(h) for h in population]
        
        # Track best
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitnesses[best_idx]
        
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Crossover and mutation
        new_population = [best_h.copy()]  # Keep best
        while len(new_population) < 50:
            p1, p2 = np.random.choice(selected, 2, replace=False)
            if np.random.random() < 0.8:
                child = crossover_h(p1, p2)
            else:
                child = mutate_h(p1)
            new_population.append(child)
        
        population = new_population
    
    return (best_h.tolist(), best_c5, n_points)
