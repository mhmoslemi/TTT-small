import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    # Compute the overlap integral C5
    dx = 2.0 / len(h)
    # Create 1-h
    one_minus_h = 1 - h
    # Compute full convolution
    overlap = correlate(h, one_minus_h, mode="full")
    # Scale by dx
    overlap *= dx
    # Return maximum value
    return np.max(overlap)

def project_to_feasible(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def mutate_h(h, n_points, rate=0.1):
    # Mutate about rate fraction of the points
    num_mutate = int(rate * len(h))
    indices = np.random.choice(len(h), num_mutate, replace=False)
    h_new = h.copy()
    h_new[indices] += 0.1 * np.random.randn(num_mutate)
    # Project back to feasible region
    return project_to_feasible(h_new, n_points)

def crossover_h(h1, h2):
    # Single point crossover
    n = len(h1)
    point = np.random.randint(1, n)
    h_new = np.concatenate([h1[:point], h2[point:]])
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Hyperparameters
    n_points = 45
    pop_size = 50
    n_generations = 12
    mutation_rate = 0.1
    
    # Initialize the population
    population = []
    
    # Add the initial solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        population.append(h_init)
    else:
        # Create an initial guess
        h_init = np.ones(n_points) * 0.5
        h_init = project_to_feasible(h_init, n_points)
        population.append(h_init)
    
    # Fill population with perturbed versions of the initial solution
    while len(population) < pop_size:
        h = mutate_h(h_init, n_points, mutation_rate)
        population.append(h)
        
    # Keep track of the best solution
    best_h = None
    best_c5 = float('inf')
    
    # Evaluation function for a single solution
    def evaluate_solution(h):
        return evaluate_erdos_solution(h)
    
    # Main GA loop
    for gen in range(n_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for i, h in enumerate(population):
            c5 = evaluate_solution(h)
            fitness.append((c5, i))
            
        # Sort by fitness
        fitness.sort()
        top_indices = [idx for _, idx in fitness[:pop_size//2]]
        
        # Update best solution
        if fitness[0][0] < best_c5:
            best_c5 = fitness[0][0]
            best_h = population[fitness[0][1]].copy()
            
        # Selection: top 50% as parents
        parents = [population[i] for i in top_indices]
        
        # Crossover and mutation to create new population
        new_population = parents[:]
        
        # Generate offspring (also filling new population)
        while len(new_population) < pop_size:
            # Randomly select two parents
            p1, p2 = np.random.choice(parents, 2, replace=False)
            # Crossover
            child = crossover_h(p1, p2)
            # Mutate
            child = mutate_h(child, n_points, mutation_rate)
            new_population.append(child)
            
        # Trim to population size
        population = new_population[:pop_size]
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_erdos_solution(best_h)
    else:
        final_c5 = evaluate_solution(population[0])
        best_h = population[0].copy()
        
    return (best_h.tolist(), final_c5, n_points)
