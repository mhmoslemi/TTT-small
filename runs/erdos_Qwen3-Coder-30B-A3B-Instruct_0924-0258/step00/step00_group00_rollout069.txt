import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h based on initial_h_values or fallback
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h += 0.05 * np.random.rand(n_points)
    
    # Ensure the constraint sum(h) * dx = 1
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 20
    alpha_choices = [0.1, 0.25, 0.5]
    
    while time.time() - start_time < budget_s - 0.1:
        # Compute correlations
        full_corr = np.correlate(h, 1 - h, mode="full")
        correlations = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            correlations[k] = full_corr[k] * dx
        
        # Find the maximum value
        max_correlation = np.max(correlations)
        # Find indices where the maximum occurs
        max_indices = np.where(np.abs(correlations - max_correlation) < 1e-10)[0]
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for k in max_indices:
            if k < n_points:
                for i in range(n_points - k):
                    gradient[i] += (1 - h[i + k])
                    if i + k < n_points:
                        gradient[i + k] -= h[i]
            else:
                k_offset = k - (n_points - 1)
                for i in range(n_points - k_offset):
                    gradient[i] += (1 - h[i + k_offset])
                    if i + k_offset < n_points:
                        gradient[i + k_offset] -= h[i]
        
        # Line search
        best_alpha = 0
        best_new_c5 = best_c5
        best_h_candidate = h
        
        for alpha in alpha_choices:
            h_candidate = h - alpha * gradient
            # Clamp to [0,1]
            h_candidate = np.clip(h_candidate, 0, 1)
            # Rescale to meet constraint
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            new_c5 = evaluate_erdos_solution(h_candidate, dx)
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_h_candidate = h_candidate
                best_alpha = alpha
        
        # Update if there was improvement
        if best_new_c5 < best_c5:
            h = best_h_candidate
            best_c5 = best_new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # If no improvement, perturb the solution
        if no_improvement_count >= max_no_improvement:
            h += 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
        
        iter_count += 1
        
        # Store best solution
        if best_c5 < evaluate_erdos_solution(best_h, dx):
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h_vals, dx):
    h = np.array(h_vals)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation) * dx
    return max_corr
