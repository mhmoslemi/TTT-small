import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise random feasible
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Normalize to correct integral
        h *= (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h *= (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    start_time = time.time()
    t = 0
    max_iter = 10000
    alpha = 0.1
    decay = 1000
    step_size = alpha
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute correlation manually
        corr = signal.correlate(h, 1 - h, mode='full')
        corr *= dx
        max_corr_idx = np.argmax(corr)
        max_corr = corr[max_corr_idx]
        
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
            
        # Compute gradient
        # Gradient w.r.t. h: for each position k, d/dh_i of the overlap integral
        # This is: (for k=0 to n-1) \int h(x)(1-h(x+k)) dx
        # Gradient at position i is: (1-h[i+k]) where k = max_corr_idx - i
        
        grad = np.zeros(n_points)
        offset = max_corr_idx - (n_points - 1)
        for j in range(n_points):
            k = j + offset
            if 0 <= k < n_points:
                grad[j] = -(1 - h[k])
        
        # Gradient descent step
        h_new = h - step_size * grad
        # Project to feasible set [0,1] with integral constraint
        # Clip and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new *= (n_points / 2) / np.sum(h_new)
        
        h = h_new
        t += 1
        step_size = alpha * np.exp(-t / decay)
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    # Vectorized computation of max overlap
    corr = signal.correlate(h, 1 - h, mode='full')
    corr *= dx
    return np.max(corr)
