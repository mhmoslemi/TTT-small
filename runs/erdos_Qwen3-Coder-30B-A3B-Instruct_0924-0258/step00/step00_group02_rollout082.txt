import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 45
    dx = 2.0 / n_points
    max_iter = 600
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Hyperparameters
    lr = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlaps
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        # Take max over all lags
        c5 = np.max(overlaps)
        
        # Update best if improved
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Compute gradient
        # Gradient of overlap_k w.r.t. h[i] is (1-h[i+k] - h[i])
        grad = np.zeros(n_points)
        for k in range(n_points):
            # Add contribution from this overlap
            grad += (1 - h[(k + np.arange(n_points)) % n_points] - h) * dx
            
        # Update with momentum
        velocity = momentum * velocity - lr * grad
        h = h + velocity
        
        # Project onto feasible set
        h = np.clip(h, 0, 1)
        # Adjust to maintain sum = n_points/2
        h = h * (n_points / 2) / np.sum(h)
        
        # Adapt learning rate
        if i > 0 and i % 10 == 0:
            lr = max(0.001, lr * 0.95)
    
    return (list(best_h), best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    # Compute overlap integral C5 = max_k int h(x)(1-h(x+k)) dx
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlaps)
