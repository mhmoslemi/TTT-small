import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available

        n_points = len(h)
        # Ensure it's feasible
        h = h / np.sum(h) * (n_points / 2)

    except NameError:
        # Fallback: uniform distribution
        n_points = 47
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure sum constraint is perfectly met
    h = h / np.sum(h) * (n_points / 2)
    
    alpha = 0.1
    momentum = 0.9
    best_h = h.copy()
    best_C5 = compute_C5(h, n_points)
    last_improvement = 0
    iteration = 0
    
    # Gradient-based update with momentum
    v = np.zeros_like(h)
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute correlations
        corr = np.correlate(h, 1 - h, mode='full')
        C5_values = corr * (2.0 / n_points)  # dx = 2.0 / n_points
        k0 = np.argmax(C5_values)
        k0_shift = k0 - (n_points - 1)
        
        # Compute gradient: partial derivative of C5 w.r.t h_i
        grad = np.zeros_like(h)
        if k0_shift >= 0:
            # h[i] contributes to term h[i] * (1 - h[i + k0_shift])
            for i in range(n_points - k0_shift):
                grad[i] += 1 - h[i + k0_shift]
            # h[i + k0_shift] contributes to term h[i + k0_shift] * (1 - h[i])
            for i in range(n_points - k0_shift):
                grad[i + k0_shift] -= h[i]
        else:
            # k0_shift < 0
            k_shift = -k0_shift
            # h[i] contributes to term h[i] * (1 - h[i - k_shift])
            for i in range(k_shift, n_points):
                grad[i] += 1 - h[i - k_shift]
            # h[i - k_shift] contributes to term h[i - k_shift] * (1 - h[i])
            for i in range(k_shift, n_points):
                grad[i - k_shift] -= h[i]
        
        # Update with momentum
        v = momentum * v - alpha * grad
        h_new = h + v
        
        # Project back to feasible region: [0,1] and sum = n_points/2
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Compute new C5
        new_C5 = compute_C5(h_new, n_points)
        
        # Accept if better or occasional random restart
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
            last_improvement = iteration
        elif iteration - last_improvement > 50:
            # Random restart
            h_new = np.random.rand(n_points) * 0.5
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            new_C5 = compute_C5(h_new, n_points)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_new.copy()
                last_improvement = iteration
        else:
            h = h_new.copy()
        
        # Adaptive step size
        if iteration % 10 == 0:
            alpha *= 0.95
    
    return (best_h.tolist(), best_C5, n_points)

def compute_C5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    C5_values = corr * dx
    return np.max(C5_values)
