import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    n_points = 42
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert np.all(h >= 0) and np.all(h <= 1)
        assert abs(np.sum(h) - n_points / 2) < 1e-6
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    # Gradient descent with projection and line search
    max_iter = 100000
    alpha = 0.1
    beta = 0.8
    eps = 1e-6
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute all overlaps to find worst lag
        dx = 2.0 / n_points
        corr = signal.correlate(h, 1 - h, mode='full')
        lags = len(corr) - n_points
        overlaps = np.array([corr[lag + lags:lag + lags + n_points] for lag in range(n_points)]) * dx
        max_overlaps = np.max(overlaps, axis=1)
        k_star = np.argmax(max_overlaps)
        C5_val = max_overlaps[k_star]
        
        # Compute gradient direction
        g = 1 - 2 * np.roll(h, k_star)
        
        # Line search
        alpha_current = alpha
        while alpha_current > eps:
            h_new_unproj = h - alpha_current * g
            h_new = np.clip(h_new_unproj, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_C5 = evaluate_erdos_solution(h_new)
            if new_C5 < C5_val - 1e-10:
                h = h_new
                if new_C5 < best_C5:
                    best_C5 = new_C5
                    best_h = h.copy()
                break
            alpha_current *= beta

    return (best_h.tolist(), best_C5, n_points)
    
def evaluate_erdos_solution(h):
    # Simple numerical evaluation
    n_points = len(h)
    dx = 2.0 / n_points
    corr = signal.correlate(h, 1 - h, mode='full')
    lags = len(corr) - n_points
    overlaps = np.array([corr[lag + lags:lag + lags + n_points] for lag in range(n_points)]) * dx
    return np.max(overlaps)
