import numpy as np
import time
from scipy import correlate

def evaluate(h, dx):
    # Compute cross-correlations for all lags
    corr = correlate(h, 1 - h, mode='full')
    # Maximize over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, n_points):
    # Ensure h is in [0,1] and sum(h) = n_points/2
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    h = h * scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Default construction
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = project(h, n_points)
    
    # Linesearch candidates
    alpha_candidates = [0.1, 0.5, 1.0]
    
    # Keep track of best solution
    best_h = h.copy()
    best_C5 = evaluate(h, 2.0 / n_points)
    
    # Perturb steps
    perturb_steps = 20
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s:
        # Compute all cross-correlations
        corr = correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        
        # Extract the maximum lag k
        k = max_k - (n_points - 1)
        
        # Compute gradient for this lag
        grad = np.zeros(n_points)
        if k >= 0:
            # h[i] contributes to corr[i+k] : grad[i] = 1 - h[i+k]
            # But we need the elements that contribute to the gradient
            for i in range(n_points - k):
                grad[i] += 1.0 - h[i + k]
        else:
            # h[i+k] contributes to corr[i] : grad[i+k] = 1 - h[i]
            # Adding k to index for correct locations
            for i in range(-k, n_points):
                grad[i] += 1.0 - h[i + k]
        
        best_alpha = 0.0
        best_h_new = h.copy()
        best_C5_new = best_C5
        
        # Try line search
        for alpha in alpha_candidates:
            h_new = h - alpha * grad
            h_new = project(h_new, n_points)
            
            C5_new = evaluate(h_new, 2.0 / n_points)
            
            if C5_new < best_C5_new:
                best_C5_new = C5_new
                best_h_new = h_new.copy()
                best_alpha = alpha
        
        # Stop if no improvement
        if best_C5_new >= best_C5:
            no_improvement_count += 1
            if no_improvement_count >= perturb_steps:
                # Add perturbation
                h += 0.01 * np.random.randn(n_points)
                h = project(h, n_points)
                no_improvement_count = 0
        else:
            h = best_h_new
            best_C5 = best_C5_new
            best_h = h.copy()
            no_improvement_count = 0
    
    return (best_h.tolist(), best_C5, n_points)
