    h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    start_time = time.time()

    # Evaluate current C5
    def evaluate_C5(h_vals):
        # Calculate correlation using numpy
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx

    # Compute initial C5
    current_C5 = evaluate_C5(h)
    best_C5 = current_C5
    best_h = h.copy()

    # Gradient approximation using finite differences
    def compute_gradient(h_vals, eps=1e-6):
        grad = np.zeros_like(h_vals)

            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_minus = h_vals.copy()
            h_minus[i] -= eps
            grad[i] = (evaluate_C5(h_plus) - evaluate_C5(h_minus)) / (2 * eps)
        return grad

    def line_search(h_vals, grad, max_alpha=1.0):
        alpha = max_alpha
        while alpha > 1e-8:
            new_h = h_vals - alpha * grad

            new_C5 = evaluate_C5(new_h)
            if new_C5 < current_C5 or alpha < 1e-6:
                return alpha, new_h
            alpha *= 0.5
        return alpha, h_vals

    iteration = 0

            grad = compute_gradient(h)

                    best_C5 = current_C5
                    best_h = h.copy()
        else:
            break
            
        iteration += 1
