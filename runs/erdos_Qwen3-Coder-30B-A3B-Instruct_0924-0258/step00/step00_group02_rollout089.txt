import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute overlap integral for all shifts
    overlap = correlate(h, 1 - h, mode="full") * dx
    # The maximum overlap is what we're trying to minimize
    return np.max(overlap)

def initialize_population(initial_h_values, n_points, pop_size=50):
    """Initialize population with feasible step functions"""
    population = [initial_h_values.copy()]
    for _ in range(pop_size - 1):
        # Create a small perturbation of the initial solution
        perturbation = np.random.normal(0, 0.05, n_points)
        h = initial_h_values + perturbation
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to maintain sum constraint
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    return population

def crossover(parent1, parent2):
    """Single-point crossover of two parents"""
    n = len(parent1)
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1):
    """Mutate a solution"""
    n = len(h)
    mutation_indices = np.random.choice(n, int(mutation_rate * n), replace=False)
    for i in mutation_indices:
        h[i] += np.random.normal(0, 0.1)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to maintain sum constraint
    h = h * (n / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization loop"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial values
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        # Fallback: create a baseline
        n_points = 45
        initial_h = np.ones(n_points) * 0.5
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # EA parameters
    pop_size = 50
    generations = 12
    mutation_rate = 0.1

    # Initialize population
    population = initialize_population(initial_h, n_points, pop_size)

    # Track the best solution
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all in population
        scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            scores.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by score
        sorted_indices = np.argsort(scores)
        population = [population[i] for i in sorted_indices]

        # Selection: top 50%
        selected = population[:pop_size//2]
        
        # Crossover
        offspring = []
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < len(selected) else selected[0]
            child1, child2 = crossover(parent1, parent2)
            offspring.extend([child1, child2])
            
        # Mutation
        mutated_offspring = []
        for child in offspring:
            mutated = mutate(child, mutation_rate)
            mutated_offspring.append(mutated)
            
        # New population: selected parents + mutated offspring
        new_pop = selected + mutated_offspring[:pop_size - len(selected)]
        population = new_pop[:pop_size]
        
        # Debug: print progress
        # print(f"Generation {gen+1} Best C5: {best_c5}")

    return (best_h.tolist(), best_c5, n_points)
