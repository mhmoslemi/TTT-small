import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: initialize with a simple construction
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.49
        h[-1] = 0.51
    
    # Ensure constraint sum(h) * dx = 1 => sum(h) = n_points / 2
    dx = 2.0 / n_points
    current_sum = np.sum(h)
    h = h * (n_points / 2) / current_sum

    # Keep track of best solution
    best_C5 = float('inf')
    best_h = h.copy()

    # Frank-Wolfe iterations
    max_iter = 500
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all cross-correlations
        corr = signal.correlate(h, 1 - h, mode='full')
        # The correlation is computed for lags from -n+1 to n-1
        # We only care about positive lags (from 0 to n-1)
        # Take the max over positive lags (since we are minimizing C5)
        C5 = np.max(corr[n_points-1:])
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
        
        # Compute gradient of the max correlation
        max_k = np.argmax(corr[n_points-1:])  # the lag that achieves the maximum
        grad = np.zeros_like(h)
        
        # Gradient computation for max_k
        # For positive lag k: h[i] contributes to corr[k] via h[i]*h[i+k]
        # For the function sum(h[i]*(1-h[i+k])), the partial derivative w.r.t h[i]
        # is (1 - h[i+k]) if the index is valid, else 0.
        for i in range(n_points):
            if i + max_k < n_points:
                grad[i] = 1 - h[i + max_k]
        
        # Frank-Wolfe step: solve subproblem <grad, v - h> s.t. v in feasible set
        # We solve: max <grad, v - h>, with v in [0,1]^n and sum(v) = n_points/2
        # Order indices by gradient values (descending), keep top n_points/2
        sorted_idx = np.argsort(-grad)
        s_idx = sorted_idx[:n_points//2]
        
        # Construct candidate solution v to maximize <grad, v - h>
        v = np.zeros_like(h)
        v[s_idx] = 1  # set top entries to 1
        # Now ensure sum(v) = n_points / 2
        v_sum = np.sum(v)
        v = v * (n_points / 2) / v_sum
        
        # Line search along direction (v - h)
        # Sample a few gamma values and estimate resulting C5
        gamma_candidates = np.array([0.01, 0.1, 0.25, 0.5, 0.75, 0.9, 1.0])
        best_gamma = 0.0
        best_new_C5 = float('inf')
        
        for gamma in gamma_candidates:
            test_h = h + gamma * (v - h)
            # Clip to [0,1]
            test_h = np.clip(test_h, 0, 1)
            # Rescale to match constraint
            test_sum = np.sum(test_h)
            test_h = test_h * (n_points / 2) / test_sum
            test_corr = signal.correlate(test_h, 1 - test_h, mode='full')
            test_C5 = np.max(test_corr[n_points-1:])
            if test_C5 < best_new_C5:
                best_new_C5 = test_C5
                best_gamma = gamma
        
        h = h + best_gamma * (v - h)
        # Make sure sum constraint is still satisfied
        h = h * (n_points / 2) / np.sum(h)
        
    # Ensure feasibility
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    corr_final = signal.correlate(h, 1 - h, mode='full')
    final_C5 = np.max(corr_final[n_points-1:])
    
    return (h, final_C5, n_points)
