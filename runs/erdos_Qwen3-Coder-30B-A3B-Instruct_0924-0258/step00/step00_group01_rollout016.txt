import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Check if initial_h_values is available
    h = None
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        # Normalize to sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
    else:
        n_points = 47
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points)
    
    # Optimizer parameters
    alpha = 0.1
    max_iter = 2000
    iter_without_improvement = 0
    max_no_improve = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute C5 using full correlation
        c5 = evaluate_c5(h, n_points)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            iter_without_improvement = 0
        else:
            iter_without_improvement += 1
            
        if iter_without_improvement > max_no_improve:
            # Random perturbation
            perturb = np.random.normal(0, 0.01, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            # Reapply sum constraint
            h = h * (n_points / 2) / np.sum(h)
            iter_without_improvement = 0
            
        # Compute gradient approximation using cross-correlations
        h1 = 1 - h
        corr = signal.correlate(h, h1, mode='full')
        corr = corr[len(h)-1:]
        max_lag = np.argmax(corr)
        
        # Compute gradient from this lag
        grad = np.zeros(n_points)
        if max_lag < n_points:
            # h[0 to n_points-1] with h1 at lag max_lag 
            for i in range(n_points):
                j = (i + max_lag) % n_points
                grad[i] += h1[j]
            # Add negative contribution from h at the same index
            for i in range(n_points):
                j = (i - max_lag) % n_points
                grad[i] -= h[j]
                
        # Update with clipped gradient
        h += alpha * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive learning rate
        if iteration % 50 == 0 and iter_without_improvement > 0:
            alpha *= 0.95
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    # Use numpy correlate to compute overlaps efficiently
    corr_values = np.correlate(h, 1 - h, mode="full")
    # Take the maximum over all lags
    max_overlap = np.max(corr_values) * dx
    return max_overlap
