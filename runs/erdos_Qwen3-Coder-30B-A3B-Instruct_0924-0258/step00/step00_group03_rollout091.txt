import time
import numpy as np
from scipy.signal import correlate

def evaluate_C5(h):
    """Evaluate C5 for a given h using correlation."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations with 1-h
    corr = correlate(h, 1 - h, mode='full')
    # Extract the overlapping parts (the valid ones)
    # The full correlation is of length 2*n - 1, we want max over k from 0 to n-1
    # For k=0, correlation at index n-1 (middle)
    # For k=1, correlation at index n-2 and n (depending on direction)
    # We want all k indices from 0 to n-1
    # Since we do full correlation, the index for k is 2*n-2-k, and reverse for negative shift
    # So for positive shifts from 0 to n-1, we look at indices 0 to n-1 (corresponding to positive k)
    # But it's easier to just take appropriate slices, using their known properties
    # The maximum is over k=0, ..., n-1 (which maps to correlation indices).
    # The valid correlations for k=0 to n-1 are indexes from n-1 to 2*n-2. 
    # But correlation is symmetric in a sense (as we're taking h*1-h, so property of cross correlation)
    # Let's take a simpler path and just grab the appropriate ranges
    max_corr = 0
    for k in range(n):
        # correlate places the 0-shift at middle, so for correlation at position p,
        # it corresponds to actual lag k = p - (n-1)
        # We want k=0,1,...,n-1 so we look at p = n-1, n, ..., 2*n-2
        # Forward overlap for lag k is at index(n-1+k) = n-1+k  
        idx = n - 1 + k
        max_corr = max(max_corr, corr[idx] * dx)
    return max_corr

def project_h(h, n_points):
    """Ensure h sums to n_points/2 and is in [0,1]."""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s > 0:
        h = h * (n_points / 2) / s
    else:
        h = np.full(n_points, 1.0 / 2)
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Number of points for discretization
    n_points = 70
    np.random.seed(seed)
    
    # Start with initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert len(h) == n_points
    else:
        # A simple construction, just a smooth bump that integrates to 1 over [0,2]
        x = np.linspace(0, 2, n_points)
        h = np.exp(-5 * (x - 1) ** 2)
        h = project_h(h, n_points)
    
    # Validate initial solution
    assert abs(np.sum(h) * 2.0 / n_points - 1.0) < 1e-10, "Initial h does not integrate to 1"
    assert np.all(h >= 0) and np.all(h <= 1), "Initial h not in [0,1]"
    
    # Evaluate the initial C5
    best_C5 = evaluate_C5(h)
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.999
    sigma0 = 0.1
    t = 0
    start_time = time.time()
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        t += 1
        # Cooling schedule
        T = T0 * (alpha ** t)
        if T < 1e-10:
            break
            
        # Mutation: add Gaussian noise and project
        sigma = sigma0 * (alpha ** (t / 200))
        h_new = h + np.random.normal(0, sigma, n_points)
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        C5_new = evaluate_C5(h_new)
        
        # Accept with probability
        delta = C5_new - best_C5
        if delta < 0 or np.random.uniform() < np.exp(-delta / T):
            h = h_new
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h_new.copy()
        
        # Occasionally log
        if t % 100 == 0:
            pass
    
    return (best_h.tolist(), best_C5, n_points)
