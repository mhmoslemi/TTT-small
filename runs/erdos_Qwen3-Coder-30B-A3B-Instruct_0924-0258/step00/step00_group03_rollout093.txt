import numpy as np
import time
from scipy import signal

def evaluate_C5(h):
    """Evaluate the maximum overlap integral for given h"""
    dx = 2.0 / len(h)
    overlap = signal.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h):
    """Project h to satisfy the integral constraint and bounds"""
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * (len(h) / 2) / sum_h
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    n_points = 70
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a reasonable initial guess
        h = np.full(n_points, 0.5)
    
    # Ensure it's feasible
    h = project_to_feasible(h)
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    # Parameters for simulated annealing
    T0 = 1.0
    alpha = 0.999
    sigma0 = 0.1
    t = 0
    start_time = time.time()
    
    # Main loop
    while time.time() - start_time < budget_s:
        t += 1
        T = T0 * (alpha ** t)
        sigma = sigma0 * (alpha ** (t / 200))
        
        # Generate candidate
        h_new = h + np.random.normal(0, sigma, n_points)
        h_new = project_to_feasible(h_new)
        
        # Evaluate
        C5_new = evaluate_C5(h_new)
        
        # Accept or reject
        if C5_new < best_C5 or np.random.rand() < np.exp(-(C5_new - best_C5) / T):
            h = h_new
            if C5_new < best_C5:
                best_h = h_new.copy()
                best_C5 = C5_new
    
    return (best_h.tolist(), best_C5, n_points)
