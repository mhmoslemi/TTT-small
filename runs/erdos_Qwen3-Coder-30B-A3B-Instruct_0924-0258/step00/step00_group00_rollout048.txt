def evaluate_h(h, dx):
    """Evaluate C5 for given h, dx"""
    # Compute cross-correlation with 1-h
    correlation = np.correlate(h, 1 - h, mode='full')
    # The full correlation has length 2*n-1
    # But only middle values contain valid lags
    # For each lag k, the effective length is 2*n-1-|k|
    # We use maximum of all valid correlations
    max_val = 0
    n = len(h)
    # Loop over all unique lags
    for k in range(-n+1, n):
        # Compute valid length for lag k
        if k >= 0:
            valid_len = n - k
        else:
            valid_len = n + k
        if valid_len <= 0:
            continue
        # Extract subset of correlation for this lag
        start_idx = n - 1 + k
        # Sum up h[i] * (1 - h[i+k]) for valid range
        # But we already computed correlation, so ...
        # We're looking at correlation[n-1+k]
        corr_val = correlation[n-1+k]
        # Multiply by dx to get normalized overlap
        val = corr_val * dx
        if val > max_val:
            max_val = val
    return max_val

def rescale_h(h, target_sum):
    """Rescale h so that sum(h) = target_sum, keeping h[i] in [0,1]"""
    current_sum = np.sum(h)
    if np.isclose(current_sum, target_sum):
        return h.copy()
    if current_sum <= 0:
        return np.full_like(h, target_sum / len(h))
    # Scale to target_sum and then clip
    new_h = h * target_sum / current_sum
    # Ensure bounds
    new_h = np.clip(new_h, 0, 1)
    # Rescale again
    actual_sum = np.sum(new_h)
    if actual_sum > 0:
        new_h = new_h * target_sum / actual_sum
    # Clip again after rescale
    new_h = np.clip(new_h, 0, 1)
    return new_h

def mutate_h(h, mutation_rate=0.1, delta=0.1):
    """Mutate h by modifying genes with probability mutation_rate"""
    # Copy input_h to avoid side effects
    new_h = h.copy()
    n = len(h)
    mutated = False
    for i in range(n):
        if np.random.random() < mutation_rate:
            mutated = True
            new_h[i] += np.random.uniform(-delta, delta)
    if not mutated:
        # Ensure at least one mutation for diversity
        i = np.random.randint(n)
        new_h[i] += np.random.uniform(-delta, delta)
    # Now clip values to [0,1]
    new_h = np.clip(new_h, 0, 1)
    
    # Rescale to target sum
    new_h = rescale_h(new_h, np.sum(h))
    return new_h

def crossover_h(parent1, parent2):
    """One-point crossover between two h vectors"""
    n = len(parent1)
    split_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:split_point], parent2[split_point:]])
    child2 = np.concatenate([parent2[:split_point], parent1[split_point:]])
    # Rescale children to sum to same value as parents
    target_sum = np.sum(parent1)
    child1 = rescale_h(child1, target_sum)
    child2 = rescale_h(child2, target_sum)
    return child1, child2

def compute_fitness(h, dx):
    """Compute fitness: smaller value is better"""
    return evaluate_h(h, dx)

def tournament_selection(population, fitnesses, tournament_size=3):
    """Tournament selection"""
    n = len(population)
    selected = []
    for _ in range(n):
        tournament_indices = np.random.choice(n, tournament_size, replace=False)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(population[winner_idx])
    return selected
