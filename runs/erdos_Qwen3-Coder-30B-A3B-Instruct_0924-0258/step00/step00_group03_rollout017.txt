import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h):
    """Evaluate the C5 constant for given h"""
    # Compute the overlap integral
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the full convolution of h with (1-h)
    conv = correlate(h, 1 - h, mode="full")
    # Take the maximum value over all lags
    max_val = np.max(conv)
    return max_val * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # sum(h) should be 35
    max_iter = int(1e6)  # larger limit to be safe
    cooling_rate = 0.999
    initial_temp = 1.0
    sigma_0 = 0.1
    t = 0
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Fallback initialization: stepped function
        h = np.ones(n_points) * 0.5
    
    # Normalize to required sum
    h = h * (target_sum / np.sum(h))
    
    # Check feasibility
    assert np.all(h >= 0) and np.all(h <= 1), "h values not in [0,1]"
    assert abs(np.sum(h) - target_sum) < 1e-10, "h sum condition violated"
    
    # Evaluate initial solution
    best_C5 = evaluate_C5(h)
    best_h = h.copy()
    
    # Timing
    start_time = time.time()
    
    while t < max_iter:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        t += 1
        
        # Update temperature
        temp = initial_temp * (cooling_rate ** t)
        
        # Mutation with decreasing noise
        sigma = sigma_0 * (cooling_rate ** (t / 200))
        noise = np.random.normal(0, sigma, n_points)
        h_new = h + noise
        
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project back to correct sum
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Evaluate
        C5_new = evaluate_C5(h_new)
        
        # Accept or reject
        if C5_new < best_C5 or np.random.rand() < np.exp(-(C5_new - best_C5) / temp):
            h = h_new
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
    
    # Final check
    final_C5 = evaluate_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
