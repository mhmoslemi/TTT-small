import numpy as np
import time
from scipy.optimize import minimize
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Reasonable size for good exploration
    dx = 2.0 / n_points
    max_iter = 600
    initial_temp = 1.0
    cooling_factor = 0.99
    min_temp = 0.01
    patience = 50
    gradient_scale = 0.1
    mutation_std = 0.05
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0, 1)
        # Ensure integral is correct
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    temperature = initial_temp
    patience_count = 0
    last_improvement = 0
    
    # Precompute correlation for efficiency
    def compute_c5(h_vec):
        # Compute correlation for all lags
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # Only take the relevant lags (first n_points-1) and normalize
        result = np.max(corr[n_points-1:]) * dx
        return result
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        iter_count += 1
        current_c5 = compute_c5(h)
        
        # Generate candidate
        # Add noise with gradient guidance if available
        noise = np.random.normal(0, mutation_std, n_points)
        h_candidate = h + noise
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Rescale to maintain integral constraint
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute candidate C5
        new_c5 = compute_c5(h_candidate)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temperature):
            h = h_candidate
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                patience_count = 0
                last_improvement = iter_count
            else:
                patience_count += 1
        else:
            patience_count += 1
            
        # Update temperature
        temperature = max(temperature * cooling_factor, min_temp)
        
        # Warm restarts
        if patience_count > patience and iter_count - last_improvement > 10:
            temperature = initial_temp
            patience_count = 0
    
    # Final validation and computation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_c5(h, dx):
    # Compute correlation for all lags
    corr = np.correlate(h, 1 - h, mode="full")
    # Only take the relevant lags (first n_points-1) and normalize
    result = np.max(corr[len(h)-1:]) * dx
    return result
