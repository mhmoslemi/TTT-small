import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Return the C5 bound for a given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral for all lags
    corr = correlate(h, 1 - h, mode="full")
    # Take maximum value and scale by dx
    max_overlap = np.max(corr) * dx
    return max_overlap

def normalize_h(h):
    """Ensure h sums to n/2 and stays in [0,1]."""
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    # Adjust to maintain exact sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute difference evenly
        h += diff / n
        h = np.clip(h, 0, 1)
    return h

def mutate_h(h, mutation_rate=0.1, noise_scale=0.1):
    """Mutate h with noise and renormalize."""
    n = len(h)
    noise = np.random.normal(0, noise_scale, n)
    h_mut = h + mutation_rate * noise
    return normalize_h(h_mut)

def crossover_h(h1, h2, crossover_rate=0.8):
    """Crossover two step functions."""
    if np.random.random() > crossover_rate:
        return h1.copy()
    n = len(h1)
    crossover_point = np.random.randint(1, n)
    h_new = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    return normalize_h(h_new)

def init_population(n_points, pop_size, initial_h_values):
    """Initialize population with diverse candidates."""
    pop = []
    # Start with initial solution
    pop.append(initial_h_values.copy())
    # Add some random but feasible solutions
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = normalize_h(h)
        pop.append(h)
    return pop

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Use initial_h_values if available or create a reasonable starting point
    if 'initial_h_values' in globals():
        h0 = initial_h_values.copy()
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5  # Simple uniform distribution

    # population size and evolution parameters
    pop_size = 50
    generations = 1000  # Estimate 1000 generations in 600s

    # Initialize population
    population = init_population(n_points, pop_size, h0)
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Tournament selection function
    def tournament_select(pop, fit_vals, tsize=5):
        idxs = np.random.choice(len(pop), tsize)
        best_idx = idxs[np.argmin([fit_vals[i] for i in idxs])]
        return pop[best_idx]
    
    for gen in range(generations):
        if elapsed() >= budget_s:
            break
            
        # compute fitness
        fitnesses = [evaluate_h(h) for h in population]
        
        # update best
        min_fit_idx = np.argmin(fitnesses)
        if fitnesses[min_fit_idx] < best_c5:
            best_c5 = fitnesses[min_fit_idx]
            best_h = population[min_fit_idx].copy()
            
        # selection and reproduction
        new_pop = []
        # keep best one
        _, best_idx = min([(fitnesses[i], i) for i in range(len(population))])
        new_pop.append(population[best_idx].copy())
        
        while len(new_pop) < pop_size:
            # Select two parents
            p1 = tournament_select(population, fitnesses)
            p2 = tournament_select(population, fitnesses)
            # Crossover
            child = crossover_h(p1, p2)
            # Mutate
            child = mutate_h(child)
            new_pop.append(child)
            
        population = new_pop
        
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_h(best_h)
        return (best_h.tolist(), float(final_c5), n_points)
    
    # fallback
    final_c5 = evaluate_h(h0)
    return (h0.tolist(), float(final_c5), n_points)
