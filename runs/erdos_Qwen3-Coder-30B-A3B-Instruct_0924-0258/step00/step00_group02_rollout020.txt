import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx=2.0/len(h)):
    """Evaluate the Erdős constant C_5 for given step function h."""
    # Compute cross-correlation h * (1 - h) with full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Return maximum value scaled by dx
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region [0,1] and scale to sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def convex_surrogate(h, n_points, dx=2.0/n_points):
    """Convex surrogate for the maximum overlap."""
    corr = np.correlate(h, 1 - h, mode='full')
    # Sum of squares of correlations (a convex surrogate)
    return np.sum(corr**2) * dx**2

def gradient_of_surrogate(h, n_points, dx=2.0/n_points):
    """Gradient of the convex surrogate with respect to h."""
    # Compute cross-correlation h * (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Compute derivatives of correlation with respect to h
    # Using the fact that it's equivalent to convolving with (1-h) then (h-1)
    grad = np.zeros_like(h)
    
    # For i in range(n_points), gradient at h[i] depends on correlations that involve i
    # This is approximated by the derivative of the sum of squares of correlations
    for i in range(n_points):
        # Partial derivative w.r.t h[i] of sum of squares of correlation terms
        # We compute a simplified approximation
        conv_derivative = 0.0
        # For each valid k, we have a term (h * (1-h))[k] = sum_{i} h[i] * (1-h[i+k])
        # but we want the derivative w.r.t h[i] of the sum over k of these squares
        # So for some k range:
        # d/dh[i] (sum_k (h * (1-h))_k^2) = 2 * (h * (1-h))_i * d/dh[i] (h * (1-h))_i
        
        # Simplified gradient: just approximate the finite difference effect
        # Manhattan-like gradient approximation for stability
        for k in range(n_points):
            if 0 <= i + k < n_points:
                conv_derivative += 2 * corr[n_points - 1 + k] * (1 - h[i + k]) * dx
            if 0 <= i - k < n_points:
                conv_derivative += 2 * corr[n_points - 1 - k] * (1 - h[i - k]) * dx
        
        grad[i] = conv_derivative * dx
    
    return grad

def optimize_at_resolution(h0, n_points, time_budget):
    """Optimize at a fixed resolution using convex surrogate."""
    dx = 2.0 / n_points
    max_time = time() + time_budget
    
    def surrogate_obj(h):
        return convex_surrogate(h, n_points, dx)
    
    def gradient_obj(h):
        return gradient_of_surrogate(h, n_points, dx)
    
    # Constraints for SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
    
    h_opt = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0, dx)
    
    # Optimizer run
    for _ in range(100):
        t = time()
        if t >= max_time:
            break
        
        # Use SLSQP with gradient info (approximate for now)
        try:
            res = minimize(surrogate_obj, h_opt, bounds=bounds, constraints=cons,
                           method='SLSQP', jac=gradient_obj, options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_opt = res.x
            else:
                h_opt = project_to_feasible(h_opt, n_points)
        except:
            h_opt = project_to_feasible(h_opt, n_points)
        
        h_opt = project_to_feasible(h_opt, n_points)
        c5 = evaluate_erdos_solution(h_opt, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
        
        # Slight perturbation
        h_opt = project_to_feasible(h_opt + np.random.normal(0, 0.01, n_points), n_points)
    
    return best_h, best_c5

        h0 = np.ones(n_points) * 0.5
        h0[10:35] = 1.0  # Make it a step function
        h0 = project_to_feasible(h0, n_points)
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Resolution levels to try: smaller first for speed
    resolutions = [15, 30, 45, 60, 75]
    
    start_time = time()
    last_improvement = 0
    for res_idx, n_points in enumerate(resolutions):
        if time() - start_time > budget_s - 5:
            break
            
        if n_points > len(best_h):
            # Interpolate from current coarse solution
            x0 = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_points)
            h_new = np.interp(x_new, x0, best_h)
            h_new = project_to_feasible(h_new, n_points)
        else:
            h_new = best_h[:n_points]
            h_new = project_to_feasible(h_new, n_points)
        
        # Perturb a few times from this level to avoid local optima
        for _ in range(3):
            if time() - start_time > budget_s - 1:
                break
            t_remaining = budget_s - (time() - start_time)
            h_test, c5_test = optimize_at_resolution(h_new + np.random.normal(0, 0.02, n_points), 
                                                     n_points, t_remaining * 0.8)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
                last_improvement = 0
            else:
                last_improvement += 1
        
        # Early exit if no improvements in 3 steps
        if last_improvement >= 3:
