import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_c5(h):
    """Evaluate the maximum overlap integral for the given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation for all lags
    corr = np.correlate(h, 1 - h, mode='full')
    # The correlation is padded; we need the middle part
    # that corresponds to the valid lags (0 to n-1 in both directions)
    # for valid shifts k in [0, n-1]
    # For our specific problem, we only consider lags 0 to n-1 (which means
    # we extract the second half of the full convolution starting from
    # n-1 index)
    c5 = np.max(corr[len(h)-1:])
    return c5 * dx

def solve_c5_optimization(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if not (np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - n_points / 2) < 1e-10):
            raise ValueError("Initial h does not meet constraints")
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    
    def objective(h_vec):
        h_reshaped = h_vec.copy()
        h_reshaped = h_reshaped * (n_points / 2) / np.sum(h_reshaped)
        c5 = evaluate_c5(h_reshaped)
        return c5
    
    def gradient(h_vec):
        # Using finite differences
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad
    
    iter_count = 0
    while True:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Local optimization
        res = minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)]*len(h), 
                       jac=gradient, options={'maxiter': 10})
        if res.success:
            h = res.x
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        else:
            # Fallback if optimization fails
            pass
            
        # Low-pass filtering to generalize
        h_filtered = gaussian_filter1d(h, sigma=0.1 * n_points)
        # Clip and rescale
        h_filtered = np.clip(h_filtered, 0, 1)
        h_filtered = h_filtered * (n_points / 2) / np.sum(h_filtered)
        h = h_filtered
        
        iter_count += 1
        if iter_count >= 100:  # Avoid too many iterations that run long
            break
            
    # Final validation
    best_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return solve_c5_optimization(seed, budget_s)
