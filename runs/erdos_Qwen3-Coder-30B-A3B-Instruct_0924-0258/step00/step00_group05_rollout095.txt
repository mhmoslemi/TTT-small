import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, n_points)
    
    # Gradient descent with adaptive step size
    max_iter = 500
    alpha = 0.1
    tol = 1e-4
    last_improvement = 0
    stagnation_count = 0
    max_stagnation = 10
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap integrals for all lags
        correlations = np.correlate(h, 1-h, mode="full")
        C5 = correlations * dx
        max_k = np.argmax(C5)
        max_val = C5[max_k]
        
        if max_val <= best_C5 * 0.999:
            best_C5 = max_val
            best_h = h.copy()
            
        # Compute gradient
        g = np.zeros(n_points)
        # For the maximum overlap lag, we compute contributions
        # For lag k, contribution to grad[i] is (1-h[i+k] - (1-h[i])) = h[i] - h[i+k]
        # We accumulate this for indices where that is the max overlap
        for k_idx in range(n_points):
            if abs(C5[k_idx] - max_val) < 1e-8:
                for i in range(n_points):
                    j = (i + k_idx) % n_points
                    g[i] += (h[i] - h[j])
        g = g * dx / n_points  # normalize
        
        # Update h
        h_new = h - alpha * g
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new C5
        new_C5 = evaluate_erdos_solution(h_new, n_points)
        
        # Accept if improvement
        if new_C5 < best_C5:
            h = h_new.copy()
            best_C5 = new_C5
            last_improvement = i
        else:
            # Perturb slightly if no improvement
            h_new = h + 0.01 * np.random.randn(n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_C5 = evaluate_erdos_solution(h_new, n_points)
            if new_C5 < best_C5:
                h = h_new.copy()
                best_C5 = new_C5
                last_improvement = i
            else:
                # Decrease alpha if stagnating
                stagnation_count += 1
                if stagnation_count > max_stagnation:
                    alpha *= 0.8
                    stagnation_count = 0
        # Decrease step size if improvement
        if i - last_improvement > 50:
            alpha *= 0.95
        if alpha < 1e-6:
            alpha = 1e-6
            
    return (best_h.tolist(), best_C5, n_points)
