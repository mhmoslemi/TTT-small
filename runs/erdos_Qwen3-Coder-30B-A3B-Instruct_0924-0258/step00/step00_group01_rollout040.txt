import numpy as np
import time
from scipy import signal

def evaluate_C5(h, dx):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx using correlation"""
    # Convolve h with (1-h) in full mode
    corr = signal.correlate(h, 1 - h, mode='full')
    # Multiply by dx to get actual integral
    corr *= dx
    # Max over all lags
    return np.max(corr)

def project_to_feasible(h, n_points, dx):
    """Project h to satisfy integral constraint and bounds"""
    h = np.clip(h, 0, 1)
    # Scale to meet integral constraint
    sum_h = np.sum(h)
    if abs(sum_h * dx - 1.0) > 1e-10:
        scale = (n_points / 2) / sum_h
        h *= scale
        h = np.clip(h, 0, 1)
    return h

    dx = 2.0 / n_points
    
    # Initialize
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Gradient descent with adaptive step
    alpha = 0.1
    stagnation_count = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s - 0.1:
        # Compute all correlations efficiently
        corr = signal.correlate(h, 1 - h, mode='full')
        corr *= dx
        k_max = np.argmax(corr)

        # If improved, keep going
        if max_val < best_C5:
            best_C5 = max_val
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                # Random restart
                h = project_to_feasible(np.random.rand(n_points) * 0.5, n_points, dx)
                stagnation_count = 0
                alpha = 0.1
                continue

        # Compute gradient with respect to h
        # Correlation of h with (1-h) gives us F(h) * F(1-h) in frequency domain
        # We reconstruct the gradient efficiently for k_max
        
        # Inverse correlation at k_max gives gradient approx?. 
        # Consider applying shift operator manually:
        # For lag k0 where max overlap occurs, compute gradient wrt each point
        
        # Simplified version:
        # At each position i, gradient contribution from lag k0:
        # If h[i] impacts h[i + k0] in overlap, etc.
        
        # Efficiently compute: h * (1 - h shifted by k)
        grad = np.zeros(n_points)
        # k_max shift to use
        k0 = k_max - (n_points - 1)  # convert full index to actual lag

        h_shift_backward = np.roll(1 - h, k0)

        # So gradient wrt h[i] is:
        # (1 - h[i+k0]) if i+k0 < n_points else 0
        # Gradient: compare to reference
        # Compute exact finite difference-like gradient
        
        # Actual gradient computation:
        # g[i] = sum over all k of d/dh[i](h[i] * (1 - h[i+k]))
        # that's [1 - h[i+k], i+k < n_points]
        # Since h[i+k] comes from h shifted by -k
        # d/dh[i](h[i] * (1 - h[i+k])) = 1 - h[i+k] if i+k < n_points
        # ∴ gradient[i] = sum over k : [1 - h[i+k]] = (1-h) convolved with ones at position i

                if j < n_points:
