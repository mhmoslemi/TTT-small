import numpy as np
import cvxpy as cp
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Define constraints for cvxpy
    h_var = cp.Variable(n_points)
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
    
    # Create optimization problem for projection
    prob = cp.Problem(cp.Minimize(cp.sum_squares(h_var - h)), constraints)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1:
        # Compute correlations
        corr = signal.correlate(h, 1 - h, mode='full')
        k0 = np.argmax(corr)
        max_corr = corr[k0]
        
        # Compute gradient for the max correlation
        grad = np.zeros(n_points)
        if k0 < n_points:
            for i in range(n_points - k0):
                grad[i] += (1 - h[i + k0]) * dx
            for i in range(k0):
                grad[i + k0] -= h[i] * dx
        else:
            k = k0 - n_points
            for i in range(n_points - k):
                grad[i + k] += (1 - h[i]) * dx
            for i in range(k):
                grad[i] -= h[i + n_points - k] * dx
        
        # Update h
        h_new = h - lr * grad
        
        # Project onto feasible set using cvxpy
        h_var.value = h_new
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
            h = h_var.value.copy()
        except:
            # fallback projection if optimization fails
            h = np.clip(h_new, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = max(signal.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Update learning rate
        iteration += 1
        if iteration % 100 == 0:
            lr *= 0.95
            
    return (best_h, best_c5, n_points)
