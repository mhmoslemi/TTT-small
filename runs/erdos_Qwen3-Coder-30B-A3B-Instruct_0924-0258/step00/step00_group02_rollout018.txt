import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given step function h"""
    dx = 2.0 / len(h)
    # Compute convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    c5 = np.max(conv) * dx
    return c5

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1, sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Fallback: construct a reasonable starting point
        n_points = 45
        h0 = np.ones(n_points) * 0.5  # Start with uniform
        h0 = project_to_feasible(h0, n_points)
        
    # PSO parameters
    n_particles = 30
    max_iter = 1000
    omega = 0.9
    c1 = 2.0
    c2 = 2.0
    v_max = 0.5
    
    # Initialize swarm
    particles = []
    velocities = []
    
    # First particle is the initial solution
    particles.append(h0.copy())
    velocities.append(np.zeros_like(h0))
    
    # Other particles with small random perturbations
    for _ in range(n_particles - 1):
        pert = np.random.normal(0, 0.1, n_points)
        h_pert = h0 + pert
        h_pert = project_to_feasible(h_pert, n_points)
        particles.append(h_pert)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
    
    # Track personal bests and global best
    pBest = [evaluate_erdos_solution(p) for p in particles]
    pBest_positions = [p.copy() for p in particles]
    
    gBest_idx = np.argmin(pBest)
    gBest = pBest[gBest_idx]
    gBest_pos = particles[gBest_idx].copy()
    
    start_time = time.time()
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(n_particles):
            current_c5 = evaluate_erdos_solution(particles[i])
            if current_c5 < pBest[i]:
                pBest[i] = current_c5
                pBest_positions[i] = particles[i].copy()
                
                if current_c5 < gBest:
                    gBest = current_c5
                    gBest_pos = particles[i].copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            # Velocity update
            velocities[i] = (omega * velocities[i] 
                           + c1 * r1 * (pBest_positions[i] - particles[i])
                           + c2 * r2 * (gBest_pos - particles[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply feasibility projection
            particles[i] = project_to_feasible(particles[i], n_points)
        
        omega *= 0.99  # Time-decaying inertia
    
    # Return best solution
    return (gBest_pos.tolist(), gBest, n_points)
