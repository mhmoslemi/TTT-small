import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    """Evaluate the overlap integral C5 for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute cross-correlations for all lags
    corr = signal.correlate(h, 1 - h, mode='full')
    # We only care about lags from -n+1 to n-1
    # But we want the maximum of the positive lags (since the problem
    # is symmetric, negative lags will be same as positive)
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Decide number of points - use initial if available, otherwise 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Ensure it's normalized
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Parameters
    perturb_steps = 20
    step_sizes = [0.1, 0.5, 1.0]
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    no_improve_count = 0
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1:
        # Evaluate current C5 and collect all cross-correlations
        dx = 2.0 / n_points
        corr = signal.correlate(h, 1 - h, mode='full')
        
        # Extract the correlation values at all positive lags
        # The full correlation has size 2*n_points - 1
        # The middle element corresponds to lag 0
        # The lag k (k >= 0) is at index n_points - 1 + k
        # We really only care about positive lags
        max_corr = -1
        max_k = 0
        for k in range(n_points):
            corr_k = corr[n_points - 1 + k]
            if corr_k > max_corr:
                max_corr = corr_k
                max_k = k
        
        # Compute gradient of correlation at max_k
        gradient = np.zeros(n_points)
        # For lag k, we look at h[i] * (1 - h[i + k]) 
        # Gradient with respect to h[i] is:
        # if i + k < n_points: then it's -(1 - h[i + k])
        # i.e. -dx * gradient[i] = (h[i + k] - 1) if i + k < n_points
        for i in range(n_points):
            if i + max_k < n_points:
                gradient[i] = 1 - h[i + max_k]
        
        # Try different step sizes
        best_h_new = None
        best_C5_new = best_C5
        
        for alpha in step_sizes:
            h_new = h - alpha * gradient
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to have correct sum
            h_new *= (n_points / 2.0) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new)
            if c5_new < best_C5_new:
                best_C5_new = c5_new
                best_h_new = h_new.copy()
        
        # If we found an improvement, accept it
        if best_h_new is not None:
            h = best_h_new
            best_C5 = best_C5_new
            no_improve_count = 0
        else:
            # No improvement found, possibly perturb
            no_improve_count += 1
            if no_improve_count >= perturb_steps:
                # Add small random noise
                noise = 0.01 * np.random.randn(n_points)
                h += noise
                # Clip and rescale
                h = np.clip(h, 0, 1)
                h *= (n_points / 2.0) / np.sum(h)
                no_improve_count = 0  # reset count
        
        # Update best solution if needed
        if best_C5_new < best_C5:
            best_C5 = best_C5_new
            best_h = h.copy()
    
    # Final evaluation of best solution
    final_C5 = evaluate_erdos_solution(best_h)
    
    # Ensure it's feasible
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) * (2.0 / n_points) - 1.0) < 1e-10, "Sum constraint violated"
    
    return (best_h.tolist(), final_C5, n_points)
