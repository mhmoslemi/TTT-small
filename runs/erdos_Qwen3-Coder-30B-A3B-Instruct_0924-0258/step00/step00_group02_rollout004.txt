import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for full mode
    corr = correlate(h, 1 - h, mode='full')
    # The overlap is max of these correlations
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from available parent or create one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization parameters
    learning_rate = 0.01
    decay_rate = 0.999
    perturb_prob = 0.05
    perturb_mag = 0.05
    
    # Iterative optimization
    iter_count = 0
    while time.time() - start_time < budget_s - 1:  # leave a bit of margin
        current_time = time.time()
        
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        
        # Find the index of maximum correlation
        max_idx = np.argmax(corr)
        
        # Compute gradient - the derivative of correlation w.r.t h
        # Using the correlation definition: (h * (1-h))[k] = sum(h[i] * (1 - h[i+k]))
        # Gradient w.r.t h[i] is: (1-h[i+k]) - h[i+k] = 1 - 2*h[i+k] for the relevant position
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # correlate through the full array, but we need local gradient
            # Approximate gradient from the shifted convolution
            # A simpler approach: L2 approximation
            k = max_idx - n_points + 1  # determine lag
            if 0 <= i - k < n_points:
                gradient[i] = 1 - 2 * h[i - k]
        
        # Use gradient to update
        h = h - learning_rate * gradient
        
        # Perturb occasionally
        if np.random.random() < perturb_prob:
            h = h + perturb_mag * (np.random.random(n_points) - 0.5)
        
        # Project back to feasible set
        h = project_h(h, n_points)
        
        # Evaluate new solution
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Anneal learning rate
        learning_rate *= decay_rate
        iter_count += 1
        
        # Early termination if we're close to target
        if best_c5 < 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
