import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for a given h."""
    # Compute correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # We only need the valid lags (non-zero padding entries)
    # The maximum overlap is the maximum of these correlations
    return np.max(corr) * dx

def rescale_h(h, target_sum):
    """Rescale h to have the desired sum."""
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.full_like(h, target_sum / len(h))
    return h * (target_sum / current_sum)

def clip_h(h):
    """Clip h to [0,1] range."""
    return np.clip(h, 0, 1)

def mutate_h(h, mutation_rate=0.1, delta=0.1):
    """Mutate a step function h."""
    h = h.copy()
    mask = np.random.random(len(h)) < mutation_rate
    h[mask] += np.random.uniform(-delta, delta, np.sum(mask))
    h = clip_h(h)
    return rescale_h(h, len(h) / 2)

def crossover_h(h1, h2):
    """Perform one-point crossover between two step functions."""
    h1 = h1.copy()
    h2 = h2.copy()
    point = np.random.randint(1, len(h1))
    h1[point:], h2[point:] = h2[point:], h1[point:]
    return h1, h2

def tournament_selection(population, fitnesses, k=3):
    """Perform tournament selection."""
    idxs = np.random.choice(len(population), size=k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = clip_h(h_parent)
        h_parent = rescale_h(h_parent, n_points / 2)
    else:
        h_parent = np.ones(n_points) * 0.5
        h_parent = rescale_h(h_parent, n_points / 2)
    
    # Initialize population
    population = [h_parent]
    for _ in range(49):
        h = h_parent.copy()
        # Random perturbation
        h += np.random.uniform(-0.2, 0.2, n_points)
        h = clip_h(h)
        h = rescale_h(h, n_points / 2)
        # Or create a random feasible solution
        if np.random.random() < 0.5:
            h = np.random.random(n_points)
            h = rescale_h(h, n_points / 2)
        population.append(h)
    
    # Main loop
    best_h = h_parent.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitnesses = []
        for h in population:
            fitness = evaluate_h(h, dx)
            fitnesses.append(fitness)
        
        # Track best individual
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
        
        # Select parents via tournament
        new_population = [best_h]  # Elitism
        while len(new_population) < 50:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.random() < 0.7:
                child1, child2 = crossover_h(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutate  
            child1 = mutate_h(child1)
            child2 = mutate_h(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:50]
    
    # Final evaluation
    final_fitnesses = [evaluate_h(h, dx) for h in population]
    final_min_idx = np.argmin(final_fitnesses)
    return (population[final_min_idx], final_fitnesses[final_min_idx], n_points)
