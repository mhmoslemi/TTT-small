import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        # Fallback: create symmetric construction
        n_points = 70
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.3
        h0[-1] = 0.7
        
    dx = 2.0 / n_points
    h = h0.copy()
    
    # Binary search bounds for C5
    low, high = 0.0, 0.38085
    start_time = time.time()
    
    # Continuous optimization with cutting planes
    while high - low > 1e-4 and time.time() - start_time < budget_s - 5:
        C5 = (low + high) / 2.0
        
        # Set up CVXPY problem
        h_vars = cp.Variable(n_points, name="h")
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add approximated constraints for each shift
        for k in range(n_points):
            # This is a linear approximation to quadratic terms
            # h[i] * (1 - h[i+k]) <= ( h[i] + 1 - h[i+k] ) / 2
            # But we derive it more carefully based on current h
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_vars[i] * (1 - h_vars[j])
            constraints.append(expr <= C5 / dx)
            
        # Objective - we minimize a dummy variable to not affect the problem
        obj = cp.Minimize(0)
        
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except:
            # if we can't solve, fallback quickly
            low = C5
            continue
            
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_new = h_vars.value
            if h_new is not None:
                h = h_new.copy()
                high = C5
            else:
                low = C5
        else:
            low = C5
            
    # Final evaluation of solution
    h_final = np.clip(h, 0, 1)
    # Normalize to satisfy integral constraint
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    c5 = max(np.correlate(h_final, 1 - h_final, mode='full')) * dx
    
    return (h_final, c5, n_points)
