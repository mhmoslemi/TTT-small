import numpy as np
import time
from scipy import signal

def evaluate_c5(h, dx):
    """Evaluate C5 = max_k integral h(x)(1-h(x+k)) dx"""
    # Compute correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Compute maximum value, scaled by dx
    max_corr = np.max(corr) * dx
    return max_corr

def rescale_h(h, dx):
    """Rescale h to ensure integral equals 1"""
    current_sum = np.sum(h)
    target_sum = 1.0 / dx  # since sum(h) * dx = 1
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def initialize_particles(n_points, n_particles, use_initial=False):
    """Initialize particles for PSO"""
    particles = []
    if use_initial and 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = rescale_h(h0, 2.0 / n_points)
    else:
        # Random initialization with sum constraint
        h0 = np.random.rand(n_points)
        h0 = rescale_h(h0, 2.0 / n_points)

    for _ in range(n_particles):
        # Perturb the initial solution slightly
        h = h0 + np.random.normal(0, 0.05, n_points)
        particles.append(rescale_h(h, 2.0 / n_points))
    return particles

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use fewer points for faster optimization
    n_points = 64  # Should be enough, small enough for speed
    dx = 2.0 / n_points
    # Initialize PSO parameters
    n_particles = 30
    max_iterations = 10000
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    velocity_limit = 0.1
    
    # Initialize particles
    particles = initialize_particles(n_points, n_particles, use_initial=True)
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Personal best
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [evaluate_c5(p, dx) for p in particles]
    
    # Global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # Optimization loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1.0:
            break
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (personal_best[i] - particles[i]) +
                social * r2 * (global_best - particles[i])
            )
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints: clip values to [0, 1] and rescale
            particles[i] = rescale_h(particles[i], dx)
            
            # Evaluate new position
            new_score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if new_score < personal_best_scores[i]:
                personal_best_scores[i] = new_score
                personal_best[i] = particles[i].copy()
            
            # Update global best
            if new_score < global_best_score:
                global_best_score = new_score
                global_best = particles[i].copy()
        
        # Check if we should terminate early due to convergence
        if iteration % 100 == 0:
            # Small tolerance
            if iteration > 0 and abs(global_best_score - personal_best_scores[global_best_idx]) < 1e-6:
                break
                
    # Final evaluation
    final_c5 = evaluate_c5(global_best, dx)
    
    # Return best solution found
    return (global_best, final_c5, n_points)
