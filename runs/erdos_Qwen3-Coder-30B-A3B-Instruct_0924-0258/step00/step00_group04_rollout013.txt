import numpy as np
import time
from scipy.signal import correlate
import math

def evaluate_c5(h):
    """Evaluate C5 bound for given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # Extract the valid correlations for k=0 to 2 (in terms of samples)
    # The full correlation has 2*n_points-1 elements, we want middle portion
    # indices: [0, 1, ..., n_points-1, ..., 2*n_points-2]
    # The shift k goes from 0 to n_points-1 in positive and negative values
    # For our constraint: k ranges from 0 to n_points-1, so we only take
    # indices from n_points-1 to 2*n_points-2
    # We want max of correlations over shifts from 0 to n_points-1
    # That's from index n_points-1 to 2*n_points-2 inclusive
    corr_values = corr[n_points-1:2*n_points-1]
    return np.max(corr_values) * dx

def project_and_rescale(h, n_points):
    """Project h to [0,1] and rescale to satisfy integral constraint"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale so sum(h) = n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, eps=1e-6):
    """Compute approximate gradient of C5 with respect to h"""
    n_points = len(h)
    dx = 2.0 / n_points
    current_c5 = evaluate_c5(h)
    grad = np.zeros(n_points)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_and_rescale(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus)
        grad[i] = (c5_plus - current_c5) / eps
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42  # Use same as current best to keep size small
    dx = 2.0 / n_points
    
    # Start with initial_h_values if available, else uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_and_rescale(h, n_points)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = project_and_rescale(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Simulated Annealing parameters
    temp = 1.0
    cooling_factor = 0.99
    min_temp = 0.01
    patience = 50
    patience_counter = 0
    gradient_scale = 0.1
    noise_scale = 0.05
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Generate candidate solution
        h_candidate = h.copy()
        
        # Add gradient guidance
        grad = compute_gradient(h)
        if np.linalg.norm(grad) > 1e-10:
            # Normalize gradient and scale it
            grad = grad / np.linalg.norm(grad) * gradient_scale
            h_candidate += grad
        
        # Add noise
        noise = np.random.normal(0, noise_scale, n_points)
        h_candidate += noise
        
        # Project and rescale
        h_candidate = project_and_rescale(h_candidate, n_points)
        
        # Evaluate
        c5_candidate = evaluate_c5(h_candidate)
        
        # Accept or reject
        delta = c5_candidate - best_c5
        if delta < 0 or np.random.rand() < math.exp(-delta / temp):
            h = h_candidate
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
                patience_counter = 0
            else:
                patience_counter += 1
        else:
            patience_counter += 1
            
        # Update temperature
        temp = max(temp * cooling_factor, min_temp)
        
        # Warm restart if too many failed attempts
        if patience_counter >= patience:
            temp = 1.0  # Reset temperature
            patience_counter = 0
    
    # Final check
    final_c5 = evaluate_c5(best_h)
    
    return (list(best_h), final_c5, n_points)
