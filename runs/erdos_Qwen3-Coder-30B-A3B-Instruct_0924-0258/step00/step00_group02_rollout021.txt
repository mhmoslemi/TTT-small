def evaluate_erdos_solution(h, dx):
    # Compute overlap integral for the given h
    # Using numpy's correlate for efficiency
    correlation = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(correlation)
    return max_overlap * dx

def project_to_feasible(h, n_points):
    # Clip h to [0,1] and rescale to sum to n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def convex_surrogate(h, n_points, dx):
    # Use a quadratic surrogate that approximates the max term
    # The superscript 2 in the original plan creates a convex problem
    correlation = np.correlate(h, 1-h, mode="full")
    # Quadratic sum of overlaps approximates max
    return np.sum(correlation**2) * dx**2

def gradient_convex_surrogate(h, n_points, dx):
    # Gradient of convex surrogate with respect to h values
    # Compute correlation of h and (1-h) to get all the terms
    # Correlation gives d/dx [h(x) * (1-h(x+k))]
    # Differentiating the surrogate:
    # d/dh_i surr = sum_k 2 * [ h * (1-h) ][k] * d/dh_i [h * (1-h) ][k]
    # We do this manually by computing the stress of 1-h and h
    h1 = np.array(h)
    h2 = 1 - np.array(h)
    corr = np.correlate(h1, h2, mode="full")
    # Now we compute the gradient using chain rule, very approximated approach
    # The gradient of the quadratic sum is 2 * each term
    # This simplified version uses standard finite difference gradient
    
    grad = np.zeros_like(h)
    eps = 1e-6
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        val_plus = convex_surrogate(h_plus, n_points, dx)
        val_minus = convex_surrogate(h_minus, n_points, dx)
        grad[i] = (val_plus - val_minus) / (2 * eps)
    return grad

def optimize_resolution(h0, n_points, dx, time_limit, elapsed_time):
    # Use SLSQP to optimize the convex surrogate
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    h = h0.copy()
    
    # Define objective and constraints
    def obj(h):
        return convex_surrogate(h, n_points, dx)
        
    def jac(h):
        return gradient_convex_surrogate(h, n_points, dx)
        
    def constraint_func(h):
        return np.sum(h) - n_points/2
    
    cons = {'type': 'eq', 'fun': constraint_func}
    
    time_left = time_limit - elapsed_time
    if time_left < 0.1:  # Not enough time to do anything
        return h0
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6})
        if res.success:
            h = res.x
    except Exception:
        pass
    
    h = project_to_feasible(h, n_points)
    return h
