import numpy as np
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    # Compute all overlaps for all lags
    correlation = np.correlate(h, 1 - h, mode="full")
    # The valid overlaps are from index n_points-1 to 2*n_points-2
    valid_correlations = correlation[len(h)-1:]
    return np.max(valid_correlations) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with provided construction or fallback to uniform
    if 'initial_h_values' in globals():

        n_points = len(h)

    dx = 2.0 / n_points
    # Ensure constraint sum(h) * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Initial evaluation
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Optimization loop
    iteration = 0
    
    while time.time() - start_time < budget_s - 5:  # Leave 5s safety
        iteration += 1
        if iteration > 100:
            break
        # Evaluate all overlaps
        # The convolution-based approach in numpy gives us overlaps
        correlation = np.correlate(h, 1 - h, mode="full")
        # Valid overlaps correspond to lags from 0 to n_points-1 (overlap_{k+1})
        valid_correlations = correlation[len(h)-1:]  # From k=0 to k=n_points-1
        max_idx = np.argmax(valid_correlations)
        max_c5 = valid_correlations[max_idx]
        # The corresponding k is max_idx
        k = max_idx
        
        if max_c5 <= best_c5 + 1e-8:
            # Already at good solution
            break
            
        # Convex optimization to find better h
        # We solve the problem of minimizing max over lags of integral h(x)(1-h(x+k))
        # Consider lag k where we have the maximum overlap
        # So we should reduce h[i]*(1-h[i+k]) at that point for all i where i+k < n
        # But this is a bilinear constraint, so we approximate it.
        # Let variables be h[i] in [0,1], and sum(h) = n_points/2  
        # Objective: minimize C
        # Constraints: for all valid i with k in [0, n_points-1]:
        # sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k]) <= C => sum_{i=0}^{n-k-1} h[i] - h[i]*h[i+k] <= C
        
        # Let's optimize by modifying h around the worst case
        # Trying a simpler modification: for worst lag k, adjust specific indices
        
        # Instead of CVXPY, let's use a straightforward gradient descent approach on some subset of indices
        # Just find two indices and adjust to decrease max overlap
        h_new = h.copy()
        
        # For the k that gives max overlap, we penalize h[i] * (1 - h[i+k])  
        # We look at i such that i+k < n, so i ranges from 0 to n-k-1
        i_indices = list(range(len(h)-k))
        # Get specific donor and recipient indices
        max_h_idx = np.argmax(h[:len(h)-k])
        min_h_idx = np.argmin(h[k:])
        h_idx = max_h_idx + k  # for h[i+k]
        # Modify h_i and h_{i+k}
        h_diff = 0.01
        # Try adjusting h[max_h_idx] and h[h_idx] to lower overlap
        h_new[max_h_idx] = max(0, min(1, h[max_h_idx] - h_diff))
        h_new[h_idx] = max(0, min(1, h[h_idx] + h_diff))
        
        # Ensure the constraint sum(h) = n_points/2 is still satisfied  
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new overlap
        curr_c5 = evaluate_c5(h_new, dx)
        if curr_c5 < best_c5:
            h = h_new
            best_c5 = curr_c5
            best_h = h.copy()
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
