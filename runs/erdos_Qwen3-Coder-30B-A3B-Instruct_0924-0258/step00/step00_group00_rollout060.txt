import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Normalize h to satisfy constraint sum(h) * dx = 1
    dx = 2.0 / n_points
    h = h * (n_points / 2.0) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    perturb_steps = 20
    no_improve_count = 0
    alpha_candidates = [0.1, 0.5, 1.0]
    
    # Gradient descent loop
    while time.time() - start_time < budget_s - 0.1:
        # Compute all cross-correlations
        # We want max_k int h(x)(1-h(x+k))dx
        # This is equivalent to max_k correlate(h, 1-h, mode='full')
        h_rev = 1.0 - h
        corr = correlate(h, h_rev, mode='full')
        # The indices correspond to lags from -n+1 to n-1
        # So the effective lags we consider are from 0 to 2*n-2, corresponding to 
        # lags k=-n+1,...,n-1. We want the max of correlation values where lag is in [0, n-1]
        # But h is defined on [0,2], so k in [0,2] -> lag in [0,n_points-1]
        # We only care about corrs where k >= 0, i.e., corr indices n_points-1 to 2*n_points-2
        # But actually, we want correlation up to lag n_points-1, and lags k = 0 to n_points-1 in our domain
        # So we need corr offset, index 0 corresponds to lag = -n_points+1, ..., index n_points-1 corresponds to lag 0 
        # We care about lags 0, ..., n_points-1, so we look at indices n_points-1 to 2*n_points-2
        corr_indices = np.arange(n_points-1, 2*n_points-1)
        max_k_idx = np.argmax(corr[corr_indices])
        max_k = max_k_idx - (n_points-1)
        
        # Compute gradient for this max_k
        # The gradient with respect to h for lag k is -h_rev[k:] for k >= 0
        # if k=0, grad[i] = -h_rev[i]  
        # if k=1, grad[i] = -h_rev[i+1] for i from 0 to n_points-k-1
        # At lag k, we compute derivative of sum h[i] (1 - h[i+k]) = sum h[i] - sum h[i] h[i+k]
        # d/dh[i] of that is 1 - h[i+k] 
        # but it's actually just gradient w.r.t. the sum h[i] (1 - h[i+k]) over all i
        # that gives us d/dh[i] (h[i] (1-h[i+k])) = (1-h[i+k]) - h[i] * d/dh[i] h[i+k]
        # But d/dh[i] h[i+k] = 0 unless i+k < n_points
        # So in our case, the gradient w.r.t. h[i] of h[i] (1 - h[i+k]) is (1-h[i+k]) , if i+k < n_points.
        # For the full sum we take the total gradient vector: a diagonal causal matrix appears
        # Actually let's think in terms of vectors ... 
        # The correlation of h with (1-h), with mode='full': 
        # If we look at the output of `correlate(h, 1-h, mode='full')`:
        # output_i = sum over j of h[j] * (1-h[j-i]) from j=0 to n-1-i (as well as ensuring j+i < n)  
        # The gradient of this single output element is -h_rev[j-i] for valid j
        # for maximum, we suppose maximum is at lag -n+1+k, i.e., i-th element of corr, where i = n-1+k
        # Then for h[i], gradient contribution is -h_rev[i+k], if i+k < n, else it's 0
        # The gradient vector for correlation k (output element index n-1+k) of h is thus:
        #gradient = np.zeros(n_points)
        #for i in range(n_points):
        #    if i+k < n_points:
        #        gradient[i] = -h_rev[i+k] 
        # But this will be applied in top of full h, so ...
        # See if we can make it efficient:
        # For lags k=0 to n_points-1 in the full correlation, retrieve corr indices
        # keep only the first n_points elements that correspond to offsets k in [0, n-1] 
        # Length of correlation is 2*npoints-1. Elements correspond to lag = -n+1, ..., n-1  
        # The actual finite interval contribution for lags 0, ..., n_points-1 is at indices
        # n_points-1 to 2*n_points-2
        # Therefore, for max_k = 0 to n_points-1 (this k), this k corresponds to index
        # n_points-1 + k, so apply step: gradient = -h_rev[k:] for i=0 to n_points-k-1
        # So we have gradient = - h_rev[max_k:] padded with zeros in front.
        # Because of zero-padding, only valid part: gradient[i] := -h_rev[i + max_k], for i from 0 to n_points - max_k - 1
        # But we only want zero for i that is out of bounds for the valid corr contribution of h[i]* h_rev[i+max_k]
        # In fact, for h[i], the influence on the correlation sum is h[i] * (1-h[i+k]) and gradient wrt h[i]
        # is (1 - h[i+k]) for i+k < n_points. 
        # For the i-th entry of the gradient vector for lag k, we have:
        # For all m such that h[m] appears in terms of the sum for output_index k + n-1, i.e. i = m + k 
        # After indexing, i -> m = i - k, i-k >= 0 -> m >= 0 and i-k < n -> m < n
        # and output_index = k + n-1 -> that is just index of the surface wrt the diagonal of length 2*n-1
        # ok, direct computation:
        gradient = np.zeros(n_points)
        grad_lag = max_k
        if grad_lag < n_points:
            for i in range(n_points - grad_lag):
                gradient[i] += -h_rev[i + grad_lag]
        
        # Try different step sizes
        best_h_new = h.copy()
        best_c5_new = best_c5
        for alpha in alpha_candidates:
            h_new = h - alpha * gradient
            # Clip h_new to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Rescale to sum(h_new) = n_points / 2
            current_sum = np.sum(h_new)
            if current_sum > 0:
                h_new = h_new * (n_points / 2) / current_sum
                
            c5_new = evaluate_c5(h_new, dx)
            
            if c5_new < best_c5_new:
                best_c5_new = c5_new
                best_h_new = h_new.copy()
        
        # If no improvement, perturb
        if best_c5_new >= best_c5:
            no_improve_count += 1
            if no_improve_count >= perturb_steps:
                # Random perturbation
                h_noisy = h + 0.01 * np.random.randn(n_points)
                h_noisy = np.clip(h_noisy, 0, 1)
                current_sum = np.sum(h_noisy)
                if current_sum > 0:
                    h_noisy = h_noisy * (n_points / 2) / current_sum
                c5_noisy = evaluate_c5(h_noisy, dx)
                if c5_noisy < best_c5:
                    best_c5 = c5_noisy
                    best_h = h_noisy.copy()
                no_improve_count = 0
        else:
            # Improvement found
            best_c5 = best_c5_new
            best_h = best_h_new.copy()
            no_improve_count = 0
        
        h = best_h.copy()
        
        if time.time() - start_time > budget_s - 0.1:
            break
            
    return best_h.tolist(), best_c5, n_points

def evaluate_c5(h_values, dx):
    h = np.array(h_values)
    h_rev = 1.0 - h
    corr = correlate(h, h_rev, mode='full')
    # corr indices from -n+1 to n-1, corresponding to lags
    # indices 0 to n-1 in corr represent lags -n+1 to -1
    # indices n to 2*n-2 represent lags 0 to n-1
    # We care about lags 0 to n-1, which are indices n to 2*n-2
    corr_indices = np.arange(len(corr)) 
    valid_indices = (corr_indices >= len(h)-1) & (corr_indices < 2*len(h)-1)
    max_corr = np.max(corr[valid_indices])
    return max_corr * dx
