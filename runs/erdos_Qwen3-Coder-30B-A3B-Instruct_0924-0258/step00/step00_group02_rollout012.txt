import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute overlap integral for all k values
    overlap = signal.correlate(h, 1 - h, mode="full") * dx
    # Return the maximum overlap
    return np.max(overlap)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 45
    dx = 2.0 / n_points
    population_size = 50
    generations = 12
    mutation_rate = 0.1
    elite_size = population_size // 2
    max_time = start_time + budget_s - 1  # Leave 1s margin
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():

    else:
        # Create a reasonable starting vector
        h0 = np.ones(n_points) * 0.5
    # Ensure sum constraint
    h0 = h0 * (n_points / 2) / np.sum(h0)
    population.append(h0)
    
    # Add random perturbations
    for _ in range(population_size - 1):
        h = h0 + np.random.randn(n_points) * 0.1
        h = np.clip(h, 0, 1)
        # Rescale to fit sum constraint
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Track best solution
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Evolve the population
    for gen in range(generations):
        if time.time() > max_time:
            break
            
        # Evaluate all candidates
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness.append(c5)
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Select elite
        elites = population[:elite_size]
        
        # Generate offspring
        offspring = []
        for _ in range(elite_size):
            # Select two parents
            parent1 = elites[np.random.randint(0, len(elites))]
            parent2 = elites[np.random.randint(0, len(elites))]
            
            # Single-point crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            mutate_indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
            for i in mutate_indices:
                child1[i] += np.random.randn() * 0.1
                child2[i] += np.random.randn() * 0.1
            
            # Ensure feasibility
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            offspring.append(child1)
            offspring.append(child2)
        
        # Replace population
        population = elites + offspring[:elite_size]
        
    # Return final solution
    return (best_h.tolist(), best_c5, n_points)
