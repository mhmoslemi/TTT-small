import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the overlap integral for given h."""
    dx = 2.0 / len(h)
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # The maximum correlation (shifts from -len+1 to len-1)
    max_corr = np.max(corr)
    return max_corr * dx

def normalize_h(h):
    """Normalize h to sum to 1 with step size 2/n."""
    n = len(h)
    dx = 2.0 / n
    scale = 1.0 / (np.sum(h) * dx)
    h = h * scale
    return np.clip(h, 0, 1)

def mutate_h(h, rate=0.1):
    """Apply bounded mutation to h."""
    n = len(h)
    mask = np.random.random(n) < rate
    noise = np.random.normal(0, 0.05, n)
    h = h + noise * mask
    return normalize_h(h)

def crossover_h(h1, h2):
    """Uniform crossover between two h."""
    n = len(h1)
    mask = np.random.random(n) > 0.5
    child = np.where(mask, h1, h2)
    return normalize_h(child)

def tournament_selection(population, fitnesses, k=5):
    """Select one individual using tournament selection."""
    idxs = np.random.choice(len(population), size=k)
    best = np.argmin([fitnesses[i] for i in idxs])
    return population[idxs[best]]
