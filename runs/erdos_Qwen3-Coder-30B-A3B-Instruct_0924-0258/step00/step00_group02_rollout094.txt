import numpy as np
from scipy import signal
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for a given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode="full")
    # Compute maximum value and scale it
    max_val = np.max(conv) * dx
    return max_val

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed for reproducibility
    random.seed(seed)
    np.random.seed(seed)
    
    # Read or create initial construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Create a simple initial construction
        n_points = 45
        h0 = np.ones(n_points) * 0.5  # uniform initial guess
        
    # Initialize PSO parameters
    n_particles = 30
    max_iter = 1000
    omega = 0.9
    c1 = 2.0
    c2 = 2.0
    velocity_max = 0.5
    velocity_min = -0.5
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # First particle is initial solution
    particles.append(h0.copy())
    pbest.append(h0.copy())
    pbest_values.append(evaluate_erdos_solution(h0))
    
    # Generate rest of particles
    for _ in range(n_particles - 1):
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        h = h0 + perturbation
        h = project_to_feasible(h, n_points)
        particles.append(h)
        pbest.append(h)
        pbest_values.append(evaluate_erdos_solution(h))
        
    # Initialize global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx]
    gbest_value = pbest_values[gbest_idx]
    
    # Initialize velocities
    for _ in range(n_particles):
        velocities.append(np.random.uniform(velocity_min, velocity_max, n_points))
        
    # PSO loop
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Update each particle
        for i in range(n_particles):
            # Evaluate current particle
            current_value = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest_values[i] = current_value
                pbest[i] = particles[i].copy()
                
            # Update global best
            if current_value < gbest_value:
                gbest_value = current_value
                gbest = particles[i].copy()
                
        # Update velocities and positions
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (omega * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], velocity_min, velocity_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply feasibility projection
            particles[i] = project_to_feasible(particles[i], n_points)
            
        # Update inertia weight (time-decaying)
        omega = max(0.4, omega * 0.99)
        
    # Return best solution found
    return (gbest.tolist(), gbest_value, n_points)
