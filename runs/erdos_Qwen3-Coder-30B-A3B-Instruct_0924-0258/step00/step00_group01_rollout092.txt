import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if abs(np.sum(h) - target_sum) > 1e-6:
            h = h * target_sum / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * target_sum / np.sum(h)
    
    def compute_C5(h):
        # using np.correlate with mode="full" to get all lags
        corr = np.correlate(h, 1 - h, mode="full")
        # take only the positive lags (for C5 definition)
        # C5 is max over all lags k from 0 to n-1 where k is the lag
        # the correlator has size 2*n-1, so we want indices from n-1 to 2*n-2
        max_corr = np.max(corr[n_points-1:])
        return max_corr * dx
    
    # Initial computation
    current_C5 = compute_C5(h)
    best_h = h.copy()
    best_C5 = current_C5
    last_improvement = 0
    iteration = 0
    alpha = 0.1
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 0.1 and iteration < max_iter:
        # Compute all correlations efficiently
        corr_full = np.correlate(h, 1-h, mode="full")
        corr_positive = corr_full[n_points-1:]
        
        max_k = np.argmax(corr_positive)
        max_val = corr_positive[max_k]
        
        # Compute gradient
        # This corresponds to ∂/∂h_i of the overlap at lag max_k
        # Let's compute manually for performance
        grad = np.zeros_like(h)
        
        # For lag k=0, the contribution is 1 - h[i]
        # For other lag k, it's related to h[(i-k)%n] or h[(i+k)%n]
        # Simplify to a direct way to compute the gradient contribution
        # Let's translate from the expression directly using convolution derivative
        
        # velocity^i = (1-h_i) [when aligned at lag] 
        # for lag k, shift 1-h by k, so the product is:
        # h[i] * (1 - h[(i+k)%n])
        # gradient: d/dh_i = (1 - h[(i+k)%n]) - h[(i-k)%n] 
        # But must also account for sum constraint later.
        # Let's approximate with the gradient of the max term estimation:
        
        # Direct computation of error
        if max_k > 0:
            # For lag k>0:
            # contribution of h[i] to overlap k is h[i] * (1 - h[(i+k)%n])
            # So ∂/∂h[i] =  1 - h[(i+k)%n] 
            # But this is  
            #     1 - h[(i+k)%n] if we're at lag k=1,      # but we have a fixed k
            #     [this is identity in the case of most weight in error]
            # Actually better: we minimize: max_k int h(x) (1-h(x+k)) dx,
            # Let us look closer: 
            # L = sum_i (h[i] * (1-h[i+k])) * dx 
            # dL/dh[i] = (1-h[i+k]) * dx
            # So at the maximum: we want to decrease the error. 
            # So we want: h[i] -= alpha * (1-h[i+k])
            # Let's experiment:
            # This changes logic slightly. For galerkin-type optimization:
            # Gradient is such that: h[i] += alpha * grad, 
            # where grad[i] is: (1 - h[(i+k)%n]) 
            # This follows from chain rule labelling.
            i_start = 0
            # exact propagation of shift from i to i+k
            grad = np.roll(1 - h, k=-max_k)  # effectively d/dh[i] = (1 - h[(i+k)%n])
        else:
            grad = 1 - h  # simple case of k=0
            
        # Modifying: h[i] += alpha * grad[i]
        # Since simplest analysis gives d/dh[i] = (1 - h[i +k])
        # but from consideration of term identity for overlap,
        # we're trying to make the overlap smaller:  
        # we add to h[-] a negative contribution to offset this.

        # Indeed (noted correctly):  
        # The overlap term is (at constant shift: h[i] * 1 - h[i+k]),  
        # So a linear decrease with change of h[i+k], 
        # Gradient w.r.t. h[i] is 1 - h[i+k].  
        # But also constrain it to keep source condition sum(h) = target_sum  
        # Approach: compute exact update for k spike
        # What happens if we knew a-lag term gives the spike?  
        # So 
        shift_towards = max_k  # which value causes this?
        
        # A crucial shift. Instead of gradient steps, direct model update:
        # h_t = h_{t-1} - alpha * grad = h + alpha * (1 - h_{[i+max_k]}) 
        # But more precise:
        # For constraint preserving, update: 
        old_h = h.copy()
        
        # Materialize gradient update
        # we compute grad[i] = (1-h[(i+max_k)%n]), then update h += alpha * grad
        # but most accurate is to actually say,
        # derivative with respect to i is: (1-h[i+max_k]), so let that be it in the gradient.
        # Now, after update, we correct for value boundary and sum constraint.
        # h += alpha * np.roll(1-h, -max_k)
        h = h + alpha * np.roll(1 - h, -shift_towards)
        
        # Keep in [0,1]
        h = np.clip(h, 0, 1)
        
        # Correct sum to match constraint while preserving ratio.
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-6:
            h = h * target_sum / h_sum

        # Check for improvement
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            last_improvement = iteration
        
        iteration += 1

        # Reduce learning rate if no improvement for a while
        if iteration - last_improvement > 20:
            alpha *= 0.95
            if alpha < 1e-6:
                alpha = 1e-6
        
        # occasional random perturbation to escape local minima
        if np.random.random() < 0.05:
            # slightly perturb
            perturb_mag = 0.02
            h = h + np.random.normal(0, perturb_mag, n_points)
            # clip
            h = np.clip(h, 0, 1)
            # adjust sum
            h_sum = np.sum(h)
            h = h * target_sum / h_sum

    # Final validation
    final_C5 = compute_C5(best_h)
    assert np.abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= 0) & (best_h <= 1))
    
    return (best_h.tolist(), final_C5, n_points)
