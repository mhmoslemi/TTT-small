import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction if available

    except:

    h_best = h0.copy()
    c5_best = evaluate_erdos_solution(h0)
    
    # Multi-resolution approach
    resolutions = [10, 20, 40, 70]
    for n in resolutions:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Initialize if needed
        if n == 10:
            h = np.random.rand(n)
            h = h / np.sum(h) * (n / 2)
        else:
            # Interpolate from previous resolution
            h = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h_best)), h_best)
            h = h / np.sum(h) * (n / 2)
        
        # Simulated Annealing
        T = 0.1
        cooling_rate = 0.99
        max_iter = 500
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Perturb
            h_new = h + np.random.normal(0, T, n)
            h_new = np.clip(h_new, 0, 1)
            # Project to satisfy constraint
            h_new = h_new / np.sum(h_new) * (n / 2)
            
            c5_new = evaluate_erdos_solution(h_new)
            
            # Accept or reject
            delta = c5_new - c5_best
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_new
                if c5_new < c5_best:
                    h_best = h_new.copy()
                    c5_best = c5_new
            
            T *= cooling_rate
            
    return (h_best, c5_best, len(h_best))

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute the full correlation
    corr = correlate(h, 1 - h, mode="full")
    # The maximum value of the overlap integral for all k
    max_corr = np.max(corr)
    return max_corr * dx
