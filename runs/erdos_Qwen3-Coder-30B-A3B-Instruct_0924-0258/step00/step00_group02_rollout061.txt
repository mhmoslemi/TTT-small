import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    dx = 2.0 / len(h_values)
    overlap = correlate(h_values, 1 - h_values, mode="full") * dx
    return np.max(overlap)

    population_size = 50
    generations = 12

        perturbation = np.random.normal(0, 0.1, n_points)
        new_h = initial_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        # Rescale to meet constraint
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        population.append(new_h)
    
    # Track best solution
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Evolution
    for gen in range(generations):

            c5 = evaluate_erdos_solution(h)
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            
        # Select elite
        elite = population[:elite_size]
        
        # Generate offspring through crossover and mutation
        offspring = []
        while len(offspring) < population_size - elite_size:
            # Select two parents randomly
            p1, p2 = np.random.choice(elite_size, 2, replace=False)
            parent1, parent2 = elite[p1], elite[p2]
            
            # Single point crossover
            crossover_point = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutate children
            for child in [child1, child2]:
                # Apply mutation
                indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
                for i in indices:
                    child[i] += 0.1 * np.random.randn()
                child = np.clip(child, 0, 1)
                
                # Rescale to meet constraint
                child = child * (n_points / 2) / np.sum(child)
                
                offspring.append(child)
                
        # Form new population
        population = elite + offspring[:population_size - elite_size]
        
    return (best_h.tolist(), best_c5, n_points)
