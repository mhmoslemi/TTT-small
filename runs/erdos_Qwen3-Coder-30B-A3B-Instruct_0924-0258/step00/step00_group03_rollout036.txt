import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C_5 bound for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    correlations = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum of these
    max_overlap = np.max(correlations) * dx
    return max_overlap

def project_h(h):
    """Project h to feasible region: [0,1] and sum = n/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    n = len(h)
    if h_sum > 0:
        h = h * (n / 2) / h_sum
    return h

def mutate_h(h, sigma):
    """Mutate h by adding Gaussian noise then projecting."""
    h_new = h + np.random.normal(0, sigma, len(h))
    return project_h(h_new)

def crossover_h(h1, h2, alpha=0.6):
    """Blend crossover between h1 and h2."""
    beta = np.random.uniform(alpha, 1 - alpha)
    h_new = beta * h1 + (1 - beta) * h2
    return project_h(h_new)

def compute_fitness(h):
    """Compute the fitness (C_5) for h."""
    return evaluate_erdos_solution(h)

    population = [h0.copy()]
    
    # Generate diversified initial population
    for _ in range(pop_size - 1):
        h_mut = mutate_h(h0, 0.1)
        population.append(h_mut)
    
    # For low computational budget: allow a few hundred generations
    max_gens = 1000
    gen = 0
    
    while gen < max_gens and (time.time() - start_time) < budget_s:

        # Evaluate fitness for all population
        fitness = [compute_fitness(h) for h in population]

        selected = []

            idx1, idx2 = np.random.randint(0, pop_size, 2)
            if fitness[idx1] < fitness[idx2]:
                selected.append(population[idx1])
            else:
                selected.append(population[idx2])
        
        # Crossover
        offspring = []
        for _ in range(pop_size // 2):
            parent1 = selected[np.random.randint(0, pop_size)]
            parent2 = selected[np.random.randint(0, pop_size)]
            child = crossover_h(parent1, parent2, 0.6)
            offspring.append(child)
        
        # Mutation
        mutated_offspring = []
        decay = min(1.0, gen / 100.0)
        sigma = 0.1 * (1 - decay)

            mutated = mutate_h(h, sigma)
            mutated_offspring.append(mutated)

        population = mutated_offspring
