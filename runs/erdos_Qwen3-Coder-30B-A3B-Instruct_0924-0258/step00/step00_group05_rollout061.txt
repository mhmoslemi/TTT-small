def evaluate_C5(h):
    # Normalize dx
    dx = 2.0 / len(h)
    # Compute correlation
    corr = signal.correlate(h, 1 - h, mode='full')
    # Take maximum over all shifts
    max_c5 = np.max(corr) * dx
    return max_c5
