import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant C5 for given step function h."""
    dx = 2.0 / len(h)
    # Compute the overlap integral for all lags
    correlation = correlate(h, 1 - h, mode='full')
    # The maximum overlap integral over all lags, scaled by dx
    c5 = np.max(correlation) * dx
    return c5

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

    # Initialize swarm: 30 particles
    n_particles = 30
    particles = []
    velocities = []
    
    # First particle is the initial solution
    particles.append(h_start.copy())
    velocities.append(np.zeros(n_points))
    
    # Additional particles with small random perturbations
    for i in range(1, n_particles):
        perturbation = np.random.normal(0, 0.1, n_points) * h_start
        h_pert = h_start + perturbation
        h_pert = project_to_feasible(h_pert, n_points)
        particles.append(h_pert)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        
    particles = np.array(particles)
    velocities = np.array(velocities)
    
    # Track personal best
    pbest_positions = particles.copy()
    pbest_scores = np.array([evaluate_erdos_solution(p) for p in particles])
    
    # Track global best
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO parameters
    w = 0.9  # Inertia weight
    c1 = 2.0  # Cognitive acceleration
    c2 = 2.0  # Social acceleration
    max_vel = 0.5
    
    iteration = 0
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        
        # Evaluate current positions
        for i in range(n_particles):
            current_score = evaluate_erdos_solution(particles[i])
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest_positions[i] = particles[i].copy()
                
            if current_score < gbest_score:
                gbest_score = current_score
                gbest_position = particles[i].copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            
            # Velocity update
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest_positions[i] - particles[i]) +
                            c2 * r2 * (gbest_position - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)

        if iteration % 10 == 0:
