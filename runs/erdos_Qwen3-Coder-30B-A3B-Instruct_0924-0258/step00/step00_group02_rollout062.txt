import numpy as np
import time
from scipy import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute overlap integral for all shifts
    overlap = correlate(h, 1 - h, mode="full") * dx
    # Return maximum overlap
    return np.max(overlap)

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0<=h<=1 and sum(h)=n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * 0.5
    return h

def mutate_h(h, mutation_rate=0.1):
    """Mutate h values with Gaussian noise and project back."""
    n_points = len(h)
    h_new = h.copy()
    n_mutate = int(n_points * mutation_rate)
    indices = np.random.choice(n_points, size=n_mutate, replace=False)
    for i in indices:
        h_new[i] += 0.1 * np.random.randn()
    return project_to_feasible(h_new, n_points)

def crossover_h(h1, h2):
    """Single-point crossover between two step functions."""
    n_points = len(h1)
    point = np.random.randint(1, n_points)
    h_new = np.concatenate([h1[:point], h2[point:]])
    return project_to_feasible(h_new, n_points)

        population.append(h)

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    for gen in range(generations):

        # Evaluate population
        fitness = []

            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Check if we found a better solution
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_h = sorted_pop[0].copy()
        
        # Select elite
        elite = sorted_pop[:elite_size]
        
        # Generate offspring
        offspring = []
        for _ in range(population_size - elite_size):
            parent1, parent2 = np.random.choice(elite, size=2, replace=False)
            child = crossover_h(parent1, parent2)
            child = mutate_h(child, mutation_rate)
            offspring.append(child)
        
        # New population
        population = elite + offspring

    return best_h, best_c5
