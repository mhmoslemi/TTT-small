import time
from scipy.signal import correlate

def evaluate_C5(h):
    """Evaluate the C5 constant for a given h"""
    # Compute the full correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # Extract the valid correlation values (excluding padding)
    # This corresponds to all valid k values for the overlap
    n = len(h)
    # The valid part is from index n-1 to 2*n-2
    valid_corr = corr[n-1:2*n-1]
    # Convert to the actual integral value
    dx = 2.0 / n
    return np.max(valid_corr) * dx
