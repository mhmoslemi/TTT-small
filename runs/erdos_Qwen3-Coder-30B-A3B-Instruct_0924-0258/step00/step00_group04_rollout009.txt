import numpy as np
from scipy.optimize import minimize
import time
from copy import deepcopy

def evaluate_C5(h):
    """Evaluate the C5 constant for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the correlation (full mode)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (excluding trivial correlations)
    max_correlation = np.max(correlation[n_points-1:])
    return max_correlation * dx

def project_and_rescale(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def crossover(parent1, parent2, n_points):
    """Perform segment-wise crossover."""
    n_points = len(parent1)
    # Choose 2-3 random crossover points
    n_crossovers = np.random.randint(2, 4)
    crossover_points = sorted(np.random.choice(n_points, n_crossovers, replace=False))
    crossover_points = [0] + crossover_points + [n_points]
    
    child = np.zeros(n_points)
    for i in range(len(crossover_points)-1):
        start = crossover_points[i]
        end = crossover_points[i+1]
        if i % 2 == 0:
            child[start:end] = parent1[start:end]
        else:
            child[start:end] = parent2[start:end]
    
    return child

def mutate(h, n_points, rate=0.05):
    """Randomly mutate some elements."""
    if np.random.rand() < 0.1:
        rate = 0.1  # increase mutation rate if stuck
    indices = np.random.choice(n_points, int(rate * n_points), replace=False)
    h[indices] += np.random.normal(0, 0.1, len(indices))
    return project_and_rescale(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42  # number of points for discretization
    pop_size = 50  # population size
    max_generations = 200  # maximum number of generations
    stagnation_limit = 50  # generations without improvement
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_and_rescale(h0, n_points)
        population.append(h0.copy())
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = project_and_rescale(h, n_points)
            population.append(h)
    else:
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = project_and_rescale(h, n_points)
            population.append(h)

    # Evolutionary loop
    best_h = None
    best_C5 = float('inf')
    stagnant_generations = 0
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            C5 = evaluate_C5(h)
            fitness.append(C5)
            
        # Update best solution
        for i, C5 in enumerate(fitness):
            if C5 < best_C5:
                best_C5 = C5
                best_h = population[i].copy()
                stagnant_generations = 0
        stagnant_generations += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:pop_size // 5]
        
        # Create new population
        new_population = [population[i] for i in top_indices]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Select parents
            parent1, parent2 = np.random.choice(top_indices, 2, replace=False)
            parent1_h = population[parent1]
            parent2_h = population[parent2]
            
            child = crossover(parent1_h, parent2_h, n_points)
            child = mutate(child, n_points, 0.05)
            
            # Local optimization using gradient-based method
            if np.random.rand() < 0.5:
                def objective(h_flat):
                    h = h_flat.copy()
                    h = h.reshape(n_points)
                    h = project_and_rescale(h, n_points)
                    return evaluate_C5(h)
                
                def constraint_func(h_flat):
                    h = h_flat.copy()
                    h = h.reshape(n_points)
                    return np.sum(h) - n_points / 2  # Sum constraint
                
                # Random sample to get starting point
                x0 = child.copy()
                result = minimize(
                    objective,
                    x0,
                    method='L-BFGS-B',
                    bounds=[(0, 1)] * n_points,
                    constraints={'type': 'eq', 'fun': constraint_func},
                    options={'maxiter': 20}
                )
                
                child = project_and_rescale(result.x, n_points)
                
            new_population.append(child)
            
        population = new_population[:pop_size]
        
        if stagnant_generations > stagnation_limit:
            # Increase diversity with high mutation rate
            for i in range(len(population)):
                population[i] = mutate(population[i], n_points, 0.1)

    if best_h is None:
        # Fallback to best from the final population
        final_fitness = [evaluate_C5(h) for h in population]
        best_idx = np.argmin(final_fitness)
        best_h = population[best_idx]
        best_C5 = final_fitness[best_idx]
        
    return (best_h.tolist(), best_C5, n_points)
