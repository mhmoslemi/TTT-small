import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given step function h."""
    # Compute cross-correlation of h with (1-h) for all lags
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx to get actual integral values
    corr *= dx
    # Return the maximum value, which is C5
    return np.max(corr)

def rescale_h(h, target_sum):
    """Rescale h so that sum(h) = target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def project_h(h, dx, target_sum):
    """Ensure h satisfies constraints: 0 <= h <= 1, sum(h)*dx = 1."""
    # Clip h to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to meet the integral constraint
    h = rescale_h(h, target_sum)
    return h

    np.random.seed(seed)

        # Create stepped pattern with constant values
        mid = n_points // 2
        h0[:mid] = 0.5
        h0[mid:] = 0.5
        h0 = rescale_h(h0, n_points / 2)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # PSO parameters
    n_particles = 30
    inertia_w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 10000
    stagnation_limit = 100
    min_improvement = 1e-6
    
    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    
    for i in range(n_particles):
        # Initialize particle from h0 with small random perturbations
        h = h0.copy()
        if i > 0:
            # Add small noise to explore
            noise_level = 0.1 / n_points
            h += np.random.normal(0, noise_level, n_points)
            h = project_h(h, dx, target_sum)
        particles.append(h.copy())
        velocities.append(np.zeros_like(h))
        val = evaluate_h(h, dx)
        personal_best.append(h.copy())
        personal_best_vals.append(val)
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_vals)
    global_best = personal_best[global_best_idx].copy()
    global_best_val = personal_best_vals[global_best_idx]
    
    # PSO iteration
    iter_count = 0
    stagnant = 0
    old_global_best_val = global_best_val + 1.0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        iter_count += 1
        
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity

            cognitive = c1 * r1 * (personal_best[i] - h)
            social = c2 * r2 * (global_best - h)
            
            v = inertia_w * v + cognitive + social
            velocities[i] = v
            particles[i] = h + v
            
            # Apply bounds and rescale
            particles[i] = project_h(particles[i], dx, target_sum)
            h = particles[i]
            
            # Evaluate
            new_val = evaluate_h(h, dx)
            
            # Update personal best
            if new_val < personal_best_vals[i]:
                personal_best[i] = h.copy()
                personal_best_vals[i] = new_val

            stagnant += 1
        else:
            old_global_best_val = global_best_val
            stagnant = 0

    return (global_best.tolist(), global_best_val, n_points)
