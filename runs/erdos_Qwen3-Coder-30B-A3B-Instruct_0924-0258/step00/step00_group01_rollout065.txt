import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
        assert abs(np.sum(h) - n_points / 2.0) < 1e-6
    except:
        # Default construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2.0) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    alpha = 0.1
    no_improve_count = 0
    max_no_improve = 100
    
    while time.time() - start_time < budget_s - 1:
        # Compute overlaps for all lags
        # We can use the full correlation (mode='full') to get all overlaps
        h_extended = np.concatenate([h, h])
        corr = correlate(h_extended, 1 - h, mode='full')
        # Extract the relevant part: from lag -(n_points-1) to +(n_points-1)
        # But we only care about non-negative lags for the problem
        # The overlapping region is from index 0 to 2*n_points - 2
        # But we really want to extract the overlaps starting at lag 0 to 2*n_points-2
        # For case k in [0, n_points-1], overlap is at corr[n_points-1+k]
        # So we compute the maximum among corr[n_points-1:n_points-1+n_points]
        overlaps = corr[n_points-1:n_points-1+n_points] * dx
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # If no improvement in a while, decrease alpha
        if no_improve_count > max_no_improve:
            alpha *= 0.95
            no_improve_count = 0
            
        # Compute update gradient (based on large overlap direction)
        # We're looking for small h[i] where h[i+k] is large (or vice versa)
        # So gradient[i] should be proportional to (1 - h[(i+k_max)%n_points]) - h[i]
        gradient = np.zeros(n_points)
        for i in range(n_points):
            gradient[i] = (1 - h[(i + k_max) % n_points]) - h[i]
            
        # Update h
        h += alpha * gradient
        
        # Clamp to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Ensure satisfying the constraint sum(h) = n_points / 2
        current_sum = np.sum(h)
        required_sum = n_points / 2.0
        if abs(current_sum - required_sum) > 1e-6:
            # Rescale h to correct sum, but keeping the relative changes
            h = h * required_sum / current_sum
            
        # Check current C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Add a small random perturbation occasionally
        if np.random.rand() < 0.05:
            perturb = np.random.normal(0, 0.005, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            h = h * required_sum / current_sum
            
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    # Use a more direct and efficient way to compute C5
    
    # Correlate h with (1 - h) using full mode to get all lags
    h_extended = np.concatenate([h, h])
    one_minus_h = 1 - h
    one_minus_h_extended = np.concatenate([one_minus_h, one_minus_h])
    
    correlation = correlate(h_extended, one_minus_h_extended, mode='full')
    # Extract overlaps: overlaps at lag k corresponds to downsampled correlation
    overlaps = correlation[len(h)-1:len(h)-1+len(h)] * dx
    
    return np.max(overlaps)
