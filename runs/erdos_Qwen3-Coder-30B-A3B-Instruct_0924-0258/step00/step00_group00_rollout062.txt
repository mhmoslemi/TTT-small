import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points)
    
    # Moving average of last 20 c5 values to detect stagnation
    recent_c5 = []
    perturb_steps = 20
    step_count = 0
    
    while time.time() - start_time < budget_s - 1:
        step_count += 1
        # Compute all cross-correlations
        # h convolved with (1-h), with mode='full'
        corr = signal.correlate(h, 1 - h, mode='full')
        # Correlation lengths
        corr_len = len(corr)
        k_indices = np.arange(-n_points + 1, n_points)
        # Select the maximum
        max_k_idx = np.argmax(corr)
        max_k = k_indices[max_k_idx]
        max_corr = corr[max_k_idx]
        
        # Create gradient: for correlation at lag k, the gradient of h[i] with respect to 
        # that correlation is (1 - h[i + k]), if i + k < n_points, and 0 otherwise.
        grad = np.zeros(n_points)
        if max_k >= 0:
            grad[max_k:] = 1 - h[:-max_k] if max_k > 0 else 1 - h
        else:
            grad[:max_k] = 1 - h[-max_k:] if max_k < 0 else 1 - h
        
        # Line search over alpha
        alpha_candidates = [0.1, 0.5, 1.0]
        best_alpha = 0.0
        best_h_new = h.copy()
        best_new_c5 = best_c5
        
        for alpha in alpha_candidates:
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            # Rescale so that sum is correct
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_c5 = evaluate_c5(h_new, n_points)
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_alpha = alpha
                best_h_new = h_new.copy()
        
        # If no improvement, perturb
        if best_new_c5 >= best_c5:
            # Add small random noise to h
            h_noisy = h + 0.01 * np.random.randn(n_points)
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = h_noisy * (n_points / 2) / np.sum(h_noisy)
            new_c5 = evaluate_c5(h_noisy, n_points)
            if new_c5 < best_c5:
                best_new_c5 = new_c5
                best_h_new = h_noisy.copy()
                best_alpha = 0.0
        
        # Update h if improvement
        if best_new_c5 < best_c5:
            h = best_h_new.copy()
            best_c5 = best_new_c5
            if best_c5 < evaluate_c5(best_h, n_points):
                best_h = h.copy()
            recent_c5 = []
        else:
            recent_c5.append(best_c5)
            if len(recent_c5) > perturb_steps:
                recent_c5.pop(0)
                # If no improvement in recent steps, perturb
                if len(recent_c5) > 1 and all(c >= recent_c5[0] for c in recent_c5):
                    h = h + 0.01 * np.random.randn(n_points)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    best_c5 = evaluate_c5(h, n_points)
                    recent_c5 = []
    
    return (best_h, best_c5, n_points)

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx
