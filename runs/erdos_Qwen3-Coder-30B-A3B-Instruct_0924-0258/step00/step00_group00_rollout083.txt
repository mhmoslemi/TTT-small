import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 45
    dx = 2.0 / n_points
    total_sum = n_points / 2.0  # Since sum(h) * dx = 1
    
    # Get initial h values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * total_sum / current_sum
    else:
        # Create a simple initial construction
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * total_sum / current_sum
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    # Frank-Wolfe algorithm
    max_iterations = 500
    tolerance = 1e-4
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all cross-correlations to find maximum
        corr_full = correlate(h, 1 - h, mode="full")
        # Extract the relevant part
        corr_max = corr_full[len(corr_full)//2:]
        max_lag_idx = np.argmax(corr_max)
        max_lag = max_lag_idx
        
        # Compute gradient of the maximum cross-correlation
        # This is the derivative of sum(h[i] * (1 - h[i + max_lag]))
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            if i + max_lag < len(h):
                gradient[i] += (1 - h[i + max_lag])
        
        # Determine the direction: select S indices with highest gradient
        # Set S = n_points/2 to maintain density
        S = int(n_points / 2)
        sorted_indices = np.argsort(gradient)[::-1][:S]
        
        # The Frank-Wolfe step: solve the subproblem
        # Minimize <gradient, v - h> s.t. v in feasible set
        # Simplified line search approach: try several stepsizes
        
        best_step = 0.0
        best_new_c5 = best_c5
        
        step_sizes = [0.1, 0.25, 0.5, 0.75, 1.0]
        
        for gamma in step_sizes:
            # Proposal direction: v = h + gamma * (direction - h)
            # direction = 1 if index in sorted_indices else 0
            direction = np.zeros_like(h)
            direction[sorted_indices] = 1.0
            direction = direction - h  # constraint: should be 'move towards feasible'
            
            # Generate trial point
            trial_h = h + gamma * direction
            # Ensure feasibility (clip to [0,1])
            trial_h = np.clip(trial_h, 0, 1)
            # Adjust magnitude to satisfy sum constraint
            actual_sum = np.sum(trial_h)
            if actual_sum > 0:
                trial_h = trial_h * total_sum / actual_sum
            
            new_c5 = compute_c5(trial_h, dx)
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_step = gamma
        
        if best_step > 0:
            direction = np.zeros_like(h)
            direction[sorted_indices] = 1.0
            direction = direction - h
            h = h + best_step * direction
            h = np.clip(h, 0, 1)
            actual_sum = np.sum(h)
            if actual_sum > 0:
                h = h * total_sum / actual_sum
                
            if best_new_c5 < best_c5:
                best_c5 = best_new_c5
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    """Computes the overlap integral C5 for given h"""
    corr_full = correlate(h, 1 - h, mode="full")
    corr_max = corr_full[len(corr_full)//2:]
    return np.max(corr_max) * dx
