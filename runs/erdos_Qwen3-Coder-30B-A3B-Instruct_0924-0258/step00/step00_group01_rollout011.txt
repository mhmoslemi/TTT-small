import numpy as np
import time
from scipy.optimize import minimize

    np.random.seed(seed)
    
    # Initialize parameters

    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization variables
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Project h to [0,1] and normalize sum
    def project_h(h_local):
        h_local = np.clip(h_local, 0, 1)
        h_local = h_local * (n_points / 2) / np.sum(h_local)
        return h_local
    
    # Compute C5
    def compute_c5(h_local):
        corr = np.correlate(h_local, 1 - h_local, mode='full')
        return np.max(corr) * dx
    
    # Gradient descent
    alpha = 0.1
    momentum = 0.9
    velocity = np.zeros(n_points)
    last_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute current C5 and correlation
        corr = np.correlate(h, 1 - h, mode='full')
        k0 = np.argmax(corr)
        max_corr = corr[k0]
        
        # Compute gradient w.r.t. active shift (k0)
        grad = np.zeros(n_points)
        if k0 < n_points:
            # h[i] * (1 - h[i + k0])
            for i in range(n_points):
                if i + k0 < n_points:
                    grad[i] += 1 - h[i + k0]
        else:
            # h[i - k0] * (1 - h[i])
            k_offset = k0 - n_points + 1
            for i in range(n_points):
                if i >= k_offset:
                    grad[i] -= h[i - k_offset]
        
        # Update with momentum and step
        velocity = momentum * velocity - alpha * grad
        h_new = h + velocity
        
        # Project and compute new C5
        h_new = project_h(h_new)
        new_c5 = compute_c5(h_new)
        
        # Accept or reject
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            last_improvement = iteration
            
        h = h_new
        
        # Adapt learning rate
        if iteration - last_improvement > 50:
            alpha *= 0.95
            last_improvement = iteration
            
        iteration += 1
        
        # Occasionally restart randomly
        if np.random.rand() < 0.05 and iteration % 50 == 0:
            h = project_h(np.random.rand(n_points))
    
    return (best_h.tolist(), best_c5, n_points)
