def evaluate(h, dx):
    # Compute cross-correlation terms
    y = correlate(h, 1 - h, mode="full")
    # Only consider the valid overlaps (not the edge cases)
    # It is not symmetric from both sides for all lags
    # For the given setup (h(x) on [0,2] with n points, dx=2/n)
    # We compute it with standard definition of convolution with appropriate scaling
    # Max of all valid correlations gives C_5
    y = y * dx
    # We only consider the relevant lags
    # For x in [0, 2] we consider lags from -1 to 1, i.e., 0 to 2n-2 indices
    # full_corr_len = 2 * len(h) - 1
    return np.max(y)

def rescale(h, target_sum):
    # Rescale h to have sum target_sum
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.ones_like(h) * (target_sum / len(h))
    return h * (target_sum / current_sum)

def mutate(h, rate=0.1, delta=0.1):
    # Mutate some elements of h
    mask = np.random.random(len(h)) < rate
    h = h.copy()
    h[mask] += np.random.uniform(-delta, delta, size=np.sum(mask))
    h = np.clip(h, 0, 1)
    return rescale(h, np.sum(h))

def crossover(p1, p2):
    # Single-point crossover
    point = np.random.randint(1, len(p1))
    child1 = np.concatenate([p1[:point], p2[point:]])
    child2 = np.concatenate([p2[:point], p1[point:]])
    child1 = rescale(child1, np.sum(p1))
    child2 = rescale(child2, np.sum(p1))
    return child1, child2

def tournament_selection(population, fitness, k=3):
    # Select one individual using tournament selection
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    max_generations = 200
    population_size = 50
    
    # Use initial_h_values if available
    try:
        h = initial_h_values()
        if np.isclose(np.sum(h), n_points / 2, rtol=1e-6):
            parent = h
        else:
            parent = rescale(h, n_points / 2)
    except:
        parent = np.ones(n_points) * 0.5
        parent = rescale(parent, n_points / 2)
    
    # Create initial population
    population = [parent.copy()]
    for _ in range(population_size - 1):
        # Largely random solutions with desired sum
        new_h = np.random.random(n_points)
        # Re-scale to desired sum
        new_h = rescale(new_h, n_points / 2)
        # Add some noise
        new_h = mutate(new_h, rate=0.2, delta=0.2)
        population.append(new_h)
    
    best_h = parent.copy()
    best_c5 = evaluate(best_h, dx)
    start_time = time.time()
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            c5 = evaluate(h, dx)
            fitness.append(c5)
        
        # Select parents and evolve
        new_population = []
        elites = np.argsort(fitness)[:1]  # keep best one
        for idx in elites:
            new_population.append(population[idx].copy())
            
        for _ in range(population_size - 1):
            p1 = tournament_selection(population, fitness)
            p2 = tournament_selection(population, fitness)
            
            if np.random.random() < 0.7:  # crossover rate
                c1, c2 = crossover(p1, p2)
                new_population.append(c1)
                if len(new_population) < population_size:
                    new_population.append(c2)
            else:
                # Just mutate one parent
                new_h = mutate(p1.copy(), rate=0.1, delta=0.1)
                new_population.append(new_h)
                
        # Keep population size consistent
        population = new_population[:population_size]
        
        # Update best solution
        current_fitness = [evaluate(h, dx) for h in population]
        min_idx = np.argmin(current_fitness)
        if current_fitness[min_idx] < best_c5:
            best_c5 = current_fitness[min_idx]
            best_h = population[min_idx].copy()
            
    # Final evaluation of best solution
    final_c5 = evaluate(best_h, dx)
    return (best_h, final_c5, n_points)
