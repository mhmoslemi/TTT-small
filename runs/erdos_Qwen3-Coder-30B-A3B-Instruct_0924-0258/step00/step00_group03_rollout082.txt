import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all lags
    corr = correlate(h, 1 - h, mode='full')
    # The maximum value gives us C5
    c5 = np.max(corr) * dx
    return c5

def normalize_h(h):
    """Ensure h sums to n/2 and stays in [0,1]."""
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Renormalize to ensure sum is exactly target
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, mutation_rate=0.1):
    """Mutate a step function with bounded noise."""
    n = len(h)
    mutation = np.random.normal(0, mutation_rate * 0.1, n)
    h_mut = h + mutation
    return normalize_h(h_mut)

def crossover_h(h1, h2):
    """Blend two step functions."""
    n = len(h1)
    mask = np.random.rand(n) > 0.5
    h_child = np.where(mask, h1, h2)
    return normalize_h(h_child)

def select_tournament(population, fitnesses, tournament_size=5):
    """Select an individual via tournament selection."""
    idx = np.random.choice(len(population), tournament_size)
    best_idx = idx[np.argmin([fitnesses[i] for i in idx])]
    return population[best_idx]
