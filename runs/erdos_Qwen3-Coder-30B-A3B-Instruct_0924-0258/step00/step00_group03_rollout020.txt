import numpy as np
import time
from scipy import signal
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    lr = 0.1
    decay_rate = 0.999
    min_lr = 1e-6
    early_stop_patience = 100
    best_c5 = float('inf')
    best_h = None

    # Get initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h[0] = 0.1
        h[-1] = 0.1
        h = h / np.sum(h) * (n_points / 2)

    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    def compute_c5(h_val):
        # Compute overlap integral
        convolved = signal.correlate(h_val, 1 - h_val, mode='full')
        # Shift and take maximum
        max_overlap = np.max(convolved)
        return max_overlap * dx

    # Optimizer
    patience = 0
    current_lr = lr
    last_c5 = compute_c5(h)

    for i in range(max_iter):
        if time_left() < 1e-3:
            break
            
        # Compute gradient direction using finite differences (simple approach)
        eps = 1e-4
        grad = np.zeros_like(h)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus / np.sum(h_plus) * (n_points / 2)
            c5_plus = compute_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus / np.sum(h_minus) * (n_points / 2)
            c5_minus = compute_c5(h_minus)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)

        # Gradient descent step
        h_new = h - current_lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Check if we improved
        new_c5 = compute_c5(h_new)
        
        if new_c5 < last_c5:
            h = h_new
            last_c5 = new_c5
            patience = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            patience += 1
            if patience > early_stop_patience:
                patience = 0
                current_lr *= 0.5
                if current_lr < min_lr:
                    break
        
        # Adaptive learning rate decay
        current_lr *= decay_rate
        if current_lr < min_lr:
            current_lr = min_lr

    if best_h is None:
        best_h = h
    if best_c5 == float('inf'):
        best_c5 = compute_c5(best_h)

    return (best_h.tolist(), best_c5, n_points)
