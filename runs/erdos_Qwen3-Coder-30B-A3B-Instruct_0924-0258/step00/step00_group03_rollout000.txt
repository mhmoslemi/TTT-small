import numpy as np
from scipy import signal
import cvxpy as cp
import time

def evaluate_h(h, dx):
    # Compute the overlap integral C5
    conv = signal.correlate(h, 1 - h, mode="full")
    max_conv = np.max(conv) * dx
    return max_conv

def project_onto_feasible(h, n_points, dx):
    # Project h onto the feasible set: 0 <= h[i] <= 1 and sum(h) * dx = 1
    # Using CVXPY for the projection
    h_var = cp.Variable(n_points)
    constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]
    prob = cp.Problem(cp.Minimize(cp.sum_squares(h_var - h)), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    return np.array(h_var.value)

def compute_gradient(h, k0, n_points, dx):
    # For convolution at lag k0, compute gradient w.r.t. h
    # We only need gradient w.r.t. the components that affect the maximum
    grad = np.zeros(n_points)
    # Error in the last version: convolution has mode="full", so we compute 
    # d/dh_i of the sum h_j * (1 - h_{j+k0}) over all j
    # If we're considering the convolution at k0 position,
    # the terms affected are those where h_j contributes to h_{j+k0} * (1 - h_j)
    # But let's compute it correctly based on how convolution works.
    # conv[i] = sum_{j=0}^{n_points-1} h[j] * (1 - h[i-j]) = sum_{j=0}^{n_points-1} h[j] - sum_{j=0}^{n_points-1} h[j] h[i-j]
    # Reverse the second sum, so h[i-j] part, if we fix i we get indices j where i-j >= 0 and i-j < n_points.
    if k0 >= 0:
        # conv[i] = sum_{j max(0, i-n_points+1)}^{min(i, n_points-1)} h[j] * (1 - h[i-j])
        # We're here to compute d/dh_k0 of conv[k0], so that's 1 - h[0]
        # Actually, let's fix what we want to differentiate!
        # conv_k = sum_{j=0}^{n_points-1} h[j] (1 - h[k-j]) if k-j >= 0 and k-j < n_points
        # We're differentiating the full convolution, so we take gradients for all h_j terms that influence it.
        # Let's go simpler. Gradient of conv[k0] w.r.t. h[i] is:
        # For a fixed k0 (valid due to the full mode), for each j, the cross-term is d/dh_i [h[j] (1 - h[k0-j])] 
        # If i == j: we get -(1 - h[k0-j])
        # If i == k0-j: we get h[j]
        # Else 0.
        # If k0 < 0, then k0 = -(k0_neg), we have j-k0 = j+k0_neg >= 0 and these terms...
        # Actually, h is of length n_points, and conv returned by correlate has 2*n_points-1 elements
        # If h is 0, size n_points, then conv has 2*n_points-1 elements, conv[i] contains elements for lag i-(n_points-1) = i - n_points + 1
        # The actual lag is k = i - n_points + 1, so i = k + n_points - 1
        # We want k = k0 (the lag that gives max), so i = k0 + n_points - 1
        # It's simpler to directly compute the derivative of that term
        # If we're computing d/dh[l] conv[k0], where k0 is the maximum, we get:
        # conv[k0] = sum_{j} h[j] (1 - h[k0-j]) if k0-j >= 0 and k0-j < n_points
        # So d/dh[l] conv[k0] = sum_{j s.t. k0=j-l} (1 - h[k0-j])  
        # It's just (1 - h[k0-l]) if l < n_points and k0 - l >= 0 and k0-l < n_points, and 0 otherwise?
        # No, it's h[j] for j such that l = k0 - j, so i = k0 + j
        # Wait: So it's
        # d/dh[l] conv[k0] = 
        # if l >= 0 and k0-l >= 0 and k0-l < n_points then 1 - h[k0-l]  
        # This j goes from 0 to n_points-1, and we need k0 - j = l => j = k0-l, so we get the contribution there.
        # For j such that k0 - j = l => l = k0 - j
        # So if that makes sense, then the contribution to the gradient is from the terms where:
        # l = k0 - j, and the term is (1 - h[k0-j])  
        # So d/dh[l] conv[k0] = (1 - h[k0-l]) if l = k0 - j for some j, and 0 otherwise.
        # But also the h[j] term has non-zero derivative if j = l.
        # Let's (re)compute:
        # d/dh[l] conv[k0] = d/dh[l] sum_{j=0..n_points-1} h[j] (1 - h[k0-j])
        # = sum_{j=0..n_points-1} d/dh[l] [h[j] (1 - h[k0-j])]
        # = sum_{j=0..n_points-1} [delta_{j,l} (1 - h[k0-j]) + h[j] d/dh[l] (1 - h[k0-j]) ]
        # = sum_{j=0..n_points-1} [delta_{j,l} (1 - h[k0-j]) - h[j] delta_{k0-j,l} ]
        # = (1 - h[k0-l]) - h[l] * delta_{k0-l,l} where delta_{a,b} is 1 if a=b else 0.
        # = 1 - h[k0-l] - h[l] delta_{k0-l,l} if k0-l is in [0, n_points-1]
        # is it simpler if we just recompute? look up the physical meaning.
        # conv[k0] = sum_{j} h[j] (1 - h[k0-j]) 
        # So d/dh[l] conv[k0] = (1 - h[k0-l]) - h[l] (if k0-l >= 0 and k0-l < n_points) 
        # If l == k0 - j, then h[j] as (1 - h[k0 - (k0-j)]) = (1 - h[j]) 
        # If j == k0 - l, Then h[k0-l] appears in the sum.
        # Let's directly sum the indices j that have h[j] in the contrib. for d/dh[l]conv[k0]; 
        # They satisfy: k0-j = l  => j = k0-l  
        # So the derivative of h[j] in conv[k0] when j = k0-l is h[j] d/dh[l] (1 - h[k0-j]) = h[j] (if l==k0-j) = h[k0-l] 
        # So we have:
        # d/dh[l] conv[k0] = (1 - h[k0-l]) (if k0-l is valid)  - h[l] (if l AND k0-l are valid)
        # = (1 - h[k0-l]) if k0-l >= 0 and k0-l < n_points  
        # = - h[l] if l >= 0 and l < n_points 
        # But that was with strict j for (k0-j) is valid. 
        # j ranges from 0 to n_points - 1. So we sum over j: j * (1 - h[k0-j]) and d/dh[l] is (1-h[k0-l]) if l = k0-j, and -h[l] if l = j 
        # Fixed j, k0-j = l => j = k0-l
        # So that's just delta_j^{k0-l} (1 - h[l]) + delta_{k0-j}^{l} (- h[l])
        # Hence: we really want the fixed equation:  
        # = delta_{j,k0-l} (1-h[l]) + delta_{j,l} (-h[l]) 
        # = (1-h[l])  if j = k0-l 
        #  + (0 if j  != k0-l but j = l) == -h[l] if l = k0-l = j and l in [0,n_points-1] 
        # But the index j varies in the sum
        # Update. 
        # The formal derivative is:
        # d/dx[l] sum_{j}  h[j] (1 - h[k0-j])
        # = sum_j [delta_{j,l}(1 - h[k0-j]) + h[j] * delta_{k0-j,l} * (-1)] 
        # = (1 - h[k0-l]) + h[l] * delta_{k0-l,l}
        # If l is in range and k0-l is in range, then it's (1 - h[k0-l]) - h[l]
        # If k0-l is invalid then we ignore the first term as it doesn't contribute to the sum over j.
        # So: d/dh[l] = -h[l] + (1 - h[k0-l]) if 0 <= k0-l < n_points. Removed one term (1 - h[k0-l]) if the 0 <= l < n_points.
        # So the gradient is composed of terms where (1 - h[k0-l]) + ( - h[l] ) but handled properly.
        # The most direct approach for the gradient is:
        # For i such that (0 <= i < n_points) and (0 <= k0 - i < n_points), we get contribution to the gradient:
        # d/dh[l] = (1 - h[k0-l]) if l = (k0-i) => l = k0-i
        # That is: d/dh[ k0-i ] = (1 - h[i])
        # Let's instead index over the correct sum directly:
        # We know j and summing over j we get:
        # For all valid j with l = k0 - j, then we contribute (1-h[l]) to d/dh[j]
        # And for the part of h[j] contributing to (1-h[k0-j]), we also contribute -h[j] to the gradient of that as well.
        # In other words:  
        # Over j = 0 to n_points-1: 
        # say S = set of valid j such that k0-j is also in [0, n_points)
        # Sum_{j in S} [ (1 - h[k0-j]) d/dh[j] h[j] + h[j] d/dh[j] (1 - h[k0-j]) ]  
        #  = sum_{j in S} [ (1 - h[k0-j]) (1) + h[j] (0) ]  
        # = sum_{j in S} (1 - h[k0-j])  
        # But now if we compute d/dh[l] where l is correct:
        # Fix l, what contribution does it get?
        # If l = k0-j for some j that's valid in [0, n_points), then
        # there's a term (1-h[k0-l]) in the derivative calculation.
        # If we're looking at d/dh[l] sum_{j} h[j] (1-h[k0-j]) 
        # = sum_{j} (1 - h[k0-j]) d/dh_l h[j] + h[j] d/dh_l(1 - h[k0-j]) 
        # = sum_{j} (1 - h[k0-j]) delta_{lj} + h[j] delta_{l, k0-j} (-1) 
        # = (1 - h[k0-l]) + (-h[l]) where that makes sense.
        # So:
        # Gradient is:
        # For valid k0-l >= 0 and k0-l < n_points:
        # grad[l] += 1 - h[k0-l]  (contributes from delta_{lj} with (1-h[k0-j]) where l = k0-j)
        # For valid l >= 0 and l < n_points:
        # grad[l] -= h[l]  (contributes from d/dh_l (1-h[k0-j]) if l = k0-j)
        grad = np.copy(h)
        # But get this:
        # d/dh[l] = (1 - h[k0-l})        if k0-l is valid index.
        #         - h[l]                 if l is valid index
        # Two such terms! Let's compute all cases properly.
        # Make it clearly:
        # In rolling convolution version, the resulting conv[i] has terms sum_{j=0}^{n_points-1} h[j](1 - h[i-j]) with i-j >= 0 and i-j < n_points.
        # These are integers ranging from 0 to 2* n_points - 2  
        # But in terms of how we index, it gets tricky.
        # Let's make a proper discretization approach. Some calls change different time points.  
        # Use:
        # for j in range(0, n_points):
        #   i = k0 + n_points - 1 (reverse index mapping from lag)
        #   So d/dh[l] conv[i] = (1-h[k0+ n_points - 1 - l])     (term from h[j] dx = l)  
        #    - h[l]                              (term from dh[k0-j], delta function)
        #   but l=j
        #    So d/dh[j] conv[k0 + n_points - 1] = (1 - h[k0 - j)] - h[j]
        #    So we take that index and twist it:
        #    d/dh[l] conv[k0 + n_points - 1] = (1 - h[k0 - l]) - h[l] (if k0-l, l valid)
        # But there's extra care needed if l or k0-l is out of range.
        # So the gradient, for position k0+ n_points -1 in the convolution result, is:   
        # d/dh[l] = if 0 <= l < n_points and 0 <= k0-l < n_points:
        #        = (1 - h[k0-l])  - h[l]
        # Else 0.
        # Now for k0 + n_points - 1 (converted back)
        # We compute d/dh[l] (conv[ k0 + n_points - 1 ] )
        # = (1-h[k0 - l]) - h[l]  
        # When both indices are valid.
        # For fixed l, we check where conv[ k0 + n_points - 1 ] contributes.
        if 0 <= k0 < 2 * n_points - 1:  # The Lag is reasonable  
            # The convolution index is k0 = k + n_points - 1 where k was lag
            # Then d/dh[l] = (1-h[k0-l]) - h[l] where the full condition is,
            # 0 <= l < n_points, 0 <= k0-l < n_points, i.e., l <= k0 and k0-l < n_points -> l > k0 - n_points.
            # That is, l >= max(0, k0 - n_points + 1) and l < min(n_points, k0 + 1)
            # We iterate over l to add the contribution.
            # But more directly: to compute one such gradient, 
            # d/dh[l]: 
            # 0 <= l < n_points 
            # 0 <= k0 - l < n_points  <=>  k0 - n_points < -l <= 0 =>  l <= k0 and l <= n_points
            grad[:] = 0
            for l in range(n_points):
                if 0 <= k0-l < n_points:
                    grad[l] = 1 - h[k0-l] - h[l]
                else:
                    grad[l] = -h[l]
        else:
            # Invalid lag; just regular "-h"
            grad = -h

    else:
        # k0 negative. Negative lag -> i-j < 0, i < j for h[j] and (1 - h[i-j]) 
        # Actually construct conv with displacement correctly
        # This is hard to make correct with full convolution.
        # But the most correct path early: 
        # Let us recalculate with logic from numpy convolution.
        # conv[k0 + n_points - 1] = sum_{j=0}^{n_points-1} h[j] (1 - h[k0 - j])
        # To take derivative:
        # d/dh[l] conv[k0 + n_points - 1] = 
        # = sum_{j=0}^{n_points-1} d/dh[l] (h[j] (1 - h[k0 - j]))
        # = sum_{j=0}^{n_points-1} ( delta_{j,l}(1-h[k0-j]) + h[j] d/dh[l] (1 - h[k0-j]) )
        # = sum_{j=0}^{n_points-1} [ delta_{j,l}(1-h[k0-j]) - h[j] *(1) * delta_{l, k0-j} ]
        # So where j == l:
        #   remaining is (1 - h[k0-l])
        # And where k0 - j == l => j = k0 - l:
        #   we get -h[k0-l] 
        # So contribution from term j: for j==l gets (1 - h[k0-l]), for j == k0-l it gets -h[l]
        # elem = 1 - h[k0-l],  -h[l] 
        # Correct version of derivative for such conv:
        # it's a tricky matrix. 
        # But simpler in algorithmic approach. Let's assume we know 
        # our max k0 term in convolution. 
        # Just recompute index cognitive error.
        # In part, no need to derive fully.
        grad = np.copy(h)
        # Loop on indices l for which we sum h[j] (1-h[k0-j]) over j  
        # For derivative w.r.t. l, we get contribution + (1 - h[k0-l])  
        # if l is normal index AND k0-l is valid index  
        # And - h[l] if l is valid (also contributed term from h[j] coefficient in front)
        # So the gradient computation for a fixed k0:
        # For all possible l in [0, n_points)  which give valid factor in the convolution sum
        # Contribution to d/dh[l] is     if k0-l is valid: (1 - h[k0-l])  
        #                              else: 0
        # Additionally, we get -h[l] added to that too
        # Since for j=l, we do get h[l] * derivative of 1-h[k0-l] = -1 , so we get -h[l].
        # And for j=k0-l, we get the derivative of h[l] (1-h[k0-j]), = h[j] * derivative w.r.t. h[l] of 1-h[k0-j]
        # If j=k0-l, and both l and k0-l are valid:  h[k0-l] * (-1) * delta_{l, k0-l} → -h[l]
        # So gradient for d/dh[l] = (1 - h[k0-l]) - h[l]
        # if 0 <= k0-l < n_points  then add condition.
        grad[:] = 0
        for l in range(n_points):
            # not to do in support we should check both conditions:
            if 0 <= k0 - l < n_points:
                grad[l] = (1 - h[k0-l]) - h[l] 
            else:
                grad[l] = -h[l]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    start_time = time.time()
    
    n_points = len(initial_h_values) 
    dx = 2.0 / n_points
    h = initial_h_values.copy() 
    h = h / np.sum(h) * (n_points / 2)

    # Grad-based approach to minimize max overlap.
    # Initialize
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    t0 = time.time()
    iter_count = 0
    learning_rate = 0.01
    step_size = 0.001
    # With time halved to allow for safety
    # Initial estimate 1000 steps with gradient descent and optimization
    # Subtract time just used to initialize
    
    num_iterations = 10000
    
    # Within n_points = 70 
    # Assume std work is approx 20 ms / iteration, with some features 
    # Budget = 600 seconds, so ~30k iterations are possible if good 
    # So let's scale to a few thousand to a few thousand  
    # Intuitively 5k to 10k iterations may be sensible to beat a state.

    for i in range(num_iterations):
        if time.time() - start_time > budget_s - 0.1:
            break

        # Evaluate current
        current_c5 = evaluate_h(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # evaluate C = max convolution (its maximum) and its index
        conv = signal.correlate(h, 1 - h, mode="full")
        #conv = np.convolve(h, 1 - h, mode="full")
        k0 = np.argmax(conv)
        # A bit unrelated but for Integral consistency
        # compute gradient of this element in convolution
        grad = compute_gradient(h, k0, n_points, dx)
        
        # Update heuristic: 
        h_new = h - learning_rate * grad
        
        # Project to feasible
        h = project_onto_feasible(h_new, n_points, dx)

        iter_count += 1
        # Dynamic step size decay
        if i > 0 and i % 100 == 0:
            learning_rate *= 0.95 
            if learning_rate < 1e-5:
                learning_rate = 1e-5 
                
    # Final check
    final_c5 = evaluate_h(best_h, dx)
    # From optimization structure, ready state with convergence check
        
    return (best_h, final_c5, n_points)
