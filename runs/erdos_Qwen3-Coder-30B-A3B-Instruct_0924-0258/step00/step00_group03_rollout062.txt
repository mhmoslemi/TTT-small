import numpy as np

def evaluate_erdos_solution(h, n_points):
    """Evaluate the Erdős overlap constant for given h."""
    # Ensure h is properly normalized
    h = np.array(h)
    dx = 2.0 / n_points
    # Compute the correlation-like integral for all shifts
    corr = np.correlate(h, 1 - h, mode='full')
    # Only consider valid indices (those where both h[i] and h[i+k] are defined)
    n = len(h)
    max_k = 2 * n - 2
    # We want to maximize over all k in range (-n+1, n-1) 
    # But array indexing from 0, so the overlap is computed for shifts 0 to 2*n-2
    # The relevant part from the entire convolution for positive shifts
    if len(corr) == 2 * n - 1:
        # Properly extract the relevant values for k in [-n+1, n-1] 
        # The center element corr[n-1] corresponds to shift 0
        # corr[i] corresponds to shift (i - (n-1))
        indices = range(len(corr))
        shifts = [i - (n-1) for i in indices]
        # Only positive shifts
        k_shifts = [s for s in shifts if s > 0]
        # corresponding correlations
        correlation_values = [corr[i] for i in indices if shifts[i] > 0]
        # Same for negative
        k_shifts_neg = [s for s in shifts if s < 0]
        correlation_values_neg = [corr[i] for i in indices if shifts[i] < 0]
        # So for k > 0, we evaluate corr[i] = shift k = i - (n-1),
        #   i.e., corr[0..n-2] = shift k = -n+1..-1, etc.
        # corr[n-1+i] = shift i for i=0..n-1
        # We want max over all valid k 
        # from -n+1 to n-1, excluding 0.
        # For positive k (k = 1..n-1), the corresponding values are corr[n-1+k]
        # For negative k (k = -1..-n+1), the corresponding values are corr[n-1+k]
        # But this isn't directly correct indexing for the positions in the vector.
        # The correct evaluation:
        # At shift k, for k=1..n-1, we sum h[i] * (1 - h[i+k]) for i from 0 to n-k-1
        # In correlate, the correlation at shift -k is at index n-1-k (if k > 0)
        max_val = 0
        for k in range(1, n):
            # shift = k corresponds to index at n-1+k (shift k)
            offset = n - 1 + k
            if offset < 2*n-1:
                corr_sum = corr[offset]
                # where k = 1..n-1 means this is our positive shifts with move right
                val = corr_sum * dx
                max_val = max(max_val, val)
        for k in range(1, n):
            # shift = -k corresponds to index at n-1-k (if k > 0)
            offset = n - 1 - k
            if offset >= 0:
                corr_sum = corr[offset]
                # where k = -1..-(n-1) means this is our negative shifts, move left
                val = corr_sum * dx
                max_val = max(max_val, val)
        return max_val
    else:
        # fallback possible
        return float('inf')

def project(h):
    h = np.array(h)
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total > 0:
        h = h * (len(h) / 2) / total
    return h.tolist()

def mutate(h, sigma):
    h = np.array(h)
    h = h + np.random.normal(0, sigma, len(h))
    return project(h.tolist())

def crossover(h1, h2, alpha=0.6):
    h1 = np.array(h1)
    h2 = np.array(h2)
    # blend with alpha in [0.5, 1.0], default 0.6 for diversity
    beta = np.random.uniform(alpha, 1 - alpha)
    h = beta * h1 + (1 - beta) * h2
    return project(h.tolist())

    # But this includes cancellations, because shift k involves sum_i of h[i] * (1-h[i+k])

    # d/dh[i] [h[i] * (1 - h[i+1])] = (1 - h[i+1]) - h[i] * (0) = (1 - h[i+1])
    # If we do full equation, it’s sum over k of topline ..., but let's just
    # heuristic approx using finite difference + projection step
