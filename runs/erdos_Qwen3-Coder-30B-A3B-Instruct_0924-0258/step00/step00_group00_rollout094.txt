import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Normalize to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Precompute dx
    dx = 2.0 / n_points
    
    # Maintain best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Track time
    start_time = time.time()
    
    # Iteration variables
    iterations_no_improvement = 0
    max_no_improvement = 20
    perturbation_magnitude = 0.01
    
    # Main loop
    while time.time() - start_time < budget_s:
        # Compute all cross-correlations
        full_corr = correlate(h, 1 - h, mode='full')
        cross_correlations = []
        for k in range(2 * n_points - 1):
            cross_correlations.append(full_corr[k] * dx)
        
        # Find maximum value and corresponding indices
        max_c5 = max(cross_correlations)
        max_indices = [k for k in range(len(cross_correlations)) if abs(cross_correlations[k] - max_c5) < 1e-10]
        
        # If we've hit our target, finish
        if max_c5 <= 0.3807:
            break
            
        if max_indices:
            # Compute gradient for the current h
            gradient = np.zeros_like(h)
            for k in max_indices:
                for i in range(n_points - k):
                    if i < len(gradient):
                        gradient[i] += (1 - h[i + k])
                    if i + k < len(gradient):
                        gradient[i + k] -= h[i]
            
            # Line search: try alpha values
            alphas = [0.1, 0.25, 0.5]
            best_alpha = 0.0
            best_new_h = h.copy()
            best_new_c5 = max_c5
            
            for alpha in alphas:
                # Propose new h
                new_h = h - alpha * gradient
                #Clip to [0,1]
                new_h = np.clip(new_h, 0, 1)
                # Rescale to sum = n_points/2
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                # Evaluate C5
                new_c5 = evaluate_c5(new_h, dx)
                if new_c5 < best_new_c5:
                    best_new_c5 = new_c5
                    best_new_h = new_h.copy()
                    best_alpha = alpha
            
            # Update if improved
            if best_new_c5 < best_c5:
                h = best_new_h.copy()
                best_c5 = best_new_c5
                iterations_no_improvement = 0
                if best_c5 < best_h.sum() * dx:
                    best_h = h.copy()
            else:
                iterations_no_improvement += 1
                
        else:
            iterations_no_improvement += 1
            
        # Escape local minima with perturbations
        if iterations_no_improvement >= max_no_improvement:
            h += perturbation_magnitude * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            # Rescale to correct sum
            h = h * (n_points / 2) / np.sum(h)
            iterations_no_improvement = 0
            
        # Prevent degenerate solutions and track time
        if time.time() - start_time > budget_s:
            break
            
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    return (list(best_h), final_c5, n_points)

def evaluate_c5(h, dx):
    # Compute max cross-correlation using np.correlate with mode='full'
    full_correlation = correlate(h, 1 - h, mode='full')
    # The full correlation array has 2*n - 1 elements
    # The valid lags are from 0 to 2*n - 2
    # The maximum lag is 2*n - 2
    # We want the lags from 0 to 2*n - 2
    max_val = max(full_correlation) * dx
    return max_val
