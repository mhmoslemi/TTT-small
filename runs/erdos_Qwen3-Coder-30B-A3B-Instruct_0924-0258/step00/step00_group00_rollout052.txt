import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45  # Starting with the provided construction size
    dx = 2.0 / n_points
    max_iter = 1000
    n_particles = 30
    inertia_w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_stagnation = 100
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = h_start / np.sum(h_start) * (n_points / 2)
    else:
        h_start = np.random.rand(n_points)
        h_start = h_start / np.sum(h_start) * (n_points / 2)
    
    particles = []
    velocities = []
    personal_best = []
    personal_best_scores = []
    
    for _ in range(n_particles):
        h = h_start.copy()
        if 'initial_h_values' not in globals():
            # Randomize within constraints, ensuring sum is correct
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        personal_best.append(h.copy())
        personal_best_scores.append(float('inf'))
    
    # Global best
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    start_time = time.time()
    
    # PSO optimization
    iteration = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        iteration += 1
        
        for i in range(n_particles):
            h = particles[i]
            
            # Clip to [0,1] and rescale to maintain sum
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            particles[i] = h
            
            # Compute overlap
            corr = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < personal_best_scores[i]:
                personal_best_scores[i] = c5
                personal_best[i] = h.copy()
                
            # Update global best
            if c5 < global_best_score:
                global_best_score = c5
                global_best = h.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Update particles
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            h = particles[i]
            v = velocities[i]
            
            # Velocity update
            v = (inertia_w * v + 
                 c1 * r1 * (personal_best[i] - h) + 
                 c2 * r2 * (global_best - h))
            
            # Position update
            h = h + v
            
            # Ensure constraints
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            
            particles[i] = h
            velocities[i] = v
            
        # Stop if stagnated
        if stagnation_count > max_stagnation:
            break
    
    # Final evaluation
    corr = signal.correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(corr) * dx
    final_h = np.clip(global_best, 0, 1)
    final_h = final_h / np.sum(final_h) * (n_points / 2)
    
    return (final_h.tolist(), final_c5, n_points)
