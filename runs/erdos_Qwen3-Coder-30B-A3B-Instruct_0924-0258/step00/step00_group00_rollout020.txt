import numpy as np
import time
from scipy import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for a given h."""
    corr = correlate(h, 1 - h, mode="full")
    # Only consider valid lags (the middle part)
    mid = len(corr) // 2
    valid_corr = corr[mid - len(h) + 1:mid + len(h) - 1]
    return np.max(valid_corr) * dx

def rescale_to_unit_sum(h, target_sum):
    """Rescale h so that sum(h) == target_sum."""
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.full_like(h, target_sum / len(h))
    return h * target_sum / current_sum

def mutate(h, mutation_rate=0.1, delta=0.1):
    """Apply mutation to a candidate solution."""
    new_h = h.copy()
    for i in range(len(new_h)):
        if np.random.random() < mutation_rate:
            new_h[i] += np.random.uniform(-delta, delta)
    # Clip to [0, 1]
    new_h = np.clip(new_h, 0, 1)
    # Adjust to maintain unit sum
    new_h = rescale_to_unit_sum(new_h, len(new_h) / 2)
    return new_h

def crossover(parent1, parent2):
    """Perform one-point crossover."""
    point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    # Keep sum fixed
    child1 = rescale_to_unit_sum(child1, len(child1) / 2)
    child2 = rescale_to_unit_sum(child2, len(child2) / 2)
    return child1, child2

def tournament_selection(population, fitnesses, k=3):
    """Select an individual using tournament selection."""
    best_idx = np.random.choice(len(population))
    best_fitness = fitnesses[best_idx]
    for _ in range(k - 1):
        idx = np.random.choice(len(population))
        if fitnesses[idx] < best_fitness:
            best_idx = idx
            best_fitness = fitnesses[idx]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load initial solution if available (after prepending the parent)
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        n = len(h_parent)
    else:
        n = 45
        h_parent = np.full(n, 0.5)  # fallback
    
    # Check or fix the normalization
    dx = 2.0 / n
    if abs(np.sum(h_parent) * dx - 1.0) > 1e-10:
        h_parent = rescale_to_unit_sum(h_parent, n / 2)

    # Parameters
    pop_size = 50
    generations = 200
    mutation_rate = 0.1
    crossover_rate = 0.7
    
    # Initialize population
    population = [h_parent]
    for _ in range(pop_size - 1):
        # Create a small perturbation
        h = h_parent + np.random.uniform(-0.2, 0.2, n)
        # Clip
        h = np.clip(h, 0, 1)
        # Rescale
        h = rescale_to_unit_sum(h, n / 2)
        population.append(h)
    
    start_time = time.time()
    
    best_h = h_parent.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Main GA loop
    for gen in range(generations):
        if time.time() - start_time >= budget_s:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Update best
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitnesses.index(min_fitness)].copy()
        
        # Selection, crossover, mutation
        new_population = []
        
        # Elitism
        sorted_indices = np.argsort(fitnesses)
        new_population.extend([population[i] for i in sorted_indices[:2]])
        
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            if np.random.rand() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
                new_population.extend([child1, child2])
            else:
                # Just mutate one
                child1 = mutate(parent1, mutation_rate)
                child2 = mutate(parent2, mutation_rate)
                new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:pop_size]

    return (best_h, best_c5, n)
