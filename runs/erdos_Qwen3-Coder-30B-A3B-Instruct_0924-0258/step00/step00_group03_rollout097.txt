import numpy as np
import time
from scipy import signal
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to correct integral
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    def compute_c5(h_vec):
        # Convolve h with (1 - h) for all lags
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Take maximum, scaled by dx
        return np.max(conv) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Gradient-based update with constraints
    def opt_step(h_vec, step_size=0.01):
        # Compute overlap for all lags
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        max_idx = np.argmax(conv)
        
        # Try to project gradient to unit box
        # Gradient of the maximum term with respect to h
        # We compute it via finite differences for robustness
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            conv_plus = signal.correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (conv_plus[max_idx] - conv[max_idx]) / eps
        
        # Update step
        h_new = h_vec - step_size * grad
        # Project to [0,1] box
        h_new = np.clip(h_new, 0, 1)
        # Project to correct integral
        current_sum = np.sum(h_new)
        h_new = h_new * (target_sum / current_sum)
        return h_new
    
    # Optimization loop
    iterations = 0
    while time.time() - start_time < budget_s - 0.1:
        iterations += 1
        # Try various step sizes
        step_sizes = [0.005, 0.01, 0.02, 0.05]
        for step_size in step_sizes:
            h_trial = opt_step(best_h, step_size)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
                break  # Accept the improvement and try again
        else:
            # No improvement found; try small perturbation
            h_perturbed = best_h + np.random.normal(0, 0.001, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            current_sum = np.sum(h_perturbed)
            h_perturbed = h_perturbed * (target_sum / current_sum)
            c5_perturbed = compute_c5(h_perturbed)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed

    return (best_h.tolist(), best_c5, n_points)
