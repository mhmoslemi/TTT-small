import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute cross-correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Extract the relevant lags
    max_corr = 0.0
    for k_idx in range(2 * n_points - 1):
        # The maximum lag is associated with index n_points - 1 in the full correlation
        corr_val = corr[k_idx] * dx
        if corr_val > max_corr:
            max_corr = corr_val
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, else create a reasonable one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a binary-like initial guess
        n_points = 45
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        h = h / np.sum(h) * (n_points / 2)  # rescale

    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    no_improvement_count = 0
    perturbation_magnitude = 0.01
    max_no_improvement = 20

    dx = 2.0 / n_points

    # Optimization loop
    while time.time() - start_time < budget_s - 0.1:
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                # Perturb to escape local minimum
                h += np.random.randn(n_points) * perturbation_magnitude
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                no_improvement_count = 0

        # Compute all correlations for gradient estimation
        corr = correlate(h, 1 - h, mode='full')
        max_c5 = 0.0
        max_indices = []
        for k_idx in range(2 * n_points - 1):
            corr_val = corr[k_idx] * dx
            if corr_val > max_c5:
                max_c5 = corr_val
                max_indices = [k_idx]
            elif corr_val == max_c5:
                max_indices.append(k_idx)

        # Compute gradient for the maximum indices
        gradient = np.zeros(n_points)
        for k_idx in max_indices:
            # The lag k goes from 0 to n_points - 1
            # Full correlation index k_idx maps to lag k = k_idx - (n_points-1)
            lag = k_idx - (n_points - 1)
            if lag >= 0:
                # h[i] * (1 - h[i + lag])
                for i in range(n_points - lag):
                    gradient[i] += (1 - h[i + lag])
                    gradient[i + lag] -= h[i]
            else:
                # lag is negative, so we use the opposite
                # h[i - lag] * (1 - h[i])
                lag_abs = -lag
                for i in range(n_points - lag_abs):
                    gradient[i - lag_abs] += (1 - h[i])
                    gradient[i] -= h[i - lag_abs]

        # Line search for step size
        step_sizes = [0.1, 0.25, 0.5]
        best_alpha = 0.0
        best_h_candidate = h.copy()
        best_candidate_c5 = current_c5
        for alpha in step_sizes:
            h_candidate = h - alpha * gradient
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            if candidate_c5 < best_candidate_c5:
                best_candidate_c5 = candidate_c5
                best_h_candidate = h_candidate.copy()
                best_alpha = alpha

        if best_alpha != 0.0:
            h = best_h_candidate
        else:
            # No improvement, use small step
            h = h - 0.01 * gradient
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (list(best_h), final_c5, n_points)
