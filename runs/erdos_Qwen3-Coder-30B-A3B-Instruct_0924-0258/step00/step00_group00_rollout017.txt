import time

    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure all values are in [0, 1]
    h = np.clip(h, 0, 1)
    
    # Evaluate initial cost
    start_time = time.time()
    dx = 2.0 / n_points
    
    # Compute full correlation
    full_corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = max(full_corr) * dx
    
    best_h = h.copy()
    best_c5 = current_c5
    
    iter_no_improve = 0
    max_iter_no_improve = 20
    
    while True:
        if time.time() - start_time > budget_s:
            break
            
        # Compute full correlation
        full_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = max(full_corr) * dx
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            iter_no_improve = 0
        else:
            iter_no_improve += 1
            
        if iter_no_improve >= max_iter_no_improve:
            # Perturb to escape local minima
            h += 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            iter_no_improve = 0
            continue
            
        # Find lag indices that contribute the most to C₅
        max_corr_val = max(full_corr)
        max_indices = [i for i, val in enumerate(full_corr) if abs(val - max_corr_val) < 1e-10]
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for k in max_indices:
            # The lags in correlate are from -n+1 to n-1
            # So k corresponds to lag: k - (n_points-1)
            lag = k - (n_points - 1)
            # For positive lags: y[i] * (1 - y[i + lag]) 
            if lag >= 0:
                # Loop through where i + lag < n_points
                for i in range(n_points - lag):
                    gradient[i] += (1 - h[i + lag])
                    if i + lag < n_points:
                        gradient[i + lag] -= h[i]
            else:  # negative lag
                # x[i] * (1 - y[i - lag]) 
                for i in range(-lag, n_points):
                    gradient[i] += (1 - h[i + lag])
                    if i + lag < n_points:
                        gradient[i + lag] -= h[i]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient /= grad_norm
        
        # Try different step sizes
        step_sizes = [0.1, 0.25, 0.5]
        best_h_candidate = None
        min_c5_candidate = np.inf
        
        for alpha in step_sizes:
            h_candidate = h - alpha * gradient
            h_candidate = np.clip(h_candidate, 0, 1)
            # Renormalize
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Evaluate C₅ for candidate
            full_corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            candidate_c5 = max(full_corr_candidate) * dx
            
            if candidate_c5 < min_c5_candidate:
                min_c5_candidate = candidate_c5
                best_h_candidate = h_candidate.copy()
                
        if best_h_candidate is not None and min_c5_candidate < current_c5:
            h = best_h_candidate
        else:
            # If no descent, perturb to get out of a flat region
            h += 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final check and recompute
    full_corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = max(full_corr_final) * dx
    
    n_points = len(best_h)
    return (best_h.tolist(), final_c5, n_points)
