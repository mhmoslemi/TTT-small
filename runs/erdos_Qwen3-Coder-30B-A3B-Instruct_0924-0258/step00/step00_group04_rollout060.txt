import numpy as np
import time
from scipy import optimize

def evaluate_c5(h):
    # Compute the overlap integral C_5
    n = len(h)
    dx = 2.0 / n
    # Compute the full convolution
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum coefficient is the C_5 value
    c5 = np.max(conv) * dx
    return c5

def project_and_rescale(h, n_points):
    # Project to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def gradient_step(h, epsilon=1e-5):
    n = len(h)
    grad = np.zeros(n)
    current_c5 = evaluate_c5(h)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = project_and_rescale(h_plus, n)
        c5_plus = evaluate_c5(h_plus)
        grad[i] = (c5_plus - current_c5) / epsilon
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    n_points = 42

    else:

    h = project_and_rescale(h, n_points)
    
    # Parameters for annealing
    temp = 1.0
    cooling_factor = 0.99
    min_temp = 0.01
    patience = 50
    patience_counter = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    start_time = time.time()
    
    # Iterate
    for iteration in range(100000):
        if time.time() - start_time > budget_s:
            break
            
        # Compute gradient
        grad = gradient_step(h)
        
        # Generate candidate
        noise = np.random.normal(0, 0.05, n_points)
        h_candidate = h - 0.1 * grad + noise
        h_candidate = project_and_rescale(h_candidate, n_points)
        
        # Evaluate candidate
        c5_candidate = evaluate_c5(h_candidate)
        
        # Accept or reject
        if c5_candidate < best_c5:
            h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
            patience_counter = 0
        elif np.random.random() < np.exp(-(c5_candidate - evaluate_c5(h)) / temp):
            h = h_candidate.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Temperature schedule
        temp = max(temp * cooling_factor, min_temp)
        
        # Reset if no progress
        if patience_counter > patience:
            temp = 1.0
            patience_counter = 0
    
    return (best_h.tolist(), best_c5, n_points)
