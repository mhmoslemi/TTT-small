import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)
        h[0] = 0.4
        h[1] = 0.6
        h[2] = 0.3
        h[3] = 0.7
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    max_iter = 500
    alpha = 0.1
    tol = 1e-4
    
    last_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all lags
        conv = signal.correlate(h, 1 - h, mode='full')
        c5_vals = conv * dx
        max_c5 = np.max(c5_vals)
        
        if abs(max_c5 - last_c5) < tol:
            break
            
        last_c5 = max_c5
        
        # Get indices of maximum values
        max_indices = np.where(c5_vals == max_c5)[0]
        
        # Compute gradient
        gradient = np.zeros_like(h)
        for idx in max_indices:
            if idx < n_points:
                # h[i] appears in overlap with 1-h[i+idx] 
                gradient += np.roll(1 - h, -idx)
            else:
                # Reverse overlap
                offset = idx - n_points + 1
                gradient += (1 - np.roll(h, offset))
                
        # Update step
        h_new = h - alpha * gradient
        
        # Project h_new back to feasible region
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final evaluation
    conv = signal.correlate(h, 1 - h, mode='full')
    c5_bound = np.max(conv * dx)
    
    return (h.tolist(), c5_bound, n_points)
