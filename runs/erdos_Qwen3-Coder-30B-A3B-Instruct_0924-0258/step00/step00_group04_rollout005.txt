import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Evaluate C5 for given h and dx."""
    n = len(h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Only consider the relevant part of correlation (margins that could be valid)
    # the overlap integral involves shifting h by k, collecting all valid overlaps
    # and taking max among them
    # the full convolution has length 2*n - 1
    # we really want the part where k ranges from 0 to n-1 (as the shifted version 
    # of h must not go out of bounds more than 1 unit in either direction, given interval [0,2])
    # so we take indices from n-1 to 2*n-2
    # but we need to adjust indices, and the actual computation is:
    # for k=0 to n-1, integrate h(x) * (1-h(x+k)) for valid regions
    # which is effectively what np.correlate computes
    # here we consider full, which is not the precise thing we want due to finite intervals.
    # The valid range in indices for k is from 0 to n-1 (so as we shift by k, we shift by at most n-1)
    # in a circular sense, so we take the central part from indices n-1 to n+1 + n-1 of the full convolution?
    # Let's correct this - we want max of overlaps for k in [0, n-1]
    # the correlation gives us values for k in [-n+1, n-1] as shift values, but we only want k in [0, n-1]
    # thus taking values for k from 0 to n-1 corresponds to indices n-1 to 2*n-2 inclusive
    # But actually, since h(x) in [0,1], and dx=2/n, the product h(x)(1-h(x+k)) bounded by 0.25
    # so, the maximum contribution is at most 0.25 * (2/n) = 0.5/n
    # Let's correct evaluation:
    # for (n-1) to (2*n-2) index of full convolution: this is shift k from 0 to n-1
    # we use domain [0,2]. We sample as h[i] for i=0,1...n-1, meaning h(x) = h[i] where i*dx = x
    # So if k = i_k*2/n, we want to implement to be ŷ_{i_k} = sum over i of h[i] * (1 - h[(i+i_k)%n]) * dx 
    # Let's explicitly compute via convolution (standard PyTorch-style):
    
    # We have a valid portion in the middle of convolution that represents shift k from 0 to n-1
    # We're interested in shift values 0 through n-1
    # In my implementation of np.correlate, it's trying to compute cov(h, 1-h), 
    # so the result has length 2*n-1, and values for shift k from (1-n) to (n-1).
    # We want k = 0 to n-1, which corresponds to indices from n-1 to 2*n-2
    # but in Python indices start at 0 so:
    # shift k=0 maps to corr_idx = n-1
    # shift k=n-1 maps to corr_idx = 2*n-2
    # So we take corr[n-1:2*n-1] which has len n
    # (from n-1 to 2*n-1 inclusive, which is a total of n elements)
    # Let's approach differently: it's better to note that the problem is
    # integrate h(x) * (1 - h(x+k)) dx for x from 0 to 2-k, so a complete range of integration
    # but considering h is step function over [0,2]
    
    # Given h is a step function sampled at n points, one represents h(x) as h[i] where x ~ i*dx.
    # We properly compute the shift-based integral. Let’s do manual loop for clarity
    # compute for each k in [0,n-1], the integral by discrete sum:
    
    n = len(h)
    max_overlap = 0
    for k in range(n):
        overlap = 0.0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        overlap *= dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap


def project_and_rescale(h, n_points):
    """Ensure h is in [0,1] and sum(h) == n_points / 2 exactly."""
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * (n_points / 2) / np.sum(h)
    else:
        # avoid division by zero
        h.fill(1.0)
        h[:n_points//2] = 0.0
    return h


def mutate(h, mutation_rate=0.05):
    """Apply mutation to a solution."""
    mask = np.random.random(len(h)) < mutation_rate
    if np.any(mask):
        h[mask] += np.random.normal(0, 0.1, np.sum(mask))
        h = project_and_rescale(h, len(h))
    return h


def crossover(h1, h2):
    """Perform uniform crossover."""
    n = len(h1)
    # Choose 2 crossover points
    points = sorted(np.random.choice(n, 2, replace=False))
    h = np.copy(h1)
    h[points[0]:points[1]] = h2[points[0]:points[1]]
    return project_and_rescale(h, n)


def local_optimize(h, dx, maxiter=50):
    """Apply local optimization to a solution."""
    n = len(h)
    def objective(h_flat):
        h = h_flat.copy()
        h = project_and_rescale(h, n)
        return evaluate_c5(h, dx)
        
    # Very rough approximation: for a few rounds, randomly nudge points and accept good ones
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    current_h = h.copy()
    current_c5 = best_c5
    
    for _ in range(1000):
        if np.random.rand() < 0.1:  # 10% chance to mutate
            # mutate with small noise
            for i in range(len(current_h)):
                if np.random.rand() < 0.05:
                    current_h[i] += np.random.normal(0, 0.05)
            current_h = project_and_rescale(current_h, n)
            current_c5 = evaluate_c5(current_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    return best_h, best_c5


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42 # 42 is used from initial_h_values, or use a reasonable small number
    
    dx = 2.0 / n_points
    pop_size = 50
    max_generations = 150
    mutation_rate = 0.05
    stagnation_threshold = 20
    start_time = time.time()

    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        pop = initial_h_values.copy()
        pop = project_and_rescale(pop, n_points)
        population.append(pop)
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = project_and_rescale(h, n_points)
            population.append(h)
    else:
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = project_and_rescale(h, n_points)
            population.append(h)

    best_c5 = float('inf')
    best_h = None

    last_improvement = 0
    generation = 0

    while generation < max_generations:
        if time.time() - start_time > budget_s - 5:
            break 
        if generation % 10 == 0:
            print(f"Generation {generation}")

        # Evaluate current population
        fitness = []
        for h in population:
            fit = evaluate_c5(h, dx)
            if fit < best_c5:
                best_c5 = fit
                best_h = h.copy()
            fitness.append((fit, h))

        fitness.sort(key=lambda x: x[0])
        
        # Keep top 20% as parents for next generation
        top_k = int(0.2 * pop_size)
        parents = [x[1] for x in fitness[:top_k]]
        
        # Generate offspring with crossover
        offspring = []
        children_count = pop_size - len(parents)  # Keep parents and add offspring
        while len(offspring) < children_count:
            if len(parents) < 2:
                # just random mutates
                h = parents[0].copy() if parents else np.random.rand(n_points)
            else:
                h1, h2 = np.random.choice(parents, 2, replace=False)
                h = crossover(h1, h2)
            offspring.append(mutate(h, mutation_rate))
            
        # Local optimization on top 10%
        top_cand = [x[1] for x in fitness[:int(0.1 * pop_size)]] 
        for h in top_cand:
            h, c5 = local_optimize(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Replace worst of population with offspring
        population = parents + offspring
        # Trim to fixed size if necessary
        if len(population) > pop_size:
            fitness = [(evaluate_c5(h, dx), h) for h in population]
            fitness.sort(key=lambda x: x[0])
            population = [x[1] for x in fitness[:pop_size]]
        
        generation += 1
        if generation - last_improvement > stagnation_threshold:
            mutation_rate = 0.1  # Increase to escape local minimum
            last_improvement = generation

    return (best_h, best_c5, n_points)
