import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h):
    """Evaluate the maximum overlap integral C5 for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations with 1-h
    corr = correlate(h, 1 - h, mode="full")
    # Take the maximum value over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy the constraint sum(h) = n_points / 2."""
    h = np.array(h)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    h = np.clip(h, 0, 1)
    return h
