import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute the full correlation
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value is the maximum of the correlation (excluding the 0-shift)
    max_overlap = np.max(corr)
    return max_overlap * dx

def initialize_population(initial_h, n_points, pop_size):
    population = []
    # Include the initial solution
    population.append(initial_h.copy())
    
    # Generate other solutions by perturbing the initial
    for i in range(pop_size - 1):
        # Perturb slightly
        pert = np.random.normal(0, 0.05, n_points)
        h_new = initial_h + pert
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Rescale to have correct sum
        h_new *= (n_points / 2) / np.sum(h_new)
        population.append(h_new)
        
    return population

def selection(population, fitness):
    # Select top 50% based on fitness (ascending)
    sorted_indices = np.argsort(fitness)
    selected_indices = sorted_indices[:len(population)//2]
    return [population[i] for i in selected_indices]

def crossover(parent1, parent2):
    n = len(parent1)
    # Single point crossover
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1):
    n = len(h)
    h_new = h.copy()
    # Mutate some positions
    indices_to_mutate = np.random.choice(n, int(n * mutation_rate), replace=False)
    for i in indices_to_mutate:
        h_new[i] += np.random.normal(0, 0.1)
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Rescale to have correct sum
    h_new *= (n / 2) / np.sum(h_new)
    return h_new
