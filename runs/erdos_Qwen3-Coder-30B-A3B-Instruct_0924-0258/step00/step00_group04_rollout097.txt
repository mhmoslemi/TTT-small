import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for lags
    correlations = correlate(h, 1 - h, mode='full')
    # Extract the valid lags that contribute to the maximum
    mid = len(correlations) // 2
    valid_correlations = correlations[mid:mid + n]
    c5 = np.max(valid_correlations) * dx
    return c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 42  # Use the provided initial construction size
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert np.all(h >= 0) and np.all(h <= 1), "Initial h must be in [0,1]"
        assert abs(np.sum(h) - n_points / 2) < 1e-6, "Sum of h must be n_points/2"
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    alpha = 0.1
    backtrack_factor = 0.8
    max_iter = int(budget_s * 10)  # Rough estimation for time limit
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        # Compute overlaps for all lags
        correlations = correlate(h, 1 - h, mode='full')
        mid = len(correlations) // 2
        valid_correlations = correlations[mid:mid + n_points]
        c5_val = np.max(valid_correlations)
        k_star = np.argmax(valid_correlations)
        
        # Compute gradient: ∂C5/∂h[j] = (2/n_points) * (1 - 2*h[(j + k*) mod n])
        grad = np.zeros(n_points)
        for j in range(n_points):
            grad[j] = 1 - 2 * h[(j + k_star) % n_points]
        grad *= 2.0 / n_points
        
        # Line search
        step_size = alpha
        h_new = h - step_size * grad
        h_proj = np.clip(h_new, 0, 1)
        # Rescale to satisfy sum constraint
        h_proj = h_proj / np.sum(h_proj) * (n_points / 2)
        
        # Backtracking line search
        while step_size > 1e-6:
            h_test = h - step_size * grad
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test / np.sum(h_test) * (n_points / 2)
            test_c5 = evaluate_erdos_solution(h_test)
            if test_c5 < c5_val:
                h = h_test
                c5_val = test_c5
                break
            step_size *= backtrack_factor
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        if step_size < 1e-6:
            break
            
    return (best_h.tolist(), best_c5, n_points)
