import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    # Compute overlap integral
    cross_corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(cross_corr)
    return max_corr * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    dx = 2.0 / n_points
    
    # Ensure initial sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Frank-Wolfe loop
    max_iter = 500
    tol = 1e-4
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlations to find maximum
        cross_corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(cross_corr)
        max_val = cross_corr[max_k]
        
        # Compute gradient of this correlation w.r.t h
        # element-wise gradient for h[i] 
        # For cross-correlation at lag k, element h[i] contributes to
        # term h[i] * (1 - h[i+k]), so gradient w.r.t h[i] = 1 - h[i+k] 
        # But we need to normalize by dx for each such gradient
        # Actually let's recompute more carefully:
        # gradient at h[i] for cross-correlation term at lag k is:
        #   d/dh[i] sum_j h[j] * (1 - h[j+k]) = 1 - h[i+k] if i+k < len(h) else 0
        
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            if i + max_k < len(h):
                gradient[i] = 1 - h[i + max_k]
            else:
                gradient[i] = 0
        
        # Sort indices by gradient (descending)
        sorted_indices = np.argsort(-gradient)
        
        # Select top indices to form active set S
        S = sorted_indices[:n_points // 2]
        
        # Frank-Wolfe direction from S
        # s = new direction vector that is 1 at indices in S and 0 elsewhere
        s = np.zeros_like(h)
        s[S] = 1.0
        s = s / np.sum(s) * (n_points / 2)  # ensure it has total mass n_points / 2
        
        # Line search over [0, 1]
        gamma_candidates = [0.1, 0.25, 0.5, 0.75, 1.0]
        best_gamma = 0.0
        min_c5 = best_c5
        
        for gamma in gamma_candidates:
            new_h = h + gamma * (s - h)
            new_h = np.clip(new_h, 0, 1)
            # Rescale to match constraint
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_c5 = evaluate_h(new_h, dx)
            if new_c5 < min_c5:
                min_c5 = new_c5
                best_gamma = gamma
                
        # Update h
        h = h + best_gamma * (s - h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Update best if improved
        if min_c5 < best_c5:
            best_c5 = min_c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
