def evaluate_erdos_solution(h, n_points):
    """Evaluate the Erdős constant for given h"""
    dx = 2.0 / n_points
    # Compute overlap integral for all shifts
    overlaps = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(overlaps) * dx
    return max_overlap
