import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    pop_size = 50
    iterations = 200
    crossover_rate = 0.7
    mutation_rate = 0.2
    mutation_delta = 0.1
    tournament_k = 3
    
    # Initialize population
    def rescale(h):
        """Ensure sum(h) = n_points / 2"""
        s = np.sum(h)
        if s == 0:
            return np.full_like(h, 0.5)
        return h * (n_points / 2) / s
    
    def create_individual():
        """Create a random individual"""
        h = np.random.random(n_points)
        h = rescale(h)
        return h
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
    else:
        population = []
    
    # Add 49 random individuals
    while len(population) < pop_size:
        individual = create_individual()
        population.append(individual)
    
    def evaluate(individual):
        """Compute C5 for individual"""
        dx = 2.0 / n_points
        corr = correlate(individual, 1 - individual, mode='full')
        # Take only the relevant part (max k corresponds to max lag)
        c5 = np.max(corr) * dx
        return c5
    
    def tournament_selection(pop, fitnesses):
        """Select one individual via tournament"""
        indices = np.random.choice(len(pop), size=tournament_k, replace=False)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        return pop[best_idx]
    
    def crossover(parent1, parent2):
        """One-point crossover"""
        if np.random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = np.random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return rescale(child1), rescale(child2)
    
    def mutate(individual):
        """Mutate individual with probability 10% for each element"""
        child = individual.copy()
        for i in range(n_points):
            if np.random.random() < 0.1:
                child[i] += np.random.normal(0, mutation_delta)
        child = np.clip(child, 0, 1)
        return rescale(child)
    
    start_time = time.time()
    
    # Evaluate all individuals
    fitnesses = [evaluate(ind) for ind in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # GA Loop
    for _ in range(iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep top individuals
        next_pop = sorted_pop[:pop_size//2]
        next_fitnesses = sorted_fitnesses[:pop_size//2]
        
        # Create offspring
        while len(next_pop) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            
            if np.random.random() < mutation_rate:
                child1 = mutate(child1)
            if np.random.random() < mutation_rate:
                child2 = mutate(child2)
                
            next_pop.extend([child1, child2])
            if len(next_pop) > pop_size:
                next_pop = next_pop[:pop_size]
        
        # Evaluate new generation
        population = next_pop
        fitnesses = [evaluate(ind) for ind in population]
        
        # Update global best
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
    
    return (best_h.tolist(), best_c5, n_points)
