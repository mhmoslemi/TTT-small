import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42  # use the provided initial construction
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        assert np.all(h >= 0) and np.all(h <= 1)
        assert abs(np.sum(h) - target_sum) < 1e-6
    else:
        h = np.random.uniform(0, 1, n_points)
        h *= target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Projective gradient descent with line search
    max_iter = 100000
    alpha0 = 0.1
    backtracking_factor = 0.8
    min_alpha = 1e-6
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find worst lag k*
        # correlate with 1-h (reversed)
        h_rev = 1 - h[::-1]
        corr_full = correlate(h, h_rev, mode='full')
        # Extract the valid part for lags 0 to n_points-1
        corr_vals = corr_full[len(h)-1 : len(h)-1+n_points] * dx
        
        k_star = np.argmax(corr_vals)
        c5_val = corr_vals[k_star]
        
        if c5_val < best_c5:
            best_h[:] = h
            best_c5 = c5_val
            
        # Compute gradient
        grad = np.zeros(n_points)
        for j in range(n_points):
            idx = (j + k_star) % n_points
            grad[j] = 1 - 2 * h[idx]
        
        # Normalize gradient
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-12:
            break
        grad /= norm_grad
        
        # Line search
        alpha = alpha0
        while alpha >= min_alpha:
            h_new = h - alpha * grad
            # Project to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to maintain sum = target_sum
            h_new *= target_sum / np.sum(h_new)
            
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < c5_val:
                h[:] = h_new
                break
            alpha *= backtracking_factor
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_c5(h, dx):
    # h is a numpy array of size n_points
    n = len(h)
    # Compute correlations for all lags
    h_rev = 1 - h[::-1]
    corr_full = correlate(h, h_rev, mode='full')
    # Extract valid correlations
    corr_vals = corr_full[n-1 : n-1+n] * dx
    return np.max(corr_vals)
