import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    # Compute overlap integral for all lags
    conv = signal.correlate(h, 1 - h, mode='full')
    # Take only the relevant part of the correlation (maximum lag)
    max_overlap = np.max(conv[len(h)-1:])
    return max_overlap * dx

def project_feasible(h, target_sum):
    # Ensure sum is correct and values stay in [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    # Ensure exact sum (rounding errors)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Adjust the largest values to make sum correct
        idx = np.argsort(h)[-min(len(h), int(abs(diff) * 100)):]
        h[idx] += diff / len(idx)
    return np.clip(h, 0, 1)

def initial_population(n_points, pop_size, initial_h_values):
    pop = []
    if initial_h_values is not None:
        # Use initial construction
        h = project_feasible(np.array(initial_h_values), n_points / 2)
        pop.append(h)
    else:
        h = np.ones(n_points) * 0.5
        h = project_feasible(h, n_points / 2)
        pop.append(h)
    # Add random solutions
    for _ in range(pop_size - len(pop)):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points / 2)
        pop.append(h)
    return pop

def crossover(p1, p2):
    # Arithmetic crossover
    alpha = 0.5
    child1 = alpha * p1 + (1 - alpha) * p2
    child2 = (1 - alpha) * p1 + alpha * p2
    return child1, child2

def mutate(h, mutation_rate, std_dev):
    # Add Gaussian noise
    mask = np.random.rand(len(h)) < mutation_rate
    noise = np.random.normal(0, std_dev, len(h))
    h = h + noise * mask
    # Clamp to feasible range and reproject
    h = project_feasible(h, len(h) / 2)
    return h

def tournament_selection(pop, fitnesses, tournament_size=3):
    idxs = np.random.choice(len(pop), tournament_size)
    best = np.argmin([fitnesses[i] for i in idxs])
    return pop[idxs[best]]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 47
    dx = 2.0 / n_points
    pop_size = 20
    mutation_rate = 0.05
    max_generations = 1000
    patience = 100
    
    # Initialize population
    population = initial_population(n_points, pop_size, initial_h_values)
    fitnesses = [evaluate_h(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    no_improve_count = 0
    
    gen = 0
    while time.time() - start_time < budget_s - 1 and gen < max_generations:
        gen += 1
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitnesses)
        # Keep top 50% and generate new offspring
        new_pop = [population[i] for i in sorted_indices[:pop_size//2]]
        new_fitnesses = [fitnesses[i] for i in sorted_indices[:pop_size//2]]
        
        # Generate rest via crossover and mutation
        while len(new_pop) < pop_size:
            p1 = tournament_selection(population, fitnesses)
            p2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(p1, p2)
            child1 = mutate(child1, mutation_rate, 0.1)
            child2 = mutate(child2, mutation_rate, 0.1)
            new_pop.extend([child1, child2])
            new_fitnesses.extend([evaluate_h(child1, dx), evaluate_h(child2, dx)])
        
        population = new_pop[:pop_size]
        fitnesses = new_fitnesses[:pop_size]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        if no_improve_count >= patience:
            break
    
    return (list(best_h), best_c5, n_points)
