import numpy as np
from scipy import optimize as spopt
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        h0 = np.array(initial_h_values)
        n_points = len(h0)

        n_points = 70
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.0
        h0[1] = 1.0
        h0[2] = 0.0
    
    dx = 2.0 / n_points
    target_C5 = 0.3807
    low = 0.0
    high = 0.38085
    threshold = 1e-4
    best_C5 = high
    best_h = h0.copy()

    def compute_C5(h_vec):
        # Compute overlap integral using correlation
        h_vec = np.array(h_vec)
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # Take max over all lags
        max_corr = np.max(corr)
        return max_corr * dx

    def feasibility_check(h_vec, C5_val):
        """Check if a given h_vec satisfies constraints for C5_val using CVXPY"""
        h_vec = np.array(h_vec)
        n = len(h_vec)
        
        # Define variables
        h = cp.Variable(n)
        
        # Constraints: h[i] in [0,1]
        constraints = [h >= 0, h <= 1]
        
        # Constraint: integral = 1
        constraints.append(cp.sum(h) * dx == 1.0)
        
        # Constraint: for all shifts k, sum h[i](1-h[i+k]) <= C5*dx
        # Create correlation-like constraints for each valid lag
        for k in range(n):
            if k == 0:
                continue  # Skip k=0 (same point)
            # Compute sum over i of h[i] * (1 - h[i+k])
            expr = 0
            for i in range(n):
                j = (i + k) % n
                expr += h[i] * (1 - h[j])
            constraints.append(expr <= C5_val * dx)
        
        # Objective: not used (feasibility check)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            return prob.status == cp.OPTIMAL or prob.status == cp.INFEASIBLE or prob.status == cp.UNBOUNDED
        except:
            return False

    def binary_search_C5():
        nonlocal best_h, best_C5
        low_C = 0.0
        high_C = 0.38085
        
        while high_C - low_C > threshold and (time.time() - start_time) < budget_s - 1:
            mid = (low_C + high_C) / 2.0
            
            # Due to solver limitations, check a few variants around mid
            test_Cs = [mid, mid * 0.99, mid * 1.01]
            feasible = False
            for C_test in test_Cs:
                try:
                    if feasibility_check(h0, C_test):
                        # Re-solve to get feasible point
                        try:
                            h_new = h0.copy()
                            C_test_value = compute_C5(h_new)
                            if C_test_value <= C_test * 1.01:  # Allow small slack
                                feasible = True
                                if C_test_value < best_C5:
                                    best_C5 = C_test_value
                                    best_h = h_new.copy()
                                break
                        except:
                            # Fall back to checked value
                            pass
                except:
                    pass
            
            if feasible:
                high_C = mid
            else:
                low_C = mid

        return high_C

    # Simple optimization with bounds
    def optimize_h():
        nonlocal best_h, best_C5
        
        def obj_func(h_vec):
            return compute_C5(h_vec)
        
        # Bounds: each h[i] in [0,1]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Constraint: sum * dx = 1
        def constraint_func(h_vec):
            return np.sum(h_vec) * dx - 1.0

        cons = {'type': 'eq', 'fun': constraint_func}
        
        try:
            res = spopt.minimize(obj_func, h0, bounds=bounds, constraints=cons,
                                method='SLSQP', options={'maxiter': 1000, 'ftol': 1e-6})

                new_h = res.x
                new_C5 = compute_C5(new_h)
                if new_C5 < best_C5:
                    best_C5 = new_C5
                    best_h = new_h.copy()
        except:
            pass

    # First we try basic optimization
    optimize_h()
    
    # Then binary search for tight bound
    binary_search_C5()

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5

    return (best_h.tolist(), float(best_C5), n_points)
