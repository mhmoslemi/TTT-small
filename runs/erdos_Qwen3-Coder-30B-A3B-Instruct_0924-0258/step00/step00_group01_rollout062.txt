def compute_c5(h, dx):
    n_points = len(h)
    # Compute all overlaps for shifts 0 to 2
    corr = correlate(h, 1 - h, mode='full')
    # Get valid shifts (from 0 to 2)
    k_indices = np.arange(n_points, 2*n_points-1)
    shifts = np.arange(n_points-1, -n_points, -1) * dx
    valid_mask = (shifts >= 0) & (shifts <= 2)
    corr_vals = corr[valid_mask]
    return np.max(corr_vals) * dx
