import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute correlation for all shifts
    corr = np.correlate(h, 1 - h, mode='full')
    # Extract the valid correlations
    start_idx = n - 1
    valid_correlations = corr[start_idx:start_idx + n]
    # Scale appropriately
    c5 = np.max(valid_correlations) * dx
    return c5

def project_h(h):
    # Clamp values to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize sum to n/2
    n = len(h)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n / 2) / current_sum
    # Ensure correction to exactly n/2
    h = h * (n / 2) / np.sum(h)
    return h

def compute_gradient(h):
    n = len(h)
    grad = np.zeros(n)
    for k in range(-n + 1, n):
        if k < 0:
            # h[i] * (1 - h[i-k]) for i in range(n+k)
            for i in range(n + k):
                grad[i] += 1 - h[i - k]
        elif k > 0:
            # h[i] * (1 - h[i+k]) for i in range(n-k)
            for i in range(n - k):
                grad[i] += 1 - h[i + k]
        else:
            # h[i] * (1 - h[i]) for i in range(n)
            for i in range(n):
                grad[i] += 1 - 2 * h[i]
    return grad

def mutate(h, sigma):
    h = h + np.random.normal(0, sigma, len(h))
    return project_h(h)

def crossover(h1, h2, alpha=0.6):
    beta = np.random.uniform(alpha, 1 - alpha)
    h = beta * h1 + (1 - beta) * h2
    return project_h(h)

def run(seed=42, budget_s=600, **kwargs):

        h0 = project_h(h0)

    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(h0)
    
    # Evolution Parameters
    population_size = 100
    generations = 400
    mutation_rate = 0.1
    learning_rate = 0.02
    decay_rate = 100.0
    
    # Initialize population
    population = [h0.copy()]
    for i in range(population_size - 1):
        h = mutate(h0.copy(), 0.1)
        population.append(h)
    
    # Main loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Evaluate population
        fitness_values = [evaluate_erdos_solution(h) for h in population]
        
        # Update best
        for h, c5 in zip(population, fitness_values):
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_values)
        top_indices = sorted_indices[:population_size // 2]
        population = [population[i] for i in top_indices]
        
        # Generate new population
        new_population = population.copy()
        
        # Crossover
        for _ in range(population_size // 3):
            if len(population) < 2:
                break
            idx1, idx2 = np.random.choice(range(len(population)), 2, replace=False)
            child = crossover(population[idx1], population[idx2])
            new_population.append(child)
        
        # Mutation
        for _ in range(population_size // 2):
            idx = np.random.choice(range(len(population)))
            h = population[idx].copy()
            sigma = mutation_rate * np.exp(-gen / decay_rate)
            child = mutate(h, sigma)
            new_population.append(child)
        
        # Local refinement with gradient descent
        top_indices = sorted_indices[:population_size // 10]
        for idx in top_indices:
            h = population[idx].copy()
            for _ in range(5):
                grad = compute_gradient(h)
                h = h - learning_rate * grad
                h = project_h(h)
            new_population.append(h)
        
        # Keep best
        population = new_population[:population_size]
    
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
