import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time


def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for different lags
    corr = correlate(h, 1 - h, mode="full")
    # The maximum correlation (over all lags) scaled by dx
    max_corr = np.max(corr)
    return max_corr * dx


def project_and_rescale(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.full(n_points, 1.0 / n_points)
    return h


def crossover(h1, h2):
    """Perform crossover between two step functions."""
    n = len(h1)
    # Randomly select split points (2-3)
    n_splits = np.random.randint(2, 4)
    splits = sorted(np.random.choice(n - 1, n_splits, replace=False))
    splits = [0] + splits + [n]
    
    # Create offspring by alternating segments
    offspring = np.zeros(n)
    for i in range(n_splits):
        start, end = splits[i], splits[i+1]
        if i % 2 == 0:
            offspring[start:end] = h1[start:end]
        else:
            offspring[start:end] = h2[start:end]
    
    return offspring


def mutate(h, mutation_rate=0.05):
    """Mutate a step function by adding noise and rescaling."""
    n = len(h)
    indices = np.random.choice(n, int(n * mutation_rate), replace=False)
    noise = np.random.normal(0, 0.1, len(indices))
    h[indices] += noise
    return project_and_rescale(h, len(h))

    """Use L-BFGS-B to locally optimize the solution."""
    def objective(h):
        # Convert to array if needed
        h = np.array(h)
        # Ensure feasibility
        h = project_and_rescale(h, n_points)
        return evaluate_c5(h)
    
    def constraint(h):
        # Sum constraint
        return np.sum(h) - n_points / 2.0
    
    # Wrap objective for scipy
    def obj(h):
        return evaluate_c5(h)
    
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use L-BFGS-B with bounds
    result = minimize(obj, h0, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': maxiter}, tol=1e-8)
    
    h_opt = result.x
    # Project to maintain feasibility
    h_opt = project_and_rescale(h_opt, n_points)
    
    return h_opt
