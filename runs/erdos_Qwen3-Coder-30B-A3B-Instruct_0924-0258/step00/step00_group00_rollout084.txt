import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Frank-Wolfe with line search
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute cross-correlations
        corr = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        max_corr = corr[k_max] * dx
        
        # Compute gradient for the max correlation
        # gradient[i] = partial derivative of max_corr w.r.t. h[i]
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Only contributions from valid correlation terms
            for j in range(n_points):
                k = j - i
                if 0 <= k + n_points - 1 < len(corr):
                    # This is simplified: just focusing on the term h[i]*(1-h[i+k])
                    if 0 <= i + k < n_points:
                        gradient[i] += (1 - h[i + k]) * dx
        
        # Simple Frank-Wolfe step
        # Select top indices to update
        s_indices = np.argsort(gradient)[-n_points//2:]
        
        # Create update direction
        v = np.zeros(n_points)
        v[s_indices] = 1
        
        # Project onto constraints
        # For linear update h' = h + gamma*(v - h)
        # Need sum(h') = n_points/2 and 0 <= h' <= 1
        # h' = h + gamma*(v - h) 
        # So h'[i] = h[i] + gamma*(v[i] - h[i])
        # We solve for gamma such that sum(h') = n_points/2
        
        # Simple line search to find best gamma
        gamma_vals = [0.05, 0.1, 0.25, 0.5, 0.75, 1.0]
        best_gamma = 0.0
        min_C5 = max_corr
        
        try:
            for gamma in gamma_vals:
                h_trial = h + gamma * (v - h)
                # Ensure constraints
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                
                C5_trial = evaluate_C5(h_trial, dx)
                if C5_trial < min_C5:
                    min_C5 = C5_trial
                    best_gamma = gamma
        except:
            pass

        if best_gamma > 0:
            h = h + best_gamma * (v - h)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            if min_C5 < best_C5:
                best_C5 = min_C5
                best_h = h.copy()
    
    # Final evaluation
    # Make sure we have a good final result
    final_C5 = evaluate_C5(best_h, dx)
    
    return best_h.tolist(), final_C5, n_points

def evaluate_C5(h, dx):
    n = len(h)
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
