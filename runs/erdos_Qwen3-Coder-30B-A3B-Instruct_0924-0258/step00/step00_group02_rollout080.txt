import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Start with a simple construction
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial C5
    def compute_C5(h):
        dx = 2.0 / n_points
        # Compute all overlaps (full convolution)
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        # The C5 is the max of overlaps over all shifts
        return np.max(overlap)
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Gradient-based optimization with projection
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)
    
    # Optimizer loop
    iterations = 0
    while time.time() - start_time < budget_s - 1:
        iterations += 1
        
        # Compute overlaps for all shifts
        dx = 2.0 / n_points
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        
        # Find dominant shifts (top 5)
        dominant_indices = np.argsort(overlaps)[-5:]
        
        # Compute gradient w.r.t. h for these dominant shifts
        grad = np.zeros_like(h)
        for k in dominant_indices:
            # This is the gradient of sum_i h[i] * (1 - h[(i+k)%n]) over h
            # For each shift k, it's: 1 - 2 * h, but where h shifts
            # More precisely, for index i: gradient term from floor(k) and ceil(k)
            offset = k - n_points + 1  # Since we do circular correlation

            # compute derivative: 
            # for overlap_k = sum h[i] * (1 - h[(i+k)%n_points]) 
            # \partial overlap_k / \partial h[i] = (1 - h[(i+k)%n_points]) - h[i] * h'[(i+k)%n_points]
            # But since h is fixed the derivative only accumulates to h[i] itself (the term)
            
            # For exact computation in numpy: 
            # Transform h to correlate with (1-h) to obtain overlap_k for shift k
            # and its Jacobian can be approximated through finite difference or interpretation
            # Since we are going to optimize with projected gradient, finite difference around current h
            # will give us raw direction, and we project onto feasible set.

            # Instead, we compute a simplified gradient approximation
            h_shifted = np.roll(h, -offset)  # roll to get h[(i+k)%] 
            grad += (1 - h_shifted) - h_shifted  # gradient of overlap_k w.r.t. h[i]

        # Adjust for array indexing with circular convolution
        # Manual derivation would require some more complex handling
        
        # Simplified but effective: approximate gradient from overlapping shifts
        # We simplify to just adding gradient from dominant overlaps (to save computation)
        # on full overlap computation
        
        # Update using momentum and learning rate
        velocity = momentum * velocity + learning_rate * grad
        h_new = h - velocity
        
        # Project h_new onto [0,1] and preserve sum
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new C5
        new_C5 = compute_C5(h_new)
        
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
            # Adaptive learning rate decrease
            learning_rate *= 0.98
        else:
            # Occasionally perturb in case of getting stuck
            if iterations % 10 == 0:
                perturbation = np.random.normal(0, 0.005, n_points)
                h_new = h + perturbation
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                new_C5 = compute_C5(h_new)
                if new_C5 < best_C5:
                    best_C5 = new_C5
                    best_h = h_new.copy()
        
        h = h_new.copy()
        if iterations % 10 == 0:
            # Occasionally update best solution
            current_C5 = compute_C5(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
    
    return (best_h.tolist(), best_C5, n_points)
