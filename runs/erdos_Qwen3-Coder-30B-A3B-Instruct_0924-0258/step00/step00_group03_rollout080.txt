import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Evaluate the overlap integral for given h
    # h is assumed to be normalized to sum to n_points/2
    # We compute max_k ∫ h(x)(1 - h(x+k)) dx
    corr = correlate(h, 1 - h, mode='full')
    # The correlation is over 2*n_points-1 values for full mode
    # We only care about the valid lags from -n_points+1 to n_points-1
    # But for max_k we want max over all valid shifts 
    # Since h is periodic in some sense due to the full mode, we just take max of all values
    return np.max(corr) * dx

def normalize_h(h, n_points):
    # Ensure h sums to n_points/2 and stays in [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        # Scale to target sum
        h = h * (n_points / 2) / current_sum
    else:
        # If all zero, set to uniform distribution scaled to desired sum
        h = np.full(n_points, 1.0 / n_points * (n_points / 2))
    return h

def mutate_h(h, noise_level=0.05, n_points=None):
    # Add Gaussian noise with bounded magnitude to h and re-normalize
    if n_points is None:
        n_points = len(h)
    noise = np.random.normal(0, noise_level, n_points)
    # Limit the noise magnitude to avoid violating bounds - this approach ensures we stay in [0,1]
    h_new = h + noise
    return normalize_h(h_new, n_points)

def crossover_h(h1, h2):
    # Simple uniform crossover: randomly select elements from each parent
    n_points = len(h1)
    mask = np.random.rand(n_points) > 0.5
    h_new = np.where(mask, h1, h2)
    return h_new

def tournament_selection(population, fitnesses, tournament_size=5):
    # Select an individual via tournament selection
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    timeout = start_time + budget_s
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction: softer step function
        n_points = 70
        h_start = np.full(n_points, 0.5) # Start with uniform 0.5 everywhere
        # Increase front part to get some structure
        for i in range(n_points // 3):
            h_start[i] = 0.8
        h_start = normalize_h(h_start, n_points)

    # Population parameters
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    num_generations = 1000  # Allow up to 1000 generations

    # Initialize population
    population = [h_start.copy()]
    for _ in range(pop_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = normalize_h(h_rand, n_points)
        population.append(h_rand)

    # Evaluate initial population
    dx = 2.0 / n_points
    fitnesses = [evaluate_h(h, dx) for h in population]
    
    # Keep track of best
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]

    # Evolution
    generation = 0
    while time.time() < timeout and generation < num_generations:
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population_sorted = [population[i] for i in sorted_indices]
        fitnesses_sorted = [fitnesses[i] for i in sorted_indices]
        
        # Select and create offspring
        new_population = [population_sorted[0]]  # Elitism: keep best
        
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_rate:
                # Crossover
                parent1 = tournament_selection(population_sorted, fitnesses_sorted)
                parent2 = tournament_selection(population_sorted, fitnesses_sorted)
                child = crossover_h(parent1, parent2)
            else:
                # Clone or mutate parent
                parent = tournament_selection(population_sorted, fitnesses_sorted)
                child = parent.copy()
            
            # Mutate with probability
            if np.random.rand() < mutation_rate:
                child = mutate_h(child, n_points=n_points)
                
            # Make sure new individual is normalized
            child = normalize_h(child, n_points)
            
            new_population.append(child)
            
        # Replace population
        population = new_population
        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Update best if needed
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            
        generation += 1

    # Final evaluation of best
    final_fitness = evaluate_h(best_h, dx)
    
    # Return the result
    return (best_h.tolist(), final_fitness, n_points)
