from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):

    target_sum = n_points / 2.0

    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)

    def compute_C5(h_vals):
        # Compute all correlations
        h_padded = np.concatenate([h_vals, h_vals])
        overlaps = []
        for k in range(n_points):
            corr = np.sum(h_vals * (1 - h_padded[k:k+n_points]))
            overlaps.append(corr * dx)
        return max(overlaps)
    
    # Function to project h onto feasible region
    def project_h(h_vals):
        # Clamp to [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Rescale to correct sum
        current_sum = np.sum(h_vals)
        if current_sum > 0:
            h_vals = h_vals * target_sum / current_sum
        return h_vals
    
    # Initial evaluation
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Run a few gradient updates on top contributing overlaps
        # Compute overlaps and identify top contributing ones
        h_padded = np.concatenate([best_h, best_h])
        overlaps = []
        for k in range(n_points):
            corr = np.sum(best_h * (1 - h_padded[k:k+n_points]))
            overlaps.append(corr * dx)
        
        # Get indices of top 5 contributing overlaps
        top_k = np.argsort(overlaps)[-5:]
        
        # Gradient update based on those dominant overlaps
        grad = np.zeros_like(best_h)
        for k in top_k:
            # Gradient w.r.t. overlap_k
            h_shifted = np.roll(best_h, k)
            grad += (1 - h_shifted) * dx
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h_new = best_h + velocity
        
        # Project to feasible set
        h_new = project_h(h_new)
        
        # Evaluate new solution
        C5_new = compute_C5(h_new)
        
        # Accept update if improvement
        if C5_new < best_C5:
            best_h = h_new.copy()

            # Adapt learning rate
            learning_rate = min(0.1, learning_rate * 1.05)

            learning_rate *= 0.95

        if iteration % 20 == 0 and time.time() - start_time < budget_s - 1:

            h_perturbed = best_h + pert
            h_perturbed = project_h(h_perturbed)
            C5_pert = compute_C5(h_perturbed)
            if C5_pert < best_C5:
                best_h = h_perturbed
                best_C5 = C5_pert

    final_h = project_h(best_h)
    final_C5 = compute_C5(final_h)
