import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure the initial h satisfies the sum constraint
    h = h * target_sum / np.sum(h)
    
    # Objective function
    def compute_C5(h_vals):
        h = np.array(h_vals)
        overlaps = np.correlate(h, 1-h, mode="full")
        return np.max(overlaps) * dx
    
    # Helper to project h onto feasible region
    def project_h(h_vals):
        h = np.array(h_vals)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Adjust to meet sum constraint
        current_sum = np.sum(h)
        if abs(current_sum) > 1e-10:
            h = h * target_sum / current_sum
        return h
    
    # Precompute indices for circular correlation
    # This computes the indices for overlap_k calculations efficiently
    def compute_gradients(h_vals, top_k_indices):
        h = np.array(h_vals)
        grad = np.zeros_like(h)
        
        # For each k dominant index, accumulate gradient
        for k in top_k_indices:
            # Compute correlation at shift k (with wraparound)
            # h[i] * (1 - h[(i+k) % n_points]) contribution to gradient
            # This is the derivative of overlap_k w.r.t. h[i]
            
            # For h[i], gradient is (1 - h[i+k]) due to chain rule
            for i in range(n_points):
                grad[i] += (1 - h[(i + k) % n_points])
            
        return grad
    
    # Initial C5
    best_C5 = compute_C5(h)
    best_h = h.copy()
    
    # Optimization parameters
    lr = 0.01
    momentum = 0.9
    v = np.zeros_like(h)
    time_start = time.time()
    step = 0
    
    while time.time() - time_start < budget_s - 1:  # Leave 1s margin
        # Compute all overlap values
        h_arr = np.array(h)
        overlaps = np.correlate(h_arr, 1 - h_arr, mode="full")
        # For the full-length correlation, we're interested in the (n_points - 1) to (2*n_points - 2) range
        # But actually, the overlap_k = sum h[i] (1 - h[i+k]) from i=0 to n_points-1
        # The full correlation of len 2*n_points-1 gets k from 0 to 2*n_points-2
        # For k=n_points-1 to 2*n_points-2, k is effectively 0 to n_points-1
        # Actually, x indices are 0 to 2*n_points-2;
        # overlap_k for k=0 to n_points-1, overlap_k index is n_points-1+k  
        overlap_vals = np.array([overlaps[n_points-1+k] for k in range(n_points)])  
        # Now we include only indices where overlap is large, e.g., top 5
        top_k_indices = np.argsort(overlap_vals)[-5:][::-1]  # Top 5 indices
        
        # Compute gradient for these
        grad = compute_gradients(h_arr, top_k_indices)
        
        # Update with momentum
        v = momentum * v - lr * grad
        h = h + v
        
        # Project h
        h = project_h(h)
        
        # Check new C5
        new_C5 = compute_C5(h)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h.copy()
        
        # Adaptive learning rate
        if step % 10 == 0 and step > 0:
            lr = max(0.001, lr * 0.95)
        
        step += 1
    
    return (best_h.tolist(), float(best_C5), int(n_points))
