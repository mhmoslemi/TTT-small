import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1-h, mode="full")
    # Only take the valid lags (from -n+1 to n-1)
    # Due to correlation being longer, we extract the relevant part
    # Correlation is of length 2*n - 1
    # The middle (n-1) values are meaningful for the overlap
    # Actually: let's compute it properly
    # For each k, integral is sum(h[i] * (1 - h[(i+k)%n])) * dx
    # The full correlation gives us all these shifted contributions
    # So we extract the overlap values (from -n+1 to n-1) 
    # and take the maximum
    max_overlap = 0
    for k in range(2*n - 1):
        offset = k - (n - 1)
        if offset == 0:
            # This would be the direct case, not an overlap
            continue
        part = 0.0
        for i in range(n):
            ri = (i + offset) % n
            part += h[i] * (1 - h[ri])
        part *= dx
        if part > max_overlap:
            max_overlap = part
    return max_overlap

def rescale_h(h, n_points):
    """Rescale h to ensure sum(h) = n_points/2"""
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum == 0:
        h = np.ones(n_points) * (target_sum / n_points)
    else:
        h = h * (target_sum / current_sum)
    return h

def project_h(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = rescale_h(h, n_points)
    return h

def compute_gradient(h, n_points, dx, eps=1e-6):
    """Compute finite difference gradient"""
    grad = np.zeros_like(h)
    base_c5 = evaluate_c5(h)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def optimize_step(h, n_points, dx, max_iter=20):
    """Single step optimization using L-BFGS"""
    # Use scipy's minimize with bounds [0,1]
    bounds = [(0, 1) for _ in range(n_points)]
    def objective(x):
        h_new = project_h(x, n_points)
        return evaluate_c5(h_new)
    
    def jac(x):
        # Approximate gradient using finite differences
        return compute_gradient(x, n_points, dx)
    
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=jac, 
                      options={'maxiter': max_iter, 'gtol': 1e-6})
    return project_h(result.x, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 42  # Started from initial_h_values
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Fallback: initialize with uniform values
        h = np.ones(n_points) * 0.5
        h = rescale_h(h, n_points)
        
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Optimization loop
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 2.0:  # Leave 2s buffer
        iteration += 1
        
        # Optimization step
        h_opt = optimize_step(h, n_points, dx, max_iter=10)
        
        # Low-pass filter to smooth
        sigma = 0.1 * n_points  # 0.1 * 42 = 4.2
        h_filtered = gaussian_filter1d(h_opt, sigma=sigma)
        
        # Project to [0,1] and rescale
        h_filtered = project_h(h_filtered, n_points)
        
        # Evaluate
        c5 = evaluate_c5(h_filtered)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_filtered.copy()
            
        # Decide whether to continue
        # Use filtered version for next iteration
        h = h_filtered
        if best_c5 < 0.3807:  # Early stop if target bound reached
            break
    
    return (best_h.tolist(), float(best_c5), n_points)
