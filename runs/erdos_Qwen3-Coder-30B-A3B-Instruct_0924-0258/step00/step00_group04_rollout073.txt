import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for lags
    corr = correlate(h, 1 - h, mode='full')
    # Extract the valid lags: from 0 to n-1 (since we have full mode, we expect full length)
    # The valid range for lags that correspond to meaningful overlaps is [0, n-1]
    # for the specific case of full correlation over 2 intervals, the full correlation
    # is n*2 - 1 elements. The valid lags are in the center of this.
    # However, since h has length n, and we are correlating h with 1-h,
    # we will consider lags from 0 to n-1 as valid ones.
    # To compute overlaps for all possible lags k=0..n-1
    # use the middle part of the correlation where it corresponds to the valid lags
    offset = n - 1  # center of full correlation
    overlaps = corr[offset:offset + n] * dx
    return np.max(overlaps)

def project_h(h, n_points):
    """Project h onto [0,1] and rescale to sum(h) = n_points / 2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.full(n_points, 1.0 / n_points)
    return h

def compute_gradient(h, k_star):
    """Compute gradient vector for worst lag k_star."""
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        i = (j + k_star) % n
        grad[j] = 1 - 2 * h[i]
    return grad

def line_search(h, grad, n_points, dx):
    """Perform backtracking line search to find good step size."""
    alpha = 0.1
    min_alpha = 1e-6
    backtrack_factor = 0.8
    current_c5 = evaluate_c5(h)
    
    while alpha >= min_alpha:
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        if new_c5 < current_c5:
            return alpha
        alpha *= backtrack_factor
    return alpha
