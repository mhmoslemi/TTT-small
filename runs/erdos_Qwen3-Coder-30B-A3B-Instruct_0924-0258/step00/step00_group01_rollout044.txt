import time
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    best_C5 = evaluate_C5(h, dx)

    def get_B_matrix(k):
        B = np.zeros(n_points)
        for i in range(n_points):
            if i + k < n_points:
                B[i] = 1
        return B

    # Optimization parameters
    alpha = 0.1
    patience = 0
    max_patience = 100
    
    # FWHT-based correlation computation
    def compute_overlaps_fwht(h_vec):
        h_fft = fft(h_vec)
        inv_h_fft = fft(1 - h_vec)
        # Compute point-wise product
        cross_fft = h_fft * inv_h_fft.conj()
        # Inverse FFT gives us full correlations
        corr_full = ifft(cross_fft).real
        # Return max over all lags
        return np.max(corr_full) * dx

    def evaluate_C5(h_vec, dx):
        # Use np.correlate for accurate evaluation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    # Gradient-based update using FWHT
    for iter_count in range(10000):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute current overlaps using full correlation
        corr = np.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_overlap = corr[max_k] * dx
        
        # Update best solution
        if max_overlap < best_C5:
            best_C5 = max_overlap
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Early stopping or restart
                break

        # Gradient estimation: approximate
        # For current max_k, compute gradient
        # Simplified gradient for Lagrange relaxation
        grad = np.zeros(n_points)
        k = max_k - (n_points - 1)  # align k with standard indexing
        
        # Compute direction affecting first sample
        for i in range(n_points):
            j = (i + k) % n_points
            if k >= 0:
                grad[i] += (1 - h[j])  # increment this sample for influence on lag k
            else:
                grad[i] -= h[j]  # reverse direction if k is negative
                
        # Apply clipped gradient
        grad = np.clip(grad, -1.0, 1.0)
        h += alpha * grad
        
        # Project onto feasible set [0,1]
        h = np.clip(h, 0, 1)
        
        # Re-scale to satisfy sum constraint
        h *= (n_points / 2) / np.sum(h)

                h += noise
                h = np.clip(h, 0, 1)
                h *= (n_points / 2) / np.sum(h)

    # Final check of solution
    final_C5 = evaluate_C5(best_h, dx)
    assert np.abs(np.sum(best_h) - n_points / 2) < 1e-6, "Sum constraint violated"

    return (list(best_h), float(final_C5), n_points)
