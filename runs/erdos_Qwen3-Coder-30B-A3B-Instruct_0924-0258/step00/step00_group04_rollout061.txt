import numpy as np
import time
from scipy import optimize

def evaluate_C5(h):
    """Evaluate the maximum overlap integral for given h."""
    n = len(h)
    dx = 2.0 / n
    # Use correlation to compute all k values efficiently
    corr = np.correlate(h, 1 - h, mode="full")
    # Take only the valid correlation values (k from -n+1 to n-1)
    # The valid range for k in terms of correlation indices
    # corresponds to the middle part of the full correlation
    mid = len(corr) // 2
    C5_vals = corr[mid - n + 1:mid + n - 1] * dx
    return np.max(C5_vals)

def project_and_rescale(h, n_points):
    """Project h to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.full(n_points, 1.0 / n_points)
    return h

def compute_gradient(h, eps=1e-6):
    """Approximate gradient of C5 w.r.t. h using finite differences."""
    n = len(h)
    grad = np.zeros(n)
    base_C5 = evaluate_C5(h)
    
    for i in range(n):
        h_eps = h.copy()
        h_eps[i] += eps
        h_eps = project_and_rescale(h_eps, n)
        C5_eps = evaluate_C5(h_eps)
        grad[i] = (C5_eps - base_C5) / eps
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42  # Using the initial construction size
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_and_rescale(h, n_points)
    else:
        h = np.full(n_points, 0.5)
        h = project_and_rescale(h, n_points)
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    # Annealing parameters
    temp = 1.0
    min_temp = 0.01
    cooling_rate = 0.99
    patience = 50
    no_improvement_count = 0
    
    # Gradient parameters
    grad_scale = 0.1
    noise_scale = 0.05
    
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < 10000:
        iter_count += 1
        
        # Compute gradient
        grad = compute_gradient(h)
        
        # Create candidate: perturb with noise and gradient
        noise = np.random.normal(0, noise_scale, n_points)
        h_candidate = h + grad_scale * grad + noise
        
        # Project and rescale to satisfy constraints
        h_candidate = project_and_rescale(h_candidate, n_points)
        
        # Evaluate candidate
        new_C5 = evaluate_C5(h_candidate)
        
        # Accept or reject
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_candidate.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update current solution
        h = h_candidate.copy()
        
        # Anneal temperature
        temp = max(temp * cooling_rate, min_temp)
        
        # Reset temperature if no improvement for too long
        if no_improvement_count > patience:
            temp = 1.0
            no_improvement_count = 0
    
    return (list(best_h), float(best_C5), int(n_points))
