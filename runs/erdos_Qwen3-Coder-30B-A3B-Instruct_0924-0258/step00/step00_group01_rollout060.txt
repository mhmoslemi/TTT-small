import numpy as np
import time
from scipy import optimize
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.full(n_points, 0.5)
        
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent
    alpha = 0.1
    momentum = 0.9
    v = np.zeros_like(h)
    last_improvement = 0
    iteration = 0
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute correlations
        correlation = np.correlate(h, 1 - h, mode='full')
        k0 = np.argmax(correlation)
        max_corr = correlation[k0]
        
        # Compute gradient for this shift
        grad = np.zeros_like(h)
        k_offset = k0 - (n_points - 1)
        
        for i in range(n_points):
            if 0 <= i - k_offset < n_points:
                grad[i] -= (1 - h[i - k_offset]) if i - k_offset >= 0 else 0
            if 0 <= i + k_offset < n_points:
                grad[i] -= h[i + k_offset] if i + k_offset < n_points else 0
        
        # Update with momentum
        v = momentum * v - alpha * grad
        h_new = h + v
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_sum_new = np.sum(h_new)
        h_new = h_new * (n_points / 2) / h_sum_new
        
        # Evaluate
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improvement = iteration
        
        # Adaptive step size
        if iteration % 10 == 0 and iteration > last_improvement + 10:
            alpha *= 0.95
            
        # Random restart
        if np.random.rand() < 0.05 or iteration > last_improvement + 50:
            h = best_h.copy() + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            
        h = h_new
        
        # Early termination for stability
        if iteration > last_improvement + 50:
            break
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx
