import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_h(h):
    """Compute the maximum overlap for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # Extract the relevant part (shifts from -n+1 to n-1)
    mid = len(corr) // 2
    max_corr = 0
    for k in range(-n + 1, n):
        # Extract the correct index in the correlation array
        idx = mid + k
        if 0 <= idx < len(corr):
            # Normalize by the number of overlapping elements
            overlap = corr[idx] * dx
            if overlap > max_corr:
                max_corr = overlap
    return max_corr

def project_h(h, n_points):
    """Ensure h is in [0,1] and has correct integral."""
    # Clamp values
    h = np.clip(h, 0, 1)
    # Normalize to sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def mutate_h(h, sigma):
    """Add Gaussian noise and project."""
    n_points = len(h)
    noise = np.random.normal(0, sigma, n_points)
    h_new = h + noise
    return project_h(h_new, n_points)

def crossover_h(h1, h2, alpha=0.6):
    """Blend crossover of two h values."""
    n_points = len(h1)
    beta = np.random.uniform(alpha, 1 - alpha)
    h_new = beta * h1 + (1 - beta) * h2
    return project_h(h_new, n_points)

def optimize_single_h(h0, n_points, max_iter=1000, lr=0.02, decay=0.99):
    """Optimize a single h using gradient descent."""
    h = h0.copy()
    
    def objective(x):
        return evaluate_h(x)
    
    def grad(x):
        # Approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(x)
        for i in range(len(x)):
            x_plus = x.copy()
            x_plus[i] += eps
            x_minus = x.copy()
            x_minus[i] -= eps
            grad[i] = (objective(x_plus) - objective(x_minus)) / (2 * eps)
        return grad
    
    for t in range(max_iter):
        g = grad(h)
        h -= lr * g
        h = project_h(h, n_points)
        lr *= decay  # decay learning rate
    
    return h

    pop_size = 50

    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 10

        fitnesses = [evaluate_h(h) for h in population]
        sorted_indices = np.argsort(fitnesses)
        
        # Select elite
        elite = [population[i] for i in sorted_indices[:elite_size]]

                parent = np.random.choice(parents)
