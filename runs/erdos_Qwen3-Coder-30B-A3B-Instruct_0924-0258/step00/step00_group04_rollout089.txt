import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum value of the convolution scaled by dx
    return np.max(conv) * dx


def project_and_rescale(h, n_points):
    # Project h to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to meet integral constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

    return grad
