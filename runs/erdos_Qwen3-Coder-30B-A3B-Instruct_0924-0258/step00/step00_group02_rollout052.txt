import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral C_5
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # The overlap integral is the maximum value of these correlations over all lags
    max_corr = np.max(corr)
    # multiply by dx to get the actual integral
    return max_corr * dx

def project_h(h, n_points):
    # clip to [0,1]
    h = np.clip(h, 0, 1)
    # rescale sum to be exactly n_points/2
    s = np.sum(h)
    if s > 0:
        h = h * (n_points / 2) / s
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize from provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # construct a simple initial solution
        n_points = 45
        h = np.ones(n_points) * 0.5
    n_points = len(h)
    
    # ensure sum(h) = n_points/2 (this might already be satisfied)
    h = project_h(h, n_points)
    
    # evaluation
    c5_bound = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = c5_bound
    
    # optimization parameters
    learning_rate = 0.01
    decay_rate = 0.9995
    perturb_prob = 0.1
    perturb_magnitude = 0.05
    
    # optimization iterations
    i = 0
    while time.time() - start_time < budget_s:
        i += 1
        # compute overlap for all shifts
        corr = correlate(h, 1 - h, mode='full')
        # max correlation index
        max_k = np.argmax(corr)
        
        # this is an approx gradient, computed as:
        # d/dh_i (h * (1-h))[k] = d/dh_i sum_j h[j]* (1-h[j+k])
        # = (1 - h[j+k]) if j==i, -h[j+k] if j==i+k, 0 elsewhere
        
        # match the shape of matrix manually
        grad = np.zeros_like(h)
        for j in range(len(h)):
            if j + max_k < len(h):
                grad[j] += (1 - h[j + max_k])
            if j - max_k >= 0:
                grad[j] -= h[j - max_k]
        
        # gradient step
        h -= learning_rate * grad
        # project to feasible region
        h = project_h(h, n_points)
        
        # random perturbation occasionally
        if np.random.rand() < perturb_prob:
            delta = np.random.uniform(-perturb_magnitude, perturb_magnitude, size=len(h))
            h += delta
            h = project_h(h, n_points)
        
        # anneal learning rate
        learning_rate *= decay_rate
        
        # evaluate
        c5_bound = evaluate_erdos_solution(h)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h.copy()
        
        # check if we're close to termination
        if time.time() - start_time > budget_s - 2:
            break
    
    return (best_h.tolist(), best_c5, n_points)
