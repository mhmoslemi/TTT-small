import numpy as np
import time
from scipy import signal

def evaluate_overlap(h, dx):
    # Compute overlap integral C5 = max_k int h(x)(1 - h(x+k)) dx
    # Using full correlation to get all lags
    corr = np.correlate(h, 1 - h, mode='full')
    # The correlation values correspond to lags from -n+1 to n-1
    # We only care about non-negative lags (0 to n-1)
    # In np.correlate, the index n-1 corresponds to lag 0
    # index n to n+n-2 = 2n-2 corresponds to lags 1 to n-1
    # We want to maximize over all valid lag indices (0 to n-1), which covers
    # the first n elements starting from index n-1
    c5 = np.max(corr[n-1:])
    return c5 * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 47 if 'initial_h_values' in globals() else 47
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert len(h) == n_points
    else:
        h = np.ones(n_points) * 0.5
    # Adjust to satisfy sum(h)*dx = 1 => sum(h) = n_points / 2
    h *= (n_points / 2) / np.sum(h)
    assert abs(np.sum(h) - n_points / 2) < 1e-10
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    last_improvement = 0
    alpha = 0.1
    iteration = 0
    while time.time() - start_time < budget_s - 0.1:
        # Compute overlaps
        c5 = evaluate_overlap(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration

        # Select the most constraining overlap lag
        corr = np.correlate(h, 1 - h, mode='full')
        lag_0_idx = n_points - 1
        max_val = corr[lag_0_idx]
        max_k = 0
        for k in range(1, n_points):
            if corr[lag_0_idx + k] > max_val:
                max_val = corr[lag_0_idx + k]
                max_k = k
        # Max_k now is the lag index with maximum overlap
        
        # Compute gradient approximation based on differencing
        # This is a reasonable heuristic, not strictly mathematical
        grad = np.zeros(n_points)
        # At each point i, we'll compute how much to change h[i] based on
        # how much h[i] contributes to overlap with h[i+max_k]
        # Approximate gradient:
        # for j in range(n_points):
        #   grad[j] += (1-h[(j+max_k)%n_points]) * 1.0
        #   grad[j] -= h[j] * 1.0 if (j+max_k) < n_points else 0
        
        # More efficient way:
        # For j=0..n-1:
        #   contribution to overlap at lag max_k from h[j] is:
        #   (1-h[(j+max_k)%n_points]) * h[j] 
        #   In cross-correlation notation:
        #   it's contribution to term corr[j + max_k] = h[j]*(1-h[j+max_k]) 
        #   But it's also contribution to term corr[(j-max_k)%n_points] = h[j]*h[j-max_k]
        #   Let's reason: 
        #   corr(k) = sum_i h[i] * (1-h[i+k]) = sum_i h[i]*1 - sum_i h[i]*h[i+k]  
        #                                     = sum_i h[i] - sum_i h[i]*h[i+k]
        #   So contribution from point i to lag k is:
        #   -h[i]*h[i+k] + h[i] = h[i]*(1-h[i+k])
        # So the gradient at i is:
        # -sum_k (h[i]*h[i+k] - h[i]) * delta_k[i,k] where delta is (i+k)%n == j
        # That's complex; let's compute:
        # Grad = sum_{all valid i, k} d/da[i] corr[a(i,k)] * grad_a[i] 
        # But we already use approx m-updt. Let's do gradient more direct:
        # Gradient of corr[l] w.r.t h[i] is:
        # -h[i+l] if (i+l) < n-points else - h[i+l-n_points]  
        # when sum all l indices to get contribution to each i:
        # To maximize for k (the lag we found), directly use the gradient based on how
        # h[i] contributes to that specific overlap:
        # The evaluated overlap is sum_i h[i] * (1 - h[i + max_k])
        # So gradient of overlap with respect to h[i] = (1-h[i+max_k]) - h[i] * 0 (as product)
        # Really, same tuples for both terms
        # We have int h(x) * (1 - h(x + k)) dx
        # So its gradient w.r.t. h[i] is:
        # - (1-h[(i+k)%n_points]) * delta[i][j]
        # Hmm let's compute for the constraint h[i]
        # The overlap at lag k = sum_i h[i] * (1 - h[i + k]) 
        # Gradient w.r.t. h[i] = (1 - h[i+k])
        # Now we have a directional shift - more precise than in usual "update most harmful"
        # Will add gradients for max_k
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_k) % n_points
            grad[i] += (1.0 - h[j])  # negative: hate when large h[i], close to 1
        
        # For the reverse term as well
        # Gradient of overlap w.r.t h[i] for lags i - k:
        # In general, Lag k given sample h[i], contributes (1-h[i+k]) to output k
        # However, if it also appears in different lags, we do consider the net influence
        # Simple approach: eschew vector math and compute it explicitly
        # Gradient w.r.t to point i from :
        # sum h[j] * (1 - h[j + max_k])
        # d/dh[i] = - h[i + max_k]
        # Maybe make a custom one:
        for i in range(n_points):
            j = (i - max_k) % n_points
            grad[i] -= h[j]
        
        # Apply clipping
        grad = np.clip(grad, -1.0, 1.0)
        
        # Update h
        h += alpha * grad
        
        # Project to [0,1]
        h = np.clip(h, 0, 1)
        
        # Project sum to correct value
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-10:
            h *= (n_points / 2) / h_sum
        
        # Adapt stepsize
        if iteration - last_improvement > 50:
            alpha *= 0.95
            last_improvement = iteration
            if alpha < 1e-6:
                alpha = 1e-6
        
        # Probably periodicity etc. - we should fast escape
        if np.random.random() < 0.05:
            # tiny random perturbation
            h += np.random.normal(0, 0.001, n_points)
            h = np.clip(h, 0, 1)
            # renormalize  
            h_sum = np.sum(h)
            h *= (n_points / 2) / h_sum
            
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate_overlap(best_h, dx)
    return (best_h, final_c5, n_points)
