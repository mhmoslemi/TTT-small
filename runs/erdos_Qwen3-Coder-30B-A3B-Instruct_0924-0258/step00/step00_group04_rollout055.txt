import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_c5(h_values):
    """Evaluate C5 for given h_values"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode="full")
    # Extract the part corresponding to valid overlaps
    middle = conv[len(h)-1 : 2*len(h)-1]
    # Maximum value scaled by dx
    max_val = np.max(middle) * dx
    return max_val

def project_and_rescale(h, n_points):
    """Ensure h sums to n_points/2 and stays in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2):
    """Perform crossover between two individuals"""
    n_points = len(parent1)
    # Randomly choose crossover points (2-3 points)
    num_points = random.randint(2, 3)
    points = sorted(random.sample(range(1, n_points), num_points))
    points = [0] + points + [n_points]
    
    # Create offspring by alternating segments
    offspring = []
    for i in range(len(points)-1):
        if i % 2 == 0:
            offspring.extend(parent1[points[i]:points[i+1]])
        else:
            offspring.extend(parent2[points[i]:points[i+1]])
    
    return np.array(offspring)

def mutate(h, mut_rate=0.05):
    """Mutate the individual with Gaussian noise"""
    n_points = len(h)
    indices = random.sample(range(n_points), int(n_points * mut_rate))
    for i in indices:
        h[i] += np.random.normal(0, 0.1)
    return project_and_rescale(h, len(h))

def local_optimize(h, n_points):
    """Apply local optimization to improve solution"""
    def objective(x):
        return evaluate_c5(x)
    
    # Use scipy to locally optimize the best individuals
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective, h, bounds=bounds, method='L-BFGS-B')
    return project_and_rescale(result.x, n_points)

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 42  # Start with the provided initial solution size
    pop_size = 50  # Population size
    max_generations = 200
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():

        h0 = np.array(initial_h_values)
        h0 = project_and_rescale(h0, n_points)
        population.append(h0)
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = project_and_rescale(h, n_points)
            population.append(h)
    else:
        # Initialize entirely randomly
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = project_and_rescale(h, n_points)
            population.append(h)
            
    best_h = None
    best_c5 = float('inf')
    
    # Loop through generations
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness (ascending, so smaller C5 is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 20% as parents
        num_parents = max(1, pop_size // 5)
        parents = population[:num_parents]
        
        # Generate offspring
        offspring = []
        
        # Crossover
        for _ in range(pop_size - num_parents):
            parent1, parent2 = random.sample(parents, 2)
            child = crossover(parent1, parent2)
            child = mutate(child, mut_rate=0.05)
            offspring.append(child)
            
        # Add children to population
        population = parents + offspring
        
        # Local optimization on top 10% periodically
        if gen % 10 == 0:
            num_top = max(1, pop_size // 10)
            for i in range(num_top):
                h = local_optimize(population[i], n_points)
                population[i] = h
        
        # Early stopping if no improvement in last 50 generations
        if gen > 50 and fitness[0] == fitness[50]:
            # Increase mutation rate a bit
            # (this is handled by higher mutation probability in mutate)
            pass

    # Final evaluation
    final_fitness = []
    for h in population:
        c5 = evaluate_c5(h)
        final_fitness.append(c5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return (list(best_h), best_c5, n_points)
