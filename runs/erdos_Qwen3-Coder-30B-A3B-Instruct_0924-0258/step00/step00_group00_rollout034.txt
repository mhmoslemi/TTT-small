import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5  # Simple uniform guess
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    
    # Frank-Wolfe algorithm
    max_iter = 500
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all cross-correlations
        corr_full = correlate(h, 1 - h, mode='full')
        # The maximum lag is at the center of the full correlation
        mid = len(corr_full) // 2
        k_max = np.argmax(corr_full[mid - n_points//2:mid + n_points//2 + 1]) - n_points//2 + mid
        max_corr = corr_full[k_max]
        
        # Gradient of the max correlation term
        # For cross-correlation at lag k_max: sum_i h[i] * (1 - h[i + k_max])
        # Gradient w.r.t. h[i] is (1 - h[i + k_max]) if i + k_max < n_points and 0 otherwise
        gradient = np.zeros_like(h)
        if k_max >= 0:
            gradient[:-k_max] = 1 - h[k_max:]
        else:
            gradient[-k_max:] = 1 - h[:k_max]
        
        # Update step
        # Select top S indices with highest gradients
        S = n_points // 2
        indices = np.argsort(gradient)[::-1][:S]
        s = np.zeros_like(h)
        s[indices] = 1.0
        
        # Line search on step size γ
        gamma_best = 0.0
        c5_best = best_c5
        
        for gamma in [0.05, 0.1, 0.25, 0.5, 0.75, 1.0]:
            h_new = h + gamma * (s - h)
            # Project back to feasible set
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5 = evaluate_c5(h_new, dx)
            if c5 < c5_best:
                c5_best = c5
                gamma_best = gamma
                
        # Advance with best gamma
        h = h + gamma_best * (s - h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h.copy()
    
    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_c5(h, dx):
    # Compute the maximum cross-correlation
    corr_full = correlate(h, 1 - h, mode='full')
    L = len(corr_full)
    mid = L // 2
    # Only consider lags within valid range
    valid_lags = corr_full[mid - len(h) + 1 : mid + len(h)]
    max_corr = np.max(valid_lags)
    return max_corr * dx
