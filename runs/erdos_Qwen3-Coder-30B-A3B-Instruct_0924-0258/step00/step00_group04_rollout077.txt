import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_c5(h):
    # This matches the problem statement's evaluation method
    n = len(h)
    dx = 2.0 / n
    # correlate h with (1-h) with full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # take the maximum over all lags (middle section)
    max_corr = np.max(corr[n-1:2*n-1])
    return max_corr * dx

def project_onto_feasible(h, n_points):
    """Project h to satisfy constraints: 0≤h≤1 and sum(h)=n_points/2."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.full(n_points, 1.0 / 2.0)
    return h

def finite_diff_grad(h, func, eps=1e-5):
    """Approximate gradient using finite differences."""
    n = len(h)
    grad = np.zeros_like(h)
    h_orig = h.copy()
    for i in range(n):
        h[i] += eps
        fp = func(h)
        h[i] -= 2*eps
        fm = func(h)
        grad[i] = (fp - fm) / (2 * eps)
        h[i] = h_orig[i]  # restore
    return grad

def local_optimize(h, n_points, max_iter=20):
    """Perform local optimization step."""
    def objective(h_vec):
        return evaluate_c5(h_vec)
    
    def callback(x):
        pass  # Can be used to log progress
    
    # Use L-BFGS-B with finite differences
    bounds = [(0, 1) for _ in range(n_points)]
    res = minimize(
        fun=objective,
        x0=h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iter, 'disp': False},
        callback=callback,
        jac=finite_diff_grad
    )
    if res.success:
        return res.x
    else:
        return h
