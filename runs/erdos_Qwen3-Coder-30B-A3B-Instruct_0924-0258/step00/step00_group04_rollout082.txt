import numpy as np
from scipy.optimize import dual_annealing, minimize
from scipy.signal import correlate
import time

def evaluate_C5(h):
    """Evaluate C5 for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlations for all lags
    corr = correlate(h, 1 - h, mode='full')
    # Extract the relevant part (positive lags)
    full_corr = corr[len(corr)//2:]
    # Compute the maximum value scaled by dx
    max_corr = np.max(full_corr) * dx
    return max_corr

def project_and_rescale(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points / 2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, n_points, rate=0.05):
    """Apply Gaussian noise with specified rate of mutation."""
    indices = np.random.choice(n_points, int(n_points * rate), replace=False)
    noise = np.random.normal(0, 0.1, len(indices))
    h_new = h.copy()
    h_new[indices] += noise
    return project_and_rescale(h_new, n_points)

def crossover(h1, h2, n_points):
    """Perform uniform crossover with 2-3 random breakpoints."""
    n_breakpoints = np.random.randint(2, 4)
    breakpoints = sorted(np.random.choice(n_points-1, n_breakpoints, replace=False))
    breakpoints = [0] + breakpoints + [n_points]
    h_new = np.zeros(n_points)
    for i in range(len(breakpoints)-1):
        start, end = breakpoints[i], breakpoints[i+1]
        if i % 2 == 0:
            h_new[start:end] = h1[start:end]
        else:
            h_new[start:end] = h2[start:end]
    return project_and_rescale(h_new, n_points)

def greedy_optimize(h, n_points, maxiter=50):
    """Apply local optimization with bounds to improve candidate."""
    dx = 2.0 / n_points
    def objective(h_vec):
        h_local = h_vec.reshape(-1)
        return evaluate_C5(h_local)
    def gradient(h_vec):
        # Simple finite difference gradient for optimization
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2*eps)
        return grad
    bounds = [(0, 1) for _ in range(n_points)]
    # Use differential evolution as a fast optimization method
    result = dual_annealing(objective, bounds, maxiter=50, seed=42)
    return project_and_rescale(result.x.reshape(-1), n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Reasonable size for optimization
    pop_size = 50
    max_gens = 200
    mutation_rate = 0.05
    top_k = pop_size // 5
    best_h = None
    best_C5 = float('inf')
    
    # Initialize population
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.random.rand(n_points)
    
    h0 = project_and_rescale(h0, n_points)
    
    population = [h0]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_and_rescale(h, n_points)
        population.append(h)
    
    for generation in range(max_gens):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all candidates
        fitness = []
        for h in population:
            c5 = evaluate_C5(h)
            fitness.append((c5, h))
        
        # Sort by C5 value
        fitness.sort(key=lambda x: x[0])
        if fitness[0][0] < best_C5:
            best_C5 = fitness[0][0]
            best_h = fitness[0][1].copy()
            
        # Select top candidates
        top_candidates = [h for _, h in fitness[:top_k]]
        
        # Apply greedy optimization to top candidates
        for h in top_candidates[:top_k//2]:
            if time.time() - start_time > budget_s - 5:
                break
            optimized = greedy_optimize(h, n_points)
            best_C5 = min(best_C5, evaluate_C5(optimized))
            if evaluate_C5(optimized) < best_C5:
                best_C5 = evaluate_C5(optimized)
                best_h = optimized.copy()
        
        # Generate new population
        new_population = top_candidates[:]
        while len(new_population) < pop_size:
            if time.time() - start_time > budget_s - 5:
                break
            # Crossover
            parents = np.random.choice(top_candidates, 2, replace=False)
            child = crossover(parents[0], parents[1], n_points)
            # Mutation
            if np.random.rand() < 0.5:
                child = mutate(child, n_points, mutation_rate)
            new_population.append(child)
            
        population = new_population[:pop_size]
    
    # Final check with local solver
    if best_h is not None:
        final_h = greedy_optimize(best_h, n_points)
        final_C5 = evaluate_C5(final_h)
        if final_C5 < best_C5:
            best_C5 = final_C5
            best_h = final_h
            
    return (best_h.tolist(), best_C5, n_points)
