import numpy as np
import time
from scipy import signal

def evaluate(h, dx):
    """Evaluate the maximum overlap integral for given step function h."""
    # Compute full correlation: h * (1 - h) with mode='full'
    corr = signal.correlate(h, 1 - h, mode='full')
    # Only consider valid lags (from -n+1 to n-1)
    n = len(h)
    # Take only the middle part corresponding to valid overlaps
    mid = len(corr) // 2
    valid_corr = corr[mid - n + 1:mid + n - 1]
    # Compute actual overlap values by scaling with dx
    overlaps = valid_corr * dx
    return np.max(overlaps)

def rescale(h, target_sum):
    """Rescale h to have sum equal to target_sum while keeping bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Avoid division by zero
    if abs(current_sum) < 1e-10:
        return np.full_like(h, target_sum / len(h))
    scaled = h * (target_sum / current_sum)
    # Clip to [0,1] bounds and rescale again if needed
    clipped = np.clip(scaled, 0, 1)
    rescaled = clipped * (target_sum / np.sum(clipped))
    return rescaled

def init_population(initial_h, n_points, pop_size=50):
    """Initialize population with the parent and random variations."""
    population = [initial_h]
    target_sum = n_points / 2.0
    
    # Add some mutated versions of the parent
    for _ in range(pop_size - 1):
        # Slightly perturb
        mutant = initial_h + np.random.uniform(-0.2, 0.2, n_points)
        # Clip and rescale to maintain constraint
        mutant = np.clip(mutant, 0, 1)
        mutant = rescale(mutant, target_sum)
        population.append(mutant)
    
    return population

def crossover(parent1, parent2):
    """One-point crossover between two parents."""
    n = len(parent1)
    point = np.random.randint(1, n - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.1, delta=0.1):
    """Mutate an individual by adding uniform noise."""
    n = len(individual)
    mutated = individual.copy()
    for i in range(n):
        if np.random.rand() < mutation_rate:
            mutated[i] += np.random.uniform(-delta, delta)
    # Clip to [0,1]
    mutated = np.clip(mutated, 0, 1)
    # Rescale to maintain sum constraint
    target_sum = n / 2.0
    mutated = rescale(mutated, target_sum)
    return mutated

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select an individual via tournament selection."""
    selected_indices = np.random.choice(len(population), tournament_size)
    best_idx = selected_indices[np.argmin([fitnesses[i] for i in selected_indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default to 45 sample points (from initial_h_values if available)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initial construction could be in global scope as `initial_h_values`
    try:
        initial_h = initial_h_values
        # Validate it
        if len(initial_h) != n_points:
            raise ValueError("Initial h has wrong length.")
        assert np.all((initial_h >= 0) & (initial_h <= 1))
        # Make sure it sums to n_points/2
        assert abs(np.sum(initial_h) - n_points/2) < 1e-6
        # if validation checks pass, we use it
    except NameError:
        # If not available, construct a simple baseline
        initial_h = np.ones(n_points) * 0.5
        initial_h = rescale(initial_h, n_points / 2)
    
    # Population management
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    
    population = init_population(initial_h, n_points, pop_size)
    
    best_h = None
    best_c5 = float('inf')
    
    # Main GA loop
    generation = 0
    max_generations = 200  # Adaptive based on time budget - 600s allowed
    
    while time.time() - start_time < budget_s - 0.1 and generation < max_generations:
        # Evaluate fitness for all candidates
        fitnesses = []
        for h in population:
            c5 = evaluate(h, dx)
            fitnesses.append(c5)
            
        # Update best solution found
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
            
        # Selection and reproduction
        new_population = []
        
        # Elitism: Keep best individual
        best_idx = np.argmin(fitnesses)
        new_population.append(population[best_idx].copy())
        
        # Fill rest of population
        while len(new_population) < pop_size:
            # Tournament selection
            p1 = tournament_selection(population, fitnesses)
            p2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                c1, c2 = crossover(p1, p2)
            else:
                c1, c2 = p1.copy(), p2.copy()
                
            # Mutation
            if np.random.rand() < 0.5:
                c1 = mutate(c1, mutation_rate, 0.1)
            if np.random.rand() < 0.5:
                c2 = mutate(c2, mutation_rate, 0.1)
                
            new_population.extend([c1, c2])
            
        # Trim to fixed pop_size
        population = new_population[:pop_size]
        generation += 1
    
    # Make sure best_h is correct for output
    if best_h is None:
        # Fallback: just return initial
        best_h = initial_h.copy()
        best_c5 = evaluate(best_h, dx)
    else:
        # Ensure it's properly rescaled
        best_h = np.clip(best_h, 0, 1)
        best_h = rescale(best_h, n_points / 2)
        best_c5 = evaluate(best_h, dx)
    
    return (best_h, best_c5, n_points)
