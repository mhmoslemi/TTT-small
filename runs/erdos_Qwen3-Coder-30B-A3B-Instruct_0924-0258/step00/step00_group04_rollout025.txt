def evaluate_overlap(h, dx):
    # Compute overlap integral for all lags
    n = len(h)
    # Use correlation with flipped h to get the overlap
    corr = np.correlate(h, 1 - h, mode='full')
    # Extract the valid range (same as the lag range)
    half_len = n - 1
    overlap_values = corr[n-1:n-1+half_len] * dx
    return np.max(overlap_values)
