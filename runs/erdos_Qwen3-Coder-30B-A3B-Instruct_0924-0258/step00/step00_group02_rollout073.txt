import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    h_pad = np.concatenate([h, np.zeros(n)])
    h_rev = np.concatenate([np.zeros(n), 1 - h])
    conv = np.convolve(h_pad, h_rev, mode='valid')
    max_val = np.max(conv) * dx
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial setup
    if 'initial_h_values' in globals():

        n_base = len(h_base)
    else:
        n_base = 45
        h_base = np.ones(n_base) * 0.5
    
    # Resolutions to try
    resolutions = [15, 30, 45, 60, 75]
    
    # Best solution tracking
    best_h = None
    best_c5 = float('inf')
    
    def surrogate_objective(h, n):
        dx = 2.0 / n
        h_pad = np.concatenate([h, np.zeros(n)])
        h_rev = np.concatenate([np.zeros(n), 1 - h])
        conv = np.convolve(h_pad, h_rev, mode='valid')
        # Use quadratic surrogate
        return np.sum(conv**2) * dx**2
    
    def jacobian(h, n):
        dx = 2.0 / n
        # Compute gradient of surrogate
        h_pad = np.concatenate([h, np.zeros(n)])
        h_rev = np.concatenate([np.zeros(n), 1 - h])
        conv = np.convolve(h_pad, h_rev, mode='valid')
        # This is a simplified gradient; actual computation is complex
        # but we'll take a simple approximation
        grad = np.zeros_like(h)
        for i in range(n):
            h_shifted = h.copy()
            h_shifted[i] += 1e-5
            h_pad_shifted = np.concatenate([h_shifted, np.zeros(n)])
            h_rev_shifted = np.concatenate([np.zeros(n), 1 - h_shifted])
            conv_shifted = np.convolve(h_pad_shifted, h_rev_shifted, mode='valid')
            grad[i] = (np.sum(conv_shifted**2) - np.sum(conv**2)) / (1e-5 * dx**2)
        return grad
    
    # Optimization parameters
    max_perturbations = 3
    max_iters_per_level = 100
    idle_count = 0
    
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s - 1:
            break
            
        # Initialize h for this resolution
        if n_points <= n_base:
            h = h_base[:n_points].copy()
        else:
            # Interpolate
            x_old = np.linspace(0, 2, n_base)
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_old, h_base)
        
        # Normalize to sum to n_points/2
        h *= (n_points / 2) / np.sum(h)
        
        # Perturbations
        for _ in range(max_perturbations):
            # Create initial guess
            current_h = h.copy()
            
            # Optimize with bounds
            def obj(h_vec):
                return surrogate_objective(h_vec, n_points)
                
            def jac(h_vec):
                return jacobian(h_vec, n_points)
                
            # Bounds
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Constraints
            def constraint_sum(h_vec):
                return np.sum(h_vec) - n_points/2
                
            constraints = {'type': 'eq', 'fun': constraint_sum}
            
            # Optimize with SLSQP
            try:
                res = minimize(obj, current_h, method='SLSQP', jac=jac, bounds=bounds,

                if res.success:
                    h_opt = res.x
                else:
                    h_opt = current_h.copy()
            except:
                h_opt = current_h.copy()
                
            # Feasibility projection
            h_opt = np.clip(h_opt, 0, 1)
            h_opt *= (n_points / 2) / np.sum(h_opt)
            
            # Evaluate
            c5 = evaluate_erdos_solution(h_opt)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                idle_count = 0
            else:
                idle_count += 1
                
            if idle_count > 20 or time() - start_time > budget_s - 1:
                break
                
            # Perturb
            noise = 0.02 * np.random.rand(n_points)
            h = np.clip(h_opt + noise, 0, 1)
            h *= (n_points / 2) / np.sum(h)
            
    if best_h is None:
        # Default fallback
        best_h = h_base[:45].copy() if 'initial_h_values' in globals() else np.ones(45) * 0.5
        best_h *= (45 / 2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h)
        
    return (best_h.tolist(), float(best_c5), len(best_h))
