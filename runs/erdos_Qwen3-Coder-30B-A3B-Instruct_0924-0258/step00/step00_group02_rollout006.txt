import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a reasonable starting point
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    lr = 0.01
    momentum = 0.9
    v = np.zeros_like(h)
    t0 = time.time()
    
    # Try to improve the solution
    iteration = 0
    while time.time() - t0 < budget_s - 0.1 and iteration < 6000:
        # Compute all overlaps with full convolution
        overlap_full = correlate(h, 1 - h, mode='full')
        # Only consider valid k values (0 to n_points-1)
        overlap = overlap_full[n_points-1:n_points-1+n_points]
        # We want to minimize max_k of these overlap integrals
        # Therefore we care about the gradient of the maximum
        k_max = np.argmax(overlap)
        dominant_k = [k_max]
        # Also consider the top few largest overlaps
        top_k = np.argsort(overlap)[-5:][::-1]
        dominant_k = list(set(dominant_k + top_k.tolist()))[:10]
        
        # Compute gradient for dominant k components
        grad = np.zeros_like(h)
        for k in dominant_k:
            # Differentiate: d/dh_i of integral h(x)(1-h(x+k))dx
            # = (1 - h(x+k))  for i = x,
            # = -h(x) (where x = i - k) for i = x+k
            # So for k-th derivative, we get:
            # grad[i] += (1 - h[(i+k)%n_points]) - h[i]
            # This needs to consider circular indexing
            shifted_h = np.roll(h, -k)
            grad += (1 - shifted_h) - h
            
        # Update with momentum + gradient descent
        v = momentum * v + lr * grad
        h = h - v
        
        # Project to feasible region: [0,1] and sum constraint
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
            
        # Evaluate
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Adjust learning rate
        if iteration % 100 == 0 and iteration > 0:
            lr = max(lr * 0.95, 1e-6)
            
        iteration += 1
        
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute the full convolution
    conv = correlate(h, 1 - h, mode='full')
    # Extract the middle part corresponding to valid k values [0, n-1]
    n = len(h)
    overlap_vals = conv[n-1:n-1+n]
    return np.max(overlap_vals) * dx
