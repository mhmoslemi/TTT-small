import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # Take the maximum correlation (excluding trivial zeros)
    max_corr = np.max(corr[len(h)-1:])
    return max_corr * dx

def project_and_rescale(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points / 2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2, n_points):
    """Perform crossover between two parents."""
    # Select crossover points
    n_crossover = random.randint(2, 3)
    crossover_points = sorted(random.sample(range(1, n_points), n_crossover))
    
    # Create offspring by combining parent segments
    offspring = np.zeros(n_points)
    start = 0
    for i, point in enumerate(crossover_points):
        if i % 2 == 0:
            offspring[start:point] = parent1[start:point]
        else:
            offspring[start:point] = parent2[start:point]
        start = point
    
    # Handle last segment
    if len(crossover_points) % 2 == 0:
        offspring[start:] = parent1[start:]
    else:
        offspring[start:] = parent2[start:]
    
    return project_and_rescale(offspring, n_points)

def mutate(h, n_points, mutation_rate=0.05):
    """Apply mutation to the given vector."""
    # Mutate a fraction of elements
    n_mutate = max(1, int(n_points * mutation_rate))
    indices = random.sample(range(n_points), n_mutate)
    for idx in indices:
        h[idx] += np.random.normal(0, 0.1)
    
    return project_and_rescale(h, n_points)

def greedy_optimize(h, n_points):
    """Apply local optimization to a candidate solution."""
    def objective(h_flat):
        h_vec = h_flat.copy()
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to correct sum
        current_sum = np.sum(h_vec)
        target_sum = n_points / 2
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return evaluate_c5(h_vec)
    
    # Run a local optimization
    bounds = [(0, 1)] * n_points
    res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
    h_opt = res.x.copy()
    h_opt = np.clip(h_opt, 0, 1)
    current_sum = np.sum(h_opt)
    if current_sum > 0:
        h_opt = h_opt * (n_points / 2) / current_sum
    return h_opt

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    pop_size = 50
    max_generations = 200
    stagnation_limit = 50
    best_c5 = float('inf')
    best_h = None
    stagnation_counter = 0
    mutation_rate = 0.05
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_and_rescale(h0, n_points)
        population.append(h0)
    else:
        # Create a random candidate, ensuring sum = n_points / 2
        h0 = np.random.rand(n_points)
        h0 = project_and_rescale(h0, n_points)
        population.append(h0)
        
    # Fill up the initial population
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = project_and_rescale(h, n_points)
        population.append(h)
    
    # Main loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all candidates
        fitness = [evaluate_c5(h) for h in population]
        sorted_indices = np.argsort(fitness)
        
        # Track best
        if fitness[sorted_indices[0]] < best_c5:
            best_c5 = fitness[sorted_indices[0]]
            best_h = population[sorted_indices[0]].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # If stagnated, increase mutation rate
        if stagnation_counter > stagnation_limit:
            mutation_rate = min(0.1, mutation_rate * 1.1)
            
        # Parents are top 20%
        parents = [population[i] for i in sorted_indices[:max(1, pop_size // 5)]]
        
        # Generate new offspring
        offspring = []
        while len(offspring) < pop_size // 2:
            # Select two parents (with replacement)
            parent1 = random.choice(parents)
            parent2 = random.choice(parents)
            
            # Crossover
            child = crossover(parent1, parent2, n_points)
            
            # Mutate
            child = mutate(child, n_points, mutation_rate)
            
            offspring.append(child)
            
        # Apply greedy optimization to top candidates
        top_indices = sorted_indices[:max(1, pop_size // 10)]
        for i in top_indices:
            if time.time() - start_time > budget_s - 1:
                break
            population[i] = greedy_optimize(population[i], n_points)
        
        # Update population
        population = parents + offspring
        
        # Re-evaluate (as mutation has changed values slightly)
        population = [project_and_rescale(h, n_points) for h in population]
    
    # Final evaluation of best solution
    if best_h is None:
        # Fallback to random solution if no progress was made
        best_h = np.random.rand(n_points)
        best_h = project_and_rescale(best_h, n_points)
        best_c5 = evaluate_c5(best_h)
    else:
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
