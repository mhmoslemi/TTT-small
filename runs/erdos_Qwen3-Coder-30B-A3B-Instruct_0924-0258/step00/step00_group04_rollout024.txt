import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    # Compute all lag overlaps
    z = correlate(h, 1 - h, mode='full')
    # Extract the valid overlaps (corresponding to lags 0 to n_points-1)
    lag_offset = n_points - 1
    z_k = z[lag_offset:lag_offset + n_points] * dx
    return np.max(z_k)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize h: either use initial_h_values or construct starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert np.all(h >= 0) and np.all(h <= 1)
        assert abs(np.sum(h) * dx - 1.0) < 1e-6
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure sum constraint is exact
    h = h * (n_points / 2.0) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    start_time = time.time()
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Compute overlaps for all lags k
        z = correlate(h, 1 - h, mode='full')
        lag_offset = n_points - 1
        z_k = z[lag_offset:lag_offset + n_points] * dx
        k_star = np.argmax(z_k)
        
        # Compute gradient
        g = np.zeros(n_points)
        for j in range(n_points):
            idx = (j + k_star) % n_points
            g[j] = 1 - 2 * h[idx]
        
        # Normalize gradient
        norm_g = np.linalg.norm(g)
        if norm_g > 1e-10:
            g = g / norm_g
        else:
            g = np.zeros_like(g)
        
        # Line search
        alpha = 0.1
        backtrack_factor = 0.8
        max_backtrack = 10
        for _ in range(max_backtrack):
            h_new = h - alpha * g
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to meet integral constraint
            h_new = h_new * (n_points / 2.0) / np.sum(h_new)
            new_c5 = evaluate_erdos_solution(h_new, n_points)
            if new_c5 < best_c5:
                h = h_new.copy()
                best_c5 = new_c5
                best_h = h.copy()
                break
            elif alpha < 1e-6:
                break
            else:
                alpha *= backtrack_factor
        
        iter_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
