import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 10000
    perturb_steps = 20
    alpha_candidates = [0.1, 0.5, 1.0]
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Ensure constraint sum(h) * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Compute all cross-correlations
        h_rev = np.flip(h)
        corr_full = correlate(h, h_rev, mode='full')
        corr_full = corr_full[len(corr_full)//2:]
        corr_full = corr_full * dx
        
        corr_full_neg = correlate(h, np.ones_like(h) - h, mode='full')
        corr_full_neg = corr_full_neg[len(corr_full_neg)//2:]
        corr_full_neg = corr_full_neg * dx
        
        # Effective correlation for maximum overlap
        # For h(x) we compute int h(x)(1-h(x+k))dx, this is equivalent to
        # correlating h with (1-h) using the right-side correlation
        corr = corr_full_neg
        
        # Find maximum correlation
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        
        # Find gradient for this lag
        k = max_k - (len(corr) - 1) // 2  # convert index back to actual lag
        gradient = np.zeros(n_points)
        if k >= 0:
            gradient[k:] = 1 - h[:-k] if k < n_points else np.zeros(0)
        else:
            gradient[:k] = 1 - h[-k:] if -k < n_points else np.zeros(0)
        
        # Gradient descent step
        candidate_h = None
        best_candidate_c5 = best_c5
        
        for alpha in alpha_candidates:
            h_candidate = h - alpha * gradient
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Renormalize
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            candidate_c5 = compute_c5(h_candidate, dx)
            if candidate_c5 < best_candidate_c5:
                best_candidate_c5 = candidate_c5
                candidate_h = h_candidate.copy()
        
        # If we found a better point, accept it
        if candidate_h is not None:
            h = candidate_h
            if best_candidate_c5 < best_c5:
                best_c5 = best_candidate_c5
                best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
            if no_improvement_count >= perturb_steps:
                # Add random noise and re-normalize
                h += 0.01 * np.random.randn(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                no_improvement_count = 0
                
        iter_count += 1
        
    # Final validation
    best_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    n_points = len(h) 
    h_rev = np.flip(h) 
    corr_full_neg = correlate(h, np.ones_like(h) - h, mode='full')
    corr_full_neg = corr_full_neg[len(corr_full_neg)//2:]
    corr_full_neg = corr_full_neg * dx
    return np.max(corr_full_neg)
