import numpy as np
import time
from scipy import correlate

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral C5 for given step function h."""
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The correlation indices correspond to lags from -len(h)+1 to len(h)-1
    # The maximum overlap integral corresponds to the maximum of the correlation
    # We extract the relevant part which corresponds to lag k from 0 to len(h)-1
    # C5 = max_k of integral of h(x)(1-h(x+k)) dx
    # For discrete case: C5 = max_k cross_correlation(k) * dx

    # The correlation array has length 2*len(h)-1
    # We take the part from position len(h)-1 to 2*len(h)-2 to get lags from 0 to len(h)-1
    corr_positive_lags = corr[len(h)-1:]
    return np.max(corr_positive_lags) * dx

def project_h_to_feasible(h, dx):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h)*dx = 1."""
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale h so that sum(h)*dx = 1
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * (1.0 / (sum_h * dx))
    # Clip again if necessary after rescaling
    h = np.clip(h, 0, 1)
    return h

def compute_dx(n_points):
    """Compute dx based on n_points."""
    return 2.0 / n_points

def run(seed=42, budget_s=600, **kwargs):

    # Set random seed
    np.random.seed(seed)
    
    # Parameters for PSO
    n_particles = 30
    n_points = 45
    max_iter = 1000
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    
    # Initialize h if available, otherwise create a random one

    else:

        h_global = np.random.rand(n_points)
        h_global = project_h_to_feasible(h_global, compute_dx(n_points))
    
    # Store best h found
    best_h = h_global.copy()
    best_c5 = evaluate_c5(best_h, compute_dx(n_points))
    
    # Initialize particles
    h_particles = []
    v_particles = []
    best_h_particles = []
    best_c5_particles = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            # Start with initial solution and perturb it slightly
            h = h_global + np.random.normal(0, 0.01, n_points)
            h = project_h_to_feasible(h, compute_dx(n_points))
        else:
            h = np.random.rand(n_points)
            h = project_h_to_feasible(h, compute_dx(n_points))
        h_particles.append(h.copy())
        v_particles.append(np.zeros(n_points))
        best_h_particles.append(h.copy())
        best_c5_particles.append(evaluate_c5(h, compute_dx(n_points)))
    
    # PSO iterations
    start_time = time.time()
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s and no_improvement_count < max_no_improvement:
        iter_count += 1
        for i in range(n_particles):
            # Update particle velocity and position
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            # Update velocity
            v_particles[i] = (inertia * v_particles[i] 
                             + cognitive * r1 * (best_h_particles[i] - h_particles[i]) 
                             + social * r2 * (best_h - h_particles[i]))
            
            # Update position
            h_particles[i] = h_particles[i] + v_particles[i]
            
            # Apply constraints
            h_particles[i] = project_h_to_feasible(h_particles[i], compute_dx(n_points))
            
            # Evaluate fitness
            c5 = evaluate_c5(h_particles[i], compute_dx(n_points))
            
            # Update particle best
            if c5 < best_c5_particles[i]:
                best_h_particles[i] = h_particles[i].copy()
                best_c5_particles[i] = c5
                
            # Update global best
            if c5 < best_c5:
                best_h = h_particles[i].copy()
                best_c5 = c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
        # Early stopping if no improvement
        if no_improvement_count >= max_no_improvement:
            break
    
    return (best_h, best_c5, n_points)
