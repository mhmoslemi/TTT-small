import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, n_points):
    # Compute the overlap integral C5
    dx = 2.0 / n_points
    # The overlap integral is the maximum of correlations
    # np.correlate with mode="full" computes all lags
    corr = correlate(h, 1 - h, mode="full")
    # The max value is at the middle of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_feasible_region(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to ensure sum(h) = n_points / 2
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    adjustment = (target_sum - current_sum) / n_points
    
    # Add adjustment to each point to get exactly target_sum
    h += adjustment
    
    # Ensure the sum is exactly target_sum
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Distribute the difference
        h += diff / n_points
        
    return np.clip(h, 0, 1)

def compute_gradient(h, n_points, max_lags):
    # Compute contribution to gradient from all max lags
    dx = 2.0 / n_points
    g = np.zeros(n_points)
    
    # For each lag in max_lags,
    # compute contribution of this lag to gradient
    for k in max_lags:
        # Compute based on convolution-like formula
        # For h[i], contribution from 1-h[i+k] with no wraparound
        for i in range(n_points):
            j = (i + k) % n_points
            # Each term is: (1 - h[j]) - (1 - h[i]) = h[i] - h[j]
            g[i] += (h[i] - h[j]) * dx
    
    return g

    best_C5 = evaluate_erdos_solution(h, n_points)
    best_h = h.copy()

        # Check time
        if time.time() - start_time > budget_s - 1:

        # Compute overlap integrals for all lags
        corr = correlate(h, 1 - h, mode="full")
        max_val = np.max(corr)
        
        # Identify lags achieving maximum C5
        max_lags = np.where(corr == max_val)[0]

        max_lags = max_lags - (n_points - 1)
        
        # Compute gradient
        g = compute_gradient(h, n_points, max_lags)
        
        # Update
        h_new = h - alpha * g
        h_new = project_onto_feasible_region(h_new, n_points)
        
        # Check new value
        new_C5 = evaluate_erdos_solution(h_new, n_points)

        if new_C5 < best_C5:

            improvement = (best_C5 - new_C5) / best_C5
