    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlap(h_vec):
        # Compute integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Only consider valid overlaps (covers all shifts)
        half_len = len(h_vec) - 1
        overlaps = corr[half_len:] * dx
        return np.max(overlaps)

    def project_h(h_vec):
        # Clamp values to [0, 1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to meet sum constraint
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec

    def gradient_step(h_vec, k_max):
        # Compute gradient of correlation at k_max
        # Using finite difference-like approach
        # We compute the gradient to decrease overlap at k_max
        grad = np.zeros_like(h_vec)
        n = len(h_vec)
        
        # Simpler approximation of gradient via finite differences
        # Gradient at position i
        for i in range(n):
            # Approximate the derivative of C5 wrt h[i]
            # This is a rough approximation by shifting h slightly
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            eps = 1e-5
            if h_vec[i] + eps <= 1:
                h_plus[i] += eps
            else:
                h_plus[i] = 1.0
            if h_vec[i] - eps >= 0:
                h_minus[i] -= eps
            else:
                h_minus[i] = 0.0
                
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[n - 1 + k_max] * dx
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[n - 1 + k_max] * dx

            if corr_plus - corr_minus != 0:
                # Gradient approximation
                grad[i] = (corr_plus - corr_minus) / (2.0 * eps)
            else:
                grad[i] = 0.0
                
        return grad

    best_h = h.copy()
