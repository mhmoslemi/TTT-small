import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Normalize initial h to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Parameters
    max_iter = 10000
    alpha = 0.1
    decay = 1000
    eps = 1e-8
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps for all k
        h_flipped = 1 - h
        overlaps = signal.correlate(h, h_flipped, mode='full')
        k_offset = len(h) - 1
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Gradient direction: derivative of overlap wrt h
        # For k at max_k, gradients come from h_flipped shifted by k
        # If max_k is offset k, then d/dh_i = -h_flipped[i + k] 
        # Or more accurately we need to consider the actual convolution gradient
        
        # Simplify the approach: gradient is rough approximation
        # Negative gradient: use the flipped h shifted to maximize overlap (detailed)
        grad = np.zeros_like(h)
        k_idx = max_k - k_offset  # Translate full index to the actual shift k
        # Step: gradient flow direction approximated by convolving h with (1-h) then taking gradient
        
        # Compute overlap with shift k_idx (we can improve this using more precise update)
        # Manual direct simple update: 
        # For computational ease and directly using max overlap part, take gradient
        # Approximation of gradient from overlap gradient signal
        flip_shifted = np.roll(h_flipped, k_idx) if k_idx != 0 else h_flipped
        grad = flip_shifted  # Simplified gradient sign

        # Adjust learning rate
        current_alpha = alpha * np.exp(-i / decay)
        
        # Update h
        h_new = h - current_alpha * grad
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Rescale to sum to n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        current_c5 = evaluate_erdos_solution(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h_values):
    # Compute C5 using numpy correlate with 1-h
    h = np.array(h_values)
    h_flipped = 1 - h
    overlaps = np.correlate(h, h_flipped, mode='full')
    k_offset = len(h) - 1
    max_k = np.argmax(overlaps)
    max_overlap = overlaps[max_k]
    return max_overlap / len(h)
