import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial solution if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except NameError:
        # Fallback: create initial construction
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        h0[10:20] = 1.0  # Step-like construction
        h0[30:40] = 0.0
        h0 = h0 / np.sum(h0) * (n_points / 2)
    
    # Make copy of initial solution
    h = h0.copy()
    dx = 2.0 / n_points
    
    # Adaptor: track and adjust parameters
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Run overlap computation
        overlaps = np.correlate(h, 1 - h, mode="full")
        # We compute max over all lags, but just sample the largest ones
        top_indices = np.argsort(overlaps)[-5:][::-1]
        
        # Gradient estimation: for top k overlaps
        grad = np.zeros_like(h)
        for k in top_indices:
            # We compute the overlap gradients manually
            # Symmetric way: consider how changes to h[i] affect overlap_k
            k_offset = k - (n_points - 1)
            # Don't include k=0 as it's just integral of h(x)
            for i in range(n_points):
                if k_offset >= 0:
                    j = (i + k_offset) % n_points
                else:
                    j = (i - k_offset) % n_points
                grad[i] += (1 - h[j]) * dx
                # And reverse
                grad[j] -= h[i] * dx
                
        # Apply momentum
        velocity = momentum * velocity - learning_rate * grad
        h += velocity
        
        # Project step back to feasible set
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h *= (n_points / 2) / h_sum  # rescale to exact sum
        
        # Evaluate C5 for new h
        c5 = evaluate_overlap(h, dx)
        
        # Update best found
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Adaptive learning rate based on improvement
        if iteration % 10 == 0:
            learning_rate *= 0.995  # slowly decay
            
        # Periodic randomization to escape local optima
        if iteration % 100 == 0 and time.time() - start_time < budget_s - 2:
            h = best_h.copy()
            noise = np.random.normal(0, 0.01, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h *= (n_points / 2) / h_sum
            
    # Ensure it meets the constraint exactly
    h_final = best_h.copy()
    h_final_sum = np.sum(h_final)
    h_final *= (n_points / 2) / h_final_sum
    final_c5 = evaluate_overlap(h_final, dx)
    
    return (h_final.tolist(), final_c5, n_points)

def evaluate_overlap(h, dx):
    """Computes the maximum overlap of h(x)(1-h(x+k)) over all k."""
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    return overlaps.max()
