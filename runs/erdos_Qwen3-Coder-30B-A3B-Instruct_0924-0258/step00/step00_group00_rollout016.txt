import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    """Compute C5 from the given step function h"""
    h = np.array(h_values)
    dx = 2.0 / len(h)
    # Compute the full correlation
    corr = correlate(h, 1 - h, mode='full')
    # Extract all correlations for lags 0 to 2n-2
    # For maximum C5, we want max over all lags [0, 2n-2]
    # The correlation for a given lag k (0 <= k <= 2n-2) 
    # corresponds to corr[n-1 + k] with length = 2n-1
    # But we need to look carefully at indexing
    # For correlation with lag k:
    # The term is sum over i of h[i] * (1 - h[i+k])
    # This contributes to the correlation at position k in the result
    # But we're doing (h * (1-h)) full correlation, so we're looking at corr[n-1 - k : n-1 + k+1]
    # Actually, interpreting full mode: corr[0:2n-1] are the correlations from k=-n+1 to k=n-1
    # We need positive lags, so we want k from 0 to n-1 that matter
    # But more directly: lag from -n+1 to n-1
    # For our problem it's natural to consider correlations over [0, 2n-2]
    # What we actually compute: lag tuples (k, max_e) = k from -n+1 to n-1
    # In np.correlate, with full mode, index corresponds to:
    # corr[0] is lag = -n+1
    # corr[n-1] is lag = 0
    # corr[n-1+k] is lag = k
    # So we extract all lags from 0 to n-1, but because of symmetry or two directions,
    # we compute over k from 0 to 2n-2 (in fact over full range of correlations for lags -n+1 to n-1)
    # We need "maximum correlation" in such: 
    # src = h, target = 1-h
    # Actually to avoid confusion:
    # Let's recompute manually for clarity:

    # The result [0:2n-1] corresponds to lags -n+1 to n-1
    #   lag -n+1: corr[0]
    #   lag -n+2: corr[1]
    #   ...
    #   lag 0: corr[n-1] 
    #   lag 1: corr[n]
    #   ...
    #   lag n-1: corr[2n-2]
    # We want max across all k in [0, n-1] i.e. positive lags but really (0 to n-1)
    # but not double count the negative lags  
    # But we are computing:
    # (1-h) lag k (seen) -> h = h[i+k] * (1 - h[i]) for 0 <= i <= n-1-k
    # That's the cross correlation contribution for lag k, starting at lag = 0.
    # But we are allowed to shift and find max lags.
    # So let:
    # C_k = \int_0^2 h(x) (1-h(x + k)) dx, for x in [0, 2-k] (restriction)
    # So it's a correlation of h with (1-h)
    # Let's consider a few values to see the correct indexing:
    # cross_correlation[0] is for lag=0, we use [0,...,n-1], [0,...,n-1] 
    # AND for [0,...,n-1-k], in loop over k, i updates to n-1-k  
    # C_k = \sum_i h[i] * (1-h[i+k]) for i = 0 to n-1-k  
    # Various indexing gymnastics.  
    # So in correlate:
    # corr = correlate(h, 1-h)
    # The correlogram (the correlation) at k-th lag has index k + n-1 (in 0:2n-2)
    # This corresponds to a lag k, where:
    #   lag = k  (since corr[n-1] = lag 0)
    # But note that we want the sum over i of h[i]* (1 - h[i+k]) still confined to valid range.
    # When we look at lag = 0, this works.
    # So it's correct to access corr[n-1 + k] for the correlation value for lag k
    # But now we want correlation for all lags, calculate C_k over range k = 0 to 2n-2
    # We compute full convolution (mode='full') and pick all k = 0 to 2n-2 <= 2n-2
    
    # Here's the clean way:
    # We compute the positive lags that matter: 1 to n points inclusion, bias, can be derived.
    # When we compute over [0,2] domain with n points, and h(x) = h[i] for x in [i*dx,(i+1)*dx)
    # Then for a fixed lag k = 0 to n-1, the integral corresponds to h[i]* (1 - h[i+k]) over appropriate indices
    # Domain for a lag k is [0, 2-k] => we have at most n-k valid indices  

    corr = correlate(h, 1 - h, mode='full')
    # Extract all valid correlations  
    # max over all allowed lag values, each associated with the same number of terms kappa; 0 <= k <= n-1
    # Re-extract all real cross-correlations manually for robustness  

    max_c5_candidate = -1.0
    for lag in range(2 * len(h) - 1):
        if lag < len(h) - 1:
            # Lag k where the overlap is limit
            k = lag - (len(h) - 1)
            # Only for lag in critical range that affects the full integrals:
            # Actually let's get the correct cross correlation.
            # Use indices from 0 to n-1 for source h, target 1-h shifted.
            # lag = k = 0 to n-1, point offset, correspond 0 to n-1 for source h.
            # to i+k legal: 0 <= i <= n-1-k
            # So correlation at lag k = index of corr = (n-1 + k) where k goes from 0 to n-1, that’s great
            # for the correlation of h with (1-h) shifted. Then multiply by dx to get normalized integral.
            # lag = 0 to n-1 -> in index = n-1 + k
            # But we must ensure i+k < n, so i = 0 to n-1-k. 
            # Correlation at index k' = k + (n-1)    
            value = corr[lag] * dx
            # But in full mode, index 0 to 2n-2, and (due to how correlate interprets),
            # The actual lags from -n+1 to n-1
            # corr [0] = lag = -n+1
            # corr [n-1] = lag = 0
            # corr [2n-2] = lag = n-1
            # If we're taking lag k from 0 to n-1, that corresponds to corr[n-1 + k].
            if k >= 0:
                value = corr[lag] * dx
                if value > max_c5_candidate:
                    max_c5_candidate = value
        else:
            # For lag negative or beyond critical range, nothing to do; ignore.
            # Or check the negative lag cases:
            # The issue points to structure where very large lags (i.e., within domain length) contribute
            # Usually more relevant is lag 0 to n-1 or 0 to some max depending on support widths.
            pass

    return max_c5_candidate

        h[:half] = 1.0  # Use binary-like warming
        h = h / np.sum(h) * (n_points / 2)  # Scale to sum = n_points/2 

    dx = 2.0 / n_points
    # Check feasibility
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Fix sum to exactly n_points/2

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    iterations = 0
    patience = 0
    max_patience = 20
    
    def project_h(h_vec):
        # Keep all values within [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale so sum = n_points/2
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec

    while time.time() - start_time < budget_s - 0.01:
        iterations += 1
        if iterations % 100 == 0:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.95:
                break

        # Evaluate current h
        current_c5 = evaluate_erdos_solution(h)
        # Rescale to maintain equality constraint
        h = project_h(h)

        # Find which lag(s) maximizes
        corr = correlate(h, 1 - h, mode='full')
        max_corr = -1.0
        max_lags = []
        for k in range(2 * n_points - 1):
            val = corr[k] * dx
            if val > max_corr:
                max_corr = val
                max_lags = [k]
            elif val == max_corr:
                max_lags.append(k)

        # Compute gradient for maximum contributor
        gradient = np.zeros(n_points)
        for k in max_lags:
            # Source h[0...n-1-k], target 1-h[k...]
            # grad[i] += (1 - h[i+k]) 
            # grad[i+k] -= h[i] 
            for i in range(n_points - k):
                gradient[i] += (1 - h[i+k])
                if i + k < n_points:
                    gradient[i+k] -= h[i]

        # Line search over step sizes
        best_step = 0.0
        best_h_trial = h.copy()
        best_trial_c5 = current_c5
        for alpha in [0.1, 0.25, 0.5]:
            h_trial = h - alpha * gradient
            h_trial = project_h(h_trial)
            trial_c5 = evaluate_erdos_solution(h_trial)
            if trial_c5 < best_trial_c5:
                best_trial_c5 = trial_c5
                best_h_trial = h_trial.copy()
                best_step = alpha

        # Apply improvement
        if best_trial_c5 < best_c5:
            h = best_h_trial
            best_c5 = best_trial_c5
            patience = 0
        else:
            patience += 1

        # Escape from local optima
        if patience >= max_patience:
            h += 0.01 * np.random.randn(n_points)
            h = project_h(h)
            patience = 0

        # Update best solution 
        if best_c5 < evaluate_erdos_solution(best_h):
            best_h = h.copy()

    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    return (list(best_h), final_c5, n_points)
