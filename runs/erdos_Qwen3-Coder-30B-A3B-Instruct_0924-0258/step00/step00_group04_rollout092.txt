import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = 42  # Using the provided initial construction size
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        if not (np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - target_sum) < 1e-10):
            h = np.clip(h, 0, 1)
            h *= target_sum / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h *= target_sum / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Optimization parameters
    max_iter = 1200
    alpha = 0.1
    tolerance = 1e-5
    last_improvement = 0
    start_time = time.time()
    
    # Perturbation frequency
    perturb_freq = 50
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute all overlap integrals
        # Compute overlap for each lag k
        overlap_values = np.zeros(n_points)
        for k in range(n_points):
            # Compute correlation: h[i] * (1 - h[(i+k)%n_points])
            # This is equivalent to computing the overlap integral for lag k
            # Using scipy's correlate with mode="full" and extracting only the relevant part
            corr = correlate(h, 1 - np.roll(h, k), mode="full")
            overlap_values[k] = corr[n_points - 1] * dx  # This is the overlap at shift k
        
        # Find worst lag
        max_k = np.argmax(overlap_values)
        max_overlap = overlap_values[max_k]
        
        # Compute gradient for worst case lag
        # Gradient is partial derivative of max_overlap w.r.t h[i]
        grad = np.zeros(n_points)
        for i in range(n_points):
            # For overlap at lag max_k:
            # grad[i] = 2/n_points * (1 - h[(i-max_k)%n_points] - h[(i+max_k)%n_points])
            grad[i] = 2 * dx * (1 - h[(i - max_k) % n_points] - h[(i + max_k) % n_points])
        
        # Update h
        h_new = h - alpha * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        # Rescale to maintain sum constraint
        h_new *= target_sum / np.sum(h_new)
        
        # Check if update improved C5
        current_c5 = evaluate_c5(h_new)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            last_improvement = iteration
        
        # Early stopping if no improvement recently
        if iteration - last_improvement > 200 and current_c5 > best_c5 + 1e-4:
            h = h_new
        else:
            h = h_new
            
        # Occasionally add small noise (perturbation)
        if iteration % perturb_freq == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h *= target_sum / np.sum(h)
    
    return (best_h.tolist(), float(best_c5), int(n_points))


def evaluate_c5(h):
    """Computes the maximum overlap integral C5 for a given h"""
    h = np.array(h)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Compute overlaps for all lags
    overlap_values = np.zeros(n_points)
    for k in range(n_points):
        # Compute the integral of h(x) * (1 - h(x + k)) for lag k
        # This is equivalent to the sum over i: h[i] * (1 - h[(i+k)%n_points]) * dx
        corr = correlate(h, 1 - np.roll(h, k), mode="full")
        overlap_values[k] = corr[n_points - 1] * dx
    
    return np.max(overlap_values)
