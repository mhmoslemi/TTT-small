import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    # Initialize
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
        h[0] = 0.5 - (n_points/2 - np.sum(h)) * dx

    # Ensure constraint ∫h(x)dx = 1
    h = h * (n_points / (2 * np.sum(h)))
    
    # Helper functions
    def evaluate_C5(h):
        # Evaluate overlap integral using full convolution
        h_padded = np.pad(h, (0, n_points), mode='constant', constant_values=0)
        h_rev = np.flip(h)
        conv = np.convolve(h_padded, h_rev, mode='full')
        # Need to compute max of ∫h(x)(1-h(x+k))dx for all k
        # This is equivalent to max over all lags of:
        # sum(h[i] * (1 - h[(i+k)%n])) * dx
        # Let's compute it as convolution, but with 1-h instead of h:
        inv_h = 1.0 - h
        inv_h_padded = np.pad(inv_h, (0, n_points), mode='constant', constant_values=0)
        conv2 = np.convolve(h, inv_h_padded, mode='full')
        # Now we need to find maximum among lags 0 to 2*n_points-1
        # But it's actually just max of all lag 0..n_points-1 positions
        # because the convolution is symmetric and we are interested in all lags
        # The maximum value in the convolution is what we want
        max_val = 0.0
        for k in range(n_points):
            val = conv2[n_points+k] * dx
            if val > max_val:
                max_val = val
        return max_val

    def project_h(h):
        # Ensure 0 <= h[i] <= 1
        h = np.clip(h, 0, 1)
        # Adjust to make sum = n_points/2
        current_sum = np.sum(h)
        if abs(current_sum - n_points/2) > 1e-10:
            # Simple projection along direction of all ones
            adjustment = (n_points/2 - current_sum) / n_points
            h = h + adjustment
            h = np.clip(h, 0, 1)
        return h

    def compute_gradient(h):
        # Get cache of current C5
        C5_current = evaluate_C5(h)
        # Find all indices k where max hit
        max_k_indices = []
        # Compute cross-correlation of h & (1-h) to find where C5 is max
        conv_result = np.correlate(h, 1-h, mode='full')
        max_val = np.max(conv_result)
        # Find positions that attained maximum
        for i, val in enumerate(conv_result):
            if abs(val - max_val) < 1e-10:
                max_k_indices.append(i)
        # Compute gradient
        g = np.zeros_like(h)
        for k_idx in max_k_indices:
            for i in range(n_points):
                if k_idx < n_points:
                    # right shift convolution case
                    j = i + k_idx
                else:
                    # left shift convolution case
                    j = i - (2*n_points - k_idx)
                if j >= 0 and j < n_points:
                    g[i] += (1 - h[j] - (1 - h[i])) * dx
            g /= len(max_k_indices)  # Normalize
        return g

    # Optimize
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    start_time = time.time()
    max_iter = 500
    alpha = 0.1
    iter_count = 0
    while iter_count < max_iter:
        if time.time() - start_time > budget_s * 0.98:
            break
        # compute gradient of current C5
        try:
            g = compute_gradient(h)
        except:
            break
        # update h
        h_new = h - alpha * g
        # project
        h_new = project_h(h_new)
        # evaluate
        new_C5 = evaluate_C5(h_new)
        if new_C5 < best_C5 and new_C5 < 0.38085:
            best_h = h_new.copy()
            best_C5 = new_C5
        # adapt step size
        if new_C5 < best_C5:
            alpha *= 1.1  # increase learning rate if improved
        else:
            alpha *= 0.7  # reduce if not improved
        # update h
        h = h_new.copy()
        iter_count += 1

    # Final check for accuracy
    best_h = project_h(best_h)
    final_C5 = evaluate_C5(best_h)
    return (list(best_h), final_C5, n_points)
