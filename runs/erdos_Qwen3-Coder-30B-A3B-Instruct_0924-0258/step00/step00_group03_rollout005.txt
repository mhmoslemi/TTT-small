import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral for a given step function h."""
    overlap = correlate(h, 1 - h, mode='full')
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.ones_like(h) * (target_sum / len(h))
    return h * (target_sum / current_sum)

def mutate_h(h, mutation_rate=0.1, noise_std=0.1):
    """Apply mutation to a step function."""
    h_mut = h.copy()
    mask = np.random.random(len(h)) < mutation_rate
    h_mut[mask] += np.random.normal(0, noise_std, np.sum(mask))
    h_mut = np.clip(h_mut, 0, 1)
    return normalize_h(h_mut, len(h) / 2)

def crossover_h(h1, h2, crossover_rate=0.8):
    """Crossover two step functions."""
    if np.random.random() > crossover_rate:
        return h1.copy(), h2.copy()
    crossover_point = np.random.randint(1, len(h1))
    h1_new = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    h2_new = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    return h1_new, h2_new

def tournament_selection(population, fitnesses, tournament_size=5):
    """Tournament selection."""
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        selected.append(population[best_idx])
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    max_generations = 2000
    population_size = 50
    
    # Initialize population
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = normalize_h(h0, n_points / 2)
        population.append(h0)
    else:
        h0 = np.ones(n_points) * 0.5
        population.append(normalize_h(h0, n_points / 2))
    
    # Add random feasible individuals
    while len(population) < population_size:
        h = np.random.random(n_points)
        h = normalize_h(h, n_points / 2)
        population.append(h)
    
    # Optimization loop
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    generation = 0
    
    while time.time() - start_time < budget_s and generation < max_generations:
        # Evaluate fitness of all individuals
        fitnesses = []
        for h in population:
            try:
                c5 = evaluate_h(h, dx)
                fitnesses.append(c5)
            except:
                fitnesses.append(float('inf'))
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Crossover and mutation
        new_population = [population[min_fitness_idx].copy()]  # Elitism
        
        while len(new_population) < population_size:
            h1, h2 = np.random.choice(selected, 2, replace=False)
            h1_new, h2_new = crossover_h(h1, h2)
            
            h1_new = mutate_h(h1_new)
            h2_new = mutate_h(h2_new)
            
            new_population.extend([h1_new, h2_new])
        
        # Truncate to population size
        population = new_population[:population_size]
        generation += 1
    
    # Final evaluation of best solution
    if best_h is not None:
        try:
            final_c5 = evaluate_h(best_h, dx)
            return (best_h, final_c5, n_points)
        except:
            pass
    # If something went wrong, return best found solution so far
    if best_h is not None:
        return (best_h, best_c5, n_points)
    else:
        # Fallback to initial solution
        h0 = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
        h0 = normalize_h(h0, n_points / 2)
        c5 = evaluate_h(h0, dx)
        return (h0, c5, n_points)
