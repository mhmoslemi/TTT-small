import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()
    
    # Parameters
    max_iter = 1200
    alpha = 0.1
    tolerance = 1e-5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Compute overlap integrals for all lags k
        c5_vals = []
        for k in range(n_points):
            # Implement correlation manually
            corr = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                corr += h[i] * (1.0 - h[j])
            c5_vals.append(corr * 2.0 / n_points)
        
        # Find worst case
        max_idx = np.argmax(c5_vals)
        max_val = c5_vals[max_idx]
        
        # Compute gradient for this max case
        grad = np.zeros(n_points)
        for i in range(n_points):
            j1 = (i - max_idx) % n_points
            j2 = (i + max_idx) % n_points
            grad[i] = 2.0 / n_points * (1.0 - h[j1] - h[j2])
        
        # Line search
        step_size = alpha
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new)
        
        # If improvement, accept the step
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Update h
        h = h_new
        
        # Perturb occasionally to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final validation
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    c5_bound = evaluate_erdos_solution(h_final)
    
    return (h_final.tolist(), c5_bound, n_points)

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations
    h_padded = np.concatenate([h, h])
    max_c5 = 0.0
    for k in range(n_points):
        corr = np.sum(h * h_padded[n_points-k:n_points-k+n_points]) * dx
        if corr > max_c5:
            max_c5 = corr
    return max_c5
