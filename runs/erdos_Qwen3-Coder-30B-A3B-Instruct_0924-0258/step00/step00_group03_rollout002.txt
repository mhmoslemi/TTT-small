import numpy as np
from scipy import signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except:
        # Fallback: create a simple initial construction
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Ensure n_points is even so that sum(h) = n_points / 2 is integer
    if n_points % 2 != 0:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Ensure h satisfies constraint: sum(h) * dx = 1
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Parameters
    max_iter = 10000
    lr = 0.01
    decay = 0.95
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute convolution values
        conv = signal.correlate(h, 1 - h, mode="full")
        k0 = np.argmax(conv)
        max_val = conv[k0]
        
        # Compute gradient of the objective with respect to h at k0
        grad = np.zeros_like(h)
        # For k0, the gradient is the convolution of 1-h with delta(x-k0)
        for j in range(len(h)):
            if k0 - j >= 0 and k0 - j < len(h):
                grad[j] += (1 - h[k0 - j])
            if k0 + j < len(h):
                grad[j] += h[j]
        
        # Update h with gradient descent
        h_new = h - lr * grad
        
        # Project onto feasible set using CVXPY
        # Variables
        x = cp.Variable(n_points)
        # Objective (minimize)
        # In this case, we directly solve projection onto the simplex
        # Constrain: sum(x) == n_points / 2, 0 <= x <= 1
        constraints = [cp.sum(x) == n_points / 2, x >= 0, x <= 1]
        prob = cp.Problem(cp.Minimize(cp.sum_squares(x - h_new)), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h = x.value
        except:
            #Fallback to projection
            h = np.clip(h_new, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Update learning rate
        if i % 100 == 0:
            lr *= decay
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    conv = signal.correlate(h, 1 - h, mode="full")
    return np.max(conv) * dx
