import numpy as np

def run(seed=42, budget_s=600, **kwargs):

    # Time tracking
    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent settings
    lr = 0.01
    decay = 0.95
    decay_freq = 100
    
    # Main optimization loop
    for i in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation and find max
        corr = np.correlate(h, 1 - h, mode="full")
        k0 = np.argmax(corr)
        max_corr = corr[k0]
        
        # Compute gradient of correlation w.r.t h at k0
        grad = np.zeros_like(h)
        for j in range(n_points):
            if k0 < n_points:
                if j + k0 < n_points:
                    grad[j] += -1
                if j - k0 >= 0:
                    grad[j] += -1
            else:
                offset = k0 - n_points
                if j + offset < n_points:
                    grad[j] += -1
                if j - offset >= 0:
                    grad[j] += -1
                    
        # Update h
        h_new = h - lr * grad
        
        # Ensure h_new is feasible using convex optimization
        h_new = project_onto_feasible_set(h_new, n_points, dx)
        
        # Update learning rate
        if i % decay_freq == 0:
            lr *= decay
            
        # Evaluation
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new
        
    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_onto_feasible_set(h, n, dx):
    # Use CVXPY for convex optimization to ensure constraints
    h_var = cp.Variable(n)
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2
    ]
    prob = cp.Problem(cp.Minimize(cp.sum(cp.square(h_var - h))), constraints)
    prob.solve(solver=cp.CVXOPT, verbose=False)
    return np.maximum(0, np.minimum(1, h_var.value))
