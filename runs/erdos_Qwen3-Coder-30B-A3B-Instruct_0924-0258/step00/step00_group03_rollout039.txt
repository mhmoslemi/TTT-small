import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Correlate h with (1-h) in full mode
    correlation = np.correlate(h, 1 - h, mode="full")
    # Extract the relevant part (max k values)
    max_k = n - 1
    # This represents C5 = max_k integral_h(x)(1-h(x+k))dx
    # Using the full convolution, we need to find max
    # correlation[0] corresponds to k=-(n-1), ..., correlation[n-1] k=0, ..., correlation[2*n-2] k=n-1
    # So the range for k from 1 to n-1 is correlation[n:2*n-1]
    values = correlation[n-1:2*n-1]
    return np.max(values) * dx

def project(h):
    # Clamp values to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to match sum constraint
    current_sum = np.sum(h)
    target_sum = len(h) / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * 0.5
    return h

def mutate(h, sigma):
    h_new = h + np.random.normal(0, sigma, len(h))
    return project(h_new)

def crossover(h1, h2, alpha):
    # Blend crossover
    beta = np.random.uniform(alpha, 1 - alpha)
    h_new = beta * h1 + (1 - beta) * h2
    return project(h_new)

def compute_gradient(h):
    n = len(h)
    # compute gradient with respect to the maximum of the overlap integrals
    grad = np.zeros_like(h)
    # For simplicity, we use finite differencing for approximate gradient
    eps = 1e-6
    base_val = evaluate_erdos_solution(h)
    for i in range(n):
        h_eps = h.copy()
        h_eps[i] += eps
        h_eps = project(h_eps)
        eps_val = evaluate_erdos_solution(h_eps)
        grad[i] = (eps_val - base_val) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70  # use the initial construction's size for consistency
    max_generations = 10000
    pop_size = 100
    elite_size = int(0.4 * pop_size)
    mutation_rate = 0.1
    crossover_alpha = 0.6
    learning_rate = 0.02
    decay_rate = 100
    
    # Initialize population
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
        population.extend([mutate(np.array(initial_h_values), 0.1) for _ in range(pop_size - 1)])
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = project(h_init)
        population = [h_init]
        population.extend([mutate(h_init, 0.1) for _ in range(pop_size - 1)])
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
            
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track current best
        if fitness[0] < best_c5:
            best_h = population[0]
            best_c5 = fitness[0]
        
        # Selection: top elite + some random
        selected_indices = sorted_indices[:elite_size]
        leftover_indices = np.random.choice(sorted_indices[elite_size:], 
                                             size=pop_size - elite_size, replace=False)
        selected_indices = np.append(selected_indices, leftover_indices)
        selected_pop = [population[i] for i in selected_indices]
        
        # Crossover
        offspring = []
        for _ in range(int(0.3 * pop_size)):
            i1, i2 = np.random.choice(len(selected_pop), size=2, replace=False)
            child = crossover(selected_pop[i1], selected_pop[i2], crossover_alpha)
            offspring.append(child)
        
        # Mutation
        for _ in range(int(0.7 * pop_size) - len(offspring)):
            parent = np.random.choice(selected_pop)
            child = mutate(parent, mutation_rate * np.exp(-gen/decay_rate))
            offspring.append(child)
        
        # Local refinement for top candidates
        top_candidates = offspring[:int(0.05 * len(offspring))]
        for h in top_candidates:
            grad = compute_gradient(h)
            h_new = h - learning_rate * np.exp(-gen/decay_rate) * grad
            h_new = project(h_new)
            offspring.append(h_new)
        
        # Evaluate new offspring
        new_fitness = [evaluate_erdos_solution(h) for h in offspring]
        
        # Replacement: replace worst with new
        worst_idx = np.argsort(fitness)[-len(offspring):]
        for i, idx in enumerate(worst_idx[:len(offspring)]):
            population[idx] = offspring[i]
            fitness[idx] = new_fitness[i]
    
    return (list(best_h), best_c5, n_points)
