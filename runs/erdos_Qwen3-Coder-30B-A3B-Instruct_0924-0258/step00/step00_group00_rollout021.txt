import numpy as np
import time
from scipy import signal

def evaluate_c5(h, dx):
    """Evaluate C5 for a given step function h."""
    corr = np.correlate(h, 1 - h, mode='full')
    # Only consider valid lags (from -n+1 to n-1)
    n = len(h)
    valid_corr = corr[n-1:2*n-1]
    return np.max(valid_corr) * dx

def rescale_to_unit_sum(h, target_sum):
    """Rescale h to have sum = target_sum, maintaining values in [0,1]."""
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.full_like(h, target_sum / len(h))
    scaled = h * (target_sum / current_sum)
    return np.clip(scaled, 0, 1)

def create_candidate(parent_h, n_points):
    """Create a candidate from parent by perturbing values."""
    h = parent_h.copy()
    # Randomly perturb 10% of values
    mask = np.random.random(n_points) < 0.1
    delta = (np.random.random(n_points) - 0.5) * 0.2
    h[mask] += delta[mask]
    h = rescale_to_unit_sum(h, n_points / 2)
    return h

def crossover(h1, h2):
    """Perform one-point crossover."""
    n = len(h1)
    point = np.random.randint(1, n)
    new_h = np.concatenate([h1[:point], h2[point:]])
    return rescale_to_unit_sum(new_h, len(new_h) / 2)

def tournament_selection(population, fitnesses, k=3):
    """Select an individual via tournament selection."""
    indices = np.random.choice(len(population), k)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 45
    dx = 2.0 / n_points
    pop_size = 50
    max_generations = 200

    population = [parent_h]
    for _ in range(pop_size - 1):
        candidate = create_candidate(parent_h, n_points)
        population.append(candidate)
    
    # Evaluate initial population
    fitnesses = [evaluate_c5(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # GA loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Selection and reproduction
        new_population = [best_h]  # Elitism
        
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover with 70% chance
            if np.random.random() < 0.7:
                child = crossover(parent1, parent2)
            else:
                # Mutation
                child = create_candidate(parent1, n_points)
                
            new_population.append(child)
            
        # Evaluate new population
        population = new_population
        fitnesses = [evaluate_c5(h, dx) for h in population]
        
        # Update best
        min_fitness = np.min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_idx = np.argmin(fitnesses)
            best_h = population[best_idx].copy()
    
    return (best_h.tolist(), best_c5, n_points)
