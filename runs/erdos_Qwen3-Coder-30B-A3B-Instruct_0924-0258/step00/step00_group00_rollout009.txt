import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute all cross-correlations of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The cross-correlations are at lags from -(n-1) to (n-1)
    # We only care about lags 0 to n-1 (non-negative lags)
    # But we want max over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():

        # Ensure it satisfies the constraints
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
    else:
        h = np.ones(n_points) * 0.5  # fallback initialization
    
    # Ensure sum constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    
    # Frank-Wolfe optimization
    max_iter = 500
    tolerance = 1e-6
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    start_time = time.time()
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute all cross-correlations with 1-h
        corr = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_corr = corr[max_lag_idx]
        max_corr *= dx
        
        # Approximate gradient at h for max_corr
        # This is simplified; actual gradient is complex for the max operator
        # Let's do a proper line search around the current h
        # For practical implementation, we switch to a better approach
        
        # Compute gradient for a single correlation value
        # Manually compute the gradient of correlation w.r.t. a specific h_i
        # For the current max lag, compute indices that contribute
        # We'll use a greedy projected gradient approach with line search
        
        # In practice, Frank-Wolfe with subgradient approach
        # Use basic gradient descent with projected updates for optimization
        # Use more efficient direct optimization
        
        # Let's use a small gradient-based approach instead:
        # Find the k that maximizes the correlation
        max_corr_val = -np.inf
        max_k = 0
        
        for k in range(n_points):
            # Compute cross-correlation for lag k (excluding symmetrically viewed lags)
            # Only compute the forward lag k: sum h[i] * (1 - h[i+k])
            corr_val = 0.0
            for i in range(n_points - k):
                corr_val += h[i] * (1 - h[i + k])
            corr_val *= dx
            if corr_val > max_corr_val:
                max_corr_val = corr_val
                max_k = k
                
        # Now we have the maximum correlation index, go to a greedy update
        # This using simple projected gradient descent approach
        # Compute gradient approximation for max_k: d/dh_i of h[i] * (1 - h[i+k])
        
        grad = np.zeros(n_points)
        for i in range(n_points - max_k):
            grad[i] += 1 - h[i + max_k]
            if i + max_k < n_points:
                grad[i + max_k] -= h[i]
        
        # Find s elements that maximize value in gradient
        s = n_points // 2
        indices = np.argsort(-grad)[:s]
        
        # Perform line search to find best step size (γ)
        candidates = [0.01, 0.05, 0.1, 0.25, 0.5, 0.75, 1.0]
        best_gamma = 0.0
        best_new_c5 = max_corr_val
        
        for gamma in candidates:
            # Update h with step gamma in direction of gradient
            new_h = h.copy()
            for i in indices:
                new_h[i] += gamma * grad[i]
            new_h = np.clip(new_h, 0, 1)
            
            # Maintain sum = target_sum
            new_sum = np.sum(new_h)
            if new_sum > 0:
                new_h = new_h * target_sum / new_sum
            
            # Evaluate new C5
            new_c5 = evaluate_h(new_h, dx)
            if new_c5 < best_new_c5:
                best_gamma = gamma
                best_new_c5 = new_c5
                
        # Perform actual update with best gamma
        if best_gamma > 0:
            update_h = h.copy()
            for i in indices:
                update_h[i] += best_gamma * grad[i]
            update_h = np.clip(update_h, 0, 1)
            new_sum = np.sum(update_h)
            if new_sum > 0:
                update_h = update_h * target_sum / new_sum
            h = update_h
          
        # Keep best solution so far
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
    # Final evaluation
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
