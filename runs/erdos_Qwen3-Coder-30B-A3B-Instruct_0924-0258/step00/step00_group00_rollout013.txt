import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def step_descent(h_vec, max_iter=20):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        
        perturb_steps = 0
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Compute all cross correlations
            corr = correlate(h_current, 1 - h_current, mode='full')
            # Find maximum lag
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (len(h_current) - 1)
            
            # Gradient direction for this lag
            # Forward direction of h[i] contributes to correlation with lag k
            # if h[i] is in position i, then (1-h[i+k]) is in position i+k
            grad = np.zeros_like(h_current)
            if max_lag >= 0:  # Positive lag
                grad[max_lag:] = -h_current[:-max_lag]
            else:  # Negative lag
                grad[:max_lag] = -h_current[-max_lag:]
            
            # Try different step sizes
            alpha_candidates = [0.1, 0.5, 1.0]
            best_alpha = 0.0
            best_new_h = h_current.copy()
            best_new_C5 = best_C5
            
            for alpha in alpha_candidates:
                candidate_h = h_current - alpha * grad
                candidate_h = project_feasible(candidate_h)
                new_C5 = compute_C5(candidate_h)
                if new_C5 < best_new_C5:
                    best_new_C5 = new_C5
                    best_new_h = candidate_h.copy()
                    best_alpha = alpha
            
            # If not improved, maybe perturb
            if best_new_C5 >= best_C5:
                perturb_steps += 1
                if perturb_steps > 20:
                    # Add noise and re-normalize
                    noise = np.random.normal(0, 0.05, n_points)
                    h_current += noise
                    h_current = project_feasible(h_current)
                    perturb_steps = 0
                    continue
            else:
                h_current = best_new_h.copy()
                best_C5 = best_new_C5
                best_h = h_current.copy()
                perturb_steps = 0
            
        return best_h, best_C5
    
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Local optimization loop
    while time.time() - start_time < budget_s - 0.1:
        # Try several local optimizations from current point
        for _ in range(5):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Perturb current point slightly
            h_perturbed = h_best + np.random.normal(0, 0.01, n_points)
            h_perturbed = project_feasible(h_perturbed)
            
            # Perform gradient descent from this point
            h_opt, C5_opt = step_descent(h_perturbed, max_iter=20)
            if C5_opt < C5_best:
                C5_best = C5_opt
                h_best = h_opt.copy()
    
    # Final check
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
