import numpy as np
from scipy import signal
import time
import random

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given step function h."""
    # Compute cross-correlation
    corr = signal.correlate(h, 1 - h, mode='full')
    # Scale by dx
    corr_scaled = corr * dx
    # Return maximum value (this is C5)
    return np.max(corr_scaled)

def rescale_h(h, dx):
    """Rescale h so that sum(h) * dx = 1, maintaining bounds [0,1]."""
    target_sum = 1.0 / dx
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Rescale
    h = h * (target_sum / current_sum)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Renormalize again to ensure sum constraint
    h = h * (target_sum / np.sum(h))
    return h

def initial_h():
    """Create initial step function h."""
    global initial_h_values
    try:
        h = np.array(initial_h_values)
        # If initial_h_values is available, use it
        return h
    except NameError:
        # Otherwise, create a simple initial construction
        n = 45  # n_points
        h = np.ones(n) * 0.5  # Start with uniform distribution
        h[0] = 0.4
        h[-1] = 0.6
        return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    random.seed(seed)
    np.random.seed(seed)
    
    # Initialize params
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 10000
    swarm_size = 30
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    
    # Create initial population
    swarm = []
    velocities = []
    
    # Initialize swarm
    for _ in range(swarm_size):
        h = initial_h().copy()
        # Add noise
        h += np.random.normal(0, 0.05, n_points)
        # Ensure feasibility after noise
        h = np.clip(h, 0, 1)
        h = rescale_h(h, dx)
        swarm.append(h)
        velocities.append(np.zeros_like(h))
    
    # Best positions tracking
    personal_best = [h.copy() for h in swarm]
    personal_best_scores = [evaluate_h(h, dx) for h in swarm]
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        for i in range(swarm_size):
            h = swarm[i]
            
            # Clip and rescale h
            h = np.clip(h, 0, 1)
            h = rescale_h(h, dx)
            swarm[i] = h
            
            # Evaluate score
            score = evaluate_h(h, dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = h.copy()
                personal_best_scores[i] = score
                
            # Update global best
            if score < global_best_score:
                global_best = h.copy()
                global_best_score = score

            # Update swarm
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * (personal_best[i] - h) +
                             social * r2 * (global_best - h))
            
            # Update h
            h += velocities[i]
            # Keep within bounds after update
            h = np.clip(h, 0, 1)
            h = rescale_h(h, dx)
            swarm[i] = h
            
        iteration += 1
    
    # Final evaluation
    final_score = evaluate_h(global_best, dx)
    return (global_best.tolist(), final_score, n_points)
