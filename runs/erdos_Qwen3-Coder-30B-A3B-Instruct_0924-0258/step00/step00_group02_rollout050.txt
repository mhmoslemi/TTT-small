import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h from initial_h_values if available, otherwise create a simple guess
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Simple initial guess: constant function
        h = np.ones(n_points) * 0.5
    
    # Ensure initial h satisfies constraint
    h = h * target_sum / np.sum(h)
    
    # Helper to evaluate C5
    def evaluate_c5(h_vals):
        # Compute correlation manually
        h = np.array(h_vals)
        one_minus_h = 1.0 - h
        # Full correlation (same as correlate with mode='full')
        corr = np.correlate(h, one_minus_h, mode='full')
        # Only take the valid part (k from -n+1 to n-1)
        # The maximum is just the maximum value in the correlation
        return np.max(corr) * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vals):
        h_proj = np.clip(h_vals, 0, 1)
        current_sum = np.sum(h_proj)
        if current_sum > 0:
            h_proj = h_proj * target_sum / current_sum
        return h_proj
    
    # Gradient-based update with constraints
    def update_h(h_vals, learning_rate):
        h = np.array(h_vals)
        # Get maximum overlap index
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        # We need to compute the gradient w.r.t. this specific shift
        # The j-th element of gradient is the derivative of max_k (h * (1-h))[k] w.r.t. h[j] where k is max_idx
        # For k-th entry of correlation of h and (1-h), the gradient w.r.t. h[i] is: (1-h[i+k]) if 0 <= i+k < n, 0 otherwise.
        # which means: gradient[i] = sum over k of (1-h)[i+k] where k = max_idx, i+k in [0,n-1]
        # This corresponds to correlation of 1-h and shifted h
        # Compute the gradient more precisely
        # For k=max_idx in correlation, this is the correlation of h and (1-h) starting at shift k
        # Derivative w.r.t. h[i] of that entry is then (1-h)[i-k], when i-k is in valid range
        grad = np.zeros_like(h)
        corr_shift = max_idx - (n_points - 1)
        for i in range(n_points):
            j = i - corr_shift
            if 0 <= j < n_points:
                grad[i] += (1.0 - h[j])
        # Update
        h_new = h - learning_rate * grad
        # Project
        h_new = project_h(h_new)
        return h_new
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    
    # Optimization loop
    iter_count = 0
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1 and iter_count < 10000:
        iter_count += 1
        # Update step
        h = update_h(h, lr)
        # Check if we improved
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Anneal learning rate
        lr *= 0.9995
        
        # Occasionally, make a random perturbation
        if iter_count % 50 == 0:
            perturb = np.random.uniform(-0.05, 0.05, n_points)
            h_pert = h + perturb
            h_pert = project_h(h_pert)
            c5_pert = evaluate_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                h = h_pert.copy()
    
    return (best_h.tolist(), best_c5, n_points)
