import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_overlap(h, dx):
    # Compute the overlap integral using correlation
    conv = correlate(h, 1 - h, mode='full') * dx
    return np.max(conv)

def make_feasible(h, dx):
    # Ensure h is in [0,1] and sum(h)*dx = 1
    h = np.clip(h, 0, 1)
    s = np.sum(h) * dx
    if s != 0:
        h = h / s * (2.0 / dx)  # Scale to make sum = 1
    return h

def mutate(h, dx, rate=0.1):
    # Add bounded noise and re-normalize
    noise = np.random.normal(0, rate, len(h))
    h_new = h + noise
    return make_feasible(h_new, dx)

def crossover(h1, h2, rate=0.5):
    # Blend two parents
    mask = np.random.random(len(h1)) < rate
    h_new = np.where(mask, h1, h2)
    return h_new

def tournament_selection(population, fitnesses, k=5):
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    dx = 2.0 / 70
    n_points = 70
    
    # Initialize from the parent construction or create a random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.random(n_points) * 0.5  # Slightly biased toward small values
        h = make_feasible(h, dx)
    
    # Always initialize population with at least the starting point
    population = [h.copy()]
    fitnesses = [evaluate_overlap(h, dx)]
    
    # Add diversity with perturbed versions
    for _ in range(19):
        h_perturbed = mutate(h, dx)
        population.append(h_perturbed.copy())
        fitnesses.append(evaluate_overlap(h_perturbed, dx))
    
    # Main loop
    generation = 0
    best_fitness = min(fitnesses)
    best_h = population[np.argmin(fitnesses)].copy()
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Selection
        new_population = []
        new_fitnesses = []
        
        # Keep best individuals
        sorted_indices = np.argsort(fitnesses)
        elite = sorted_indices[:10]
        for idx in elite:
            new_population.append(population[idx].copy())
            new_fitnesses.append(fitnesses[idx])
        
        # Generate rest via selection, crossover, mutation
        while len(new_population) < 50:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            if random.random() < 0.8:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            
            child = mutate(child, dx)
            child_fitness = evaluate_overlap(child, dx)
            
            new_population.append(child)
            new_fitnesses.append(child_fitness)
        
        population = new_population
        fitnesses = new_fitnesses
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
    
    final_fitness = evaluate_overlap(best_h, dx)
    return (best_h.tolist(), final_fitness, n_points)
