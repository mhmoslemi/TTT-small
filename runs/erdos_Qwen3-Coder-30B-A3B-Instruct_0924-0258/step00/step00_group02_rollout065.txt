import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős minimum overlap constant for given h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all shifts
    correlation = signal.correlate(h, 1 - h, mode="full")
    # Scale by dx and find maximum
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available, otherwise start from scratch
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Create a random initial solution
        n_points = 45
        h_init = np.random.rand(n_points)
        h_init = project_to_feasible(h_init, n_points)
        
    # PSO parameters
    n_particles = 30
    max_iter = 1000
    
    # Initialize swarm
    particles = []
    velocities = []
    
    # First particle is the initial solution
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Other particles are small random perturbations
    for _ in range(n_particles - 1):
        h_perturbed = h_init + np.random.uniform(-0.1, 0.1, n_points)
        h_perturbed = project_to_feasible(h_perturbed, n_points)
        particles.append(h_perturbed)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    
    # Evaluate initial particles
    particle_costs = [evaluate_erdos_solution(p) for p in particles]
    pbest = particles.copy()
    pbest_costs = particle_costs.copy()
    gbest_idx = np.argmin(pbest_costs)
    gbest = pbest[gbest_idx].copy()
    gbest_cost = pbest_costs[gbest_idx]
    
    # PSO parameters
    w_start = 0.9
    w_decay = 0.99
    c1 = 2.0
    c2 = 2.0
    
    start_time = time.time()
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Update particles
        w = w_start * (w_decay ** iteration)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) + 
                             c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -0.5, 0.5)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply feasibility projection
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate new cost
            cost = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if cost < pbest_costs[i]:
                pbest[i] = particles[i].copy()
                pbest_costs[i] = cost
                
                # Update global best if needed
                if cost < gbest_cost:
                    gbest = particles[i].copy()
                    gbest_cost = cost
        
        # Report progress occasionally
        if iteration % 100 == 0:
            print(f"Iteration {iteration}, Best C5 = {gbest_cost:.6f}")
    
    # Return final result
    return (gbest, gbest_cost, n_points)
