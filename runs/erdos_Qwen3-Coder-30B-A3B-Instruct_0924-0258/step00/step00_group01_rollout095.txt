import numpy as np
import time
import random
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given h"""
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # Only take the valid lags
    n = len(h)
    valid_corr = corr[n-1:2*n-1]
    return np.max(valid_corr) * dx

def project_to_feasible(h):
    """Project h to satisfy sum(h) = n_points / 2 and h[i] in [0,1]"""
    h = np.clip(h, 0, 1)
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust values
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        # Add diff/n evenly to each
        h = h + diff / n
    else:
        # Need to decrease values
        h = h + diff / n
    return np.clip(h, 0, 1)

def crossover_mutation(parent1, parent2, mutation_rate=0.05, noise_std=0.05):
    """Generate offspring from two parents via crossover and mutation"""
    n = len(parent1)
    # Crossover - average parents
    offspring1 = (parent1 + parent2) / 2
    offspring2 = (parent1 + parent2) / 2
    
    # Mutation - add Gaussian noise
    mask1 = np.random.random(n) < mutation_rate
    offspring1 += np.random.normal(0, noise_std, n) * mask1
    mask2 = np.random.random(n) < mutation_rate
    offspring2 += np.random.normal(0, noise_std, n) * mask2
    
    # Project to feasible region
    offspring1 = project_to_feasible(offspring1)
    offspring2 = project_to_feasible(offspring2)
    
    return offspring1, offspring2

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select one individual using tournament selection"""
    selected_idx = random.randint(0, len(population) - 1)
    for _ in range(tournament_size - 1):
        candidate_idx = random.randint(0, len(population) - 1)
        if fitnesses[candidate_idx] < fitnesses[selected_idx]:
            selected_idx = candidate_idx
    return population[selected_idx]

def run(seed=42, budget_s=600, n_points=47, population_size=20):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial h from parent construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0)
    else:
        # Generate random valid initial h
        h0 = np.random.random(n_points)
        h0 = project_to_feasible(h0)
    
    # Population with initial h plus random candidates
    population = [h0]
    for _ in range(population_size - 1):
        h_rand = np.random.random(n_points)
        h_rand = project_to_feasible(h_rand)
        population.append(h_rand)
    
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        # Evaluate population
        fitnesses = []
        for h in population:
            fit = evaluate_h(h, dx)
            fitnesses.append(fit)
        
        # Track best solution
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitnesses.index(min_fitness)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Selection + reproduction
        new_population = []
        # Elitism: keep best solution
        new_population.append(best_h.copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover_mutation(parent1, parent2)
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:population_size]
        generation += 1
        
    return (best_h, best_c5, n_points)
