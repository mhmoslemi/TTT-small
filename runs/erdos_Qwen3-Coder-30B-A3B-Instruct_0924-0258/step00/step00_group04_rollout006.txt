import numpy as np
from scipy.optimize import minimize
import time
from copy import deepcopy

def evaluate_c5(h, dx):
    """Compute C5 bound for given h."""
    n = len(h)
    # correlate h with (1-h) in full mode
    # note: np.correlate with mode='full' is equivalent to convolution with flipped kernel
    # (1-h) has the same shape as h
    corr = np.correlate(h, 1-h, mode="full")
    # only consider valid shifts [0, 1, ..., n-1]
    valid_corr = corr[n-1:2*n-1]
    return np.max(valid_corr) * dx

def rescale_to_constraint(h, target_sum):
    """Ensure h sums to target_sum (keeping values in [0,1])."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    new_h = h * target_sum / current_sum
    return np.clip(new_h, 0, 1)

def mutate_individual(h, mutation_rate=0.05):
    """Apply mutation to a single individual."""
    n = len(h)
    mutation_indices = np.random.choice(n, int(n * mutation_rate), replace=False)
    for i in mutation_indices:
        h[i] += np.random.normal(0, 0.1)
    # project and rescale
    h = np.clip(h, 0, 1)
    h = rescale_to_constraint(h, len(h) / 2)
    return h

def crossover(parent1, parent2, n_crossover=2):
    """Perform crossover between two parents, producing a child."""
    n = len(parent1)
    # choose crossover points (at least 2)
    if n_crossover < 2:
        n_crossover = 2
    if n_crossover > n-1:
        n_crossover = n-1
    points = np.sort(np.random.choice(n-1, n_crossover, replace=False)) + 1
    points = np.concatenate(([0], points, [n]))
    
    child = np.zeros(n)
    start = 0
    take_parent2 = False
    for i, pt in enumerate(points[1:]):
        if take_parent2:
            child[start:pt] = parent2[start:pt]
        else:
            child[start:pt] = parent1[start:pt]
        start = pt
        take_parent2 = not take_parent2
    # enforce constraints
    child = rescale_to_constraint(child, n / 2)
    return child

def local_optimize(h, dx, maxiter=200):
    """Use L-BFGS to improve a solution."""
    n = len(h)
    def objective(h_flat):
        h_reconstructed = h_flat.reshape(n)
        return evaluate_c5(h_reconstructed, dx)
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n / 2
    
    def constraint_bounds(h_flat):
        return np.concatenate([h_flat, 1-h_flat])  # h >= 0, h <= 1

    # Initialize with current solution
    initial_h_flat = h.flatten()

    # Construct bounds (0 to 1 for each element)
    bounds = [(0, 1) for _ in range(n)]
    # Provide a simpler constraint using scipy -
    # we'll do a gradient-based minimizing approach with constraints
    # This trick uses scipy to minimize C5, keeping everything in [0,1]
    result = minimize(objective, initial_h_flat, method='L-BFGS-B', bounds=bounds, options={'maxiter': maxiter})
    
    # clean up the result to meet constraints exactly
    optimized_h = result.x.reshape(n)
    optimized_h = np.clip(optimized_h, 0, 1)
    optimized_h = rescale_to_constraint(optimized_h, n / 2)
    return optimized_h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # problem setup
    n_points = 42  # Use initial_h_values size unless changing
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    population = []
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if h0.shape[0] != n_points:
            # resize
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
    else:
        h0 = np.random.uniform(0, 1, n_points)
    
    h0 = rescale_to_constraint(h0, n_points / 2)
    population.append(h0.copy())
    
    for i in range(pop_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = rescale_to_constraint(h, n_points / 2)
        population.append(h.copy())

    best_solution = None
    best_c5 = float('inf')
    stagnation_counter = 0
    mutation_rate = 0.05
    
    gen = 0
    while time.time() - start_time < budget_s - 1 and gen < 300:
        gen += 1
        
        # Evaluate all
        fitnesses = []
        for ind in population:
            c5 = evaluate_c5(ind, dx)
            fitnesses.append(c5)
        
        # Track best solution
        min_idx = np.argmin(fitnesses)
        min_c5 = fitnesses[min_idx]
        if min_c5 < best_c5:
            best_c5 = min_c5
            best_solution = deepcopy(population[min_idx])
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            if stagnation_counter > 50:
                mutation_rate = 0.1
        
        # Selection
        sorted_indices = np.argsort(fitnesses)
        parents = [population[i] for i in sorted_indices[:pop_size//5]]

        # Crossover and Mutation
        offspring = []
        while len(offspring) < pop_size - len(parents):
            p1, p2 = np.random.choice(parents, size=2, replace=False)
            child = crossover(p1, p2, 2)
            child = mutate_individual(child, mutation_rate)
            offspring.append(child)
            
        # Local optimization on best
        top_k = population[sorted_indices[:pop_size//10]]
        for i in range(len(top_k)):
            opt = local_optimize(top_k[i], dx)
            offspring.append(opt)
            
        # Create new population
        population = parents + offspring[:pop_size - len(parents)]
        # Keep diversity
        if len(population) == 0:
            population = [h0]
    
    # Final check and return
    final_c5 = evaluate_c5(best_solution, dx)
    return (best_solution, final_c5, n_points)
