import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h with the provided initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Fallback initialization
        n_points = 70
        h = np.full(n_points, 0.5)
        h[0] = 0.4
        h[1] = 0.6
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    max_iter = 500
    alpha = 0.1
    tol = 1e-4
    last_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlations for all lags
        corr = correlate(h, 1 - h, mode='full')
        c5_val = np.max(corr) * dx
        
        # Find the set of lags that achieve the maximum
        max_lags = np.where(corr == np.max(corr))[0]
        # Convert to relative indices in [-n+1, n-1]
        lags = max_lags - (n_points - 1)
        
        # Compute the gradient
        grad = np.zeros(n_points)
        for k in lags:
            if k >= 0:
                # h[i] contributes to overlap with (1-h[i+k])
                for i in range(n_points - k):
                    grad[i] += -dx
                for i in range(k, n_points):
                    grad[i] += -dx
            else:
                # k < 0, so it's (1-h[i]) contributes to overlap with h[i+k]
                for i in range(-k, n_points):
                    grad[i] += -dx
                for i in range(n_points + k):
                    grad[i] += -dx
        
        # Updated h
        h_new = h - alpha * grad
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        
        # Ensure sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update h
        h = h_new
        
        # Check convergence
        if abs(last_c5 - c5_val) < tol:
            break
        last_c5 = c5_val

    # Final evaluation
    corr = correlate(h, 1 - h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (h.tolist(), c5_bound, n_points)
