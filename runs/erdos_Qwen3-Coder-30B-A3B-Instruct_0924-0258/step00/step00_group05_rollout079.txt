import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    # Compute all overlaps
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integral is the max of the correlation values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction if available

        n_points = len(h0)
    except NameError:
        # Fallback construction: piecewise constant with 70 points

        h0 = np.ones(n_points) * 0.5
        h0[:n_points//4] = 1.0
        h0[n_points//4:3*n_points//4] = 0.5
        h0[3*n_points//4:] = 0.0

    dx = 2.0 / n_points
    
    # Binary search on C5
    low = 0.0
    high = 0.38085
    tolerance = 1e-4
    start_time = time.time()
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0, n_points)
    
    while high - low > tolerance and time.time() - start_time < budget_s - 1:
        mid = (low + high) / 2.0
        
        # Check feasibility using convex optimization
        h_vars = cp.Variable(n_points)
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2.0
        ]
        
        # Add overlap constraints
        for k in range(1, n_points):
            # The overlap for shift k: sum h[i] * (1 - h[i+k]) for valid i
            # Convert to a linear constraint
            corr = 0.0
            for i in range(n_points - k):
                corr += h_vars[i] * (1 - h_vars[i+k])
            constraints.append(corr <= mid / dx)
            
        # No objective needed, just feasibility check
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False, timelimit=budget_s - (time.time() - start_time) - 1)
            if prob.status == cp.OPTIMAL or prob.status == cp.SOLVER_ERROR:
                # Solution is feasible, try to improve h
                # Check if we got a new better solution
                if prob.status == cp.OPTIMAL:
                    h_new = h_vars.value
                    c5_new = evaluate_erdos_solution(h_new, n_points)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                high = mid
            else:
                low = mid
        except:
            low = mid

    # Final refinement with gradient-based optimization
    try:
        # Use scipy for better optimization
        from scipy.optimize import minimize
        
        def obj(h):
            # We want to minimize the overlap, so return negative of the max overlap
            # But with the constraint we are basically ensuring any feasible solution
            # has overlap <= C5
            corr_full = np.correlate(h, 1 - h, mode="full")
            return -np.max(corr_full)
            
        def constraint_func(h):
            return np.sum(h) - n_points / 2.0  # Sum == 1
            
        cons = {'type': 'eq', 'fun': constraint_func}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-6})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            c5_final = evaluate_erdos_solution(h_final, n_points)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    # Ensure final solution is within bounds
    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
