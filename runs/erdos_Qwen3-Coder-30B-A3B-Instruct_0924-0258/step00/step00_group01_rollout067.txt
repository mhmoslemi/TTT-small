import numpy as np
import time
from scipy.fft import fft, ifft
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 47 if 'initial_h_values' in globals() else 50
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h.copy()
        # Ensure constraints
        h_sum = np.sum(h)
        if abs(h_sum - n_points/2.0) > 1e-6:
            h = h * (n_points/2.0) / h_sum
    else:
        # Start with uniform function
        h = np.ones(n_points) * 0.5
        h = h * (n_points/2.0) / np.sum(h)
        
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Pre-compute binary masks for cross-correlations
    B_k = {}
    B_negk = {}
    for k in range(n_points):
        B_k[k] = np.zeros(n_points)
        B_negk[k] = np.zeros(n_points)
        for i in range(n_points):
            if i + k < n_points:
                B_k[k][i] = 1
            if i - k >= 0:
                B_negk[k][i] = 1
    
    alpha = 0.1
    last_improvement = 0
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Compute overlaps using FFT for efficiency
        # Overlap at lag k = sum(h[i] * (1 - h[i+k])) 
        # = sum(h[i]) - sum(h[i]*h[i+k])
        h_fft = fft(h)
        one_minus_h = 1 - h
        one_minus_h_fft = fft(one_minus_h)
        # Cross correlation via FFT
        cc = ifft(h_fft * np.conj(one_minus_h_fft)).real
        # We want max over all lags
        overlaps = np.concatenate([cc[-(n_points-1):], cc[:n_points]])
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        if max_overlap <= best_C5:
            # No improvement, possibly downhill
            pass
        else:
            best_C5 = max_overlap
            best_h = h.copy()
        
        # Update h in the direction of largest overlap
        # For high accuracy, we can just use simple nn update
        # but to save time, we directly compute gradient
        
        # Gradient: (1 - h[k]) * B_k - h * B_{-k}
        if max_k >= n_points:
            lag = max_k - n_points
        else:
            lag = max_k
            
        if lag == 0:
            # Gradient is h * 0 - h * 0 = 0. This is a saddle point; use random walk
            h += np.random.normal(0, alpha * 0.01, n_points)
        else:
            grad = np.zeros(n_points)
            for i in range(n_points):
                # For lag k, we look at h[i+k] for the 'a' in 'h[i]*h[i+k]'
                if i + lag < n_points:
                    grad[i] += (1 - h[i+lag])  # Contribution from positive correlation
                # For the h[i] * (1 - h[i+k]) part
                if i - lag >= 0:
                    grad[i] -= h[i-lag]  # Contribution to negative lags

            # Gradient has wrong sign for minimization
            grad *= -alpha
            
            # Clip gradient
            grad = np.clip(grad, -1.0, 1.0)
            h += grad
            
            # Project h into [0, 1]
            h = np.clip(h, 0, 1)
            
            # Adjust sum to be exactly n_points/2
            h_sum = np.sum(h)
            if abs(h_sum - n_points/2.0) > 1e-6:
                h = h * (n_points/2.0) / h_sum 
        
        # Adaptive learning rate
        if iter_count % 50 == 0:
            if last_improvement == iter_count:
                alpha *= 0.95
                if alpha < 1e-6:
                    alpha = 1e-6
            last_improvement = iter_count
            
        # Checker for escape from local minima
        if np.random.random() < 0.05:
            h += np.random.normal(0, alpha * 0.05, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points/2.0) > 1e-6:
                h = h * (n_points/2.0) / h_sum
          
    # Final re-evaluation
    final_C5 = evaluate_C5(best_h, dx)
    return (best_h.tolist(), final_C5, n_points)

def evaluate_C5(h, dx):
    n = len(h)
    h = np.array(h)
    # Use scipy.signal.correlate for validation
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_corr = np.max(corr)
    return max_corr
