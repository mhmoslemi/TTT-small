import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe algorithm parameters
    max_iter = 1000
    tolerance = 1e-6
    learning_rate = 0.9
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # Leave some buffer
            break
            
        # Compute cross-correlations
        corr_full = correlate(h, 1 - h, mode='full')
        # Take only positive lags (full correlation has negative lags too)
        corr_positive = corr_full[len(corr_full)//2:]
        
        # Find maximum cross-correlation
        max_k_idx = np.argmax(corr_positive)
        max_corr = corr_positive[max_k_idx] * dx
        
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
        
        # Compute gradient of max_corr w.r.t. h
        # For correlation h * (1 - h[k]), gradient wrt h[i] is (1-h[k+i]) (when k+i is valid)
        # We'll set up the subproblem as looking for the direction of improvement
        grad = np.zeros(n_points)
        
        # Build gradient based on correlation structure
        for i in range(n_points):
            if i + max_k_idx < n_points:
                grad[i] = -(1 - h[i + max_k_idx])
        
        # Frank-Wolfe step: find vertex s that maximizes <grad, s>
        # Select top indices based on gradient values
        sorted_indices = np.argsort(-grad)
        s_indices = sorted_indices[:n_points // 2]
        
        # Create vertex s (sparse vector with 1s at s_indices)
        s = np.zeros(n_points)
        s[s_indices] = 1
        s = s * (n_points / 2) / np.sum(s)  # normalize to correct sum
        s = np.clip(s, 0, 1)
        
        # Line search for step size gamma
        gamma_candidates = [0.1, 0.25, 0.5, 0.75, 1.0]
        best_gamma = 0.0
        min_c5 = max_corr
        
        for gamma in gamma_candidates:
            new_h = h + gamma * (s - h)
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)  # Fix sum
            
            # Compute new C5
            new_corr = correlate(new_h, 1 - new_h, mode='full')
            new_corr_pos = new_corr[len(new_corr)//2:]
            new_c5 = np.max(new_corr_pos) * dx
            
            if new_c5 < min_c5:
                min_c5 = new_c5
                best_gamma = gamma
        
        # Update using the best gamma
        if best_gamma > 0:
            h = h + best_gamma * (s - h)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)  # Fix sum again
        else:
            # No improvement, back to best
            h = best_h.copy()
            # Increase learning rate slightly to encourage more exploration?
            learning_rate *= 1.01
            
        if abs(max_corr - min_c5) < tolerance:
            break
    
    # Final evaluation
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_corr_pos = final_corr[len(final_corr)//2:]
    final_c5 = np.max(final_corr_pos) * dx
    
    return (best_h.tolist(), final_c5, n_points)
