import numpy as np
from scipy import signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Fallback: construct a simple initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    n_points = len(h)
    dx = 2.0 / n_points
    max_iter = 10000
    learning_rate = 0.01
    decay_factor = 0.95
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Compute correlations for all lags
        corr = np.correlate(h, 1 - h, mode="full")
        k0 = np.argmax(corr)
        max_corr = corr[k0]
        
        # Compute gradient (approximate)
        grad = np.zeros(n_points)
        if k0 < n_points:
            grad[:n_points-k0] += (1 - h[k0:])
        else:
            grad[k0-n_points:] += (1 - h[:n_points])
            
        # Update with gradient
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto feasible set using convex optimization
        # Variables
        x = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            x >= 0,
            x <= 1,
            cp.sum(x) == n_points / 2.0
        ]
        
        # Objective (just to form problem, not used for projection directly)
        obj = cp.Minimize(0)
        prob = cp.Problem(obj, constraints)
        
        # Solve projection
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h_projected = x.value
        except:
            # Fallback projection if optimization fails
            h_projected = h_new.copy()
            h_projected = np.clip(h_projected, 0, 1)
            h_projected = h_projected * (n_points / 2.0) / np.sum(h_projected)
            
        h = h_projected
        
        # Update best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Decay learning rate
        if i % 100 == 0:
            learning_rate *= decay_factor
            
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
