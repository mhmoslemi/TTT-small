def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # parameters

    target_sum = n_points / 2.0
    
    # initial solution

        # fallback starting point
        h = np.ones(n_points) * 0.5
    
    # normalize to target sum
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # learning rate
    lr = 0.01
    decay = 0.999
    iterations = 0
    
    while time() - start_time < budget_s - 1:
        iterations += 1
        
        # compute correlations
        corr = np.correlate(h, 1-h, mode="full")
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        
        # compute gradient
        # computing gradient of h[i] w.r.t. overlap at max_k
        # For correlation h*(1-h), gradient around k=0 is (1-2*h)
        # Correlation gradient: estimate it as derivative of shift-correlation
        # But simpler: Use gradient from linear approximation to overwrite
        
        # compute local gradients from centers
        gradient = np.zeros_like(h)
        stride = 1
        
        # compute gradient at max_k via finite differences on shifted h
        for i in range(n_points):
            # shift h slightly
            h_plus = h.copy()
            h_plus[i] += 1e-4
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * target_sum / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1-h_plus, mode="full")
            corr_orig = np.correlate(h, 1-h, mode="full")
            
            gr = (corr_plus[max_k] - corr_orig[max_k]) / 1e-4
            gradient[i] = gr
        
        # update step
        h = h - lr * gradient
        
        # projection to feasible set
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        # compute new C5
        c5 = evaluate_erdos_solution(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        lr *= decay

        # early stop if we're close enough
        if best_c5 < 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
