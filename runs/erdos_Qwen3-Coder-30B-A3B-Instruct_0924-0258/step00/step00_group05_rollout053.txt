import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initial solution from parent
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Create initial step function
        h_init = np.ones(n_points) * 0.5
        # Slightly perturb for diversity
        h_init = np.clip(h_init + np.random.normal(0, 0.05, n_points), 0, 1)
    
    # Normalize initial solution
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Binary search parameters
    low = 0.0
    high = 0.38085
    precision = 1e-4
    best_h = h_init.copy()
    best_c5 = high
    
    while time.time() - start_time < budget_s - 1 and high - low > precision:
        mid = (low + high) / 2.0
        
        # Create variables
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Overlap constraints
        for k in range(n_points):
            # Compute convolution term manually to avoid full mode overhead
            # This is for shift k, so sum h[i] * (1 - h[i + k]) for valid indices
            # This uses a tridiagonal type approach since h is a step function
            # We compute the maximum overlap over all shifts k
            # For efficiency, we're using numpy
            overlap_sum = 0
            valid_count = 0
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    overlap_sum += h[i] * (1 - h[j])
                    valid_count += 1
            if valid_count > 0:
                constraints.append(overlap_sum <= mid / dx)
        
        # Objective (dummy since we don't optimize any objective, just satisfy constraints)
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLOP, verbose=False, time_limit=budget_s - (time.time() - start_time))
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                # Solution is feasible, update best and reduce upper bound
                h_sol = h.value
                if h_sol is not None:
                    # Check the actual overlap
                    max_overlap = 0
                    for k in range(n_points):
                        overlap = 0
                        for i in range(n_points):
                            j = i + k
                            if j < n_points:
                                overlap += h_sol[i] * (1 - h_sol[j])
                        max_overlap = max(max_overlap, overlap * dx)
                    
                    if max_overlap <= mid:
                        high = mid
                        best_c5 = max_overlap
                        best_h = h_sol.copy()
                        if high - low < precision:
                            break
                    else:
                        low = mid
            else:
                # Problem is infeasible: increase lower bound
                low = mid
        except:
            low = mid

    # Final evaluation of best solution
    max_overlap = 0
    for k in range(n_points):
        overlap = 0
        for i in range(n_points):
            j = i + k
            if j < n_points:
                overlap += best_h[i] * (1 - best_h[j])
        max_overlap = max(max_overlap, overlap * dx)
    
    # Return final results
    return (best_h.tolist(), max_overlap, n_points)
