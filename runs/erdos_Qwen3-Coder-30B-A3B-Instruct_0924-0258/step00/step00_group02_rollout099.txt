def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute overlaps using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # Scale by dx for proper integration
    overlap_scaled = overlap * dx
    # Return the maximum overlap value
    return np.max(overlap_scaled)

def project_to_feasible(h, n_points):
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def convex_surrogate(h, n_points):

    dx = 2.0 / n_points
    h = np.array(h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Quadratic approximation of max
    return np.sum(conv**2) * dx**2

        return np.concatenate([h, 1 - h])  # lower and upper bounds

    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint for sum
    constraints = [{'type': 'eq', 'fun': constraint_sum}]

        h_opt = res.x

        h_opt = h0
