import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, n_points):
    # Evaluate the overlap integral for given h
    dx = 2.0 / n_points
    # Compute all correlations for all lags
    # correlate with 1-h (reversed)
    corr = correlate(h, 1 - h, mode='full')
    # The max should be at middle + 1 to n_points (or equivalently, between n_points and 2*n_points-1)
    max_k = np.argmax(corr)
    # But due to mode="full", for length n_points, we have 2*n_points - 1 convolved values
    # The valid range for k is from 0 to n_points - 1
    # and the overlap integral corresponds to values from n_points - 1 to 2*n_points-2
    # Actually, we want values from index n_points - 1 onward which correspond to k = 0 to n_points - 1
    # But correlated returns from -n_points+1 to n_points-1 (index 0 to 2*n_points-2)
    # so the value for k=0 is at index n_points - 1 (the middle element)
    # Value for k=1 is at index n_points (shifted by n_points-1)
    # So for k=0, it's index n_points-1, k=1 is index n_points, etc.
    # Therefore we take values from n_points-1 to 2*n_points-2 (which is indices after middle element)
    # But k=0,1,...,n_points-1 means we look at index n_points-1 through 2*n_points-2
    # The overlap integral for k is proportional to corr[n_points-1 + k]
    # However, for JSON convention and stability, just get max throughout the full range
    # which we already did (index max_k)
    # The evaluation needs the maximum of the convolution scaled by dx
    max_corr_value = corr[max_k]
    # the peak overlap value in the full correlation
    return max_corr_value * dx
