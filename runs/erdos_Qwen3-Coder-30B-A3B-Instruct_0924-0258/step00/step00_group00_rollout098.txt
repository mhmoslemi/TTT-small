import numpy as np
import time
from scipy import correlate

def evaluate_h(h_values, dx):
    """Evaluate the maximum overlap integral for given h values."""
    corr = correlate(h_values, 1 - h_values, mode='full')
    # We're interested in the maximum value across all possible lags
    # The overlap integral involves convolving h with (1-h)
    # For the full convolution, we want to look at the maximum of the correlation
    # The valid range of lags corresponds to the symmetric extension
    # But in this case, taking max of all values gives us the desired result
    return np.max(corr) * dx

def rescale_h(h):
    """Rescale h to satisfy the constraint sum(h) = n_points / 2."""
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Adjust all values to match the target sum
    # We can simply scale the whole vector and then clip to [0,1]
    scale_factor = target_sum / current_sum
    h_scaled = h * scale_factor
    h_clipped = np.clip(h_scaled, 0, 1)
    
    # Rescale again to correct the sum
    new_sum = np.sum(h_clipped)
    if abs(new_sum) < 1e-10:
        return h_clipped
    h_rescaled = h_clipped * (target_sum / new_sum)
    return np.clip(h_rescaled, 0, 1)

def mutate_h(h, mutation_rate=0.1, delta=0.1):
    """Mutate h with given probability and magnitude."""
    h_new = h.copy()
    for i in range(len(h_new)):
        if np.random.rand() < mutation_rate:
            h_new[i] += np.random.uniform(-delta, delta)
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Rescale to maintain sum constraint
    h_new = rescale_h(h_new)
    return h_new

def crossover_h(h1, h2):
    """Perform one-point crossover between two h vectors."""
    n = len(h1)
    point = np.random.randint(1, n)
    h_new = np.concatenate([h1[:point], h2[point:]])
    # Rescale to maintain sum constraint
    h_new = rescale_h(h_new)
    return h_new

def tournament_selection(fitnesses, k=3):
    """Select index of best candidate from k random choices."""
    indices = np.random.choice(len(fitnesses), size=k, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return best_idx
