import numpy as np
from scipy import optimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except:
        # Default initialization
        n_points = 70
        h0 = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Binary search on C5
    low = 0.0
    high = 0.38085
    tolerance = 1e-4
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    
    while high - low > tolerance and time.time() - start_time < budget_s - 5:
        mid = (low + high) / 2.0
        
        # Try to solve with current C5 bound
        try:
            h_opt = solve_relaxed_problem(h0, n_points, dx, mid)
            if h_opt is not None:
                c5_val = evaluate_erdos_solution(h_opt)
                if c5_val < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_val
                high = mid
            else:
                low = mid
        except:
            low = mid
    
    # Final evaluation and refinement
    try:
        h_final = solve_relaxed_problem(best_h, n_points, dx, high)
        if h_final is not None:
            c5_final = evaluate_erdos_solution(h_final)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    except:
        pass
    
    return (best_h, best_c5, n_points)

def solve_relaxed_problem(h0, n_points, dx, c5_bound):
    # Simple approach: start with h0, then do box constrained optimization
    # We're maximizing sum(h[i]*(1-h[i+k])) subject to bounds and sum(h)=1
    
    # Using scipy.optimize with bounds and constraints
    def objective(h):
        # This is actually minimizing - we want to minimize the maximum overlap
        # so we minimize the (largest) correlation value
        h_array = np.array(h)
        conv = np.correlate(h_array, 1 - h_array, mode="full")
        # Maximum of the full convolution 
        return -np.max(conv) / dx  # Negative because we minimize
    
    def constraint_sum(h):
        return np.sum(h) * dx - 1.0
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Choose optimizer parameters
    try:
        result = optimize.minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=constraints,
                                   options={'maxiter': 100, 'ftol': 1e-6})
        if result.success:
            h_result = result.x
            # Project to ensures bounds and normalization
            h_result = np.clip(h_result, 0, 1)
            h_result = h_result * (1.0 / np.sum(h_result) * n_points * dx)  # Adjust to make integral = 1
            return h_result
    except:
        pass
    
    return None
