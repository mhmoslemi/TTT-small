import numpy as np
from scipy import signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[0] = 0.2
        h[-1] = 0.2
        h[1] = 0.8
        h[-2] = 0.8
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)  # scale to exact sum
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Precompute valid (i, j) pairs for all lag k
    valid_pairs = {}
    for k in range(n_points):
        pairs = []
        for i in range(n_points):
            j = i + k
            if j < n_points:
                pairs.append((i, j))
        valid_pairs[k] = pairs
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 5:
        iteration += 1
        
        # Evaluate current C5
        overlaps = np.correlate(h, 1-h, mode="full")
        c5_current = np.max(overlaps) * dx
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Solve convex relaxation
        # Variables
        x = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(x) == target_sum,
            x >= 0, x <= 1,
            C >= 0
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            pairs = valid_pairs[k]
            if not pairs:
                continue
            # Constraint: sum_{i,j} h[i] - h[i] * h[j] <= C * dx
            # which is same as sum_{i,j} h[i] - h[i] * h[j] <= C * dx
            # write as sum(h[i]) - sum(h[i] * h[j]) <= C * dx
            # where j=i+k for valid pairs
            sum_h_i = cp.sum([x[i] for i, j in pairs])
            sum_h_i_h_j = cp.sum([x[i] * x[j] for i, j in pairs])
            # Add constraint for all k: sum(h[i]) - sum(h[i]*h[j]) <= C * dx
            # i.e. sum(h[i]) - sum(h[i]*h[j]) - C * dx <= 0
            constraints.append(sum_h_i - sum_h_i_h_j - C * dx <= 0)
            
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = np.array(x.value).clip(0, 1)
                if np.abs(np.sum(h_new) - target_sum) < 1e-6:
                    h = h_new
                else:
                    # fallback
                    h = np.clip(h + 0.05 * (np.random.rand(n_points) - 0.5), 0, 1)
                    h = h * target_sum / np.sum(h)
            else:
                # fallback to random perturbation
                h = np.clip(h + 0.05 * (np.random.rand(n_points) - 0.5), 0, 1)
                h = h * target_sum / np.sum(h)
        except:
            # fallback to simple perturbation
            h = np.clip(h + 0.02 * (np.random.rand(n_points) - 0.5), 0, 1)
            h = h * target_sum / np.sum(h)
            
        if iteration % 10 == 0:
            print(f"Iteration {iteration}, C5 = {c5_current:.6f}")
    
    # Final evaluation
    overlaps = np.correlate(best_h, 1-best_h, mode="full")
    final_c5 = np.max(overlaps) * dx
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_c5, len(best_h))
