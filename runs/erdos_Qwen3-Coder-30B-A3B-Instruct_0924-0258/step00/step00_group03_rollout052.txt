import numpy as np
from scipy import signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure initial h satisfies constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization parameters
    learning_rate = 0.01
    decay_factor = 0.95
    max_iter = 10000
    iter_count = 0
    start_time = time.time()
    
    # Gradient descent with projection
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute convolution and find maximum
        conv = signal.correlate(h, 1 - h, mode='full')
        k0 = np.argmax(conv)
        max_val = conv[k0]
        
        # Compute gradient w.r.t h for maximum convolution
        grad = np.zeros_like(h)
        for j in range(n_points):
            if k0 < n_points:
                if j + k0 < n_points:
                    grad[j] += (1 - h[j + k0])
            else:
                if j >= k0 - n_points:
                    grad[j] += h[j - (k0 - n_points)]
        
        # Update
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto feasible set
        h_projected = project_onto_feasible(h_new, n_points, dx)
        
        # Update h if it improves objective
        c5_new = evaluate_erdos_solution(h_projected)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_projected.copy()
            
        h = h_projected
        
        # Decay learning rate
        if i % 100 == 0:
            learning_rate *= decay_factor
        
        iter_count += 1
        
    return best_h.tolist(), best_c5, n_points

def project_onto_feasible(h, n_points, dx):
    """Project h onto feasible set: 0 <= h[i] <= 1 and sum(h) * dx = 1"""
    # Create optimization problem
    h_var = cp.Variable(n_points)
    
    # Objective: minimize ||h_var - h||^2
    objective = cp.Minimize(cp.sum_squares(h_var - h))
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) * dx == 1
    ]
    
    # Solve problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        return h_var.value
    else:
        # If optimization fails, fallback to simple projection
        h_proj = np.clip(h, 0, 1)
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        return h_proj

def evaluate_erdos_solution(h):
    """Evaluate C5 for the given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx
