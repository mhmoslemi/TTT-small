import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for a given step function h."""
    dx = 2.0 / len(h)
    # Compute the correlation: h(x) * (1 - h(x + k))
    overlap = signal.correlate(h, 1 - h, mode="full") * dx
    # Return maximum overlap
    return np.max(overlap)

def project_to_feasible(h, n_points):
    """Ensure h is feasible: 0 ≤ h[i] ≤ 1 and sum(h) = n_points / 2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.full(n_points, 1.0 / 2)
    return h

def mutate_h(h, n_points, mutation_rate=0.1):
    """Apply mutation to a step function h."""
    h_new = h.copy()
    num_mutate = int(mutation_rate * n_points)
    indices = np.random.choice(n_points, num_mutate, replace=False)
    for i in indices:
        h_new[i] += 0.1 * np.random.randn()
    return project_to_feasible(h_new, n_points)

def crossover_h(parent1, parent2, n_points):
    """Perform single-point crossover between two parent step functions."""
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return project_to_feasible(child1, n_points), project_to_feasible(child2, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set parameters
    n_points = 45
    pop_size = 50
    num_generations = 12
    mutation_rate = 0.1
    elite_size = 25
    
    # Initialize population with a good starting solution
    if 'initial_h_values' in globals():

    else:
        h0 = np.ones(n_points) * 0.5

    # Create initial population
    population = [h0]
    for _ in range(pop_size - 1):
        h_perturbed = h0 + np.random.normal(0, 0.1, n_points)
        h_perturbed = project_to_feasible(h_perturbed, n_points)
        population.append(h_perturbed)
    
    # Initialize best solution
    best_h = h0
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Evolution loop
    for gen in range(num_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals in the population
        fitness_scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness_scores.append(c5)
            
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if fitness_scores[0] < best_c5:
            best_h = population[0]
            best_c5 = fitness_scores[0]
            
        # Select elite individuals
        selected_parents = population[:elite_size]
        
        # Generate offspring through crossover and mutation
        offspring = []
        while len(offspring) < pop_size - elite_size:
            parent1, parent2 = np.random.choice(selected_parents, 2, replace=False)
            child1, child2 = crossover_h(parent1, parent2, n_points)
            
            child1 = mutate_h(child1, n_points, mutation_rate)
            child2 = mutate_h(child2, n_points, mutation_rate)
            
            offspring.extend([child1, child2])

        # Keep only the required number of offspring
        offspring = offspring[:pop_size - elite_size]
        
        # Create new population
        population = selected_parents + offspring

    # Final evaluation of best solution
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure that the returned h satisfies all constraints
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    
    return (best_h.tolist(), best_c5, n_points)
