import numpy as np
from scipy import optimize
import time
from math import sqrt

def evaluate_overlap(h, dx):
    n = len(h)
    # Compute all overlap integrals
    overlap = np.correlate(h, 1 - h, mode='full')
    # Only consider the relevant part [0, n-1] for k=0 to n-1
    # But we want to bend the indices for circular shift
    # Instead, compute the actual overlaps with circular shift:
    max_overlap = 0
    for k in range(n):
        # h[i] * (1 - h[(i+k)%n])
        current_overlap = 0
        for i in range(n):
            current_overlap += h[i] * (1 - h[(i+k) % n])
        current_overlap *= dx
        max_overlap = max(max_overlap, current_overlap)
    return max_overlap

def project_h(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_overlap(best_h, dx)
    
    # Optimizer parameters
    lr = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    last_improvement = 0
    patience = 10
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        
        # Compute current overlaps
        overlaps = []
        for k in range(n_points):
            overlap = 0.0
            for i in range(n_points):
                overlap += best_h[i] * (1 - best_h[(i+k) % n_points])
            overlaps.append(overlap * dx)
            
        # Find top contributing k values (top 5 for compute efficiency)
        top_ks = np.argsort(overlaps)[-5:][::-1]  # Top 5 indices
        
        # Compute gradient with respect to top k contributions
        grad = np.zeros(n_points)
        for k in top_ks:
            for i in range(n_points):
                # Derivative of h[i] * (1 - h[(i+k)%n]) w.r.t. h[i]
                grad[i] += (1 - best_h[(i+k) % n_points]) * dx
                # Derivative w.r.t. h[(i+k)%n] -> negative sign
                grad[(i+k) % n_points] -= best_h[i] * dx
        
        # Update velocity with momentum
        velocity = momentum * velocity - lr * grad
        
        # Update h
        new_h = best_h + velocity
        new_h = project_h(new_h, n_points)
        
        # Evaluate
        new_c5 = evaluate_overlap(new_h, dx)
        
        # Accept if better
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            last_improvement = 0
            lr = max(0.001, lr)  # Reset learning rate on improvement
        else:
            last_improvement += 1
            # Reduce learning rate if no improvement
            lr *= 0.995
            
        # Adaptive restart
        if last_improvement > patience:
            # Perturb the solution
            noise = np.random.normal(0, 0.001, n_points)
            new_h = best_h + noise
            new_h = project_h(new_h, n_points)
            new_c5 = evaluate_overlap(new_h, dx)
            if new_c5 < best_c5:

                best_c5 = new_c5
