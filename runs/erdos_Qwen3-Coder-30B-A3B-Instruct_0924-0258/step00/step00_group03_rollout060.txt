import numpy as np
import time
from scipy.optimize import minimize
from math import sqrt

def compute_fitness(h):
    """Compute the maximum overlap integral for the given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1-h, mode='full')
    # Extract the valid correlations (non-zero shifts)
    start = n - 1
    end = 2 * n - 2
    valid_corr = corr[start:end]
    # Return maximum overlap
    return np.max(valid_corr) * dx

def project(h):
    """Project h to satisfy constraints: 0 <= h[i] <= 1, sum(h) = n/2."""
    # Clamp values
    h = np.clip(h, 0, 1)
    # Normalize sum to n/2
    current_sum = np.sum(h)
    target_sum = len(h) / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def mutate(h, sigma):
    """Add Gaussian noise and project."""
    h_new = h + np.random.normal(0, sigma, len(h))
    return project(h_new)

def crossover(h1, h2, alpha=0.6):
    """Blend crossover with alpha in [0.5,1.0]."""
    beta = np.random.uniform(alpha, 1 - alpha)
    h_new = beta * h1 + (1 - beta) * h2
    return project(h_new)

def gradient_step(h, step_size=0.02):
    """Perform a gradient step using finite differences."""
    n = len(h)
    # Compute current maximum overlap
    current_max = compute_fitness(h)
    # Perturb each value and compute gradient
    grad = np.zeros(n)
    epsilon = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = project(h_plus)
        f_plus = compute_fitness(h_plus)
        grad[i] = (f_plus - current_max) / epsilon
    
    # Update h
    h_new = h - step_size * grad
    return project(h_new)
