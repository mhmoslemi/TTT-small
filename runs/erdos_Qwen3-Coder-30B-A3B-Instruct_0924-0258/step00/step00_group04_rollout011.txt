import numpy as np
import time
from scipy.optimize import minimize
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42  # Use the provided initial construction size
    dx = 2.0 / n_points
    max_iter = 600
    patience = 50
    initial_temp = 1.0
    cooling_factor = 0.99
    min_temp = 0.01
    gradient_scale = 0.1
    mutation_noise = 0.05
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Evaluate initial C5
    def compute_C5(h):
        h = np.array(h)
        # Compute correlation: h * (1 - h_{shifted})
        correlations = np.correlate(h, 1 - h, mode='full')
        # Take the maximum among all lag values
        max_corr = np.max(correlations)
        # Apply scaling factor dx
        return max_corr * dx
    
    current_C5 = compute_C5(h)
    best_C5 = current_C5
    best_h = h.copy()
    
    # Track recent improvements
    last_improvement = 0
    temp = initial_temp
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        iteration += 1
        
        # Make a candidate solution
        # Add Gaussian noise to h
        noise = np.random.normal(0, mutation_noise, n_points)
        h_candidate = h + noise * gradient_scale
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Rescale to meet sum constraint
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Evaluate candidate
        new_C5 = compute_C5(h_candidate)
        
        # Accept or reject the candidate
        if new_C5 < current_C5 or np.random.rand() < np.exp((current_C5 - new_C5) / temp):
            h = h_candidate
            current_C5 = new_C5
            
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
                last_improvement = iteration
        
        # Update temperature
        temp = max(temp * cooling_factor, min_temp)
        
        # Warm restart if no improvement in patience
        if iteration - last_improvement > patience:
            temp = initial_temp
            last_improvement = iteration
    
    # Final validation
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
