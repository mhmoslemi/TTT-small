import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # Take the maximum among all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h):
    """Project h to satisfy constraints: 0 <= h <= 1, sum(h) = n_points/2."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (0.5 * len(h)) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

        h = project_to_feasible(h)
    else:
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    start_time = time.time()
    
    # Gradient-based optimization approach
    max_iter = 100000
    lr = 0.1
    decay_rate = 0.9999
    min_lr = 1e-6
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlaps
        corr_full = correlate(h, 1 - h, mode='full')
        # Get only the valid lags (from -n+1 to n-1)
        n = len(h)
        corr_slice = corr_full[n-1:2*n-1]
        corr_vals = corr_slice / n  # normalize by n
        max_idx = np.argmax(corr_vals)
        best_k = max_idx - (n - 1)  # Convert to actual lag
        
        # Compute gradient w.r.t. h for this worst case
        grad = np.zeros_like(h)
        for j in range(n):
            # This is one of the indices in the correlation
            j_minus_k = (j - best_k) % n
            j_plus_k = (j + best_k) % n
            grad[j] = 1 - h[j_minus_k] - h[j_plus_k]
        
        # Normalize gradient magnitude for better step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            grad = grad / grad_norm
        
        # Line search for step size
        step_size = lr
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new)
        c5_new = evaluate_c5(h_new)
        
        # Backtracking line search if needed
        iter_count = 0
        while iter_count < 10 and c5_new >= best_c5 and step_size > min_lr:
            step_size *= 0.5
            h_new = h - step_size * grad
            h_new = project_to_feasible(h_new)
            c5_new = evaluate_c5(h_new)
            iter_count += 1
            
        # Update h
        h = h_new
        lr = max(lr * decay_rate, min_lr)
        
        # Update best solution
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
    return (best_h.tolist(), best_c5, len(best_h))
