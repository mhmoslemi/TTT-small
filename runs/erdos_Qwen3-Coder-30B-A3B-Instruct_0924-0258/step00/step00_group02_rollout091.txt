import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts k
    corr = correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum value scaled by dx
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

    particles = []
    velocities = []
    pBest = []
    pBest_vals = []
    
    # First particle is the initial solution
    particles.append(initial_h_values.copy())
    pBest.append(initial_h_values.copy())
    pBest_vals.append(evaluate_erdos_solution(initial_h_values))
    
    # Other particles are initial solution perturbed
    for _ in range(n_particles - 1):
        h = initial_h_values + np.random.uniform(-0.1, 0.1, n_points)
        h = project_to_feasible(h, n_points)
        particles.append(h)
        pBest.append(h.copy())
        pBest_vals.append(evaluate_erdos_solution(h))
    
    # Initialize velocities
    for _ in range(n_particles):
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    
    # Global best
    gBest_idx = np.argmin(pBest_vals)
    gBest = pBest[gBest_idx].copy()
    gBest_val = pBest_vals[gBest_idx]
    
    start_time = time.time()
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_init * (w_decay ** iteration)
        
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Compute current value
            current_val = evaluate_erdos_solution(h)
            
            # Update personal best
            if current_val < pBest_vals[i]:
                pBest[i] = h.copy()
                pBest_vals[i] = current_val
            
            # Update global best
            if current_val < gBest_val:
                gBest = h.copy()
                gBest_val = current_val
                
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = w * v + c1 * r1 * (pBest[i] - h) + c2 * r2 * (gBest - h)
            
            # Limit velocity
            v = np.clip(v, -0.5, 0.5)
            velocities[i] = v
            
            # Update position
            h_new = h + v
            h_new = project_to_feasible(h_new, n_points)
            
            particles[i] = h_new
            
        # Re-evaluate all after update
        for i in range(n_particles):
            pBest_vals[i] = evaluate_erdos_solution(pBest[i])
            
        gBest_idx = np.argmin(pBest_vals)
        gBest = pBest[gBest_idx].copy()
        gBest_val = pBest_vals[gBest_idx]
    
    return (gBest, gBest_val, n_points)
