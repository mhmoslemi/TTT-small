import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the Erdos constant for given h_values"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute overlaps for all shifts
    correlations = np.correlate(h_values, 1 - np.array(h_values), mode='full')
    # The maximum overlap is the maximum of all correlations (excluding edge effects)
    max_overlap = np.max(correlations[n_points-1:])
    return max_overlap * dx

def project(h):
    """Project h to satisfy constraints: 0 <= h[i] <= 1, sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    n_points = len(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_fitness(h):
    """Compute the fitness as the maximum overlap"""
    return evaluate_erdos_solution(h)

def mutate(h, sigma):
    """Mutate h with Gaussian noise and project back"""
    h_new = h + np.random.normal(0, sigma, len(h))
    return project(h_new)

def crossover(h1, h2, alpha=0.6):
    """Blend crossover between two individuals"""
    beta = np.random.uniform(alpha, 1 - alpha)
    h_new = beta * h1 + (1 - beta) * h2
    return project(h_new)
