        h[1] = 0.8

    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe algorithm parameters
    max_iter = 1000
    learning_rate = 0.9
    tol = 1e-6
    iter_count = 0
    
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Compute cross-correlations
        corr = signal.correlate(h, 1 - h, mode='full')
        k_idx = np.argmax(corr)
        max_corr = corr[k_idx]
        c5_val = max_corr * (2.0 / n_points)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Compute gradient for max correlation index
        k = k_idx - (n_points - 1)  # Convert index to lag
        grad = np.zeros(n_points)
        if k >= 0:
            grad[k:] += h[:-k]  # h[i] contributes to grad[i+k] if k >= 0
        else:
            grad[:k] += h[-k:]  # k < 0
        grad = -grad  # negative because we want to minimize
            
        # Sort indices by gradient in descending order
        sorted_indices = np.argsort(grad)[::-1]
        
        # Select top indices to form s
        s_indices = sorted_indices[:n_points // 2]
        s = np.zeros(n_points)
        s[s_indices] = 1.0
        
        # Frank-Wolfe step: find step size that minimizes C5
        best_gamma = 0.0
        best_new_c5 = float('inf')
        
        # Test a few gamma values
        gamma_candidates = [0.1, 0.25, 0.5, 0.75, 1.0]
        for gamma in gamma_candidates:
            new_h = (1 - gamma) * h + gamma * s
            new_h = np.clip(new_h, 0, 1)
            h_sum = np.sum(new_h)
            if h_sum > 0:
                new_h = new_h * (n_points / 2) / h_sum
            else:
                new_h = np.ones(n_points) * (n_points / 2) / n_points

            new_c5 = corr_new[np.argmax(corr_new)] * (2.0 / n_points)
            
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_gamma = gamma
        
        # Update h
        if best_gamma > 0:
            h = (1 - best_gamma) * h + best_gamma * s
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points / 2) / h_sum
        else:
            break
            
        iter_count += 1
    
    # Final evaluation
    corr_final = signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = corr_final[np.argmax(corr_final)] * (2.0 / n_points)
    
    return (best_h.tolist(), final_c5, n_points)
