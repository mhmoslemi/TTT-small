import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except:
        # Fallback: uniform construction with sum = n_points/2
        n_points = 70
        h = np.full(n_points, 0.5)
        h[0] = 0.25  # Adjust first point to make sum correct
        h[1] = 0.75
        h[2] = 0.25
        h[3] = 0.75
        h = h * (n_points / 2) / np.sum(h)
    
    # Normalize
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_C5(h_vec):
        # Compute correlations for all lags
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # Slicing to get valid overlaps
        valid_corr = corr[len(h_vec)-1:]
        max_corr = np.max(valid_corr)
        return max_corr * dx
    
    def project_h(h_vec):
        # Clamp between 0 and 1
        h_vec = np.clip(h_vec, 0, 1)
        # Adjust to ensure sum = n_points/2
        current_sum = np.sum(h_vec)
        if abs(current_sum - n_points/2) > 1e-10:
            # Simple adjustment: subtract excess from largest elements
            delta = current_sum - n_points/2
            if delta > 0:
                # Too much, reduce from max values
                indices = np.argsort(h_vec)[-int(delta):]
                # Clip down to 0
                for i in indices:
                    h_vec[i] = max(0, h_vec[i] - delta/len(indices))
            else:
                # Not enough, increase from min values
                indices = np.argsort(h_vec)[:int(-delta)]
                # Clip up to 1
                for i in indices:
                    h_vec[i] = min(1, h_vec[i] - delta/len(indices))
        # Ensure exact sum
        h_vec = h_vec * (n_points/2) / np.sum(h_vec)
        return h_vec
    
    def compute_gradient(h_vec, max_k):
        # Compute gradient for the active lags only
        n = len(h_vec)
        grad = np.zeros(n)
        
        for k in max_k:
            # Add gradient contribution from lag k
            # D/dh[i] \int h(x)(1-h(x+k)) dx
            # = (1 - h[i+k] - (1 - h[i])) * dx
            # = (h[i] - h[i+k]) * dx
            for i in range(n):
                j = (i + k) % n
                grad[i] += (h_vec[i] - h_vec[j]) * dx
                
        return grad
    
    # Initialize
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    alpha = 0.1  # Initial step size
    min_alpha = 1e-6
    convergence_thresh = 1e-6
    last_improvement = 0
    improvement_count = 0
    max_iter = 500
    last_C5 = best_C5
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break

        # Evaluate all correlations
        corr = np.correlate(h, 1 - h, mode="full")
        valid_corr = corr[len(h)-1:]
        max_val = np.max(valid_corr)
        max_k = np.where(valid_corr == max_val)[0]
        
        # Compute gradient
        grad = compute_gradient(h, max_k)
        
        # Gradient descent step
        h_new = h - alpha * grad
        h_new = project_h(h_new)
        
        # Evaluate new C5
        new_C5 = evaluate_C5(h_new)
        
        # Accept new solution if better and feasible
        if new_C5 < best_C5:
            h = h_new
            best_C5 = new_C5
            last_improvement = it
            improvement_count += 1
        else:
            # Maybe perturb a bit if stuck
            h_perturbed = h + np.random.normal(0, 0.001, n_points)
            h_perturbed = project_h(h_perturbed)
            perturbed_C5 = evaluate_C5(h_perturbed)
            if perturbed_C5 < best_C5:
                h = h_perturbed
                best_C5 = perturbed_C5
                last_improvement = it
                improvement_count += 1
                
        # Adaptive step size
        if abs(new_C5 - last_C5) / (abs(last_C5) + 1e-10) < convergence_thresh:
            alpha *= 0.9
        else:
            alpha *= 1.02
            
        if alpha < min_alpha:
            alpha = min_alpha
            
        last_C5 = new_C5
    
    # Final validation
    best_h = project_h(best_h)
    final_C5 = evaluate_C5(best_h)
    
    return (best_h, final_C5, n_points)
