import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for given step function h"""
    dx = 2.0 / len(h)
    # Compute correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Scale by dx
    corr *= dx
    # Return the maximum value
    return np.max(corr)

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 ≤ h[i] ≤ 1 and sum(h) = n_points/2"""
    # Clip values
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Create default initialization
        n_points = 45
        h_init = np.ones(n_points) * 0.5
        h_init[0] = 0.3
        h_init[1] = 0.7
        h_init[-1] = 0.3
        h_init[-2] = 0.7
        
    # Initialize swarm
    n_particles = 30
    particles = []
    velocities = []
    
    # First particle is the initial solution
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Generate other particles with small random perturbations
    for _ in range(n_particles - 1):
        perturbation = np.random.normal(0, 0.1, n_points)
        h = h_init + perturbation
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append((np.random.rand(n_points) - 0.5) * 0.5)
    
    # Evaluate all particles
    particle_scores = [evaluate_erdos_solution(p) for p in particles]
    pBest = particle_scores.copy()
    pBest_particles = [p.copy() for p in particles]
    
    # Initialize global best
    gBest_idx = np.argmin(pBest)
    gBest = pBest[gBest_idx]
    gBest_particle = pBest_particles[gBest_idx].copy()
    
    # PSO parameters
    omega = 0.9  # inertia weight
    c1 = 2.0     # cognitive acceleration
    c2 = 2.0     # social acceleration
    max_vel = 0.5
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (omega * velocities[i] + 
                           c1 * r1 * (pBest_particles[i] - particles[i]) + 
                           c2 * r2 * (gBest_particle - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply feasibility projection
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            current_score = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if current_score < pBest[i]:
                pBest[i] = current_score
                pBest_particles[i] = particles[i].copy()
                
                # Update global best
                if current_score < gBest:
                    gBest = current_score
                    gBest_particle = particles[i].copy()
            
        # Update inertia weight
        omega = max(0.4, omega * 0.99)
        
    return (gBest_particle, gBest, n_points)
