import numpy as np
from scipy import signal
import time
import random

def evaluate_h(h, dx):
    # Compute the overlap integral for all lags
    corr = signal.correlate(h, 1-h, mode='full')
    # The maximum overlap for C_5
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Ensure h is within [0,1] and sum = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust by scaling
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    return h

def initialize_population(n_points, pop_size, initial_h=None):
    population = []
    if initial_h is not None:
        population.append(project_to_feasible(initial_h.copy(), n_points))
    else:
        # Random initial population
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    return population

def tournament_selection(population, fitnesses, tournament_size=2):
    indices = np.random.choice(len(population), size=tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def crossover(parent1, parent2):
    n_points = len(parent1)
    # Simple arithmetic crossover
    alpha = np.random.rand()
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    # Re-project to feasible
    offspring1 = project_to_feasible(offspring1, n_points)
    offspring2 = project_to_feasible(offspring2, n_points)
    return offspring1, offspring2

def mutate(h, mutation_rate=0.05, sigma=0.1):
    n_points = len(h)
    mask = np.random.rand(n_points) < mutation_rate
    if not np.any(mask):
        # Ensure at least one mutation
        idx = np.random.randint(0, n_points)
        mask[idx] = True
    h = h.copy()
    h[mask] += np.random.normal(0, sigma, np.sum(mask))
    return project_to_feasible(h, n_points)

def genetic_algorithm(n_points, pop_size=20, max_time=600, initial_h=None):
    start_time = time.time()
    population = initialize_population(n_points, pop_size, initial_h)
    dx = 2.0 / n_points

    best_h = None
    best_fitness = float('inf')
    no_improvement_count = 0
    generation = 0
    while time.time() - start_time < max_time - 1 and no_improvement_count < 100:
        # Evaluate fitness
        fitnesses = []
        for h in population:
            fitness = evaluate_h(h, dx)
            fitnesses.append(fitness)
            if fitness < best_fitness:
                best_fitness = fitness
                best_h = h.copy()
        no_improvement_count += 1
        # Selection and reproduction
        new_population = []
        # Keep best
        sorted_indices = np.argsort(fitnesses)
        new_population.extend([population[i] for i in sorted_indices[:2]])
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            offspring1, offspring2 = crossover(parent1, parent2)
            offspring1 = mutate(offspring1)
            offspring2 = mutate(offspring2)
            new_population.extend([offspring1, offspring2])
        population = new_population[:pop_size]
        generation += 1
    return best_h, best_fitness

    return (h.tolist(), c5_bound, n_points)
