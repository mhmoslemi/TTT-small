import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for shifts k=0,1,...,2n-2
    corr = signal.correlate(h, 1 - h, mode='full')
    # Take maximum over all lags and multiply by dx
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.full(n_points, 1.0 / n_points)
    return h
