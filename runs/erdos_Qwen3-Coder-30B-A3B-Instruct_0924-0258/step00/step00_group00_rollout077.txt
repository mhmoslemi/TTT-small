import numpy as np
from scipy import signal
import time
import random

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    corr = signal.correlate(h, 1 - h, mode='full')
    # Only take the relevant part of the correlation (excluding padding)
    n = len(h)
    corr_vals = corr[n-1:2*n-1]
    return np.max(corr_vals) * dx

def rescale_h(h, target_sum):
    """Rescale h to have the target sum while keeping values in [0,1]"""
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.ones_like(h) * (target_sum / len(h))
    h_scaled = h * (target_sum / current_sum)
    # Clamp to [0,1]
    h_clamped = np.clip(h_scaled, 0, 1)
    # Rescale again to ensure target sum
    actual_sum = np.sum(h_clamped)
    if actual_sum > 0:
        h_rescaled = h_clamped * (target_sum / actual_sum)
    else:
        h_rescaled = np.ones_like(h) * (target_sum / len(h))
    return h_rescaled

def initialize_swarm(n_points, n_particles, initial_h=None):
    """Initialize swarm of particles"""
    if initial_h is not None:
        # Use provided initial construction
        h_base = np.array(initial_h)
        h_base = rescale_h(h_base, n_points / 2)
    else:
        # Generate a random construction
        h_base = np.random.rand(n_points)
        h_base = rescale_h(h_base, n_points / 2)
    
    swarm = []
    for _ in range(n_particles):
        if initial_h is not None:
            # Perturb the base slightly
            h = h_base + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
        else:
            h = np.random.rand(n_points)
            h = rescale_h(h, n_points / 2)
        swarm.append(h)
    return swarm

def pso_optimize(n_points, initial_h=None, budget_s=600):
    """Run PSO to optimize h for minimal C5"""
    n_particles = 30
    omega = 0.7
    phi_p = 2.0
    phi_g = 2.0
    max_iter = 1000
    dx = 2.0 / n_points
    
    # Initialize swarm
    swarm = initialize_swarm(n_points, n_particles, initial_h)
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Track personal bests
    personal_bests = [None for _ in range(n_particles)]
    personal_best_scores = [float('inf') for _ in range(n_particles)]
    
    # Track global best
    global_best = None
    global_best_score = float('inf')
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            h = swarm[i]
            # Ensure h is within bounds and sum is correct
            h = np.clip(h, 0, 1)
            h = rescale_h(h, n_points / 2)
            
            # Evaluate current fitness
            score = evaluate_c5(h, dx)
            
            # Update personal best
            if score < personal_best_scores[i] or personal_best_scores[i] == float('inf'):
                personal_bests[i] = h.copy()
                personal_best_scores[i] = score
                if score < global_best_score:
                    global_best = h.copy()
                    global_best_score = score
            
            # Update swarm position if needed
            swarm[i] = h
            
        # Update velocities and positions
        for i in range(n_particles):
            if personal_bests[i] is not None:
                r1, r2 = random.random(), random.random()
                # Update velocity with PSO formula
                velocities[i] = (omega * velocities[i] + 
                                phi_p * r1 * (personal_bests[i] - swarm[i]) + 
                                phi_g * r2 * (global_best - swarm[i]))
                
                # Update position
                swarm[i] += velocities[i]
                
                # Clamp and rescale
                swarm[i] = np.clip(swarm[i], 0, 1)
                swarm[i] = rescale_h(swarm[i], n_points / 2)
    
    return global_best, global_best_score

    random.seed(seed)

    # Ensure final result is valid
    if h_opt is not None:
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = rescale_h(h_opt, n_points / 2)
        return h_opt.tolist(), c5_best, n_points
    else:
        # Fallback to initial construction if nothing good found
        if 'initial_h_values' in globals() and initial_h_values is not None:
            return initial_h_values, evaluate_c5(np.array(initial_h_values), 2.0/n_points), n_points
        else:
            # Generate a fallback random construction
            h_fallback = np.random.rand(n_points)
            h_fallback = rescale_h(h_fallback, n_points / 2)
            return h_fallback.tolist(), evaluate_c5(h_fallback, 2.0/n_points), n_points
