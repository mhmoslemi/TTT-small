import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_c5(h):
    """Evaluate the maximum overlap for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    correlations = np.correlate(h, 1 - h, mode='full')
    # Take only the valid overlaps (excluding boundary effects)
    valid_correlations = correlations[n-1:2*n-1]
    return np.max(valid_correlations) * dx

def project_to_feasible(h):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    # Rescale to satisfy integral constraint
    current_sum = np.sum(h)
    target_sum = len(h) / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def gradient_descent_step(h, step_size=1e-3):
    """Perform one step of gradient descent on the maximum overlap."""
    n = len(h)
    dx = 2.0 / n
    # Compute current overlaps
    correlations = np.correlate(h, 1 - h, mode='full')
    valid_correlations = correlations[n-1:2*n-1]
    max_idx = np.argmax(valid_correlations)
    max_val = valid_correlations[max_idx]
    
    # Approximate gradient using finite differences
    eps = 1e-5
    grad = np.zeros_like(h)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus)
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        corr_plus_valid = corr_plus[n-1:2*n-1]
        grad[i] = (corr_plus_valid[max_idx] - max_val) / eps
    
    # Update h
    h_new = h - step_size * grad
    h_new = project_to_feasible(h_new)
    return h_new

def smooth_and_normalize(h, sigma_factor=0.1):
    """Apply Gaussian smoothing and normalize."""
    n = len(h)
    sigma = sigma_factor * n
    h_smooth = gaussian_filter1d(h, sigma=sigma)
    h_smooth = project_to_feasible(h_smooth)
    return h_smooth

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42  # Good balance between resolution and speed
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure the integral constraint holds
    h = project_to_feasible(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 300:
        # LOCAL OPTIMIZATION STEP
        h = gradient_descent_step(h, step_size=1e-3)
        
        # SMOOTHING STEP
        h = smooth_and_normalize(h, sigma_factor=0.1)
        
        # Update best solution
        c5_current = evaluate_c5(h)
        if c5_current < best_c5:
            best_h = h.copy()
            best_c5 = c5_current
            
        iteration += 1
        
        # Occasionally check if we're making progress
        if iteration % 20 == 0:
            if time.time() - start_time > budget_s - 1:
                break
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
