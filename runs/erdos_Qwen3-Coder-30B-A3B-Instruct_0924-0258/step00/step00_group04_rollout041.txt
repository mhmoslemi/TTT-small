import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Fix number of points
    n_points = 42
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert len(h) == n_points
        assert np.all(h >= 0) and np.all(h <= 1)
        assert abs(np.sum(h) - n_points/2) < 1e-10
    else:
        h = np.full(n_points, 0.5)
        h *= n_points / (2 * np.sum(h))
    
    # Ensure sum constraint
    h = h * n_points / (2 * np.sum(h))
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h, n_points)
    
    start_time = time.time()
    
    alpha = 0.1
    iter_count = 0
    last_improvement = 0
    patience = 50
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute all overlaps
        # This is a bit tricky since we want max_k of the correlate
        # Let's compute for all lags
        h_extended = np.concatenate([h, h])  # Double length for convolution-like behavior
        C = []
        for k in range(n_points):
            # h[i] * (1 - h[(i+k)%n_points])
            corr_val = np.sum(h * np.roll(1 - h, k))
            C.append(corr_val)
        
        # Adjust for the factor
        adjust = 2.0 / n_points
        C = np.array(C) * adjust
        
        # Find the worst case lag
        k_star = np.argmax(C)
        C5_current = C[k_star]
        
        if C5_current < best_C5:
            best_C5 = C5_current
            best_h = h.copy()
        
        # Compute gradient w.r.t. h for the worst lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            # derivative of h[i] * (1 - h[(i-k_star) % n_points])
            # with respect to h[i] is (1 - h[(i-k_star) % n_points]) 
            # Plus terms for the shifted positions
            # we're looking at h[i] * (1 - h[(i + k_star) % n_points]) 
            # plus terms from the same cross correlation formula
            term1 = 1 - h[(i - k_star) % n_points]
            term2 = 1 - h[(i + k_star) % n_points]
            grad[i] = adjust * (term1 + term2 - 1)
        
        # Steepest descent step
        h_new = h - alpha * grad
        
        # Clip to feasible bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy the integral constraint
        h_new *= n_points / (2 * np.sum(h_new))
        
        # Simple backtracking line search for alpha
        trial_h = h_new.copy()
        trial_C5 = evaluate_C5(trial_h, n_points)
        
        if trial_C5 < C5_current:
            h = trial_h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            if last_improvement > patience:
                alpha *= 0.5
                last_improvement = 0

        if abs(C5_current - best_C5) < 1e-5:
            break
            
        # Occasionally add noise to escape local minima
        if iter_count % 50 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h *= n_points / (2 * np.sum(h))
    
    # Final validate and return best solution
    final_C5 = evaluate_C5(best_h, n_points)
    return (best_h.tolist(), final_C5, n_points)

def evaluate_C5(h, n_points):
    h = np.array(h)
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # This gives us the full correlation, which includes negative lags
    # The valid range we care about for the overlap integral
    # is from -n_points+1 to n_points-1
    mid = len(corr) // 2
    valid_corr = corr[mid - n_points + 1 : mid + n_points]
    # Scale by dx which is 2/n_points
    dx = 2.0 / n_points
    valid_corr = valid_corr * dx
    return np.max(valid_corr)
