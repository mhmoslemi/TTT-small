import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    # Compute all correlations: h(x) * (1 - h(x + k))
    correlations = correlate(h_values, 1 - h_values, mode='full')
    # Take maximum over all lags
    max_correlation = np.max(correlations)
    return max_correlation * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points / 2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-10:
        # Simple projection: adjust towards average
        avg_val = current_sum / n_points
        adjustment = (target_sum - current_sum) / n_points
        h += adjustment
        h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, n_points):
    """Compute gradient for current configuration, so that we can descend from max overlap"""
    dx = 2.0 / n_points
    # Create mirrored version for correlation
    h_extended = np.concatenate([h, h])  # Double length for convenience
    # Compute all correlations (all lags)
    correlations = correlate(h_extended, 1 - h, mode='full')
    
    # Find lags that achieve the maximum overlap
    max_correlation = np.max(correlations)
    max_indices = np.where(np.abs(correlations - max_correlation) < 1e-10)[0]
    
    # Gradient calculation
    grad = np.zeros_like(h)
    for k in max_indices:
        k_mod = k % n_points
        # We want d/dh[i] (h[i] * (1 - h[i + k_mod]))
        # For i in [0, n_points), and i + k_mod in [0, n_points) with modulo
        for i in range(n_points):
            j = (i + k_mod) % n_points
            grad[i] += (1 - h[j] - (1 - h[i])) * dx
    
    return grad
