import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h."""
    dx = 2.0 / len(h)
    correlation = signal.correlate(h, 1 - h, mode="full")
    c5 = np.max(correlation) * dx
    return c5

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points / 2 and 0 <= h[i] <= 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.full(n_points, 1.0 / n_points)
    return h

def mutate_h(h, n_points, mutation_rate=0.1):
    """Mutate h by perturbing a fraction of values."""
    h_mut = h.copy()
    num_mutate = int(mutation_rate * n_points)
    indices = np.random.choice(n_points, num_mutate, replace=False)
    for i in indices:
        h_mut[i] += 0.1 * np.random.randn()
    return project_to_feasible(h_mut, n_points)

def crossover_h(h1, h2, n_points):
    """Single-point crossover between two individuals."""
    crossover_point = np.random.randint(1, n_points)
    h_child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    h_child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    return project_to_feasible(h_child1, n_points), project_to_feasible(h_child2, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points
    
    # Initial solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.ones(n_points) * 0.5

    # Population initialization
    population_size = 50
    population = [h0.copy()]
    while len(population) < population_size:
        h_perturbed = h0 + np.random.uniform(-0.1, 0.1, n_points)
        h_feasible = project_to_feasible(h_perturbed, n_points)
        population.append(h_feasible)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        c5 = evaluate_erdos_solution(h)
        fitness.append(c5)
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # GA parameters
    num_generations = 12
    num_parents = population_size // 2
    mutation_rate = 0.1
    
    for gen in range(num_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track the best in this generation
        current_best = population[0]
        current_best_c5 = fitness[0]
        if current_best_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_best_c5
            
        # Select parents
        parents = population[:num_parents]
        
        # Generate offspring
        offspring = []
        for i in range(0, len(parents), 2):
            p1 = parents[i]
            p2 = parents[(i + 1) % len(parents)]
            child1, child2 = crossover_h(p1, p2, n_points)
            offspring.extend([child1, child2])
            
        # Mutate offspring
        mutated_offspring = []
        for child in offspring:
            mutated_child = mutate_h(child, n_points, mutation_rate)
            mutated_offspring.append(mutated_child)
            
        # Build new population
        new_population = parents + mutated_offspring[:num_parents]
        population = new_population
        
        # Evaluate new population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness.append(c5)
    
    # Final check and return
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
