import numpy as np
import time
from scipy.optimize import minimize
from math import sqrt

def evaluate_c5(h):
    """Compute C5 value for given h"""
    n = len(h)
    dx = 2.0 / n
    # compute correlation for all lags
    corr = np.correlate(h, 1 - h, mode="full")
    # take maximum
    max_corr = np.max(corr)
    return max_corr * dx

def project_and_rescale(h, n_points):
    """Project h to [0,1] and rescale to sum to n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def gradient_descent_step(h, n_points, eps=1e-6):
    """Estimate gradient via finite difference"""
    h_copy = h.copy()
    grad = np.zeros_like(h)
    for i in range(n_points):
        h_plus = h_copy.copy()
        h_plus[i] += eps
        h_plus = project_and_rescale(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus)
        
        h_minus = h_copy.copy()
        h_minus[i] -= eps
        h_minus = project_and_rescale(h_minus, n_points)
        c5_minus = evaluate_c5(h_minus)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 42  # start from the parent construction
    dx = 2.0 / n_points

    # Ensure initial sum is correct
    h = project_and_rescale(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Simulated Annealing parameters
    temperature = 1.0
    min_temp = 0.01
    cooling_factor = 0.99
    patience = 50
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Early termination check
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        grad = gradient_descent_step(h, n_points)
        
        # Generate candidate solution with gradient guidance
        noise = np.random.normal(0, 0.05, n_points)
        step = 0.1 * (grad + noise)
        h_candidate = h + step
        
        # Project and rescale
        h_candidate = project_and_rescale(h_candidate, n_points)
        
        # Evaluate
        c5_candidate = evaluate_c5(h_candidate)
        
        # Accept or reject
        if c5_candidate < best_c5:
            h = h_candidate
            best_c5 = c5_candidate
            no_improvement_count = 0
            if best_c5 < 0.3807:
                break
        else:
            # Accept with probability based on temperature
            delta = c5_candidate - best_c5
            if np.random.rand() < np.exp(-delta / temperature):
                h = h_candidate
        
        # Update temperature
        temperature = max(temperature * cooling_factor, min_temp)
        
        # Check if we should reset temperature
        no_improvement_count += 1
        if no_improvement_count > patience:
            temperature = 1.0  # restart temperature
            no_improvement_count = 0
    
    # Final check on best solution
    best_h = project_and_rescale(best_h, n_points)
    best_c5 = evaluate_c5(best_h)
