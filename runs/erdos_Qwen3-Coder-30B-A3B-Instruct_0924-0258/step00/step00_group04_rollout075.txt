import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_erdos_solution(h):
    """Evaluate the overlap integral C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute overlap for all lags using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # Take the maximum over all lags (excluding the zero-lag which is not relevant)
    max_overlap = np.max(corr[n-1:])  # Only consider positive lags
    return max_overlap * dx

def compute_gradient(h, eps=1e-6):
    """Compute gradient of C5 using finite differences"""
    n = len(h)
    dx = 2.0 / n
    base_c5 = evaluate_erdos_solution(h)
    
    grad = np.zeros_like(h)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        # Ensure boundary constraints
        h_plus = np.clip(h_plus, 0, 1)
        # Rescale to maintain integral constraint
        h_plus *= (n / 2) / np.sum(h_plus)
        c5_plus = evaluate_erdos_solution(h_plus)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Start with uniform distribution
        n_points = 42
        h = np.ones(n_points) * 0.5
    
    # Ensure the integral constraint is met
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    # Iteratively optimize in phases: local optimization + low-pass filtering
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Local optimization step
        def objective(x):
            # Clip to [0,1] and rescale to meet integral constraint
            x = np.clip(x, 0, 1)
            x = x * (n_points / 2) / np.sum(x)
            return evaluate_erdos_solution(x)
        
        # Optimization with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                              options={'maxiter': 20})
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            c5_opt = evaluate_erdos_solution(h_opt)
        except:
            h_opt = h
            c5_opt = best_c5
            pass
        
        # Apply low-pass filter
        sigma = 0.1 * n_points
        h_filtered = gaussian_filter1d(h_opt, sigma=sigma, mode='wrap')
        h_filtered = np.clip(h_filtered, 0, 1)
        h_filtered = h_filtered * (n_points / 2) / np.sum(h_filtered)
        
        # Update if better
        c5_filtered = evaluate_erdos_solution(h_filtered)
        if c5_filtered < best_c5:
            best_c5 = c5_filtered
            best_h = h_filtered.copy()
        
        h = h_filtered
        
        # Early exit condition
        if best_c5 < 0.3807:
            break
            
    return (list(best_h), best_c5, len(best_h))
