import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default initialization
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization parameters
    learning_rate = 0.01
    decay = 0.999
    perturb_mag = 0.05
    max_iter = budget_s  # Roughly 1 sec/iter
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        conv = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        
        # Update best if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the maximum overlap lag
        max_idx = np.argmax(conv)
        # Conv is full length 2*n-1, so lag index from center
        lag = max_idx - (n_points - 1)
        
        # Compute gradient w.r.t. lag
        if lag >= 0:
            # h[i] * (1 - h[i+lag]) terms contribute to this lag
            grad = np.zeros_like(h)
            for j in range(n_points - lag):
                grad[j] += 1 - h[j + lag]
            for j in range(lag, n_points):
                grad[j] -= h[j - lag]
        else:
            # Reverse case
            lag = -lag
            grad = np.zeros_like(h)
            for j in range(n_points - lag):
                grad[j] += h[j + lag]
            for j in range(lag, n_points):
                grad[j] -= (1 - h[j - lag])
        
        # Apply gradient step
        h = h - learning_rate * grad
        
        # Project onto feasible region
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        # Perturb occasionally to escape local minima
        if i % 10 == 0:
            h += perturb_mag * np.random.uniform(-1, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            
        # Decay learning rate
        learning_rate *= decay
        
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
