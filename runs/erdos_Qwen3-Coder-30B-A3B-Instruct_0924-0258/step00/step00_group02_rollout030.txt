import numpy as np
from scipy import optimize
import time

def evaluate_h(h, dx):
    # computes C5 = max_k integral of h(x)(1-h(x+k)) dx
    h_rev = np.flip(h)
    corr = np.correlate(h, h_rev, mode='full')
    # corr[k] = sum h[i] h[i+k] for k = -(n-1) to n-1
    # we want sum h[i] (1-h[i+k]) = sum h[i] - sum h[i] h[i+k]
    # = sum h[i] - corr[n-1+k] = 1 - corr[n-1+k]
    # indices in corr go from -n+1 to n-1
    # so corr[n-1+k] corresponds to k in range [-n+1, n-1]
    # since we're iterating k from 0 to n-1, corr[n-1+k] is indices n-1+k for k = 0..n-1
    # we want max over k from 0 to n-1 of (1 - corr[n-1+k])
    # which is same as 1 - min over k from 0 to n-1 of corr[n-1+k]
    overlaps = 1.0 - corr[n-1:n-1+n]
    return np.max(overlaps)

def project_h(h, dx, n_points):
    # Project h to feasible region
    h = np.clip(h, 0, 1)
    # Fix sum constraint by rescaling
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 45
    dx = 2.0 / n_points
    
    h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
    h = project_h(h, dx, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Optimizer: gradient descent with momentum
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # evaluate
        c5 = evaluate_h(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # compute dominant overlaps
        h_rev = np.flip(h)
        corr = np.correlate(h, h_rev, mode='full')
        # overlap_k = 1 - corr[n-1+k] for k = 0 to n-1
        overlaps = 1.0 - corr[n-1:n-1+n]
        # find top contributing indices
        top_k = np.argsort(overlaps)[-5:][::-1]
        # compute gradient of the sum of top overlaps
        grad = np.zeros_like(h)
        for k in top_k:
            # gradient of integral h(x) * (1 - h(x + k))
            # approximately: since we're using discrete samples
            # let d/dx [h(x) * (1 - h(x + k))] = 
            # h'(x) * (1 - h(x + k)) + h(x) * (-h'(x+k))
            # for discrete, grad[i] = (1 - h[wrap(i+k)]) + (h[wrap(i)], 
            # see derivative using finite differences
            # to gain accuracy, compute derivative properly by hand of correlation
            # for a particular k, the contribution to overlap_k at i is h[i](1-h[i+k])
            # so d/dh[i] grad = (1-h[i+k]) and d/dh[i+k] = -h[i]
            # we need to handle array wrap in a proper way
            for i in range(n_points):
                grad[i] += (1 - h[(i + k) % n_points])
                grad[(i + k) % n_points] -= h[i]
        grad *= dx

        # update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h += velocity
        
        # project h into feasible set
        h = project_h(h, dx, n_points)
        
        # adaptive learning rate
        if c5 - best_c5 < 1e-8:
            learning_rate *= 0.95
        if learning_rate < 1e-6:
            learning_rate = 1e-6
            
    return (best_h, best_c5, n_points)
