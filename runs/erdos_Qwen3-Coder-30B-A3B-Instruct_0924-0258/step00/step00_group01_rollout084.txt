def evaluate_c5(h):

    corr = np.correlate(h, 1-h, mode='full')
    # The maximum overlap is the maximum of these correlations (scaled by dx)
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_onto_feasible(h, n_points):
    """Project h onto the feasible region: [0,1] and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h
