import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h):
    """Evaluate C5 for given h."""
    n_points = len(h)
    dx = 2.0 / n_points
    h_corr = signal.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(h_corr) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h onto feasible set: 0 <= h <= 1, sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2.0) / h_sum
    return h

def compute_gradient(h, k_max, n_points):
    """Compute gradient of overlap at k_max shift."""
    dx = 2.0 / n_points
    grad = np.zeros_like(h)
    # The gradient is the derivative of the correlation with respect to h
    # In the context of correlation, this relates to overlapping terms
    # A simple approximation: we center the gradient over the region
    # where the correlation term is maximized
    lk = n_points - 1 + k_max  # index in full correlation
    for i in range(n_points):
        if i + k_max < n_points:
            grad[i] -= 1  # because (1 - h[i+k_max]) contributes to the integral
        if i - k_max >= 0:
            grad[i] += 1  # because h[i-k_max] contributes (via 1-h)
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Use the provided initial construction size
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
    else:
        # Create a basic initial function
        h = np.ones(n_points) * 0.5
        
    h = project_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    lr = 0.01
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Compute overlaps for all shifts
        h_corr = signal.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(h_corr)
        c5 = np.max(h_corr) * dx
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Compute gradient (simplified)
        grad = compute_gradient(h, max_k - (n_points - 1), n_points)
        
        # Update using gradient ascent (minimize C5)
        h = h - lr * grad
        h = project_h(h, n_points)
        
        # Adaptive learning rate decay
        if iter_count % 20 == 0:
            lr *= 0.97
            
        # Occasionally perturb the solution
        if iter_count % 50 == 0:
            perturb = np.random.uniform(-0.05, 0.05, size=n_points)
            h += perturb
            h = project_h(h, n_points)
            
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
