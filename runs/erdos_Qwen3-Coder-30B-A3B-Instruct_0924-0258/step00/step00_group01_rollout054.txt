def compute_c5(h):
    """Compute C5 bound for a given h."""
    h = np.array(h)
    n_points = len(h)
    dx = 2.0 / n_points
    overlaps = np.correlate(h, 1 - h, mode='full')
    return np.max(overlaps) * dx
