import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        assert np.all(h >= 0) and np.all(h <= 1), "Initial h not in [0,1]"
        assert abs(np.sum(h) - target_sum) < 1e-6, "Initial sum not correct"
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Projective gradient descent parameters
    max_iter = 10000
    alpha = 0.1
    backtracking_factor = 0.8
    min_alpha = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps and find worst lag
        overlaps = np.correlate(h, 1-h, mode='full')[:n_points] * dx
        k_star = np.argmax(overlaps)
        
        # Compute gradient (derivatives of the maximal overlap term)
        g = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_star) % n_points
            g[i] = 1 - 2 * h[j]
        
        # Normalize gradient
        norm_g = np.linalg.norm(g)
        if norm_g > 1e-12:
            g = g / norm_g
        
        # Line search
        alpha_search = alpha
        while alpha_search >= min_alpha:
            h_new_unproj = h - alpha_search * g
            h_new = np.clip(h_new_unproj, 0, 1)
            # Rescale to satisfy constraint
            h_new = h_new * target_sum / np.sum(h_new)
            
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < best_c5:
                h = h_new
                best_c5 = new_c5
                best_h = h.copy()
                break
            else:
                alpha_search *= backtracking_factor
        
        # If alpha becomes too small, reduce learning rate
        if alpha_search < min_alpha:
            alpha *= 0.9
    
    return (best_h.tolist(), float(best_c5), n_points)

def evaluate_c5(h, dx):
    # Compute overlap integrals for all lags
    # mask of 1-h for circular correlation
    conv = correlate(h, 1-h, mode='full')
    # Extract valid lags (0 to N-1)
    overlaps = conv[len(h)-1:] * dx
    return np.max(overlaps)
