        conv = signal.correlate(h_array, 1 - h_array, mode='full')
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    start_time = time.time()
    alpha = 0.1
    no_improvement_count = 0
    max_no_improvement = 50
    
    # Precompute lag indices for efficiency
    lag_indices = np.arange(1, n_points)

        C5 = evaluate_C5(h)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early stopping
        if no_improvement_count > max_no_improvement:
            # Random perturbation
            if np.random.rand() < 0.05:
                perturb = np.random.normal(0, 0.02, n_points)
                h += perturb
                h = np.clip(h, 0, 1)

        conv_full = signal.correlate(h, 1 - h, mode='full')

        lags = np.arange(-n_points + 1, n_points)
        max_lag_idx = np.argmax(conv_full)
        k0 = lags[max_lag_idx]

        # Create shifting matrices for efficient computation
        grad = np.zeros(n_points)
        if k0 > 0:
            # h[i] * (1 - h[i+k0])  
            grad[:-k0] += 1 - h[k0:] 
            grad[k0:] -= h[:-k0]
        elif k0 < 0:
            # h[i] * (1 - h[i - k0])
            k0_positive = -k0
            grad[k0_positive:] += 1 - h[:-k0_positive] 
            grad[:-k0_positive] -= h[k0_positive:]
        else:
            # Circular case: lag 0
            grad[:] += 1 - 2*h
        
        # Clip gradient
        grad = np.clip(grad, -1.0, 1.0)

        h += alpha * grad
        
        # Project to [0,1]

        if abs(current_sum - target_sum) > 1e-6:
            h *= (target_sum / current_sum)
        
        # Adaptive step size
        if no_improvement_count % 50 == 0 and no_improvement_count > 0:
            alpha *= 0.95
            if alpha < 1e-6:
                alpha = 1e-6
    
    # Final evaluation
    final_C5 = evaluate_C5(best_h)
    
    return (best_h.tolist(), float(final_C5), int(n_points))
