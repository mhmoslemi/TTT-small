import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 70
        h = np.full(n_points, 0.5)
        # Adjust to ensure sum constraint
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
    
    dx = 2.0 / n_points
    max_iter = 500
    alpha = 0.1
    tol = 1e-4
    start_time = time.time()
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        # Clip to [0, 1]
        h_vec = np.clip(h_vec, 0, 1)
        # Adjust to make sum = n_points/2
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * (n_points / 2) / h_sum
        return h_vec
    
    # Compute C5
    def compute_C5(h_vec):
        # Correlate h with (1-h) - full mode gives all lags
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # The max is over all valid lags
        # This corresponds to the maximum correlation value scaled by dx
        return np.max(corr) * dx
    
    # Compute gradient w.r.t. h
    def compute_gradient(h_vec, max_lag_idx):
        # For a given maximum lag index, we compute the gradient
        # This dependency is from correlation with 1-h
        grad = np.zeros_like(h_vec)
        lag = max_lag_idx - (n_points - 1)  # Convert to actual lag in range
        # Gradient 1/N terms
        if lag >= 0:
            # Take lags where h[i] contributes: up to length of h
            # Ignore negative indices
            for i in range(n_points):
                j = i + lag
                if j < n_points:
                    grad[i] += 1.0 * dx  # from h[i] appearing in correlation
                    grad[j] += -1.0 * dx  # from 1-h[j] appearing in correlation
        else:
            # Add -1 contribution from lagged terms
            for i in range(n_points):
                j = i + lag
                if j >= 0:
                    grad[i] += 1.0 * dx
                    grad[j] += -1.0 * dx
        return grad
    
    C5_prev = compute_C5(h)
    last_improvement = 0.0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute all correlations
        corr = signal.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx] * dx
        
        # Compute gradient for the maximum 
        grad = compute_gradient(h, max_idx)
        
        h_new = h - alpha * grad
        h_new = project_h(h_new)
        
        C5_new = compute_C5(h_new)
        
        # Check improvement step
        improvement = C5_prev - C5_new
        if improvement < tol:
            last_improvement += 1
        else:
            last_improvement = 0
            
        if last_improvement > 5:
            break
            
        h = h_new
        C5_prev = C5_new
            
    C5_result = compute_C5(h)
    
    return (h.tolist(), float(C5_result), n_points)
