import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    alpha = 0.1
    convergence_threshold = 1e-4
    perturbation_std = 0.05
    max_perturbations = 10
    
    # Get initial values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Initialize best solution
    best_h = h.copy()
    current_C5 = evaluate_erdos_solution(h, n_points)
    best_C5 = current_C5
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all lags
        corr = correlate(h, 1 - h, mode="full")
        lags = np.arange(-n_points + 1, n_points)
        max_overlap = np.max(corr)
        
        # Find all lags that achieve maximum overlap
        max_indices = np.where(np.abs(corr - max_overlap) < 1e-10)[0]
        max_lags = lags[max_indices]
        
        # Compute gradient based on max lags
        gradient = np.zeros(n_points)
        for k in max_lags:
            if k > 0:
                # h[i] contributes to lag k only when we look at (h[i] * (1 - h[i+k]))
                # So gradient[i] -= (1 - h[i+k]) + (1 - h[i])
                for i in range(n_points):
                    j = (i + k) % n_points
                    gradient[i] -= (1 - h[j])
                    gradient[i] -= (1 - h[i])
            elif k < 0:
                # h[i] contributes to lag k only when we look at (h[i] * (1 - h[i+k]))
                for i in range(n_points):
                    j = (i + k) % n_points
                    gradient[i] -= (1 - h[j])
                    gradient[i] -= (1 - h[i])
            else:
                # k == 0: case where we have h[i] * (1 - h[i])
                gradient += -2 * h
    
        gradient *= dx
    
        # Update h using gradient descent with projected step
        h_new = h - alpha * gradient
        
        # Project to feasible region [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum = n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_C5 = evaluate_erdos_solution(h_new, n_points)
        
        # Update if better
        if new_C5 < best_C5:
            best_h = h_new.copy()
            best_C5 = new_C5
            h = h_new.copy()
            alpha = min(1.0, alpha * 1.1)  # Increase step size when improving
        else:
            alpha = max(1e-6, alpha * 0.9)  # Decrease step size after no improvement
            
            # Try perturbations to escape local minima
            for _ in range(max_perturbations):
                if time.time() - start_time > budget_s - 1:
                    break
                noise = np.random.normal(0, perturbation_std, n_points)
                h_perturbed = h + noise
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
                perturbed_C5 = evaluate_erdos_solution(h_perturbed, n_points)
                if perturbed_C5 < best_C5:
                    best_h = h_perturbed.copy()
                    best_C5 = perturbed_C5
                    h = h_perturbed.copy()
                    break
    
    return (best_h.tolist(), best_C5, n_points)

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
