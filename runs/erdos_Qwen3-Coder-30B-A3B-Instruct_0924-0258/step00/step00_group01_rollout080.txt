import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Fallback: uniform distribution
        n_points = 47
        h = np.full(n_points, 1.0 / n_points)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vals):
        # Compute overlap integral C5
        h_vals = np.array(h_vals)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    def constraint_bounds(h_vals):
        return np.min(h_vals), np.max(h_vals) - 1.0
    
    def optimize_step(h_vals, current_C5):
        # Gradient approximation using finite differences
        eps = 1e-5
        grad = np.zeros_like(h_vals)
        h_copy = h_vals.copy()
        
        for i in range(len(h_vals)):
            h_plus = h_copy.copy()
            h_minus = h_copy.copy()
            
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Ensure bounds
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            C5_plus = compute_C5(h_plus)
            C5_minus = compute_C5(h_minus)
            
            grad[i] = (C5_plus - C5_minus) / (2 * eps)
        
        # Move along negative gradient
        step_size = 0.01
        new_h = h_vals - step_size * grad
        
        # Project back into feasible set
        new_h = np.clip(new_h, 0, 1)
        
        # Rescale to meet sum constraint
        current_sum = np.sum(new_h)
        if abs(current_sum) > 1e-10:
            new_h = new_h * target_sum / current_sum
            
        return new_h
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Main optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        new_h = optimize_step(best_h, best_C5)
        new_C5 = compute_C5(new_h)
        
        if new_C5 < best_C5:
            best_h = new_h.copy()
            best_C5 = new_C5
            
        if best_C5 < 0.3807:
            break
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    return (list(best_h), final_C5, len(best_h))
