import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    # Compute the overlap integral C5 using the provided method
    n_points = len(h_values)
    dx = 2.0 / n_points
    # np.correlate with mode='full' computes all lags
    full_correlation = correlate(h_values, 1 - h_values, mode='full')
    # Extract the relevant part and multiply by dx
    C5 = np.max(full_correlation) * dx
    return C5

        # Compute all cross-correlations
        full_correlation = correlate(h_values, 1 - h_values, mode='full')
        C5_vals = full_correlation * dx
        C5 = np.max(C5_vals)

        max_val = np.max(C5_vals)
        max_indices = np.where(np.abs(C5_vals - max_val) < 1e-10)[0]

            # Fallback: single index case
            max_idx = np.argmax(C5_vals)
            max_indices = [max_idx]

        # Build gradient
        gradient = np.zeros(n_points)

            # The contribution to the gradient for each h[j] from lag k
            # Specifically, the term in the correlation sum:
            # h[i] * (1 - h[i+k]) for i = 0 to n-k-1
            # The derivative of this with respect to h[j] is:
            # j == i -> (1 - h[i+k])
            # j == i+k -> -h[i]
            # others -> 0
            
            # Case k = 0 (correlation with itself):
            if k == n_points - 1:
                for i in range(n_points):
                    gradient[i] += (1 - h_values[i])
            else:
                # Lag k corresponds to offset k - (n_points-1) in full array
                # i = 0 to n_points-1-k
                for i in range(n_points - k):
                    gradient[i] += 1.0 - h_values[i + k]
                    if i + k < n_points:
                        gradient[i + k] -= h_values[i]

        norm_grad = np.linalg.norm(gradient)
        if norm_grad > 0:
