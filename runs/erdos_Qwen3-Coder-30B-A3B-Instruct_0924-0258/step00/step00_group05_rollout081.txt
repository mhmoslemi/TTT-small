import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    # This function evaluates the overlap integral for given h_values
    # This is a placeholder for the actual evaluation function provided in the environment
    dx = 2.0 / len(h_values)
    # Correlate h with (1-h) using full mode
    corr = correlate(h_values, 1 - np.array(h_values), mode="full")
    # Multiply by dx to get the integral approximations
    corr *= dx
    # Return the maximum value
    return np.max(corr)

def project_h(h, n_points):
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to satisfy sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    n_points = 70
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure sum constraint
    h = project_h(h, n_points)
    
    # Time tracking
    start_time = time.time()
    max_iter = 500
    alpha = 0.1
    tol = 1e-4
    last_c5 = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        # Compute correlations
        corr = correlate(h, 1 - h, mode="full")
        corr *= dx
        max_val = np.max(corr)
        max_indices = np.where(corr == max_val)[0]
        
        # Get the middle lag indices for the maximum correlation
        # We care about the lags that give max correlation
        middle_index = len(corr) // 2
        max_lags = [idx - middle_index for idx in max_indices]
        
        # Compute gradient for min of max overlaps
        # Gradient w.r.t. h[i]
        grad = np.zeros_like(h)
        for lag in max_lags:
            if lag == 0:
                # This is when h[i] multiplies with (1-h[i])
                grad += -1
            elif lag > 0:
                # h[i] multiplies with (1-h[i+lag])
                for j in range(len(h)-lag):
                    grad[j] -= 1
            else:
                # lag < 0
                # h[i] multiplies with (1-h[i+lag])
                for j in range(-lag, len(h)):
                    grad[j] -= 1
        
        # Apply gradient descent
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        
        h = h_new
        
        c5_new = evaluate_erdos_solution(h)
        
        if abs(last_c5 - c5_new) < tol:
            break
        last_c5 = c5_new
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h)
    
    return (h.tolist(), final_c5, n_points)
