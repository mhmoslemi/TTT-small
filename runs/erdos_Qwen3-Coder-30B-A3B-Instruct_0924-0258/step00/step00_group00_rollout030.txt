import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Initialize with a simple construction
        n_points = 45
        h = np.full(n_points, 0.5)
        h[0] = 0.0
        h[-1] = 1.0
        h[1:-1] = 0.0
        # Adjust to meet constraint
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Frank-Wolfe parameters
    max_iter = 500
    tol = 1e-4
    lr = 0.9
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all cross-correlations
        corr = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        max_corr = corr[k_max]
        
        # Compute gradient of this correlation with respect to h[i]
        # This is h[i + k_max] for the first term, and -h[i + k_max] for the second
        # The gradient at position i is the coefficient of h[i] in the derivative
        # This is just the coefficient from the correlation structure
        
        grad = np.zeros_like(h)
        if k_max < n_points:
            # Forward correlation
            for i in range(n_points - k_max):
                grad[i] += 1 - h[i + k_max]
        else:
            # Backward correlation
            k = k_max - n_points + 1
            for i in range(n_points - k):
                grad[i + k] -= h[i]
                
        # Apply Frank-Wolfe update: find s that maximizes gradient and update
        # We could use a more complex method here, but let's use a simple line search
    
        # Try some line search steps
        best_gamma = 1.0
        min_C5 = best_C5
        
        candidates = [0.1, 0.25, 0.5, 0.75, 1.0]
        for gamma in candidates:
            h_new = h + gamma * (1 - h) * (grad > 0) - gamma * (1 - h) * (grad < 0)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            
            C5_new = evaluate_C5(h_new, dx)
            if C5_new < min_C5:
                min_C5 = C5_new
                best_gamma = gamma
                
        # Update h with best gamma
        h_new = h + best_gamma * (1 - h) * (grad > 0) - best_gamma * (1 - h) * (grad < 0)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        h = h_new
        
        if min_C5 < best_C5:
            best_C5 = min_C5
            best_h = h.copy()
    
    return (best_h.tolist(), best_C5, n_points)

def evaluate_C5(h, dx):
    n_points = len(h)
    corr = correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)
