def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    else:
        # Create a reasonable starting point if initial_h_values is not available
        h = np.ones(n_points) * 0.5
        
    # Normalize to meet constraint
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    # Gradient-based optimizer using projected gradient descent
    learning_rate = 0.01
    decay = 0.9995
    max_iter = int(budget_s / 1.0)  # Roughly 1s per iteration
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        h_flip = 1 - h
        correlations = np.correlate(h, h_flip, mode='full')
        max_shift_idx = np.argmax(correlations)
        max_overlap = correlations[max_shift_idx]
        
        # Compute gradient
        # We are computing d/dh of sum(h_i * (1 - h_{i+k}))
        # For the k that achieves max overlap:
        grad = np.zeros_like(h)
        k = max_shift_idx - (n_points - 1)  # The actual shift
        
        if k >= 0:
            grad[k:] += -h_flip[:-k]
            grad[:-k] += h_flip[k:]
        else:
            k = -k
            grad[k:] += -h_flip[:-k]
            grad[:-k] += h_flip[k:]
        
        # Update h using projected gradient descent
        h = h - learning_rate * grad
        
        # Project h onto feasible set
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        # Evaluate current solution
        c5 = evaluate_erdos_solution(h)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Anneal learning rate
        learning_rate *= decay
        
        # Occasionally add noise to escape local minima
        if i % 100 == 0:
            noise = np.random.uniform(-0.01, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
