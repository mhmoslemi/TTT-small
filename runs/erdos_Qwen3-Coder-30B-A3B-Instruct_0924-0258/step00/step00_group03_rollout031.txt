def mutate_h(h, mutation_rate=0.1):
    """Mutate h to get a new feasible solution."""
    n = len(h)
    # Create a copy
    new_h = h.copy()
    # Randomly mutate some elements
    num_mutations = int(mutation_rate * n)
    indices = np.random.randint(0, n, num_mutations)
    # For each index, add small random change and make sure it stays in [0,1]
    for i in indices:
        # Uniform on [-0.1, 0.1], but ensure h stays in [0,1]
        delta = (np.random.rand() - 0.5) * 0.2
        new_h[i] = np.clip(new_h[i] + delta, 0, 1)
    # Renormalize the vector to sum to n/2. Just rescale to match desired sum
    current_sum = np.sum(new_h)
    desired_sum = n / 2.0
    if current_sum > 0:
        new_h = new_h * desired_sum / current_sum
    return new_h

def crossover_h(h1, h2):
    """Create a new h from two parents."""
    n = len(h1)
    # Simple uniform crossover
    mask = np.random.rand(n) > 0.5
    new_h = np.where(mask, h1, h2)
    # Renormalize
    current_sum = np.sum(new_h)
    desired_sum = n / 2.0
    if current_sum > 0:
        new_h = new_h * desired_sum / current_sum
    return new_h

    # Track the best solution
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    elapsed = 0
    gen = 0
    start_time = time.time()
    
    while elapsed < budget_s and gen < max_gen:
        gen += 1
        # Selection via tournament
        selected = []
        for _ in range(pop_size):
            contestants = np.random.choice(pop_size, size=5, replace=False)
            best_of_tournament = contestants[np.argmin([fitnesses[i] for i in contestants])]
            selected.append(best_of_tournament)
        
        # Crossover+Mutation to create new population
        new_population = []
        for i in range(0, pop_size, 2):
            ind1 = selected[i]
            ind2 = selected[i+1] if i+1 < pop_size else selected[0]
            child1 = crossover_h(population[ind1], population[ind2])
            child2 = crossover_h(population[ind2], population[ind1])
            
            # Mutate children
            child1 = mutate_h(child1, mutation_rate=0.1)
            child2 = mutate_h(child2, mutation_rate=0.1)
            
            new_population.extend([child1, child2])
        
        # Keep only first pop_size individuals
        new_population = new_population[:pop_size]
        
        # Evaluate new population
        new_fitnesses = []
        for h in new_population:
            fit = evaluate_h(h)
            new_fitnesses.append(fit)
            if fit < best_c5:
                best_c5 = fit
                best_h = h.copy()
        
        # Update population and fitnesses
        population = new_population
        fitnesses = new_fitnesses
        
        # Update elapsed time
        elapsed = time.time() - start_time
    
    # Final evaluation
    best_c5 = evaluate_h(best_h)
    
    return (best_h.tolist(), best_c5, len(best_h))
