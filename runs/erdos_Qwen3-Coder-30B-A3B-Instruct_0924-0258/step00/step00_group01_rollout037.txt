import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 47
        h = np.full(n_points, 0.5)
    
    # Ensure the constraint sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Time tracking
    start_time = time.time()
    
    # Optimization parameters
    alpha = 0.1
    momentum = 0.9
    beta = 0.95
    patience = 50
    best_C5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent with momentum
    v = np.zeros_like(h)
    
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s:
        iter_count += 1
        
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        k0 = np.argmax(corr)
        max_overlap = corr[k0] * (2.0 / n_points)
        
        # If this is our best solution so far, update
        if max_overlap < best_C5:
            best_C5 = max_overlap
            best_h = h.copy()
            last_improvement = iter_count
            
        # Compute gradient for active shift k0
        grad = np.zeros_like(h)
        if k0 < n_points:
            # Gradient due to h[i] * (1 - h[i + k0])
            for i in range(n_points - k0):
                grad[i] += 1 - h[i + k0]
            # Gradient due to h[i + k0] * (1 - h[i])
            for i in range(k0, n_points):
                grad[i] -= h[i - k0]
        else:
            # m = k0 - n_points + 1, conflicts with np.correlate logic
            k_shift = k0 - n_points + 1
            for i in range(n_points - k_shift):
                grad[i] += 1 - h[i + k_shift]
            for i in range(k_shift, n_points):
                grad[i] -= h[i - k_shift]
        
        # Update using momentum and gradient
        v = momentum * v - alpha * grad
        h_new = h + v
        
        # Project onto [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust sum to maintain constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive step size
        if iter_count - last_improvement > patience:
            alpha *= beta
            last_improvement = iter_count
            # Random restart
            if np.random.rand() < 0.05:
                h = np.full(n_points, 0.5)
                h = h * (n_points / 2) / np.sum(h)
                
    return (best_h.tolist(), best_C5, n_points)
