        # correlation will map 1-h into frequency domain and multiply
        # Using np.correlate as reference method for now
        h1 = h.copy()
        h2 = 1 - h.copy()
        
        # Compute full cross-correlation using numpy
        corr = np.correlate(h1, h2, mode='full')
        # Note: only part of this is valid; treat it as all useful data points
        return corr
    
    # Wrapper to effectively compute C5
    def calculate_C5(h):
        h1 = h.copy()
        h2 = 1 - h.copy()
        corr = np.correlate(h1, h2, mode='full')
        # Extract the relevant portion - the overlap integrals are at positions
        # from n_points to 2*n_points-2, because we compute a full correlation
        # So corr[n_points-1:2*n_points-1] represents all relevant terms
        valid_corr = corr[n_points-1:2*n_points-1]
        return np.max(valid_corr) * dx
    
    # Optimization loop
    alpha = alpha_start
    last_improvement_iter = 0
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        iter_count += 1
        
        # Compute overlaps using convolution
        overlap_values = compute_overlaps(h)
        # Extract valid values
        valid_overlaps = overlap_values[n_points-1:2*n_points-1]
        k_max = np.argmax(valid_overlaps)

        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            last_improvement_iter = iter_count
            
        # Compute gradient manually to avoid numerical instability
        # We want to minimize the max overlap
        # This is done by moving in direction of steepest descent of overlap
        
        # Optimize gradient computation carefully:
        # Inefficient approach for illustration - simply relying on correlate
        # But we can use FWHT trick for that if needed in precision or speed
        # For now we approximate gradient of overlap with respect to h

        # compute gain in removing h[i] from overlap with h[i+lag]

        # Effectively this is grad of max violation
        h_shift = np.roll(h, lag)
        h_reverse_shift = np.roll(h, -lag)
        grad = -(1 - h_shift) - h_reverse_shift
        
        # Clip to reasonable delta
        grad = np.clip(grad, -1, 1)
        
        # Update
        h += alpha * grad
        
        # Project back to [0,1]
        h = np.clip(h, 0, 1)
        
        # Adjust sum to exactly n_points / 2
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-6:
            scale_factor = (n_points / 2) / h_sum
            h *= scale_factor
            
        # Decay learning rate
        if iter_count % 50 == 0 and iter_count > last_improvement_iter + 50:
            alpha *= alpha_decay
            
        # Random perturbation to escape local minima
        if np.random.rand() < perturbation_prob:
            pert = np.random.normal(0, 0.01, n_points)
            h += pert
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h *= (n_points / 2) / h_sum
            
    # Final C5 evaluation
    final_c5 = calculate_C5(best_h)
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    actual_sum = np.sum(best_h)
    best_h *= (n_points / 2) / actual_sum
    
    return (best_h.tolist(), final_c5, n_points)
