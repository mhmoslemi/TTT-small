import numpy as np
from scipy.optimize import minimize
import time
import random
from math import sqrt

def compute_fitness(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # Take the maximum (excluding edge effects)
    max_corr = max(corr[n-1:])
    return max_corr * dx

def project(h):
    # Clamp values
    h = np.clip(h, 0, 1)
    # Normalize sum to n/2
    target_sum = len(h) / 2.0
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def mutate(h, sigma):
    h_new = h + np.random.normal(0, sigma, len(h))
    return project(h_new)

def crossover(h1, h2, alpha=0.6):
    # Blend crossover
    beta = np.random.uniform(alpha, 1 - alpha)
    h_new = beta * h1 + (1 - beta) * h2
    return project(h_new)

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Initial construction
    if 'initial_h_values' in globals():

        n_points = len(h_base)
    else:

        n_points = 70
        h_base = np.full(n_points, 0.5)
        # Adjust to satisfy integral constraint
        h_base = project(h_base)
    
    # Population parameters
    population_size = 100
    generations = 1000
    elite_size = int(0.4 * population_size)
    random_size = int(0.1 * population_size)
    crossover_size = int(0.3 * population_size)
    
    # Record start time
    start_time = time.time()
    
    # Initialize population
    population = [h_base.copy()]
    for _ in range(population_size - 1):
        h_mut = mutate(h_base, 0.1)
        population.append(h_mut)
    
    # Evaluate initial population
    fitnesses = [compute_fitness(h) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Tournament selection
        selected = []
        
        # Elite selection
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for idx in elite_indices:
            selected.append(population[idx].copy())
        
        # Random selection
        for _ in range(random_size):
            idx = random.randint(0, population_size - 1)
            selected.append(population[idx].copy())
        
        # Crossover
        offspring = []
        for _ in range(crossover_size):
            idx1, idx2 = random.sample(range(len(selected)), 2)
            child = crossover(selected[idx1], selected[idx2])
            offspring.append(child)
        
        # Mutation for offspring
        mutated_offspring = []
        sigma = 0.1 * np.exp(-gen / 100)
        for child in offspring:
            child_mut = mutate(child, sigma)
            mutated_offspring.append(child_mut)
            
        # Local refinement for top 5%
        top_indices = np.argsort(fitnesses)[:int(0.05 * population_size)]
        for idx in top_indices:
            h_current = population[idx].copy()
            # Simple gradient descent step
            learning_rate = 0.02 * np.exp(-gen / 100)
            # Approximate gradient: finite difference
            h_grad = np.zeros_like(h_current)
            eps = 1e-4
            base_fitness = compute_fitness(h_current)
            for i in range(len(h_current)):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = project(h_plus)
                fitness_plus = compute_fitness(h_plus)
                h_grad[i] = (fitness_plus - base_fitness) / eps
            
            # Update with gradient step
            h_new = h_current - learning_rate * h_grad
            h_new = project(h_new)
            
            mutated_offspring.append(h_new)
            
        # Evaluate new population
        new_population = selected + mutated_offspring
        new_fitnesses = [compute_fitness(h) for h in new_population]

        min_idx = np.argmin(new_fitnesses)
        if new_fitnesses[min_idx] < best_fitness:
            best_fitness = new_fitnesses[min_idx]
            best_h = new_population[min_idx].copy()
        
        population = new_population
        fitnesses = new_fitnesses
        
        # Periodic report

    # Final evaluation
    final_fitness = compute_fitness(best_h)
    
    # Return results
    return (list(best_h), final_fitness, len(best_h))
