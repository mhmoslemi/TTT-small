import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the overlap integral for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # Extract relevant part (k from -n+1 to n-1)
    k_indices = np.arange(-n + 1, n)
    # Take max of positive k values to get C5
    max_corr = np.max(corr[n-1:])
    return max_corr * dx

def make_feasible(h):
    """Ensure h is feasible: 0 <= h[i] <= 1 and sum(h) = n/2."""
    n = len(h)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n / 2
    if abs(current_sum - target_sum) > 1e-10:
        # Rescale to achieve target sum
        h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Crossover two parent solutions."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    n = len(parent1)
    # Single-point crossover
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, noise_scale=0.1):
    """Mutate a candidate solution."""
    if np.random.rand() > mutation_rate:
        return h.copy()
    n = len(h)
    noise = np.random.normal(0, noise_scale, n)
    h_mut = h + noise
    return make_feasible(h_mut)

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection."""
    indices = np.random.choice(len(population), size=tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()
