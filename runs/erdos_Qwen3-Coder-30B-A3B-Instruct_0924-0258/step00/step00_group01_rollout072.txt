import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute the overlap integral using correlation
    corr = correlate(h, 1 - h, mode='full')
    # Take the maximum over all lags
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_feasible(h, n_points):
    # Ensure h is feasible: sum = n_points/2 and values in [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    # Adjust by scaling to preserve relative distribution
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        # If sum is zero, spread evenly
        h = np.full(n_points, 1.0 / n_points)
    return h

def initialize_population(n_points, pop_size, initial_h_values=None):
    # Create a starting population
    population = []
    if initial_h_values is not None and len(initial_h_values) == n_points:
        # Use the provided initial solution
        h = np.array(initial_h_values)
        h = project_feasible(h, n_points)
        population.append(h)
    else:
        # Default construction
        h = np.full(n_points, 0.5)
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Fill with random feasible individuals
    while len(population) < pop_size:
        h = np.random.random(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    return np.array(population)

def selection(population, fitnesses, tournament_size=3):
    # Tournament selection
    selected = []
    n = len(population)
    for _ in range(n):
        indices = np.random.choice(n, tournament_size, replace=False)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        selected.append(population[best_idx])
    return np.array(selected)

def crossover(parent1, parent2, crossover_rate=0.8):
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    # Point crossover
    n = len(parent1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.05, std=0.1):
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h[i] += np.random.normal(0, std)
    h = project_feasible(h, len(h))
    return h

def ga_optimize(n_points, initial_h_values=None, pop_size=20, max_generations=1000, budget_s=600):
    start_time = time.time()
    best_h = None
    best_c5 = np.inf
    
    # Initialize population
    population = initialize_population(n_points, pop_size, initial_h_values)
    dx = 2.0 / n_points
    
    generation = 0
    no_improvement_count = 0
    last_best = np.inf
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Check for stagnation
        if best_c5 < last_best:
            no_improvement_count = 0
            last_best = best_c5
        else:
            no_improvement_count += 1
        
        # Selection
        selected = selection(population, fitnesses)
        # Crossover and mutation for new population
        new_population = []
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[min(i+1, len(selected)-1)]
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.append(child1)
            new_population.append(child2)
        
        # Keep the best from previous generation
        population = new_population[:pop_size]
        # Ensure we still have the best individual
        if best_h is not None and len(population) > 0:
            # Replace worst with best if there's a space
            worst_idx = np.argmax([evaluate_h(h, dx) for h in population])
            if evaluate_h(population[worst_idx], dx) > best_c5:
                population[worst_idx] = best_h
        
        generation += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 47  # Use the same as initial_h_values
    initial_h_values = kwargs.get('initial_h_values', None)
    
    h_values, c5_bound = ga_optimize(
        n_points=n_points,
        initial_h_values=initial_h_values,
        pop_size=20,
        budget_s=budget_s
    )
    return (h_values.tolist(), c5_bound, n_points)
