import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial h values or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default construction: simple step function
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Precompute indices
    indices = np.arange(n_points)
    
    def compute_c5(h):
        """Compute C5 from h"""
        # Compute all overlaps
        h_rev = np.flip(h)
        corr = correlate(h, h_rev, mode='full')
        # Overlaps are indexed from 0 to 2*n_points-2
        # We are interested in k in [0, n_points-1] corresponds to indices n_points-1 to 2*n_points-2
        # For overlap_k, it's the value at index n_points-1+k
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            overlaps[k] = correlate(h, 1-h, mode='full')[n_points-1+k] * dx
        return np.max(overlaps)
    
    def project_feasible(h):
        """Project h onto feasible set [0,1], sum(h) = target_sum"""
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        else:
            h = np.ones_like(h) * (target_sum / n_points)
        return h
    
    def update_h(h, grad, momentum, lr):
        """Update h using momentum and learning rate"""
        momentum = 0.9 * momentum - lr * grad
        h = h + momentum
        return h, momentum
    
    # Start timing
    start_time = time.time()
    
    # Initialize momentum
    momentum = np.zeros_like(h)
    
    # Evaluate initial solution
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    # Set initial learning rate
    lr = 0.01
    
    # Optimization loop
    iteration = 0
    
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        
        # Compute all overlaps
        h_rev = np.flip(h)
        corr = correlate(h, h_rev, mode='full')
        # Overlaps for all valid shifts
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            overlaps[k] = corr[n_points-1+k] * dx
        # Get indices of top contributing k's (say top 5)
        top_k_indices = np.argpartition(overlaps, -5)[-5:]
        
        # Manually compute gradient for the dominant overlaps
        grad = np.zeros_like(h)
        for k in top_k_indices:
            # Gradients for the k-th overlap term:
            # Let overlap_k = sum(h[i] * (1 - h[i+k])) * dx
            # d/dh[i] = (1 - h[i+k]) * dx   if i+k < n_points
            # d/dh[i] = (1 - h[i+k-n_points]) * dx  if i+k >= n_points  
            # indices are i, i+k or i+k-n_points
            # We consider shift k for (index i) -> (index i+k_rel)
            # wrap-around using modulo
            for i in range(n_points):
                # Compute h[i] * (1 - h[(i+k)%n_points])
                # Partial derivative w.r.t. h[i] is (1 - h[(i+k)%n_points]) * dx
                # Partial derivative w.r.t. h[(i+k)%n_points] is  -h[i] * dx
                i_k = (i + k) % n_points
                grad[i] += (1 - h[i_k]) * dx
                # We compute Jacobian of the max over all k
                # But partial derivative wrt h[i] from overlap_l is 0 if l != k
                # So just take those with k in top_k_indices
                if k in top_k_indices and i_k < n_points:
                    # i_k = i + k => i_k - k = i
                    # so d/dh[i] = -delta_{i_k} + dx for each relevant k only
                    grad[i] += -h[i_k] * dx  # -h[i+k] for i \in [0, n_points-1]
        
        # Update h with momentum
        h, momentum = update_h(h, grad, momentum, lr)
        
        # Project to feasible set
        h = project_feasible(h)
        
        # Compute new C5
        c5 = compute_c5(h)
        
        # Update best if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            lr = max(0.001, lr * 1.1)  # Increase learning rate if improving
        else:
            lr = max(0.0001, lr * 0.95)  # Decrease LR if not improving
            
        # Early stopping: no large improvement for 50 iterations
        if iteration > 50 and lr < 0.0001:
            break
            
    return (best_h.tolist(), best_c5, n_points)
