import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Fallback: create a reasonable initial guess
        n_points = 70
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Parameters
    learning_rate = 0.01
    decay_factor = 0.95
    decay_interval = 100
    
    for iteration in range(10000):  # Limit iterations but check time
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the full correlation
        corr = correlate(h, 1 - h, mode="full")
        k0 = np.argmax(corr)
        max_val = corr[k0]
        
        # Compute gradient w.r.t h at the max correlation position
        grad = np.zeros(n_points)
        # Refresh: since we're computing for correlation at max k0
        # This is a simplification of gradient computation (the conv. gradient)
        # We'll approximate derivative by finite differences for the maximum
        # This is a simplified approach that capitalizes on continuous approx
        # The correct gradient for overlap involves computing derivative w.r.t max k location
        
        # Instead, we simulate a gradient-based update:
        # The symmetric gradient for h is the derivative of max overlap
        # To take an efficient step:
        # we first compute the lags
        # And update h by a gradient step in direction of decreasing max overlap
        grad_p = np.zeros(n_points)
        # Create a triangular reflection of h and book-keep indices
        for i in range(n_points):
            shifted = (i - k0 + n_points) % n_points
            if shifted < n_points:
                grad_p[i] = 1 - h[shifted]
            else:
                grad_p[i] = 0  # default for out-of-bound (which is symmetrized)
            # due to symmetry constraints: use simple measure
        # Compute reasonable approximation:
        # For k0 shift, the gradient of the overlap is approximately
        # -h[j] + 1 for j such that j + k0 < n_points for the shifted position
        # Let's flip the logic:
        # we're adjusting the entries of h to reduce max_val in the time-domain
        
        # Simplified update step:
        # Use projection to maintain bounds and sum
        # g = 1-h for forward correlation
        # accurate: grad[i] = contribution of h[i] to max_corr
        
        # The -1 is due to derivative of 1-x wrt x = -1
        # Proposition: gradient is the "mechanical" part of corr derivative
        # There's a specific lag k0 where we get the maximum
        # The gradient of the max with respect to h at lag k0 is 
        # (1 - h) at appropriate indices
        
        # Since this is complex to compute directly, we'll use:
        # 1. A large step in gradient direction (or we won't go anywhere)
        # 2. Project back into feasible set
        
        # Analyze from the zero input perspective:
        # For a smooth version, if h(x) = y we have again 1-h(x+k) so
        # In linear marginal, this contributes y to correlation.
        # So, for max contribution at k0, take:
        # A step toward the direction of steepest descent for one of the lags
        # Vectorized: take gradient w.r.t specific lag
        # Now project via CVXPY:
        
        # Gradient of h[0], that contributes to h[0] + h[k0]
        # When max_lag is k0, i.e. h[0] * 1-h[k0], gradient is 1-h[k0]
        # But we want to reduce the value at this index.
        
        # Defining gradient easily 
        # Simplified approach:
        # For correlation at lag k0
        # h[i] * (1-h[i+k0]) 
        # grad[0] = 1 - h[k0]
        # grad[1] = 1 - h[k0+1] etc. for the contribution to corr[k0]
        
        # Reverse engineering gradient:
        k0_shifted = k0 - (n_points - 1)
        # Build exact gradient using sliding window effect
        # Let's compute the actual gradient w.r.t sum_j h[j] * (1-h[j+k0])
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k0_shifted) % n_points
            grad[i] = - (1 - h[j])  # Current prediction of gradient
            
        # Scale to potential improvement
        # if max_val is large, gradient direction should shift h[i] (when j is influenced)
        
        # Stochastic version for simplicity
        h_new = h - learning_rate * grad
        # Project onto feasible set: sum(h) = 1, 0 <= h[i] <= 1
        h_projected = project_onto_feasible(h_new, n_points, dx)
        h = h_projected
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        if iteration % decay_interval == 0:
            learning_rate *= decay_factor
                
    return (best_h.tolist(), best_c5, n_points)

def project_onto_feasible(h, n_points, dx):
    # Using Convex Optimization with CVXPY for projection
    # min ||x - h||^2 subject to 0 <= x <= 1 and sum(x) = n_points / 2
    # That is equivalent to sum(x) * dx = 1
    x = cp.Variable(n_points)
    constraints = [
        x >= 0,
        x <= 1,
        cp.sum(x) == n_points / 2
    ]
    objective = cp.Minimize(cp.sum_squares(x - h))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        return np.array(x.value)
    except:
        # Fallback if CVXPY fails (which it shouldn't in typical cases)
        projected = np.clip(h, 0, 1)
        projected = projected * (n_points / 2) / np.sum(projected)
        return projected

def compute_c5(h, dx):
    # Compute the overlap integral using full convolution
    corr_full = correlate(h, 1 - h, mode="full")
    return np.max(corr_full) * dx
