import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():

        h[:n_points//2] = 1.0
        h += 0.05 * np.random.randn(n_points)
        h = np.clip(h, 0, 1)
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    
    # Keep track of best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Iteration parameters
    max_no_improvement = 20
    no_improvement_count = 0
    step_sizes = [0.1, 0.25, 0.5]
    pert_magnitude = 0.01
    
    def compute_C5(h_vals):
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        max_corr = 0
        for k in range(len(corr)):
            # Correct for indexing
            if k >= len(h_vals) - 1:
                # This correctly maps to lag values 0 to n_points-1
                lag = k - (len(h_vals) - 1)
                val = corr[k] * dx
                if val > max_corr:
                    max_corr = val
        return max_corr

    def compute_gradient(h_vals, max_lags):
        grad = np.zeros_like(h_vals)
        for k in max_lags:
            for i in range(len(h_vals) - k):
                if i < len(h_vals):
                    grad[i] += (1 - h_vals[i + k])
                if i + k < len(h_vals):
                    grad[i + k] -= h_vals[i]
        return grad

    def project_feasible(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        return h_vals

    iteration = 0
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        # Compute correlations
        corr = correlate(h, 1 - h, mode='full')
        # Compute maximum correlation and indices
        max_val = -1
        max_indices = []
        for k in range(len(corr)):
            lag = k - (n_points - 1)
            if lag >= 0 and lag < n_points:
                val = corr[k] * dx
                if val > max_val:
                    max_val = val
                    max_indices = [lag]
                elif val == max_val:
                    max_indices.append(lag)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for lag in max_indices:
            k = lag
            for i in range(n_points - k):
                if i < n_points:
                    grad[i] += (1 - h[i + k])
                if i + k < n_points:
                    grad[i + k] -= h[i]
        
        # Try different step sizes
        best_h_candidate = h.copy()
        best_C5_candidate = compute_C5(h)
        
        for alpha in step_sizes:
            h_candidate = h - alpha * grad
            h_candidate = project_feasible(h_candidate)
            c5_candidate = compute_C5(h_candidate)
            if c5_candidate < best_C5_candidate:
                best_C5_candidate = c5_candidate
                best_h_candidate = h_candidate.copy()
        
        # Check if we improved
        if best_C5_candidate < best_C5:
            h = best_h_candidate.copy()
            best_C5 = best_C5_candidate
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Escape strategy
        if no_improvement_count >= max_no_improvement:
            noise = np.random.randn(n_points) * pert_magnitude
            h += noise
            h = project_feasible(h)
            no_improvement_count = 0

    # Ensure best solution is valid
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    # Double check final solution and return
    assert abs(np.sum(best_h) * dx - 1.0) < 1e-6, "Sum constraint violated"
    assert all(0 <= x <= 1 for x in best_h), "Bounds violated"
    
    return (best_h.tolist(), final_C5, n_points)
