import numpy as np
from scipy import optimize
import time
from math import sqrt

def evaluate_erdos_solution(h_values):
    """Evaluate the overlap integral for given step function h."""
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    # Compute cross-correlations for all lags
    corr = np.correlate(h, 1 - h, mode="full")
    # Only take the valid lags where the correlation is computed
    # The valid range is from lag -(n-1) to +(n-1)
    # But we want the maximum over all valid lags, ensuring 100% overlap
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points):
    """Project h to ensure sum(h) = n_points / 2 and 0 <= h[i] <= 1."""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s == 0:
        return np.full(n_points, 0.5)
    h = h * (n_points / 2) / s
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):

    best_c5 = evaluate_erdos_solution(h)
    
    # Frank-Wolfe-like optimization
    max_iter = 1000
    alpha = 0.05  # Learning rate
    tolerance = 1e-5
    min_improvement = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all cross-correlations
        h = project_to_feasible(h, n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        # Get valid lags from the middle
        n_corr = len(corr)
        valid_corr = corr[n_points - 1: 2 * n_points - 1]
        max_index = np.argmax(valid_corr)

        # Update gradient estimate
        # The gradient of correlation_k w.r.t. h[i] is 1 - h[i + k] if i+k >= 0

        # Compute direction vector - only the relevant gradient used
        # Compute derivatives for maximum correlation
        k = max_index - (n_points - 1)  # get actual lag
        
        # We compute partials for the top cross-correlation
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                # Incremental gradient
                gradient[i] += (1 - h[j]) * dx  # Derivative from correlate
        
        # Use Frank-Wolfe style update
        # QP subproblem: minimize gradient * d subject to constraints
        # Linearized problem:
        # maximize gradient * h (since we want to minimize negative)
        # subject to sum(h) = n_points / 2, 0 <= h[i] <= 1
        
        # Simple projection step for update direction
        # Take gradient direction
        # Sort indices in descending order of gradient
        sorted_idx = np.argsort(-gradient)
        
        # Control which elements to increase (some that are small)
        # For efficient Frank-Wolfe: use any initial set that is valid
        # In this version, use top indices based on gradient
        # Let's take 1/3 of points with highest gradient
        updates = int(n_points / 3)
        if updates == 0: updates = 1
        indices_to_increase = sorted_idx[:updates]
        
        # Line search over steps to minimize C5
        best_gamma = 0.0

    final_c5 = evaluate_erdos_solution(best_h)
