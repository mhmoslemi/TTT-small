import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Fallback: uniform distribution
        n_points = 47
        h = np.ones(n_points) * 0.5
        h = h * (n_points/2) / np.sum(h)  # normalize to sum=2
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    # Main optimization loop
    for it in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Gradient descent step
        grad = compute_gradient(h, dx)
        
        # Line search to find step size
        alpha = 0.1
        for _ in range(10):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            sum_h = np.sum(h_new) * dx
            h_new = h_new * (n_points/2) / sum_h  # rescale to sum = n_points/2
            
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < best_c5:
                h = h_new
                best_c5 = new_c5
                best_h = h.copy()
                break
            alpha *= 0.5
            
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    # Compute overlap integral C5 = max_k ∫ h(x)(1 - h(x+k)) dx
    conv = np.correlate(h, 1-h, mode="full")
    max_conv = np.max(conv)
    return max_conv * dx

def compute_gradient(h, dx):
    # Compute gradient using finite differences
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-6
    
    base_c5 = compute_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        sum_h = np.sum(h_plus) * dx
        h_plus = h_plus * (n/2) / sum_h
        
        c5_plus = compute_c5(h_plus, dx)
        
        grad[i] = (c5_plus - base_c5) / eps
        
    return grad
