import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure constraint sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_C5(h):
        # Compute the full correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # Max correlation is in the middle, but we'll compute all for precision
        max_val = np.max(corr) * dx
        return max_val
    
    def project_h(h):
        # Project to [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to meet constraint
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def finite_diff_gradient(h, eps=1e-4):
        grad = np.zeros_like(h)
        base_val = evaluate_C5(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            grad[i] = (evaluate_C5(h_plus) - base_val) / eps
        return grad
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    # Annealing parameters
    temp = 1.0
    cooling_factor = 0.99
    min_temp = 0.01
    max_patience = 50
    patience = 0
    gradient_scale = 0.1
    noise_std = 0.05
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        current_C5 = evaluate_C5(h)
        
        # Compute gradient
        try:
            grad = finite_diff_gradient(h)
        except:
            grad = np.zeros_like(h)
        
        # Generate candidate
        noise = np.random.normal(0, noise_std, n_points)
        h_candidate = h - gradient_scale * grad + noise
        h_candidate = project_h(h_candidate)
        
        new_C5 = evaluate_C5(h_candidate)
        
        # Accept or reject
        if new_C5 < current_C5:
            h = h_candidate
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_candidate.copy()
            patience = 0
        else:
            if np.random.random() < np.exp((current_C5 - new_C5) / temp):
                h = h_candidate
                patience = 0
            else:
                patience += 1
        
        # Update temperature
        temp = max(temp * cooling_factor, min_temp)
        
        # Warm restart if no improvement
        if patience >= max_patience:
            temp = 1.0
            patience = 0
        
        # Occasionally recheck best solution
        if iteration % 10 == 0:
            temp = max(temp * 0.995, min_temp)
    
    # Final evaluation
    final_C5 = evaluate_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
