import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Setup
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 2400  # Roughly 600s budget at ~0.25s/iter
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Normalize to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    perturb_steps = 20
    last_improved = 0
    alpha_candidates = [0.1, 0.5, 1.0]
    
    start_time = time.time()
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute all cross-correlations
        h1 = 1 - h
        correlations = correlate(h, h1, mode="full")
        k_indices = np.arange(-n_points + 1, n_points)
        max_k_idx = np.argmax(correlations)
        max_k = k_indices[max_k_idx]
        max_corr = correlations[max_k_idx]
        
        # Compute gradient for this lag
        grad = np.zeros(n_points)
        if max_k >= 0:
            grad[max_k:] = -h[:-max_k]
        else:
            grad[:max_k] = -h[-max_k:]
        
        # Try different step sizes
        best_h_new = h.copy()
        best_c5_new = best_c5
        
        # Try a few step sizes
        for alpha in alpha_candidates:
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            
            # Rescale to sum = n_points/2
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5_new:
                best_c5_new = c5_new
                best_h_new = h_new.copy()
        
        # If we improved
        if best_c5_new < best_c5:
            h = best_h_new.copy()
            best_c5 = best_c5_new
            last_improved = iter_num
        else:
            # Perturb if no improvement in a while
            if iter_num - last_improved > perturb_steps:
                # Add small noise
                noise = 0.01 * np.random.randn(n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                # Re-evaluate
                c5_new = compute_c5(h, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                last_improved = iter_num
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    n = len(h)
    # Compute all cross-correlations
    h1 = 1 - h
    correlations = correlate(h, h1, mode="full")
    # Get the maximum value scaled by dx
    return np.max(correlations) * dx
