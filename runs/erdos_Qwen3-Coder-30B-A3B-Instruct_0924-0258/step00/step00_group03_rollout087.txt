import numpy as np
from scipy.optimize import differential_evolution
import time
from math import sqrt

def initial_h_values():
    # Generate initial construction as described in the task
    # They are centered around initial_h_values if available
    # Create a simple symmetric construction for now
    n = 70
    h = np.ones(n) * 0.5
    # Adjust for integral constraint
    h[20:50] = 1.0
    h[:20] = 0.0
    h[50:] = 0.0
    h = h * (n / 2.0) / np.sum(h)
    return h

def evaluate_erdos_solution(h_values):
    # Precomputed function in the context
    n = len(h_values)
    dx = 2.0 / n
    max_val = 0
    for k in range(1-n, n):
        corr = np.correlate(h_values, 1-h_values, mode='full')
        if k >= 0:
            val = corr[n-1+k]
        else:
            val = corr[n-1+k]
        max_val = max(max_val, val * dx)
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction or start with a simple construction
    try:
        h0 = initial_h_values()
    except:
        h0 = np.ones(70) * 0.5
        h0[20:50] = 1.0
        h0[:20] = 0.0
        h0[50:] = 0.0
        h0 = h0 * (70 / 2.0) / np.sum(h0)
    
    n_points = len(h0)
    dx = 2.0 / n_points

    def objective(h):
        # clip within bounds [0,1]
        h = np.clip(h, 0, 1)
        # normalize to correct integral
        h = h * (n_points / 2.0) / np.sum(h)
        
        max_val = 0
        for k in range(1-n_points, n_points):
            # Compute convolution manually
            # When k >= 0, h[i+k] for i in [0, ..., n_points-k-1]
            # When k < 0, h[i] for i in [-k, ..., n_points-1]
            if k >= 0:
                # h[i] * (1 - h[i+k]) for i = 0..n_points-k-1
                overlap = np.sum(h[:-k] * (1 - h[k:]))
            else:
                # h[i] * (1 - h[i+k]) for i = -k..n_points-1
                overlap = np.sum(h[-k:] * (1 - h[:-k]))
            val = overlap * dx
            max_val = max(max_val, val)
        return max_val

    bounds = [(0, 1)] * n_points
    # Use differential evolution with a population size of 30
    # and a maximum of 200 generations
    res = differential_evolution(objective, bounds, seed=seed, maxiter=200, popsize=30, 
                                mutation=(0.5, 1), recombination=0.7, 
                                disp=False, polish=True)
    
    if not res.success:
        h_best = h0
    else:
        h_best = np.clip(res.x, 0, 1)
        h_best = h_best * (n_points / 2.0) / np.sum(h_best)
    
    # Evaluate final solution
    c5_bound = evaluate_erdos_solution(h_best)
    
    return (h_best, c5_bound, n_points)
