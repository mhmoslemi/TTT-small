import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_h(h, dx):
    """Evaluate C5 for a given step function h"""
    # Correlate h with (1-h) for all lags
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags, scaled by dx
    c5 = np.max(corr) * dx
    return c5

def project_feasible(h, n_points):
    """Ensure h is feasible: sum = n_points/2, values in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-10:
        # Linearly adjust to meet sum constraint
        h = h * target_sum / current_sum
        h = np.clip(h, 0, 1)
    return h

def initial_population(n_points, pop_size, initial_h):
    """Initialize population with given initial solution and random others"""
    pop = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        pop.append(h)
    return pop

def selection_tournament(pop, fitnesses, tournament_size=3):
    """Select one individual using tournament selection"""
    best_idx = random.randint(0, len(pop) - 1)
    for _ in range(tournament_size - 1):
        idx = random.randint(0, len(pop) - 1)
        if fitnesses[idx] < fitnesses[best_idx]:
            best_idx = idx
    return pop[best_idx]

def crossover(parent1, parent2):
    """Arithmetic crossover"""
    # Blend children
    alpha = 0.5
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(h, mutation_rate=0.05, std=0.1):
    """Gaussian mutation with projection to feasible region"""
    mask = np.random.rand(len(h)) < mutation_rate
    if np.any(mask):
        h = h + np.random.normal(0, std, len(h)) * mask
        h = project_feasible(h, len(h))
    return h

def genetic_optimization(initial_h, n_points, pop_size=20, generations=1000,

    """Genetic algorithm to minimize C5"""
    # Initialize population
    pop = initial_population(n_points, pop_size, initial_h)
    best_h = None

    start_time = time.time()
    
    for g in range(generations):
        if time.time() - start_time > 590:
            break
        
        # Evaluate all individuals

        for h in pop:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sorting pop by fitness for easier selection and replacement
        sorted_indices = np.argsort(fitnesses)
        pop = [pop[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Create new population
        new_pop = [pop[0], pop[1]]  # Elitism: keep best two
        while len(new_pop) < pop_size:
            # Tournament selection for parents
            parent1 = selection_tournament(pop, fitnesses)
            parent2 = selection_tournament(pop, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1, mutation_rate, 0.1)
            child2 = mutate(child2, mutation_rate, 0.1)
            
            # Add to new population
            new_pop.append(child1)
            if len(new_pop) < pop_size:
                new_pop.append(child2)
        
        pop = new_pop
    
    return best_h, best_c5
