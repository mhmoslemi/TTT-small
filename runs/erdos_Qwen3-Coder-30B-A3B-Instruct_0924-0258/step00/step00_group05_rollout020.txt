import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 70
        h = np.full(n_points, 0.5)
        h[0] = 0.4
        h[1] = 0.6
        h[-1] = 0.4
        h[-2] = 0.6
    
    # Ensure constraint sum(h) * dx = 1
    dx = 2.0 / n_points
    h_sum = np.sum(h)
    h *= (n_points / 2) / h_sum
    
    # Initialize best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, n_points)
    
    # Gradient descent parameters
    alpha = 0.1
    max_iter = 500
    improvement_threshold = 1e-4
    stagnation_count = 0
    max_stagnation = 10
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap integrals for all lags
        conv = np.correlate(h, 1 - h, mode="full")
        current_C5_vals = conv * dx
        max_idx = np.argmax(current_C5_vals)
        max_C5 = current_C5_vals[max_idx]
        
        # Identify lags achieving current maximum
        K_max = [max_idx - (n_points - 1)]
        if K_max[0] < 0:
            K_max = [K_max[0] + n_points]
            
        # Gradient computation based on K_max
        g = np.zeros(n_points)
        for k in K_max:
            for i in range(n_points):
                j = (i + k) % n_points
                g[i] += (1 - h[j]) - h[i]
        g *= dx
        
        # Update with gradient descent
        h_new = h - alpha * g
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        h_new *= (n_points / 2) / h_new_sum
        
        # Evaluate
        new_C5 = evaluate_erdos_solution(h_new, n_points)
        
        # Accept if improvement or periodically
        if new_C5 < best_C5:
            h = h_new.copy()
            best_C5 = new_C5
            best_h = h.copy()
            stagnation_count = 0
            alpha = min(0.1, alpha * 1.05)  # Increase alpha if improving
        else:
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                # Perturb and try again
                perturb = np.random.normal(0, 0.05, n_points)
                h_perturbed = h + perturb
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed_sum = np.sum(h_perturbed)
                h_perturbed *= (n_points / 2) / h_perturbed_sum
                new_C5_pert = evaluate_erdos_solution(h_perturbed, n_points)
                if new_C5_pert < best_C5:
                    h = h_perturbed.copy()
                    best_C5 = new_C5_pert
                    best_h = h.copy()
                else:
                    alpha *= 0.9  # Reduce step size
                stagnation_count = 0
        
        # Adaptive adjustment of alpha
        if iteration % 10 == 0 and iteration > 0:
            # Use small backward search for better step size
            test_h = h - alpha * g
            test_h = np.clip(test_h, 0, 1)
            test_h_sum = np.sum(test_h)
            test_h *= (n_points / 2) / test_h_sum
            test_C5 = evaluate_erdos_solution(test_h, n_points)
            if test_C5 < best_C5:
                alpha *= 1.1
            else:
                alpha *= 0.9
    
    return (best_h.tolist(), best_C5, n_points)

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr) * dx
    return max_val
