import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = 47  # Using the initial construction size
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if abs(np.sum(h) - target_sum) > 1e-6:
            # Normalize to correct sum
            h = h * target_sum / np.sum(h)
    else:
        h = np.full(n_points, 0.5)
    
    # Project h to [0,1] and ensure correct sum
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    # Optimizer parameters
    alpha = 0.1
    momentum = 0.9
    decay_factor = 0.95
    patience = 50
    patience_counter = 0
    best_C5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    
    # Precompute indices for correlation
    # size of full correlation is 2*n_points-1
    full_len = 2 * n_points - 1
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Compute correlations for all shifts
        corr = correlate(h, 1 - h, mode='full')
        # Select only the valid part
        # The maximum overlap occurs at shifts from 0 to 2*n_points-2
        # But we're interested in the actual shifts in [0, 2) which correspond to 
        # indices from 0 to n_points-1 and n_points+1 to 2*n_points-2  
        # Actually in our case, we want the maximum of these overlaps, the shift field.
        # Let's compute C5 based on full correlation values, it has 2*n_points-1 elements,
        # so it includes shift k = 0 to k = 2*n_points-2 (which is [0,2) in x space)
        # Normalized by dx
        corr = corr * dx
        max_corr_ind = np.argmax(corr)
        max_overlap = corr[max_corr_ind]  # The actual overlap with that shift
        if max_overlap < best_C5:
            best_C5 = max_overlap
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= patience:
                alpha *= decay_factor
                patience_counter = 0
        
        # Compute gradient with respect to active shift
        # The active shift k0 is at index max_corr_ind
        k0 = max_corr_ind
        
        # compute derivative of h_i w.r.t C5 (which is integral of h(x) * (1 - h(x + k0)))
        # As we go over all i, and for each i, we find the contribution to the integral.
        # There are two terms: 
        # h_i * (1-h_{i+k0}) for i < n_points and i+k0 < n_points
        # h_i * (1-h_{i-k0}) for i >= k0 and i-k0 < n_points
        
        # compute the gradient as gradient[i] = d/dh[i]( sum of h[i] * (1 - h[i+k0]) )
        # from shifted convolution in correlate_result, we get exact the form
        # if at shift k0, we compute contribution of h[i] * (1 - h[i+k0]) to the integral
        # We need to find how to compute partial derivatives of this over h
    
        # Direct calculation of gradient for the maximum overlapping
        # Let's reset gradient array
        gradient = np.zeros(n_points)
        # Assuming k0 is index in full correlation 
        # In correlate, if we do h conv (1-h), position max_corr_ind corresponds to shift k0
        # where shift k0 means h(x) * (1 - h(x+k0))
        # Elements [i] of gradient = (1 - h[i+k0]), all valid i, k0 boundary constrained
        # So, derivative of the term h[i] * (1 - h[i+k0]) 
        # w.r.t. h[i] is 1 - h[i+k0]
        # derivative of the term h[j] * (1 - h[j+k0]) w.r.t. h[i] is 0 if j!=i 
        # This is slightly to be interpreted: h[i] contributes to the result with each shift k0 based on h[i] * (1 - h[i+k0])
        
        # Better: go through each element of full correlation
        # gradient[i] refers to derivative of C5 w.r.t h[i]
        # let the overlap integral be: \int h(x) (1 - h(x + k0)) dx
        # whole integral is computed in correlate
        # For each element i, this is for current shift k0, contribution of h[i] * (1 - h[i+k0])
        # gradient w.r.t. h[i] = 1 - h[i+k0] (for valid i+k0)
        # and for h[i-k0] if k0<=i
        # Actually want derivative of the total integral as function of h[i]
        # ===>
        # C5 = \int h(x) (1 - h(x + k0)) dx
        # dC5/dh_i = (1 - h[i+k0]) if (i+k0)<len(h)
        # dC5/dh_i = -h[i-k0] if (i-k0) >= 0
        for i in range(n_points):
            contrib = 0
            if i + k0 < n_points:
                contrib += (1 - h[i + k0]) * dx
            # we need to consider only shifts where k0 <= i, i.e. where h[i-k0] term contributes
            if k0 <= i:
                contrib -= h[i - k0] * dx  # coef due to derivative wrt h[i]
            gradient[i] = contrib

        # Update h
        h_new = h - alpha * gradient
        
        # Project:
        h_new = np.clip(h_new, 0, 1)
        
        # Ensure the sum is correct
        current_sum = np.sum(h_new)
        diff = target_sum - current_sum
        if abs(diff) > 1e-10:
            # Add the difference to every element with equal proportion
            h_new += diff / n_points
        
        # Optionally add momentum (small imaginary change, if just looking at vector)
        # Then, we just continue with h_new in next iteration
        
        h = h_new
        
        # Random restart  to avoid local minima
        if iteration % 50 == 0:
            if np.random.rand() < 0.05:
                # Randomly perturb h towards a different local solution
                h = np.clip(h + np.random.normal(0, 0.02, n_points), 0, 1)
                # Re-normalize
                h = h * target_sum / np.sum(h)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    corr = corr * dx
    final_C5 = np.max(corr)
    if best_C5 > final_C5:
        best_C5 = final_C5
    return (best_h, best_C5, n_points)
