import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Problem parameters
    n_points = 47  # Based on initial_h_values
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Normalize initial guess to meet constraint
    h0 = h0 * target_sum / np.sum(h0)
    
    # Define objective function to minimize
    def evaluate(h):
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Adjust to meet sum constraint
        h = h * target_sum / np.sum(h)
        
        # Compute overlap integral for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        # Only consider relevant lags (0 to 2)
        # In full mode: indices from -n+1 to n-1, so we take middle n points
        # that correspond to lags 0 to n-1 -> we take indices n-1 to 2*n-2
        mid_idx = len(corr) // 2
        overlaps = corr[mid_idx:mid_idx + n_points]  # This is for lags 0 to n-1
        
        # We also need to consider negative shifts, so we take maximum
        # of all computed overlaps
        c5 = np.max(overlaps) * dx
        return c5
    
    # Use scipy.optimize.minimize with bounds and constraints
    # We'll try a gradient-free approach like Nelder-Mead since this is not smooth
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Start with our initial guess but make a small random perturbation for diversity
    x0 = h0.copy()
    # Small random perturbations
    x0 += np.random.normal(0, 0.01, n_points)
    x0 = np.clip(x0, 0, 1)
    # Project back to correct sum
    x0 = x0 * target_sum / np.sum(x0)
    
    # Define constraint: sum must equal target_sum
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    # Optimization callback for tracking
    best_c5 = float('inf')
    best_h = x0.copy()
    
    def callback(xk, **kwargs):
        nonlocal best_c5, best_h
        c5 = evaluate(xk)
        if c5 < best_c5:
            best_c5 = c5
            best_h = xk.copy()
    
    start_time = time.time()
    
    # Use L-BFGS-B which respects bounds and has good global behavior
    try:
        # We first try a simple method like L-BFGS-B for a few steps
        res = optimize.minimize(
            lambda x: evaluate(x),
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 500, 'ftol': 1e-6},
            callback=callback if budget_s >= 10 else None
        )
        if res.success:
            x_final = res.x
        else:
            x_final = best_h
        
    except Exception:
        # Fallback method
        x_final = best_h
    
    # Correct final result to ensure sum constraint
    x_final = x_final * target_sum / np.sum(x_final)
    
    # Final evaluation
    c5_final = evaluate(x_final)
    
    # Return final result
    return (x_final.tolist(), c5_final, n_points)
