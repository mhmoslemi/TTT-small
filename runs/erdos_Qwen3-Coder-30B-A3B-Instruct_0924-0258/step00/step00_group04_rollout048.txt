import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 42
    dx = 2.0 / n_points
    max_iter = 10000
    alpha0 = 0.1
    beta = 0.8
    min_alpha = 1e-6
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        assert np.all(h >= 0) and np.all(h <= 1)
        assert abs(np.sum(h) * dx - 1.0) < 1e-10
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Precompute lag offset for full correlation
    lag_offset = n_points - 1
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, lag_offset)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute correlations for all lags
        corr = correlate(h, 1 - h, mode='full')
        z = corr[lag_offset:lag_offset + n_points] * dx
        k_star = np.argmax(z)
        c5_val = z[k_star]
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_star) % n_points
            grad[i] = 1 - 2 * h[j]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
            
        # Line search
        alpha = alpha0
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        new_c5 = compute_c5(h_new, dx, lag_offset)
        while new_c5 >= c5_val and alpha > min_alpha:
            alpha *= beta
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            new_c5 = compute_c5(h_new, dx, lag_offset)
            
        h = h_new
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx, lag_offset):
    corr = correlate(h, 1 - h, mode='full')
    z = corr[lag_offset:lag_offset + len(h)] * dx
    return np.max(z)
