import numpy as np
from scipy import signal
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h"""
    # Compute cross-correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum value gives us the C5 bound
    return np.max(corr) * dx

def rescale_h(h, dx, target_sum):
    """Rescale h to have sum equal to target_sum while maintaining bounds"""
    current_sum = np.sum(h) * dx
    if current_sum > 0:
        h = h * target_sum / current_sum
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale again to make sum match the target
    current_sum = np.sum(h) * dx
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def initialize_h(n_points, use_initial=False):
    """Initialize h vector"""
    if use_initial and 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(h)), h)
    else:
        h = np.random.rand(n_points)
        h = rescale_h(h, 2.0 / n_points, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Use a reasonable size for optimization
    dx = 2.0 / n_points
    target_sum = 1.0
    max_particles = 30
    max_iter = 10000
    inertia_w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_stagnation = 100
    convergence_threshold = 1e-6
    
    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_c5 = []
    
    # Use initial construction as first particle if available
    first_h = initialize_h(n_points, use_initial=True)
    particles.append(first_h.copy())
    velocities.append(np.zeros(n_points))
    personal_best.append(first_h.copy())
    personal_best_c5.append(evaluate_c5(first_h, dx))
    
    # Add remaining particles randomly
    for _ in range(max_particles - 1):
        h = initialize_h(n_points, use_initial=False)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        personal_best.append(h.copy())
        personal_best_c5.append(evaluate_c5(h, dx))
    
    # Global best
    global_best_idx = np.argmin(personal_best_c5)
    global_best = personal_best[global_best_idx].copy()
    global_best_c5 = personal_best_c5[global_best_idx]
    
    start_time = time.time()
    iteration = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        iteration += 1
        
        for i in range(max_particles):
            h = particles[i]
            
            # Update particle
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - h) + 
                           c2 * r2 * (global_best - h))
            
            # Apply velocity (clipping to bounds)
            h = h + velocities[i]
            
            # Ensure h stays in [0,1] and rescale to maintain sum
            h = np.clip(h, 0, 1)
            h = rescale_h(h, dx, target_sum)
            
            # Store the updated particle
            particles[i] = h
            
            # Evaluate C5
            c5 = evaluate_c5(h, dx)
            
            # Update personal best
            if c5 < personal_best_c5[i]:
                personal_best[i] = h.copy()
                personal_best_c5[i] = c5
                
                # Check if this is a new global best
                if c5 < global_best_c5:
                    global_best = h.copy()
                    global_best_c5 = c5
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
                
        if stagnation_count > max_stagnation:
            # Reset some particles to exploration if no progress
            for i in range(max_particles):
                if np.random.rand() < 0.3:
                    particles[i] = initialize_h(n_points, use_initial=False)
                    velocities[i] = np.zeros(n_points)
            stagnation_count = 0
        
        # Occasionally rescale all particles to ensure feasibility
        if iteration % 100 == 0:
            for i in range(max_particles):
                h = particles[i]
                h = rescale_h(h, dx, target_sum)
                particles[i] = h
            
            # Update personal bests after rescaling
            for i in range(max_particles):
                c5 = evaluate_c5(personal_best[i], dx)
                if c5 < personal_best_c5[i]:
                    personal_best_c5[i] = c5
                    if personal_best_c5[i] < global_best_c5:
                        global_best_c5 = personal_best_c5[i]
    
    # Return best solution found
    return (global_best, global_best_c5, n_points)
