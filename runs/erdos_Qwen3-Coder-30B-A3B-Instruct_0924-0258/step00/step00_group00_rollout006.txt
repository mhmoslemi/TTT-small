import numpy as np
from scipy.signal import correlate
import time

def evaluate_C5(h, dx):
    """Evaluate C5 = max_k int h(x)(1 - h(x+k)) dx"""
    corr = correlate(h, 1 - h, mode="full")
    # The correlation is computed over all lags, but we want the overlap integral.
    # The maximum is taken over all valid lags.
    # For discrete signals of length n, the full correlation has 2n-1 elements.
    # The valid lags from -n+1 to n-1.
    # But we don't care about exact lags, just the maximum overlap integral.
    # We take max of all correlations, scaled by dx
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with available construction or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
        # Ensure sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    # Normalize to ensure sum == n_points / 2 and h[i] in [0,1]
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Frank-Wolfe iterations
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlations
        corr = correlate(h, 1 - h, mode="full")
        max_corr_idx = np.argmax(corr)
        max_lag = max_corr_idx - (n_points - 1)  # Convert from full index to lag

        # Gradient of the maximum correlation
        grad = np.zeros_like(h)
        # This is a simplified gradient estimation
        # We want to know how to move h to reduce correlation(max_lag)
        # Linear approximation at current h
        for k in range(n_points):
            # Find the relevant term (h[i] * (1 - h[i+lag])) contribution
            if max_lag >= 0:
                if k + max_lag < n_points:
                    grad[k] += 1 - h[k + max_lag]
            else:
                if k - max_lag >= 0:
                    grad[k] += 1 - h[k - max_lag]

        # Sort by gradient magnitude to find most beneficial directions
        # Pick top-half of indices to update
        s = np.argsort(-np.abs(grad))[:n_points//2]
        s = np.array(s)

        # Line search to determine step size
        best_gamma = 0.0
        best_cand = np.copy(h)
        
        candidates = [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
        for gamma in candidates:
            # Try to find a good gamma (0 to 1) that improves solution
            # We update with a direction based on gradient
            h_candidate = h.copy()
            # Move towards a feasible direction
            direction = np.zeros_like(h)
            for i in s:
                direction[i] = 1 if grad[i] > 0 else -1
            
            # Scale direction to have a small step size
            step_direction = direction * gamma * 0.01
            h_candidate = h + step_direction
            
            # Enforce bounds
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Ensure mass constraint sum(h_candidate) == n_points / 2
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            c5_candidate = evaluate_C5(h_candidate, dx)
            if c5_candidate < best_cand:
                best_cand = c5_candidate
                best_gamma = gamma

        # Accept the best update for the next iteration
        h = best_cand
        new_C5 = evaluate_C5(h, dx)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h.copy()
    
    # Final evaluation
    final_C5 = evaluate_C5(best_h, dx)
    
    return (best_h.tolist(), final_C5, n_points)
