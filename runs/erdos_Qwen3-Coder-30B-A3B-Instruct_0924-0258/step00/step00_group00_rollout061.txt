import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Initialize with a simple construction
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Ensure constraints
    h = h / np.sum(h) * (n_points / 2.0)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points)
    
    # Optimization parameters
    perturb_steps = 20
    step_size_candidates = [0.1, 0.5, 1.0]
    max_iter = int(budget_s * 10)  # Rough estimate based on timing
    start_time = time.time()
    last_improvement = 0
    
    # Gradient-based optimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Get correlations for all lags
        corr = correlate(h, 1 - h, mode='full')
        k_indices = np.arange(-n_points + 1, n_points)
        max_k = k_indices[np.argmax(corr)]
        max_corr = np.max(corr)
        
        # Compute gradient in the direction of max_corr
        grad = np.zeros(n_points)
        if max_k >= 0:
            # h[i] contributes to correlation at lag max_k if i + max_k < n_points
            grad[:n_points - max_k] = -(1 - h[max_k:])
        else:
            # h[i + max_k] contributes to correlation at lag max_k if i + max_k >= 0
            grad[-max_k:] = -(1 - h[:n_points + max_k])
            
        # Try different step sizes
        best_step_h = h.copy()
        best_step_c5 = best_c5
        
        for alpha in step_size_candidates:
            candidate_h = h - alpha * grad
            candidate_h = np.clip(candidate_h, 0, 1)
            candidate_h = candidate_h / np.sum(candidate_h) * (n_points / 2.0)
            candidate_c5 = evaluate_c5(candidate_h, n_points)
            
            if candidate_c5 < best_step_c5:
                best_step_h = candidate_h
                best_step_c5 = candidate_c5
                
        # If no improvement found, add perturbation
        if best_step_c5 >= best_c5:
            last_improvement += 1
            if last_improvement > perturb_steps:
                noise = np.random.normal(0, 0.01, n_points)
                h = np.clip(h + noise, 0, 1)
                h = h / np.sum(h) * (n_points / 2.0)
                last_improvement = 0
        else:
            h = best_step_h
            best_c5 = best_step_c5
            last_improvement = 0
            
        # Update best solution
        if best_c5 < evaluate_c5(best_h, n_points):
            best_h = h.copy()
    
    # Final validation
    final_c5 = evaluate_c5(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
