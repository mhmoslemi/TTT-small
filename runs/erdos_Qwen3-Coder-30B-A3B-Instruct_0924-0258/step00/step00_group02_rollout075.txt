import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a reasonable starting point
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)
    
    # dx for normalization
    dx = 2.0 / n_points
    
    # Evaluate initial solution
    def evaluate_h(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(correlation) * dx
        return max_overlap
    
    best_h = h.copy()
    best_c5 = evaluate_h(best_h)
    
    # Gradient-based optimization with projection
    learning_rate = 0.01
    max_iter = int(budget_s / 1.0) if budget_s > 1 else 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlations
        correlation = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(correlation)
        max_overlap = np.max(correlation) * dx
        
        # Compute gradient w.r.t. h for the maximal lag
        # Since correlation[k] = sum(h[i] * (1 - h[i+k]))
        gradient = np.zeros_like(h)
        for i in range(n_points):
            for j in range(n_points):
                k = j - i
                if 0 <= i < n_points and 0 <= j < n_points:
                    # Add to gradient of h[i] the impact of h[j] when h[i] is shifted by k
                    if 0 <= k + n_points - 1 < 2 * n_points - 1:
                        # This is simplified: we compute gradient as partial derivative for max overlap term
                        if k == max_idx - (n_points - 1):
                            gradient[i] -= 1 - h[j]
                        elif k == -max_idx + (n_points - 1):
                            gradient[i] -= h[j]
        
        # Apply gradient step and clip to [0,1]
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        
        # Rescale to maintain sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new solution
        current_c5 = evaluate_h(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Perturb occasionally to escape local minima
        if iteration % 50 == 0:
            perturbation = np.random.uniform(-0.05, 0.05, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (list(best_h), best_c5, n_points)
