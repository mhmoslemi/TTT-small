import numpy as np
import time
from scipy.fft import fft, ifft

def evaluate_c5(h, dx):
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # Only consider the valid overlaps (not padding)
    middle = len(corr) // 2
    overlaps = corr[middle:]
    return np.max(overlaps) * dx

def project_to_feasible(h, n_points):
    # Ensure sum(h) = n_points / 2 and all values are in [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-10:
        # Scale to reach target sum
        h = h * (target_sum / current_sum)
        # Clip again to maintain constraints
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Determine problem parameters
    n_points = 47 if 'initial_h_values' in globals() else 64
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert len(h) == n_points
    else:
        h = np.full(n_points, 0.5)
    h = project_to_feasible(h, n_points)

    # Precompute useful arrays
    # For efficiency in updating, we precompute shifts
    max_iter = 2000
    alpha = 0.1
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    # Track number of successful updates
    no_improvement_count = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps using fast Walsh-Hadamard-like correlation
        # We're using FFT for convolution of h and (1-h)
        # Cross-correlation with full wraparound
        # But we can do it efficiently with flips and FFT
        
        # To get cross-correlation: convolve h with reverse(1-h)
        h1 = np.array(h)
        h2 = np.array(1 - h)
        # For full cross-correlation, pad with zeros if needed
        pad_len = 2 * n_points - 1
        h1_pad = np.pad(h1, (0, pad_len - len(h1)), 'constant')
        h2_pad = np.pad(h2, (0, pad_len - len(h2)), 'constant')
        
        # fft-based cross correlation
        H1 = fft(h1_pad)
        H2 = fft(h2_pad)
        # convolution (but h2 is reversed for cross correlation)
        corr = ifft(H1 * np.conj(H2))
        corr = np.real(corr)
        
        # only consider middle part with actual cross-correlations
        start_idx = n_points - 1
        corr_vals = corr[start_idx:start_idx + n_points]
        # compute maximum
        max_k = np.argmax(corr_vals)
        c5 = np.max(corr_vals) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
            last_improvement = iteration
        else:
            no_improvement_count += 1

        # Update step
        if max_k < n_points:
            # Create gradient update direction
            # We want to decrease the overlap at lag max_k
            # This is roughly: update h[i] by -alpha * (grad[i])
            # The gradient is related to how much h[i] contributes to overlap at lag max_k
            # Which involves h[i] and h[(i + max_k) % n_points] (and corresponding terms)
            # So we can compute in the frequency domain or use finite difference approach
            
            # Approach: For cross-correlation computation, 
            # using overlap h * (1 - h[max_k]) and h * (1 - h[0])
            # We're going to construct a direct update
            # We should decrease the contribution from overlap at max_k
            
            # A reasonable heuristic:
            # Look at current term: h[i + max_k] * (1 - h[i])
            # To decrease, we increment h[i] if h[i] is small and h[i+max_k] is large
            # Or decrement h[i] if h[i] is large and h[i+max_k] is small
            
            # Direct gradient-like notion for update:
            
            # Get the positions involved in this cross-correlation
            # at k = max_k:
            # The correlation sum involves terms: h[i] * (1 - h[i + k])
            # So for improving this change h[i] to reduce the value
            
            # Let's do a simpler gradient descent-y approach:
            # Use direct correlation calculation for gradient
            
            # We want to minimize h[i] * (1 - h[(i + max_k) % n_points])
            # so if h[i] = 1, the contribution from h[(i+max_k)%n_points] is zero.
            # But the relative influence depends on h[i] and h[i+max_k].
            # If we change h[i] by delta, we affect that term.
            # For simplicity, let's take a normal update approach:
            
            # More nuanced update:
            # Compute Finite difference gradient for C5 at Lag max_k
            
            # For now, just update to reduce this correlation directly
            # Logic: either reduce h at the i which makes high contribution,
            # or increase h for low values that contribute much to this lag.
            
            # A safe safer gradient to compute is:
            # For each i in [0, n_points), compute contribution to correlation at k = max_k:
            # term = h[i] * (1 - h[(i+k) % n_points])
            # If we are at maximum, it means we want to reduce this term:
            # If h[i] is high and (1-h[(i+k)%n_points]) is high, this contributes
            # Risk to fire too much for a safety update, therefore let's be gentle
            
            grad = np.zeros_like(h)
            for i in range(n_points):
                idx = (i + max_k) % n_points
                grad[i] = (1 - h[idx]) - h[i]
            
            # Grad can have any sign, but let's clip and follow direction
            clip_grad = np.clip(grad, -0.5, 0.5)
            h += alpha * clip_grad
            
            # Ensure sum is maintained
            h = project_to_feasible(h, n_points)
            
            # Scaling back to sum = n_points / 2 may alter the balance
            # Only scale if sum is off by more than tiny_amount
            current_sum = np.sum(h)
            target_sum = n_points / 2.0
            if abs(current_sum - target_sum) > 1e-5:
                h *= (target_sum / current_sum)
                h = project_to_feasible(h, n_points)
        
        # Adaptive learning rate
        if iteration - last_improvement > 20:
            alpha *= 0.95
            last_improvement = iteration
            
        # Small random perturbation
        if np.random.random() < 0.05:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = project_to_feasible(h, n_points)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
