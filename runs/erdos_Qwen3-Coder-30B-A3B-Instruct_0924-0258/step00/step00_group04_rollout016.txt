import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 42  # Use the default from initial_h_values
    dx = 2.0 / n_points
    max_iter = 1200
    
    # Get initial values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert len(h) == n_points
    else:
        h = np.full(n_points, 0.5)
    
    # Adjust to ensure sum(h) = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap integrals for all lags
        # Using the full convolution (cross-correlation with flipped kernel)
        h_flipped = np.flip(1 - h)
        correlations = correlate(h, h_flipped, mode='full')
        # The correlation results are longer than needed, extract relevant part
        # But actually we need the overlap integrals in the right way.
        # Skip computing overlaps directly as it will be expensive
        # Instead, use gradient update as a Newton step for approximation:
        
        # Compute the maximum overlap and corresponding k
        max_z = -1.0
        max_k = 0
        correlations = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            # compute overlap integral for lag k
            if k < n_points:
                shift = k
                z = np.sum(h * np.roll(1 - h, shift))
            else:
                shift = k - (2 * n_points - 2)
                z = np.sum(h * np.roll(1 - h, -shift))
            correlations[k] = z
            if z > max_z:
                max_z = z
                max_k = k
        
        # Compute the gradient w.r.t. h for this maximum overlap
        g = np.zeros(n_points)
        if max_k < n_points:
            # h[i] * (1 - h[i + k]) term 
            for i in range(n_points):
                j1 = (i - max_k) % n_points
                j2 = (i + max_k) % n_points
                g[i] = 1 - h[j1] - h[j2]
        else:
            shift = max_k - (2 * n_points - 2)
            for i in range(n_points):
                j1 = (i + shift) % n_points
                j2 = (i - shift) % n_points
                g[i] = 1 - h[j1] - h[j2]
        g *= 2.0 / n_points
        g -= np.mean(g)  # Mass conservation correction
    
        # Line search
        step_size = 0.1
        h_new = h - step_size * g
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new, dx)
        
        # Accept if better
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        # Update h
        h = h_new
    
    # Perturb sometimes to escape local minima
    if iteration % 50 == 0 and iteration > 0:
        noise = np.random.normal(0, 0.01, n_points)
        h = h + noise
        h = np.clip(h, 0, 1) 
        h = h * (n_points / 2) / np.sum(h)
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)


def compute_c5(h, dx):
    # Compute the overlap integral for all parameters
    n_points = len(h)
    max_val = 0.0
    
    # Compute for all lags from 0 to n_points - 1
    for k in range(n_points):
        # Correlation based on sliding window
        shift = k
        inner = 0.0
        for i in range(n_points):
            j = (i + shift) % n_points
            inner += h[i] * (1 - h[j])
        val = inner * dx
        if val > max_val:
            max_val = val
    return max_val
