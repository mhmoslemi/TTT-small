import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    # Evaluate max of correlation with (1-h)
    full_corr = correlate(h, 1 - h, mode='full')
    # Extract the relevant part for lags from 0 to 2*n-2
    # We want the lag 0 to n-1 contribution
    C5 = np.max(full_corr[len(h)-1:])
    return C5 * dx

def project_to_feasible(h, n_points):
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    # Scale to sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, max_lags, n_points):
    # Compute gradient for max_lags
    gradient = np.zeros(n_points)
    for k in max_lags:
        # For each term h[i] * (1 - h[i+k]) in the correlation
        # Partial derivative for h[i]: (1 - h[i+k]) 
        # Partial derivative for h[i+k]: -h[i]
        for i in range(n_points - k):
            if i < n_points:
                gradient[i] += (1 - h[i + k])
            if i + k < n_points:
                gradient[i + k] -= h[i]
    return gradient

def get_best_alpha(h, gradient, n_points, dx):
    # Try different alpha values
    alphas = [0.1, 0.25, 0.5]
    best_alpha = 0
    best_c5 = np.inf
    for alpha in alphas:
        h_candidate = h - alpha * gradient
        h_candidate = project_to_feasible(h_candidate, n_points)
        c5 = evaluate_c5(h_candidate, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_alpha = alpha
    return best_alpha, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = np.inf
    
    n_points = 100  # Start with a reasonable small size
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
    else:
        # Create initial construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = project_to_feasible(h, n_points)
    
    no_improvement_count = 0
    max_no_improvement = 20
    
    while time.time() - start_time < budget_s - 0.1:
        # Evaluate current
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count > max_no_improvement:
            # Add some noise to escape local minima
            h += 0.01 * np.random.randn(n_points)
            h = project_to_feasible(h, n_points)
            no_improvement_count = 0
        
        # Compute all correlations
        full_corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(full_corr[len(h)-1:])
        max_lags = [i for i in range(len(full_corr)) if full_corr[i] == max_corr]
        
        # Compute gradient for max lags
        gradient = compute_gradient(h, max_lags, n_points)
        
        # Line search
        alpha, new_c5 = get_best_alpha(h, gradient, n_points, dx)
        
        # Apply update
        h = h - alpha * gradient
        h = project_to_feasible(h, n_points)
        
        # Early stopping if we're close
        if best_c5 < 0.38075:
            break
    
    if best_h is None:
        best_h = h
    return (list(best_h), best_c5, n_points)
