import time
from scipy.signal import correlate

    np.random.seed(seed)
    start_time = time.time()

    h *= (n_points / 2) / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Optimization parameters
    alpha = 0.1
    momentum = 0.9
    old_grad = np.zeros_like(h)
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s:
        iter_count += 1
        
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        C5 = np.max(corr)
        
        # Track best
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            # Random restart with 5% probability
            if np.random.rand() < 0.05:
                h = np.random.rand(n_points) * 0.5
                h *= (n_points / 2) / np.sum(h)
                h = np.clip(h, 0, 1)
            else:
                # Early stopping
                break
                
        # Find worst shift (maximum correlation)
        max_idx = np.argmax(corr)
        k0 = max_idx - (n_points - 1)  # convert to shift index
        
        # Compute gradient w.r.t. the active shift
        grad = np.zeros_like(h)
        if k0 >= 0:
            # grad[i] += (1 - h[i + k0]) for i + k0 < n_points
            for i in range(n_points - k0):
                grad[i] += (1 - h[i + k0])
            # grad[i + k0] += -h[i] for i < n_points - k0
            for i in range(n_points - k0):
                grad[i + k0] += -h[i]
        else:
            # k0 < 0, so we're looking at negative shifts
            k0_abs = -k0
            for i in range(k0_abs, n_points):
                grad[i] += (1 - h[i - k0_abs])
            for i in range(n_points - k0_abs):
                grad[i] += -h[i + k0_abs]

        # Apply momentum and update step
        old_grad = momentum * old_grad + (1 - momentum) * grad
        h_new = h - alpha * old_grad
        
        # Projection: clip to [0,1] and adjust sum
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        target_sum = n_points / 2
        
        if abs(h_new_sum - target_sum) > 1e-6:
            # redistribute to maintain sum
            diff = target_sum - h_new_sum
            # Simple proportional adjustment
            scale = diff / np.sum(h_new)
            h_new = np.clip(h_new + scale * h_new, 0, 1)
            # Then normalize again
            h_new *= target_sum / np.sum(h_new)
        
        h = h_new
        
        # Reduce learning rate occasionally
        if iter_count % 10 == 0 and no_improvement_count > 0:
            alpha *= 0.95
            
    # Final check of validity and bounding
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_C5 = np.max(corr)
    
    return (best_h.tolist(), final_C5, n_points)
