import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar

def evaluate_overlap(h, dx):
    # Compute the maximum overlap integral
    corr = correlate(h, 1 - h, mode='full')
    # Only take the relevant part of the correlation
    mid = len(corr) // 2
    max_corr = np.max(corr[mid:])
    return max_corr * dx

def project_and_rescale(h, n_points):
    # Project h onto [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to ensure sum(h) = n_points / 2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, dx, eps=1e-4):
    # Compute gradient approximation using finite differences
    n = len(h)
    grad = np.zeros(n)
    base_val = evaluate_overlap(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_and_rescale(h_plus, n)
        val_plus = evaluate_overlap(h_plus, dx)
        
        grad[i] = (val_plus - base_val) / eps
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_and_rescale(h, n_points)
    else:
        h = np.full(n_points, 0.5)
        h = project_and_rescale(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Parameters
    temp = 1.0
    min_temp = 0.01
    cooling_factor = 0.99
    patience = 50
    patience_count = 0
    gradient_scale = 0.1
    noise_std = 0.05
    
    while time.time() - start_time < budget_s - 1:
        # Compute gradient
        grad = compute_gradient(h, dx)
        
        # Generate candidate
        noise = np.random.normal(0, noise_std, n_points)
        # Combine noise and gradient
        candidate = h - gradient_scale * grad + noise
        candidate = project_and_rescale(candidate, n_points)
        
        # Evaluate candidate
        candidate_c5 = evaluate_overlap(candidate, dx)
        
        # Accept or reject
        if candidate_c5 < best_c5:
            h = candidate
            best_h = candidate
            best_c5 = candidate_c5
            patience_count = 0
        else:
            # Accept with probability based on temperature
            if np.random.rand() < np.exp((best_c5 - candidate_c5) / temp):
                h = candidate
                patience_count = 0
            else:
                patience_count += 1
        
        # Update temperature
        temp = max(temp * cooling_factor, min_temp)
        
        # Warm restart if needed
        if patience_count >= patience:
            temp = 1.0
            patience_count = 0
    
    return (best_h, best_c5, n_points)
