import numpy as np
import time
from scipy import optimize

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute overlap integral for all shifts
    corr = np.correlate(h, 1-h, mode='full')
    # Only the valid overlaps (shifts from -n+1 to n-1)
    # The middle n-1 elements correspond to valid overlaps
    valid_corr = corr[n-1:2*n-1]
    # Scale by dx to get actual integral values
    c5 = np.max(valid_corr) * dx
    return c5

def project_h(h):
    # Project h to [0,1] and normalize so sum = n/2
    h = np.clip(h, 0, 1)
    n = len(h)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n / 2) / current_sum
    return h

def mutate(h, sigma):
    # Add Gaussian noise scaled by sigma
    noise = np.random.normal(0, sigma, len(h))
    h_new = h + noise
    return project_h(h_new)

def crossover(h1, h2, alpha=0.6):
    # Blend crossover
    beta = np.random.uniform(alpha, 1 - alpha)
    h_new = beta * h1 + (1 - beta) * h2
    return project_h(h_new)

def optimize_h(h0, budget_s):
    # Parameters
    n_points = len(h0)
    dx = 2.0 / n_points
    max_generations = int(budget_s * 1000)  # Rough estimate
    
    # Adaptive parameters
    mutation_base_sigma = 0.1
    decay_tau = 100.0
    
    # Set up population
    pop_size = 100
    population = []
    population.append(h0)
    for _ in range(pop_size - 1):
        h_mut = mutate(h0, mutation_base_sigma * 0.5)
        population.append(h_mut)
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
            
        # Compute fitness for all population members
        fitnesses = [evaluate_erdos_solution(h) for h in population]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitnesses[min_fitness_idx]
        
        # Selection: tournament selection
        tournament_size = 3
        selected_indices = []
        for _ in range(pop_size):
            tournament = np.random.choice(len(population), tournament_size)
            winner_idx = tournament[np.argmin([fitnesses[i] for i in tournament])]
            selected_indices.append(winner_idx)
        
        # Create new population
        new_population = []
        
        # Add elite individuals
        top_indices = np.argsort(fitnesses)[:pop_size//4]
        for idx in top_indices:
            new_population.append(population[idx].copy())
        
        # Crossover
        while len(new_population) < 3 * pop_size // 4:
            i1, i2 = np.random.choice(selected_indices, 2, replace=False)
            child = crossover(population[i1], population[i2])
            new_population.append(child)
        
        # Mutation
        for h in new_population[len(top_indices):]:
            sigma = mutation_base_sigma * np.exp(-gen / decay_tau)
            h_mut = mutate(h, sigma)
            new_population.append(h_mut)
            
        # Ensure size
        population = new_population[:pop_size]
    
    # Refine with local optimization
    def objective(h_flat):
        h = h_flat.reshape(-1)
        return evaluate_erdos_solution(h)
    
    def constraint(h_flat):
        h = h_flat.reshape(-1)
        return np.sum(h) - n_points / 2
    
    # Try to refine the best solution using scipy
    h_init = best_h.copy()
    x0 = h_init.flatten()
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        # Use SLSQP with constraints
        result = optimize.minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 500}
        )
        if result.success:
            refined_h = result.x.reshape(-1)
            refined_h = project_h(refined_h)
            new_c5 = evaluate_erdos_solution(refined_h)
            if new_c5 < best_c5:
                best_h = refined_h
                best_c5 = new_c5
    except:
        pass  # If refine fails, keep existing best
    
    return best_h, best_c5

        n_points = 70
        h0 = np.zeros(n_points)
        # Divided into two regions: high and low values
        mid = n_points // 2
        h0[:mid] = 0.5
        h0[mid:] = 0.0
        h0 = project_h(h0)
    
    h_opt, c5_opt = optimize_h(h0, budget_s)
    
    return (h_opt.tolist(), float(c5_opt), len(h_opt))
