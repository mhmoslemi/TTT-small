import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 47
        h = 0.5 * np.ones(n_points)
    
    # Ensure sum(h) = n_points / 2 exactly
    h *= n_points / (2 * np.sum(h))
    
    # Check constraints
    assert np.isclose(np.sum(h), n_points / 2), "Sum constraint not satisfied"
    assert np.all(h >= 0) and np.all(h <= 1), "Bounds constraint not satisfied"
    
    # Optimization parameters
    max_iter = 10000
    alpha = 0.1
    alpha_decay = 0.95
    early_stop = 50
    momentum = 0.9
    restart_prob = 0.05
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = compute_C5(h)
    
    start_time = time.time()
    
    # For momentum
    h_prev = np.zeros_like(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlaps for all shifts
        correlation = np.correlate(h, 1 - h, mode='full')
        k0 = np.argmax(correlation)
        
        # Compute gradient w.r.t. h for the active shift
        grad = np.zeros_like(h)
        if k0 < len(h):
            # Gradient contribution from h[i] * (1 - h[i + k0])
            # d/dh[i] h[i] * (1 - h[i + k0]) = 1 - h[i + k0]
            for i in range(len(h)):
                if i + k0 < len(h):
                    grad[i] += 1 - h[i + k0]
        else:
            # Handle other case - values before the end
            for i in range(len(h)):
                if i >= k0 - len(h) + 1:
                    grad[i] -= h[i - k0 + len(h) - 1]
        
        # Apply momentum and gradient step
        h_new = h + momentum * (h - h_prev) - alpha * grad
        
        # Clip to bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Project to maintain sum condition
        h_new *= (n_points / 2) / np.sum(h_new)
        
        # Check if we should restart
        if np.random.random() < restart_prob and iteration > 0 and iteration % 50 == 0:
            h_new = 0.5 * np.ones_like(h_new)
            h_new *= (n_points / 2) / np.sum(h_new)
        
        # Update h and h_prev
        h = h_new
        h_prev = h_new.copy()
        
        # Compute current C5
        c5 = compute_C5(h)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Decrease step size if no improvement
        if iteration % 10 == 0 and iteration > 0:
            if iteration % 100 == 0:
                alpha = max(1e-6, alpha * alpha_decay)
    
    return (best_h.tolist(), best_c5, n_points)

def compute_C5(h):
    # Compute correlation assuming h is periodic with period 2
    correlation = np.correlate(h, 1 - h, mode='full')
    # Return the maximum overlap (should be on the full array)
    max_correlation = np.max(correlation)
    dx = 2.0 / len(h)
    return max_correlation * dx
