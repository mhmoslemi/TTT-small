import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

        h = h * (n_points / 2.0) / np.sum(h)
    
    dx = 2.0 / n_points
    start_time = time.time()
    
    def compute_C5(h):
        # Compute correlation using numpy
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def objective(h):
        return compute_C5(h)
    
    # Initial value
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Simple gradient ascent approach
    max_iter = 10000
    step_size = 0.001
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute finite differences for gradient estimation
        grad = np.zeros(n_points)
        eps = 1e-5
        current_C5 = compute_C5(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            # Ensure within bounds
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            # Re-normalize
            h_plus = h_plus * (n_points / 2.0) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2.0) / np.sum(h_minus)
            
            grad[j] = (compute_C5(h_plus) - compute_C5(h_minus)) / (2 * eps)
        
        # Update direction (negative gradient for minimization)
        update = -grad
        
        # Project onto feasible set with constraint sum(h) = n_points/2
        # Maintain the sum constraint
        mean_update = np.mean(update)
        update -= mean_update
        
        # Line search
        alpha = step_size
        while alpha > 1e-8:
            h_new = h + alpha * update
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to maintain sum constraint
            h_new = h_new * (n_points / 2.0) / np.sum(h_new)
            
            new_C5 = compute_C5(h_new)
            if new_C5 < current_C5:
                h = h_new
                current_C5 = new_C5
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = h.copy()
                break
            alpha *= 0.5
    
    # Final check
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
