import numpy as np
from scipy import correlate
import time
import random

def evaluate_h(h, dx):
    # Compute all correlations for lags -n+1 to n-1
    corr = correlate(h, 1-h, mode="full")
    # The overlap integral for each lag
    overlaps = corr * dx
    # Maximum overlap
    return np.max(overlaps)

def project_and_normalize(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize sum to n_points/2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, k_max, n_points):
    # Compute gradient of correlation at lag k_max
    grad = np.zeros(n_points)
    if k_max >= 0:
        # For k >= 0, grad[i] = 1 - h[i + k]
        for i in range(n_points - k_max):
            grad[i] += (1 - h[i + k_max])
    else:
        # For k < 0, grad[i + |k|] = 1 - h[i]
        for i in range(n_points + k_max):
            grad[i - k_max] += (1 - h[i])
    return grad

def compute_correlations(h, n_points):
    # Returns all correlation values
    corr = correlate(h, 1-h, mode="full")
    return corr

def step_size_search(h, grad, n_points, dx, alpha_candidates=[0.1, 0.5, 1.0]):
    current_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    best_alpha = 0.0
    
    for alpha in alpha_candidates:
        h_new = h - alpha * grad
        h_new = project_and_normalize(h_new, n_points)
        new_c5 = evaluate_h(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            best_alpha = alpha
    return best_h, best_c5, best_alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available, otherwise start from scratch
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    h = project_and_normalize(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    perturb_steps = 20
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 0.1:
        # Compute all correlations
        corr = compute_correlations(h, n_points)
        # Find maximum correlation and corresponding lag
        max_idx = np.argmax(corr)
        k_max = max_idx - (n_points - 1)
        
        # Compute gradient for this lag
        grad = compute_gradient(h, k_max, n_points)
        
        # Line search
        h_new, new_c5, best_alpha = step_size_search(h, grad, n_points, dx)
        
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Perturbation if no improvement for a while
        if no_improvement_count >= perturb_steps:
            # Add noise
            noise = 0.02 * np.random.randn(n_points)
            h = h + noise
            h = project_and_normalize(h, n_points)
            no_improvement_count = 0

        # Early exit condition
        if new_c5 < 0.3807:
            break
    
    return (best_h, best_c5, n_points)
