import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialization
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with uniform values and normalize sum
        h = np.full(n_points, 0.5)
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Project to feasible set if needed
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Objective function for one step
    def compute_C5(h):
        corr = signal.correlate(h, 1 - h, mode='full')
        # Take the maximum over all lags
        max_val = np.max(corr)
        return max_val * dx
    
    # Gradient computation based on lag contributing to max
    def compute_gradient(h, max_lags):
        n = len(h)
        grad = np.zeros(n)
        for k in max_lags:
            # For lags from -n+1 to n-1
            if k >= 0:
                # Compute d/dh[i] for contributions to overlap
                for i in range(n):
                    if i + k < n:
                        grad[i] += -dx  # d/dh[i] of (1-h[i+k]) term in overlap integral
                        grad[i+k] += -dx  # d/dh[i+k] of (1-h[i]) term in overlap integral
            else:
                for i in range(n):
                    if i - k < n:
                        grad[i] += -dx  # d/dh[i] of (1-h[i+k]) term
                        grad[i-k] += -dx  # d/dh[i-k] of (1-h[i]) term
        return grad
    
    # Optimization parameters
    max_iter = 500
    alpha = 0.1
    tol = 1e-4
    start_time = time.time()
    
    current_C5 = compute_C5(h)
    prev_C5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations
        corr = signal.correlate(h, 1 - h, mode='full')
        
        # Find maximum value and corresponding lags
        max_val = np.max(corr)
        max_lags = np.where(np.abs(corr - max_val) < 1e-10)[0]
        
        # Convert indices to actual lag values
        lags = np.arange(-n_points + 1, n_points)
        actual_lags = lags[max_lags - (n_points - 1)]
        
        # Compute approximate gradient (average of gradients over all max-lags)
        grad = np.zeros(n_points)
        for k in actual_lags:
            k_idx = k + n_points - 1  # index in correlation
            # Reflect symmetry for efficient gradient
            if k >= 0:
                for i in range(n_points - k):
                    grad[i] -= dx
                    grad[i + k] -= dx
            else:
                for i in range(n_points + k):
                    grad[i] -= dx
                    grad[i - k] -= dx
        
        # Normalize gradient
        grad_sum = np.linalg.norm(grad)
        if grad_sum > 0:
            grad = grad / grad_sum
            
        h_new = h - alpha * grad
        h = project(h_new)
        
        # Check convergence
        prev_C5 = current_C5
        current_C5 = compute_C5(h)
        
        if abs(prev_C5 - current_C5) < tol:
            break
    
    # Final optimization to make sure we minimized
    while time.time() - start_time < budget_s - 1:
        corr = signal.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        max_lags = np.where(np.abs(corr - max_val) < 1e-10)[0]
        lags = np.arange(-n_points + 1, n_points)
        actual_lags = lags[max_lags - (n_points - 1)]
        
        # Gradient over max lags
        grad = np.zeros(n_points)
        for k in actual_lags:
            if k >= 0:
                for i in range(n_points - k):
                    grad[i] -= dx
                    grad[i + k] -= dx
            else:
                for i in range(n_points + k):
                    grad[i] -= dx
                    grad[i - k] -= dx
                    
        if np.linalg.norm(grad) > 0:
            h -= alpha * grad / np.linalg.norm(grad)
        h = project(h)
        
        current_C5 = compute_C5(h)
        # Early termination if improvement negligible
        if prev_C5 - current_C5 < tol:
            break
        prev_C5 = current_C5
        
    # Final C5 computation
    final_C5 = compute_C5(h)
    return (h.tolist(), final_C5, n_points)
