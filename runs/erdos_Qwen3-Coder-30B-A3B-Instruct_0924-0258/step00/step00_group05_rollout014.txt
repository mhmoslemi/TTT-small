import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_low = 10
    n_high = 70
    n_levels = int(np.log2(n_high / n_low))
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_low / 2)
    else:
        h = np.random.rand(n_low)
        h = h / np.sum(h) * (n_low / 2)
    
    # Initial temperature and cooling schedule
    T = 0.1
    cooling_factor = 0.99
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Multi-resolution optimization
    for level in range(n_levels + 1):
        n = n_low * (2 ** level)
        if level > 0:
            # Interpolate to higher resolution
            h = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h)), h)
            # Ensure sum constraint
            h = h / np.sum(h) * (n / 2)
        
        # Optimization at this resolution
        for iteration in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Perturb h
            h_new = h.copy()
            noise = np.random.normal(0, T, n)
            h_new += noise
            # Clip to [0, 1]
            h_new = np.clip(h_new, 0, 1)
            # Adjust sum to exact constraint
            h_new = h_new / np.sum(h_new) * (n / 2)
            
            # Compute new C5
            c5_new = compute_c5(h_new)
            
            # Metropolis acceptance
            delta_c5 = c5_new - best_c5
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            
            # Update temperature
            T *= cooling_factor
            
    return (best_h.tolist(), best_c5, len(best_h))

def compute_c5(h):
    # Calculate overlap integral C5
    n = len(h)
    dx = 2.0 / n
    corr = signal.correlate(h, 1 - h, mode='full')
    # The correlation length is 2*n - 1, but we only need max in range
    # The full length gives us all overlapping regions, we extract max
    corr_max = np.max(corr)
    return corr_max * dx
