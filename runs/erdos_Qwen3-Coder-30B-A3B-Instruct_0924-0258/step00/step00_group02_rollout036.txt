import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute overlap integral for all shifts
    overlap = signal.correlate(h, 1 - h, mode="full") * dx
    # Return maximum overlap
    return np.max(overlap)

def project_to_feasible(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points / 2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, n_points, mutation_rate=0.1):
    """Mutate a candidate solution"""
    h_new = h.copy()
    n_mutate = int(len(h) * mutation_rate)
    indices = np.random.choice(len(h), size=n_mutate, replace=False)
    for i in indices:
        h_new[i] += 0.1 * np.random.randn()
    return project_to_feasible(h_new, n_points)

def crossover(parent1, parent2):
    """Single-point crossover between two parents"""
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return offspring1, offspring2

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    pop_size = 50
    generations = 12
    mutation_rate = 0.1
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = initial_h_values
    else:
        # Create a simple initial construction
        h0 = np.ones(n_points) * 0.5
    
    # Ensure feasibility of initial
    h0 = project_to_feasible(h0, n_points)
    population.append(h0)
    
    # Add some random perturbations
    for _ in range(pop_size - 1):
        h = h0 + 0.1 * np.random.randn(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
        
    # Track best solution
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Start timing
    start_time = time.time()
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:  # Leave 5s for cleanup
            break
            
        # Evaluate all population members
        scores = []
        for h in population:
            score = evaluate_erdos_solution(h)
            scores.append(score)
            
        # Sort by score
        sorted_indices = np.argsort(scores)
        population = [population[i] for i in sorted_indices]
        scores = [scores[i] for i in sorted_indices]
        
        # Check for new best
        if scores[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = scores[0]
            
        # Selection: keep top 25%
        parents = population[:pop_size//2]
        
        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            parent1 = parents[i]
            parent2 = parents[i+1] if i+1 < len(parents) else parents[0]
            child1, child2 = crossover(parent1, parent2)
            offspring.extend([child1, child2])
            
        # Mutate offspring
        mutated_offspring = [mutate(h, n_points, mutation_rate) for h in offspring]
        
        # New population: keep best parents + mutated offspring
        new_population = parents + mutated_offspring[:pop_size - len(parents)]
        population = new_population
        
    return (best_h, best_c5, n_points)
