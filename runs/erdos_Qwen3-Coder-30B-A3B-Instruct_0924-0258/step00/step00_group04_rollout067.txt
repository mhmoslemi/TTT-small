import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # Extract the relevant part and normalize
    c5_vals = corr[len(h)-1:] * dx
    return np.max(c5_vals)

def project_h(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to correct sum
    current_sum = np.sum(h) * dx
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, k_star, n_points, dx):
    """Compute gradient for a given lag k_star"""
    grad = np.zeros_like(h)
    for i in range(n_points):
        # Forward difference (i+k_star) mod n_points: h[(i + k_star) % n_points] 
        # Backward difference (i-k_star) mod n_points: h[(i - k_star) % n_points]
        grad[i] = 1 - h[(i - k_star) % n_points] - h[(i + k_star) % n_points]
    return grad * dx

def line_search(h, grad, n_points, dx, current_c5, max_iter=10):
    """Backtracking line search"""
    alpha = 1.0
    step = 0
    while step < max_iter:
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points, dx)
        new_c5 = evaluate_h(h_new, dx)
        if new_c5 <= current_c5 * 0.9999 or alpha < 1e-8:
            return h_new, alpha
        alpha *= 0.5
        step += 1
    return h_new, alpha

def run(seed=42, budget_s=600, **kwargs):
    # Seed for reproducibility
    np.random.seed(seed)
    
    # Parameters and initialization
    n_points = 42  # As given in task and initial_h_values
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Validate and handle the shape
        if len(h) != n_points:
            h = np.resize(h, n_points)
    else:
        # Default initialization
        h = np.full(n_points, 0.5)
        
    # Plug in initial solution if not properly constrained
    if not np.all((h >= 0) & (h <= 1)):
        h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    if abs(current_sum - n_points/2) > 1e-10:
        h = h * (n_points / 2 / current_sum)
        
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    start_time = time.time()
    
    # Main loop
    iter_count = 0
    while time.time() - start_time < budget_s:
        # Determine the current maximum integral over lag k
        corr = correlate(h, 1 - h, mode='full')
        c5_vals = corr[len(h)-1:] * dx
        k_max = np.argmax(c5_vals)
        c5_current = c5_vals[k_max]
        
        # Compute gradient
        grad = compute_gradients(h, k_max, n_points, dx)
        
        # Line search
        h_new, alpha = line_search(h, grad, n_points, dx, c5_current)
        h = h_new
        
        # Re-evaluate and track best
        c5_new = evaluate_h(h, dx)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
            
        iter_count += 1
        
        # Occasionally add some noise to escape local minima
        if iter_count % 50 == 0 and time.time() - start_time < budget_s - 10:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = project_h(h, n_points, dx)
    
    # Final validation
    final_c5 = evaluate_h(best_h, dx)
    return (best_h.tolist(), float(final_c5), int(n_points))
