import numpy as np
import time
from scipy import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 constant for given step function h."""
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Extract the relevant part and compute maximum
    mid = len(corr) // 2
    max_corr = np.max(corr[mid:])
    return max_corr * dx

def rescale_h(h, target_sum):
    """Rescale h to have target_sum while keeping values in [0,1]."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Linear rescaling to match target sum
    ratio = target_sum / current_sum
    h_scaled = h * ratio
    # Clip to [0,1]
    h_scaled = np.clip(h_scaled, 0, 1)
    # Adjust again to hit exact sum
    if np.sum(h_scaled) != target_sum:
        diff = target_sum - np.sum(h_scaled)
        # Find the smallest element to increase
        idx = np.argmin(h_scaled)
        h_scaled[idx] = min(1.0, h_scaled[idx] + diff)
    return h_scaled

def mutate_h(h, mutation_rate=0.1, delta=0.1):
    """Mutate h with probability mutation_rate, adding noise."""
    h_new = h.copy()
    n_points = len(h_new)
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            h_new[i] += np.random.normal(0, delta)
    return rescale_h(h_new, np.sum(h) if 'target_sum' not in locals() else target_sum)

def crossover_h(h1, h2):
    """Perform one-point crossover between h1 and h2."""
    n_points = len(h1)
    split = np.random.randint(1, n_points - 1)
    h_new = np.concatenate([h1[:split], h2[split:]])
    return rescale_h(h_new, np.sum(h1))

def tournament_selection(pop, fitnesses, k=3):
    """Select indices of individuals via tournament selection."""
    selected = []
    for _ in range(len(pop)):
        idxs = np.random.choice(len(pop), k, replace=False)
        best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
        selected.append(best_idx)
    return selected

def run(seed=42, budget_s=600, n_points=45):
    np.random.seed(seed)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.ones(n_points) * 0.5
    assert np.all(h0 >= 0) and np.all(h0 <= 1), "Initial h out of bounds"
    h0 = rescale_h(h0, target_sum)

    # Population initialization
    population_size = 50
    population = [h0.copy()]
    for _ in range(population_size - 1):
        h_pert = h0 + np.random.uniform(-0.2, 0.2, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = rescale_h(h_pert, target_sum)
        population.append(h_pert)

    # GA parameters
    max_generations = 200
    mutation_rate = 0.1
    crossover_rate = 0.7

    # Evolution loop
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate all individuals
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Tournament selection
        selected_indices = tournament_selection(population, fitnesses)
        selected_pop = [population[i] for i in selected_indices]
        
        # Reproduction
        new_population = []
        idx = 0
        while len(new_population) < population_size - 1:
            if np.random.random() < crossover_rate and idx + 1 < len(selected_pop): 
                p1, p2 = selected_pop[idx], selected_pop[idx + 1]
                child = crossover_h(p1, p2)
                new_population.append(child)
                idx += 2
            else:
                h = selected_pop[idx % len(selected_pop)]
                child = mutate_h(h, mutation_rate)
                new_population.append(child)
                idx += 1
                
        # Keep best individual from previous generation   
        new_population.insert(0, best_h.copy())
        population = new_population[:population_size]

    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
    else:
        final_c5 = evaluate_h(h0, dx)
        
    return (best_h.tolist(), final_c5, n_points)
