import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create a simple one
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        n_points = 45
        h_init = np.ones(n_points) * 0.5  # Simple uniform distribution
    
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # PSO parameters
    n_particles = 30
    c1, c2 = 2.0, 2.0
    inertia = 0.9
    max_vel = 0.5
    
    # Initialize swarm
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    pBest = np.full(n_particles, np.inf)
    pBest_h = np.zeros((n_particles, n_points))
    
    # Set first particle to initial solution
    particles[0] = h_init
    pBest[0] = evaluate_erdos_solution(h_init)[1]
    pBest_h[0] = h_init.copy()
    
    # Initialize other particles with small random perturbations
    for i in range(1, n_particles):
        pert = np.random.normal(0, 0.1, n_points)
        particles[i] = np.clip(h_init + pert, 0, 1)
        # Rescale to meet constraint
        particles[i] *= target_sum / np.sum(particles[i])
        cost, _ = evaluate_erdos_solution(particles[i])
        pBest[i] = cost
        pBest_h[i] = particles[i].copy()
    
    # Global best
    gBest_idx = np.argmin(pBest)
    gBest_cost = pBest[gBest_idx]
    gBest_h = pBest_h[gBest_idx].copy()
    
    # PSO iterations
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Update all particles
        for i in range(n_particles):
            # Compute current cost
            cost, _ = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if cost < pBest[i]:
                pBest[i] = cost
                pBest_h[i] = particles[i].copy()
            
            # Update global best
            if cost < gBest_cost:
                gBest_cost = cost
                gBest_h = particles[i].copy()
                gBest_idx = i
        
        # Update velocities and positions
        for i in range(n_particles):
            # Random components
            r1, r2 = np.random.rand(), np.random.rand()
            
            # Compute velocity update
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (pBest_h[i] - particles[i]) + 
                            c2 * r2 * (gBest_h - particles[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply feasibility constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= target_sum / np.sum(particles[i])
        
        # Reduce inertia over time
        inertia *= 0.99
        
        # Break if we're under time pressure
        if time.time() - start_time > budget_s - 0.1:
            break
    
    # Final evaluation
    final_cost, _ = evaluate_erdos_solution(gBest_h)
    
    return (gBest_h.tolist(), final_cost, n_points)

def evaluate_erdos_solution(h_values):
    """
    Function to compute the C5 bound for given h values.
    Returns (cost, C5_value)
    """
    h = np.array(h_values)
    dx = 2.0 / len(h)
    # Compute overlap integral
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr, max_corr
