import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 45
    h = initial_h_values.copy() if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
    h = h / np.sum(h) * n_points / 2
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Gradient-based optimization
    def gradient_descent_step(h, lr):
        # Compute overlap for all shifts
        h_flip = 1 - h
        cross_corr = np.correlate(h, h_flip, mode='full')
        max_shift_idx = np.argmax(cross_corr)
        max_shift = max_shift_idx - (len(h) - 1)
        # Gradient w.r.t. h for this shift
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if 0 <= i - max_shift < len(h):
                grad[i] = -h_flip[i - max_shift]
            if 0 <= i + max_shift < len(h):
                grad[i] += h_flip[i + max_shift]
        return grad
    
    lr = 0.01
    for i in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
        # Calculate gradient
        grad = gradient_descent_step(h, lr)
        # Update h
        h -= lr * grad
        # Projection to feasible set
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        # Evaluate
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr *= 0.999

    # Perturbation-based search
    for i in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        h_pert = h + 0.01 * np.random.uniform(-1, 1, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert / np.sum(h_pert) * n_points / 2
        c5 = evaluate_erdos_solution(h_pert)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pert.copy()
    
    return (best_h.tolist(), best_c5, n_points)
