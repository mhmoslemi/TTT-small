import numpy as np
from scipy.optimize import minimize
import time
from copy import deepcopy

def evaluate_c5(h):
    """Evaluate C5 for a step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations
    corr = np.correlate(h, 1-h, mode="full")
    # Extract only the valid correlations (non-zero lag)
    mid = len(corr) // 2
    valid_corr = corr[mid - (n_points - 1):mid + n_points]
    c5 = np.max(valid_corr) * dx
    return c5

def project_and_rescale(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def mutation(h, rate=0.05):
    """Mutate a solution by adding noise and projecting"""
    n_points = len(h)
    indices = np.random.choice(n_points, size=int(n_points * rate), replace=False)
    noise = np.random.normal(0, 0.1, len(indices))
    h_new = h.copy()
    h_new[indices] += noise
    return project_and_rescale(h_new, n_points)

def crossover(h1, h2, n_crossover_points=2):
    """Crossover two solutions"""
    n_points = len(h1)
    if n_crossover_points == 0:
        return h1 if np.random.rand() < 0.5 else h2
    points = sorted(np.random.choice(n_points, size=n_crossover_points, replace=False))
    h_new = h1.copy()
    start = 0
    for i, p in enumerate(points):
        if i % 2 == 0:
            h_new[start:p] = h2[start:p]
        start = p
    if n_crossover_points % 2 == 0:
        h_new[start:] = h2[start:]
    return h_new

def local_optimize(h, n_points):
    """Use gradient-based local optimization for a candidate"""
    def objective(x):
        # Reshape x into a valid h vector
        h_vec = x.copy()
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * (n_points / 2) / h_sum
        return evaluate_c5(h_vec)
    
    # Use scipy minimize to try to improve the solution
    res = minimize(objective, h, method='L-BFGS-B', bounds=[(0,1)]*n_points)
    return project_and_rescale(res.x, n_points)

    np.random.seed(seed)
    n_points = 64  # Starting with a reasonable number of samples
    dx = 2.0 / n_points
    
    # Initialize population with feasible candidates
    pop_size = 80
    population = []
    
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = project_and_rescale(h_init, n_points)
        population.append(h_init)
    else:
        # Create random initial candidates
        h_init = np.random.rand(n_points)
        h_init = project_and_rescale(h_init, n_points)
        population.append(h_init)
    
    # Fill up initial population
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_and_rescale(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    costs = [evaluate_c5(h) for h in population]
    
    # Track best solution
    best_idx = np.argmin(costs)
    best_h = population[best_idx].copy()
    best_c5 = costs[best_idx]
    
    start_time = time.time()
    gen_count = 0
    stagnation_count = 0
    mutation_rate = 0.05
    
    while time.time() - start_time < budget_s - 10:
        gen_count += 1
        
        # Selection: top 20%
        top_n = max(5, pop_size // 5)
        sorted_indices = np.argsort(costs)[:top_n]
        parents = [population[i] for i in sorted_indices]
        
        # Create new offspring
        offspring = []
        
        # Crossover and mutation
        while len(offspring) < pop_size // 2:
            # Select two parents
            p1, p2 = np.random.choice(parents, size=2, replace=False)
            # Crossover
            child = crossover(p1, p2, n_crossover_points=2 if gen_count < 50 else 3)
            child = project_and_rescale(child, n_points)
            # Mutate
            child = mutation(child, mutation_rate)
            offspring.append(child)
        
        # Local optimization for top offspring
        top_offspring = offspring[:max(1, len(offspring) // 10)]
        for i, h in enumerate(top_offspring):
            improved_h = local_optimize(h, n_points)
            offspring[i] = improved_h
            current_c5 = evaluate_c5(improved_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = improved_h.copy()
        
        # Create new population
        population = parents + offspring
        costs = [evaluate_c5(h) for h in population]
        
        # Update best solution if needed
        current_best_idx = np.argmin(costs)
        if costs[current_best_idx] < best_c5:
            best_c5 = costs[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Increase mutation rate if stagnation detected
        if stagnation_count > 50:
            mutation_rate = min(0.1, mutation_rate + 0.01)
            stagnation_count = 0
        
        # Keep population size consistent
        if len(population) > pop_size:
            worst_indices = np.argsort(costs)[-10:]
            new_pop = []
            for i in range(len(population)):
                if i not in worst_indices:
                    new_pop.append(population[i])
            population = new_pop
            costs = [evaluate_c5(h) for h in population]
    
    return (best_h.tolist(), best_c5, n_points)
