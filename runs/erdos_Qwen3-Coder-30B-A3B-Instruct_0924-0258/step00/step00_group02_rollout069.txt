import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    # Compute the overlap integral for the given step function h
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode (equivalent to convolution)
    corr = correlate(h, 1 - h, mode='full')
    # The maximum value of the correlation (multiplied by dx)
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n_points):
    # 1. Clip each element to [0, 1]
    h = np.clip(h, 0, 1)
    # 2. Rescale to satisfy sum constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    
    # Initialize swarm parameters
    n_particles = 30
    max_iter = int(budget_s / 5) if budget_s > 5 else 100
    inertia_weight = 0.9
    c1, c2 = 2.0, 2.0
    vel_max = 0.5
    
    # Initialize particles
    particles = []
    velocities = []
    pBest = []
    pBest_scores = []
    
    # First particle is the initial solution
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    score = evaluate_erdos_solution(h_init)
    pBest.append(h_init.copy())
    pBest_scores.append(score)
    
    # Generate rest of particles with slight random perturbations
    for i in range(1, n_particles):
        perturb = np.random.normal(0, 0.05, n_points)
        h = h_init + perturb
        h = project_to_feasible(h, n_points)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        score = evaluate_erdos_solution(h)
        pBest.append(h.copy())
        pBest_scores.append(score)
    
    # Track global best
    gBest_idx = np.argmin(pBest_scores)
    gBest = pBest[gBest_idx].copy()
    gBest_score = pBest_scores[gBest_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Update each particle
        for i in range(n_particles):
            h = particles[i]
            score = evaluate_erdos_solution(h)
            
            # Update personal best
            if score < pBest_scores[i]:
                pBest_scores[i] = score
                pBest[i] = h.copy()
            
            # Update global best
            if score < gBest_score:
                gBest_score = score
                gBest = h.copy()
            
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            vel = inertia_weight * velocities[i] + \
                  c1 * r1 * (pBest[i] - h) + \
                  c2 * r2 * (gBest - h)
            # Velocity clamping
            vel = np.clip(vel, -vel_max, vel_max)
            velocities[i] = vel
            
            # Update position
            new_h = h + vel
            # Feasibility projection
            new_h = project_to_feasible(new_h, n_points)
            particles[i] = new_h
            
        # Inertia weight decay
        inertia_weight *= 0.99
        
    return (gBest, gBest_score, n_points)
