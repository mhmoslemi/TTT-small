import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Fallback: uniform distribution
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Normalize to sum to n_points / 2
        h *= (n_points / 2) / np.sum(h)
    
    # dx for integration
    dx = 2.0 / n_points
    
    def compute_overlap(h_vals):
        # Compute overlap integral using correlation
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        # Max overlap
        return np.max(corr) * dx
    
    def objective(h_vals):
        return compute_overlap(h_vals)
    
    # Build bounds: 0 to 1 for each point
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = compute_overlap(h)
    
    # Optimization algorithm: sequential quadratic programming (SLSQP)
    # We run for a fixed number of steps or until time budget expires
    iter_count = 0
    max_iter = 5000 if n_points < 1000 else 1000  # adaptive iteration count
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        try:
            # Perform optimization
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                              options={'maxiter': 10, 'ftol': 1e-6})
            
            if result.success:
                h = result.x
                # Normalize to maintain sum constraint
                h = h * (n_points / 2) / np.sum(h)
                
                # Update best solution if improved
                current_c5 = compute_overlap(h)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                
                # Small perturbation to avoid local maxima
                noise = np.random.normal(0, 0.01, n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            
        except:
            pass
            
        iter_count += 1
    
    # Final check of best solution
    best_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
