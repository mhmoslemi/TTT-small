import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

def evaluate_C5(h):
    """Evaluate C5 for given h vector. h should be normalized to sum to 1."""
    n = len(h)
    dx = 2.0 / n
    # Create correlation with 1-h
    corr = np.correlate(h, 1 - h, mode='full')
    # Only take the relevant part (positive lags)
    corr = corr[n-1:]
    c5 = np.max(corr) * dx
    return c5

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points / 2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def mutate(h, rate=0.05):
    """Mutate h by adding Gaussian noise to some elements"""
    h = h.copy()
    n = len(h)
    n_mutate = max(1, int(n * rate))
    indices = np.random.choice(n, n_mutate, replace=False)
    noise = np.random.normal(0, 0.1, n_mutate)
    h[indices] += noise
    return project_to_feasible(h, len(h))

def crossover(h1, h2, n_crossover=2):
    """Crossover two individuals at n_crossover points"""
    n = len(h1)
    points = sorted(np.random.choice(n-1, n_crossover, replace=False) + 1)
    points = [0] + points + [n]
    new_h = np.zeros(n)
    for i in range(len(points)-1):
        start, end = points[i], points[i+1]
        if i % 2 == 0:
            new_h[start:end] = h1[start:end]
        else:
            new_h[start:end] = h2[start:end]
    return project_to_feasible(new_h, n)

def lbfgs_optimize(h, maxiter=100):
    """Apply local optimization to improve h"""
    n = len(h)
    dx = 2.0 / n
    def obj(h_vec):
        # Use a smoother version for optimization
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * (n / 2) / h_sum
        return evaluate_C5(h_vec)
    def grad(h_vec):
        # Custom gradient approximation as direct gradient is difficult
        # For now return zero gradient to indicate dirichlet
        return np.zeros_like(h_vec)
    res = minimize(obj, h, method='L-BFGS-B', bounds=[(0,1)]*n, options={'maxiter': maxiter})
    h_opt = np.clip(res.x, 0, 1)
    h_opt = project_to_feasible(h_opt, n)
    return h_opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Configuration
    n_points = 60  # Reasonable size for speed
    pop_size = 50
    max_gens = 200
    elite_size = pop_size // 5
    crossover_rate = 0.5
    mutate_rate = 0.05
    lbfgs_freq = 10
    patience = 50
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        # Use initial construction and generate some random ones
        h_base = np.array(initial_h_values)
        h_base = project_to_feasible(h_base, n_points)
        population.append(h_base)
        for _ in range(pop_size - 1):
            h_rand = np.random.rand(n_points)
            h_rand = project_to_feasible(h_rand, n_points)
            population.append(h_rand)
    else:
        # All random
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    stale_count = 0
    
    while generation < max_gens and time.time() - start_time < budget_s - 1:
        # Evaluate all individuals in population
        fitness = []
        for h in population:
            c5 = evaluate_C5(h)
            fitness.append(c5)
        
        # Sort by fitness
        idxs = np.argsort(fitness)
        population = [population[i] for i in idxs]
        fitness = [fitness[i] for i in idxs]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            stale_count = 0
        else:
            stale_count += 1
        
        # Selection
        elites = population[:elite_size]
        
        # Create new population
        new_pop = elites.copy()
        
        # Crossover
        for _ in range(pop_size - len(new_pop)):
            if np.random.rand() < crossover_rate:
                p1, p2 = np.random.choice(elite_size, 2, replace=False)
                child = crossover(population[p1], population[p2])
            else:
                p = np.random.choice(elite_size)
                child = population[p].copy()
            # Mutation
            if np.random.rand() < 0.5:
                child = mutate(child, mutate_rate)
            new_pop.append(child)
        
        # Local optimization for top 10% if needed
        if generation % lbfgs_freq == 0:
            for i in range(max(1, elite_size // 2)):
                idx = i
                h = new_pop[idx]
                h = lbfgs_optimize(h)
                new_pop[idx] = h
        
        population = new_pop
        generation += 1
        
        # Early stopping if no progress
        if stale_count >= patience:
            mutate_rate = 0.1  # Increase mutation rate to escape local minima
            
    # Final return best
    if best_h is None:
        # If nothing was found, use final population member
        final_fitness = [evaluate_C5(h) for h in population]
        best_idx = np.argmin(final_fitness)
        best_h = population[best_idx]
        best_c5 = final_fitness[best_idx]
    
    return (best_h, best_c5, n_points)
