import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup with n = 70 points from pre-imported initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        h = h * n_points / (2 * np.sum(h))  # normalize sum to n_points/2
    else:
        # Fallback to random construction
        n_points = 70
        h = np.random.rand(n_points)
        h = h * n_points / (2 * np.sum(h))  # normalize sum to n_points/2
    
    # Energy function
    def compute_C5(h_vals):
        dx = 2.0 / len(h_vals)
        result = correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(result) * dx

    # Normalize to sum = n/2
    def normalize_h(h_vals):
        return h_vals * (n_points / 2) / np.sum(h_vals)
    
    # Warm start at low resolution
    n_low = 10
    h_low = np.ones(n_low) * 0.5
    h = normalize_h(h_low)
    n_points = n_low
    best_h = h.copy()
    best_C5 = compute_C5(h)
    current_C5 = best_C5
    
    # Simulated Annealing parameters
    temp = 0.1
    cooling_rate = 0.99
    max_iter_per_level = 500
    
    # Levels of resolution
    resolutions = [10, 20, 30, 40, 50, 60, 70]
    for idx, n_level in enumerate(resolutions):
        if n_level > n_points:
            # Resample to higher resolution
            h_new = np.interp(np.linspace(0, 2, n_level), np.linspace(0, 2, n_points), h)
            h = h_new
            n_points = n_level
            temp = 0.1  # reset temperature at each level

        # Run simulated annealing
        for _ in range(max_iter_per_level):
            if time.time() - start_time > budget_s - 1.0:
                break
                
            h_new = h + np.random.normal(0, temp * 0.1, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new)
            
            new_C5 = compute_C5(h_new)
            delta_C5 = new_C5 - current_C5
            
            if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / temp):
                h = h_new
                current_C5 = new_C5
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = h.copy()
            
            temp *= cooling_rate
            
        if time.time() - start_time > budget_s - 1.0:
            break
    
    # Final Refinement step at highest resolution
    for _ in range(100):
        if time.time() - start_time > budget_s - 1.0:
            break
        h_new = h + np.random.normal(0, 0.001, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new)
        new_C5 = compute_C5(h_new)
        if new_C5 < current_C5:
            h = h_new
            current_C5 = new_C5
    
    return (best_h.tolist(), best_C5, len(best_h))
