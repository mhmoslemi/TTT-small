import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Determine n_points and initial h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        assert abs(np.sum(h) - n_points / 2.0) < 1e-6, "Initial h does not satisfy sum constraint"
    else:
        n_points = 47
        h = np.ones(n_points, dtype=float) * 0.5
        h *= n_points / 2.0 / np.sum(h)
    
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Precompute some useful values
    h_copy = h.copy()
    
    def compute_c5(h):
        # Use scipy correlate
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def project_h(h):
        # Project to feasible region
        h = np.clip(h, 0, 1)
        # Adjust sum to be exactly n_points / 2
        current_sum = np.sum(h)
        target_sum = n_points / 2.0
        if abs(current_sum - target_sum) > 1e-10:
            scale_factor = target_sum / current_sum
            h *= scale_factor
        return h
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Gradient-based optimization
    alpha = 0.1
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        if iteration % 50 == 0:
            alpha *= 0.95
            
        # Compute correlations using fast convolution approach
        corr_full = signal.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr_full)
        max_k = max_idx - (len(h) - 1)
        
        # Compute gradient estimate using explicit differences
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # For lag k, we have h[i] * (1 - h[(i+k) % n]) 
            # gradient w.r.t. h[i] is (1 - h[(i+k) % n]) for forward and 
            # -h[(i-k) % n] for backward
            # but with full correlator, we take the contribution to lag max_k
            # For simplicity: we compute gradient based on that max_k
            
            # We compute the derivative of correlation with respect to h[i] 
            # which is (1 - h[(i+max_k) % N]) when forward and -h[(i-max_k) % N] when backward  
            
            i_plus_k = (i + max_k) % n_points
            i_minus_k = (i - max_k) % n_points
            grad[i] = (1 - h[i_plus_k]) - h[i_minus_k]
        
        # Clip gradient to avoid large jumps
        grad = np.clip(grad, -1, 1)
        
        # Update h
        h += alpha * grad
        h = project_h(h)
        
        # Check solution quality
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            last_improvement = iteration
            # reset learning rate on improvement
            alpha = 0.1
            
        # Occasionally add small noise to escape local minima
        if np.random.random() < 0.05:
            noise = np.random.normal(0, 0.01, n_points)
            h += noise
            h = project_h(h)
            
        # Hard stop if no improvement in a while
        if iteration - last_improvement > 200 and alpha < 0.0001:
            break
            
    # Final validation
    c5_final = compute_c5(best_h)
    
    return (best_h.tolist(), float(c5_final), n_points)
