import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """ Evaluates the C5 bound for given h """
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum correlation value
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_h(h, n_points):
    """ Projects h onto feasible set: [0,1] and sum = n_points/2 """
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to fix sum to exactly n_points/2
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, max_lags, n_points):
    """ Compute gradient with respect to max_lags """
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    # For each lag in max_lags, compute additional contribution
    for k in max_lags:
        if k == 0:
            # Direct term
            grad += (1 - h) * dx
        else:
            # Off-diagonal terms
            if k > 0:
                # h[i] contributes to term with i+k
                end_idx = min(n_points, n_points - k)
                grad[:end_idx] += (-h[k:end_idx+k]) * dx
                # h[i+k] contributes to term with i
                grad[k:end_idx+k] += (-h[:end_idx]) * dx
            else:
                # k < 0, symmetrical case
                k_abs = -k
                end_idx = min(n_points, n_points + k_abs)
                grad[k_abs:end_idx] += (-h[:end_idx-k_abs]) * dx
                grad[:end_idx-k_abs] += (-h[k_abs:end_idx]) * dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
        h = project_h(h, n_points)
    
    # Initialize optimization parameters
    max_iter = 500
    alpha = 0.1
    tol = 1e-4
    
    # Iterative optimization
    current_c5 = evaluate_erdos_solution(h)
    prev_c5 = current_c5 + 1.0
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1.0:
            break
            
        # Compute correlations
        corr = correlate(h, 1-h, mode="full")
        max_val = np.max(corr)
        # find all lags with this max value
        max_lags = np.where(corr == max_val)[0]
        # Convert indices to actual lags
        lags = max_lags - (n_points - 1)
        # There may be multiple lags with same max, we include all
        
        # Compute gradient
        grad = np.zeros(n_points)
        for k in lags:
            if k == 0:
                # h[i] contributes to h[i] (since 1-h[i] contribution)
                grad += (-1.0) * np.ones(n_points)
            elif k > 0:
                # h[i] contributes to term involving h[i+k]
                end_idx = min(n_points, n_points - k) 
                grad[:end_idx] += 1.0  # the -h[i+k]
            else:
                # k < 0, symmetrical case
                k_abs = -k
                end_idx = min(n_points, n_points + k_abs)
                grad[k_abs:end_idx] += 1.0  # the -h[i-k]

        # Normalize gradient by number of contributing lags
        grad = grad / len(lags)
        # Update h
        h_new = h - alpha * grad
        # Project h_new
        h_new = project_h(h_new, n_points)
        
        # Evaluate new C5
        new_c5 = evaluate_erdos_solution(h_new)
        
        # Accept update if beneficial
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if abs(prev_c5 - current_c5) < tol:
                break
            prev_c5 = current_c5
    
    # Final check and return
    final_c5 = evaluate_erdos_solution(h)
    return (h.tolist(), final_c5, n_points)
