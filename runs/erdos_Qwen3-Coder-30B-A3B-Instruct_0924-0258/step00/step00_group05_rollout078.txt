def evaluate_erdos_solution(h, dx):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    # Compute overlaps for all shifts
    overlaps = np.correlate(h, 1 - h, mode="full")
    # Get the maximum overlap
    max_overlap = np.max(overlaps)
    # Scale by dx to get actual C5
    return max_overlap * dx

        if k == 0:
            overlap = cp.sum(cp.multiply(h, 1 - h)) * dx

            if k < n_points:
