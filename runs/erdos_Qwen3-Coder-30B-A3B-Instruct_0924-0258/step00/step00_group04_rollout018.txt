import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 42
        h = np.full(n_points, 0.5)
    
    # Ensure initial h satisfies constraint sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Validate initial h
    assert np.all(h >= 0) and np.all(h <= 1), "h must be in [0,1]"
    assert abs(np.sum(h) - n_points / 2) < 1e-10, "sum(h) must equal n_points/2"
    
    # Precompute dx
    dx = 2.0 / n_points
    
    # Objective function to minimize
    def compute_c5(h_vals):
        # Compute full convolution of h with (1-h)
        conv = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Take maximum over all lags
        max_val = np.max(conv)
        return max_val * dx
    
    # Gradient computation
    def compute_gradient(h_vals, k_star):
        n = len(h_vals)
        grad = np.zeros(n)        
        # For each index j, compute the partial derivative of z_k_star
        # with respect to h[j]
        for j in range(n):
            # Get indices for the two terms
            left_idx = (j - k_star) % n
            right_idx = (j + k_star) % n
            grad[j] = 2.0 / n * (1 - h_vals[left_idx] - h_vals[right_idx])            
        return grad * dx
    
    # Try to maintain the sum constraint exactly by rescaling
    def project_to_feasible(h_vals):
        # Clip to [0, 1]
        h_new = np.clip(h_vals, 0, 1)
        # Project to sum = n_points / 2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Helper to get the index such that the FFT order is compatible with numpy's correlate
    # but we compute convolve over full
    def compute_convolutions(h_vals):
        n = len(h_vals)
        # max over all possible lags
        convs = []
        for k in range(1, n):  # k: 1 to n-1
            # Corresponds to correlation with time-shifted version of (1-h)
            # Since we're not using numpy's correlate in full mode directly with lag 0
            # we simulate it by taking the appropriate window
            # correlate does (i, i+k) with h[i] * (1 - h[i+k], similar to what slice we use)
            corr = 0.0
            for i in range(n):
                corr += h_vals[i] * (1 - h_vals[(i + k) % n])
            corr *= dx
            convs.append(corr)
        return convs
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    # It might be expensive if trying to compute 1000+ gradients at each step
    # For better performance stick with small (say 50) random restarts or batch steps
    # But here go with a gradient descent approach optimized for speed
    
    iter_count = 0
    max_iter = int(budget_s * 2)  # ~2 iter/sec
    step_size = 0.05
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Compute all overlap integrals for all lags
        # We compute integral based on correlation as provided in the problem:
        # each term `int h(x)(1 - h(x+k)) dx` for k = 0 to n-1
        # In discrete form for lags:
        lags = range(1, n_points)  # Skip 0 since would be 0
        values = []
        for k in lags:
            corr = 0.0
            for i in range(n_points):
                corr += h[i] * (1 - h[(i + k) % n_points])
            values.append(corr * dx)
        
        # Find worst lag
        max_idx = np.argmax(values)
        k_star = lags[max_idx]
        
        # Compute full gradient for that lag
        grad = compute_gradient(h, k_star)
        
        # Compute new h
        h_new = h - step_size * grad
        
        # Project to feasible set
        h_new = project_to_feasible(h_new)
        
        current_c5 = compute_c5(h_new)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        
        h = h_new.copy()
        
        iter_count += 1
        # Occasionally perturb to escape local minima
        if iter_count % 200 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h += noise
            h = project_to_feasible(h)
    
    # Ensure best_h is feasible at the end
    final_h = project_to_feasible(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
