import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    # Compute the overlap integral using correlation
    # h is assumed to be normalized
    n = len(h)
    dx = 2.0 / n
    # Compute correlation with 1-h
    corr = correlate(h, 1-h, mode='full')
    # Compute maximum correlation value scaled by dx
    max_corr = np.max(corr) * dx
    return max_corr

def normalize_h(h):
    # Normalize h so that sum(h)*dx = 1.0
    n = len(h)
    dx = 2.0 / n
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n / 2) / h_sum
    else:
        h = np.ones_like(h) * 0.5
    return np.clip(h, 0, 1)

def mutate_h(h, mutation_rate=0.1):
    # Apply small random noise and re-normalize
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, 0.05, len(h))
        h = h + noise
        h = normalize_h(h)
    return h

def crossover_h(h1, h2, crossover_rate=0.8):
    # Blend two solutions
    if np.random.random() < crossover_rate:
        alpha = np.random.random()
        h = alpha * h1 + (1 - alpha) * h2
        h = normalize_h(h)
    else:
        h = h1.copy()
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    # Select best individual from tournament
    indices = np.random.choice(len(population), size=tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()
