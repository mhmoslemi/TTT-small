import numpy as np
from scipy import signal
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    
    # Use initial construction if available, otherwise create a random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Multi-resolution annealing
    resolutions = [10, 20, 40, 70]
    temperature = 0.1
    cooling_rate = 0.9995
    iterations_per_res = 300
    
    for res_idx, n in enumerate(resolutions):
        if time.time() - start_time > budget_s * 0.98:
            break
        
        # Interpolate to current resolution if needed
        if n > n_points:
            # Linear interpolation
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n)
            h = np.interp(x_new, x_orig, h)
            n_points = n
        elif n < n_points:
            # Downsample
            h = h[::n_points//n]
            n_points = n
            
        # Simulated Annealing at this resolution
        for _ in range(iterations_per_res):
            if time.time() - start_time > budget_s * 0.98:
                break
                
            # Perturb h
            h_new = h + np.random.normal(0, temperature, n_points)
            h_new = np.clip(h_new, 0, 1)
            
            # Project to correct sum
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            
            # Compute overlap
            corr = signal.correlate(h_new, 1 - h_new, mode='full')
            c5 = np.max(corr) * (2.0 / n_points)
            
            # Accept or reject
            delta = c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
                h = h_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            
            temperature *= cooling_rate
            
    # Final validation
    if best_h is not None:
        corr = signal.correlate(best_h, 1 - best_h, mode='full')
        c5_final = np.max(corr) * (2.0 / len(best_h))
        return (best_h.tolist(), c5_final, len(best_h))
    
    # Fallback if no better solution found
    corr = signal.correlate(h, 1 - h, mode='full')
    c5_final = np.max(corr) * (2.0 / len(h))
    return (h.tolist(), c5_final, len(h))
