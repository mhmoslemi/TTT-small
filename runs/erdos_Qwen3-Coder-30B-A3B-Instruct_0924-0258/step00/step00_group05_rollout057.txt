import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h):
    # Evaluate the overlap integral for the given step function h
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h) using full mode
    correlation = signal.correlate(h, 1 - h, mode="full")
    # Take maximum value scaled by dx
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_onto_feasible_region(h, n_points):
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    # Ensure the sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, max_lags, n_points, dx):
    # Initialize gradient vector
    grad = np.zeros_like(h)
    # For each lag that achieves the maximum
    for k in max_lags:
        # For each x position
        for i in range(n_points):
            # Value of 1-h at position i+k
            if i + k < n_points:
                grad[i] += dx
            # Value of h at position i+k
            if i + k < n_points:
                grad[i] -= dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)
        # Make sum equal to n_points/2
        h = h * (n_points / 2.0) / np.sum(h)
    
    # Ensure it satisfies the constraint exactly
    h = project_onto_feasible_region(h, n_points)
    
    # Parameters
    max_iter = 500
    alpha = 0.1
    tol = 1e-4
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    prev_c5 = best_c5 + 1.0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Compute all correlations
        correlation = signal.correlate(h, 1 - h, mode="full")
        max_val = np.max(correlation)
        max_lags = np.where(correlation == max_val)[0]
        # Adjust indices: full mode produces 2*n-1 correlations
        # We care about lags -n+1 to n-1
        actual_lags = max_lags - (n_points - 1)
        
        # Update using projected gradient descent
        grad = compute_gradient(h, actual_lags, n_points, dx)
        h_new = h - alpha * grad
        
        # Project to feasible set
        h_new = project_onto_feasible_region(h_new, n_points)
        h = h_new
        
        # Evaluate C5
        c5 = evaluate_erdos_solution(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Check for convergence
        if abs(c5 - prev_c5) < tol:
            break
        prev_c5 = c5
        
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
