import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute overlap integral for all shifts
    overlap = correlate(h, 1 - h, mode='full') * dx
    # Return maximum overlap
    return np.max(overlap)

def project_to_feasible(h, n_points):
    """Ensure h is feasible: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def mutate_h(h, n_points, mutation_rate=0.1):
    """Mutate h by adding random Gaussian noise and reprojecting"""
    h_new = h.copy()
    n_mutate = int(mutation_rate * len(h_new))
    indices = np.random.choice(len(h_new), n_mutate, replace=False)
    for i in indices:
        h_new[i] += 0.1 * np.random.randn()
    return project_to_feasible(h_new, n_points)

def crossover_h(h1, h2):
    """Single-point crossover of two step functions"""
    n = len(h1)
    if n != len(h2):
        raise ValueError("h1 and h2 must have same length")
    crossover_point = np.random.randint(1, n)
    h_new = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use initial construction if available
    initial_h = initial_h_values if 'initial_h_values' in globals() else None

        n_points = 45
        # Create a simple initial construction
        h0 = np.ones(n_points) * 0.5

    # Ensure h0 is feasible
    h0 = project_to_feasible(h0, n_points)

    # Settings
    population_size = 50
    generations = 12
    top_k = population_size // 2
    mutation_rate = 0.1
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    # Create initial population
    population = [h0]
    for _ in range(population_size - 1):
        h_perturbed = h0 + np.random.normal(0, 0.05, n_points)
        h_perturbed = project_to_feasible(h_perturbed, n_points)
        population.append(h_perturbed)

    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness.append((c5, h))
        
        # Sort by fitness (ascending)
        fitness.sort(key=lambda x: x[0])
        best_in_gen = fitness[0][0]
        if best_in_gen < best_c5:
            best_c5 = best_in_gen
            best_h = fitness[0][1].copy()
        
        # Selection
        parents = [h for _, h in fitness[:top_k]]
        
        # Crossover and mutation
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                child1 = crossover_h(parents[i], parents[i+1])
                child2 = crossover_h(parents[i+1], parents[i])
            else:
                child1 = parents[i].copy()
                child2 = parents[i].copy()
            # Mutate
            child1 = mutate_h(child1, n_points, mutation_rate)
            child2 = mutate_h(child2, n_points, mutation_rate)
            offspring.extend([child1, child2])
        
        # Replace population (keep best parents and all offspring)
        population = parents[:top_k] + offspring[:population_size-top_k]
    
    return (best_h, best_c5, n_points)
