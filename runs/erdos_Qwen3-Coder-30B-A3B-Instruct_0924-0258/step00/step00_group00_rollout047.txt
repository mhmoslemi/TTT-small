import numpy as np
import time
from scipy import correlate

def evaluate_h(h, dx):
    # Compute the correlation for all lags and find maximum
    corr = correlate(h, 1 - h, mode='full')
    # Only consider the valid overlapping parts
    # For a sequence of length n, the full correlation has length 2*n-1
    # The valid range for k is from -(n-1) to (n-1)
    n = len(h)
    # Take the center portion representing valid overlaps
    mid = len(corr) // 2
    # We want the maximum of non-negative lags (overlaps)
    # but the problem asks for max over ALL k, including negative
    # The correlations for negative lags are at the beginning, positive at the end
    # ignoring the middle which is for k=0
    valid_corr = corr[mid - (n - 1):mid + n - 1]
    # But we also need to get the parts for positive lags, and negative lags
    # The actual valid overlaps have no common portion ala case k=0
    # So we exclude that middle one, and take all other entries and normalize
    # For now, just take the maximum in absolute terms of all correlations:
    # in our use case, the symmetric values (with consistent k) appear as
    # the center cross-correlation contributes the 0-lag, and cross-correlation
    # of shifted h with 1-h gives us overlapping values;
    # The function definition says: C5 = max_k int h(x)(1 - h(x+k)) dx
    # so our correlate would give sum over index i of h[i] * (1 - h[i+k]),
    # which is the sum of overlap values for shift k, so that is exactly what we need
    # the cross_correlation result is a correlation, so we get the maximum of that
    return np.max(np.abs(valid_corr)) * dx

def rescale_h(h, target_sum):
    """Rescale h to have a specific sum."""
    current_sum = np.sum(h)
    if np.abs(current_sum) < 1e-10:
        # Avoid division by zero if sum is 0
        h.fill(target_sum / len(h))
    else:
        h *= target_sum / current_sum
    return h

def mutate_h(h, mutation_rate=0.1, delta=0.1):
    h = h.copy()
    mask = np.random.rand(len(h)) < mutation_rate
    h[mask] += np.random.uniform(-delta, delta, size=np.sum(mask))
    h = np.clip(h, 0, 1)
    h = rescale_h(h, len(h) / 2)
    return h

def crossover_h(h1, h2):
    # One-point crossover
    n = len(h1)
    point = np.random.randint(1, n)
    h = h1[:point].copy()
    h = np.concatenate([h, h2[point:]])
    h = rescale_h(h, len(h) / 2)
    return h

def tournament_selection(population, fitnesses, tournament_size=3):
    selected = []
    for _ in range(len(population)):
        idxs = np.random.choice(len(population), tournament_size)
        best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
        selected.append(population[best_idx])
    return selected

def compute_fitness(h_vals, dx):
    return evaluate_h(h_vals, dx)

        current_best_idx = np.argmin(fitnesses)

        selected = tournament_selection(population, fitnesses, tournament_k)

        new_population = [best_h.copy(), selected[0].copy()]  # Elitism

                child1 = crossover_h(p1, p2)
                child2 = crossover_h(p2, p1)
                new_population.extend([child1, child2])
            else:
                parent = np.random.choice(selected)
                child = mutate_h(parent, mutation_rate, 0.1)
                new_population.append(child)
                
        # Trim to exact population size
        population = new_population[:pop_size]
        
    # Final evaluation of best solution
    final_fitness = compute_fitness(best_h, dx)
    return (best_h.tolist(), final_fitness, n_points)
