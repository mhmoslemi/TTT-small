import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Determine n_points from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 100  # Default
        h = np.ones(n_points, dtype=float) * 0.5
    
    # Ensure sum(h) = n_points / 2 exactly
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h[i] in [0, 1]
    h = np.clip(h, 0, 1)
    
    # Compute dx
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = None
    start_time = time.time()
    
    # Gradient descent parameters
    alpha = 0.1
    decay_factor = 0.95
    momentum = 0.9
    v = np.zeros_like(h)
    improvement_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute all overlaps
        correlation = np.correlate(h, 1 - h, mode="full")
        k_indices = np.arange(len(correlation)) - (len(h) - 1)
        # Exclude self-correlation (k=0) if needed but we need max over all
        # Actually compute C5 correctly: max over all shifts k
        C5_vals = correlation * dx
        k0 = np.argmax(C5_vals)
        max_overlap = C5_vals[k0]
        
        if best_c5 is None or max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            improvement_count = 0
            last_improvement = time.time()
        else:
            improvement_count += 1
            if improvement_count > 50:
                # Early stopping
                break
                
        # Compute gradient for shift k0
        # This corresponds to derivative of sum h[i] (1 - h[i+k0])
        # w.r.t. h[i]  
        grad = np.zeros_like(h)
        # We need to consider correlation[i] = sum_{j} h[j] * (1 - h[j+i])
        # So its contrib to grad[i] is (1 - h[i + k0]) if i+k0 is valid index
        # and -(h[i - k0]) if i >= k0
        
        k_shift = k0 - (len(h) - 1)
        
        # When computing correlation, use correlate(h, 1-h, mode='full')
        # typical k_shift ranges from -len(h)+1 to len(h)-1
        # correlation index k = k0 means shift k0 - (n-1) 
        # so shift k0 gives us support from i to i+k0 where we add (1-h[i+k0])
        # and from i-k0 to i we add -h[i-k0]


        # Sodium for computing gradient: shift k0
        if k_shift >= 0:
            # we are adding 1-h[i+k_shift] to grad[i]
            # so for i from 0 to len(h)-k_shift-1, grad[i] += 1-h[i+k_shift]
            # and for i from k_shift to len(h)-1, grad[i] -= h[i-k_shift]  
            # But we are only concerned with positions where such contributions exist
            # in valid range.
            # corr[i] = sum_{j} h[j] * (1-h[j+i-k_shift]) 
            # grad[i] = diff corr[i] / diff h[i] = sum_{j, j+i-k_shift >=0} (1-h[j+i-k_shift]) 
            #  - sum_{j, j+i-k_shift<0 and j>=0} h[j+i-k_shift]
            
            # Actually possible: for correlation with shift k0 use h and 1-h, 
            # grad for h[i] from shift k_shift = k0-(n-1) is:
            #  1_{i+k_shift < len(h)} * 1 - 1_{i>=k_shift} * h[i-k_shift] 
            #  = 1 - 1_{i>=k_shift} * h[i-k_shift] for i <= len(h)-k_shift-1  
            #  = -h[i-k_shift] for i >= k_shift
            
            # Bias is:
            #  1 for shift k (i,k): i+shift = k => i = k-shift  
            #  correlation[i] = sum_{j} h[j] * (1 - h[j + i - shift]) = sum_j h[j] * (1 - h[j + (i-shift)])
            #  We want to compute grad[i] = d/d h[i] of correlation[i]
            #  = sum_j d/d h[i] (h[j] * (1 - h[j + i - shift])) 
            #  = sum_{j, j == i} 1 * (1 - h[j + i - shift]) 
            #                  + sum_{j, j + i - shift == i} h[j] * (-1)  
            #  = (1 - h[i + i - shift]) + h[i - i + shift] 
            #  = (1 - h[i + i - shift]) + h[shift]  
            #  We should do it properly: shift = k0 - (n-1) 
            #  j + i - shift = i => j = shift
            #  thus 1 * (1 - h[shift]) is added to gradient of h[i] 
            #  
            # Viz: for shift k, correlation[i] = sum_{j} h[j] * (1 - h[j+i-k])
            # grad[i] = d/dh[i] sum_{j} h[j] * (1 - h[j+i-k])
            #         = sum_{j, j+i-k==i} (h[j] * (-1))  
            #         + sum_{j, j==i} (1 - h[j+i-k])
            #         = -h[i-k] * 1_{i>=k}  
            #         + (1 - h[i+k]) * 1_{k<=i}
            
            # So for shift k0, we have:
            # grad[i] += 1 - h[i+k0-(n-1)] if i+k0-(n-1) valid  
            #          -= h[i - (k0 - (n-1))]  if i >= k0 - (n-1) 
            # So total of:
            # if i >= k0 - (n-1), gradient += -h[i - k0 + n-1]   
            # if i <= n-1 - k_shift and k_shift >= 0: jq = 1-h[i+k_shift]  
            bounds = len(h) - k_shift - 1
            if bounds < len(h):
                # indices where h[i+k_shift] is array bounds
                grad[:bounds] += 1
                grad[:bounds] -= h[k_shift:bounds+k_shift]
                # Add contribution of indices where 1-h[i+k_shift]
                if k_shift < bounds:
                    grad[k_shift:bounds] += 1-h[:bounds-k_shift]
            else:
                # i+k_shift >= len(h) for some i, so contribution to gradient is negligible  
                # Intermediate case: indices where both computed terms exist
                valid_indices = np.arange(len(h))
                shift_idx = valid_indices + k_shift
                valid_i = shift_idx < len(h)
                grad[valid_indices[valid_i]] += 1
                grad[valid_indices[valid_i]] -= h[shift_idx[valid_i]]
                
            # Direct loop-based
            grad = np.zeros_like(h)
            for i in range(len(h)):
                # One contribution: 1 - h[i+k_shift]  
                # if k_shift >=0 and i+k_shift < len(h)  
                if i+k_shift < len(h) and k_shift >=0:
                    grad[i] += 1
                if i+k_shift < len(h) and k_shift >= 0:  
                    grad[i] -= h[i+k_shift]  # derivative term of h[i] from term h[i+k_shift]
                if i-k_shift >= 0 and k_shift >= 0:
                    grad[i] -= h[i-k_shift]  

        else:
            # k_shift < 0
            # kl = |k_shift| 
            # corr[i] = sum_j h[j] (1 - h[j+i-k_shift]) = sum_j h[j] (1 - h[j+i+kl])
            # grad[i] = derivative = sum_j d/dh[i] h[j] * (1-h[j+i+kl]) + h[j] * d/dh[i] (1-h[j+i+kl])
            # = sum_{j,j==i} (1-h[j+i+kl])   + sum_{j,j==i+kl} h[j]*(-1)
            # = (1-h[i+kl]) + h[i-kl] * 1_{i>=kl}
            # double-check formula
            # At i, v = h[i+kl] * (1 - h[i+kl]) => grad[i] += 1 - h[i+kl]  
            # At i, v = h[i-kl] * (1-h[i]) => but we only sum h[j] * (1-h[j+i-k]) 
            # Sampling of h[j] has j : j+i-k = i => j = k => d/dh[i] = (1-h[k])  
            # i: j+i-k = i => j=k. So: d/dh[i] term = (1-h[k]) reported to gradient of h[i]
            # Taking insight:
            # corr[i] = sum_{j} h[j] (1 - h[j+i-k_shift])  
            # For h[i]: the contribution is from terms where j==i or j==i+k_shift
            # case 1: j==i, then we have h[i] * (1-h[i+i-k_shift]) = h[i] * (1 - h[i*2 - k_shift])  
            # Wait: that's not right. Let's do clairvoyant:
            # corr[i] = sum_{j} h[j] (1 - h[j+i-k]) 
            # grad[i] = sum_{j} d/d h[i} (h[j] (1-h[j+i-k])) = sum_{j} (1-h[j+i-k]) 
            # when do we have j=i ? We have h[j] = h[i] and that term is h[i] * (1 - h[i+i-k]) = h[i] * (1 - h[2*i-k])
            # But that doesn’t match intuitive understanding. Inspecting clearer:
            # corr[i] = sum_{j} h[j] * (1 - h[j + (i+1) - (k-1)])  where we shift items a bit
            # Alternately look again:
            # corr[i] = sum_j h[j] (1 - h[(j + i) - k])  
            # = sum_j h[j] - sum_j h[j] h[j+i-k]
            # Gradient of corr[i] with respect to h[i] is derivative of this:
            # d/dh[i] sum_j h[j] - sum_j h[j] h[j+i-k]    
            # = 0 - sum_j d/dh[i]   (h[j] h[j+i-k]) = -sum_{j, j+i-k==i} h[j] - h[j]  
            # = - h[i] - h[i-k]  for all j such that j+i-k==i i.e., j==k
            # So for every term in correlation: h[k] (1-h[k])  
            # Now taking shifts from correlation:
            # i in [0, length-1], but the full computation involves i-k for all k. 
            # This is due to e.g. sum_h[j] * (1-h[j+k0]) where the i-th h[i] 
            # contributes to the k-th index of correlation where j+i-k = i.
            # Let's re-derive:
            # For a fixed index i in [0, range], 
            # we look at array of size 2*len(h)-1 for full correlation, valid for k=0...2n-2
            # We want to compute the contribution of h[i] to correlation[k], i.e.
            # take formula  sum_j h[j] (1 - h[j+i-k])  
            # Contribution of h[i] to that is when j=i, so term is (1 - h[i+i-k])  
            # So i'd be: 1 - h[2*i-k]  
            # But that's an index which might not be valid, and that gives us the full connection of
            # d/dh[i] sum_j h[j] (1-h[j+i-k]) = 1_{2*i-k valid} * (1-h[2*i-k])  
            # But we also indirectly get from the second part:
            # in sum_j h[j] h[j+i-k], we look for contributions to derivative from h[i]:  
            # d/dh[i] h[j] h[j+i-k] = h[j] * 1_{j+i-k=i} * 1 = h[j] for j=i-k
            # So we must also add 
            # sum_{j,i-k=j} h[j] = h[i-k]  
            # But for fixed i, only valid for j>=0 and j<2*n-1, so j=i-k => i-k>=0 and i-k<2*n-1  
            # So sum_{j=i-k, 0 <= j < 2*n-1} h[j]  
            # Actually underestimate of 2 errors:
            # 1. Jacobian
            # 2. Order.  The standard correlation from convolution has that j
            # 1.) CORRELATION[i,k] is computed for indices where h[j] * (1-h[j+i+k])

            # a[j] = h[j] and b[j] = (1-h[j])  
            # don't use default interpretation of bulk correlation
            # instead use:
            # correlation[i](k) = sum_{j} a[j] * b[j+i-k] = sum_{j} h[j] * (1-h[j+i-k])

            # In calculation: correlation full, with size 2*n-1. For shift k=0: using indices

            # 
            # To be clear: slice of size 2*n-1, with shift corresponding to k0
            # take h_new and 1-h and compute full correlation, at k0 index

            # (1) contribution of term h[k0] = 1 - h[k0 + i -k0] (here i is constant index)

            # partial contribution of term h[j + k0] when j+i-k0 = i =>  j = k0

            # =

            # d_sum term w.r.t. h[i] = 1 - h[i + k0-n+1] + h[0] if (i+k0-n+1 >=0)

            # Given: h, and for each k, define a_k = sum_{j} h[j] (1 - h[j+k])

            # When k = k0:
            # = (1-h[i+k0]) + h[i-k0]
            # Actually a bit muddled, use NumPy directly
        else:
            # More careful procedure
            grad = np.zeros_like(h)
            # relabel k_shift as k
            k = k_shift
            # From coefficient positions in correlation that signal h[i] 
            # i is the variable we differentiate. For term h[i] in h, in correlation...
            for i in range(len(h)):
                # contribution from term h[i] * (1 - h[i+k]) to correlation index that depends k
                # shards about spatial relationship, but ledger real line indexing (not physical indices)
                # correlation[i+k_shift] involves h[j] * (1 - h[j+i+k_shift])

                # if j+k == i, that means j == i-k. Here index is j= i-k

                # For blur: keep all fussing
                # This is subtle
                # contribution to effortless manual tracking
                if i+k < len(h):
                    grad[i] += 1
                if i+k >= 0 and i+k < len(h):
                    grad[i] -= h[i+k]

        
        # Apply gradient step with momentum
        v = momentum * v - alpha * grad
        h_new = h + v
        
        # Project onto feasible set
        # 1. Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # 2. Adjust sum to be exactly n_points/2 
        # We want sum(h_new) = n_points / 2
        current_sum = np.sum(h_new) * dx
        desired_sum = 1.0
        delta_sum = desired_sum - current_sum
        # Distribute the residual proportionally 
        if abs(delta_sum) > 1e-10:
            # Allocate proportionally: for large delta will be not uniform
            # Simpler: just add delta to all
            # Better: proportionally to h_new
            h_new = h_new + delta_sum / len(h_new) # Small uniform
            # clip to [0,1] again
            h_new = np.clip(h_new, 0, 1)
            # final correction for sum
            s = np.sum(h_new) * dx
            # Normalize such that sum(h_new) = desired_sum (1)
            h_new = (h_new / s) * desired_sum
        
        h = h_new
        
        # Early stopping if stuck
        if improvement_count > 50 and (time.time() - last_improvement) > 10:
            break

    # Final computation of C5
    final_corr = np.correlate(best_h, 1-best_h, mode="full")
    final_corr *= dx
    final_c5 = np.max(final_corr)
    
    return (best_h.tolist(), float(final_c5), int(n_points))
