import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract the valid correlations (excluding padding)
    valid_corr = corr[n - 1:2 * n - 1]
    # Normalize and find maximum
    c5 = np.max(valid_corr) * dx
    return c5

def project_h(h):
    """Project h to satisfy constraints: h[i] in [0,1] and sum(h) = n/2"""
    # Clamp values to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to satisfy integral constraint
    current_sum = np.sum(h)
    target_sum = len(h) / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, sigma):
    """Mutate h with Gaussian noise and project back."""
    h_new = h + np.random.normal(0, sigma, len(h))
    return project_h(h_new)

def crossover_h(h1, h2, alpha=0.6):
    """Blend crossover between two solutions."""
    beta = np.random.uniform(alpha, 1 - alpha)
    h_new = beta * h1 + (1 - beta) * h2
    return project_h(h_new)

def compute_fitness(h):
    """Compute the fitness (C5 value) of h."""
    return evaluate_erdos_solution(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    global initial_h_values
    try:
        h0 = np.array(initial_h_values)
    except:
        # Fallback: initialize with a piecewise constant construction
        n_points = 70
        h0 = np.ones(n_points) * 0.5
        h0[0:n_points//2] = 1.0
        h0[n_points//2:] = 0.0
        h0 = project_h(h0)

    n_points = len(h0)
    # Population of size 100
    population_size = 100
    population = [h0.copy()]
    # Generate diversified initial population
    for _ in range(population_size - 1):
        h = mutate_h(h0, 0.1)
        population.append(h)
    
    # Track best solution
    best_h = h0.copy()
    best_c5 = compute_fitness(best_h)
    
    # Optimization loop
    max_generations = 1000
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate population
        fitness_scores = [compute_fitness(h) for h in population]
        
        # Track best
        min_fitness = min(fitness_scores)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitness_scores.index(min_fitness)].copy()
        
        # Selection: Top 40% plus some random for diversity
        sorted_indices = np.argsort(fitness_scores)
        selected_indices = sorted_indices[:int(0.4 * population_size)]
        # Add some random
        random_indices = np.random.choice(population_size, int(0.1 * population_size))
        selected_indices = np.concatenate([selected_indices, random_indices])
        selected_indices = list(set(selected_indices)) # Remove duplicates
        selected_population = [population[i] for i in selected_indices]
        
        # Produce offspring
        offspring = []
        # Crossover
        for k in range(int(0.3 * population_size)):
            if len(selected_population) < 2:
                break
            i1, i2 = np.random.choice(len(selected_population), 2, replace=False)
            h1, h2 = selected_population[i1], selected_population[i2]
            h_child = crossover_h(h1, h2)
            offspring.append(h_child)
        # Mutation
        for _ in range(population_size - len(offspring)):
            if len(selected_population) == 0:
                break
            h = selected_population[np.random.randint(len(selected_population))]
            h_child = mutate_h(h, 0.1 * np.exp(-gen / 100.0))
            offspring.append(h_child)
        
        # Local refinement on best candidates
        top_indices = sorted_indices[:int(0.05 * population_size)]
        for i in top_indices:
            h_refine = population[i].copy()
            # Simple gradient descent step
            learning_rate = 0.02 * np.exp(-gen / 100.0)
            for _ in range(5):  # few steps to avoid overfitting
                h_new = h_refine + learning_rate * np.random.normal(0, 0.01, n_points)
                h_new = project_h(h_new)
                if compute_fitness(h_new) < compute_fitness(h_refine):
                    h_refine = h_new
                else:
                    break
            offspring.append(h_refine)
        
        # Replace bottom half of population
        bottom_indices = sorted_indices[int(0.5 * population_size):]
        for i, idx in enumerate(bottom_indices):
            if i < len(offspring):
                population[idx] = offspring[i]
    
    # Final check
    final_c5 = compute_fitness(best_h)
    # Clamp final solution for clean output
    best_h = project_h(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
