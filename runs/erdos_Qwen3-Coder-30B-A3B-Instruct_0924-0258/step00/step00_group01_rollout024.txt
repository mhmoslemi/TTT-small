import numpy as np
from scipy import correlate
import time
import random

def evaluate_h(h, dx):
    """Evaluate the overlap integral for a given step function h."""
    # h is a numpy array of length n_points
    # We compute max_k int h(x)(1 - h(x+k)) dx
    # Using correlate with mode='full' and then taking max over all lags
    corr = correlate(h, 1 - h, mode='full')
    # Take only the relevant part (from -n+1 to n-1)
    n = len(h)
    mid = len(corr) // 2
    relevant = corr[mid - n + 1: mid + n - 1]
    return np.max(relevant) * dx

def project_feasible(h):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n/2."""
    h = np.clip(h, 0, 1)
    n = len(h)
    current_sum = np.sum(h)
    if abs(current_sum - n / 2) < 1e-10:
        return h
    
    # Adjust to make sum equal to n/2
    scale = (n / 2) / current_sum
    h = h * scale
    h = np.clip(h, 0, 1)
    return h

def initialize_population(n_points, pop_size, initial_h):
    """Initialize population with given initial solution and random ones."""
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select an individual using tournament selection."""
    selected_idx = random.randint(0, len(population) - 1)
    for _ in range(tournament_size - 1):
        idx = random.randint(0, len(population) - 1)
        if fitnesses[idx] < fitnesses[selected_idx]:
            selected_idx = idx
    return population[selected_idx]

def crossover(parent1, parent2):
    """Uniform crossover of two parents."""
    n = len(parent1)
    mask = np.random.rand(n) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    
    # Project to make sum constraint valid
    child1 = project_feasible(child1)
    child2 = project_feasible(child2)
    
    return child1, child2

def mutate(h, mutation_rate=0.05, noise_std=0.1):
    """Mutate h by adding Gaussian noise."""
    h = h.copy()
    n = len(h)
    for i in range(n):
        if random.random() < mutation_rate:
            h[i] += np.random.normal(0, noise_std)
    
    # Project to make sum constraint valid
    h = project_feasible(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    random.seed(seed)

    # Get initial construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        # Create a simple initial construction
        n_points = 47
        initial_h = np.ones(n_points) * 0.5
        initial_h[:n_points//2] = 1.0  
        initial_h[n_points//2:] = 0.0  
        initial_h = project_feasible(initial_h)

    n_points = len(initial_h)
    dx = 2.0 / n_points
    pop_size = 20
    max_generations = 1000
    tournament_size = 3
    mutation_rate = 0.05
    
    # Initialize population
    population = initialize_population(n_points, pop_size, initial_h)
    
    best_fitness = float('inf')
    best_h = None

    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Evaluate fitness for all individuals
        fitnesses = []
        for h in population:
            fit = evaluate_h(h, dx)
            fitnesses.append(fit)
            
            if fit < best_fitness:
                best_fitness = fit
                best_h = h.copy()
        
        # Sort population by fitness
        sorted_indices = sorted(range(len(fitnesses)), key=lambda i: fitnesses[i])
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Create new generation
        new_population = [population[0], population[1]]  # Elitism
        
        # Generate rest of population through crossover and mutation
        while len(new_population) < pop_size:
            p1 = tournament_selection(population, fitnesses)
            p2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(p1, p2)
            
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population
        
    # Final evaluation of best solution
    if best_h is not None:
        final_fitness = evaluate_h(best_h, dx)
    else:
        final_fitness = evaluate_h(initial_h, dx)
        
    return (best_h.tolist(), final_fitness, n_points)
