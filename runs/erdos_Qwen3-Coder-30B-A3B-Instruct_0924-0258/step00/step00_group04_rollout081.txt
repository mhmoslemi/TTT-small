import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import random

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # Take the maximum (excluding trivial cases)
    max_corr = np.max(corr[n-1:])
    return max_corr * dx

def project_and_rescale(h, n_points):
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def crossover(parent1, parent2, n_points):
    # Create 2-3 crossover points
    n_crossovers = random.randint(2, 3)
    crossover_points = sorted(random.sample(range(1, n_points), n_crossovers))
    offspring = np.copy(parent1)
    start = 0
    for i, point in enumerate(crossover_points):
        if i % 2 == 0:
            offspring[start:point] = parent2[start:point]
        start = point
    if len(crossover_points) % 2 == 0:
        offspring[start:] = parent2[start:]
    return offspring

def mutate(h, n_points, rate=0.05):
    # Mutate 5-10% of the elements
    n_mutate = int(n_points * rate)
    indices = random.sample(range(n_points), n_mutate)
    for i in indices:
        h[i] += np.random.normal(0, 0.1)
    return project_and_rescale(h, n_points)

def greedy_optimize(h, n_points, max_iter=100):
    def obj(h_vec):
        h_new = h_vec.reshape(-1)
        return evaluate_c5(h_new)
    def grad(h_vec):
        # Placeholder - actual gradient would be complex
        # Use finite differences for smoother gradient estimation
        eps = 1e-5
        grad_vec = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            grad_vec[i] = (obj(h_plus) - obj(h_minus)) / (2 * eps)
        return grad_vec
    # Simple gradient descent with projection
    h_flat = h.flatten()
    learning_rate = 0.01
    for _ in range(max_iter):
        g = grad(h_flat)
        h_flat -= learning_rate * g
        h_flat = project_and_rescale(h_flat, n_points)
        if np.allclose(h_flat, h, atol=1e-5):
            break
    return h_flat.reshape(h.shape)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    time_start = time.time()
    
    n_points = 42  # Starting with the initial solution
    pop_size = 50
    best_h = None
    best_c5 = float('inf')
    steady_state_count = 0
    mutation_rate = 0.05
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if np.abs(np.sum(initial_h) - n_points / 2) > 1e-6:
            initial_h = project_and_rescale(initial_h, n_points)
        population = [initial_h] + [np.random.rand(n_points) for _ in range(pop_size - 1)]
    else:
        population = [np.random.rand(n_points) for _ in range(pop_size)]
    
    for i in range(len(population)):
        population[i] = project_and_rescale(population[i], n_points)
    
    generation = 0
    max_generations = 200
    
    while time.time() - time_start < budget_s - 1 and generation < max_generations:
        # Evaluate all candidates
        scores = []
        for h in population:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            scores.append(c5)
        
        # Sort population
        sorted_indices = np.argsort(scores)
        population = [population[i] for i in sorted_indices]
        scores = [scores[i] for i in sorted_indices]
        
        # apply greedy optimization to top 10%
        top_n = max(1, pop_size // 10)
        for i in range(top_n):
            population[i] = greedy_optimize(population[i], n_points)
        
        # Update best solution
        if scores[0] < best_c5:
            best_c5 = scores[0]
            best_h = population[0].copy()
        
        # Crossover and mutation
        offspring = []
        # Elitism
        offspring.extend(population[:pop_size//4])
        
        # Create new offspring
        while len(offspring) < pop_size:
            # Select two parents (tournament)
            parent1_idx = random.randint(0, pop_size//2)
            parent2_idx = random.randint(0, pop_size//2)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            child = crossover(parent1, parent2, n_points)
            
            # Mutate
            if random.random() < 0.5:  # 50% chance of mutation
                child = mutate(child, n_points, rate=mutation_rate)
            
            offspring.append(child)
        
        population = offspring
        
        # Periodically increase mutation if no improvement in last 50 gens
        if generation % 10 == 0:
            steady_state_count += 1
            if steady_state_count > 5:
                mutation_rate = 0.1
                steady_state_count = 0
        
        generation += 1
        
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_c5(best_h)
    else:
        # Fallback to first member
        best_h = population[0]
        final_c5 = evaluate_c5(best_h)
        
    # Ensure final solution is feasible
    best_h = project_and_rescale(best_h, n_points)
    final_c5 = evaluate_c5(best_h)
        
    return (best_h.tolist(), final_c5, n_points)
