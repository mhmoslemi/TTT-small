import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute overlap integral for all shifts
    overlap = correlate(h, 1 - h, mode="full") * dx
    # Return maximum overlap
    return np.max(overlap)

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.full(n_points, 1.0 / n_points)
    return h

def mutate(h, mutation_rate=0.1):
    """Mutate h by perturbing some values and rescaling."""
    n_points = len(h)
    h_mut = h.copy()
    indices = np.random.choice(n_points, size=int(n_points * mutation_rate), replace=False)
    for i in indices:
        h_mut[i] += 0.1 * np.random.randn()
    return project_to_feasible(h_mut, n_points)

def crossover(h1, h2):
    """Perform single-point crossover on two parents."""
    n_points = len(h1)
    point = np.random.randint(1, n_points)
    h_child1 = np.concatenate([h1[:point], h2[point:]])
    h_child2 = np.concatenate([h2[:point], h1[point:]])
    return h_child1, h_child2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    population_size = 50
    generations = 12
    mutation_rate = 0.1
    n_evaluations = 0
    
    # Initialize population
    population = []

    else:

    # Add initial solution to population
    population.append(initial_h.copy())
    n_evaluations += 1
    
    # Add random perturbations
    for _ in range(population_size - 1):
        h = initial_h + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
        n_evaluations += 1
    
    # Evaluate initial population
    fitness = []
    for i, h in enumerate(population):
        c5 = evaluate_erdos_solution(h)
        fitness.append(c5)
        if i == 0:
            best_h = h.copy()
            best_c5 = c5
    
    best_fitness = min(fitness)
    
    # Evolution loop
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:  # Save 1s for cleanup
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution so far
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Select top 50% as parents
        parents = population[:population_size//2]
        
        # Generate offspring through crossover (100% of new population from crossover)
        new_population = parents.copy()
        while len(new_population) < population_size:
            # Randomly pair parents and perform crossover
            idx1, idx2 = np.random.choice(len(parents), 2, replace=False)
            child1, child2 = crossover(parents[idx1], parents[idx2])
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            new_population.append(child1)
            if len(new_population) < population_size:

        population = new_population[:population_size]
        
        # Evaluate new population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness.append(c5)
            n_evaluations += 1
    
    # Final validation
    if evaluate_erdos_solution(best_h) < best_c5:
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
