import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h_values, dx):
    """Evaluate the maximum overlap integral for given h_values"""
    # Compute the full cross-correlation between h and (1-h)
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The maximum overlap is the maximum value in the correlation
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_h(h, dx, n_points):
    """Project h to satisfy constraint sum(h) * dx = 1 and bounds [0,1]"""
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Rescale to satisfy integral constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    # Ensure final integral = 1
    h = np.clip(h, 0, 1)
    return h

        h_plus[i] += eps

        h_minus[i] -= eps

        val_plus = evaluate_h(h_plus, dx)
        val_minus = evaluate_h(h_minus, dx)
        grad[i] = (val_plus - val_minus) / (2 * eps)
    return grad

            alpha *= 1.2  # Increase step size

            alpha *= 0.5  # Reduce step size
        if alpha < 1e-8:
            break
    return best_h, best_val
