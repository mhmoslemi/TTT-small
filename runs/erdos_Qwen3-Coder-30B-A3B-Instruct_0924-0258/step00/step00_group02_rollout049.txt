import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_vals, n_points):
    dx = 2.0 / n_points
    h = np.array(h_vals)
    # Compute convolution h * (1 - h) with full mode
    conv = np.correlate(h, 1 - h, mode="full")
    # Maximum value scaled by dx
    c5 = np.max(conv) * dx
    return c5

def project_to_feasible(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale so that sum(h) = n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def surrogate_objective(h, n_points):
    dx = 2.0 / n_points
    h = np.array(h)
    # Convolution h * (1 - h)
    conv = np.correlate(h, 1 - h, mode="full")
    # Use sum of squares of all values as surrogate
    return np.sum(conv**2) * dx**2

def surrogate_gradient(h, n_points):
    dx = 2.0 / n_points
    h = np.array(h)
    inv_dx = 1.0 / dx
    # Convolution h * (1 - h)
    conv = np.correlate(h, 1 - h, mode="full")
    # Gradient of sum of squares 
    # Each term contributes 2 * conv[k] * d(conv[k])/dh[i]
    grad = np.zeros_like(h)
    conv_len = len(conv)
    # d(conv[k])/dh[i] is 1 if k == i or k == n_points-1-i, etc.
    # For full convolution, it's a tricky derivative but for small perturbations
    # we can compute it numerically or use symmetry
    for i in range(len(h)):
        # Use finite difference to compute gradient approximation
        eps = 1e-6
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        val_plus = surrogate_objective(h_plus, n_points)
        val_minus = surrogate_objective(h_minus, n_points)
        grad[i] = (val_plus - val_minus) / (2 * eps)

    return grad * inv_dx**2

def optimize_at_resolution(h_start, n_points, max_iter=100, time_budget_s=1.0):
    t0 = time.time()

    def obj(h):
        return surrogate_objective(h, n_points)

    def grad(h):
        return surrogate_gradient(h, n_points)

    def constraint_func(h):
        return np.sum(h) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint_func}
    bounds = [(0, 1) for _ in range(n_points)]

    result = minimize(obj, h_start, method='SLSQP', jac=grad, bounds=bounds, 
                      constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-8})

    h_opt = result.x
    h_feasible = project_to_feasible(h_opt, n_points)
    c5 = evaluate_erdos_solution(h_feasible, n_points)
    
    return h_feasible, c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    resolutions = [15, 30, 45, 60, 75]
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    t0 = time.time()
    
    # Start from initial construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        assert len(h_current) == 45
    else:
        # Initialize to a fixed pattern
        h_current = np.ones(45) * 0.5
        
    n_current = 45
    h_current = project_to_feasible(h_current, n_current)
    
    # Optimization parameters
    max_iter_per_level = 200
    num_perturbations = 3

    for level, n_points in enumerate(resolutions):
        if time.time() - t0 > budget_s * 0.95:
            break
            
        # Interpolate to current resolution
        if n_points > n_current:
            x_old = np.linspace(0, 2, n_current)
            x_new = np.linspace(0, 2, n_points)
            h_current = np.interp(x_new, x_old, h_current)
        elif n_points < n_current:
            x_old = np.linspace(0, 2, n_current)
            x_new = np.linspace(0, 2, n_points)
            h_current = np.interp(x_new, x_old, h_current)
            
        h_current = project_to_feasible(h_current, n_points)
        n_current = n_points
        
        # Perturb and optimize multiple times
        best_h_level = h_current
        best_c5_level = evaluate_erdos_solution(h_current, n_points)
        
        for iter_num in range(num_perturbations):
            if time.time() - t0 > budget_s * 0.95:
                break
                
            h_perturbed = h_current.copy()
            # Add small random perturbation
            noise = np.random.normal(0, 0.01, n_points)
            h_perturbed += noise
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            
            h_opt, c5_opt = optimize_at_resolution(h_perturbed, n_points, max_iter_per_level, 
                                                  budget_s * 0.1)
            
            if c5_opt < best_c5_level:
                best_c5_level = c5_opt
                best_h_level = h_opt
                
        if best_c5_level < best_c5:
            best_c5 = best_c5_level
            best_h = best_h_level
            best_n = n_points
            
        h_current = best_h_level
        
    if best_h is not None:
        return (list(best_h), best_c5, best_n)
    else:
        # Return self-init from 45 if nothing worked
        return (list(h_current), evaluate_erdos_solution(h_current, 45), 45)
