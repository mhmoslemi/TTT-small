import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_overlap(h):
    """Evaluate C5 for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # compute full correlation
    corr = correlate(h, 1-h, mode='full')
    # take max over all lags (k values)
    max_corr = np.max(corr)
    # rescale by dx
    return max_corr * dx

def project_and_rescale(h, n_points):
    """Project h to [0,1] and rescale to sum = n_points/2"""
    # Project to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def finite_diff_gradient(h, n_points):
    """Estimate gradient of overlap w.r.t. h using finite differences"""
    eps = 1e-4
    grad = np.zeros_like(h)
    base_c5 = evaluate_overlap(h)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_and_rescale(h_plus, n_points)
        c5_plus = evaluate_overlap(h_plus)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

    best_h = h.copy()
    best_c5 = evaluate_overlap(h)
    
    # Simulated Annealing parameters
    temp = 1.0
    min_temp = 0.01
    cooling_factor = 0.99
    patience = 50
    no_improve_count = 0
    
    # Mutation parameters
    noise_std = 0.05
    grad_scale = 0.1
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate current value
        current_c5 = evaluate_overlap(h)
        
        # Compute approximate gradient
        grad = finite_diff_gradient(h, n_points)
        
        # Generate candidate by adding gradient-guided noise
        noise = np.random.normal(0, noise_std, n_points)
        h_candidate = h + grad_scale * grad + noise
        
        # Project and rescale
        h_candidate = project_and_rescale(h_candidate, n_points)
        
        # Evaluate candidate
        new_c5 = evaluate_overlap(h_candidate)
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_candidate
            if new_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = new_c5
            no_improve_count = 0
        else:
            # Accept with probability
            if np.random.rand() < np.exp((current_c5 - new_c5) / temp):
                h = h_candidate
                no_improve_count = 0
            else:
                no_improve_count += 1
        
        # Update temperature
        temp = max(temp * cooling_factor, min_temp)
        
        # Reset temperature if no improvement for patience steps
        if no_improve_count > patience:
            temp = 1.0
            no_improve_count = 0
    
    # Final validation of best solution
    best_h = project_and_rescale(best_h, n_points)  # Ensure final constraint
