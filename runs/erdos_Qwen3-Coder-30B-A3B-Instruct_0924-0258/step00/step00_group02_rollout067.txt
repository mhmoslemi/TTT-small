import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant C5 for given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute overlap integral for all shifts
    conv = correlate(h, 1 - h, mode='full')
    # Take maximum over all shifts
    max_val = np.max(conv) * dx
    return max_val

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def pso_erdos_optimize(initial_h, n_particles=30, budget_s=600, seed=42):
    """Optimize h using Particle Swarm Optimization for Erdos constant."""
    np.random.seed(seed)
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Initialize swarm
    particles = []
    velocities = []
    
    # First particle is the initial solution
    particles.append(initial_h.copy())
    velocities.append(np.zeros(n_points))
    
    # Other particles are perturbed versions
    for _ in range(n_particles - 1):
        pert = initial_h + np.random.normal(0, 0.1, n_points)
        particles.append(project_feasible(pert, n_points))
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    
    # Initialize personal bests and global best
    pBest = [evaluate_erdos_solution(p) for p in particles]
    gBest_idx = np.argmin(pBest)
    gBest_val = pBest[gBest_idx]
    gBest_pos = particles[gBest_idx].copy()

    # PSO parameters
    omega = 0.9
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.5
    decay_rate = 0.99
    
    start_time = time.time()
    
    # Main loop
    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            # Update personal best
            current_val = evaluate_erdos_solution(particles[i])
            if current_val < pBest[i]:
                pBest[i] = current_val
                if current_val < gBest_val:
                    gBest_val = current_val
                    gBest_pos = particles[i].copy()
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            # Velocity update with adaptive bounds
            velocities[i] = (omega * velocities[i] + 
                           c1 * r1 * (particles[i] - particles[i]) + 
                           c2 * r2 * (gBest_pos - particles[i]))
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i], n_points)
        
        # Decrease inertia weight
        omega *= decay_rate
        
    return gBest_pos, gBest_val
