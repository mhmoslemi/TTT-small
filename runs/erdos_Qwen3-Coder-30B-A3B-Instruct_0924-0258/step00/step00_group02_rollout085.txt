import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute overlap for all shifts
    overlap = correlate(h, 1 - h, mode='full') * dx
    # Return maximum overlap
    return np.max(overlap)

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def mutate_h(h, n_points, mutation_rate=0.1):
    """Mutate h in place by adding Gaussian noise and reprojecting."""
    num_mutate = int(mutation_rate * len(h))
    indices = np.random.choice(len(h), num_mutate, replace=False)
    h[indices] += 0.1 * np.random.randn(num_mutate)
    return project_feasible(h, n_points)

def crossover_h(h1, h2, n_points):
    """Single-point crossover between h1 and h2."""
    n = len(h1)
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate((h1[:crossover_point], h2[crossover_point:]))
    child2 = np.concatenate((h2[:crossover_point], h1[crossover_point:]))
    child1 = project_feasible(child1, n_points)
    child2 = project_feasible(child2, n_points)
    return child1, child2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    # Initialize population
    population_size = 50
    generation_size = 75
    generations = 12
    mutation_rate = 0.1

    else:
        h_init = np.ones(n_points) * 0.5
    # Create initial population
    population = [h_init.copy()]
    for i in range(population_size - 1):
        h_perturb = h_init + np.random.uniform(-0.1, 0.1, n_points)
        population.append(project_feasible(h_perturb, n_points))
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        # Evaluation
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness.append((c5, h))
        fitness.sort(key=lambda x: x[0])
        # Keep top 50% and update best
        best_in_gen = fitness[0][0]
        if best_in_gen < best_c5:
            best_c5 = best_in_gen
            best_h = fitness[0][1].copy()
        # Selection (top 50%)
        selected = [h for _, h in fitness[:population_size // 2]]
        # Crossover and mutation
        new_pop = selected[:]
        # Generate offspring
        for _ in range(population_size // 2):  # pairs
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover_h(parent1, parent2, n_points)
            new_pop.extend([child1, child2])
        # Mutate
        for i in range(len(new_pop)):
            if np.random.rand() < 0.8:
                mutate_h(new_pop[i], n_points, mutation_rate)
        # Truncate to generation size
        population = new_pop[:generation_size]
    
    return (best_h, best_c5, n_points)
