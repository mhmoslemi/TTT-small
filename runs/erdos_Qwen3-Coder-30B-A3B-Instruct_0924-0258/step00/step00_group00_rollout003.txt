import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 10000
    n_particles = 30
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    max_velocity = 0.1
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    
    particles = [h_init.copy() for _ in range(n_particles)]
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Personal best
    pbest = [p.copy() for p in particles]
    pbest_scores = [float('inf')] * n_particles
    
    # Global best
    gbest = h_init.copy()
    gbest_score = float('inf')
    
    start_time = time.time()
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Clip to [0,1] and rescale
            h = np.clip(particles[i], 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            
            # Evaluate C5
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < pbest_scores[i]:
                pbest[i] = h.copy()
                pbest_scores[i] = c5
            
            # Update global best
            if c5 < gbest_score:
                gbest = h.copy()
                gbest_score = c5
            
            # Update velocity and position
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (pbest[i] - particles[i]) + 
                           social * r2 * (gbest - particles[i]))
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
    
    return (gbest, gbest_score, n_points)
