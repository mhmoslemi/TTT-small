import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Fallback initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Ensure sum(h) = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    last_improvement = 0
    iter_count = 0
    
    perturb_steps = 20
    alpha_candidates = [0.1, 0.5, 1.0]
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute all cross-correlations
        corr = correlate(h, 1 - h, mode='full')
        lags = np.arange(-n_points + 1, n_points)
        max_k = lags[np.argmax(corr)]
        max_corr = np.max(corr)
        
        # Compute gradient
        grad = np.zeros(n_points)
        if max_k >= 0:
            grad[:n_points - max_k] = 1 - h[max_k:]
        else:
            grad[-max_k:] = 1 - h[:n_points + max_k]
        
        # Try different step sizes
        best_h_new = h.copy()
        best_c5_new = best_c5
        
        for alpha in alpha_candidates:
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            
            # Rescale to ensure sum = n_points/2
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < best_c5_new:
                best_c5_new = c5_new
                best_h_new = h_new.copy()
        
        # If no improvement, add small Gaussian noise
        if best_c5_new >= best_c5:
            h_noisy = h + 0.01 * np.random.randn(n_points)
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = h_noisy * (n_points / 2) / np.sum(h_noisy)
            c5_noisy = evaluate_c5(h_noisy, dx)
            if c5_noisy < best_c5:
                best_c5_new = c5_noisy
                best_h_new = h_noisy
                
        # Update if improvement
        if best_c5_new < best_c5:
            h = best_h_new.copy()
            best_c5 = best_c5_new
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Perturb if no improvement in recent steps
        if last_improvement >= perturb_steps:
            h += 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = 0
            
        # Track best
        if best_c5_new < evaluate_c5(best_h, dx):
            best_h = best_h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute all cross-correlations
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
