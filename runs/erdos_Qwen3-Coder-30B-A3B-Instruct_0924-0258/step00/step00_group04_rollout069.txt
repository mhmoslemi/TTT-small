import numpy as np
from time import time
import math

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initialize from initial_h_values if available, otherwise create a uniform one
    n_points = 42
    if 'initial_h_values' in globals():

        # Ensure feasibility
        assert np.all(h >= 0) and np.all(h <= 1)
        assert abs(np.sum(h) - n_points / 2.0) < 1e-10
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2.0) / np.sum(h)
    
    # Compute C5 from current h
    def compute_C5(h):
        h_ext = np.concatenate([h, h])
        c5_vals = []
        dx = 2.0 / n_points
        for k in range(n_points):
            # With circular indexing
            c5 = np.sum(h * np.concatenate([h_ext[n_points-k:], h_ext[:n_points-k]]) * dx)
            c5_vals.append(c5)
        return max(c5_vals)

    # Compute gradient for k-th correlation
    def compute_gradient(h, k):
        n = len(h)
        grad = np.zeros(n)
        dx = 2.0 / n
        for i in range(n):
            j1 = (i - k) % n
            j2 = (i + k) % n
            grad[i] = 2.0 * dx * (1.0 - h[j1] - h[j2])
        return grad

    # Backtracking line search
    def line_search(h, grad, max_iter=10):
        alpha = 1.0
        c5_current = compute_C5(h)
        for _ in range(max_iter):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2.0) / np.sum(h_new)
            c5_new = compute_C5(h_new)
            if c5_new < c5_current:
                return alpha, h_new
            alpha *= 0.5
        return alpha, h_new

    # Initialize best
    best_h = h.copy()
    best_c5 = compute_C5(h)
    
    # Start optimization
    start_time = time()
    iter_count = 0
    last_improvement = 0
    
    while time() - start_time < budget_s - 0.1 and iter_count < 10000:
        # Compute all correlation values
        h_ext = np.concatenate([h, h])
        c5_vals = []
        dx = 2.0 / n_points
        
        # Compute all z_k values
        for k in range(n_points):
            c5 = np.sum(h * np.concatenate([h_ext[n_points-k:], h_ext[:n_points-k]]) * dx)
            c5_vals.append(c5)
        
        # Find worst overlap
        k_star = np.argmax(c5_vals)
        
        # Compute gradient for the worst case
        grad = compute_gradient(h, k_star)
        
        # Perform line search
        alpha, h_new = line_search(h, grad)
        
        # Update h
        h = h_new
        c5 = compute_C5(h)
        
        # Check for improvement
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iter_count
            
        # Occasionally add noise to escape local optima
        if iter_count % 50 == 0 and iter_count > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2.0) / np.sum(h)
            
        iter_count += 1
        
        # Early stopping if no improvement in 500 iterations
        if iter_count - last_improvement > 500:
            break
    
    return (best_h.tolist(), best_c5, n_points)
