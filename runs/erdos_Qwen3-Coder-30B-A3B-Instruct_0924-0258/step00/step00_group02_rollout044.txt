import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant for given step function h."""
    dx = 2.0 / len(h)
    # Compute the overlap integral for all shifts
    conv = correlate(h, 1 - h, mode="full")
    # Max over all possible shifts in [0,2]
    max_overlap = np.max(conv) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Fallback construction
        n_points = 45
        h0 = np.ones(n_points) * 0.5
    
    # Initialize swarm
    n_particles = 30
    particles = []
    velocities = []
    pBest = []
    pBest_vals = []
    
    # Initial particle is the known solution
    particles.append(h0.copy())
    velocities.append(np.zeros(n_points))
    pBest.append(h0.copy())
    pBest_vals.append(evaluate_erdos_solution(h0))
    
    for i in range(1, n_particles):
        # Slightly perturb the initial solution
        perturbation = np.random.normal(0, 0.1, n_points)
        h_pert = h0 + perturbation
        h_pert = project_to_feasible(h_pert, n_points)
        particles.append(h_pert.copy())
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        pBest.append(h_pert.copy())
        pBest_vals.append(evaluate_erdos_solution(h_pert))
    
    # Global best
    gBest_idx = np.argmin(pBest_vals)
    gBest = pBest[gBest_idx].copy()
    gBest_val = pBest_vals[gBest_idx]
    
    # PSO parameters
    omega = 0.9
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.5
    
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        for i in range(n_particles):
            # Evaluate current particle
            current_val = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if current_val < pBest_vals[i]:
                pBest[i] = particles[i].copy()
                pBest_vals[i] = current_val
            
            # Update global best
            if current_val < gBest_val:
                gBest = particles[i].copy()
                gBest_val = current_val
        
        # Update velocities and positions
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            # Velocity update
            v_prev = velocities[i]
            pbest_diff = pBest[i] - particles[i]
            gbest_diff = gBest - particles[i]
            
            velocities[i] = (omega * v_prev 
                           + c1 * r1 * pbest_diff 
                           + c2 * r2 * gbest_diff)
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
        
        # Decay inertia
        omega *= 0.995
    
    return (gBest, gBest_val, n_points)
