import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    # Compute all cross-correlations
    corr = correlate(h, 1 - h, mode="full")
    # We only care about the valid lags [0, 2*dx, ..., (n-1)*dx]
    # Since we are looking at h(x) * (1 - h(x + k))
    # the valid range of k corresponds to lags that are
    # around the middle of the full correlation, the middle length is n
    # From the full correlation with length 2*n - 1, 
    # the valid lags 'k' in [0, 2] correspond to indices [n-1, n, ..., n-1+2*(n-1)] 
    # Which is the full range, but we want only k in [0, 2], so we want to look at index range
    # [n-1:2*n-2] , since we want h at i, and h at i+k -> i ranges up to n-1-k (due to bounds)
    # i in [0, n-1-k] -> we want to select the corresponding full length segments
    # the entries for the lags 0 to 2 is from full [n-1 : n-1+2*(n-1)+1]
    # Actually there's a simpler way:
    # The valid lags k we want are from 0 to 2 with step dx -> k in [0, dx, 2*dx]?
    # No, k is a continuous variable, so running the correlate
    # The `correlate` gives the convolution of h and 1-h, which corresponds to
    # int h(x) * (1-h(x + tau)) dtau, but with discrete indices.
    
    # We compute the maximum of the correlation, but we need to map the correct lag indices
    # Summarize the shift by k that ranges from 0 to 2:
    n = len(h)
    # We take the full correlation (length 2*n - 1) and look at valid segments that correspond to k
    # To go from correlation index to shift k:
    # correlation_index = n - 1 + shift
    # shift = correlation_index - (n - 1)
    # valid shifts in [0, 2] where shift = k, 
    # shift in [0, 2] =>
    # k = shift = corr_index - (n-1) => corr_index = k + n-1
    # where k is in [0, 2], so corr_index in [n-1, n+1]
    # but the valid indices of cumulative overlap (also taking in account permutation) are 
    # from shift = 0=k to 2=k, so from index n-1 to n+1, and again we can get k in [0, 2] exactly?
    
    # Valid lag indices we care about (correlate indices that map to k in [0, 2])  
    # Assume dx = 2/n, so k can be approximated as s*dx for s in [0, n]  
    # But for k in [0, 2], the corresponding lag indices are ones that shift  
    # go from 0 to 2 corresponds to from 0 to n shifts, so it makes sense given  
    # stretch of the valid correlation index will depend on what we care about
    # Anyway, the key is looking over k in [0, 2], so that's from index 0 to len(h) including ends.
    # so from i = n-1 to n-1+2*(n-1)=2n-3, yes.
    # But let's just use all the definitive values and just find the correct j such that from 0 to n we can map k to indices.
    # Let's trust the correlate capability directly.
    
    # So for a vector h of size n
    # the result of correlate(h, 1-h, mode='full') 
    # is a vector of size 2*n-1,
    # and the i-th element gives the correlation with lag (i-n+1)  
    # so lag 0 = element n-1, lag 1 = element n, etc.
    # The full indices go from:
    # -n+1 to n-1 => -1 to 1 when n=3
    # So for lag in [0, 2], we want corr[n-1:n+1] (valid values)  
    # which i-th element corresponds to k = i - (n-1) 
    # so i from n-1 to n+1, k from 0 to 2
    # We then want actual values for k in [0, 2] thus we take 
    # indices from [n-1] to [n+1], which is 3 elements
    # Typically we'd want all the corrections to be captured, but since effect 
    # ranges up to kerne size of 2, the correlation values starting from middle
    # and we want from k = 0 to k = 2, it thus corresponds to a trim
    # of this way: 
    # c = correlate(h, 1 - h, mode="full")
    # l = len(c)
    # middle = (l - 1) // 2 
    # This is the index of lag 0, so lg = c[middle + k] = c[n-1+k]  
    # Want lag where lag from 0 to 2 so we need k in [0,2] which means index in [n-1, n+1]
    corr_max = np.max(corr[n-1:n+1]) * dx
    return corr_max

def run(seed=42, budget_s=600, **kwargs):
    # Initialize parameters
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction if exists
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
        n_points = len(h)
    else:
        # default initialization
        n_points = 45
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        # Scale to ensure sum(h) * dx = 1
        dx = 2.0 / n_points
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure feasibility: sum = 1/dx, each h[i] in [0,1]
    dx = 2.0 / n_points
    assert abs(np.sum(h) * dx - 1.0) < 1e-6, "Sum constraint broken"
    
    # Settings
    n_iter = 500
    tolerance = 1e-4
    lrate = 0.9
    
    # Frank-Wolfe algorithm
    best_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    
    for it in range(n_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations to find max lag
        corr = correlate(h, 1 - h, mode="full")
        # Get index range of [0,2] scale
        # lag_index = (lag + n - 1)  
        # for lag in [0, 2], we get index in [n-1, n+1]
        valid_corr = corr[n_points - 1: n_points + 1] 
        max_corr_idx = np.argmax(valid_corr)
        max_lag = max_corr_idx - (n_points - 1)  # Should be 0, 1, or 2
        # Get the corresponding correlation value
        true_max_corr = valid_corr[max_corr_idx] * dx
        
        if abs(true_max_corr - best_c5) < tolerance:
            continue
            
        # Compute the gradient (importance) of correlation k
        # d/dh_i ( sum_j h_j (1 - h_{j+k}) ) = (1 - h_{i+k}) 
        # So we take effect of changing h[i] on that correlation (key sum)
        # Check this more carefully - gradient w.r.t. one specific h_i for a given lag 
        # i.e., h_i contributes to overall contrib of the lag j = i + k where i is varied
        # Let's derive this from simpler perspective: 
        # the correlate sums over indices: 
        # for all i, legacy_sum += h[i] * (1- h[i + k]), we analyze the effect
        # of changing h[i]  on the output of that lag k sum. 
        # The h[i] is involved in the term where h[i] * (1 - h[i + k])
        # Its contribution to the k-th correlation is via the terms h[i] * (1 - h[i+k])  
        # Considering h[i] is fixed and we vary h[i], the gradient itself is:
        # d/dh[i]( h[i] * (1 - h[i + k]) ) = (1 - h[i + k])  
        # But this is not valid if i+k is out of bounds...
        # So let’s reason directly:
        # Consider all of correlate(h, 1-h) wrote explicitly with functions:
        # (h*1-h)_k  = sum_{i=0}^{n-1} h[i] * (1-h[i+k]) when i+k < n
        # So contribution to sum from each i:
        # 1. if i + k < n, then i depends on term h[i+k], so its gradient is -h[i+k]
        # 2. if i + k >= n, we're.carrying back, but that is messy in more involved style.
        # Use h.copy() to compute correlation gradient  
        # J(k, i) = d/dh[i] corr[k] = d/dh[i] sum_{j} h[j] * (1 - h[j+k]) = if j = i then (1 - h[i+k]) 
        # For fixed lag k, what contributes to gradient w.r.t. h[i]?
        # We need h[j] * (1 - h[j+k]) where j = i: that term contributes (1 - h[i+k])  
        # But it’s also at j = i only, so the contribution per i becomes:
        # gradient of corr_k w.r.t. h_i = (1 - h[i+k])  
        # Only valid as long as i+k < n
        # All valid h_i that can contribute to lag d
        # i is from 0 to n-1, k = d, then i+k in [k, n-1+k]
        # so h[i] contributes to lag d if 
        # if k is the lag index in correlate who corresponds to actual space shift k
        # we compute from the full corr, where corr[n-1 + d] is the contribution by lag d
        # so from now on, we use array aliasing:
        corr_index = n_points - 1 + max_lag
        # Better yet:
        # corr = correlate(h, 1-h, mode="full") 
        # corr is len 2*n-1
        # lag k = 0 corresponds to index = n-1 (middle)
        # i-th element of h * 1-h corresponds to h[i]

        # compute the gradient for the max correlated index at lag max_lag
        gradient = np.zeros_like(h)
        # say lag index in correlate is corr[n-1 + max_lag]   if max_lag >=0, it’s in range
        for i in range(n_points):
            j = i + max_lag  # This is the idx in second vector where j >= 0 and j < n
            if 0 <= j < n_points:
                # i contributes to correlation by including h[i] * (1 - h[j])  
                # gradient w.r.t h[i] is (1 - h[j])
                gradient[i] = (1 - h[j])
                
        # Perform Frank-Wolfe step:
        # Find direction (s) given the gradient (this involves selecting top 100 values)
        selected_indices = np.argsort(gradient)[::-1][:n_points//2]
        s = np.zeros(n_points)
        s[selected_indices] = 1
        
        # Ensure s satisfies sum constraint scale
        # s is binary vector of 1s at `n_points//2` indices
        # so sum(s) = n_points/2, but h should also sum to n_points/2
        # We make s compatible so that h is scale-invariant and sum correctly.
        
        # Find the descent direction; direction s is the same structure but we're not setting it by stackoverflow copy
        # Must do linear optimization along s.
        # Follow Frank-Wolfe approach: 
        # We have to perform projection of h in direction of s
        # Let h_new = (1 - gamma) * h + gamma * new_h, where new_h is the minimum norm element in feasible set 
        # such that (1 - gamma) * h + gamma * new_h = feasible  
        # This often leads to line search updates toward s.  
        # But let's recompute the exact line search for gamma
        
        # Let's try to find gamma in [0, 1] that minimizes C5  
        # i.e., minimize C5 for h = h + gamma * (s - h)
        
        best_gamma = 0.0 
        best_c5_step = best_c5
        step_list = np.array([0.05, 0.10, 0.25, 0.5, 0.75, 0.90, 1.0])
        
        for gamma in step_list:
            h_test = h * (1 - gamma) + s * gamma
            # Clip to [0,1]
            h_test = np.clip(h_test, 0, 1)
            # Re-scale to get correct sum:
            new_sum = np.sum(h_test)  
            if abs(new_sum) > 1e-10:
                h_test = h_test * (n_points/2) / new_sum
            test_c5 = evaluate_h(h_test, dx) 
            if test_c5 < best_c5_step:
                best_c5_step = test_c5
                best_gamma = gamma
        
        if best_gamma > 0:
            h = h * (1 - best_gamma) + s * best_gamma
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            if best_c5_step < best_c5:
                best_c5 = best_c5_step
                best_h = h.copy()
            
    return (best_h.tolist(), best_c5, n_points)
