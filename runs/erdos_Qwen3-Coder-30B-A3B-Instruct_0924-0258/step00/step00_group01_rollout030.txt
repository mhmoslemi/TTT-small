import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    # Compute the overlap integral C5
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Default construction: uniform distribution
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[1] = 0.6
        h[-1] = 0.4
        h[-2] = 0.6
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Project onto feasible set
    def project(h):
        h = np.clip(h, 0, 1)
        # Rescale to meet the constraint sum(h) = n_points / 2
        scale = (n_points / 2) / np.sum(h)
        h = h * scale
        return h
    
    # Use gradient descent-like method for optimization
    max_iter = 2000
    step_size = 0.01
    reg = 1e-8
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximatively using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        # Compute current overlap
        current_c5 = evaluate_overlap(h, dx)
        
        # Approximate gradient via finite difference
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project(h_plus)
            c5_plus = evaluate_overlap(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project(h_minus)
            c5_minus = evaluate_overlap(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Step in direction of negative gradient
        new_h = h - step_size * grad
        new_h = project(new_h)
        
        # Evaluate new overlap
        new_c5 = evaluate_overlap(new_h, dx)
        
        # Update if improvement
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            
        # Accept step only if improvement or small enough
        if new_c5 <= current_c5 or np.random.random() < 0.1:
            h = new_h.copy()
            
        # Slightly reduce step size over time
        step_size *= 0.9995
    
    return (best_h.tolist(), best_c5, len(best_h))
