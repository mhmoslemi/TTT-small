import numpy as np
from scipy import signal
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Evaluate C5 using np.correlate."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47  # Use the provided initial construction
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if abs(np.sum(h) - target_sum) > 1e-6:
            h = h * target_sum / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * target_sum / np.sum(h)
    
    assert abs(np.sum(h) - target_sum) < 1e-6
    assert np.all(h >= 0) and np.all(h <= 1)
    
    # Initial evaluation
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Optimization loop
    max_iter = 5000
    alpha = 0.1
    stagnation_counter = 0
    last_improvement = 0
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.99:
            break
            
        # Compute all overlaps using np.correlate
        corr = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        # Find the lag with maximum overlap
        # In full correlation index k_max, lag is k_max - (n_points - 1)
        lag = k_max - (n_points - 1)
        
        # Compute diagonals for the maximum overlapping lag
        grad = np.zeros(n_points)
        if lag >= 0:
            # h[i] * (1 - h[i+lag]) part (forward)
            grad[lag:] = 1 - h[:-lag] if lag > 0 else 1 - h
            # (1 - h[i]) * h[i+lag] part (reverse) 
            rev_grad = np.zeros(n_points)
            rev_grad[:-lag] = h[lag:] if lag > 0 else h
            grad += rev_grad
        else:
            # lag < 0: reverse case
            lag_abs = -lag
            grad[:-lag_abs] = 1 - h[lag_abs:]
            rev_grad = np.zeros(n_points)
            rev_grad[lag_abs:] = h[:-lag_abs] if lag_abs > 0 else h
            grad += rev_grad
        
        # Normalize gradient
        grad = grad / (np.max(np.abs(grad)) + 1e-10)
        
        # Update h
        h_new = h + alpha * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum constraint
        h_new_sum = np.sum(h_new)
        if abs(h_new_sum) > 1e-6:
            h_new = h_new * target_sum / h_new_sum
            
        # Check if we improved
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            last_improvement = i
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        h = h_new.copy()
        
        # Decrease learning rate
        if i - last_improvement > 50 and alpha > 1e-6:
            alpha *= 0.95
            
        # Random perturbation to escape local minima
        if np.random.rand() < 0.02 and i > 100:
            perturb = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturb, 0, 1)
            h = h * target_sum / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
