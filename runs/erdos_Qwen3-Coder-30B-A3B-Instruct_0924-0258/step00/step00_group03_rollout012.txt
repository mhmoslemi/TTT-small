import numpy as np
import time
from scipy.optimize import minimize
from math import sqrt

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    # Compute all correlations for different shifts
    correlations = np.correlate(h_values, 1 - h_values, mode="full")
    # Extract the valid ones (positive shifts)
    valid_correlations = correlations[n-1:]
    max_correlation = np.max(valid_correlations)
    return max_correlation * dx

def project(h, n_points):
    """Project h to satisfy integral constraint"""
    h = np.clip(h, 0, 1)
    scale = (n_points / 2.0) / np.sum(h)
    h = h * scale
    return h

def mutate(h, sigma):
    """Mutate by adding Gaussian noise and project"""
    h_new = h + np.random.normal(0, sigma, len(h))
    return project(h_new, len(h))

def crossover(h1, h2, alpha=0.6):
    """Blend crossover operator"""
    beta = np.random.uniform(alpha, 1 - alpha)
    h_new = beta * h1 + (1 - beta) * h2
    return h_new
