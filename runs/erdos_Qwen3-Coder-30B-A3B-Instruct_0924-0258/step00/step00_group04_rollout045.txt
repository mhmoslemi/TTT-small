import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 42  # Use the provided initial construction size
    dx = 2.0 / n_points
    max_iter = 6000  # Rough estimate for 600s budget
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        assert np.all(h >= 0) and np.all(h <= 1)
        assert abs(np.sum(h) * dx - 1.0) < 1e-10
    else:
        # Fall back to uniform initialization
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)

    best_h, best_C5 = h.copy(), float('inf')
    start_time = time.time()
    
    # Gradient descent with line search
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.99:  # Leave some buffer
            break
            
        # Compute overlaps for all lags
        # Mode full: output length = 2*n - 1
        corr_full = correlate(h, 1 - h, mode='full')
        # Extract the central part (corresponding to valid lags)
        corr_valid = corr_full[len(h)-1:len(h)-1+n_points]  # indices: 0 to n-1 for lag 0 to n-1
        # But actually we want to map this to portion where lag ranges from 0 to n-1
        # corr_full: indices [0, 1, ..., 2*n-2]
        # lag 0 is at index (n-1), lag 1 at index n, ..., lag n-1 at index 2*n-2
        # But we only care about (0 to n-1) inclusive. So that's indices [n-1, n, ..., 2*n-2]
        # We need the first n items starting from index n-1.
        # Method: shift corr_full of size 2n-1 to be of size n, starting from lag = 0
        C5_vals = corr_valid * dx  # Note the *dx in the original formulation from simulator!
        
        # C5 is the max k -> integral
        k_star = np.argmax(C5_vals)
        C5_val = C5_vals[k_star]
        
        if C5_val < best_C5:
            best_C5 = C5_val
            best_h = h.copy()

        # Compute gradient w.r.t. h
        # ∂(overlap_k) / ∂h[j] = (1 - 2 h[(j + k_star) % n])
        grad = np.zeros(n_points)
        for j in range(n_points):
            idx = (j + k_star) % n_points
            grad[j] = 1 - 2 * h[idx]
        
        # Line search
        alpha = 0.1
        step_size = alpha
        while True:
            # Temporarily update
            h_new = h - step_size * grad
            
            # Project to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Rescale to sum = n_points / 2
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new C5
            corr_full_new = correlate(h_new, 1 - h_new, mode='full')
            corr_valid_new = corr_full_new[len(h_new)-1:len(h_new)-1+n_points]
            C5_new = np.max(corr_valid_new * dx)
            
            if C5_new < C5_val or step_size < 1e-6:
                break
            step_size *= 0.8

        # Final update step
        h = h_new
    
    return (best_h.tolist(), best_C5, n_points)
