import numpy as np
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Computes the C5 bound for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts k
    correlation = np.correlate(h, 1 - h, mode="full")
    # The maximum value is in the middle of the correlation array
    max_overlap = np.max(correlation)
    return max_overlap * dx

def project_to_feasible(h, n_points):
    """Projects h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def convex_surrogate(h, n_points):
    """A convex surrogate for the maximum overlap to enable gradient-based optimization"""
    dx = 2.0 / n_points
    correlation = np.correlate(h, 1 - h, mode="full")
    # Use sum of squares as surrogate for max to make it convex
    return np.sum(correlation**2) * dx**2

def gradient_of_surrogate(h, n_points):
    """Analytical gradient of the convex surrogate"""
    dx = 2.0 / n_points
    # Compute correlation and its gradient
    h_flip = 1 - h
    correlation = np.correlate(h, h_flip, mode="full")
    
    # Compute derivative terms
    # Using the chain rule on the surrogate sum of squares
    # Gradient of sum of squares: 2 * each correlation * gradient of correlation
    # We compute a simplified approximation of the gradient
    # For practical purposes we compute as:
    # ∂/∂h[i]  sum_j (h * (1-h))[j]^2  
    # This involves convolution gradients which are complex
    # Here's a simpler gradient approach: use finite differences for robustness
    
    # Since we cannot easily compute exact gradient, we use the fact that we want 
    # to minimize max_k (h * (1 - h))[k], so let's use finite differences for the gradient
    # It's more robust to compute numerical gradient for the actual C5 estimate
    # That would require many evaluations, so for now, return what we have
    
    # If we compute the direct gradient computation for surrogate, we want:
    # ∇_h (sum_k (h * (1-h))[k]^2) dx^2
    # But since the simple form is computationally complex, we just return zeros and 
    # rely on the built-in optimizer and slight stochastic modification during optimization
    
    # For simplicity in code, just return zeros, but keep it so the caller can use

    return np.zeros_like(h)  # placeholder

def optimize_at_resolution(h0, n_points, time_start, budget_s):
    """Optimize at a given resolution level"""
    # Clip initial
    h = project_to_feasible(h0, n_points)
    
    # Low-resolution, use easier approach  
    # Use a simple gradient descent-style approach with repeated local search
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    best_c5 = best_c5
    max_iter = 1000
    
    # Initialize variables for optimization
    dx = 2.0 / n_points
    constraint_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Random perturbations to escape local minima
    iter_count = 0
    for perturb_iter in range(10):
        if time() - time_start > budget_s - 5:
            break
        # Noise to slightly perturb
        h_perturb = h + 0.05 * (np.random.rand(n_points) - 0.5)
        h_perturb = project_to_feasible(h_perturb, n_points)
        
        res = minimize(
            convex_surrogate,
            h_perturb,
            args=(n_points,),
            method='SLSQP',
            bounds=bounds,
            constraints=constraint_sum,
            options={'maxiter': 100},
        )
        
        if res.success:
            h_opt = project_to_feasible(res.x, n_points)
            c5_opt = evaluate_erdos_solution(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        
        iter_count += 1
        if iter_count > max_iter:
            break
            
    return best_h, best_c5
