import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h to satisfy sum constraint
    h = h * target_sum / np.sum(h)
    
    # Optimizer settings
    lr = 0.01
    momentum = 0.9
    v = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Compute overlap integrals for all k
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        # Only consider relevant k values (top contributors to max)
        top_k_indices = np.argsort(overlaps)[-5:][::-1]
        # Compute gradient w.r.t. h for these overlaps
        grad = np.zeros_like(h)
        for k in top_k_indices:
            # Gradient for overlap_k = sum(h[i] * (1 - h[(i+k)%n_points]))
            # ∂/∂h[i] overlap_k = 1 - 2*h[i] - h[(i-k)%n_points] (approx.)
            # Using finite differences for approximation
            grad += np.roll(1 - 2 * h - np.roll(h, k), -k) * dx
        
        # Gradient descent step with momentum
        v = momentum * v + lr * grad
        h_new = h - v
        
        # Project onto feasible set: [0,1] and sum=1
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        h = h_new
        # Adapt learning rate
        if abs(new_c5 - best_c5) < 1e-6:
            lr *= 0.95
        else:
            lr *= 1.01
        
        # Occasionally perturb for escape
        if np.random.random() < 0.01:
            h = h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            
    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    # Compute all overlaps
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlaps)
