import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, n_points):
    # Compute overlap integral for all lags
    dx = 2.0 / n_points
    # Correlate h with (1-h) using full mode
    full_corr = correlate(h, 1 - h, mode='full')
    # The maximum C5 is the maximum of the valid lags
    max_k = len(full_corr) // 2
    c5 = np.max(full_corr[max_k - n_points + 1:max_k + n_points])
    return c5 * dx

def project_to_feasible(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Moderate size for efficiency
    
    # Initialize h from initial_h_values or with uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.resize(h, n_points)
    else:
        # Initialize with uniform values and rescale
        h = np.full(n_points, 0.5)
    
    h = project_to_feasible(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points)
    start_time = time.time()
    
    # For gradient computation, we'll use a deterministic step size
    alpha = 0.1
    max_iter = 10000
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute overlap values for all lags
        dx = 2.0 / n_points
        full_corr = correlate(h, 1 - h, mode='full')
        max_k = len(full_corr) // 2
        corr_values = full_corr[max_k - n_points + 1:max_k + n_points]
        
        # Find worst case lag
        k_star = np.argmax(corr_values)
        
        # Compute gradient for lag k_star
        g = np.zeros(n_points)
        for j in range(n_points):
            # For lag k_star, gradient is 2/n_points * (1 - h[(j-k_star)%n_points] - h[(j+k_star)%n_points])
            # But let's use the exact formula from signal correlation
            # Gradient of overlap at lag k_star w.r.t. h[j]
            # z_k = sum_i h[i] * (1 - h[i+k]) for i=0,...,n_points-1
            # ∂z_k/∂h[j] = 1 - h[(j-k)%n_points] - h[(j+k)%n_points] 
            left_idx = (j - k_star) % n_points
            right_idx = (j + k_star) % n_points
            g[j] = 1 - h[left_idx] - h[right_idx]
        
        g *= 2.0 / n_points
        
        # Compute new h
        h_new = h - alpha * g
        
        # Project to feasible region
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new C5
        c5_new = evaluate_c5(h_new, n_points)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        # Sometimes we want to perturb to avoid local minima
        if iter_count % 50 == 0:
            perturbation = np.random.normal(0, 0.005, n_points)
            h = h + perturbation
            h = project_to_feasible(h, n_points)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, n_points)
    return (best_h.tolist(), final_c5, n_points)
