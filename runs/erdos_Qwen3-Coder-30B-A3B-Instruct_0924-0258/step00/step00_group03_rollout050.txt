import numpy as np
from scipy import signal
import cvxpy as cp
import time

def evaluate_h(h_values):
    h = np.array(h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations with 1-h
    corr = signal.correlate(h, 1-h, mode="full")
    # Maximum correlation (scaled by dx)
    max_corr = np.max(corr) * dx
    return max_corr

def project_onto_feasible(h, n_points):
    # Use convex optimization to project onto the feasible set:
    # 0 <= h[i] <= 1, sum(h) * dx = 1
    dx = 2.0 / n_points
    h_proj = cp.Variable(n_points)
    constraints = [h_proj >= 0, h_proj <= 1, cp.sum(h_proj) == n_points / 2]
    prob = cp.Problem(cp.Minimize(cp.sum_squares(h_proj - h)), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    return h_proj.value

        k_max = np.argmax(corr)

            h_minus[i] = np.clip(h_minus[i] - eps, 0, 1)

        h_current = h_new[:]
