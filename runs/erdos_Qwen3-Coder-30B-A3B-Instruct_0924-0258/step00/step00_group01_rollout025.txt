import numpy as np
import cvxpy as cp
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h with initial_h_values if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Fallback to uniform step function
        n_points = 100
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
    
    # Ensure integral constraint
    dx = 2.0 / n_points
    current_sum = np.sum(h) * dx
    h = h * (1.0 / current_sum) * 0.5  # Adjust so that integral is 1

    # Precompute valid pairs for all lags
    valid_pairs = {}
    for k in range(n_points):
        pairs = []
        for i in range(n_points):
            j = i + k
            if j < n_points:
                pairs.append((i, j))
        valid_pairs[k] = pairs

    best_h = h.copy()
    best_c5 = float('inf')
    
    iter_count = 0
    while time.time() - start_time < budget_s - 2 and iter_count < 100:
        iter_count += 1
        # Evaluate C5
        corr = correlate(h, 1-h, mode='full')
        c5_val = np.max(corr) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # If we're already good enough, terminate early
        if best_c5 <= 0.3807:
            break

        # Identify worst constraint (largest overlap)
        max_corr = np.max(corr)
        max_lag = np.argmax(corr)
        
        # Create optimization problem
        # Variable
        h_var = cp.Variable(n_points)
        # Constraint: h in [0, 1]
        constraints = [h_var >= 0, h_var <= 1]
        # Constraint: integral = 1
        constraints.append(cp.sum(h_var) * dx == 1.0)
        
        # Constraint for the worst lag
        lag = max_lag - (n_points - 1)
        pairs = valid_pairs[abs(lag)]
        if lag >= 0:
            # h[i] * (1 - h[i + lag]) <= C for i = 0, ..., n_points - lag -1
            for i, j in pairs:
                constraints.append(h_var[i] * (1 - h_var[j]) <= max_corr * dx)
        else:  # lag < 0
            # h[i - lag] * (1 - h[i]) <= C for i = -lag, ..., n_points - 1
            for i, j in pairs:
                constraints.append(h_var[j] * (1 - h_var[i]) <= max_corr * dx)
        # Objective
        objective = cp.Minimize(max_corr * dx)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h = h_var.value
            else:
                # If optimization fails, continue with current h
                pass
        except:
            # If problem solving fails, continue with current h
            pass
        
        # Ensure integral constraint
        current_sum = np.sum(h) * dx
        h = h * (1.0 / current_sum) * 0.5  # Normalize so integral equals 1

    # Final evaluation
    corr = correlate(best_h, 1-best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), float(final_c5), int(n_points))
