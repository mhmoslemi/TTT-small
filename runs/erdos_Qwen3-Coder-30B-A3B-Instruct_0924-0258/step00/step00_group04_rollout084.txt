import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_c5(h, n_points):
    """Evaluate the C5 constant for a given step function h."""
    # The full convolution with 1-h
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum value of the convolution
    max_val = np.max(conv)
    # Normalize by dx
    dx = 2.0 / n_points
    return max_val * dx

def project_h(h, n_points):
    """Ensure h is within [0,1] and sums to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-10:
        scale_factor = target_sum / current_sum
        h = h * scale_factor
    return h

def initialize_population(n_points, pop_size, seed):
    """Create initial population."""
    random.seed(seed)
    population = []
    
    if 'initial_h_values' in globals():
        # Use provided initial construction
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
        population.append(h)
        # Generate remaining random solutions
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = project_h(h, n_points)
            population.append(h)
    else:
        # Generate entirely random solutions
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = project_h(h, n_points)
            population.append(h)
    
    return population

def mutate(h, mutation_rate):
    """Mutate a candidate solution."""
    n_points = len(h)
    num_mutate = int(mutation_rate * n_points)
    indices = random.sample(range(n_points), num_mutate)
    for i in indices:
        h[i] += np.random.normal(0, 0.1)
    return project_h(h, n_points)

def crossover(h1, h2):
    """Perform crossover between two parents."""
    n_points = len(h1)
    num_cuts = random.randint(2, 3)
    cuts = sorted(random.sample(range(1, n_points), num_cuts))
    
    # Start with h1
    child = h1.copy()
    last_cut = 0
    swap = False
    
    for cut in cuts:
        if swap:
            child[last_cut:cut] = h2[last_cut:cut]
        swap = not swap
        last_cut = cut
    
    # Handle final segment
    if swap:
        child[last_cut:] = h2[last_cut:]
    
    return project_h(child, n_points)

def greedy_optimization(h, n_points, max_iter=100):
    """Apply gradient-based optimization to improve a solution."""
    def objective(h_flat):
        h = h_flat.reshape(-1)
        return evaluate_c5(h, n_points)
    
    def gradient(h_flat):
        h = h_flat.reshape(-1)
        # Simple finite difference gradient approximation
        grad = np.zeros_like(h_flat)
        epsilon = 1e-6
        for i in range(len(h_flat)):
            h_plus = h_flat.copy()
            h_plus[i] += epsilon
            h_minus = h_flat.copy()
            h_minus[i] -= epsilon
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * epsilon)
        return grad
    
    # Use scipy's L-BFGS-B
    h_flat = h.flatten()
    result = minimize(objective, h_flat, method='L-BFGS-B', jac=gradient,
                     bounds=[(0, 1)] * len(h_flat),
                     options={'maxiter': max_iter, 'gtol': 1e-6})
    
    return result.x.reshape(h.shape)

def run(seed=42, budget_s=600, **kwargs):
    # Control parameters
    n_points = 100
    pop_size = 50
    mutation_rate = 0.05
    max_generations = 200
    elite_ratio = 0.2
    mutation_stagnation_threshold = 50
    no_improvement_count = 0
    best_h = None
    best_c5 = float('inf')
    
    # Initialize
    start_time = time.time()
    population = initialize_population(n_points, pop_size, seed)
    
    # Store indices for better fit
    elite_size = int(elite_ratio * pop_size)
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate all candidates
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, n_points)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort indices by fitness
        sorted_indices = np.argsort(fitness)
        
        # Keep elite
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Create new population
        new_population = elite.copy()
        
        # Crossover from elite
        while len(new_population) < pop_size:
            parent1, parent2 = random.sample(elite, 2)
            child = crossover(parent1, parent2)
            new_population.append(child)
        
        # Mutate offspring
        for i in range(len(new_population) - elite_size):
            new_population[elite_size + i] = mutate(new_population[elite_size + i], mutation_rate)
        
        # Optional local optimization
        if generation % 10 == 0 and len(elite) > 0:
            for i in range(len(new_population[:elite_size])):
                try:
                    h_opt = greedy_optimization(new_population[i], n_points)
                    c5_opt = evaluate_c5(h_opt, n_points)
                    if c5_opt < fitness[i]:
                        new_population[i] = h_opt
                except:
                    pass  # If local optimization fails, keep original
        
        population = new_population
        
        # Check for stagnation
        if generation > 0 and fitness[sorted_indices[0]] < fitness[sorted_indices[1]]:
            no_improvement_count = 0
        elif generation > 0:
            no_improvement_count += 1
            
        if no_improvement_count >= mutation_stagnation_threshold:
            mutation_rate = min(0.1, mutation_rate + 0.01)
    
    # Make sure final best solution is valid
    if best_h is not None:
        best_h = project_h(best_h, n_points)
        final_c5 = evaluate_c5(best_h, n_points)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to a random solution if something went wrong
        h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.random.rand(n_points)
        h = project_h(h, n_points)
        final_c5 = evaluate_c5(h, n_points)
        return (h.tolist(), final_c5, n_points)
