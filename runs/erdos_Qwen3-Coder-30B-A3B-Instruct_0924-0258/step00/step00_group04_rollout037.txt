import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_overlap(h, dx):
    """Evaluate the overlap integral for given h"""
    n = len(h)
    # Compute convolution of h and (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Take the maximum value and scale by dx
    max_corr = np.max(corr)
    return max_corr * dx

def project_and_rescale(h, n_points):
    """Project h to [0,1] and rescale to sum to n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.full(n_points, 1.0 / n_points)
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    random.seed(seed)
    
    # Problem parameters
    n_points = 42  # Use the initial construction size if available
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_and_rescale(h, n_points)
    else:
        h = np.full(n_points, 0.5)
        h = project_and_rescale(h, n_points)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Simulated Annealing parameters
    temperature = 1.0
    cooling_factor = 0.99
    min_temp = 0.01
    patience = 50
    patience_counter = 0
    gradient_scale = 0.1
    noise_std = 0.05
    
    start_time = time.time()
    
    for iteration in range(100000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Store current
        current_h = h.copy()
        current_c5 = evaluate_overlap(h, dx)
        
        # Create candidate using gradient guidance with random noise
        # This is a heuristic approximation of gradient direction
        h_candidate = h.copy()
        
        # Add random noise scaled by current gradient effect
        noise = np.random.normal(0, noise_std, n_points)
        h_candidate += noise * gradient_scale
        
        # Project and rescale
        h_candidate = project_and_rescale(h_candidate, n_points)
        
        # Evaluate
        new_c5 = evaluate_overlap(h_candidate, dx)
        
        # Accept or reject the move
        if new_c5 < current_c5 or \
           np.exp((current_c5 - new_c5) / max(temperature, 1e-10)) > random.random():
            h = h_candidate
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                patience_counter = 0
        else:
            patience_counter += 1
            
        # Adjust temperature
        temperature = max(temperature * cooling_factor, min_temp)
        
        # Restart temperature if no improvement
        if patience_counter > patience:
            temperature = 1.0
            patience_counter = 0
            
    return (best_h.tolist(), best_c5, n_points)
