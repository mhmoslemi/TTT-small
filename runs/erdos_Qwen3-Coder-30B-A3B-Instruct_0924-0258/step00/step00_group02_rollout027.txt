import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def compute_gradient(h_vec, k_max):
        # The gradient is the derivative of the correlation w.r.t. h
        # For shift k_max, we compute gradient of (h * (1-h))[k_max]
        # Gradient is: 1 - 2 * h (shifted by k_max)
        # But we're dealing with correlation directly
        # This is a simplified gradient update
        # See that correlation gradient for h[i]: 
        # For term h[i] * (1 - h[i + k_max]) -> 1 - 2*h[i] for i, and -h[i-k_max] for i+k_max
        # Simplified: чуть проще делать NUMERICALLY incremental
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        k_max_actual = np.argmax(corr)
        # Compute finite diff gradient approximately
        eps = 1e-4
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_test = h_vec.copy()
            h_test[i] += eps
            h_test = project_h(h_test)
            corr_test = np.correlate(h_test, 1 - h_test, mode='full')
            grad[i] = (corr_test[k_max_actual] - corr[k_max_actual]) / eps
        return grad
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    max_iter = 5000
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute current overlap
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        c5_current = corr[k_max] * dx
        
        # Compute gradient (simplified gradient descent approach)
        # For better convergence we just perform a small gradient step
        eps = 1e-5
        grad = np.zeros(len(h))
        corr_base = np.correlate(h, 1 - h, mode='full')
        k_max_base = np.argmax(corr_base)
        base_val = corr_base[k_max_base]
        
        for i in range(len(h)):
            h_test = h.copy()
            h_test[i] += eps
            h_test = project_h(h_test)
            corr_test = np.correlate(h_test, 1 - h_test, mode='full')
            grad[i] = (corr_test[k_max_base] - base_val) / eps
        
        # Gradient update
        h = h - lr * grad
        h = project_h(h)
        
        # Compute new C5 and update best
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        # Anneal learning rate
        lr *= 0.999
        
        # Occasionally add noise
        if iteration % 100 == 0:
            noise = 0.01 * np.random.uniform(-1, 1, len(h))
            h = h + noise
            h = project_h(h)
            
        iteration += 1
        
        # Early termination if not improving much
        if iteration > max_iter:
            break
    
    return (best_h.tolist(), best_c5, n_points)
