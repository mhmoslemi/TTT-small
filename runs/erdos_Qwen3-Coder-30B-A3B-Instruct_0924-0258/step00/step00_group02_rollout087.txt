import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    overlap = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def crossover(parent1, parent2):
    n = len(parent1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, rate=0.1):
    h = h.copy()
    n = len(h)
    n_mutate = int(rate * n)
    indices = np.random.choice(n, n_mutate, replace=False)
    for i in indices:
        h[i] += 0.1 * np.random.randn()
    return project_to_feasible(h, len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    n_points = 45
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.ones(n_points) * 0.5
    
    population_size = 50
    population = [initial_h]
    for _ in range(population_size - 1):
        perturbed = initial_h + 0.1 * np.random.randn(n_points)
        population.append(project_to_feasible(perturbed, n_points))
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Evolution loop
    generations = 12
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness.append((c5, h))
        
        # Sort by fitness
        fitness.sort(key=lambda x: x[0])
        best_in_gen = fitness[0][0]
        if best_in_gen < best_c5:
            best_c5 = best_in_gen
            best_h = fitness[0][1].copy()
        
        # Select top 50%
        selected = [h for _, h in fitness[:population_size//2]]
        
        # Generate offspring
        offspring = []
        while len(offspring) < population_size - len(selected):
            if time.time() - start_time > budget_s - 1:
                break
            p1, p2 = np.random.choice(selected, 2, replace=False)
            c1, c2 = crossover(p1, p2)
            c1 = mutate(c1)
            c2 = mutate(c2)
            offspring.extend([c1, c2])
        
        # Replace population
        population = selected + offspring[:population_size - len(selected)]
    
    return (best_h.tolist(), best_c5, n_points)
