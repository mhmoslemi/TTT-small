import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a feasible starting point
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        # Normalize to sum to n_points / 2
        h *= (n_points / 2) / np.sum(h)
    
    # dx for discretization
    dx = 2.0 / n_points
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent parameters
    perturb_steps = 20
    step_size_candidates = [0.1, 0.5, 1.0]
    
    # Optimization loop
    step_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 0.1:
        step_count += 1
        
        # Compute all cross-correlations
        corr = correlate(h, 1 - h, mode='full')
        # The maximum lag is at the center of the correlation
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(corr) - 1) // 2
        
        # Compute gradient for the maximum lag
        # h[i] contributes to the correlation with 1 - h[i + max_lag] (when max_lag > 0)
        # or with 1 - h[i - max_lag] (when max_lag < 0)
        grad = np.zeros(n_points)
        if max_lag > 0:
            # For lag k = max_lag, d/dh[i] cross_correlation = 1 - h[i + k]
            # So gradient is 1 - h[i + k] for i < n - k
            for i in range(n_points - max_lag):
                grad[i] = 1 - h[i + max_lag]
        elif max_lag < 0:
            # For lag k = max_lag, d/dh[i] cross_correlation = 1 - h[i - k]
            # So gradient is 1 - h[i - k] for i >= -k
            for i in range(-max_lag, n_points):
                grad[i] = 1 - h[i + max_lag]
        else:
            # For lag 0, max_lag = 0: gradient is all 1's
            grad.fill(1)
            
        # Try different step sizes
        best_h_new = h.copy()
        best_new_c5 = best_c5
        improved = False
        for alpha in step_size_candidates:
            # Compute new h
            h_new = h - alpha * grad
            # Clip to [0, 1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to sum to n_points / 2
            if np.sum(h_new) > 0:
                h_new = h_new * (n_points / 2) / np.sum(h_new)
            # Compute new C5
            h_new_c5 = compute_c5(h_new, dx)
            if h_new_c5 < best_new_c5:
                best_new_c5 = h_new_c5
                best_h_new = h_new.copy()
                improved = True
        
        if improved:
            h = best_h_new.copy()
            if best_new_c5 < best_c5:
                best_c5 = best_new_c5
                best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Perturb if no improvement
        if no_improvement_count > perturb_steps:
            h = h + 0.01 * np.random.randn(n_points)
            # Clip
            h = np.clip(h, 0, 1)
            # Normalize
            if np.sum(h) > 0:
                h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
            
        # Check time limit
        if time.time() - start_time >= budget_s - 0.1:
            break
    
    # Final validation
    assert np.all(h >= 0) and np.all(h <= 1), "h not in [0,1]"
    assert abs(np.sum(h) - n_points / 2) < 1e-10, "Sum constraint not met"
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def compute_c5(h, dx):
    """Compute C5 = max_k int h(x)(1 - h(x+k)) dx"""
    # In numpy this is equivalent to computing correlate(h, 1-h, mode='full')
    # and taking the maximum
    corr = correlate(h, 1 - h, mode='full')
    # Maximum value scaled by dx (area under corr = integral)
    return np.max(corr) * dx
