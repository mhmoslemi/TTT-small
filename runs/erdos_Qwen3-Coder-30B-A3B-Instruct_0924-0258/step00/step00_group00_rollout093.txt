import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
        n_points = len(h)
    else:
        # Binary-like construction
        n_points = 45
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
    
    # Ensure sum is correct
    h = h * (n_points / 2) / np.sum(h)
    
    # Normalize dx
    dx = 2.0 / n_points
    
    def evaluate(h_vec):
        # Full correlation: h * (1 - h)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(corr) * dx
        return max_val
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to ensure sum = n_points/2
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    def compute_gradient(h_vec, max_k):
        # Compute gradient for correlation with lag k
        grad = np.zeros_like(h_vec)
        for k in max_k:
            for i in range(len(h_vec) - k):
                if i < len(h_vec):
                    grad[i] += (1 - h_vec[i + k])
                if i + k < len(h_vec):
                    grad[i + k] -= h_vec[i]
        return grad
    
    best_c5 = evaluate(h)
    best_h = h.copy()
    
    iterations_without_improvement = 0
    max_iterations_without_improvement = 20
    step_sizes = [0.1, 0.25, 0.5]
    
    while time.time() - start_time < budget_s - 0.1:
        # Compute all cross-correlations
        corr = correlate(h, 1 - h, mode='full')
        all_correlations = corr * dx
        
        # Find maximum value and corresponding lags
        max_val = np.max(all_correlations)
        max_indices = np.where(all_correlations == max_val)[0]
        
        # Compute gradient based on max indices
        gradient = np.zeros_like(h)
        for k in max_indices:
            if k < len(h):  # Only consider valid k values
                for i in range(len(h) - k):
                    # Gradient w.r.t. h[i]
                    gradient[i] += (1 - h[i + k])
                    # Gradient w.r.t. h[i + k]
                    if i + k < len(h):
                        gradient[i + k] -= h[i]
        
        # Ensure gradient is not zero
        if np.linalg.norm(gradient) < 1e-12:
            # Perturb
            h = project_feasible(h + 0.01 * np.random.randn(len(h)))
            iterations_without_improvement += 1
            continue
            
        # Normalize gradient
        gradient = gradient / (np.linalg.norm(gradient) + 1e-12)
        
        # Line search
        best_h_candidate = h.copy()
        best_c5_candidate = best_c5
        
        for alpha in step_sizes:
            h_candidate = h - alpha * gradient
            h_candidate = project_feasible(h_candidate)
            c5_candidate = evaluate(h_candidate)
            
            if c5_candidate < best_c5_candidate:
                best_c5_candidate = c5_candidate
                best_h_candidate = h_candidate.copy()
        
        # Update
        if best_c5_candidate < best_c5:
            h = best_h_candidate.copy()
            best_c5 = best_c5_candidate
            iterations_without_improvement = 0
        else:
            iterations_without_improvement += 1
            
        # Perturbation escape
        if iterations_without_improvement >= max_iterations_without_improvement:
            h = project_feasible(h + 0.01 * np.random.randn(len(h)))
            iterations_without_improvement = 0
            
        # Early termination check
        if best_c5 < 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate(h)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
        
    return (best_h.tolist(), float(best_c5), int(n_points))
