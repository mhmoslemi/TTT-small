import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    dx = 2.0 / len(h)
    correlations = correlate(h, 1-h, mode="full") * dx
    return np.max(correlations)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Fallback initialization
        n_points = 70
        h = np.full(n_points, 0.5)
        
    # Ensure sum constraint
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum

    # Parameters
    max_iter = 500
    alpha = 0.1
    tol = 1e-4
    dx = 2.0 / n_points
    
    start_time = time.time()
    
    # Precompute full correlation indices
    full_len = 2 * n_points - 1
    full_indices = np.arange(full_len)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations
        correlations = correlate(h, 1 - h, mode="full") * dx
        max_idx = np.argmax(correlations)
        max_val = correlations[max_idx]
        
        # Gradient computation
        # Only the terms involving the maximum correlation contribute
        # For the maximum overlap, we consider positions where max occurs
        
        # Gradient estimate (only approximate, but works empirically)
        g = np.zeros(n_points)
        # Approximate gradient: for each position, count how many times h[i] contributes
        # to the maximum correlation term
        
        # A simpler heuristic approach:
        # We compute the gradient from the correlation structure
        # The gradient for the maximum is approximated by the partial derivatives
        
        # More accurate gradient: compute how much h[i] contributes to all 
        # correlations involving it, then weight by which is max
        
        # Build gradient strictly for the max
        # For maximum at index max_idx, we have `max_idx` as a lag
        # This means the correlation is due to h[i] and h[i + max_idx - n_points + 1]
        # let's iterate through all positions and see how they contribute
        
        temp_g = np.zeros_like(h)
        
        # Case 1: when lag = max_idx - n_points + 1
        # corr[i] = sum_{j} h[i+j] * (1 - h[j]) for some j
        # so g[i] = 1 - h[i + lag] for some lags
        # 
        # Using continuity of correlation derivative with respect to h[i]
        # if C = sum h[i] (1 - h[i + k]), then dC/dh[i] = 1 - h[i + k]  (if valid position)
        
        # Effective way: compute the gradient 1 by one using the maximum lag
        lag = max_idx - n_points + 1  # lag goes from -(n_points-1) to (n_points-1)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                temp_g[i] += (1 - h[j]) * dx
        
        # Total max estimated gradient
        g = temp_g
        
        # Update h
        h_new = h - alpha * g
        
        # Project: 
        # 1. Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # 2. Normalize to sum to n_points/2
        h_sum_new = np.sum(h_new)
        h_new = h_new * (n_points / 2) / h_sum_new
        
        h = h_new
        
        # Check convergence
        new_val = evaluate_erdos_solution(h)
        if abs(new_val - max_val) < tol:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h)
    
    return (h.tolist(), final_c5, len(h))
