import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    # Compute correlation integral using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Take only the relevant part (k=0 to n-1)
    c5_vals = corr[n-1:] * dx
    return np.max(c5_vals)

def project_and_rescale(h, n_points):
    """Project h to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def crossover(parent1, parent2, n_points):
    """Perform crossover of two parent step functions."""
    # Choose 2-3 cut points
    n_cuts = random.randint(2, 3)
    cut_points = sorted(random.sample(range(1, n_points), n_cuts))
    
    child = parent1.copy()
    start = 0
    use_first = True
    for cut in cut_points + [n_points]:
        if use_first:
            child[start:cut] = parent2[start:cut]
        use_first = not use_first
        start = cut
    return child

def mutate(h, n_points, mutation_rate=0.05):
    """Mutate step function with given probability."""
    h = h.copy()
    num_mutate = int(n_points * mutation_rate)
    indices = random.sample(range(n_points), num_mutate)
    for i in indices:
        h[i] += np.random.normal(0, 0.1)
    return project_and_rescale(h, n_points)

    return h_trial
