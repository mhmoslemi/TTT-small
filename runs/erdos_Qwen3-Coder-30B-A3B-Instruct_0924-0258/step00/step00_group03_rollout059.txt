import numpy as np
import time
from scipy import signal

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for a given h."""
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Convert to integral value
    max_overlap = np.max(corr) * dx
    return max_overlap

def ensure_feasible(h, dx):
    """Ensure h is in [0,1] and sum(h)*dx = 1."""
    h = np.clip(h, 0, 1)
    total = np.sum(h) * dx
    if total != 1.0:
        h = h * (1.0 / total)
    return h

def mutate_h(h, dx, rate=0.1):
    """Mutate h by adding noise and re-normalizing."""
    noise = np.random.normal(0, rate * 0.1, len(h))
    h_new = h + noise
    h_new = ensure_feasible(h_new, dx)
    return h_new

def crossover_h(h1, h2):
    """Uniform crossover of two h vectors."""
    mask = np.random.rand(len(h1)) > 0.5
    h_new = np.where(mask, h1, h2)
    return h_new

def tournament_selection(population, fitnesses, k=5):
    """Select the best individual from a tournament."""
    idxs = np.random.choice(len(population), size=k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    max_generations = 1000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(ensure_feasible(np.array(initial_h_values), dx))
    else:
        population.append(ensure_feasible(np.random.rand(n_points), dx))
    
    # Add random individuals to start
    for _ in range(population_size - len(population)):
        population.append(ensure_feasible(np.random.rand(n_points), dx))
    
    best_fitness = float('inf')
    best_h = None
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute fitness for entire population
        fitnesses = []
        for ind in population:
            fit = evaluate_overlap(ind, dx)
            fitnesses.append(fit)
        
        # Track best solution
        min_fit_idx = np.argmin(fitnesses)
        if fitnesses[min_fit_idx] < best_fitness:
            best_fitness = fitnesses[min_fit_idx]
            best_h = population[min_fit_idx].copy()
            
        # Selection + reproduction
        new_population = []
        for _ in range(population_size):
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.rand() < 0.8:
                offspring = crossover_h(parent1, parent2)
            else:
                offspring = parent1
            
            # Mutation
            offspring = mutate_h(offspring, dx, rate=0.1)
            
            new_population.append(offspring)
            
        population = new_population
    
    if best_h is None:
        # Fallback to initial if no improvement
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
        else:
            best_h = np.random.rand(n_points)
        best_h = ensure_feasible(best_h, dx)
        best_fitness = evaluate_overlap(best_h, dx)
        
    return (best_h.tolist(), best_fitness, n_points)
