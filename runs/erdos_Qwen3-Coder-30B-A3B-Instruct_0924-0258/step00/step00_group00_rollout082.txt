import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5  # initial guess
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # discretization
    dx = 2.0 / n_points
    
    # Frank-Wolfe algorithm
    max_iter = 500
    tolerance = 1e-4
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # compute cross-correlations
        c = np.correlate(h, 1 - h, mode='full')
        c = c * dx
        k_max = np.argmax(c)
        max_corr = c[k_max]
        
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
        
        # compute subgradient for the maximized correlation
        if k_max < n_points:
            # when k = 0 to n_points-1, we look at the case h[i] * (1 - h[i+k])
            grad = np.zeros(n_points)
            for i in range(n_points):
                k = k_max - i + n_points - 1  # aligning with positive indices in full correlation
                if k >= 0 and k < n_points:
                    grad[i] = 1.0 - h[i+k]  # gradient of h[i] * (1 - h[i+k]) w.r.t. h[i]
        else:
            # otherwise, adjust the indexing
            grad = np.zeros(n_points)
            for i in range(n_points):
                k = k_max - i + n_points - 1  # same as above
                if k >= 0 and k < n_points:
                    grad[i] = 1.0 - h[i+k]
        
        # sort by gradient and select top S points
        S = n_points // 2
        indices = np.argsort(-grad)  # descending order
        s_indices = indices[:S]
        
        # line search in direction of gradient
        gamma_values = [0.01, 0.05, 0.1, 0.25, 0.5, 0.75, 1.0]
        best_gamma = 0.0
        temp_h = h.copy()
        local_min_c5 = max_corr
        
        for gamma in gamma_values:
            trial_h = h.copy()
            trial_h[s_indices] += gamma * (1.0 - h[s_indices])
            trial_h = np.clip(trial_h, 0.0, 1.0)
            # rescale to enforce sum = n_points/2
            trial_h = trial_h * (n_points / 2) / np.sum(trial_h)
            
            trial_c = np.correlate(trial_h, 1 - trial_h, mode='full') * dx
            trial_max = np.max(trial_c)
            
            if trial_max < local_min_c5:
                local_min_c5 = trial_max
                best_gamma = gamma
                temp_h = trial_h.copy()
        
        h = temp_h.copy()
        
        # early stopping
        if (iteration > 10) and (abs(max_corr - local_min_c5) < tolerance):
            break
    
    # final evaluation
    final_c = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(final_c)
    
    return (best_h.tolist(), final_c5, n_points)
