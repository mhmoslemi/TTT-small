import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate C5 bound for given h"""
    # Compute cross-correlation: h * (1 - h_shifted)
    corr = correlate(h, 1 - h, mode='full')
    # The maximum value represents the worst-case overlap
    max_corr = np.max(corr)
    return max_corr * dx

def rescale_h(h, target_sum):
    """Rescale h to have sum = target_sum, keeping values in [0,1]"""
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.full_like(h, 1.0 / len(h))
    scale = target_sum / current_sum
    h_scaled = h * scale
    # Clip to [0,1]
    h_clipped = np.clip(h_scaled, 0, 1)
    # Rescale again to meet target sum
    final_sum = np.sum(h_clipped)
    if final_sum > 0:
        h_final = h_clipped * (target_sum / final_sum)
    else:
        h_final = np.full_like(h_clipped, target_sum / len(h_clipped))
    return h_final

def init_particles(n_points, n_particles, initial_h=None):
    """Initialize swarm particles"""
    if initial_h is not None:
        # Use initial solution if available
        particles = [initial_h.copy()]
        particles.extend([initial_h.copy() + 0.05 * (np.random.rand(n_points) - 0.5) 
                         for _ in range(n_particles - 1)])
    else:
        # Generate random starting particles
        particles = [np.random.rand(n_points) for _ in range(n_particles)]
    
    # Rescale each particle to have correct sum
    target_sum = n_points / 2.0
    for i in range(len(particles)):
        particles[i] = rescale_h(particles[i], target_sum)
    return particles

def update_velocity(swarm_pos, swarm_vel, pbest_pos, gbest_pos, 
                    inertia_w=0.7, c1=2.0, c2=2.0, max_vel=0.5):
    """Update particle velocities according to PSO"""
    n_points = len(swarm_pos[0])
    new_vel = []
    for i in range(len(swarm_pos)):
        r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
        vel = (inertia_w * swarm_vel[i] + 
               c1 * r1 * (pbest_pos[i] - swarm_pos[i]) + 
               c2 * r2 * (gbest_pos - swarm_pos[i]))
        # Clip velocity to prevent stagnation
        vel = np.clip(vel, -max_vel, max_vel)
        new_vel.append(vel)
    return new_vel

def optimize_h(n_points, n_particles, max_iter, initial_h, time_budget):
    """Optimize h using PSO algorithm"""
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    
    # Initialize particles
    particles = init_particles(n_points, n_particles, initial_h)
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Track best solutions
    pbest = [p.copy() for p in particles]
    pbest_scores = [evaluate_h(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # Evolve
    iteration = 0
    while iteration < max_iter:
        if time.time() - start_time > time_budget - 2:
            break
            
        for i in range(n_particles):
            # Update particle position
            particles[i] += velocities[i]
            
            # Ensure values are in [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            # Project back to sum constraint
            particles[i] = rescale_h(particles[i], target_sum)
            
            # Evaluate new solution
            score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                
                # Update global best if needed
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    
        # Update velocities
        velocities = update_velocity(particles, velocities, pbest, gbest)
        
        iteration += 1
        
    return gbest, gbest_score

    # PSO parameters
    n_particles = 20
    max_iter = 500
    
    # Run optimization
    best_h, best_c5 = optimize_h(n_points, n_particles, max_iter, initial_h, budget_s)
    
    # Verify constraints
    assert 0 <= np.min(best_h) <= 1, "h must be in [0,1]"
    assert 0 <= np.max(best_h) <= 1, "h must be in [0,1]"
    dx = 2.0 / n_points
    assert abs(np.sum(best_h) * dx - 1.0) < 1e-8, "Sum constraint violated"
    
    return (best_h.tolist(), float(best_c5), n_points)
