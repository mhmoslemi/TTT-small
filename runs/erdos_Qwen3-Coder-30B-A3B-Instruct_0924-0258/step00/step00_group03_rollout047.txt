import numpy as np
import time
from scipy import signal
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        # Initialize with a simple construction
        h = np.ones(n_points) * 0.5
        h[0] = 0.1
        h[-1] = 0.1
        h = h / np.sum(h) * target_sum
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * target_sum
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try small perturbations
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.01)
        h_new[idx] += delta
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * target_sum
        
        # Evaluate
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Accept with small probability to escape local minima
            if np.random.random() < 0.01:
                h = h_new.copy()
    
    # Final refinement using scipy optimization
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * target_sum
        return evaluate_c5(x, dx)
    
    # Use a subset of points for optimization to save time
    subset_size = min(30, n_points)
    indices = np.random.choice(n_points, subset_size, replace=False)
    x0 = h[indices]
    
    result = minimize(objective, x0, method='L-BFGS-B', 
                     bounds=[(0, 1) for _ in range(subset_size)],
                     options={'maxiter': 500})
    
    if result.success:
        refined = h.copy()
        refined[indices] = np.clip(result.x, 0, 1)
        refined = refined / np.sum(refined) * target_sum
        c5_refined = evaluate_c5(refined, dx)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = refined.copy()
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute overlap integral
    h_array = np.array(h)
    # The overlap integral is max_k integral h(x)(1-h(x+k)) dx
    # Using correlation: h * (1-h) with full mode
    conv = signal.correlate(h_array, 1 - h_array, mode='full')
    # Normalize by dx
    conv = conv * dx
    # Find maximum
    max_val = np.max(conv)
    return max_val
