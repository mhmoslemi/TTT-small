import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)
        h[0] = 0.4
        h[1] = 0.6
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    max_iter = 500
    alpha = 0.1
    tol = 1e-4
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find contributing lags for gradient computation
        max_lags = np.where(corr == np.max(corr))[0]
        if len(max_lags) == 0:
            max_lags = [len(corr) // 2]
            
        # Initialize gradient
        grad = np.zeros(n_points)
        
        # Compute gradient from all contributing lags
        for lag in max_lags:
            k = lag - (n_points - 1)  # actual lag in range [-n+1, n-1]
            # Gradient contribution from the specific lag
            for i in range(n_points):
                j = i - k
                if 0 <= j < n_points:
                    grad[i] += dx
                    
        # Project gradient descent step
        h_new = h - alpha * grad
        
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Re-scale to satisfy sum constraint
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (n_points / 2) / current_sum
        
        # Update h
        h = h_new
        
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, len(best_h))
