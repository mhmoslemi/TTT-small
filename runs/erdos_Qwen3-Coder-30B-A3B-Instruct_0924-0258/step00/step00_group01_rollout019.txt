    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initial evaluation
    dx = 2.0 / n_points
    overlaps = np.correlate(h, 1 - h, mode="full")
    c5_bound = np.max(overlaps) * dx
    
    best_h = h.copy()
    best_c5 = c5_bound
    
    # Main optimization loop
    alpha = 0.1
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        # Compute overlaps at all lags
        overlaps = np.correlate(h, 1 - h, mode="full")
        c5_current = np.max(overlaps) * dx
        
        # Find max overlap lag
        k_max = np.argmax(overlaps)
        
        # Compute pseudo-gradient
        grad = np.zeros(n_points)
        # Forward differences (same as convolution)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] += (1 - h[j])
            grad[j] -= h[i]
        
        # Apply gradient with clipping
        grad = np.clip(grad, -1, 1)
        h += alpha * grad
        
        # Project to feasible set
        h = np.clip(h, 0, 1)
        
        # Maintain sum constraint
        h *= target_sum / np.sum(h)
        
        # Check for improvement
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Adaptive learning rate
        if no_improvement_count > 50:
            alpha *= 0.95
            no_improvement_count = 0
        
        # Occasionally perturb to avoid local minima
        if np.random.random() < 0.05:
            perturbation = np.random.normal(0, 0.01, n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            h *= target_sum / np.sum(h)
            
        # Break early if no improvement for a while
        if no_improvement_count > max_no_improvement:
            break
            
    return (best_h.tolist(), best_c5, n_points)
