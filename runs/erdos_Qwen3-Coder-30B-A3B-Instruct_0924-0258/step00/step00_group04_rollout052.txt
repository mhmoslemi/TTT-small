import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_c5(h):
    """Evaluate the maximum overlap integral C_5 for step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Use np.correlate with mode='full' to compute all lags
    correlation = np.correlate(h, 1 - h, mode="full")
    # The valid correlations are the ones that don't involve padding
    # The center corresponds to lag 0, and we have n_points - 1 lags on each side
    # We only consider the valid correlations: from index n_points-1 to 2*n_points-2
    c5 = np.max(correlation[n_points-1:2*n_points-1]) * dx
    return c5

def prepare_initial_h(n_points):
    """Prepare an initial h function."""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            raise ValueError(f"initial_h_values has length {len(h)}, expected {n_points}")
        # Check constraints
        assert np.all(h >= 0) and np.all(h <= 1), "h must be in [0,1]"
        assert abs(np.sum(h) - n_points / 2) < 1e-10, "sum(h) must be n_points/2"
        return h.copy()
    else:
        # Default initialization
        h = np.full(n_points, 0.5)
        h[0] = 0.5 + (n_points / 2 - np.sum(h)) / n_points
        return h.copy()

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points / 2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        # Rescale to meet the sum constraint
        h = h * (n_points / 2) / current_sum
        h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, dx=2.0/42, n_points=42):
    """Estimate gradient of C_5 w.r.t. h using finite differences."""
    eps = 1e-6
    grad = np.zeros_like(h)
    base_c5 = evaluate_c5(h)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def optimize_step(h, n_points, dx=2.0/42):
    """Perform a local optimization step."""
    def objective(h_vec):
        h = h_vec.reshape(-1)
        return evaluate_c5(h)
    
    def gradient(h_vec):
        h = h_vec.reshape(-1)
        return compute_gradient(h, dx, n_points).flatten()
    
    # Ensure h is within valid range and sum constraint
    h = project_h(h, n_points)
    
    # Use L-BFGS-B for optimization
    result = minimize(objective, h, method='L-BFGS-B', jac=gradient, bounds=[(0,1)]*n_points, options={'maxiter': 20})
    h_opt = result.x.reshape(-1)
    return h_opt

def run(seed=42, budget_s=600, n_points=42):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    h = prepare_initial_h(n_points)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1.0 and iteration < 300:
        iteration += 1
        
        # Local optimization step
        h_opt = optimize_step(h, n_points)
        h_opt = project_h(h_opt, n_points)
        
        # Low-pass filter and rescale to maintain sum constraint
        h_filtered = gaussian_filter1d(h_opt, sigma=0.1 * n_points)
        h_filtered = project_h(h_filtered, n_points)
        
        # Evaluate new solution
        c5 = evaluate_c5(h_filtered)
        
        if c5 < best_c5:
            best_h = h_filtered.copy()
            best_c5 = c5
        
        h = h_filtered
        
    return (best_h.tolist(), best_c5, n_points)
