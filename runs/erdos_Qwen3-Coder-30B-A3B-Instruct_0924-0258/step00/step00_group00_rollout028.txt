import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral C5 for a given step function h."""
    # Compute cross-correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Extract the valid correlation values (not padding)
    c5 = np.max(corr) * dx
    return c5

def rescale_h(h, dx):
    """Ensure the integral constraint is satisfied."""
    target_sum = 1.0 / dx
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    return h
