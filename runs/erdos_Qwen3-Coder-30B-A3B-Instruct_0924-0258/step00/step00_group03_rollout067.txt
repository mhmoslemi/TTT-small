import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h):
    """Evaluate the maximum overlap for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for different lags
    corr = correlate(h, 1 - h, mode="full")
    # Only take the relevant part of the correlation (upper half)
    corr = corr[n-1:]  # Start from lag 0 onwards to n-1
    # Scale by dx
    corr *= dx
    # Return maximum value
    return np.max(corr)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initialize from the imported initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a simple initial guess: a step function
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Make sure sum is right
        h *= 35.0 / np.sum(h)
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Validate initial conditions
    assert np.isclose(np.sum(h) * dx, 1.0), "Initial h does not integrate to 1"
    assert np.all(h >= 0) and np.all(h <= 1), "Initial h out of bounds"
    
    # Initialize best solution
    best_h = h.copy()
    best_C5 = evaluate_C5(best_h)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.999
    sigma0 = 0.1
    t = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        t += 1
        # Update temperature
        T = T0 * (alpha ** t)
        
        # Update mutation std
        sigma = sigma0 * (alpha ** (t / 200))
        
        # Generate candidate
        h_new = h + np.random.normal(0, sigma, n_points)
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project back to satisfy constraint
        h_new *= (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        C5_new = evaluate_C5(h_new)
        
        # Accept or reject
        if C5_new < best_C5 or np.random.rand() < np.exp(-(C5_new - best_C5) / T):
            h = h_new
            if C5_new < best_C5:
                best_h = h.copy()
                best_C5 = C5_new
                
    return (best_h.tolist(), best_C5, n_points)
