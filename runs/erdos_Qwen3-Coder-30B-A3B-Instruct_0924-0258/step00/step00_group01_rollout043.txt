    assert np.abs(np.sum(h) * dx - 1.0) < 1e-6
    assert np.all(h >= 0) and np.all(h <= 1)
    
    def compute_C5(h):
        # Using scipy correlate for accuracy
        corr = signal.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        # Correct sum to exactly n_points / 2
        h_sum = np.sum(h)
        desired_sum = n_points / 2
        if abs(h_sum - desired_sum) > 1e-6:
            scale = desired_sum / h_sum
            h = h * scale
        return h
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Set initial learning rate
    alpha = 0.1
    last_improvement = 0
    iteration = 0
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        old_C5 = best_C5
        
        # Compute overlaps and find worst case
        corr = signal.correlate(h, 1 - h, mode='full')

        max_overlap = corr[k_max] * dx
        
        # Find the lag that produces this maximum overlap
        lag = k_max - (n_points - 1)
        if lag < 0:
            lag += n_points
        
        # Compute approximate gradient using finite differences
        # We will modify h at positions where the overlap is largest
        h_grad = np.zeros_like(h)
        for i in range(n_points):
            # Forward difference in the overlap
            h_new = h.copy()
            h_new[i] += 1e-4
            h_new = project_h(h_new)
            overlap_new = compute_C5(h_new)
            h_grad[i] = (overlap_new - max_overlap) / 1e-4
            
        # Apply gradient step
        h += alpha * h_grad
        h = project_h(h)
        
        # Update step size
        C5_new = compute_C5(h)
        if C5_new < best_C5:
            best_C5 = C5_new
            best_h = h.copy()
            last_improvement = iteration

    # Final check and polishing
    final_C5 = compute_C5(best_h)
