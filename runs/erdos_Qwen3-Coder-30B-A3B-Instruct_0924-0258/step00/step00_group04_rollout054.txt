import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_c5(h):
    """Evaluate the maximum overlap integral for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract overlapping parts for k = 0 to n-1
    c5_vals = corr[n-1:] * dx
    return np.max(c5_vals)

def prepare_initial_h(n_points):
    """Prepare initial h values"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            raise ValueError("initial_h_values has wrong length")
        if not (np.all(h >= 0) and np.all(h <= 1)):
            raise ValueError("initial_h_values not in [0,1]")
        if abs(np.sum(h) - n_points/2.0) > 1e-10:
            raise ValueError("initial_h_values does not satisfy integral constraint")
        return h.copy()
    else:
        # Default initialization
        h = np.full(n_points, 0.5)
        return h

def optimize_step(h, n_points, dx):
    """Optimize h to reduce C5 using local gradient descent with constraints"""
    def obj(h_vec):
        h_copy = h_vec.copy()
        h_copy = np.clip(h_copy, 0, 1)
        # Rescale to maintain integral constraint
        h_copy = h_copy * (n_points / 2.0) / np.sum(h_copy)
        return evaluate_c5(h_copy)
    
    # For gradient estimation, use finite differences
    def grad(h_vec):
        eps = 1e-6
        grad_vals = np.zeros_like(h_vec)
        base_val = obj(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            grad_vals[i] = (obj(h_plus) - base_val) / eps
        return grad_vals
    
    # Create bounds
    bounds = [(0, 1)] * n_points
    
    # Optimize with bounds
    result = minimize(
        obj,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        jac=grad,
        options={'maxiter': 100}
    )
    
    if result.success:
        h_opt = result.x.copy()
    else:
        h_opt = h.copy()
    
    # Ensure constraints
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2.0) / np.sum(h_opt)
    return h_opt

def smooth_h(h, sigma_factor=0.1):
    """Apply smoothing to h"""
    n = len(h)
    sigma = sigma_factor * n
    h_smooth = gaussian_filter1d(h, sigma)
    # Clip and rescale
    h_smooth = np.clip(h_smooth, 0, 1)
    # Fix rescaling
    h_smooth = h_smooth * (len(h) / 2.0) / np.sum(h_smooth)
    return h_smooth

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Initial setup
    h = prepare_initial_h(n_points)
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1.0:  # Leave some margin
        iter_count += 1
        # Local optimization step
        h_opt = optimize_step(h, n_points, dx)
        
        # Apply smoothing
        h_smooth = smooth_h(h_opt)
        
        # Evaluate the C5
        c5 = evaluate_c5(h_smooth)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_smooth.copy()
            
        # Update h for next iteration
        h = h_smooth.copy()
        
    if best_h is None:
        # Fallback in case no good solution was found
        h_final = prepare_initial_h(n_points)
        best_c5 = evaluate_c5(h_final)
        best_h = h_final.copy()
        
    return (best_h.tolist(), best_c5, n_points)
