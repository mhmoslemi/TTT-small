import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h += 0.1 * np.random.rand(n_points)
        h = np.clip(h, 0, 1)
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    max_no_improve = 20
    no_improve_count = 0
    perturbation_magnitude = 0.01
    
    while time.time() - start_time < budget_s - 0.1:
        # Compute full correlation
        correlation = signal.correlate(h, 1 - h, mode='full')
        # Extract the relevant parts for each lag k
        cross_correlations = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            # The lag k corresponds to correlation[k] in the full array
            cross_correlations[k] = correlation[k] * dx
        
        # Find the maximum C5 and corresponding lags
        max_c5 = np.max(cross_correlations)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        if no_improve_count >= max_no_improve:
            # Perturb the solution
            h += perturbation_magnitude * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improve_count = 0
            continue
        
        # Identify the indices of lags that contribute to max C5
        max_indices = np.where(np.abs(cross_correlations - max_c5) < 1e-10)[0]
        
        # Compute gradient for h
        gradient = np.zeros(n_points)
        for k_idx in max_indices:
            # For lag k, compute the gradient contribution
            # cross_correlation_k = sum_{i=0}^{n_points-1-k} h[i] * (1 - h[i+k])
            # d/dh[j] = (if j = i) (1 - h[i+k]) or (if j = i+k) -h[i]
            for i in range(n_points - k_idx):
                if i + k_idx < n_points:
                    gradient[i] += (1 - h[i + k_idx])
                    gradient[i + k_idx] -= h[i]
        
        # Line search over alpha
        step_sizes = [0.1, 0.25, 0.5]
        best_alpha = None
        best_new_c5 = max_c5
        
        for alpha in step_sizes:
            h_candidate = h - alpha * gradient
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Evaluate candidate
            candidate_correlation = signal.correlate(h_candidate, 1 - h_candidate, mode='full')
            candidate_cross_correlations = np.zeros(2 * n_points - 1)
            for k in range(2 * n_points - 1):
                candidate_cross_correlations[k] = candidate_correlation[k] * dx
            candidate_c5 = np.max(candidate_cross_correlations)
            
            if candidate_c5 < best_new_c5:
                best_new_c5 = candidate_c5
                best_alpha = alpha
        
        if best_alpha is not None:
            h = h - best_alpha * gradient
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improve_count = 0
        else:
            # If no good alpha found, try a small random perturbation
            h += 0.001 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improve_count += 1
    
    # Final evaluation
    final_correlation = signal.correlate(best_h, 1 - best_h, mode='full')
    final_cross_correlations = np.zeros(2 * n_points - 1)
    for k in range(2 * n_points - 1):
        final_cross_correlations[k] = final_correlation[k] * dx
    final_c5 = np.max(final_cross_correlations)
    
    return (best_h.tolist(), final_c5, n_points)
