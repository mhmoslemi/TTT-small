import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Initialize with a simple construction
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Ensure constraints
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    start_time = time.time()
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 20
    
    while time.time() - start_time < budget_s - 1:
        # Compute all cross-correlations
        # np.correlate computes correlation with flipped second signal
        # For our case, we want corr(h, 1-h)
        # So we compute corr(h, 1-h) but this is equivalent to corr(1-h, h) with different indexing
        
        # Compute 1-h
        h1 = 1.0 - h
        # Cross-correlation with mode='full'
        # This returns correlation at all lags from -n+1 to n-1
        corr = np.correlate(h, h1, mode='full')
        
        # The overlap integral is the correlation scaled by dx
        # We want to maximize this for each lag, then minimize the maximum
        # So we compute the max correlation over all lags, but we know it's computed with dx from corr
        # corr has length 2*n - 1, corresponding to lags -n+1 to n-1
        # The actual lag values are: -n+1, -n+2, ..., -1, 0, 1, ..., n-1
        corr_scaled = corr * dx
        
        # Find the maximum value and the corresponding lag
        max_corr_idx = np.argmax(corr_scaled)
        max_corr_val = corr_scaled[max_corr_idx]
        lag = max_corr_idx - (n_points - 1)  # Convert index to actual lag
        
        # Compute gradient for the maximum lag
        # We want to minimize the maximum correlation at lag 'lag'
        # The gradient of correlation_k with respect to h[i] is:
        # - if i + k < n_points: 1 - h[i + k] 
        # - otherwise: 0
        
        # Compute the gradient vector for the current maximum lag
        grad = np.zeros(n_points)
        if lag >= 0:
            # lag is non-negative, so we want values at positions [0, 1, ..., n_points-1-lag]
            # to be subtracted by 1-h[i + lag]
            grad[:n_points - lag] = -(1.0 - h[lag:])
        else:
            # lag is negative, so we want values at positions [-lag, ..., n_points-1] to be subtracted by 1-h[i]
            # This means for h[i], the contribution comes at position i - lag
            grad[-lag:] = -(1.0 - h[:-lag])
        
        # Try different step sizes
        alpha_candidates = [0.1, 0.5, 1.0]
        best_new_h = None
        best_new_C5 = best_C5
        
        for alpha in alpha_candidates:
            # Compute candidate h
            h_new = h - alpha * grad
            
            # Clip to [0,1] range
            h_new = np.clip(h_new, 0, 1)
            
            # Reproject to sum(h) = n_points / 2
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new C5
            new_C5 = compute_C5(h_new, dx)
            
            if new_C5 < best_new_C5:
                best_new_C5 = new_C5
                best_new_h = h_new.copy()
        
        # If we found an improvement
        if best_new_h is not None:
            h = best_new_h
            if best_new_C5 < best_C5:
                best_C5 = best_new_C5
                best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count >= max_no_improvement:
                # Random perturbation
                h = np.clip(h + 0.01 * np.random.randn(n_points), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                no_improvement_count = 0
        
        iter_count += 1
        
    return best_h.tolist(), best_C5, n_points

def compute_C5(h, dx):
    # For the desired evaluation with 1-h
    h1 = 1.0 - h
    # Compute all correlations with mode='full'
    # Result is a vector of 2*n - 1 values from lag -n+1 to n-1
    corr = np.correlate(h, h1, mode='full')
    # Then we pick the max of all the scaled correlations (scaled by dx)
    corr_scaled = corr * dx
    return np.max(corr_scaled)
