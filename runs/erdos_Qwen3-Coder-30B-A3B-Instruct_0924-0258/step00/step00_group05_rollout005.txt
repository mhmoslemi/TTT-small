import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)
        h[0] = 0.4
        h[1] = 0.6
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    max_iter = 500
    alpha = 0.1
    threshold = 1e-4
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        corr = signal.correlate(h, 1 - h, mode="full")
        corr *= dx
        c5 = np.max(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find indices where maximum is achieved
        max_indices = np.where(corr == c5)[0]
        # For numerical stability, we might get multiple indices
        max_idx = max_indices[0]  # use first
        k = max_idx - (n_points - 1)  # convert to lag

        # Compute gradient
        g = np.zeros(n_points)
        if k >= 0:
            # h[i] contributes to overlap with (1 - h[i+k])
            for i in range(n_points - k):
                g[i] += 1.0 * dx
                g[i+k] += -1.0 * dx
        else:
            # k < 0; h[i+k] contributes to overlap with (1 - h[i])
            for i in range(-k, n_points):
                g[i] += -1.0 * dx
                g[i+k] += 1.0 * dx

        # Gradient descent step
        h_new = h - alpha * g
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        scale = (n_points / 2) / np.sum(h_new)
        h_new *= scale
        
        h = h_new
        
    # Final evaluation
    corr = signal.correlate(best_h, 1 - best_h, mode="full")
    corr *= dx
    final_c5 = np.max(corr)
    
    return (best_h.tolist(), final_c5, len(best_h))
