import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial values if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        n_points = 70
        h0 = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    start_time = time()
    
    # Binary search on C5
    low = 0.0
    high = 0.38085
    eps = 1e-4
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    
    while high - low > eps and time() - start_time < budget_s - 5:
        C_mid = (low + high) / 2.0
        
        # Formulate relaxed LP
        h = cp.Variable(n_points)
        constraints = [h[i] >= 0 for i in range(n_points)]
        constraints += [h[i] <= 1 for i in range(n_points)]
        constraints += [cp.sum(h) == n_points / 2.0]
        
        # Add relaxed constraints for all lags
        for k in range(2 * n_points - 1):
            # Compute correlation term using linear relaxation
            # h[i] * (1 - h[(i+k)%n_points]) <= (h[i] + 1 - h[(i+k)%n_points]) / 2
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h[i] + 1 - h[j]
            # Note: This is a conservative relaxation
            constraints += [expr / 2.0 <= C_mid / dx]
        
        # Solve the problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h.value
                # Project to bounds and normalize
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2.0) / np.sum(h_sol)
                c5_test = evaluate_erdos_solution(h_sol)
                if c5_test < best_c5:
                    best_h = h_sol.copy()
                    best_c5 = c5_test
                high = C_mid
            else:
                low = C_mid
        except:
            low = C_mid
            
    # Final check with actual computation
    c5_final = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), c5_final, n_points)
