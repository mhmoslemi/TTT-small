import time

def run(seed=42, budget_s=600, **kwargs):

        overlaps = np.correlate(best_h, 1 - best_h, mode="full")

        h = cp.Variable(n_points)

        constraints = []

        constraints.append(h >= 0)

        full_overlaps = np.correlate(best_h, 1 - best_h, mode="full")

                continue
                
            sum_expr = cp.sum(cp.multiply(h[i_vals], 1 - h[j_vals]))
            constraints.append(sum_expr <= C)

        objective = cp.Minimize(C)

        prob = cp.Problem(objective, constraints)

def evaluate_c5(h, dx):
    # Compute overlaps: C5 = max_k int h(x)(1-h(x+k)) dx
    # This is equivalent to: max_k sum_i h[i] * (1 - h[i+k])
    overlaps = np.correlate(h, 1 - h, mode="full")
    return np.max(overlaps) * dx

def compute_gradient(h, dx):
    # Compute approximate gradient w.r.t. h
    grad = np.zeros_like(h)
    for i in range(len(h)):
        d = 1e-6
        h_plus = h.copy()
        h_plus[i] += d
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = h_plus * (len(h) / 2) / np.sum(h_plus)
        h_minus = h.copy()
        h_minus[i] -= d
        h_minus = np.clip(h_minus, 0, 1)
        h_minus = h_minus * (len(h) / 2) / np.sum(h_minus)
        c5_plus = evaluate_c5(h_plus, dx)
        c5_minus = evaluate_c5(h_minus, dx)
        grad[i] = (c5_plus - c5_minus) / (2 * d)
    return grad
