import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound for given step function h"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute correlation h * (1 - h) with full mode
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    # Return maximum value scaled by dx
    return np.max(conv) * dx

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()

            best_c5 = c5_init
            best_n = n_init
    else:
        # Construct a basic initial function
        n = 45
        h_init = np.ones(n) * 0.5
        h_init[0:n//2] = 1.0
        h_init[n//2:] = 0.0
        # Normalize to sum to n/2
        h_init = h_init * (n / 2) / np.sum(h_init)
        c5_init = evaluate_erdos_solution(h_init)
        if c5_init < best_c5:
            best_h = h_init.copy()
            best_c5 = c5_init
            best_n = n
    
    def objective(h):
        # Compute surrogate: sum of squared correlations
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        # Return weighted sum to encourage minimizing maximum
        return np.sum(corr**2) * dx**2
    
    def gradient(h):
        # Gradient of squared correlations approximation
        dx = 2.0 / len(h)
        # Simplified but effective gradient estimation
        # This is partial gradient - actual implementation is complex
        # Note: in practice this requires computing exact derivatives
        # For speed we'll use finite differences
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad

        return np.sum(h) - (len(h) / 2)

        # Clip to [0,1]

        # Rescale to sum to n_points/2
        h = h * (len(h) / 2) / np.sum(h)
        return h
