import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_erdos_solution(h):
    # Compute the maximum overlap integral with proper scaling
    n = len(h)
    dx = 2.0 / n
    # The full convolution sums over all possible shifts k
    # For the overlap integral: integral of h(x)(1-h(x+k))dx
    # corr = correlate(h, 1-h, mode="full")
    # But we want maximum over all shifts
    # Actually, compute all shifts explicitly to avoid ambiguity
    maxv = 0.0
    for k in range(1 - n, n):
        if k >= 0:
            # h[i] * (1 - h[i+k])
            # valid for i in [0, n-k-1]
            # sum = sum(h[i] * (1 - h[i+k]) for i in range(n-k))
            # This can be computed as: 
            if k == 0:
                continue  # not needed, would be just integral of h
            # Align start of h with shifted 1-h  
            sh = n-k
            s = sum(h[i] * (1 - h[i+k]) for i in range(sh))
        else:
            # h[i+k] * (1 - h[i]) 
            # valid for i in [-k, n-1]
            # n+k is valid for i in [0, n+k-1] 
            # i = 0 to n-|k|-1, use h[i+k]* (1-h[i])
            sh = n + k
            s = sum(h[i+k] * (1 - h[i]) for i in range(sh))
        val = s * dx
        if val > maxv:
            maxv = val
    return maxv

def project(h, n_points):
    # Clamp values to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize sum to n_points / 2
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * (n_points / 2) / sum_h
    return h

def mutate(h, sigma):
    h = h + np.random.normal(0, sigma, len(h))
    return project(h, len(h))

def crossover(h1, h2, alpha=0.6):
    # Blend crossover
    beta = np.random.uniform(alpha, 1 - alpha)
    h = beta * h1 + (1 - beta) * h2
    return project(h, len(h))

    # partial of integral h(x) * (1 - h(x+k)) w.r.t. h(i)  
    # The contribution to gradient of item i comes from terms where either:
    # h[i] appears (k=0 case) or h[i+k] appears  
    # We approximate gradient here by finite difference approximation
    # For more precise gradient, must compute this analytically per shift

    # Consider that when shift k, overlapping terms are
    #     h[i] * (1 - h[i+k]) for valid i
    #     So, for h[i], contributes to gradient of:
    #     - (1 - h[i+k]) if there's a valid shift k' such that i+k' = i
    #     - -h[i+k] if there's a valid shift k' such that i+k' = i  
    # Er, simpler approach:
    # We compute how much h[i] contributes to the objective J = sum(h[i] * (1-h[i+k]) over all i and k
    # For gradient of h[i], dJ/dh[i] = sum over k of d/dh[i] h[i] (1 - h[i+k]) 
    #                = sum over k of [ (1-h[i+k]) if i+k valid] + [ -h[previ] if i = previ+k] If not, then again we're in the sum over valid overlapping terms.
    # Better: j = h[i] * (1 - h[i+k])
    #     dJ/dh[i] = 1 - h[i+k] if h[i] appears in this sum, which happens for shift k'
    # For shift k' = 0, contribution to gradient of h[i] = 1 - h[i]
    # For shift k' = 0, contribution to gradient of h[i] = 1 - h[i]
    # But be precise: contribution of h[i] term depends on whether this term is part of the overlap integral under shift k'=k  
    # if for a particular shift k', we have h[i]*(1-h[i+k']) in the overlap integral, then:  
    # dJ/dh[i] = 1 - h[i+k']
    # and also if h[i+k']*(1-h[i]) is there, h[i+k'] appears. 
    # not easy to do this cleanly for all k simultaneously.
    # Let's just compute full gradient for one fixed k_effition and do approximate gradients

    # For any term in this sum where we differentiate w.r.t. h[i], we get:

    # That is: 
    # let S be the set of valid pairs (i, k) where both h[i] and h[i+k] are in the domain
    # and for i in range(n), if we consider terms h[i] (1-h[i+k]) over shift k, 
    # then dJ/dh[i] = sum_k (1-h[i+k]) for valid k such that h[i+k] exists.
    # The scale to dx is factor that was taken out in function computation

def gradient_descent_step(h, lr=0.01):
    n = len(h)
    h = h.copy()
    dx = 2.0/n
    # Estimate gradients and take step
    # This is expensive. Do an example step directly:
    # We directly try perturbing h slightly and see what happens
    # calls predict or compute change to objective:
    # Simple: Gradient relevance is built into how the function is computed.
    # For a step: use objective change to k neighbors
    # Use finite difference approximation to gradient by computing: 
    # we want gradient to minimize h*(1-h) over all overlaps
    # Actually better: explicitly compute influence from the overlap terms.
    # In approximation mode: Here's one idea:
    # use SGD line search idea: use small perturbations to estimate it.
    # For time constraints, just try polynomial search in gradient direction to more carefully explore nearby.
    # Or even simpler:
    # Just approach I:
    # Try a few gradient steps with a small learning rate
    # Let's try sampling particles and test if good step:
    step_vec = np.zeros_like(h)
    # per i compute contribution to gradient  
    for i in range(n):
        # We want contribution from j where overlap gets estimated
        # h[i] (1 - h[i+k]) 
        # Let's compute actual estimated impact for shifts up to some valid k-range, 
        # Assume that for shift k, we are computing change in h[i] terms for both occurrences.  
        # Local search heuristic: search a few directions, evaluate, see if we move toward better objective
        impact_sum = 0
        num_valid_shifts = 0
        for k in range(1 - n, n):
            if k == 0:
                continue  # skip self
            if i + k >= 0 and i + k < n:
                # Value h[i] * (1-h[i+k]) contributes to total overlap
                # So d/dh[i] of this term = 1 - h[i+k]
                impact_sum += (1-h[i+k])
                num_valid_shifts += 1
        step_vec[i] = impact_sum / n if num_valid_shifts > 0 else 0
    # Now we take gradient step
    new_h = h - lr * step_vec
    return project(new_h, n)

    best_c5 = evaluate_erdos_solution(best_h)

    num_generations = 1000  # bounded by time
    mutation_sigma = 0.1
    crossover_alpha = 0.6 
    lr = 0.01
    decay_rate = 100

    valid_candidates = []
    fitness_scores = []
    
    # Initialize population around the provided solution
    valid_candidates = [best_h]
    fitness_scores = [best_c5]

    for i in range(pop_size - 1):
        h = mutate(initial_h.copy(), 0.1)
        f = evaluate_erdos_solution(h)
        valid_candidates.append(h)
        fitness_scores.append(f)

    # Sort by fitness score (lower is better)
    zipped = list(zip(valid_candidates, fitness_scores))
    zipped.sort(key=lambda x: x[1])
    valid_candidates, fitness_scores = zip(*zipped)

    # Main loop
    gen = 0
    while time.time() - start_time < budget_s * 0.95 and gen < num_generations:
        gen += 1
        # Selection
        selected = valid_candidates[:int(0.4 * pop_size)]  # Keep top 40%
        # Add some random candidates to preserve diversity
        for _ in range(int(0.1 * pop_size)):
            h = mutate(initial_h.copy(), 0.1)
            f = evaluate_erdos_solution(h)
            selected.append(h)
        # Reproduce with crossover
        offsprings = []
        for _ in range(int(0.3 * pop_size)):
            # Choose 2 parents randomly among top 10 or selected
            parent1 = selected[np.random.randint(len(selected))]
            parent2 = selected[np.random.randint(len(selected))]
            offspring = crossover(parent1, parent2, alpha=crossover_alpha)
            # Mutate
            offspring = mutate(offspring, mutation_sigma * np.exp(-gen/decay_rate))
            offsprings.append(offspring)
        
        # Local refinement of top candidates  
        for h in valid_candidates[:int(0.05 * pop_size)]:
            h_refined = gradient_descent_step(h, lr * np.exp(-gen/decay_rate))
            f = evaluate_erdos_solution(h_refined)
            if f < best_c5:
                best_h = h_refined.copy()
                best_c5 = f
                
        # Evaluate new candidates
        all_candidates = selected + offsprings
        all_fitness = []
        for h in all_candidates:
            f = evaluate_erdos_solution(h)
            all_fitness.append(f)
            if f < best_c5:
                best_h = h.copy()
                best_c5 = f
        
        # Replace bottom 50% candidates
        zipped_new = list(zip(all_candidates, all_fitness))
        zipped_new.sort(key=lambda x: x[1])
        new_top = [x[0] for x in zipped_new[:int(0.5 * len(all_candidates))]]
        valid_candidates = new_top
        
    return (best_h.tolist(), best_c5, n_points)
