import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction or create one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a uniform initial guess
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Make it sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
        
    dx = 2.0 / n_points
    max_iter = 1000
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Line search parameters
    alpha_max = 0.1
    tolerance = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            c5_plus = evaluate_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            if h_minus[i] < 0:
                h_minus[i] = 0
            c5_minus = evaluate_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Project gradient to feasible set (sum to 0)
        grad_mean = np.mean(grad)
        grad = grad - grad_mean
        
        # Perform line search
        alpha = alpha_max
        while alpha > 1e-8:
            h_new = h - alpha * grad
            # Ensure bounds
            h_new = np.clip(h_new, 0, 1)
            # Rescale to maintain sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_c5(h_new, dx)
            
            if c5_new < best_c5:
                h = h_new
                best_h = h.copy()
                best_c5 = c5_new
                break
            else:
                alpha *= 0.5
        
        # If no improvement with current step, reduce learning rate
        if alpha <= 1e-8:
            alpha_max *= 0.9
            if alpha_max < 1e-6:
                break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_c5(h, dx):
    # Using full convolution to get overlap at all lags
    corr = np.correlate(h, 1 - h, mode="full")
    # The length of correlation is 2 * n_points - 1
    # We're taking the maximum of the middle values which are valid
    mid_idx = len(corr) // 2
    central_values = corr[mid_idx:(mid_idx + len(h))]
    return np.max(central_values) * dx
