import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure sum constraint
    def normalize_h(h_vec):
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    h = normalize_h(h)
    
    # Evaluate current C5
    def evaluate_C5(h_vec):
        # Compute all correlations
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # Only keep the valid ones (length 2*n_points - 1)
        corr = corr[n_points-1:]
        # Convert to integral form
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_C5 = evaluate_C5(best_h)
    
    start_time = time.time()
    
    # Gradient-based optimization with adaptive step size
    max_iter = 5000
    alpha = 0.05  # Initial learning rate
    tolerance = 1e-5
    prev_C5 = best_C5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Evaluate current C5 and find max lag
        corr_full = np.correlate(h, 1 - h, mode="full")
        corr = corr_full[len(h)-1:]  # Only valid correlations
        max_lags = np.where(corr == np.max(corr))[0]
        max_k = max_lags[0]  # Take the first one
        
        # Compute gradient (approximate using finite differences)
        # For each point, the gradient based on the maximum lag
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Contribution from the maximum lag k
            j = (i + max_k) % n_points
            grad[i] += (1 - h[j]) - h[i]
            
        grad *= dx
        
        # Update with gradient descent
        h_new = h - alpha * grad
        
        # Projection step: clip to [0,1] and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new)
        
        # Evaluate new solution
        new_C5 = evaluate_C5(h_new)
        
        # Accept improvement
        if new_C5 < best_C5:
            h = h_new
            best_C5 = new_C5
            # Adjust step size
            alpha *= 1.05
        else:
            # Reduce step size on no improvement
            alpha *= 0.9
            
        # Early stopping
        if abs(prev_C5 - best_C5) < tolerance:
            alpha *= 0.9
            if alpha < 1e-6:
                break
        prev_C5 = best_C5
    
    return (best_h.tolist(), best_C5, n_points)
