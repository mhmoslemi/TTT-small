def solve_convex_relaxation(h_start, n_points, dx, target_c5):
    h = cp.Variable(n_points)
    constraints = []
    
    # Bounds
    for i in range(n_points):
        constraints.append(h[i] >= 0)
        constraints.append(h[i] <= 1)
    
    # Measure constraint
    constraints.append(cp.sum(h) == n_points / 2.0)
    
    # Relax quadratic terms using linear approximation
    for k in range(2 * n_points - 1):
        # For full convolution at shift k
        left = k % n_points
        right = (k - n_points + 1) % n_points
        # Compute indices for correlation
        corr_sum = 0.0
        for i in range(n_points):
            j = (i + k) % n_points
            # Use linearization: h[i]*(1-h[j]) <= (h[i] + 1 - h[j])/2
            corr_sum += (h[i] + 1 - h[j]) / 2.0
        constraints.append(corr_sum <= target_c5 / dx)
    
    objective = cp.Minimize(0)  # This is a feasibility problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value
            # Project back to valid range
            h_sol = np.clip(h_sol, 0, 1)
            # Adjust to match exact sum constraint
            h_sol *= (n_points / 2.0) / np.sum(h_sol)
            return h_sol
    except:
        pass
    
    return None

def evaluate_erdos_solution(h_vals, n_points, dx):
    # Convert to array
    h = np.array(h_vals)
    # Evaluate all correlations
    conv = correlate(h, 1 - h, mode='full')
    # Only consider valid shifts
    max_val = np.max(conv[:n_points])
    return max_val * dx
