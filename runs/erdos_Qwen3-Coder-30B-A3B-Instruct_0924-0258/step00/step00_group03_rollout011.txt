import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_erdos_solution(h, n_points):
    """Evaluate the C5 constant for given step function h."""
    dx = 2.0 / n_points
    # Compute correlations
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all shifts (excluding zero shift)
    max_val = np.max(conv[:len(conv)//2])  # Only look at positive shifts
    return max_val * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def mutate_h(h, sigma, n_points):
    """Mutate h with Gaussian noise and project."""
    h_new = h + np.random.normal(0, sigma, len(h))
    return project_h(h_new, n_points)

def crossover_h(h1, h2, alpha):
    """Blend crossover between two step functions."""
    beta = np.random.uniform(alpha, 1 - alpha)
    h_new = beta * h1 + (1 - beta) * h2
    return h_new

def compute_fitness(h, n_points):
    """Compute the overlap integral for given h."""
    return evaluate_erdos_solution(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    """Run the optimization to find better h for C5."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Fallback initial construction
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    h0 = project_h(h0, n_points)
    best_h = h0.copy()
    best_c5 = compute_fitness(h0, n_points)
    current_time = time.time()
    
    # Population settings
    pop_size = 100
    generations = 1000
    mutation_rate = 0.1
    crossover_alpha = 0.6
    elite_size = int(0.1 * pop_size)
    tournament_size = 5
    local_refine_ratio = 0.05

    population = [h0.copy()]
    for _ in range(pop_size - 1):
        h_new = mutate_h(h0, 0.05, n_points)
        population.append(h_new)
    
    # Main optimization loop
    for g in range(generations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute fitness for all individuals
        fitnesses = []
        for h in population:
            fit = compute_fitness(h, n_points)
            fitnesses.append(fit)
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = population[0].copy()
        
        # Selection
        selected_indices = []
        for _ in range(pop_size - elite_size):
            # Tournament selection
            tournament = np.random.choice(len(population), tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament]
            winner_idx = tournament[np.argmin(tournament_fitnesses)]
            selected_indices.append(winner_idx)
        
        # Elitism
        elite_indices = sorted_indices[:elite_size]
        selected_indices.extend(elite_indices)
        
        # Crossover and mutation
        offspring = []
        for _ in range(pop_size - len(elite_indices)):
            parent1_idx = selected_indices[np.random.randint(0, len(selected_indices))]
            parent2_idx = selected_indices[np.random.randint(0, len(selected_indices))]
            h1 = population[parent1_idx]
            h2 = population[parent2_idx]
            h_new = crossover_h(h1, h2, crossover_alpha)
            h_new = mutate_h(h_new, mutation_rate, n_points)
            offspring.append(h_new)
        
        # Add elites and offspring
        new_population = [population[i] for i in elite_indices]
        new_population.extend(offspring)
        
        # Local refinement for top candidates
        num_local = max(1, int(local_refine_ratio * pop_size))
        for i in range(num_local):
            h = new_population[i]
            # Simple gradient-based update for local refinement
            # Approximate gradient using finite differences
            eps = 1e-4
            old_fit = compute_fitness(h, n_points)
            
            for j in range(len(h)):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = project_h(h_plus, n_points)
                fit_plus = compute_fitness(h_plus, n_points)
                
                grad = (fit_plus - old_fit) / eps
                # Step size schedule
                step = 0.01 * (1 - g / generations)
                h[j] -= step * grad
                
            h = project_h(h, n_points)
            new_population[i] = h
        
        # Update population
        population = new_population
        
        # Adaptive mutation rate
        if g % 50 == 0:
            mutation_rate *= 0.95

    return (best_h.tolist(), best_c5, n_points)
