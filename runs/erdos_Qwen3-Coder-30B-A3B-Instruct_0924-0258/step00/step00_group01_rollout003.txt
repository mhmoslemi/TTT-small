import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h, dx=2.0/len(h)):
    """Evaluate the overlap integral for a given step function h."""
    correlation = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(correlation) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize starting point from provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Or create a simple uniform distribution as fallback
        n_points = 47
        h = np.ones(n_points) * 0.5

    # Normalize h such that sum(h) * dx = 1 (hence sum(h) = n_points / 2)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Find all valid (i,j) pairs for correlations
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        max_val = corr[max_corr_idx]
        
        # If we have improvement, update best
        if max_val * dx < best_c5:
            best_h = h.copy()
            best_c5 = max_val * dx

        # Create subproblem using cvxpy
        # Minimize C such that for all lags k: sum(h[i] * (1 - h[i+k])) <= C*dx
        # This is equivalent to sum(h[i] - h[i]*h[i+k]) <= C*dx
        # The leading term sum(h[i]) is constant, so we just minimize sum(h[i]*h[i+k])

        # Track max correlation and penalty
        # Instead we'll solve a penalized relaxation: 
        # Minimize (1+epsilon)*C + sum( (h[i] - h[i]*h[i+k]) )
        # where epsilon is small, and we enforce each correlation <= C*dx
        
        # Simple gradient-like descent step:
        # Try local search: update h slightly to reduce the max correlation
        
        # Create an augmented softmax-based update
        # This aims to push h away from regions that cause high overlap
        gradient = np.zeros_like(h)
        h_extend = np.concatenate([h, [0]*(n_points-1)])
        for i in range(n_points):
            corr_val = 0
            for j in range(n_points):
                k = j - i
                if k >= 0:
                    corr_val += h[i] * (1 - h_extend[k + i])
            gradient[i] = corr_val * dx - best_c5  # Simple gradient

        # Apply gradient with small step
        h = h - 0.01 * gradient
        
        # Clamp values
        h = np.clip(h, 0, 1)

        # Re-normalize
        h = h * target_sum / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
