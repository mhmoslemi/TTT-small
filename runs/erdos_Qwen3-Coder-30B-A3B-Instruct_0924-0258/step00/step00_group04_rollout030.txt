import numpy as np
from scipy.optimize import minimize
import time
from copy import deepcopy

def evaluate_overlap(h, n_points):
    """Evaluate C5 for given h using fast convolution"""
    dx = 2.0 / n_points
    # np.correlate with mode='full' computes all overlaps
    overlaps = np.correlate(h, 1 - h, mode="full")
    # Trim to valid range
    overlaps = overlaps[n_points-1:2*n_points-1]
    return np.max(overlaps) * dx

def project_and_rescale(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, n_points, mutation_rate=0.05):
    """Apply mutation with given rate"""
    h = h.copy()
    n_mutate = max(1, int(len(h) * mutation_rate))
    indices = np.random.choice(len(h), n_mutate, replace=False)
    for i in indices:
        h[i] += np.random.normal(0, 0.1)
    return project_and_rescale(h, n_points)

def crossover(h1, h2, n_points):
    """Create offspring from two parents with 2-3 crossover points"""
    h1, h2 = h1.copy(), h2.copy()
    n_crossover = np.random.randint(2, 4)
    points = sorted(np.random.choice(n_points-1, n_crossover, replace=False))
    points = [0] + points + [n_points]
    
    new_h = np.zeros(n_points)
    for i in range(len(points)-1):
        start, end = points[i], points[i+1]
        if i % 2 == 0:
            new_h[start:end] = h1[start:end]
        else:
            new_h[start:end] = h2[start:end]
    return project_and_rescale(new_h, n_points)

def greedy_optimize(h, n_points, maxiter=200):
    """Apply local optimization to improve h"""
    def objective(h_flat):
        h_vec = h_flat.reshape(-1)
        h_vec = project_and_rescale(h_vec, n_points)
        return evaluate_overlap(h_vec, n_points)
    
    def constraint_sums(h_flat):
        h_vec = h_flat.reshape(-1)
        return np.sum(h_vec) - n_points / 2
    
    # Initialize with current h
    h_flat = h.copy().reshape(-1)
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sums}
    
    try:
        result = minimize(objective, h_flat, method='L-BFGS-B',
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': maxiter, 'ftol': 1e-6})
        if result.success:
            h_opt = result.x.reshape(-1)
            return project_and_rescale(h_opt, n_points)
    except:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100  # A manageable size that's fast to compute
    pop_size = 50
    max_generations = 200
    stagnation_limit = 50
    best_c5 = float('inf')
    best_h = None
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = project_and_rescale(initial_h, n_points)
        population.append(initial_h)
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = project_and_rescale(h, n_points)
            population.append(h)
    else:
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = project_and_rescale(h, n_points)
            population.append(h)
    
    generation = 0
    stagnation_count = 0
    mutation_rate = 0.05
    
    while time.time() - start_time < budget_s - 5 and generation < max_generations:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_overlap(h, n_points)
            fitness.append(c5)
        
        # Track best solution
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[np.argmin(fitness)].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Apply stagnation logic
        if stagnation_count > stagnation_limit:
            mutation_rate = 0.1
        else:
            mutation_rate = 0.05
            
        # Selection
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:pop_size//5]]
        
        # Create new population
        new_population = elite.copy()
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(elite, 2, replace=False)
            child = crossover(parent1, parent2, n_points)
            child = mutate(child, n_points, mutation_rate)
            new_population.append(child)
        
        # Local optimization for best individuals
        for i in range(len(new_population) // 10):
            idx = sorted_indices[i]
            if time.time() - start_time >= budget_s - 5:
                break
            if np.random.rand() < 0.75:
                optimized = greedy_optimize(population[idx], n_points, maxiter=50)
                new_population.append(optimized)
            else:
                new_population.append(population[idx])
                
        population = new_population[:pop_size]
        generation += 1
        
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_overlap(best_h, n_points)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback evaluation
        h = population[0]
        c5 = evaluate_overlap(h, n_points)
        return (h.tolist(), c5, n_points)
