import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h from initial_h_values if available, else create random feasible h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
    
    # Ensure h is within bounds
    h = np.clip(h, 0, 1)
    
    # Project to satisfy constraint
    h = h / np.sum(h) * target_sum
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    time_limit = start_time + budget_s
    
    # Main optimization loop
    for iteration in range(10000):
        if time.time() > time_limit:
            break
            
        # Compute correlations
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Take gradient of the max correlation with respect to h
        max_k = np.argmax(corr)
        # The correlation at k = max_k corresponds to h * (1 - h_shifted)
        # Gradient is (1 - h_shifted) for h[i] where h_shifted is offset by max_k
        # We use simpler approach: perturb h locally to find gradient
        
        # Estimate gradient by finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        
        # For each point, compute grad[i] by shifting h to get correlations
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            # Project back to sum constraint
            h_plus = h_plus / np.sum(h_plus) * target_sum
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
        
        # Update h with gradient descent
        step_size = 0.01
        h_new = h - step_size * grad
        
        # Project back to valid range
        h_new = np.clip(h_new, 0, 1)
        # Project back to sum constraint
        h_new = h_new / np.sum(h_new) * target_sum
        
        h = h_new
        
        # Evaluate new C5
        c5_new = evaluate_c5(h, dx)
        
        # Update best solution
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        
        # Adaptive step size
        step_size *= 0.999
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given h"""
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5
