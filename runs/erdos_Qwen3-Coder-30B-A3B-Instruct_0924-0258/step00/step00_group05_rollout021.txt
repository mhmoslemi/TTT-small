import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    overlap = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(overlap) * dx
    return max_overlap

def normalize_h(h, n_points):
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h *= target_sum / current_sum
    return h

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    return normalize_h(h, n_points)

def compute_gradient(h, n_points, max_lags):
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    for k in max_lags:
        if k == 0:
            continue
        # k is the lag in correlation; we need to account for circular shift
        # For correlation h * (1 - h_{k}), we compute gradient w.r.t. h[i]
        # which contributes to correl(i, k) = h[i] * (1 - h[i-k])
        # i.e., gradient[i] += (1 - h[i-k] - (1 - h[i])) = h[i] - h[i-k]
        for i in range(n_points):
            i_k = (i - k) % n_points
            grad[i] += (h[i] - h[i_k])
    grad *= dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    alpha = 0.1
    tol = 1e-4
    min_alpha = 1e-6
    perturb_freq = 20
    perturb_scale = 0.01
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        h = np.full(n_points, 0.5)
        h = project_h(h, n_points)
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, n_points)
    
    start_time = time.time()
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap integrals
        overlap = correlate(h, 1 - h, mode='full')
        max_val = np.max(overlap)
        max_lags = np.where(overlap == max_val)[0]
        
        # Convert to actual lags (handle the full correlation)
        actual_lags = max_lags - (n_points - 1)
        
        # Compute gradient
        grad = compute_gradient(h, n_points, actual_lags)
        
        # Gradient descent update
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate
        new_C5 = evaluate_erdos_solution(h_new, n_points)
        
        # Accept if improvement
        if new_C5 < best_C5:
            h = h_new
            best_C5 = new_C5
            last_improvement = iteration
        else:
            # Perturb occasionally if no improvement
            if iteration - last_improvement > perturb_freq:
                noise = np.random.normal(0, perturb_scale, n_points)
                h_perturbed = h + noise
                h_perturbed = project_h(h_perturbed, n_points)
                new_C5_pert = evaluate_erdos_solution(h_perturbed, n_points)
                if new_C5_pert < best_C5:
                    h = h_perturbed
                    best_C5 = new_C5_pert
                    last_improvement = iteration
        
        # Adapt step size
        if new_C5 < best_C5:
            # Increase step size if there was improvement
            alpha = min(1.0, alpha * 1.05)
        else:
            # Decrease step size if no improvement
            alpha = max(min_alpha, alpha * 0.9)
    
    # Final check
    final_h = project_h(best_h.copy(), n_points)
    final_C5 = evaluate_erdos_solution(final_h, n_points)
    
    return (final_h.tolist(), final_C5, n_points)
