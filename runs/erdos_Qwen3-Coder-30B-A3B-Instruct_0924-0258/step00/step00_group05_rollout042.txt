import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Binary search on C5
    low = 0.0
    high = 0.38085
    tolerance = 1e-4
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    while high - low > tolerance and time.time() - start_time < budget_s * 0.95:
        mid = (low + high) / 2.0
        
        # Formulate LP with linearized constraints
        h_vars = cp.Variable(n_points)
        
        # Constraint: sum(h) = target_sum
        constraints = [cp.sum(h_vars) == target_sum]
        
        # Add linearized constraints
        for k in range(n_points):
            # Constraint: \sum h[i] * (1 - h[i+k]) <= mid/dx
            # Linear relaxation: h[i] * (1 - h[i+k]) <= (h[i] + 1 - h[i+k]) / 2
            # That is: h[i] + 1 - h[i+k] - 2 * h[i] * h[i+k] >= 0
            # Which means: 2 * h[i] * h[i+k] <= h[i] + 1 - h[i+k]
            # Using the relaxation h[i] * (1 - h[i+k]) <= (h[i] + 1 - h[i+k])/2
            
            constraint_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                # Linear approximation: h[i] * (1 - h[j]) <= (h[i] + 1 - h[j]) / 2
                constraint_sum += (h_vars[i] + 1 - h_vars[j]) / 2
            constraints.append(constraint_sum <= mid / dx)
            
        # Objective: minimize max shift
        # Reduce by imposing that all shifts are bounded by C5, use relaxed constraints
        # This is less direct; but since we're using relaxation, we make a binary search + solving approach
        
        # Check if the relaxation is feasible
        try:
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.GLOP, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_vars.value
                # Clip to [0, 1]
                h_sol = np.clip(h_sol, 0, 1)
                # Rescale to meet constraint sum
                h_sol = h_sol * target_sum / np.sum(h_sol)
                
                c5_eval = evaluate_erdos_solution(h_sol, n_points)
                if c5_eval < best_c5:
                    best_h = h_sol.copy()
                    best_c5 = c5_eval
                    
                high = mid
            else:
                low = mid
        except Exception:
            low = mid
            
    # Final check to ensure best_h is valid (hard constraints)
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * target_sum / np.sum(h_final)
    
    final_c5 = evaluate_erdos_solution(h_final, n_points)
    
    return (h_final.tolist(), final_c5, n_points)

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    h_array = np.array(h)
    # 1 - h for correlation kernel
    kernel = 1 - h_array
    # Full correlation with mode="full"
    corr = np.correlate(h_array, kernel, mode="full")
    # Take maximum over all lags
    max_corr = np.max(corr) * dx
    return max_corr
