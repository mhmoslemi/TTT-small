import time

def run(seed=42, budget_s=600, **kwargs):

    max_iter = 1000
    start_time = time.time()

    def compute_overlap(h_vec):
        # Compute full correlation between h and (1-h)
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # Return maximum value scaled by dx
        return np.max(corr) * dx
    
    def objective(h_vec):
        return compute_overlap(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) * dx - 1.0
    
    def gradient(h_vec):
        # Compute approximated gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        orig_val = objective(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            # Ensure bounds
            h_plus = np.clip(h_plus, 0, 1)
            grad[i] = (objective(h_plus) - orig_val) / eps
        return grad
    
    # Main optimization loop
    best_h = h.copy()
    best_c5 = compute_overlap(h)
    
    # Use scipy.optimize with bounds
    bounds = [(0, 1) for _ in range(n_points)]

    # Project function to ensure bounds
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        scale = np.sum(h_vec) * dx
        if scale > 0:
            h_vec = h_vec * (1.0 / scale)
        return h_vec
    
    it = 0
    while time.time() - start_time < budget_s - 1 and it < max_iter:

        # Current value and gradient
        current_c5 = compute_overlap(h)
        grad = gradient(h)
        
        # Gradient descent step
        step_size = 0.05
        new_h = h - step_size * grad
        
        # Ensure bounds and sum constraint
        new_h = project_h(new_h)

            if new_c5 < best_c5:

        best_h = h.copy()
        best_c5 = compute_overlap(h)
    
    # Return the solution
    return (best_h.tolist(), best_c5, n_points)
