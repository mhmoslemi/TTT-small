import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

    h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h_vals):
        # Compute all overlaps
        corr = correlate(h_vals, 1 - h_vals, mode="full")
        # Extract the overlapping parts (excluding padding)
        mid = len(corr) // 2
        overlaps = corr[mid:mid+n_points]
        return np.max(overlaps) * (2.0 / n_points)

    # Helper function to project h onto feasible region
    def project(h_vals):
        # Clip to [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Rescale to ensure sum = n_points / 2
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        return h_vals

    # Evaluate initial solution
    best_c5 = compute_c5(h)
    best_h = h.copy()

    start_time = time.time()
    iteration = 0
    alpha = 0.1
    tolerance = 1e-5
    last_improvement = 0

    while time.time() - start_time < budget_s - 1:
        iteration += 1

        # Compute all overlaps efficiently
        h_copy = h.copy()
        corr = correlate(h_copy, 1 - h_copy, mode="full")
        mid = len(corr) // 2
        overlaps = corr[mid:mid+n_points]

        # Find the worst overlap (max)
        k_star = np.argmax(overlaps)
        max_overlap = overlaps[k_star]

        # Build gradient for this k_star
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_prev = (i - k_star) % n_points
            i_next = (i + k_star) % n_points
            grad[i] = 1 - h_copy[i_prev] - h_copy[i_next]

        # Gradient step
        h_new = h - alpha * grad

        # Project back
        h_new = project(h_new)

        # Compute new C5
        c5_new = compute_c5(h_new)

        # Line search: if not improving, try smaller step
        if c5_new >= best_c5:
            alpha *= 0.5

            h = h_new
            if c5_new < best_c5:

                    # Perturb to escape local minima
                    h += np.random.normal(0, 0.01, n_points)

                    last_improvement = 0

        # Occasionally reduce learning rate
        if iteration % 100 == 0 and alpha > 1e-6:
            alpha *= 0.9
