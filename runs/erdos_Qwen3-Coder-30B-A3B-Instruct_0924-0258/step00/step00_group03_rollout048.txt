import numpy as np
from scipy import fft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Start from initial construction or generate random feasible h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Parameters
    max_iter = 10000
    alpha_start = 0.1
    decay_rate = 1000.0
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlap integrals using FFT
        H = fft.fft(h)
        overlap_spectrum = np.abs(H)**2
        overlap_full = fft.ifft(overlap_spectrum).real
        
        # Compute C5 as maximum overlap (excluding zero lag)
        C5 = np.max(overlap_full[1:]) * dx  # Skip index 0 (zero lag)
        
        # Track best
        if C5 < best_c5:
            best_c5 = C5
            best_h = h.copy()
            
        # Compute gradient (approximating the derivative of overlap wrt h)
        # For overlap at k = 1 (since k=0 is 0 lag)
        k_star = np.argmax(overlap_full[1:]) + 1
        # Simplified gradient estimate
        grad = np.zeros_like(h)
        # Approximate finite difference gradient
        # Use the fact that for small changes in h
        # Overlap at position k is about h(i) * (1-h(i+k)) + etc
        # So the gradient is approximately (1 - h[i+k]) for each i
        for j in range(n_points):
            k = (j + k_star) % n_points
            grad[j] += (1.0 - h[k]) * dx
            
        # Update step
        t = float(i)
        alpha = alpha_start * np.exp(-t / decay_rate)
        h_new = h - alpha * grad
        
        # Projection steps to maintain constraints
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        h_new = h_new / current_sum * (n_points / 2)
        
        # For stability, mix with old value
        h = 0.9 * h + 0.1 * h_new
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), float(final_c5), n_points)
