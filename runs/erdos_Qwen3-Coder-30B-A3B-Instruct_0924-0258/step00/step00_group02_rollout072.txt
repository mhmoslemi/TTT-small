import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    # Compute overlap for all shifts
    h = np.array(h_values)
    h_rev = np.flip(h)
    correlation = np.correlate(h, h_rev, mode='full')
    # The indices corresponding to the k values for 1-h
    # The correlation of h with (1-h) is the same as correlation of h with (-h) + 1
    # So we compute correlation of h with 1-h using correlation of h with 1 and h with -h
    h_neg = -h
    h_one = np.ones_like(h)
    corr_h_neg = np.correlate(h, h_neg, mode='full')
    corr_h_one = np.correlate(h, h_one, mode='full')
    # Overlap integral = h(x) * (1 - h(x + k))
    # This is computed as - correlation(h, h) + correlation(h, 1) = corr_h_one - corr_h_neg
    overlap = corr_h_one - corr_h_neg
    # We only take the part where k >= 0 for the first half of the full correlation
    # overlapping by k from 0 to n-1
    max_overlap = np.max(overlap)
    return max_overlap * dx

    # Resolution levels to try
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n = n_points
    last_improvement = 0
    
    # Helper to ensure h is valid (within bounds and sum constraint)
    def project_h(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s > 0:
            h = h * (n_points / 2) / s
        return h
    
    # Surrogate objective: smooth approximation to max overlap
    def surrogate_objective(h):
        h = np.array(h)
        dx = 2.0 / n_points
        # Compute overlap for all shifts using convolution
        h_rev = np.flip(h)
        overlap_vals = np.correlate(h, h_rev, mode='full')
        # Compute overlap with (1 - h) 
        h_one = np.ones_like(h)
        h_neg = -h
        corr_h_one = np.correlate(h, h_one, mode='full')
        corr_h_neg = np.correlate(h, h_neg, mode='full')
        overlap = corr_h_one - corr_h_neg
        # Using the squared maximum as proxy for max
        max_over = np.max(overlap)
        return max_over if max_over > 0 else 1e-10  # Ensure positive
    
    constraints = []
    bounds = [(0, 1)] * n_points
    
    def sum_constraint(h):
        return np.sum(h) - n_points / 2
    
    constraints.append({'type': 'eq', 'fun': sum_constraint})
    
    def wrapped_objective(h):
        return surrogate_objective(h)
    
    for res in resolutions:
        n_points = res
        if time() - start_time > budget_s - 10:

        # Interpolate starting point if needed
        if len(h_start) < n_points:
            # Linear interpolation
            old_x = np.linspace(0, 2, len(h_start))
            new_x = np.linspace(0, 2, n_points)
