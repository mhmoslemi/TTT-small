def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute cross-correlation using full mode
    correlation = np.correlate(h, 1 - h, mode="full")
    # The max overlap is the maximum of these values
    max_overlap = np.max(correlation)
    return max_overlap * dx

def project_to_feasible(h, n_points):
    """Project h onto feasible region: sum(h) = n_points/2 and 0 <= h[i] <= 1."""
    h = np.clip(h, 0, 1)
    scale_factor = (n_points / 2) / np.sum(h)
    h = h * scale_factor
    return h
