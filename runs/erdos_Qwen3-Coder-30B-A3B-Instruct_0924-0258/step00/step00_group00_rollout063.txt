import numpy as np
import time
from scipy import signal

def evaluate_c5(h, dx):
    # Compute cross-correlation of h with (1-h)
    cross_corr = signal.correlate(h, 1 - h, mode='full')
    # The lags go from -(n-1) to (n-1), but we only care about the positive lags
    # The first n-1 elements correspond to negative lags
    # The middle element corresponds to lag 0
    # The last n-1 elements correspond to positive lags
    # We want the maximum positive lag (from the last n-1 elements)
    max_corr = np.max(cross_corr[len(h)-1:])
    return max_corr * dx

def project_to_feasible(h, n_points, dx):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to correct sum
    current_sum = np.sum(h) * dx
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def line_search(h, grad, dx, n_points, alpha_candidates):
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    for alpha in alpha_candidates:
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, n_points, dx)
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    h = project_to_feasible(h, n_points, dx)
    
    # Optimization parameters
    max_iter = 10000
    perturb_steps = 20
    alpha_candidates = [0.1, 0.5, 1.0]
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    iter_count = 0
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 0.1 and iter_count < max_iter:
        iter_count += 1
        
        # Compute cross-correlations
        cross_corr = signal.correlate(h, 1 - h, mode='full')
        # The lags go from -(n-1) to (n-1)
        # We want positive lags (from the last n-1 elements)
        max_corr_idx = np.argmax(cross_corr[len(h)-1:])
        max_lag = max_corr_idx - (len(h) - 1)
        
        # Compute gradient for this specific lag
        # The gradient d C_k / d h[i] = 1 - h[i + k] if i + k < n
        # But we compute it for the lag that gives max correlation
        grad = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                grad[i] = 1 - h[i + max_lag]
        else:
            # Invert the situation
            max_lag = -max_lag
            for i in range(n_points - max_lag):
                grad[i + max_lag] = 1 - h[i]
        
        # Perform line search
        h_new, c5_new = line_search(h, grad, dx, n_points, alpha_candidates)
        
        # Accept if improvement
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
        else:
            # Perturb every few iterations
            if iter_count % perturb_steps == 0:
                h = h + 0.01 * np.random.randn(n_points)
                h = project_to_feasible(h, n_points, dx)
    
    best_h = project_to_feasible(best_h, n_points, dx)
    return list(best_h), best_c5, n_points
