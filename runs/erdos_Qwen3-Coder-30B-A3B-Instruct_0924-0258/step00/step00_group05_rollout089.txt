import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode='full')
    # The maximum value is the one we want
    c5 = np.max(corr) * dx
    return c5

def project_to_feasible(h, n_points):
    """Project h to sum to n_points/2 and clip to [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def perturb_h(h, sigma):
    """Add Gaussian noise and project to feasible set."""
    h_new = h + np.random.normal(0, sigma, len(h))
    return project_to_feasible(h_new, len(h))
