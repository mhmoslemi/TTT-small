import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute the maximum overlap
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create uniform step function
        n_points = 47
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # Initial evaluation
    best_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 2:
        iteration += 1
        # Compute overlaps
        corr = correlate(h, 1 - h, mode='full')
        # Indices for full correlation (lags from -n+1 to n-1)
        lags = np.arange(-n_points + 1, n_points)
        max_idx = np.argmax(corr)
        max_lag = lags[max_idx]
        max_value = corr[max_idx]
        
        if max_value * dx <= best_c5:
            break
            
        # Try to improve by solving a convex relaxation
        # We fix the constraint that corresponds to the max overlap
        # The constraint is: sum over i of h[i] * (1 - h[i + max_lag])
        # i can range from 0 to n - 1 - max_lag if max_lag > 0
        # or from -max_lag to n - 1 if max_lag < 0
        # For fixed lag k, we consider sum_i h[i] * (1 - h[i + k]) = sum_i h[i] - sum_i h[i] * h[i + k]
        # subject to sum h[i] = n_points / 2
        
        # This is a non-convex problem. We use a relaxation.
        # We will instead try a simple improvement by projecting the current h
        # onto a new constraint: we want sum_i h[i](1-h[i+max_lag]) <= new_c5
        
        # Let's sample a few more points to avoid getting stuck
        # Add small random noise
        new_h = h + np.random.normal(0, 0.001, n_points)
        new_h = np.clip(new_h, 0, 1)
        # Renormalize
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate
        c5 = evaluate_h(new_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = new_h.copy()
            h = new_h.copy()
        else:
            # Small perturbation, try again
            h = new_h
        # If we're at a local minimum, nudge more
        if iteration % 10 == 0:
            h += np.random.normal(0, 0.0005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
