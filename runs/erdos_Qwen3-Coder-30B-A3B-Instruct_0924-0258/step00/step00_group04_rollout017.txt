import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_points = 42  # use the same size as initial construction
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # initialize h from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * target_sum
    
    best_h = h.copy()
    best_C5 = float('inf')
    
    # helper to evaluate C5
    def compute_C5(h_vec):
        # compute full correlation
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # take the maximum value across all lags (excluding the middle one which is 0)
        max_corr = np.max(corr)
        return max_corr * dx
    
    # helper to project h back to feasible region
    def project_h(h_vec):
        # clip to [0, 1]
        h_vec = np.clip(h_vec, 0, 1)
        # re-scale to have correct sum
        h_vec = h_vec / np.sum(h_vec) * target_sum
        return h_vec
    
    # optimization parameters
    max_iter = 10000
    alpha = 0.1
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 0.1 and iter_count < max_iter:
        # compute overlaps for all lags
        # we'll compute the correlation values beforehand since it's O(n^2)
        # Compute full correlation efficiently using numpy
        h_vec = h.copy()
        h_rev = 1 - h_vec[::-1]  # reversed vector for correlation
        corr = correlate(h_vec, h_rev, mode='full')
        # These are not final yet - need more work to normalize
        
        # compute all values (will be in C5_style index)
        # Compute correlation coefficients for the "overlap integral"
        full_corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # scaled by dx
        full_corr_scaled = full_corr * dx
        # Find the maximum value (positive lags only)
        C5_val = np.max(full_corr_scaled)
        
        # Check if this is the best so far
        if C5_val < best_C5:
            best_C5 = C5_val
            best_h = h.copy()
            last_improvement = iter_count
            
        # Compute gradient w.r.t. the worst lag
        # Lag k -> max how large the correlation values
        # the lags are from -(n-1) to (n-1), so we shift it
        k_indices = np.arange(- (n_points - 1), n_points)
        # take all values where k > 0 and k < n_points (since in full mode k goes 2n-2 = 82)
        # The actual values that represent the overlap integral
        # full_corr_scaled indices correspond to k = -41 .. +41
        max_k_idx = np.argmax(full_corr_scaled)
        max_k = k_indices[max_k_idx]
        
        # now compute gradient for the max_k lag
        # compute gradient: d/dh[j] of h[j] * (1 - h[j+k])
        # For current system, we compute:
        # grad = [1 - h[(j-k) % n_points] - h[(j+k) % n_points] for j in range(n_points)]
        # But we need to be careful since convolve uses different indexing conventions
        # Let's use the actual loaded full_corr expression
        grad = np.zeros(n_points)
        for j in range(n_points):
            # the derivative is 1 - h[(j - k) % n_points] - h[(j + k) % n_points]
            left_j = (j - max_k) % n_points
            right_j = (j + max_k) % n_points
            grad[j] = 1 - h[left_j] - h[right_j]
        # normalize since correlate does * 2
        grad *= 2 * dx / n_points
        
        # update h
        h = h - alpha * grad
        h = project_h(h)
        
        iter_count += 1
        
        # Occasionally consider perturbation to avoid local minima
        if iter_count % 100 == 0:
            h += np.random.normal(0, 0.001, n_points)
            h = project_h(h)
        
        # Early exit if no improvement in a while
        if iter_count - last_improvement > 5000 and C5_val < 0.39:
            break
    
    # Final check
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
