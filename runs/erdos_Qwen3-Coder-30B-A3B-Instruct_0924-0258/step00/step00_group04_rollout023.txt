import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up problem parameters
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        assert np.all(h >= 0) and np.all(h <= 1), "Initial h must be in [0,1]"
        assert abs(np.sum(h) * dx - 1.0) < 1e-10, "Initial h must satisfy integral constraint"
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Helper to compute C5
    def compute_C5(h_vec):
        # compute correlation with 1-h
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Extract the valid correlations
        lags = len(corr) - len(h_vec) + 1
        corr_vals = corr[len(h_vec)-1 : len(h_vec)-1 + lags]
        return np.max(corr_vals) * dx
    
    # Project h to maintain constraints
    def project_h(h_vec):
        # Clip to bounds
        h_proj = np.clip(h_vec, 0, 1)
        # Rescale to correct integral
        integral = np.sum(h_proj) * dx
        if abs(integral - 1.0) > 1e-10:
            h_proj = h_proj * (1.0 / integral)
        return h_proj
    
    # Main optimization loop
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    start_time = time.time()
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode='full')
        lags = len(corr) - len(h) + 1
        corr_vals = corr[len(h)-1 : len(h)-1 + lags]
        # Find worst lag
        k_star = np.argmax(corr_vals)
        
        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + k_star) % len(h)
            grad[i] = 1 - 2 * h[j]
        
        # Normalize gradient
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-12:
            break
        grad = grad / norm_grad
        
        # Line search
        alpha = 0.1
        backtracking_factor = 0.8
        min_alpha = 1e-6
        while alpha >= min_alpha:
            h_new = h - alpha * grad
            h_proj = project_h(h_new)
            new_C5 = compute_C5(h_proj)
            if new_C5 < best_C5:
                h = h_proj
                if new_C5 < best_C5:
                    best_C5 = new_C5
                    best_h = h.copy()
                break
            alpha *= backtracking_factor
            
        if alpha < min_alpha:
            # If we can't improve, break
            break
    
    # Final correction
    h_final = project_h(best_h)
    final_C5 = compute_C5(h_final)
    
    return (h_final.tolist(), final_C5, n_points)
