import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h):
    """Evaluate the C5 constant for a given h"""
    # Compute the overlap for all shifts
    n = len(h)
    dx = 2.0 / n
    # Create the flipped h
    flipped_h = 1 - h
    # Compute all correlations
    correlations = correlate(h, flipped_h, mode='full')
    # Only take the valid ones (k from -n+1 to n-1)
    # But we need to take only the positive shifts part for max
    center = len(correlations) // 2
    # The effective k range is from 0 to 2 (we consider 0 to n-1 in original)
    # But we want max over all valid k (shifts from -n+1 to n-1)
    # Actually, for k from 0 to n-1, we need to consider the shifted correlation
    # We use the definition: for k = 0 to n-1, C_5 = max over all k
    # so we look at correlations from center to center + n-1 
    # because they correspond to shift k from 0 to n-1
    max_overlap = 0
    # We evaluate C_5 as max over valid integer k shifts
    # For each k from 0 to n-1, we evaluate the integral 
    # This is equivalent to: for k = 0 to n-1, compute the sum of h[i] * (1 - h[i+k])
    # where i+k < n, so i from 0 to (n-1-k)
    # This equals correlations about index center + k 
    # Just need to be careful of indexing
    # Actually simpler to recompute using the direct expression
    # max_k \int h(x) (1 - h(x+k)) dx
    # This means finding the maximum of the sum over valid i of h[i] * (1 - h[i+k]) 
    # over all valid k (0 to n-1)
    max_val = 0
    # Let's compute directly
    for k in range(n):
        # Integral from 0 to 2 of h(x) * (1 - h(x+k)) dx
        # Equals sum h[i] * (1 - h[i+k]) for valid i
        # Valid i: i+k < n => i <= n-k-1, so i from 0 to n-k-1
        sum_val = 0
        for i in range(n - k):
            sum_val += h[i] * (1 - h[i + k])
        max_val = max(max_val, sum_val * dx)  # dx = 2/n
    return max_val
