def evaluate_erdos_solution(h, n_points):
    """Evaluate the C5 bound for the given step function h"""
    dx = 2.0 / n_points
    # Compute all overlaps
    overlaps = correlate(h, 1 - h, mode="full")
    # Return maximum overlap scaled by dx
    return np.max(overlaps) * dx
