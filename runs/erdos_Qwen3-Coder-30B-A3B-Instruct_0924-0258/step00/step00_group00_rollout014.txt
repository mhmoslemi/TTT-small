import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    iteration = 0
    perturb_steps = 0
    max_perturb_steps = 20
    
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        
        # Compute all cross-correlations
        corr = signal.correlate(h, 1 - h, mode='full')
        lags = np.arange(-n_points + 1, n_points)
        max_idx = np.argmax(corr)
        max_lag = lags[max_idx]
        max_corr = corr[max_idx]
        
        # Compute gradient direction
        grad = np.zeros(n_points)
        if max_lag >= 0:
            grad[max_lag:] = -h[:-max_lag] if max_lag < n_points else 0
        else:
            grad[:max_lag] = -h[-max_lag:] if -max_lag < n_points else 0
            
        # Line search for step size
        alpha_candidates = [0.1, 0.5, 1.0]
        best_alpha = 0.0
        best_new_c5 = best_c5
        for alpha in alpha_candidates:
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_c5 = evaluate_erdos_solution(h_new)
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_alpha = alpha
        
        # If we found a better solution, update
        if best_alpha > 0:
            h = h - best_alpha * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            if best_new_c5 < best_c5:
                best_c5 = best_new_c5
                best_h = h.copy()
                perturb_steps = 0
            else:
                perturb_steps += 1
        else:
            # Perturb and rescale
            h += 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            perturb_steps += 1
            
        # Possibly perturb if no improvement recently
        if perturb_steps >= max_perturb_steps:
            h += 0.02 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            perturb_steps = 0
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h):
    # Compute C5 using numpy correlate
    n_points = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    lags = np.arange(-n_points + 1, n_points)
    # Take maximum value from all correlations
    max_corr = np.max(corr)
    # Multiply by dx = 2.0 / n_points
    dx = 2.0 / n_points
    return max_corr * dx
