import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Compute C5 bound for given h"""
    dx = 2.0 / len(h)
    # Compute full correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum correlation value and multiply by dx
    c5 = np.max(corr) * dx
    return c5

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum(h) = target_sum using scaling"""
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.ones_like(h) * (target_sum / len(h))
    return h * (target_sum / current_sum)

def perturb_h(h, T, noise_scale=0.1):
    """Add perturbation to h"""
    # Generate noise
    noise = np.random.normal(0, noise_scale * T, len(h))
    h_new = h + noise
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Project to correct sum
    target_sum = len(h) / 2.0
    h_new = project_to_constraint(h_new, target_sum)
    return h_new

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Start with initial values if available

    else:

        h = project_to_constraint(h, 5.0)
    
    n_points = len(h)
    c5_best = evaluate_c5(h)
    best_h = h.copy()
    
    # Multi-resolution optimization with simulated annealing
    resolutions = [10, 20, 40, 70]
    
    for res_idx, n_level in enumerate(resolutions):
        if time.time() - start_time > budget_s - 10:
            break
            
        if n_level > n_points:
            # Interpolate to higher resolution
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_level)
            h = np.interp(x_new, x_orig, h)
            n_points = n_level
        
        # Starting temperature
        T = 0.1
        cooling_rate = 0.99
        max_iter = 500
        
        for it in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            h_new = perturb_h(h, T)
            
            c5_new = evaluate_c5(h_new)
            
            # Accept with probability based on cost difference
            delta = c5_new - c5_best
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_new
                if c5_new < c5_best:
                    c5_best = c5_new
                    best_h = h.copy()
                
            # Cool down
            T *= cooling_rate
            
    # Final validation
    c5_final = evaluate_c5(best_h)
    
    return (best_h.tolist(), c5_final, len(best_h))
