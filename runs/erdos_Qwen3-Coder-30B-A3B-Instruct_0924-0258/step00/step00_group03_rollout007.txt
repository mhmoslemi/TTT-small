import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given step function h."""
    # Compute the full correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # Take the maximum correlation value
    max_corr = np.max(corr)
    # Return the value times dx
    return max_corr * dx

def normalize_h(h, target_sum):
    """Normalize h to have the specified sum while keeping values in [0,1]."""
    total = np.sum(h)
    if total > 0:
        h = h * target_sum / total
    # Clamp to [0,1] and re-normalize to ensure constraints
    h = np.clip(h, 0, 1)
    actual_sum = np.sum(h)
    if actual_sum > 0:
        h = h * target_sum / actual_sum
    return h

def mutate_h(h, mutation_rate, n_points, dx):
    """Mutate a step function h."""
    # Create small random changes
    noise = np.random.normal(0, 0.05, len(h))
    h_new = h + noise * mutation_rate
    # Normalize to keep sum constraint
    h_new = normalize_h(h_new, n_points * dx / 2)
    return h_new

def crossover_h(h1, h2):
    """Crossover two step functions."""
    n = len(h1)
    # Use uniform crossover
    mask = np.random.rand(n) > 0.5
    h_new = np.where(mask, h1, h2)
    return h_new

def tournament_selection(population, fitnesses, tournament_size):
    """Select an individual via tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx]

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    # Assume initial_h_values exists and has reasonable size
    # Use n_points = len(initial_h_values) to match current implementation
    n_points = len(initial_h_values)
    dx = 2.0 / n_points  # Grid spacing
    
    # Initialize population
    pop_size = 50
    population = []
    
    # Add initial_h_values
    population.append(normalize_h(np.array(initial_h_values), n_points * dx / 2))
    
    # Add some random feasible individuals
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = normalize_h(h, n_points * dx / 2)
        population.append(h)
    
    # Parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = max(2, pop_size // 10)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s:
        generation += 1
        # Evaluate fitness of all individuals
        fitnesses = []
        for h in population:
            c5 = evaluate_overlap(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Create new population
        new_population = []
        
        # Elitism - keep best
        elite_idx = np.argmin(fitnesses)
        new_population.append(population[elite_idx].copy())
        
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child = crossover_h(parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            child = mutate_h(child, mutation_rate, n_points, dx)
            
            new_population.append(child)
        
        population = new_population
    
    # Final re-evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_overlap(best_h, dx)
        return (list(best_h), final_c5, n_points)
    else:
        # Fallback
        h_eval = normalize_h(np.array(initial_h_values), n_points * dx / 2)
        c5_eval = evaluate_overlap(h_eval, dx)
        return (list(h_eval), c5_eval, n_points)
