import numpy as np

from scipy import correlate

def evaluate_h(h, dx):
    # Compute all cross-correlations for lags from -n+1 to n-1
    # The cross-correlation at lag k is sum h[i] * (1 - h[i+k])
    # We compute h * (1-h) with full mode in numpy
    corr = correlate(h, 1-h, mode="full")
    # The full correlation has length 2n-1
    # We map k from -n+1 to n-1 into indices [0, 2n-2]
    # We want the maximum value among all lags
    max_corr = np.max(corr) * dx
    return max_corr

def project_h(h, target_sum):

    # Use clipping and rescaling
    h = np.clip(h, 0, 1)

        return h
    scale_factor = target_sum / current_sum
    h = h * scale_factor
    # Clip again after scaling to ensure bounds
    h = np.clip(h, 0, 1)
    return h
