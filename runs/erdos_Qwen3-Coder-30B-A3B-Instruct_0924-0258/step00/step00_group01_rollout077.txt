import cvxpy as cp
import time

def evaluate_c5(h, dx):
    """Evaluate C5 using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

    best_c5 = evaluate_c5(h, dx)

    max_iters = 100
