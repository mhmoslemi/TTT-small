import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Initial guess; can be adjusted
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    max_iter = 10000
    alpha_init = 0.1
    backtrack_factor = 0.8
    min_alpha = 1e-6
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        assert len(h) == n_points
        assert np.all(h >= 0) and np.all(h <= 1)
        assert abs(np.sum(h) - target_sum) < 1e-6
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)  # Scale to target sum

    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode='full')
        lag_offset = n_points - 1
        z = corr[lag_offset:lag_offset + n_points] * dx
        k_star = np.argmax(z)
        c5_val = z[k_star]

        # Compute gradient
        g = np.zeros(n_points)
        for i in range(n_points):
            g[i] = 1 - 2 * h[(i + k_star) % n_points]
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 0:
            g = g / g_norm
        else:
            continue  # No gradient, skip update
            
        # Line search for step size
        alpha = alpha_init
        while alpha >= min_alpha:
            h_new = h - alpha * g
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < c5_val:
                h = h_new
                break
            alpha *= backtrack_factor
        else:
            # If line search fails, reduce step size and try again
            h = np.clip(h - 0.01 * g, 0, 1)
            h = h * target_sum / np.sum(h)
            
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    n_points = len(h)
    lag_offset = n_points - 1
    z = corr[lag_offset:lag_offset + n_points] * dx
    return np.max(z)
