import time

    alpha = 0.1
    tol = 1e-4
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        # Compute all correlations and find maximum
        correlation = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(correlation)
        max_val = correlation[max_idx]
        max_overlap = max_val * dx
        
        # Find lags that contribute to max
        # For this simple gradient descent approach, we'll use all relevant indices
        grad = np.zeros_like(h)
        
        # Gradient direction for the contributing kernel
        for i in range(n_points):
            for k in range(n_points):
                j = i + k
                if j < n_points:
                    grad[i] += (1 - h[j]) * dx
                if i >= k:
                    j = i - k
                    if j >= 0:
                        grad[i] += (-h[j]) * dx
        
        # Update h
        h_new = h - alpha * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new *= (n_points / 2) / np.sum(h_new)

        current_c5 = evaluate_erdos_solution(h_new, n_points)

            h = h_new.copy()

    # Final validation
