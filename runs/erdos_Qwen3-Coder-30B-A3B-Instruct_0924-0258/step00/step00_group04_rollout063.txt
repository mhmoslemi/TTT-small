import numpy as np
import time
import random
from scipy import optimize

def evaluate_c5(h):
    """Evaluate the C5 constant for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value and scale it appropriately
    c5 = np.max(corr) * dx
    return c5

def project_and_rescale(h, n_points):
    """Project h into [0,1] and rescale to meet integral constraint"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        # Rescale to make sum = n_points/2
        h = h * (n_points / 2) / current_sum
    return np.clip(h, 0, 1)

    h_start = project_and_rescale(h_start, n_points)
    
    # Annealing parameters
    temp_initial = 1.0
    temp_min = 0.01
    cooling_rate = 0.99
    patience_limit = 50
    gradient_scale = 0.1
    mutation_std = 0.05
    max_iter = int(budget_s)
    
    # Initialization
    h_current = h_start.copy()
    current_c5 = evaluate_c5(h_current)
    best_h = h_current.copy()
    best_c5 = current_c5
    
    # For patience tracking
    patience = 0
    last_improvement = 0
    
    start_time = time.time()
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Annealing temperature
        temp = max(temp_initial * (cooling_rate ** i), temp_min)
        
        # Compute gradient approximation
        grad = np.zeros_like(h_current)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project_and_rescale(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = project_and_rescale(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
        # Generate candidate
        # Add Gaussian noise
        noise = np.random.normal(0, mutation_std, n_points)
        # Bias towards improving gradient direction
        h_candidate = h_current - gradient_scale * grad + noise
        h_candidate = project_and_rescale(h_candidate, n_points)
        
        # Evaluate candidate
        candidate_c5 = evaluate_c5(h_candidate)
        
        # Accept or reject
        if candidate_c5 < current_c5 or np.random.random() < np.exp((current_c5 - candidate_c5) / temp):
            h_current = h_candidate
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                patience = 0
                last_improvement = i
            else:
                patience += 1
        else:
            patience += 1
            
        # Check for early restart
        if patience > patience_limit:
            temp = temp_initial
            patience = 0
            
    return (best_h.tolist(), best_c5, n_points)
