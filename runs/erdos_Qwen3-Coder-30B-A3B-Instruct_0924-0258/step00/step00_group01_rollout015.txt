import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47  # Using the initial construction size
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if abs(np.sum(h) * dx - 1.0) > 1e-6:
            h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.full(n_points, 0.5)
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure constraints
    assert np.all(h >= 0) and np.all(h <= 1), "Initial h values not in [0,1]"
    assert abs(np.sum(h) * dx - 1.0) < 1e-6, "Initial sum not equal to 1"
    
    # Precompute binary matrices for all lags
    B = {}
    for k in range(n_points):
        B[k] = np.zeros(n_points)
        B[k][k:] = 1
        B[k] = B[k].astype(int)
    
    B_neg = {}
    for k in range(n_points):
        B_neg[k] = np.zeros(n_points)
        B_neg[k][:k] = 1
        B_neg[k] = B_neg[k].astype(int)
        
    # Objective function
    def compute_C5(h):
        corr = np.correlate(h, 1-h, mode="full")
        C5 = np.max(corr) * dx
        return C5
    
    # fwht-based overlap computation (for optimization)
    def compute_overlaps_fwht(h):
        # FWHT of h and (1-h)
        h_fft = fft(h)
        inv_h_fft = fft(1 - h)
        # Cross-correlation
        cross_corr_fft = h_fft * np.conj(inv_h_fft)
        cross_corr = np.real(ifft(cross_corr_fft))
        return cross_corr * dx
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Optimization parameters
    alpha = 0.1
    n_iter = 0
    no_improve_count = 0
    escape_probability = 0.05
    
    while time.time() - start_time < budget_s - 2:
        n_iter += 1
        
        # Compute overlaps
        overlaps = compute_overlaps_fwht(h)
        k0 = np.argmax(overlaps)
        C5 = overlaps[k0]
        
        # Update best solution
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Adaptive learning rate decay
        if no_improve_count % 50 == 0 and no_improve_count > 0:
            alpha *= 0.95
            if alpha < 1e-6:
                alpha = 1e-6
        
        # Compute gradient approximation
        # h[i] = h[i] + alpha * (B[k0][i] * (1 - h[(i+k0)%n_points]) - B_neg[k0][i] * h[(i-k0)%n_points])
        
        # Using simplified gradient
        grad = np.zeros(n_points)
        if k0 < n_points:
            # For k0 lag
            for i in range(n_points):
                grad[i] += B[k0][i] * (1 - h[(i+k0) % n_points]) - B_neg[k0][i] * h[(i-k0) % n_points]
        
        grad *= alpha
        
        # Apply gradient
        h += grad
        
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        
        # Ensure constraint on sum
        current_sum = np.sum(h)
        target_sum = n_points / 2
        scale_factor = target_sum / current_sum
        h *= scale_factor
        h = np.clip(h, 0, 1)
        
        # Escape local minima
        if np.random.random() < escape_probability:
            perturb = np.random.normal(0, 0.02, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h *= target_sum / np.sum(h)
    
    # Final validation
    final_C5 = compute_C5(best_h)
    return (best_h.tolist(), final_C5, n_points)
