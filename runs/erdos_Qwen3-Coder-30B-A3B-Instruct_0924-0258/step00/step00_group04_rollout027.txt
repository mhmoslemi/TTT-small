def evaluate_c5(h, dx):
    """Compute the maximum overlap integral C5"""
    n = len(h)
    # Compute cross-correlation of h and (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Take the maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

        c5_minus = evaluate_c5(h_minus, dx)
