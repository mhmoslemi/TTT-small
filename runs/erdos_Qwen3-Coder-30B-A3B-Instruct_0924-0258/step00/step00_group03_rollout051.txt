import numpy as np
from scipy import signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Initial evaluation
    conv = signal.correlate(h, 1 - h, mode='full')
    max_k = np.argmax(conv)
    c5_bound = conv[max_k] * dx
    
    best_h = h.copy()
    best_c5 = c5_bound
    
    # Gradient-based optimization with projection
    learning_rate = 0.01
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Compute gradient for the maximum overlap lag
        k0 = max_k
        grad = np.zeros(n_points)
        
        # Gradient contribution from the k0 shift
        # For convolution at lag k0, the gradient with respect to h[i] 
        # is (1 - h[i + k0]) if i + k0 < n_points, or (1 - h[i - k0]) if we're mirroring
        # Since we're computing full convolution, the derivative is based on shifted indices
        
        # Mirror boundary conditions in convolution
        forward_indices = np.arange(n_points)
        backward_indices = np.arange(n_points-1, -1, -1)
        
        # Gradient computation: see which h[i] contribute to max overlap at lag k0
        for i in range(n_points):
            j = i + k0 - (n_points - 1)  # Shift in the convolution
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        
        # Define optimization variables
        h_var = cp.Variable(n_points)
        constraints = [h_var >= 0, h_var <= 1]
        
        # Compute sum constraint
        objectives = []
        for i in range(n_points):
            # Gradient step
            grad_i = grad[i] * learning_rate
            objectives.append(h_var[i] - grad_i)
        
        obj = cp.Minimize(cp.sum(objectives))
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h_new = np.array(h_var.value).reshape(-1)
        except Exception:
            h_new = h - learning_rate * grad
            h_new = np.clip(h_new, 0, 1)
        
        # Project onto sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        conv_new = signal.correlate(h_new, 1 - h_new, mode='full')
        max_k_new = np.argmax(conv_new)
        c5_new = conv_new[max_k_new] * dx
        
        h = h_new
        max_k = max_k_new
        c5_bound = c5_new
        
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
            
        # Adaptive learning rate
        if iter_count % 100 == 0:
            learning_rate *= 0.95
            
    return (best_h.tolist(), best_c5, n_points)
