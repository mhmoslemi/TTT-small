import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound for given h values."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute all correlations for lags 0 to n-1
    corr = correlate(h_values, 1 - h_values, mode="full")
    # Extract the valid correlations (from lag 0 to n-1) and multiply by dx
    overlaps = corr[n - 1 : 2 * n - 1] * dx
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Validate constraints
    assert np.all(h >= 0) and np.all(h <= 1), "h must be in [0,1]"
    assert abs(np.sum(h) - n_points / 2) < 1e-6, "sum(h) must equal n_points/2"
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    alpha = 0.1
    backtracking = 0.8
    max_iter = 100000
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Compute all overlaps for each lag
        dx = 2.0 / n_points
        corr = correlate(h, 1 - h, mode="full")
        overlaps = corr[n_points - 1 : 2 * n_points - 1] * dx
        k_star = np.argmax(overlaps)
        
        # Compute gradient: dC5/dh[j] = 1 - 2*h[(j + k_star) % n_points]
        grad = np.zeros(n_points)
        for j in range(n_points):
            idx = (j + k_star) % n_points
            grad[j] = 1 - 2 * h[idx]
        
        # Normalize gradient to unit length
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad /= grad_norm
            
        # Line search
        alpha_new = alpha
        h_new = h - alpha_new * grad
        
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy integral constraint
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Compute new C5
        c5_new = evaluate_erdos_solution(h_new)
        
        # Backtracking line search
        while alpha_new > 1e-6 and c5_new >= best_c5:
            alpha_new *= backtracking
            h_new = h - alpha_new * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            c5_new = evaluate_erdos_solution(h_new)
            
        # Update
        h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        
        alpha = alpha_new
        
        # Occasionally reduce step size to prevent overshoot
        if iter_count % 50 == 0:
            alpha *= 0.95
    
    return (best_h.tolist(), best_c5, n_points)
