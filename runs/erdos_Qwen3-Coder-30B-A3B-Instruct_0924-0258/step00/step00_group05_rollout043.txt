import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_precision = 1e-4
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5, dtype=float)
    
    # Normalize to sum to 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Binary search on C5
    low = 0.0
    high = 0.38085
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points, dx)
    
    while high - low > target_precision and (time.time() - start_time) < budget_s - 1:
        c_mid = (low + high) / 2.0
        
        # Create variables and constraints using CVXPY
        h_vars = cp.Variable(n_points, name='h')
        constraints = []
        
        # Bounds
        constraints += [h_vars >= 0, h_vars <= 1]
        
        # Sum constraint
        constraints += [cp.sum(h_vars) == n_points / 2]
        
        # Overlap constraints
        # Compute indices and create correlation pattern
        for k in range(2 * n_points - 1):
            # For correlation with shifted 1-h
            t = 0
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_vars[i] * (1 - h_vars[j])
                t += 1
            constraints += [expr <= c_mid / dx]
        
        # Objective (minimize C5)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                solution = h_vars.value
                # Check constraints after solving
                if solution is not None:
                    solution = np.clip(solution, 0, 1)
                    norm_factor = (n_points / 2) / np.sum(solution)
                    solution = solution * norm_factor
                    
                    # Check if it's better
                    c5_val = evaluate_erdos_solution(solution, n_points, dx)
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = solution.copy()
                    
                    # Update search range
                    high = c_mid
                    h = solution
                else:
                    low = c_mid
            else:
                low = c_mid
        except Exception:
            low = c_mid
    
    # Ensure we return the best solution found
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h, n_points, dx):
    # Compute the overlap integral in a vectorized way
    h_np = np.array(h)
    # Padding for convolution
    h_extended = np.concatenate([h_np, h_np])
    h_shifted = 1 - h_np
    # Use numpy.correlate for full convolution
    result = np.correlate(h_extended, h_shifted, mode='full')
    # Take the upper half, corresponding to the maximum over k
    max_corr = np.max(result[n_points - 1:])
    return max_corr * dx
