import numpy as np
import time
from scipy.optimize import minimize

def evaluate_c5(h):
    """Compute the C5 constant for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the full convolution of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum value among all lags
    c5 = np.max(corr) * dx
    return c5

def project_and_rescale(h, n_points):
    """Project h to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, eps=1e-6):
    """Compute finite difference gradient of C5 w.r.t. h."""
    n = len(h)
    grad = np.zeros_like(h)
    c5_base = evaluate_c5(h)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_and_rescale(h_plus, n)
        c5_plus = evaluate_c5(h_plus)
        grad[i] = (c5_plus - c5_base) / eps
    return grad
