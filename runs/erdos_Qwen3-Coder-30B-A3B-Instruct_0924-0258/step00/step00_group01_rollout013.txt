import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 47
        h = 0.5 * np.ones(n_points)
    
    # Adjust to meet sum constraint exactly
    h *= (n_points / 2) / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Parameters
    alpha = 0.1
    momentum = 0.9
    last_improvement = 0
    no_improvement_count = 0
    
    start_time = time.time()
    iteration = 0
    
    # Precompute correlation indices
    correlation_indices = []
    for k in range(n_points):
        correlation_indices.append((k, slice(None, n_points-k) if k < n_points else slice(-k, None)))
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Compute all overlaps
        correlation = np.correlate(h, 1 - h, mode='full')
        C5 = np.max(correlation) * dx
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count > 50:
            # Random restart with 5% probability
            if np.random.rand() < 0.05:
                h = np.random.rand(n_points)
                h *= (n_points / 2) / np.sum(h)
                h = np.clip(h, 0, 1)
                no_improvement_count = 0
                
        # Calculate gradient for the worst shift
        k0 = np.argmax(correlation)
        gradient = np.zeros(n_points)
        
        if k0 < n_points:
            # dC5/dh_i = 1 - h_{i+k0} for valid indices
            if k0 < n_points:
                for i in range(n_points):
                    if i + k0 < n_points:
                        gradient[i] += 1 - h[i + k0]
                    if i >= k0:
                        gradient[i] -= h[i - k0]
        else:
            # From the right side (negative k)
            k0_shift = k0 - n_points + 1
            for i in range(n_points):
                if i + k0_shift < n_points:
                    gradient[i] += 1 - h[i + k0_shift]
                if i >= k0_shift:
                    gradient[i] -= h[i - k0_shift]
        
        # Update with momentum and gradient
        h_new = h - alpha * gradient
        
        # Apply projection
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain sum constraint
        old_sum = np.sum(h)
        new_sum = np.sum(h_new)
        h_new += (n_points / 2 - new_sum) / n_points
        
        # Ensure true constraint
        h_new = np.clip(h_new, 0, 1)
        h_new *= (n_points / 2) / np.sum(h_new)
        
        # Apply momentum update
        h = momentum * h + (1 - momentum) * h_new
        
        # Update step size
        if no_improvement_count % 10 == 0 and no_improvement_count > 0:
            alpha *= 0.95
        
        # Early stopping if no improvement for too long
        if no_improvement_count > 100:
            break
    
    # Final validation
    final_correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_C5 = np.max(final_correlation) * dx
    
    return (best_h.tolist(), final_C5, n_points)
