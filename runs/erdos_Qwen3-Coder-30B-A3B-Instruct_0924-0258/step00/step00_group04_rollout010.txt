def evaluate_C5(h_values):
    """Evaluate the C5 constant for given h values."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h_values, 1 - h_values, mode='full')
    # Take max of all correlations (excluding the zero lag which is not relevant)
    max_corr = np.max(corr[n-1:])  # Only consider lags from 0 to n-1
    return max_corr * dx

def project_and_rescale(h, target_sum):
    """Project h to [0,1] and rescale to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h
