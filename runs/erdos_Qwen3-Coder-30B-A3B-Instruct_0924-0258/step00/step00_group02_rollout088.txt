import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute overlap integral for all shifts
    overlap = correlate(h, 1 - h, mode="full") * dx
    # Return maximum overlap
    return np.max(overlap)

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points / 2 and h[i] ∈ [0, 1]"""
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * (n_points / 2) / sum_h
    return h

def create_initial_population(initial_h_values, n_points, pop_size):
    """Create initial population around the provided initial solution"""
    population = []
    # Add the initial solution
    population.append(initial_h_values.copy())
    
    # Add random perturbations
    for _ in range(pop_size - 1):
        # Perturb initial_h_values with noise
        perturbed = initial_h_values + 0.1 * np.random.randn(n_points)
        # Project to feasible region
        perturbed = project_to_feasible(perturbed, n_points)
        population.append(perturbed)
    
    return population

def single_point_crossover(parent1, parent2):
    """Perform single-point crossover between two individuals"""
    n = len(parent1)
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.1):
    """Mutate an individual by adding Gaussian noise to some elements"""
    n = len(individual)
    num_mutation = int(n * mutation_rate)
    indices = np.random.choice(n, num_mutation, replace=False)
    mutated = individual.copy()
    mutated[indices] += 0.1 * np.random.randn(num_mutation)
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 45
    pop_size = 50
    generations = 12
    mutation_rate = 0.1
    elite_size = 25
    
    # Get initial solution if available
    try:
        initial_h = initial_h_values.copy()
    except:
        # Create an initial feasible solution
        initial_h = np.ones(n_points) * 0.5
        initial_h = project_to_feasible(initial_h, n_points)
    
    # Initialize population
    population = create_initial_population(initial_h, n_points, pop_size)
    
    # Evaluate initial population
    fitness_scores = []
    for i, h in enumerate(population):
        c5 = evaluate_erdos_solution(h)
        fitness_scores.append((c5, i))
    
    # Sort population by fitness (lower is better)
    fitness_scores.sort()
    best_c5 = fitness_scores[0][0]
    best_h = population[fitness_scores[0][1]].copy()
    
    # Begin evolution
    start_time = time.time()
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness_scores = []
        for i, h in enumerate(population):
            c5 = evaluate_erdos_solution(h)
            fitness_scores.append((c5, i))
        
        # Sort by fitness
        fitness_scores.sort()
        
        # Update best
        if fitness_scores[0][0] < best_c5:
            best_c5 = fitness_scores[0][0]
            best_h = population[fitness_scores[0][1]].copy()
        
        # Select elite
        elite_indices = [idx for _, idx in fitness_scores[:elite_size]]
        elite_population = [population[i] for i in elite_indices]
        
        # Generate offspring through crossover and mutation
        offspring = []
        while len(offspring) < pop_size - elite_size:
            # Select two parents
            parent1_idx, parent2_idx = np.random.choice(elite_size, 2, replace=False)
            parent1 = elite_population[parent1_idx]
            parent2 = elite_population[parent2_idx]
            
            # Crossover
            child1, child2 = single_point_crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Project to feasible region
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            
            offspring.append(child1)
            if len(offspring) < pop_size - elite_size:
                offspring.append(child2)
        
        # Form new population
        population = elite_population + offspring
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
