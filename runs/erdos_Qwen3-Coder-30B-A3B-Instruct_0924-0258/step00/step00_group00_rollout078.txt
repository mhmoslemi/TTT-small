import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given step function h."""
    # Compute the overlap integral for all lags
    corr = correlate(h, 1 - h, mode='full')
    # The full correlation has length 2*n_points - 1
    # The valid range of lags is from 0 to n_points - 1
    # But we need max over all shifts
    max_corr = np.max(corr)
    return max_corr * dx

def rescale_h(h, dx):
    """Rescale h to maintain integral constraint."""
    target_sum = 1.0 / dx  # since ∫h(x)dx = 1
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial values if available, otherwise generate a simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Generate initial construction with simple structure
        n_points = 45
        h = np.ones(n_points) * 0.5
        # Make a binary-like construction that sums to 1
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h = rescale_h(h, 2.0 / n_points)
    
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize swarm for PSO
    nparticles = 30
    particles = [h.copy() for _ in range(nparticles)]
    velocities = [np.zeros_like(h) for _ in range(nparticles)]
    personal_best = [particles[i].copy() for i in range(nparticles)]
    personal_best_scores = [evaluate_h(particles[i], dx) for i in range(nparticles)]
    
    # Global best
    best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[best_idx].copy()
    global_best_score = personal_best_scores[best_idx]
    
    # PSO parameters
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive weight
    c2 = 2.0  # social weight
    max_iter = 5000
    stagnation_counter = 0
    stagnation_limit = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.5:
            break
            
        for i in range(nparticles):
            # Update velocity and position
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (personal_best[i] - particles[i]) + 
                             c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = rescale_h(particles[i], dx)
            
            # Evaluate
            score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = score
                # Update global best
                if score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = score
                    stagnation_counter = 0
            else:
                stagnation_counter += 1
                
        # Convergence check
        if stagnation_counter >= stagnation_limit:
            # Random restart
            for i in range(nparticles):
                particles[i] = rescale_h(np.random.rand(n_points) * 0.5 + 0.25, dx)
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = evaluate_h(particles[i], dx)
            stagnation_counter = 0
            # Re-evaluate global best
            best_idx = np.argmin(personal_best_scores)
            global_best = personal_best[best_idx].copy()
            global_best_score = personal_best_scores[best_idx]
    
    # Final check of global best
    final_score = evaluate_h(global_best, dx)
    return (global_best.tolist(), final_score, len(global_best))
