def evaluate_erdos_solution(h, dx):
    # Compute overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def optimize_step(h, n_points, dx, maxiter=200):

    # Use SLSQP for constrained optimization
    # Define bounds: [0,1] for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint for sum(h) = n_points/2
    def sum_constraint(x):
        return np.sum(x) - n_points / 2.0
    
    constraints = [{'type': 'eq', 'fun': sum_constraint}]
    
    # Run optimization
    try:
        result = minimize(
            lambda h: convex_surrogate(h, n_points, dx),
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        return project_feasible(result.x, n_points)
    except Exception:
        return h

def upsample_h(h_coarse, n_points):
    # Linear interpolation from coarse to fine
    h_fine = np.interp(
        np.linspace(0, 2, n_points),
        np.linspace(0, 2, len(h_coarse)),
        h_coarse
    )
    return project_feasible(h_fine, n_points)
