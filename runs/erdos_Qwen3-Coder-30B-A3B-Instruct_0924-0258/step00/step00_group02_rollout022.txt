import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    h = np.array(h)
    # correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1-h, mode="full")
    # compute maximum overlap
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def generate_perturbed_start(h0, n_points, noise_level=0.1):
    """Generate a perturbed version of h0 that maintains constraints"""
    h = h0 + np.random.normal(0, noise_level, n_points)
    h = project_to_feasible(h, n_points)
    return h

def compute_surrogate_gradient(h, n_points):
    """Compute gradient of surrogate objective: sum_k (h * (1-h))[k]^2 * dx^2"""
    dx = 2.0 / n_points
    h = np.array(h)
    diff = 1 - h
    # correlation of h with diff
    corr = np.correlate(h, diff, mode="full")
    # approximate gradient using Jacobian of correlation
    grad = np.zeros_like(h)
    # Gradient of sum_k corr[k]^2 is 2 * sum_k corr[k] * d corr[k]/d h
    # approximate d corr[k]/d h_i using finite differences
    for i in range(n_points):
        eps = 1e-6
        h_plus = h.copy()
        h_plus[i] += eps
        corr_plus = np.correlate(h_plus, diff, mode="full")
        grad_i = 2 * np.sum(corr * (corr_plus - corr)) / eps * dx**2
        grad[i] = grad_i
    return grad

def optimize_at_resolution(h0, n_points, max_time):
    """Optimize on a fixed resolution level"""
    dx = 2.0 / n_points
    start_time = time()
    
    # Define constraints and bounds
    bounds = [(0, 1) for _ in range(n_points)]
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        h = np.array(h)
        corr = np.correlate(h, 1-h, mode="full")
        # Use sum of squares of correlation values as convex surrogate
        return np.sum(corr**2) * dx**2 - 0.9 * np.max(corr) * dx
    
    def jacobian(h):
        return compute_surrogate_gradient(h, n_points)
    
    # Initial guess is already projected
    h = project_to_feasible(np.array(h0), n_points)
    
    # minimize with SLSQP
    res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                   jac=jacobian, options={'maxiter': 1000, 'ftol':1e-8})
    
    if res.success:
        h_final = project_to_feasible(res.x, n_points)
        c5 = evaluate_erdos_solution(h_final, n_points)
        return h_final, c5
    else:
        # fallback: return original projected version
        h_final = project_to_feasible(h, n_points)
        c5 = evaluate_erdos_solution(h_final, n_points)
        return h_final, c5
