import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h):
    # Compute the overlap integral for all shifts
    # Using correlate with mode='full' computes all shifts
    # h is already normalized to sum to n_points/2
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute integral = sum(h(x)(1-h(x+k)) dx)
    # This is equivalent to: sum(h[i] * (1 - h[i + k]) for all valid pairs)
    # For the full correlation:
    corr = correlate(h, 1 - h, mode="full")
    # The max of correlations is C5
    # Convert from discrete correlation to integral (multiply by dx)
    # The correlation is over n_points*2-1 values, and we only take the effective
    # portion that corresponds to valid overlaps, which is from indices n_points-1 to 2*n_points-2
    max_corr = np.max(corr[n_points-1:2*n_points-1]) * dx
    return max_corr

def project_to_feasible(h, n_points):
    # Ensure h sums to n_points/2 and all values are in [0,1]
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        # If all are zero, initialize with uniform values
        h = np.full(n_points, 1.0 / 2)
    return h
