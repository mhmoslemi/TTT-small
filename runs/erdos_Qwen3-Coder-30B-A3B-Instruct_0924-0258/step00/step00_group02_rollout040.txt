import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for given step function h"""
    dx = 2.0 / len(h)
    # Compute convolution h * (1 - h) with full mode
    conv = correlate(h, 1 - h, mode="full")
    # Scale by dx and return maximum
    return np.max(conv) * dx

def project_feasible(h, n_points):
    """Project h onto feasible region: [0,1] and sum = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    old_sum = np.sum(h)
    if old_sum > 0:
        h = h * (n_points / 2) / old_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    if 'initial_h_values' in globals():

    else:

        n_points = 45
        h_init = np.ones(n_points) * 0.5
    n_points = len(h_init)
    
    # PSO Parameters
    n_particles = 30
    inertia_weight = 0.9
    c1 = 2.0
    c2 = 2.0
    velocity_max = 0.5
    
    # Initialize particles
    particles = []
    velocities = []
    
    # First particle: initial solution
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Remaining particles: perturbed versions
    for i in range(1, n_particles):
        h_perturb = h_init + np.random.uniform(-0.1, 0.1, n_points)
        h_perturb = project_feasible(h_perturb, n_points)
        particles.append(h_perturb)
        velocities.append(np.random.uniform(-0.2, 0.2, n_points))
    
    # Evaluate initial solutions
    fitness = [evaluate_erdos_solution(p) for p in particles]
    
    # Track personal best
    pBest = particles.copy()
    pBest_fitness = fitness.copy()
    
    # Track global best
    gBest_idx = np.argmin(fitness)
    gBest = particles[gBest_idx].copy()
    gBest_fitness = fitness[gBest_idx]
    
    start_time = time.time()
    
    # PSO main loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Update inertia weight (time decay)
        inertia_weight *= 0.99
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                            c1 * r1 * (pBest[i] - particles[i]) + 
                            c2 * r2 * (gBest - particles[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply feasibility projection
            particles[i] = project_feasible(particles[i], n_points)
            
            # Evaluate fitness
            new_fitness = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if new_fitness < pBest_fitness[i]:
                pBest[i] = particles[i].copy()
                pBest_fitness[i] = new_fitness
            
            # Update global best
            if new_fitness < gBest_fitness:
                gBest = particles[i].copy()
                gBest_fitness = new_fitness
    
    # Final evaluation of best solution
    c5_bound = evaluate_erdos_solution(gBest)
    return (gBest, c5_bound, n_points)
