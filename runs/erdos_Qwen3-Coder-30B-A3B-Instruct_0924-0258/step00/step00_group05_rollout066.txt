def compute_c5_direct(h, n_points, dx):
    """Compute C5 directly using correlate"""
    h1 = 1 - h
    corr = correlate(h, h1, mode='full')
    # compute max over all shifts
    max_overlaps = corr[-n_points+1:]
    return np.max(max_overlaps) * dx

def evaluate_erdos_solution(h, n_points, dx):
    """Evaluate solution"""
    h1 = 1 - h
    corr = correlate(h, h1, mode='full')
    max_overlaps = corr[-n_points+1:]
    return np.max(max_overlaps) * dx
