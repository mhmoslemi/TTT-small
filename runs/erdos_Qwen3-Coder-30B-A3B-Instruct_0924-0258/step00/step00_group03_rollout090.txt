import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h):
    """Evaluate the maximum overlap integral for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for different shifts
    correlations = correlate(h, 1 - h, mode='full')
    # Only consider relevant parts (the ones that make sense for overlap)
    # Maximum is the highest correlation, but we are looking at max k
    max_overlap = np.max(correlations)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2  # sum(h) should be n_points / 2
    
    # Initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a random feasible starting point
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
    
    # Ensure bounds and sum
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * target_sum
    
    # Current best
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    # Annealing parameters
    T0 = 1.0
    alpha = 0.999
    sigma0 = 0.1
    t = 0
    start_time = time.time()
    
    def project(h):
        """Project h to feasible region."""
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
        return h
    
    while time.time() - start_time < budget_s:
        t += 1
        T = T0 * (alpha ** t)
        sigma = sigma0 * (alpha ** (t / 200))
        
        # Mutate
        h_new = h + np.random.randn(n_points) * sigma
        h_new = project(h_new)
        
        # Evaluate
        C5_new = evaluate_C5(h_new)
        
        # Accept or reject
        if C5_new < best_C5 or np.random.rand() < np.exp(-(C5_new - best_C5) / T):
            h = h_new
            if C5_new < best_C5:
                best_h = h_new.copy()
                best_C5 = C5_new
        
        # Occasionally print progress
        if t % 100 == 0:
            if time.time() - start_time > budget_s * 0.95:
                break
    
    # Final check
    final_C5 = evaluate_C5(best_h)
    
    return (list(best_h), float(final_C5), n_points)
