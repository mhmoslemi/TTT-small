        # Project h onto box constraints [0,1]
        h_proj = np.clip(h_vec, 0, 1)
        # Then adjust to meet sum constraint using CVXPY
        h_var = cp.Variable(n_points)
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        prob = cp.Problem(cp.Minimize(cp.sum_squares(h_var - h_proj)), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return h_var.value
        except:
            # Fallback to simple projection if CVXPY fails
            h_proj = np.clip(h_vec, 0, 1)
            current_sum = np.sum(h_proj)
            if current_sum > target_sum:
                # Distribute excess evenly among largest elements
                excess = current_sum - target_sum
                sort_idx = np.argsort(h_proj)[::-1]
                for i in sort_idx:
                    if excess <= 0: break
                    take = min(excess, h_proj[i])
                    h_proj[i] -= take
                    excess -= take
