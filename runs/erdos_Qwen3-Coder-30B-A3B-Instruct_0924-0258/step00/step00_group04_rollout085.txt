import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute full correlation
    corr = correlate(h, 1 - h, mode="full")
    # Take the maximum value and scale it
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_bounds(h):
    """Project h to [0, 1] bounds."""
    return np.clip(h, 0, 1)

def rescale_to_constraint(h, n_points):
    """Rescale h to satisfy ∫₀² h(x) dx = 1."""
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

    perturbation = np.random.normal(0, noise_scale, n_points)
    h_plus = project_to_bounds(h + perturbation)
    h_minus = project_to_bounds(h - perturbation)
    
    h_plus = rescale_to_constraint(h_plus, n_points)
