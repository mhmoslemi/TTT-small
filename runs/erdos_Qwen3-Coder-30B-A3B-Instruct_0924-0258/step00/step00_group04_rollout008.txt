import numpy as np
from scipy.optimize import minimize
import time
from copy import deepcopy

def evaluate_h(h):
    """Evaluate C5 for a given h using the full correlation."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Take the maximum value (excluding trivial zeros)
    max_corr = np.max(corr[n//2:])  # Only consider valid overlaps
    return max_corr * dx

def project_and_rescale(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def mutate(h, n_points, mutation_rate=0.05):
    """Apply mutation to a solution h"""
    h = h.copy()
    num_mutate = max(1, int(len(h) * mutation_rate))
    indices = np.random.choice(len(h), size=num_mutate, replace=False)
    for i in indices:
        h[i] += np.random.normal(0, 0.1)
    return project_and_rescale(h, n_points)

def crossover(h1, h2, n_points):
    """Uniform crossover between two solutions"""
    h1 = h1.copy()
    h2 = h2.copy()
    # Choose crossover points
    n_crossover = np.random.randint(2, 4)
    points = sorted(np.random.choice(len(h1), size=n_crossover, replace=False))
    points = [0] + points + [len(h1)]
    
    result = np.zeros_like(h1)
    for i in range(len(points) - 1):
        start, end = points[i], points[i+1]
        if i % 2 == 0:
            result[start:end] = h1[start:end]
        else:
            result[start:end] = h2[start:end]
    return project_and_rescale(result, n_points)

def greedy_optimize(h, n_points, maxiter=10):
    """Use local optimization on a candidate"""
    def objective(x):
        h_test = project_and_rescale(x, n_points)
        return evaluate_h(h_test)
    
    # Flatten constraints to pass to minimize
    bounds = [(0, 1) for _ in range(len(h))]
    # Use scipy minimize with L-BFGS-B
    # We'll use a random subset for faster evaluation if needed
    res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': maxiter, 'ftol': 1e-6})
    return project_and_rescale(res.x, n_points)

    # Initialize population
    pop_size = 50
    population = []
    
    if 'initial_h_values' in globals():

        # Validate and add

            population.append(initial_h)
        else:
            initial_h = np.random.rand(n_points)
            initial_h = project_and_rescale(initial_h, n_points)
            population.append(initial_h)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = project_and_rescale(initial_h, n_points)
        population.append(initial_h)
    
    # Fill remaining with random candidates
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = project_and_rescale(h, n_points)
        population.append(h)
    
    # Keep track of best solution
    best_h = population[0].copy()
    best_c5 = evaluate_h(best_h)
    
    start_time = time.time()
    generation = 0
    stagnation_count = 0
    mutation_rate = 0.05
    
    while time.time() - start_time < budget_s * 0.95:
        generation += 1
        # Evaluate population
        fitness = []
        for h in population:
            score = evaluate_h(h)
            fitness.append(score)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        current_best_c5 = fitness[0]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Adapt mutation rate if stagnation detected
        if stagnation_count > 50:
            mutation_rate = 0.1

        # Select parents (top 20%)
        num_parents = max(1, pop_size // 5)
        parents = population[:num_parents]

        # Create offspring
        offspring = []
        while len(offspring) < pop_size - num_parents:
            # Crossover two random parents
            idx1, idx2 = np.random.choice(len(parents), size=2, replace=False)
            child = crossover(parents[idx1], parents[idx2], n_points)
            # Mutate
            if np.random.rand() < 0.5:
                child = mutate(child, n_points, mutation_rate)
            offspring.append(child)
            
        # Greedy local optimization on top 10%
        num_top = max(1, pop_size // 10)
        for i in range(num_top):
            if np.random.rand() < 0.3:
                optimized = greedy_optimize(population[i], n_points)
                offspring.append(optimized)

        # Keep new offspring
        population = parents + offspring[:pop_size - len(parents)]
        
        # Rescale all to ensure constraints
        for i in range(len(population)):
            population[i] = project_and_rescale(population[i], n_points)
    
    # Final evaluation of best solution
    final_c5 = evaluate_h(best_h)
    return (list(best_h), final_c5, n_points)
