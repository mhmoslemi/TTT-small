import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
    else:
        h = np.ones(n_points) * 0.5
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1:
        # Local optimization step
        # We need to compute the max correlation
        def objective(h_vec):
            # Evaluate C5 for current h_vec
            h = h_vec.copy()
            c5 = evaluate_c5(h, dx)
            return c5
        
        # Use L-BFGS-B with bounds [0,1]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Let's compute a better gradient estimate or just use the evaluation directly
        # Using a simplified approach - we'll compute gradient numerically, but
        # that's expensive, so we'll do less frequent local optimizations
        
        # Use a direct gradient-based method for better initial convergence:
        # Starting optimization with trust-constr for better handling
        
        try:
            result = minimize(objective, h.copy(), method='L-BFGS-B', bounds=bounds, 
                            options={'maxiter': 20})
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                h_sum = np.sum(h_new)
                h_new = h_new * target_sum / h_sum
                h = h_new.copy()
            else:
                # Reset to last good solution if optimization failed
                h = best_h.copy()
        except:
            # On error, revert to last best
            h = best_h.copy()
            
        # Apply Gaussian smoothing with σ = 0.1 * n_points
        sigma = 0.1 * n_points
        h_smooth = gaussian_filter1d(h, sigma=sigma, mode='wrap')
        h_smooth = np.clip(h_smooth, 0, 1)
        h_sum = np.sum(h_smooth)
        h = h_smooth * target_sum / h_sum
        
        # Evaluate new C5
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early exit for very small improvement or by budget
        if best_c5 <= 0.3807:
            break
            
    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_c5(h, dx):
    # Compute correlations for all lags
    # Using numpy correlate with mode="full"
    # But we want the maximum of (h * (1 - h_shifted)) integral
    h_full = np.concatenate([h, h])  # Duplicate to avoid wrap-around issues
    max_c = 0.0
    
    for k in range(len(h)):
        # This handles cross-correlation: h[i] * (1 - h[(i+k)%n])
        h_shift = np.roll(h, k)
        product = h * (1.0 - h_shift)
        c = np.sum(product) * dx
        max_c = max(max_c, c)
    
    return max_c
