```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import convolve
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_overlap(h, dx):
    # Compute all correlations of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum is over all lag indices
    max_corr = np.max(corr)
    return max_corr * dx

def compute_c5(h_values, n_points):
    dx = 2.0 / n_points
    return evaluate_overlap(h_values, dx)

def initialize_h(n_points):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if not (np.all(h >= 0) and np.all(h <= 1)):
            h = np.clip(h, 0, 1)
        if abs(np.sum(h) - n_points / 2.0) > 1e-10:
            h = h * (n_points / 2.0) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    return h

def smooth_and_rescale(h, sigma=0.1):
    h_smooth = gaussian_filter1d(h, sigma=sigma)
    h_smooth = np.clip(h_smooth, 0, 1)
    scale = (len(h) / 2.0) / np.sum(h_smooth)
    h_smooth = h_smooth * scale
    return h_smooth

def objective(h, n_points, dx):
    # This function estimates the C5 value using finite differences
    # We are maximizing C5, so we return negative for minimization
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    return -max_val * dx

def constraint_sum(h, n_points):
    dx = 2.0 / n_points
    return np.sum(h) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    h = initialize_h(n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    start_time = time.time()
    
    tol = 1e-6
    
    # Buffers for constraints
    cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, n_points)}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    while time.time() - start_time < budget_s - 0.1:
        # Local optimization step
        result = minimize(
            objective, 
            h, 
            args=(n_points, dx),
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': tol, 'gtol': tol, 'maxiter': 50}
        )
        
        if result.success:
            h = result.x
        else:
            # fallback: slightly perturb and retry
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2.0) / np.sum(h)
        
        # Low-pass filtering for smoothing
        sigma = 0.1 * n_points
        h = smooth_and_rescale(h, sigma)
        
        # Evaluate current solution
        c5 = compute_c5(h, n_points)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Early termination if we're close to the target
        if best_c5 < 0.3807:
            break
    
    h_final = best_h.copy()
    c5_bound = best_c5
    
    return (h_final.tolist(), c5_bound, n_points)
```
