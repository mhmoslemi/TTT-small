import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def evaluate_c5(h):
    """Compute the maximum overlap integral C5 for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all cross-correlations
    correlations = np.correlate(h, 1 - h, mode="full")
    # Take only the part corresponding to valid overlaps
    # The valid range corresponds to lags k in [0, n-1]
    # which are indices n-1 to 2*n-2 in full correlation
    valid_correlations = correlations[n-1:2*n-1]
    # Compute C5
    c5 = max(valid_correlations) * dx
    return c5

def project_to_feasible(h):
    """Project h to [0,1] and rescale to satisfy sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def get_gradient(h, eps=1e-5):
    """Get approximate gradient of C5 with respect to h using finite differences"""
    n = len(h)
    dx = 2.0 / n
    c5_base = evaluate_c5(h)
    grad = np.zeros(n)
    for i in range(n):
        h_eps = h.copy()
        h_eps[i] += eps
        h_eps = project_to_feasible(h_eps)
        c5_plus = evaluate_c5(h_eps)
        grad[i] = (c5_plus - c5_base) / eps
    return grad

def optimize_step(h, max_iter=100):
    """Perform a local optimization step using L-BFGS"""
    n = len(h)
    dx = 2.0 / n
    def objective(h_vec):
        h_proj = project_to_feasible(h_vec)
        return evaluate_c5(h_proj)
    
    def grad(h_vec):
        h_proj = project_to_feasible(h_vec)
        return get_gradient(h_proj)
    
    # Try different optimization steps
    try:
        sol = minimize(objective, h, method='L-BFGS-B', jac=grad, 
                       bounds=[(0,1)]*n, options={'maxiter': max_iter})
        if sol.success:
            h_new = sol.x
        else:
            h_new = h
    except:
        h_new = h
    h_new = project_to_feasible(h_new)
    return h_new

def smooth_and_rescale(h, sigma_factor=0.1):
    """Apply Gaussian smoothing and rescale to maintain constraints"""
    n = len(h)
    sigma = sigma_factor * n
    h_smooth = gaussian_filter1d(h, sigma=sigma)
    h_smooth = project_to_feasible(h_smooth)
    return h_smooth

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up problem parameters
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Check if it satisfies constraints, if not, reinitialize
        if not (np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - n_points / 2) < 1e-10):
            h = np.full(n_points, 0.5)
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure sum constraint
    h = project_to_feasible(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Local optimization
        h_opt = optimize_step(h, max_iter=50)
        # Smoothing
        h_smooth = smooth_and_rescale(h_opt, sigma_factor=0.1)
        # Evaluate
        c5_current = evaluate_c5(h_smooth)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_smooth.copy()
        # Move to new h
        h = h_smooth.copy()
        
        # Early termination if we're getting close to target
        if best_c5 <= 0.3807:
            break
            
    return (list(best_h), best_c5, n_points)
