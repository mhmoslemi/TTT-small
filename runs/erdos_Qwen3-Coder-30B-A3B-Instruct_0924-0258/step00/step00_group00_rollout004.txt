```python
import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral C5 for a given h."""
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Take the maximum value among all lags
    max_corr = np.max(corr)
    return max_corr * dx

def rescale_h(h, target_sum):
    """Rescale h to have the given target sum."""
    current_sum = np.sum(h)
    if current_sum == 0:
        return h
    return h * (target_sum / current_sum)

def clip_and_rescale(h, target_sum):
    """Clip h to [0,1] and rescale to target sum."""
    h = np.clip(h, 0, 1)
    return rescale_h(h, target_sum)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2  # sum(h) * dx = 1 => sum(h) = n_points / 2
    
    # Initialize h using a potentially better starting point
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        h = clip_and_rescale(h, target_sum)
    else:
        # Default construction: binary-like structure
        h = np.random.rand(n_points)
        h = clip_and_rescale(h, target_sum)
    
    best_h = h.copy()
    best_C5 = evaluate_h(h, dx)
    best_time = time.time()
    
    # PSO parameters
    n_particles = 30
    w = 0.7  # Inertia weight
    c1 = c2 = 2.0  # Learning rates
    max_iter = 1000
    no_improvement_limit = 100
    no_improvement_count = 0
    
    # Initialize particles
    particles = [h.copy() for _ in range(n_particles)]
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [evaluate_h(p, dx) for p in particles]
    
    global_best = best_h.copy()
    global_best_score = best_C5
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.98:
            break
            
        for i in range(n_particles):
            # Update particle
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            pbest_h = personal_best[i]
            gbest_h = global_best
            
            v = w * v + c1 * r1 * (pbest_h - h) + c2 * r2 * (gbest_h - h)
            
            # Update position
            h += v
            
            # Clip and rescale
            h = clip_and_rescale(h, target_sum)
            
            # Store updated particle
            particles[i] = h
            
            # Evaluate
            score = evaluate_h(h, dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = h.copy()
                personal_best_scores[i] = score
                
                # Update global best if needed
                if score < global_best_score:
                    global_best = h.copy()
                    global_best_score = score
                    no_improvement_count = 0
                    if score < best_C5:
                        best_h = h.copy()
                        best_C5 = score
                        best_time = time.time()
                else:
                    no_improvement_count += 1
                    
            # Check for convergence
            if no_improvement_count >= no_improvement_limit:
                break
                
        if no_improvement_count >= no_improvement_limit:
            break
    
    # Final evaluation
    final_C5 = evaluate_h(best_h, dx)
    
    return (list(best_h), final_C5, n_points)
```
