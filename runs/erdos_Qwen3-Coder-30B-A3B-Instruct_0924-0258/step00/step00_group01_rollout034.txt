import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback to uniform distribution
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[0] = 0.45
        h[-1] = 0.55
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Gradient approximation using finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            h_minus[i] = max(0.0, h[i] - eps)
            c5_plus = evaluate_c5(h_plus, dx)
            c5_minus = evaluate_c5(h_minus, dx)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update direction: negative gradient
        update_dir = -grad
        # Project to ensure sum is preserved
        update_dir = update_dir - np.mean(update_dir)
        
        # Line search
        alpha = 1.0
        step_size = 0.1
        
        for _ in range(10):
            new_h = h + alpha * update_dir
            # Clamp to bounds
            new_h = np.clip(new_h, 0, 1)
            # Rescale
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            new_c5 = evaluate_c5(new_h, dx)
            if new_c5 < best_c5:
                h = new_h
                best_c5 = new_c5
                best_h = h.copy()
                break
            else:
                alpha *= 0.5
        
        # If no improvement, reduce step size to avoid getting stuck
        if iteration % 100 == 0:
            step_size *= 0.9
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute correlation between h and 1-h
    corr = np.correlate(h, 1 - h, mode="full")
    # Scale and return the maximum value
    return np.max(corr) * dx
