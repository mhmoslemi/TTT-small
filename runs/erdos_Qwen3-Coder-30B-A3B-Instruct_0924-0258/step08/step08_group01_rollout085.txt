import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 0
    max_generations = int(budget_s * 0.7 / (n_points * 1e-4)) if n_points > 0 else 1000
    no_improvement_count = 0
    max_no_improvement = 200
    mutation_step = 0.05
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.1
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Function to ensure sum constraint
    def enforce_sum(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Function to get fitness for population
    def evaluate_population(pop):
        fitness = []
        for h in pop:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        return np.array(fitness)
    
    # Simple tournament selection
    def tournament_selection(pop, fitness, k=3):
        indices = np.random.choice(len(pop), k)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        return pop[best_idx].copy()
    
    # Blend crossover
    def blend_crossover(h1, h2, alpha=0.5):
        h_child = alpha * h1 + (1 - alpha) * h2
        return h_child
    
    # Mutation
    def mutate(h, rate, step):
        mask = np.random.rand(len(h)) < rate
        h = h.copy()
        if np.any(mask):
            delta = np.random.uniform(-step, step, size=len(h)) * mask
            h = h + delta
        return h
    
    # Main GA loop
    while time() - start_time < budget_s * 0.95 and generations < max_generations:
        # Evaluate fitness
        fitness = evaluate_population(population)
        
        # Track best
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_c5 = fitness[min_idx]
            best_h = population[min_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            break
        
        # Selection + crossover + mutation
        new_population = []
        
        # Elitism: keep best half
        elite_indices = np.argsort(fitness)[:population_size // 2]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate new population
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            if np.random.rand() < crossover_rate:
                # Crossover
                alpha = np.random.rand() * 0.4 + 0.3  # Alpha in [0.3, 0.7]
                child = blend_crossover(parent1, parent2, alpha)
            else:
                # Just inherit one parent
                child = parent1.copy()
            
            # Mutation
            child = mutate(child, mutation_rate, mutation_step)
            child = enforce_sum(child)
            new_population.append(child)
        
        population = new_population[:population_size]
        
        # Periodic SLSQP refinement
        if generations % 100 == 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 500, 'ftol': 1e-10})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        
        generations += 1
        mutation_step *= 0.98  # Decay mutation
    
    # Final SLSQP refinement on best
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
