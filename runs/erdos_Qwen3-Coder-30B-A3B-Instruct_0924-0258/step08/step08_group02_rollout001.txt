import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter_sa = 100000
    epsilon = 0.1
    
    # Initial temperature and parameters
    T = T_init
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    c5_current = evaluate_erdos_solution(h_current, dx)
    
    # Keep track of best
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Annealing iterations
    for i in range(max_iter_sa):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb two random indices
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        h_new = h_current.copy()
        
        # Random perturbation
        delta = np.random.uniform(-epsilon, epsilon)
        h_new[idx1] += delta
        h_new[idx2] -= delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Compute new c5
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        delta_c5 = c5_new - c5_current
        
        if delta_c5 < 0:
            # Always accept improvement
            h_current = h_new
            c5_current = c5_new
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-delta_c5 / T):
                h_current = h_new
                c5_current = c5_new
        
        # Cool down
        T = max(T * alpha, T_min)
    
    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)
        
        # Optimize from best found
        res = minimize(
            objective,
            h_best,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        
        if res.success:
            h_refined = res.x
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < c5_best:
                h_best = h_refined
                c5_best = c5_refined
                
    except Exception:
        pass

    return (list(h_best), c5_best, n_points)
