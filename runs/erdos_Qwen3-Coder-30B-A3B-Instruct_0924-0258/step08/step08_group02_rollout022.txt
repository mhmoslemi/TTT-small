import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create binary-like structure with alternating 0s and 1s
        h_start = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                h_start[i] = 1.0
            else:
                h_start[i] = 0.0
        # Add some randomization to break symmetry
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize threshold and cooling parameters
    threshold = 0.1
    cooling_rate = 0.95
    min_threshold = 0.01
    max_iterations = 500
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps and find maximum overlap lag
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        max_overlap = np.max(overlaps)
        k_max = np.argmax(overlaps)
        
        # Compute gradient contribution for each coordinate
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient[i] = -2 * dx * (1 - best_h[j] - best_h[i])
        
        # Sort indices by gradient magnitude
        sorted_indices = np.argsort(np.abs(gradient))[::-1]
        
        # Select top 50% of coordinates to threshold
        top_k = n_points // 2
        indices_to_threshold = sorted_indices[:top_k]
        
        # Make a copy of current solution for thresholding
        h_new = best_h.copy()
        
        # Apply thresholding
        for i in indices_to_threshold:
            if h_new[i] >= 1 - threshold:
                h_new[i] = 1.0
            elif h_new[i] <= threshold:
                h_new[i] = 0.0
        
        # Ensure sum constraint by redistribution
        current_sum = np.sum(h_new)
        if abs(current_sum - n_points / 2) > 1e-6:
            # Redistribute excess/deficit 
            diff = (n_points / 2) - current_sum
            if diff > 0:
                # Need to increase values
                # Increase by placing extra mass to smallest values
                sort_indices = np.argsort(h_new)
                for i in range(int(diff / 0.1 + 1)):  # Ensure we have enough
                    if i < len(sort_indices):
                        idx = sort_indices[i]
                        increase = min(0.1, diff)
                        h_new[idx] = min(1.0, h_new[idx] + increase)
                        diff -= increase
                        if diff <= 0:
                            break
            else:
                # Need to decrease values
                diff = -diff
                sort_indices = np.argsort(h_new)[::-1]
                for i in range(int(diff / 0.1 + 1)):
                    if i < len(sort_indices):
                        idx = sort_indices[i]
                        decrease = min(0.1, diff)
                        h_new[idx] = max(0.0, h_new[idx] - decrease)
                        diff -= decrease
                        if diff <= 0:
                            break

        # Ensure final constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate and compare
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        # Cool threshold
        threshold = max(min_threshold, threshold * cooling_rate)
        
        # Early stopping condition if threshold is too low
        if threshold < min_threshold:
            break
    
    # Coordinate-wise refinement
    for i in range(200):  # Iterations
        if time() - start_time > budget_s * 0.95:
            break
        h_refined = best_h.copy()
        
        # Golden section search on each coordinate
        # Only optimize a few coordinates at a time to reduce computation
        indices_to_optimize = np.random.choice(n_points, size=min(5, n_points), replace=False)
        
        for idx in indices_to_optimize:
            # Golden section search to find optimal h[idx]
            def objective(val):
                h_test = h_refined.copy()
                h_test[idx] = val
                h_test = h_test * (n_points/2) / np.sum(h_test)
                overlaps = np.correlate(h_test, 1 - h_test, mode='full') * dx
                return np.max(overlaps)
            
            # Small optimization around current value using golden section
            # For simplicity, use a binary search with small steps
            left, right = 0.0, 1.0
            for _ in range(20):
                mid1 = left + (right - left) / 3
                mid2 = right - (right - left) / 3
                
                if objective(mid1) < objective(mid2):
                    right = mid2
                else:
                    left = mid1
                    
            new_val = (left + right) / 2
            h_refined[idx] = new_val
            
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        new_c5 = evaluate_erdos_solution(h_refined, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_refined.copy()
    
    # Final L-BFGS refinement
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)
        
        res = minimize(
            objective,
            best_h.copy(),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-12}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
