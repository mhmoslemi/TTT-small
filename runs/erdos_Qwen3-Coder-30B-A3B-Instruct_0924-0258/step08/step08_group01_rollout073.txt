import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale optimization
    max_scales = int(np.log2(n_points)) + 3
    current_h = h_start.copy()
    current_c5 = best_c5
    scale = 0
    mu = 0.05
    
    while scale < max_scales and (time() - start_time) < budget_s * 0.95:
        current_n = 2 ** scale
        if current_n < n_points:
            # Upsample current solution
            current_h = np.interp(np.linspace(0, 1, current_n), 
                                 np.linspace(0, 1, len(current_h)), 
                                 current_h)
        elif current_n > n_points:
            # Downsample current solution
            current_h = current_h[::current_n // n_points] if current_n // n_points > 0 else current_h[:n_points]
        
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (current_n / 2) / np.sum(current_h)
        
        # Perturb with noise
        perturbed_h = current_h + mu * (2 * np.random.random(len(current_h)) - 1)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (current_n / 2) / np.sum(perturbed_h)
        
        # Evaluate
        perturbed_c5 = evaluate_erdos_solution(perturbed_h, 2.0 / current_n)
        
        if perturbed_c5 < current_c5:
            current_h = perturbed_h
            current_c5 = perturbed_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        # Local SQP refinement
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / len(h))
        
        def constraint_sum(h):
            return np.sum(h) - len(h) / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(len(current_h))]
        
        try:
            res = minimize(objective, current_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-12})
            if res.success:
                refined_h = res.x
                refined_c5 = objective(refined_h)
                if refined_c5 < current_c5:
                    current_h = refined_h
                    current_c5 = refined_c5
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = current_h.copy()
        except Exception:
            pass
        
        # Update parameters
        mu *= 0.95
        scale += 1
    
    # Final refinement if needed
    if (time() - start_time) < budget_s * 0.1:
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / len(h))
        
        def constraint_sum(h):
            return np.sum(h) - len(h) / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(len(best_h))]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-14})
            if res.success:
                final_h = res.x
                final_c5 = objective(final_h)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except Exception:
            pass
    
    # Ensure final constraints are met exactly
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (len(best_h) / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, len(best_h))
