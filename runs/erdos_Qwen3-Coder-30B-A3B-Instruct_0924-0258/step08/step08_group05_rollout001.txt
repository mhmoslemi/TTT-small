import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Use initial construction if available

        h_start[5:15] = 1.0  # Step up from index 5 to 14
        h_start[15:25] = 0.0  # Step down from index 15 to 24
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    # Renormalize to satisfy ∫ h(x) dx = 1
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Hyperparameters
    max_iter = 10000
    step_size = 0.02
    tolerance = 1e-10
    no_improvement_limit = 100
    
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps for all lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        lag_indices = np.arange(len(corr))
        lags = lag_indices - (len(h_current) - 1)
        overlap_values = corr * dx
        
        # Find the lag with maximum overlap
        max_idx = np.argmax(overlap_values)
        max_lag = lags[max_idx]
        
        # Compute gradient of overlap w.r.t. h for specific lag
        # Using finite differences approach
        h_gradient = np.zeros_like(h_current)
        eps = 1e-5
        
        # Finite difference approximation
        for i in range(len(h_current)):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            overlap_plus = corr_plus * dx

            overlap_minus = corr_minus * dx
            
            h_gradient[i] = (overlap_plus[max_idx] - overlap_minus[max_idx]) / (2 * eps)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(h_gradient)
        if grad_norm > 0:
            h_gradient = h_gradient / grad_norm

        sum_h = np.sum(h_current)
        projection = (n_points / 2 - sum_h) / len(h_current)

        h_new = h_current - step_size * h_gradient + step_size * projection
        
        # Clamp values to [0,1] and re-normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)

        c5_new = evaluate_erdos_solution(h_new, dx)

    def objective(h):
        return evaluate_erdos_solution(h, dx)

            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
