import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Population size
    pop_size = 50
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a balanced initial solution
        h_start = np.zeros(n_points)
        half = n_points // 2
        h_start[:half] = 1.0
        if n_points % 2 == 1:
            h_start[half] = 0.5
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # Initialize population
    population = []
    population.append(h_start.copy())
    
    # Fill rest of population with diverse solutions
    for i in range(1, pop_size):
        while True:
            # Create random solution with same sum constraint
            h = np.random.rand(n_points)
            h = h * (n_points/2) / np.sum(h)
            # Ensure bounds
            h = np.clip(h, 0, 1)
            # Check sum constraint
            if abs(np.sum(h) - n_points/2) < 1e-6:
                population.append(h.copy())
                break
    
    # Helper function to adjust sum constraint
    def normalize(h):
        h = h * (n_points/2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fitness.append(evaluate_erdos_solution(h, dx))
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Mutation and crossover operations
    def mutate(h):
        h = h.copy()
        # Flip mutation
        idx1, idx2 = random.sample(range(n_points), 2)
        h[idx1] = min(1.0, max(0.0, h[idx1] + 0.1))
        h[idx2] = min(1.0, max(0.0, h[idx2] - 0.1))
        
        # Perturb mutation (random scaling)
        indices = random.sample(range(n_points), max(1, int(0.1 * n_points)))
        for i in indices:
            factor = random.uniform(0.9, 1.1)
            h[i] = min(1.0, max(0.0, h[i] * factor))
        return normalize(h)
    
    def crossover(h1, h2):
        # Uniform crossover
        child = np.zeros(n_points)
        for i in range(n_points):
            if random.random() < 0.5:
                child[i] = h1[i]
            else:
                child[i] = h2[i]
        return normalize(child)
    
    def tournament_selection(population, fitness, k=5):
        selected = []
        for _ in range(2):  # Select 2 parents
            tournament = []
            for _ in range(k):
                idx = random.randint(0, len(population)-1)
                tournament.append((idx, fitness[idx]))
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0])
        return selected[0], selected[1]
    
    # Main evolution loop
    iteration = 0
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Selection
        parent1_idx, parent2_idx = tournament_selection(population, fitness)
        parent1 = population[parent1_idx]
        parent2 = population[parent2_idx]
        
        # Crossover
        child = crossover(parent1, parent2)
        
        # Mutation
        if random.random() < 0.8:  # 80% chance of mutation
            child = mutate(child)
        
        # Evaluate child
        child_fitness = evaluate_erdos_solution(child, dx)
        
        # Replacement: Replace worst individual with child
        worst_idx = np.argmax(fitness)
        population[worst_idx] = child
        fitness[worst_idx] = child_fitness
        
        # Update best solution
        if child_fitness < best_c5:
            best_c5 = child_fitness
            best_h = child.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            # Increase mutation rate if no improvement
            if random.random() < 0.3:
                # Try a different mutation approach
                idx = random.randint(0, n_points-1)
                child[idx] = random.uniform(0, 1)
                child = normalize(child)
                child_fitness = evaluate_erdos_solution(child, dx)
                if child_fitness < best_c5:
                    best_c5 = child_fitness
                    best_h = child.copy()
            no_improvement_count = 0
            
        # Break early if we have spent most time
        if iteration % 100 == 0 and time() - start_time > budget_s * 0.9:
            break
    
    # Final refinement using local search with SLSQP
    try:
        from scipy.optimize import minimize
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
