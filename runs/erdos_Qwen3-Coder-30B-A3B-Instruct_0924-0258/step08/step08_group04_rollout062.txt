import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # Parameters for PSO
    num_particles = 75
    inertia_start = 0.8
    inertia_end = 0.4
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    perturbation_rate = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    particle_best = []
    particle_best_values = []
    
    for _ in range(num_particles):
        h = h_start.copy()
        if 'initial_h_values' not in globals():
            # Add small random perturbations
            h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        particle_best.append(h.copy())
        particle_best_values.append(evaluate_erdos_solution(h, dx))
    
    # Global best solution tracking
    global_best_idx = np.argmin(particle_best_values)
    global_best = particle_best[global_best_idx].copy()
    global_best_value = particle_best_values[global_best_idx]
    
    max_iter = int(1e6)
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Update inertia linearly
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_count / max_iter)
        
        for i in range(num_particles):
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            cognitive = cognitive_weight * r1 * (particle_best[i] - particles[i])
            social = social_weight * r2 * (global_best - particles[i])
            
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible set
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate fitness
            current_value = evaluate_erdos_solution(particles[i], dx)
            
            # Update particle best
            if current_value < particle_best_values[i]:
                particle_best[i] = particles[i].copy()
                particle_best_values[i] = current_value
                
                # Update global best
                if current_value < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = current_value
        
        iter_count += 1
        
        # Occasionally perturb particles to escape local minima
        if iter_count % 1000 == 0:
            for i in range(num_particles):
                if np.random.rand() < perturbation_rate:
                    particles[i] += np.random.normal(0, 0.02, n_points)
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
        # Check if we're stagnating
        if iter_count > 5000 and iter_count % 1000 == 0:
            if all(abs(particle_best_values[i] - global_best_value) < 1e-6
                   for i in range(num_particles)):
                # Need to perturb the global best and particles
                global_best += np.random.normal(0, 0.01, n_points)
                global_best = np.clip(global_best, 0, 1)
                global_best = global_best * (n_points/2) / np.sum(global_best)
                
                for i in range(num_particles):
                    particles[i] = global_best.copy()
    
    # Final local refinement on global best
    h_final = global_best.copy()
    
    return (list(h_final), global_best_value, n_points)
