import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Parameters for convex relaxation approach
    max_iterations = 2000
    k_top = 10  # Number of top overlaps to consider in each convex relax
    filter_width = 5
    
    def apply_low_pass_filter(h, width):
        """Apply a simple moving average filter"""
        if width <= 0:
            return h
        kernel = np.ones(width) / width
        return np.convolve(h, kernel, mode='same')
    
    def convex_optimize_top_k(h, k):
        """Solve convex relaxation considering top k overlaps"""
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        # Get indices of top k largest absolute values
        top_k_indices = np.argsort(np.abs(corr))[-k:]
        
        # Form the convex problem
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Objective: minimize sum of weighted overlaps
        objective = 0
        for i in top_k_indices:
            lag = i - (n_points - 1)
            # Compute contribution to the objective
            if lag >= 0:
                # h[i]*h[i+lag] term appears in correlation
                # With 1-h_k  this becomes h[i]*(1-h[i+lag]) = h[i] - h[i]*h[i+lag]
                # So we have -h[i]*h[i+lag] in the negative correlation
                # But since we want to minimize sum over all, we consider:
                # Negative of the correlation sum
                if i + lag < n_points:
                    objective += h_vars[i] * h_vars[i + lag]
            else:
                lag = -lag
                if i >= lag:
                    objective += h_vars[i] * h_vars[i - lag]
        
        objective = cp.Minimize(objective)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, ep=1e-8)
            if prob.status == cp.OPTIMAL:
                return np.array(h_vars.value)
        except:
            pass
        return h.copy()

    # Main iteration loop
    for iteration in range(max_iterations):
        elapsed = time() - start_time
        if elapsed > budget_s * 0.95:
            break
            
        # Optimize using convex relaxation
        h_opt = convex_optimize_top_k(h, k_top)
        
        # Apply low-pass filter to smooth
        h_filtered = apply_low_pass_filter(h_opt, filter_width)
        
        # Clamp and renormalize
        h_filtered = np.clip(h_filtered, 0, 1)
        h_filtered = h_filtered * (n_points / 2) / np.sum(h_filtered)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_filtered, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_filtered.copy()
            h = h_filtered.copy()
        else:
            # Adjust parameters if no improvement
            if k_top < 30:
                k_top += 2
            if filter_width > 2:
                filter_width -= 1
            if iteration % 100 == 0 and k_top > 10:
                k_top = max(10, k_top - 5)
        
        # Early stopping criteria
        if iteration > 0 and iteration % 50 == 0 and c5_new >= best_c5:
            if filter_width > 1:
                filter_width = max(1, filter_width - 2)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
