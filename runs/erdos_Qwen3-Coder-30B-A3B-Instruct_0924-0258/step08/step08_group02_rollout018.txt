import numpy as np
from cvxpy import *
import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a piecewise constant function that satisfies sum constraint
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Split the budget: 70% for convex relaxation steps, 30% for local refinement
    relax_budget = 0.7 * budget_s
    refine_budget = 0.3 * budget_s
    
    # Parameters for convex relaxation
    lambda_reg = 0.1
    alpha_reg = 0.01
    
    # Main iterative process
    no_improvement_count = 0
    max_no_improvement = 50
    
    # Convex relaxation solver
    def solve_convex_relaxation(h_current, lambda_reg, alpha_reg):
        h = Variable(n_points)
        # Define the objective: convex relaxation with regularization
        objective = Minimize(
            lambda_reg * sum_squares([sum(h[i] * (1 - h[(i + k) % n_points]) for i in range(n_points)) for k in range(n_points)]) +
            alpha_reg * sum_squares(h)
        )
        
        # Constraints
        constraints = [
            sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        
        # Solve
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status == 'optimal':
                return np.array(h.value).flatten()
        except:
            pass
            
        # Fallback: return the current values normalized
        return h_current * (n_points/2) / np.sum(h_current)
    
    # Coordinate descent refinement
    def coordinate_descent_step(h_current, fixed_vars, num_to_optimize=10):
        h_new = h_current.copy()
        
        # Select indices to optimize
        indices_to_optimize = []
        all_indices = list(range(n_points))
        np.random.shuffle(all_indices)
        indices_to_optimize = all_indices[:num_to_optimize]
        indices_to_fix = [i for i in range(n_points) if i not in indices_to_optimize]
        
        # Optimize selected variables using gradient descent
        max_iter = 50
        step_size = 0.01
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
            
            grad = np.zeros(n_points)
            for i in indices_to_optimize:
                for k in range(n_points):
                    j = (i + k) % n_points
                    grad[i] += -2 * dx * (1 - h_new[j] - h_new[i])
            
            # Update with gradient descent
            h_new[indices_to_optimize] -= step_size * grad[indices_to_optimize]
            
            # Project to [0,1]
            h_new[indices_to_optimize] = np.clip(h_new[indices_to_optimize], 0, 1)
            
            # Keep fixed variables constant
            # Normalize to satisfy constraint
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
        return h_new
    
    # Main loop
    last_improvement_time = time.time()
    
    while time.time() - start_time < budget_s * 0.95:
        # Convex relaxation step (60% of relaxation budget)
        relax_start = time.time()
        h_relaxed = solve_convex_relaxation(best_h, lambda_reg, alpha_reg)
        h_relaxed = h_relaxed * (n_points/2) / np.sum(h_relaxed)
        
        while time.time() - relax_start < relax_budget * 0.6 and time.time() - start_time < budget_s * 0.95:
            if np.allclose(h_relaxed, best_h, atol=1e-6):
                break
            current_c5 = evaluate_erdos_solution(h_relaxed, dx)
            if current_c5 < best_c5:
                best_h = h_relaxed.copy()
                best_c5 = current_c5
                last_improvement_time = time.time()
            else:
                break
            # Practical reduce
            lambda_reg = max(0.01, lambda_reg * 0.95)
            h_relaxed = solve_convex_relaxation(h_relaxed, lambda_reg, alpha_reg)
        
        # Coordinate descent forward
        h_local = coordinate_descent_step(best_h, [], num_to_optimize=int(n_points * 0.2))
        h_local = h_local * (n_points/2) / np.sum(h_local)
        
        # Evaluate
        current_c5 = evaluate_erdos_solution(h_local, dx)
        if current_c5 < best_c5:
            best_h = h_local.copy()
            best_c5 = current_c5
            last_improvement_time = time.time()
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            break
            
        # Early exit
        if time.time() - last_improvement_time > 10.0:
            break
    
    return (list(best_h), best_c5, n_points)
