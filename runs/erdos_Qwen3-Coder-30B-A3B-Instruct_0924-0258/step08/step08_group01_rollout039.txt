from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    # Genetic algorithm parameters
    population_size = 50
    num_generations = 10000
    crossover_rate = 0.8
    mutation_rate = 0.1
    mutation_step = 0.05
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5 + 0.25
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5 + 0.25
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    no_improvement_count = 0
    stagnation_limit = 200
    
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            f = evaluate_erdos_solution(h, dx)
            fitness.append(f)
            
        # Select parents using tournament selection
        selected = []
        for _ in range(population_size // 2):
            tournament_size = 3
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Generate offspring
        offspring = []
        for i in range(0, len(selected), 2):
            if i + 1 >= len(selected):
                break

            parent2 = selected[i+1]

                alpha = random.uniform(0.3, 0.7)
                child = alpha * parent1 + (1 - alpha) * parent2
                # Ensure sum is preserved
                child = child * (n_points / 2) / np.sum(child)
                offspring.append(child)
            else:
                offspring.append(parent1.copy())
                offspring.append(parent2.copy())
            
            # Mutation
            if random.random() < mutation_rate:
                child = offspring[-1].copy()
                # Add random noise
                noise = np.random.normal(0, mutation_step, n_points)
                child += noise

                # Adjust to maintain sum
                child = child * (n_points / 2) / np.sum(child)
                offspring[-1] = child

        all_individuals = population + offspring

        return evaluate_erdos_solution(h, dx)

        return np.sum(h) - n_points / 2.0

            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
