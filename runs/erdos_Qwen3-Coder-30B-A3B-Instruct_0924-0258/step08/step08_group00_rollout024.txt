import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def repair(h, n_points):
    h = np.clip(h, 0, 1)
    if np.sum(h) == 0:
        return h
    h = h * (n_points/2) / np.sum(h)
    return h

    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        n_points = len(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-start optimization with Latin Hypercube sampling
    n_starts = 50
    best_solution = (best_h.copy(), best_c5)
    
    # Generate Latin Hypercube samples
    samples = np.random.uniform(0, 1, size=(n_starts, n_points))
    for i in range(n_starts):
        samples[i] = repair(samples[i], n_points)
    
    # Add the initial solution if it exists
    if 'initial_h_values' in globals():
        samples[0] = h_start.copy()
    
    # Run SLSQP optimization for each start
    for i, h0 in enumerate(samples):
        if time() - start_time > budget_s * 0.95:
            break
            
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h0, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                h_final = repair(h_final, n_points)
                c5_final = objective(h_final)
                
                if c5_final < best_solution[1]:
                    best_solution = (h_final.copy(), c5_final)
        except Exception:
            continue
    
    # Final check
    return (list(best_solution[0]), best_solution[1], n_points)
