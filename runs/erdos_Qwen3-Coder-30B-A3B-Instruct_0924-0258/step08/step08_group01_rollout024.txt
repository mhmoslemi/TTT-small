import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale approach
    max_scales = int(np.log2(n_points)) + 1
    current_h = h_start.copy()
    current_n = n_points
    scale = 0
    perturbation_magnitude = 0.05
    decay_factor = 0.95
    
    while scale < max_scales and (time() - start_time) < budget_s * 0.95:
        # Double the resolution
        new_n = current_n * 2
        new_h = np.zeros(new_n)
        
        # Interpolate old values
        for i in range(current_n):
            new_h[2*i] = current_h[i]
            new_h[2*i+1] = current_h[i]
        
        # Add some noise with current perturbation magnitude
        for i in range(new_n):
            if np.random.random() < 0.7:  # 70% chance to perturb
                pert = perturbation_magnitude * (np.random.random() - 0.5) * 2
                new_h[i] += pert
                new_h[i] = max(0, min(1, new_h[i]))  # Clip to [0,1]
        
        # Adjust to maintain sum constraint
        current_sum = np.sum(new_h)
        new_h = new_h * (new_n / 2) / current_sum
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, 2.0/new_n)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            current_h = new_h.copy()
            current_n = new_n
            perturbation_magnitude = 0.05  # Reset
        else:
            perturbation_magnitude *= decay_factor
            current_h = new_h.copy()
            current_n = new_n
            
        # Local refinement with SLSQP
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / len(h))
        
        def constraint_sum(h):
            return np.sum(h) - len(h) / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(current_n)]
        
        try:
            res = minimize(objective, current_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-12})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined
                    current_h = h_refined
        except Exception:
            pass
            
        scale += 1
        
    # Final refinement on best solution
    try:
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / len(h))
        
        def constraint_sum(h):
            return np.sum(h) - len(h) / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(len(best_h))]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            final_h = res.x
            final_c5 = objective(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, len(best_h))
