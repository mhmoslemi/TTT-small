import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search on C5
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    # Sample shifts to use in feasibility check
    shifts = np.random.choice(range(1, n_points), size=min(100, n_points-1), replace=False)
    
    while time() - start_time < budget_s * 0.95 and high - low > tolerance:
        mid = (low + high) / 2.0
        
        # Formulate convex feasibility problem
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2.0
        ]
        
        # Overlap constraints for sampled shifts
        for k in shifts:
            # Create the constraint for shift k
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h_var[i] * (1 - h_var[j])
            constraints.append(overlap_expr <= mid * dx)
        
        # Solve feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_feasible = h_var.value
                if h_feasible is not None:
                    # Check if this solution is actually better
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_c5 = c5_feasible
                        best_h = h_feasible.copy()
                    high = mid
                    continue
        except:
            pass
        low = mid

    # Local refinement using gradient descent
    h_refine = best_h.copy()
    step_size = 0.1
    for it in range(500):
        if time() - start_time > budget_s * 0.99:
            break
            
        # Compute correlations for all shifts
        corr = np.correlate(h_refine, 1 - h_refine, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n_points - 1)
        
        # Compute gradient with respect to max overlap
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_shift
            j = j % n_points
            grad[i] = 1 - 2 * h_refine[i] - h_refine[j]
        
        h_new = h_refine - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_refine = h_new.copy()
        else:
            step_size *= 0.99
            if step_size < 1e-6:
                break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
