import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a better structured initial guess
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1.0
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use convex relaxation approach for better bounds
    # Solve convex relaxation using CVXPY
    def solve_convex_relaxation(h_init, n_points, dx):
        # Variables
        h = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points/2, C >= 0]
        
        # Overlap constraints using McCormick envelopes
        # We'll add constraints for a subset of lags to make it tractable
        max_lag = n_points // 2
        for lag in range(-max_lag, max_lag + 1):
            if lag == 0:
                continue
            # For lag k, correlate h with 1-h
            # Sum_{i} h[i] * (1 - h[i + k]) <= C
            # This is equivalent to Sum_{i} (h[i] - h[i]*h[i+k]) <= C
            # We'll use i from 0 to n_points-1-|k|
            
            if lag > 0:
                # h[i] * (1 - h[i + lag])
                # Equivalent to h[i] - h[i] * h[i + lag]
                lag_indices = np.arange(n_points - lag)
                # Use z[i][i+lag] for h[i] * h[i+lag] where i in 0..n_points-1-lag
                # and use McCormick envelopes for the bilinear term
                # This creates a convex relaxation
                upper_bound = sum(h_init[i] - h_init[i]*h_init[i + lag] for i in range(n_points - lag))
                constraints.append(sum(h[i] - h[i]*h[i + lag] for i in range(n_points - lag)) <= C)
            elif lag < 0:
                # h[i+lag] * (1 - h[i])
                # Lag is negative, so h[i] * (1 - h[i - lag])
                lag_abs = -lag
                lag_indices = np.arange(lag_abs, n_points)
                constraints.append(sum(h[i] - h[i]*h[i - lag_abs] for i in range(lag_abs, n_points)) <= C)
        
        # Objective
        objective = cp.Minimize(C)
        problem = cp.Problem(objective, constraints)
        
        try:
            problem.solve(solver=cp.CVXOPT, verbose=False, max_iters=1000)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                solution = h.value
                c5_value = C.value
                return solution, c5_value
        except Exception:
            pass
        
        return h_init, evaluate_erdos_solution(h_init, dx)
    
    # Convex relaxation step
    if time() - start_time < budget_s * 0.7:
        h_relaxed, c5_relaxed = solve_convex_relaxation(best_h, n_points, dx)
        if c5_relaxed < best_c5 and h_relaxed is not None:
            best_h = h_relaxed
            best_c5 = c5_relaxed
    
    # Local refinement with SLSQP
    if time() - start_time < budget_s * 0.95:
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    # Final check with a tabu search like local improvement
    if time() - start_time < budget_s * 0.98:
        # Try a gradient-based improvement for residual
        # This is a quick local search to improve known solution
        h_current = best_h.copy()
        # Optimize using multiple restarts
        best_local_h = h_current.copy()
        best_local_c5 = evaluate_erdos_solution(h_current, dx)
        
        for _ in range(10):  # Increase number of local search attempts
            if time() - start_time > budget_s * 0.98:
                break
            # Small perturbation and gradient update
            perturb = np.random.normal(0, 0.01, n_points)
            h_perturbed = h_current + perturb
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_local_c5:
                best_local_c5 = c5_perturbed
                best_local_h = h_perturbed.copy()
                
        if best_local_c5 < best_c5:
            best_c5 = best_local_c5
            best_h = best_local_h
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
