import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with binary pattern
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Binary threshold optimization
    # Try different thresholds for sparsification
    thresholds = np.linspace(0.1, 1.0, 10)
    
    for t in thresholds:
        if time() - start_time > budget_s * 0.6:
            break
            
        h = h_start.copy()
        h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
        
        # Simple iterative update with threshold
        for iter in range(50):
            if time() - start_time > budget_s * 0.6:
                break
                
            # Compute overlaps for all lags
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            max_overlap = np.max(overlaps)
            l_max = np.argmax(overlaps)
            
            # Estimate gradient for max overlap
            gradient = np.zeros(n_points)
            for i in range(n_points):
                j = (i + l_max) % n_points
                gradient[i] = -2 * dx * (1 - h[j] - h[i])
            
            # Apply threshold
            update_mask = np.abs(gradient) > t
            if np.sum(update_mask) > 0:
                # Update based on gradient sign
                update_direction = np.sign(gradient)
                h_new = h - 0.01 * update_direction
                h_new = np.clip(h_new, 0, 1)
                
                # Maintain sum constraint
                current_sum = np.sum(h_new)
                target_sum = n_points / 2
                
                if abs(current_sum - target_sum) > 1e-6:
                    # Simple adjustment
                    adjustment = (target_sum - current_sum) / n_points
                    h_new = np.clip(h_new + adjustment, 0, 1)
                
                h = h_new
                
            # Track best solution
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Phase 2: Local refinement with constrained optimization
    if time() - start_time < budget_s * 0.9:
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Use np.random initialization for diversity
            x0 = best_h.copy()
            x0 = x0 * (n_points/2) / np.sum(x0)
            
            # L-BFGS-B with sum constraint
            def objective(x):
                return evaluate_erdos_solution(x, dx)
                
            def constraint_func(x):
                return np.sum(x) - n_points / 2  # Sum constraint
            
            cons = {'type': 'eq', 'fun': constraint_func}
            
            res = minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-9}
            )
            
            if res.success:
                refined_h = res.x
                refined_c5 = evaluate_erdos_solution(refined_h, dx)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
            
        except Exception:
            pass
            
    # Final validation
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
