import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h onto feasible set: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to bounds
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    h = h * (n_points/2) / np.sum(h)
    return h

def compute_gradients(h, max_lags, n_points):
    """Compute combined gradient from top max_lags"""
    gradient = np.zeros(n_points)
    for lag in max_lags:
        if lag >= 0:
            for i in range(n_points - lag):
                gradient[i] += 1 - h[i + lag]
                if i + lag < n_points:
                    gradient[i + lag] -= h[i]
        else:
            for i in range(n_points + lag):
                gradient[i] += 1 - h[i - lag]
                if i - lag >= 0:
                    gradient[i - lag] -= h[i]
    return gradient

def line_search(h, grad, step_size, n_points, dx):
    """Backtracking line search for optimal step size"""
    c1 = 1e-4
    alpha = step_size
    current_c5 = evaluate_erdos_solution(h, dx)
    
    while alpha > 1e-12:
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 <= current_c5 - c1 * alpha * np.dot(grad, grad):
            return alpha, h_new
        alpha *= 0.5
    return alpha, h

    max_iter = 10000
    iter_count = 0
    no_improve_count = 0
    K = 5  # Number of top lags to consider

        corr = np.correlate(h_current, 1 - h_current, mode='full')

        top_lags_idx = np.argsort(corr)[-K:][::-1]
        # Get the corresponding lags (positive and negative)
        max_lags = []
        for idx in top_lags_idx:
            lag = idx - (len(h_current) - 1)
            max_lags.append(lag)

        gradient = compute_gradients(h_current, max_lags, n_points)
        
        # Perform line search
        step_size, h_new = line_search(h_current, gradient, step_size, n_points, dx)

        h_new = project_h(h_new, n_points)
