import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm Parameters
    population_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 5
    stagnation_limit = 100
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)
    
    # Genetic Algorithm
    def tournament_selection(population, fitnesses, tournament_size=3):
        indices = np.random.choice(len(population), size=tournament_size)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        return population[best_idx]
    
    def blend_crossover(parent1, parent2, alpha=0.5):
        child1 = np.zeros_like(parent1)
        child2 = np.zeros_like(parent2)
        for i in range(len(parent1)):
            gamma = (1 + 2 * alpha) * np.random.rand() - alpha
            child1[i] = gamma * parent1[i] + (1 - gamma) * parent2[i]
            child2[i] = gamma * parent2[i] + (1 - gamma) * parent1[i]
        return child1, child2
    
    def mutate(h, mutation_rate, generation, max_generations):
        step_size = 0.05 * (1 - generation / max_generations)
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.normal(0, step_size)
        # Rebalance to meet sum constraint
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        h = np.clip(h, 0, 1)
        return h
    
    # Main GA loop
    no_improvement_count = 0
    for gen in range(generations):
        if time() - start_time > budget_s * 0.7:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            fitness = evaluate_erdos_solution(h, dx)
            fitnesses.append(fitness)
        
        # Track best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        if no_improvement_count >= stagnation_limit:
            break
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Create new population
        new_population = elite[:]
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            if np.random.rand() < crossover_rate:
                child1, child2 = blend_crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            child1 = mutate(child1, mutation_rate, gen, generations)
            child2 = mutate(child2, mutation_rate, gen, generations)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
    
    # Final local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
