import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 0
    max_generations = int(budget_s * 0.7 / (n_points * 0.01)) if n_points > 0 else 10000
    no_improvement_count = 0
    max_no_improvement = 200
    mutation_step = 0.05
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        population[0] = population[0] * (n_points / 2) / np.sum(population[0])
    else:
        population.append(np.ones(n_points) * 0.5)
    
    while len(population) < population_size:
        h = np.random.rand(n_points) * 0.8 + 0.1  # Keep away from boundaries
        h = h * (n_points / 2) / np.sum(h)  # Enforce sum constraint
        population.append(h)
    
    # Main GA loop
    while time() - start_time < budget_s * 0.95 and generations < max_generations:
        # Evaluate fitness for all population members
        fitness = []
        for h in population:
            f = evaluate_erdos_solution(h, dx)
            fitness.append(f)
            
        # Tournament selection
        selected = []
        for _ in range(population_size // 2):
            i1, i2 = np.random.randint(0, population_size, 2)
            if fitness[i1] < fitness[i2]:
                selected.append(population[i1].copy())
            else:
                selected.append(population[i2].copy())
        
        # Generate offspring
        offspring = []
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[i + 1] if i + 1 < len(selected) else selected[0]
            
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand() * 0.4 + 0.3  # Blend range [0.3, 0.7]
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(0, n_points)
                child1[idx] += np.random.randn() * mutation_step
                child1[idx] = np.clip(child1[idx], 0, 1)
                # Maintain sum constraint
                diff = np.sum(child1) - n_points / 2
                if abs(diff) > 1e-10:
                    # Distribute difference evenly among elements
                    child1 += -diff / n_points
                
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(0, n_points)
                child2[idx] += np.random.randn() * mutation_step
                child2[idx] = np.clip(child2[idx], 0, 1)
                # Maintain sum constraint
                diff = np.sum(child2) - n_points / 2
                if abs(diff) > 1e-10:
                    # Distribute difference evenly among elements
                    child2 += -diff / n_points
            
            offspring.append(child1)
            offspring.append(child2)
            
        # Combine population with offspring
        # Keep top individuals from both populations
        all_individuals = population + offspring
        all_fitness = fitness + [evaluate_erdos_solution(h, dx) for h in offspring]
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(all_fitness)
        population = [all_individuals[i] for i in sorted_indices[:population_size]]
        
        # Update best
        current_best_fitness = all_fitness[sorted_indices[0]]
        if current_best_fitness < best_c5:
            best_c5 = current_best_fitness
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            # Reset with new random population
            population = []
            for _ in range(population_size):
                h = np.random.rand(n_points) * 0.8 + 0.1
                h = h * (n_points / 2) / np.sum(h)
                population.append(h)
            no_improvement_count = 0
            
        # Periodically refine with SLSQP
        if generations % 100 == 0 and generations > 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
                
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 1000, 'ftol': 1e-12})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
                
        generations += 1
        mutation_step *= 0.98  # Decay mutation
        if mutation_step < 0.005:
            mutation_step = 0.005
    
    # Final SLSQP refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
        
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    # Final verification of constraints
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) - n_points / 2) < 1e-10
    
    return (list(best_h), best_c5, n_points)
