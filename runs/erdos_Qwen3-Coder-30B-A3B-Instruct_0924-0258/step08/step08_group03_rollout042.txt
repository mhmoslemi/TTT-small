import numpy as np
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search bounds
    low = 0.38
    high = 0.38085
    tol = 1e-5
    
    # Precompute shifts
    shifts = np.arange(-n_points + 1, n_points)
    
    # SDP solver parameters
    # Need to do binary search on C5 and try to solve SDP feasibility
    while time() - start_time < budget_s * 0.95 and high - low > tol:
        mid = (low + high) / 2.0
        C5 = mid
        
        # Use SDP relaxation to check feasibility
        # Variables: Y = h*h^T, h
        Y = Semidefinite(n_points)
        h_var = Variable(n_points)
        
        # Constraints
        constraints = [
            Y >> 0,  # Y positive semidefinite
            diag(Y) == h_var,  # Diagonal elements of Y are h
            sum(h_var) == n_points / 2,  # Constraint on sum
        ]
        
        # Overlap constraints for all shifts
        for k in shifts[::max(1, len(shifts)//200)]:  # Sample shifts
            if k >= 0:
                expr = sum_entries(h_var[:n_points-k]) - sum_entries(Y[np.arange(n_points-k), np.arange(k,n_points)])
            else:
                expr = sum_entries(h_var[-k:]) - sum_entries(Y[np.arange(-k,n_points), np.arange(n_points+k)])
                
            constraints.append(expr <= C5 * dx)
        
        # Create the problem and solve
        objective = Minimize(0)  # dummy objective (feasibility)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=MOSEK, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                # Get solution
                h_sol = h_var.value
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                c5_sol = evaluate_erdos_solution(h_sol, dx)
                
                if c5_sol < best_c5:
                    best_c5 = c5_sol
                    best_h = h_sol.copy()
                high = C5
            else:
                low = C5
        except:
            low = C5
            
    # Local refinement with gradient descent
    h = best_h.copy()
    step_size = 0.001
    for iter in range(10000):
        if time() - start_time > budget_s * 0.95:
            break
        # Compute overlap
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h[i] - h[j]
            else:
                grad[i] = 1 - 2 * h[i]
        
        # Gradient descent step with projection
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            step_size *= 0.995
            
    return (list(best_h), best_c5, n_points)
