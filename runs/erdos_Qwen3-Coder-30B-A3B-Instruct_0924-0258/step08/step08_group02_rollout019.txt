import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convex relaxation parameters
    lambda_reg = 0.1
    alpha_reg = 0.01
    
    def convex_relaxation(h_init):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        
        # Create overlap terms for all lags
        overlap_exprs = []
        for k in range(n_points):
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h[i] * (1 - h[j])
            overlap_exprs.append(expr)
        
        # Regularized objective
        obj = cp.Minimize(lambda_reg * cp.sum_squares(overlap_exprs) + 
                         alpha_reg * cp.sum_squares(h))
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                sol = h.value
                sol = np.clip(sol, 0, 1)
                sol = sol * (n_points/2) / np.sum(sol)
                return sol
        except:
            pass
        return h_init
    
    def local_coordinate_descent(h, num_vars=7):
        h_new = h.copy()
        indices = np.random.permutation(n_points)[:num_vars]
        
        def objective_grad(vars_vals):
            h_temp = h_new.copy()
            for i, idx in enumerate(indices):
                h_temp[idx] = vars_vals[i]
            
            overlaps = np.correlate(h_temp, 1 - h_temp, mode='full') * dx
            max_overlap = np.max(overlaps)
            l_max = np.argmax(overlaps)
            
            # Approximate gradient for coordinate update
            gradient = np.zeros(len(indices))
            for i, idx in enumerate(indices):
                j = (idx + l_max) % n_points
                gradient[i] = -2 * dx * (1 - h_temp[j] - h_temp[idx])
            
            return max_overlap, gradient
        
        # Use L-BFGS-B for coordinate optimization
        bounds = [(0, 1) for _ in range(len(indices))]
        initial_vals = [h_new[i] for i in indices]
        
        try:
            res = minimize(
                lambda x: objective_grad(x)[0],
                initial_vals,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 50, 'ftol': 1e-9}
            )
            if res.success:
                for i, idx in enumerate(indices):
                    h_new[idx] = res.x[i]
        except:
            pass
        
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        return h_new
    
    # Main optimization loop
    iter_count = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95:
        iter_count += 1
            
        # Convex relaxation step
        relaxed_h = convex_relaxation(best_h)
        
        # Coordinate descent step
        coord_h = local_coordinate_descent(relaxed_h)
        
        # Evaluate and keep best
        current_c5 = evaluate_erdos_solution(coord_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = coord_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= 50:
            break
            
        # Adaptive step size for coordinate descent
        num_vars = max(3, int(n_points * 0.2) - iter_count // 5)
    
    # Final refinement
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
