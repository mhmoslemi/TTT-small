import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Multiresolution approach: coarse grid first
    coarse_n = 15
    coarse_dx = 2.0 / coarse_n
    fine_n = n_points
    fine_dx = dx
    
    # Create population of coarse solutions
    pop_size = 50
    population = []
    for _ in range(pop_size):
        h_temp = np.random.rand(coarse_n)
        h_temp = h_temp * (coarse_n/2) / np.sum(h_temp)
        population.append(h_temp)
    
    # Keep track of time
    remaining_time = budget_s * 0.90
    
    # Genetic algorithm on coarse grid
    for gen in range(200):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, coarse_dx)
            fitness.append(-c5)
        
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            idx1, idx2 = np.random.randint(0, pop_size, 2)
            if fitness[idx1] > fitness[idx2]:
                selected.append(population[idx1])
            else:
                selected.append(population[idx2])
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            p1 = selected[i]
            p2 = selected[i+1] if i+1 < pop_size else selected[0]
            
            # Crossover
            if np.random.rand() < 0.5:
                crossover_point = np.random.randint(1, coarse_n)
                child1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
                child2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
            else:
                child1, child2 = p1.copy(), p2.copy()
            
            # Mutation
            for j in range(coarse_n):
                if np.random.rand() < 0.2:
                    child1[j] += np.random.normal(0, 0.05)
                if np.random.rand() < 0.2:
                    child2[j] += np.random.normal(0, 0.05)
            
            # Clamp and rescale
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (coarse_n/2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (coarse_n/2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
    
    # Find best coarse solution
    best_coarse = None
    best_coarse_c5 = float('inf')
    for h in population:
        c5 = evaluate_erdos_solution(h, coarse_dx)
        if c5 < best_coarse_c5:
            best_coarse = h
            best_coarse_c5 = c5
    
    # Upscale to fine grid
    h_fine = np.interp(np.linspace(0, 2, fine_n), 
                       np.linspace(0, 2, coarse_n), 
                       best_coarse)
    h_fine = h_fine * (fine_n/2) / np.sum(h_fine)
    
    # Refine with gradient-based method
    step_size = 0.01
    max_iter = 10000
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all correlations and find dominant lags
        corr = np.correlate(h_fine, 1 - h_fine, mode='full')
        # Get dominant overlaps for most influence
        sorted_lags = np.argsort(corr)[-5:]  # Top 5 lags
        dominant_lags = sorted_lags - (len(h_fine) - 1)
        
        # Compute approximate gradient for dominant lags
        gradient = np.zeros(fine_n)
        for lag in dominant_lags:
            if lag >= 0:
                for i in range(fine_n - lag):
                    gradient[i] += 1 - h_fine[i + lag]
                    if i + lag < fine_n:
                        gradient[i + lag] -= h_fine[i]
            else:
                for i in range(fine_n + lag):
                    gradient[i] += 1 - h_fine[i - lag]
                    if i - lag >= 0:
                        gradient[i - lag] -= h_fine[i]
        
        # Gradient descent step
        h_new = h_fine - step_size * gradient
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (fine_n/2) / np.sum(h_new)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, fine_dx)
        
        # Accept step if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # If not improving, reduce step size
        if c5_new >= evaluate_erdos_solution(h_fine, fine_dx):
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            h_fine = h_new
            step_size *= 1.05  # Increase step size slightly
        
        iter_count += 1
    
    return (list(best_h), best_c5, fine_n)
