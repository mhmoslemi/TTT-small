import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale optimization
    max_scales = int(np.floor(np.log2(max(n_points, 16))) + 1)
    scale = 0
    current_points = n_points
    current_h = h_start.copy()
    mu = 0.05  # perturbation magnitude
    decay_factor = 0.95
    max_iters_per_scale = 10
    
    while scale < max_scales and (time() - start_time) < budget_s * 0.95:
        # Upsample: double the resolution
        if scale > 0:
            current_points = 2 * current_points
            current_h = np.interp(np.linspace(0, 2, current_points), 
                                np.linspace(0, 2, len(current_h)), 
                                current_h)
        
        # Ensure sum constraint
        current_h = current_h * (current_points / 2) / np.sum(current_h)
        
        # Perturbation and refinement within scale
        for iter_scale in range(max_iters_per_scale):
            if (time() - start_time) >= budget_s * 0.95:
                break
                
            # Perturb solution
            h_perturb = current_h.copy()
            for i in range(len(h_perturb)):
                pert = mu * (2 * np.random.random() - 1)
                h_perturb[i] += pert
            
            # Clip and normalize
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (current_points / 2) / np.sum(h_perturb)
            
            # Evaluate
            c5_perturb = evaluate_erdos_solution(h_perturb, 2.0 / current_points)
            
            # Accept improvement
            if c5_perturb < best_c5:
                best_c5 = c5_perturb
                best_h = h_perturb.copy()
                current_h = h_perturb.copy()
                mu = 0.05  # reset
            else:
                mu *= decay_factor
            
            # Local refinement with SQP
            def objective(h):
                return evaluate_erdos_solution(h, 2.0 / current_points)
            
            def constraint_sum(h):
                return np.sum(h) - current_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(current_points)]
            
            try:
                res = minimize(objective, current_h, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 50, 'ftol': 1e-12})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_c5 = c5_refined
                        best_h = h_refined
                        current_h = h_refined
            except Exception:
                pass
                
        scale += 1
    
    # Final refinement with SLSQP
    if (time() - start_time) < budget_s * 0.95:
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / len(best_h))
        
        def constraint_sum(h):
            return np.sum(h) - len(best_h) / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(len(best_h))]
        
        try:
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    # Final check and return
    final_c5 = evaluate_erdos_solution(best_h, 2.0 / len(best_h))
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (len(best_h) / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, 2.0 / len(best_h))
    
    return (list(best_h), final_c5, len(best_h))
