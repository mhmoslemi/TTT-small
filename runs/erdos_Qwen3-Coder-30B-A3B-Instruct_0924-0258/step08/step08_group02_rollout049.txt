import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a binary-like construction
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1.0
        # Randomize some positions
        np.random.shuffle(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Threshold-based approach with iterative refinement
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Initial threshold and parameters
    threshold = 0.1
    cooling_rate = 0.95
    max_iterations = 500
    min_threshold = 0.01
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        max_lag = np.argmax(overlaps)
        
        # Compute gradient for the maximum overlap lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_lag) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Apply thresholding only to coordinates with highest gradient magnitude
        # Rank by absolute gradient
        grad_idx = np.argsort(np.abs(gradient))[::-1]
        top_k = min(n_points // 2, n_points - 1)  # Top 50%
        top_indices = grad_idx[:top_k]
        
        # Apply thresholding  
        h_new = h.copy()
        for i in top_indices:
            if h[i] >= 1 - threshold:
                h_new[i] = 1.0
            elif h[i] <= threshold:
                h_new[i] = 0.0
            # Keep intermediate values
        
        # Ensure sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate new solution
        current_c5 = evaluate_erdos_solution(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        
        # Update h and threshold
        h = h_new
        threshold = max(min_threshold, threshold * cooling_rate)
    
    # Coordinate-wise refinement step
    for _ in range(100):
        if time() - start_time > budget_s * 0.98:
            break

        h_refined = h.copy()
        # For each coordinate, optimize it while fixing others
        for i in range(n_points):
            # Create candidate values for h[i] using golden section search
            def obj_val(x):
                h_temp = h_refined.copy()
                h_temp[i] = x
                # Normalize to maintain sum
                h_temp = h_temp * (n_points/2) / np.sum(h_temp)
                h_temp = np.clip(h_temp, 0, 1)
                return evaluate_erdos_solution(h_temp, dx)
            
            # Golden section search within [0,1]
            a, b = 0.0, 1.0
            res = 0.618
            x1 = a + (1 - res) * (b - a)
            x2 = a + res * (b - a)
            
            for _ in range(20):  # max iterations
                if abs(b - a) < 1e-6:
                    break
                if obj_val(x1) < obj_val(x2):
                    b = x2
                    x2 = x1
                    x1 = a + (1 - res) * (b - a)
                else:
                    a = x1
                    x1 = x2
                    x2 = a + res * (b - a)
            h_refined[i] = (a + b) / 2
        
        # Normalize
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        h_refined = np.clip(h_refined, 0, 1)
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
        h = h_refined
    
    # Final L-BFGS refinement
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(corr)
        
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_h = np.clip(refined_h, 0, 1)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
