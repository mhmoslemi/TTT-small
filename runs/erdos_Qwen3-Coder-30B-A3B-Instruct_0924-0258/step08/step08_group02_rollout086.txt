import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search optimization
    # Start with coarse search
    for t in np.linspace(1.0, 0.01, 50):
        if time() - start_time > budget_s * 0.95:
            break
        
        h = best_h.copy()
        h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
        
        # Compute max overlap and lag
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Apply thresholding
        mask = np.abs(gradient) > t
        if np.sum(mask) > 0:
            # Update h based on gradient signs
            for i in range(n_points):
                if mask[i]:
                    # Update based on sign and constraint
                    if gradient[i] > 0:
                        h[i] = min(1.0, h[i] + 0.05)
                    else:
                        h[i] = max(0.0, h[i] - 0.05)
            
            # Enforce sum constraint
            h = h * (n_points/2) / np.sum(h)
            
            # Check if better solution
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Local refinement with scipy
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def gradient(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            max_overlap = np.max(overlaps)
            l_max = np.argmax(overlaps)
            
            grad = np.zeros_like(h)
            for i in range(len(h)):
                j = (i + l_max) % len(h)
                grad[i] = -2 * dx * (1 - h[j] - h[i])
            return grad
        
        # Optimization with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use L-BFGS-B 
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception as e:
        pass
    
    return (list(best_h), best_c5, n_points)
