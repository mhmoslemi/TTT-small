import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Use FFT for faster correlation computation
    def compute_overlap_fft(h):
        # Compute FFT of h and 1-h
        H = np.fft.fft(h)
        G = np.fft.fft(1 - h)
        # Compute cross-correlation in frequency domain
        corr = np.fft.ifft(H * np.conj(G))
        # Extract the full correlation (with proper offset)
        corr_vals = np.abs(corr) * dx
        return corr_vals
    
    # Main optimization loop
    max_iter = 20000
    step_size = 0.02
    tolerance = 1e-10
    no_improvement_count = 0
    no_improvement_limit = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps using FFT
        corr_vals = compute_overlap_fft(h_current)
        max_idx = np.argmax(corr_vals)
        max_lag = max_idx - (len(h_current) - 1)
        
        # Compute gradient (this is an estimate)
        gradient = np.zeros_like(h_current)
        for i in range(len(h_current)):
            if max_lag >= 0 and i + max_lag < len(h_current):
                gradient[i] = -1.0
            elif max_lag < 0 and i - max_lag >= 0:
                gradient[i] = -1.0
                
        # Perform gradient step
        h_new = h_current - step_size * gradient
        
        # Project onto feasible region
        sum_h = np.sum(h_new)
        proj = (n_points/2 - sum_h) / len(h_new)
        h_new = h_new + proj
        
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check improvement
        c5_current = evaluate_erdos_solution(h_new, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        h_current = h_new
        
        # Adaptive step size
        if no_improvement_count > 50:
            step_size *= 0.9
            no_improvement_count = 0
            
        if step_size < 1e-6:
            step_size = 1e-6
            
    # Final refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 2000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
