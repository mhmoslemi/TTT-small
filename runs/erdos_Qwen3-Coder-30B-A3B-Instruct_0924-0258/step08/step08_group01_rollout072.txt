import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale optimization
    scale = 0
    max_scales = int(np.log2(n_points)) + 2
    scale_points = [max(16, 2**scale)]
    while scale < max_scales:
        current_points = scale_points[-1]
        if current_points > 1000:
            break
        scale_points.append(current_points * 2)
        scale += 1
    
    for scale_idx, current_n in enumerate(scale_points[:-1]):  # Exclude the last one which is too big
        if time() - start_time > budget_s * 0.95:
            break
            
        # Upsample the best solution
        if current_n <= len(best_h):
            h_upsampled = best_h[:current_n].copy()
        else:
            # Linear interpolation for upscaling
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, current_n)
            h_upsampled = np.interp(x_new, x_old, best_h)
            
        # Normalize sum
        h_upsampled = h_upsampled * (current_n / 2) / np.sum(h_upsampled)
        
        # Perturbations with annealing
        perturbation_magnitude = 0.05
        max_iterations = 20
        no_improvement_count = 0
        max_no_improvement = 10
        
        for iter_idx in range(max_iterations):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Create perturbed candidate
            h_perturbed = h_upsampled + perturbation_magnitude * (np.random.random(current_n) - 0.5)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (current_n / 2) / np.sum(h_perturbed)
            
            # Evaluate
            c5_perturbed = evaluate_erdos_solution(h_perturbed, 2.0 / current_n)
            
            if c5_perturbed < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_perturbed
                no_improvement_count = 0
                # Reset perturbation magnitude
                perturbation_magnitude = 0.05
            else:
                no_improvement_count += 1
                perturbation_magnitude *= 0.95  # Decrease magnitude
                
            if no_improvement_count >= max_no_improvement:
                break
                
        # Local refinement using SLSQP
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / current_n)
            
        def constraint_sum(h):
            return np.sum(h) - current_n / 2.0
            
        bounds = [(0, 1) for _ in range(current_n)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-12})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
            
    # Final evaluation 
    evaluated_c5 = evaluate_erdos_solution(best_h, 2.0 / len(best_h))
    
    return (list(best_h), evaluated_c5, len(best_h))
