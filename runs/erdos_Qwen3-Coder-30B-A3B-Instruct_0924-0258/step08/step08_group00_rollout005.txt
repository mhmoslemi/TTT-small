import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different strategies to escape local optima
    
    # Strategy 1: Random restarts with local search
    max_restarts = 50
    for restart in range(max_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize with random perturbation
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Local optimization with simulated annealing inspired approach
        temp = 1.0
        cooling_rate = 0.995
        min_temp = 1e-6
        max_iter_per_temp = 100
        
        for iter_temp in range(5000):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb one element
            idx = np.random.randint(0, n_points)
            old_val = h[idx]
            new_val = old_val + np.random.normal(0, 0.02)
            new_val = np.clip(new_val, 0, 1)
            
            # Adjust other elements to maintain sum constraint
            diff = new_val - old_val
            h[idx] = new_val
            
            # Distribute the difference
            for i in range(n_points):
                if i != idx:
                    h[i] -= diff / (n_points - 1)
                    if h[i] < 0:
                        h[i] = 0
                    elif h[i] > 1:
                        h[i] = 1
                        
            # Renormalize
            h = h * (n_points/2) / np.sum(h)
            
            # Accept or reject based on temperature
            current_c5 = evaluate_erdos_solution(h, dx)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            else:
                # Occasionally accept worse solutions
                if np.random.rand() < np.exp(-(current_c5 - best_c5) / temp):
                    # Accept worse solution
                    pass
                else:
                    # Revert
                    h[idx] = old_val
                    h = h * (n_points/2) / np.sum(h)
            
            if temp > min_temp:
                temp *= cooling_rate
                
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try multiple random initializations
        best_final = best_c5
        best_final_h = best_h.copy()

            if time() - start_time > budget_s * 0.95:
                break

            res = minimize(objective, h_init, method='SLSQP',

                c5_final = objective(h_final)
                if c5_final < best_final:
                    best_final = c5_final
                    best_final_h = h_final
        
        if best_final < best_c5:
            best_c5 = best_final
            best_h = best_final_h
            
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
