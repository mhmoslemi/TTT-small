import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with provided construction or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search bounds
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    # Number of shifts to sample for feasibility check (reduced for speed)
    num_shifts = 100
    
    while time() - start_time < budget_s * 0.9 and high - low > tolerance:
        mid = (low + high) / 2
        
        # Sample shifts
        shifts = np.random.choice(n_points, num_shifts, replace=False)
        
        # Create feasibility problem
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Add overlap constraints
        for k in shifts:
            # Compute overlap for shift k
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h_var[i] * (1 - h_var[j])
            constraints.append(overlap <= mid * dx)
        
        # Solve feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.SOLVED:
                h_solution = h_var.value
                if h_solution is not None:
                    h = h_solution
                    c5 = evaluate_erdos_solution(h, dx)
                    if c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = c5
                    high = mid
                else:
                    low = mid
            else:
                low = mid
        except:
            low = mid
    
    # Local refinement with gradient descent
    h = best_h.copy()
    step_size = 0.1
    max_iter = 500
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.98:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h[i] - h[j]
            else:
                grad[i] = 1 - 2 * h[i]
        
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            step_size *= 0.95
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
