import numpy as np
from scipy.optimize import minimize
from time import time
import math

def project(h, n_points):
    """Project h onto the feasible set: [0,1]^n and sum(h) = n/2"""
    h_clipped = np.clip(h, 0, 1)
    if np.sum(h_clipped) == 0:
        return h_clipped
    h_scaled = h_clipped * (n_points / 2.0) / np.sum(h_clipped)
    return h_scaled

def compute_gradient(h, n_points, dx):
    """Compute gradient of the overlap integral with respect to h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_lag = max_idx - (n_points - 1)
    
    # Create shifted version of (1-h) with the lag that maximizes overlap
    shifted = np.roll(1 - h, max_lag)
    # Gradient: derivative of the overlap with respect to each component of h
    gradient = 2 * (1 - 2*h) * shifted
    return gradient

def evaluate_erdos_solution(h, dx):
    """Evaluate the maximum overlap of h with 1-h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = project(h_start.copy(), n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Parameters
    max_iter = 1000
    patience = 50
    step_size = 0.1
    no_improvement = 0
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient and update
        grad = compute_gradient(h, n_points, dx)
        h_new = h - step_size * grad
        h_new = project(h_new, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Check if improved
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Update h
        h = h_new.copy()
        
        # Adapt step size
        if no_improvement >= patience:
            step_size *= 0.5
            no_improvement = 0
            if step_size < 1e-8:
                break

    # Final optimization using scipy
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 200, 'ftol': 1e-9})
        
        if res.success:
            h_final = project(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
