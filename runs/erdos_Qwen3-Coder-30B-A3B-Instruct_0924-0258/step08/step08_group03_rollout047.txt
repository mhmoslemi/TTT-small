import numpy as np
from time import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Add convex perturbation to escape local minima
    max_iter = 5000
    epsilon_0 = 0.001
    beta = 0.99
    top_k = 100
    tolerance = 1e-6
    last_improvement = 0
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    # Outer optimization loop with convex perturbation
    for t in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlations
        corr = np.correlate(h, 1 - h, mode='full')
        top_shifts = np.argsort(corr)[-top_k:][::-1]  # Top k shifts by descending correlation
        
        # Create convex program
        h_var = cp.Variable(n_points)
        objective = 0
        for k in top_shifts:
            # Compute kernel for shifted correlation
            if k >= n_points:
                shifted_kernel = np.concatenate([np.zeros(k - n_points + 1), np.ones(n_points - (k - n_points + 1))])
            elif k == 0:
                shifted_kernel = np.ones(n_points)
            else:
                shifted_kernel = np.concatenate([np.zeros(k), np.ones(n_points - k)])
            
            # Add elastic term from correlation
            objective += cp.sum(cp.multiply(h_var, 1 - h_var))
        
        # Add convex regularizer
        objective += epsilon_0 * (beta ** t) * cp.pnorm(h_var, 2)**2
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h_new = np.array(h_var.value).flatten()
        except:
            # Fallback: gradient update
            h_new = h.copy()
        
        # Ensure feasibility - projection
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Track best solution
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improvement = t
        elif t - last_improvement > 100:
            # Early termination if no progress
            break
            
        # Update h
        h = h_new
        if t % 500 == 0:
            epsilon_0 *= 0.95  # Reduce early perturbation over time
    
    # Final gradient-free refinement
    if time() - start_time < budget_s * 0.99:
        for _ in range(100):
            if time() - start_time > budget_s * 0.99:
                break
            h_perturbed = h + np.random.normal(0, 0.001, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            c5_pert = evaluate_erdos_solution(h_perturbed, dx)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_perturbed.copy()
                
    return (list(best_h), best_c5, n_points)
