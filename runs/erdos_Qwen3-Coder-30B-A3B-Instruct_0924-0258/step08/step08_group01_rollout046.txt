import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale approach
    max_scales = int(np.log2(n_points)) + 3
    scale = 0
    current_h = h_start.copy()
    current_n = n_points
    perturbation_magnitude = 0.05
    
    while scale < max_scales and time() - start_time < budget_s * 0.9:
        # Double the resolution each scale
        next_n = current_n * 2
        if next_n > 1000:  # Limit max points
            break
            
        # Upsample current solution
        upsampled = np.interp(np.linspace(0, 2, next_n, endpoint=False), 
                             np.linspace(0, 2, current_n, endpoint=False), 
                             current_h)
        
        # Apply perturbations
        mutated = upsampled + perturbation_magnitude * (2 * np.random.random(next_n) - 1)
        mutated = np.clip(mutated, 0, 1)
        
        # Adjust to maintain sum constraint
        mutated = mutated * (next_n / 2) / np.sum(mutated)
        
        # Evaluate
        c5_mutated = evaluate_erdos_solution(mutated, 2.0/next_n)
        
        if c5_mutated < best_c5:
            best_c5 = c5_mutated
            best_h = mutated.copy()
            current_h = mutated.copy()
            current_n = next_n
            perturbation_magnitude = 0.05  # Reset magnitude
        else:
            perturbation_magnitude *= 0.95  # Decay magnitude
            
        # Local refinement with SLSQP
        def objective(h):
            return evaluate_erdos_solution(h, 2.0/next_n)
        
        def constraint_sum(h):
            return np.sum(h) - next_n / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(next_n)]
        
        try:
            res = minimize(objective, current_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-12})
            if res.success:
                refined_h = res.x
                c5_refined = objective(refined_h)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = refined_h
                    current_h = refined_h
                    current_n = next_n
        except Exception:
            pass
            
        scale += 1
        
        # Periodic reset
        if scale % 5 == 0:
            # Restart with best solution from previous scale
            current_h = best_h.copy()
            current_n = len(current_h)
            perturbation_magnitude = 0.05
            
    # Final refinement using only best solution
    def objective(h):
        return evaluate_erdos_solution(h, 2.0/len(best_h))
    
    def constraint_sum(h):
        return np.sum(h) - len(best_h) / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(len(best_h))]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            final_h = res.x
            c5_final = objective(final_h)
            if c5_final < best_c5:
                best_h = final_h
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, len(best_h))
