import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Trust Region method with cutting planes
    max_iter = 1000
    trust_radius = 0.2
    improvement_threshold = 1e-8
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Precompute all lags
    all_lags = np.arange(1 - n_points, n_points)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h_current, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            
        # Check for convergence
        if abs(current_c5 - best_c5) < improvement_threshold:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            
        if no_improvement_count > max_no_improvement:
            break
            
        # Solve convex optimization with trust region and cutting planes
        try:
            # Build model with cutting planes
            h_vars = cp.Variable(n_points)
            
            # Objective
            objective = cp.Minimize(cp.Variable())
            
            # Constraints
            constraints = [
                cp.sum(h_vars) == n_points / 2,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Add overlap constraints for cutting planes
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            overlap_values = corr * dx
            max_idx = np.argmax(overlap_values)
            max_lag = all_lags[max_idx]
            
            # Add some constraints for most violated lags
            top_violations = np.argsort(overlap_values)[-10:][::-1] # Top 10
            for idx in top_violations:
                lag = all_lags[idx]
                if lag >= 0:
                    # Only consider valid indices
                    valid_range = slice(0, n_points - lag)
                    expr = cp.sum(cp.multiply(h_vars[valid_range], 
                                            1 - h_vars[valid_range + lag]))
                else:
                    valid_range = slice(-lag, n_points)
                    expr = cp.sum(cp.multiply(h_vars[valid_range], 
                                            1 - h_vars[valid_range - lag]))
                
                constraints.append(expr <= cp.Variable())

            # Create and solve the problem
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                # Project onto trust region
                delta = h_new - h_current
                if np.linalg.norm(delta) > trust_radius:
                    h_new = h_current + trust_radius * delta / np.linalg.norm(delta)
                
                # Clip to feasible bounds
                h_new = np.clip(h_new, 0, 1)
                
                # Renormalize
                h_new = h_new * (n_points/2) / np.sum(h_new)
                h_current = h_new
                
                # Adjust trust radius based on improvement
                new_c5 = evaluate_erdos_solution(h_current, dx)
                if new_c5 < current_c5:
                    trust_radius *= 1.2  # Increase
                else:
                    trust_radius *= 0.8  # Decrease
                    trust_radius = max(trust_radius, 0.01)
                    
        except Exception:
            # Fall back to local optimization
            h_new = h_current.copy()
            step_size = 0.01
            
            # Compute all overlaps for all lags
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            lag_indices = np.arange(len(corr))
            lags = lag_indices - (len(h_current) - 1)
            overlap_values = corr * dx
            
            # Find the lag with maximum overlap
            max_idx = np.argmax(overlap_values)
            max_lag = lags[max_idx]
            
            # Compute gradient of overlap w.r.t. h for specific lag
            gradient = np.zeros_like(h_current)
            for i in range(len(h_current)):
                if max_lag >= 0:
                    if i + max_lag < len(h_current):
                        gradient[i] = -1.0
                else:
                    if i - max_lag >= 0:
                        gradient[i] = -1.0
                        
            # Apply gradient descent with constraint projection
            h_new = h_current - step_size * gradient
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            h_current = h_new
    
    # Final refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
