import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter_sa = 100000
    epsilon = 0.1
    
    # Initialize with normalized h
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    
    T = T_init
    current_c5 = evaluate_erdos_solution(h, dx)
    
    iter_count = 0
    stagnation_count = 0
    stagnation_limit = 1000
    
    while T > T_min and (time() - start_time) < budget_s * 0.9:
        if iter_count >= max_iter_sa:
            break
            
        # Perturb two random indices
        i, j = np.random.randint(0, n_points, 2)
        delta = np.random.uniform(-epsilon, epsilon)
        
        # Make sure new values are within bounds
        new_h = h.copy()
        new_h[i] = h[i] + delta
        new_h[j] = h[j] - delta
        
        # Enforce bounds
        new_h = np.clip(new_h, 0, 1)
        
        # Maintain sum constraint
        sum_diff = np.sum(new_h) - (n_points / 2)
        if abs(sum_diff) > 1e-10:
            new_h[0] = new_h[0] - sum_diff
            new_h = np.clip(new_h, 0, 1)
        
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject based on SA criterion
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        if stagnation_count > stagnation_limit:
            T = max(T_min, T * alpha)
            stagnation_count = 0
            
        iter_count += 1
        T = max(T_min, T * alpha)
    
    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return np.max(np.correlate(h, 1 - h, mode='full') * dx)
        
        # Normalize for initial condition
        h_refine = best_h.copy()
        h_refine = h_refine * (n_points/2) / np.sum(h_refine)
        
        res = minimize(
            objective,
            h_refine,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
