import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with alternating pattern to avoid local minima
        h_start = np.zeros(n_points)
        h_start[::2] = 0.5
        h_start[1::2] = 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for the hybrid approach
    max_iterations = 500
    no_improvement_limit = 50
    no_improvement_count = 0
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Convex relaxation parameters
    lambda_reg = 0.1
    alpha_reg = 0.01
    
    # Coordinate descent parameters
    coord_subset_size = max(1, n_points // 5)  # 20% of variables
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convex relaxation step
        try:
            # Formulate the relaxed optimization problem
            h_relaxed = cp.Variable(n_points)
            objective = cp.Minimize(
                lambda_reg * cp.sum_squares(cp.matmul(cp.eye(n_points), h_relaxed) - cp.sum(cp.multiply(h_relaxed, 1 - h_relaxed))) +
                alpha_reg * cp.sum_squares(h_relaxed)
            )
            constraints = [
                cp.sum(h_relaxed) == n_points/2,
                h_relaxed >= 0,
                h_relaxed <= 1
            ]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_relaxed.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                h = h_new
        except:
            pass
            
        # Coordinate descent step
        indices_to_update = np.random.choice(n_points, size=coord_subset_size, replace=False)
        h_old = h.copy()
        
        # Simple gradient descent on selected variables
        learning_rate = 0.1
        inner_iter = 50
        for _ in range(inner_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute current overlaps and gradient
            overlaps = np.correlate(h, 1-h, mode='full') * dx
            max_idx = np.argmax(overlaps)
            
            # Gradient estimation for selected indices
            grad = np.zeros(n_points)
            for i in indices_to_update:
                j = (i + max_idx) % n_points
                grad[i] = -2 * dx * (1 - h[j] - h[i])
            
            # Update
            h_update = h - learning_rate * grad
            h_update = np.clip(h_update, 0, 1)
            h_update = h_update * (n_points/2) / np.sum(h_update)
            
            h = h_update
            
        # Check for improvement
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > no_improvement_limit:
            break
    
    # Final refinement using scipy
    try:
        from scipy.optimize import minimize
        
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)
        
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-12}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
