import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    population_size = 50
    num_generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    mutation_step = 0.05
    slsqp_freq = 100
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)
    
    no_improvement_count = 0
    generation = 0
    
    while time() - start_time < budget_s * 0.9:
        # Evaluate population
        fitness = []
        for h in population:
            f = evaluate_erdos_solution(h, dx)
            fitness.append(f)
        
        # Find best in current population
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 200:
            break
            
        # Tournament selection
        selected_indices = []
        for _ in range(population_size // 2):
            idx1, idx2 = np.random.choice(population_size, 2, replace=False)
            if fitness[idx1] < fitness[idx2]:
                selected_indices.append(idx1)
            else:
                selected_indices.append(idx2)

        offspring = []
        for i in range(0, len(selected_indices), 2):
            if np.random.rand() < crossover_rate:
                parent1 = population[selected_indices[i]]
                parent2 = population[selected_indices[i+1]]
                
                # Blend crossover
                alpha = np.random.uniform(0.3, 0.7)
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
                
                # Mutation
                if np.random.rand() < mutation_rate:
                    child1 += np.random.normal(0, mutation_step, n_points)
                    child1 = np.clip(child1, 0, 1)
                    # Normalize to satisfy constraint
                    child1 = child1 * (n_points / 2) / np.sum(child1)
                    
                if np.random.rand() < mutation_rate:
                    child2 += np.random.normal(0, mutation_step, n_points)
                    child2 = np.clip(child2, 0, 1)

                    child2 = child2 * (n_points / 2) / np.sum(child2)
                    
                offspring.append(child1)
                offspring.append(child2)

                offspring.append(population[selected_indices[i]])

        elite_indices = np.argsort(fitness)[:population_size//2]

        population = elite + offspring[:population_size - len(elite)]

        mutation_step *= 0.98

                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
