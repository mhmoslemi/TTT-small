import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def ensure_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Multi-resolution optimization
    resolutions = [35, 70, 140, 280]  # increasing resolution
    
    # Simulated Annealing parameters
    t_initial = 1.0
    cooling_rate = 0.99995
    min_temp = 1e-8
    max_iter_per_res = 5000
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_current = np.interp(np.linspace(0, 1, res), 
                                np.linspace(0, 1, n_points), 
                                h_start)
        else:
            h_current = h_start[:res]
            if len(h_current) < res:
                h_current = np.pad(h_current, (0, res - len(h_current)), 
                                 mode='constant', constant_values=h_current[-1])
        
        n_points_current = res
        dx_current = 2.0 / n_points_current
        h_current = ensure_feasible(h_current)
        current_obj = evaluate_erdos_solution(h_current, dx_current)
        
        if current_obj < best_c5:
            best_c5 = current_obj
            best_h = h_current.copy()
            
        temp = t_initial
        patience_count = 0
        max_patience = 500
        
        iter_count = 0
        while iter_count < max_iter_per_res:
            if time() - start_time > budget_s * 0.95:
                break
                
            iter_count += 1
            if patience_count >= max_patience:
                break
                
            # Make a small random perturbation
            h_new = h_current.copy()
            idx = np.random.randint(len(h_new))
            delta = np.random.normal(0, temp * 0.1)
            h_new[idx] += delta
            h_new = ensure_feasible(h_new)
            
            new_obj = evaluate_erdos_solution(h_new, dx_current)
            
            # Accept or reject the new solution based on SA criterion
            if new_obj < current_obj:
                h_current = h_new
                current_obj = new_obj
                patience_count = 0
                if current_obj < best_c5:
                    best_c5 = current_obj
                    best_h = h_current.copy()
            else:
                # Accept with probability based on temperature
                prob = np.exp(-(new_obj - current_obj) / temp)
                if np.random.rand() < prob:
                    h_current = h_new
                    current_obj = new_obj
                    patience_count = 0
                else:
                    patience_count += 1
                    
            temp *= cooling_rate
            if temp < min_temp:
                temp = min_temp
        
        # Refinement at each resolution level using SLSQP
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx_current)
            
            def constraint_sum(h):
                return np.sum(h) - n_points_current / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points_current)]
            
            res = minimize(objective, h_current, 
                          method='SLSQP', 
                          bounds=bounds, 
                          constraints=cons_sum, 
                          options={'maxiter': 200, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
            
        h_start = best_h[:n_points] if len(best_h) > n_points else best_h
        
    # Final verification
    final_obj = evaluate_erdos_solution(best_h, 2.0 / len(best_h))
    
    return (list(best_h), final_obj, len(best_h))
