import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Configuration for convex optimization
    max_iterations = 2000
    K = 10  # Number of top overlaps to consider
    filter_bandwidth = 5
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Get top K overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        top_k_indices = np.argsort(corr)[-K:][::-1]
        
        # Formulate convex problem
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Objective: minimize sum of top K overlaps
        objective = 0
        for idx in top_k_indices:
            shift = idx - (n_points - 1)
            obj_term = 0
            for i in range(n_points):
                j = i + shift
                if 0 <= j < n_points:
                    obj_term += h_var[i] * (1 - h_var[j])
                else:
                    obj_term += h_var[i] * (1 - 0)  # Assuming 0 outside
            objective += obj_term
        
        # Solve convex relaxation
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                if np.all(h_new >= 0) and np.all(h_new <= 1):
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                else:
                    h_new = h.copy()
            else:
                h_new = h.copy()
        except:
            h_new = h.copy()
        
        # Apply low-pass filter
        filtered_h = np.zeros(n_points)
        for i in range(n_points):
            total = 0
            count = 0
            for j in range(max(0, i - filter_bandwidth), min(n_points, i + filter_bandwidth + 1)):
                total += h_new[j]
                count += 1
            filtered_h[i] = total / count
        
        h = filtered_h.copy()
        
        # Evaluate solution
        c5_new = evaluate_erdos_solution(h, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Increase K if needed to increase focus
        if iteration % 100 == 0 and K < 30:
            K = min(K + 2, 30)
            
        # Adjust filter bandwidth
        if iteration % 50 == 0:
            filter_bandwidth = max(3, filter_bandwidth - 1)
            
        # Early stopping
        if iteration > 0 and iteration % 200 == 0 and c5_new >= best_c5:
            if K == 30:
                break
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
