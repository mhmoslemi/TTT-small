import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Regularization parameters
    lambda_reg = 0.1
    alpha_reg = 0.01
    
    def solve_convex_relaxation(h_current):
        # Convex relaxation: minimize sum_k (sum_i h_i * (1 - h_{i+k}))^2 + alpha * sum h_i^2
        h = Variable(n_points)
        objective = 0
        # Simplified approximation - ignore full correlation matrix for convexity
        # Instead use a regularization trick that mimics the behavior
        for k in range(n_points):
            corr_term = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_term += h[i] * (1 - h[j])
            objective += square(corr_term) 
        
        objective += alpha_reg * sum_squares(h)
        constraints = [sum(h) == n_points/2, h >= 0, h <= 1]
        prob = Problem(Minimize(objective), constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            return np.array(h.value).flatten()
        except:
            return h_current.copy()
            
    def coordinate_descent_step(h_current, fixed_ratio=0.8):
        # Fixed coordinate selection
        indices = np.random.permutation(n_points)
        fixed_indices = set(indices[:int(n_points * fixed_ratio)])
        var_indices = [i for i in range(n_points) if i not in fixed_indices]
        
        # Create a new array with ones and zeros for seeded initialization
        h_new = h_current.copy()
        
        # Optimize only the variable indices
        for _ in range(20):
            if time.time() - start_time > budget_s * 0.95:
                break
            for i in var_indices:
                # Simple gradient-based update for single variable
                old_val = h_new[i]
                best_val = old_val
                best_score = float('inf')
                
                # Try a few nearby values
                candidates = [max(0, min(1, old_val + delta)) for delta in [-0.1, -0.05, 0, 0.05, 0.1]]
                current_corr = np.correlate(h_new, 1 - h_new, mode='full') * dx
                max_corr = np.max(current_corr)
                base_score = max_corr
                
                for val in candidates:
                    h_test = h_new.copy()
                    h_test[i] = val
                    test_corr = np.correlate(h_test, 1 - h_test, mode='full') * dx
                    test_max = np.max(test_corr)
                    test_score = test_max
                
                    if test_score < best_score:
                        best_score = test_score
                        best_val = val
                        
                h_new[i] = best_val
                
            # Post-process: ensure sum constraint
            h_new = h_new * (n_points/2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            
        return h_new
        
    # Main optimization loop
    no_improvement_count = 0
    max_no_improvement = 50
    
    for iteration in range(500):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Convex relaxation step
        relaxed_h = solve_convex_relaxation(best_h)
        relaxed_h = relaxed_h * (n_points/2) / np.sum(relaxed_h)
        relaxed_h = np.clip(relaxed_h, 0, 1)
        
        # Coordinate descent refinement
        refined_h = coordinate_descent_step(relaxed_h)
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(refined_h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = refined_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final refinement using scipy
    try:
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)
        
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            final_h = res.x
            final_h = final_h * (n_points/2) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5  # type: ignore
    except:
        pass
        
    return (list(best_h), best_c5, n_points)
