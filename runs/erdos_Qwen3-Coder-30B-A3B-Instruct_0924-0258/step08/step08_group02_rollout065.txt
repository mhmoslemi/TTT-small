import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize with feasible solution
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    
    # Hybrid approach: alternating convex relaxation and coordinate descent
    max_iter = 1000
    no_improvement_count = 0
    max_no_improvement = 50
    
    def project_h(h, n_points, dx):
        """Project h to satisfy integral constraint"""
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def solve_convex_relaxation(h, n_points, dx, lambda_reg=0.1, alpha_reg=0.01):
        """Solve convex relaxation of the problem"""
        # Create variables
        h_vars = cp.Variable(n_points)
        
        # Objective: minimize sum of squared overlaps + regularization
        obj = 0
        for k in range(n_points):
            # Compute the correlation term for lag k
            corr_k = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_k += h_vars[i] * (1 - h_vars[j])
            obj += cp.square(corr_k) * dx
        
        obj += alpha_reg * cp.sum_squares(h_vars)
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Solve
        prob = cp.Problem(cp.Minimize(obj), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return h_vars.value
        except:
            return h.copy()
    
    # Coordinate descent with fixed subset of variables
    def coordinate_descent_step(h, n_points, dx, fixed_vars=None, iter_count=50):
        if fixed_vars is None:
            # Randomly select 20% to optimize
            n_optimize = max(1, n_points // 5)
            fixed_vars = np.random.choice(n_points, n_points - n_optimize, replace=False)
        
        h_new = h.copy()
        
        # Optimize the complement variables using gradient descent
        for k in range(iter_count):
            # Compute gradient for current approximation of max overlap
            overlaps = np.correlate(h_new, 1 - h_new, mode='full') * dx
            l_max = np.argmax(overlaps)
            
            # Gradient for fixed coordinates only
            grad = np.zeros(n_points)
            for i in range(n_points):
                if i in fixed_vars:
                    continue
                j = (i + l_max) % n_points
                grad[i] = -2 * dx * (1 - h_new[j] - h_new[i])
            
            # Gradient step
            step_size = 0.01
            h_new = h_new - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            
            # Ensure sum constraint
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Update fixed variables from original
            for i in fixed_vars:
                h_new[i] = min(1.0, max(0.0, h_new[i]))
        
        return h_new, fixed_vars
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convex relaxation step
        try:
            h_relaxed = solve_convex_relaxation(h, n_points, dx)
            h = project_h(h_relaxed, n_points, dx)
        except:
            h = project_h(h, n_points, dx)
        
        # Coordinate descent refinement
        for _ in range(3):
            # Choose random variables to fix for this step
            n_optimize = max(1, n_points // 5)
            fixed_vars = np.random.choice(n_points, n_points - n_optimize, replace=False)
            
            h_new, _ = coordinate_descent_step(h, n_points, dx, fixed_vars)
            h = project_h(h_new, n_points, dx)
        
        # Evaluate and store best
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            break
    
    # Final refinement with local optimizer
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def obj(h_vec):
            overlaps = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
            return -np.max(overlaps)  # Negative because we minimize
        
        res = minimize(
            obj,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
