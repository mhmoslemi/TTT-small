import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    num_generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    stagnation_limit = 100
    refinement_period = 100
    
    # Initialize population
    population = []
    population_fitness = []
    
    # Add initial solution first
    population.append(h_start.copy())
    population_fitness.append(evaluate_erdos_solution(h_start, dx))
    
    # Generate rest of population
    for _ in range(pop_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        h_rand = np.clip(h_rand, 0, 1)
        population.append(h_rand)
        population_fitness.append(evaluate_erdos_solution(h_rand, dx))
    
    # Best solution tracking
    best_idx = np.argmin(population_fitness)
    best_individual = population[best_idx].copy()
    best_fitness = population_fitness[best_idx]
    
    # GA loop
    generation = 0
    stagnation_count = 0
    
    while time() - start_time < budget_s * 0.7 and generation < num_generations:
        # Sort population by fitness
        sorted_indices = np.argsort(population_fitness)
        population = [population[i] for i in sorted_indices]
        population_fitness = [population_fitness[i] for i in sorted_indices]
        
        # Update best if improved
        if population_fitness[0] < best_fitness:
            best_fitness = population_fitness[0]
            best_individual = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count >= stagnation_limit:
            # Restart with random solutions
            for i in range(1, pop_size):
                h_rand = np.random.rand(n_points)
                h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
                h_rand = np.clip(h_rand, 0, 1)
                population[i] = h_rand
                population_fitness[i] = evaluate_erdos_solution(h_rand, dx)
            stagnation_count = 0
        
        # Create new population
        new_population = [population[0].copy()]  # Elitism
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = np.random.randint(0, pop_size // 2)
            parent2_idx = np.random.randint(0, pop_size // 2)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover with probability
            if np.random.rand() < crossover_rate:
                # Blend crossover (one-point crossover variation)
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            mutation_strength = mutation_rate * (0.05 ** (generation / num_generations))
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, mutation_strength)
                    child2[i] += np.random.normal(0, mutation_strength)
                    
            # Ensure feasibility
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            # Add to new population
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        # Evaluate new population
        population = new_population[:pop_size]
        population_fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        generation += 1
        
        # Periodic SLSQP refinement
        if generation % refinement_period == 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_individual, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-12})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_fitness:
                        best_fitness = c5_refined
                        best_individual = h_refined
            except Exception:
                pass
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_individual, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            final_h = res.x
            final_c5 = objective(final_h)
            if final_c5 < best_fitness:
                best_individual = final_h
                best_fitness = final_c5
    except Exception:
        pass
    
    return (list(best_individual), best_fitness, n_points)
