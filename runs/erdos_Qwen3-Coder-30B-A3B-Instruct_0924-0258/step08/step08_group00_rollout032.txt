import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    random.seed(seed)
    start_time = time()

    # Strategy 1: Local search with simulated annealing approach
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    current_c5 = evaluate_erdos_solution(h, dx)
    temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-8
    max_iter = 10000
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break

        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05)
        h_new = h.copy()
        h_new[idx] += delta
        h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject with simulated annealing probability
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temp):
            h = h_new
            current_c5 = new_c5
            
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp
            
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

            return evaluate_erdos_solution(h, dx)

                break

                h_refined = h.copy()

                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                
                res = minimize(objective, h_refined, method=method, 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 500, 'ftol': 1e-9})
                
                if res.success:
                    h_final = res.x
                    c5_final = objective(h_final)
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
            except Exception:
                continue

            max_lag = max_lag_idx - (n_points - 1)

            else:
                step *= 0.5
                if step < 1e-6:
                    break
    
    return (list(best_h), best_c5, n_points)
