import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        for i in range(n_points // 2, n_points):
            h_start[i] = 1.0
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Thresholding-based optimization
    threshold = 0.1
    cooling = 0.95
    max_iterations = 500
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        max_overlap = np.max(overlaps)
        k_max = np.argmax(overlaps)
        
        # Compute gradient (approximate)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient[i] = -2 * dx * (1 - best_h[j] - best_h[i])
        
        # Sort indices by gradient magnitude
        idx_sorted = np.argsort(np.abs(gradient))[::-1]
        
        # For top 50%, apply thresholding
        threshold_indices = idx_sorted[:n_points // 2]
        
        # Save old values for redistribution
        old_h = best_h.copy()
        
        # Apply thresholding
        for i in threshold_indices:
            if best_h[i] >= 1 - threshold:
                best_h[i] = 1.0
            elif best_h[i] <= threshold:
                best_h[i] = 0.0
            else:
                # Keep it as is, but we'll make sure sum constraint is met

        current_sum = np.sum(best_h)

            diff = n_points/2 - current_sum

                # Need to add more mass
                idx_to_increase = np.argmax(np.abs(gradient))
                best_h[idx_to_increase] = min(1.0, best_h[idx_to_increase] + diff)
            else:
                # Need to remove mass
                idx_to_decrease = np.argmax(np.abs(gradient))
                best_h[idx_to_decrease] = max(0.0, best_h[idx_to_decrease] + diff)

        best_h = np.clip(best_h, 0, 1)

        best_h = best_h * (n_points/2) / np.sum(best_h)
        
        # Evaluate current
        current_c5 = evaluate_erdos_solution(best_h, dx)
        
        if current_c5 < best_c5:

            # best_h is already updated

        if iter % 10 == 0:

    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    
    return (list(best_h), best_c5, n_points)
