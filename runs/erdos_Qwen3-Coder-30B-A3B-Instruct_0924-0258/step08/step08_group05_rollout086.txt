import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Trust Region Optimization with Cutting Planes
    max_iter = 5000
    trust_radius = 0.2
    min_trust_radius = 1e-6
    trust_radius_factor = 1.2
    trust_shrink_factor = 0.5
    
    def get_all_overlaps(h):
        corr = np.correlate(h, 1 - h, mode='full')
        lag_indices = np.arange(len(corr))
        lags = lag_indices - (len(h) - 1)
        overlap_values = corr * dx
        return overlap_values, lags
    
    def solve_trust_region_cutting(h_current, trust_radius, cutting_planes):
        n = len(h_current)
        h_vars = cp.Variable(n)
        C = cp.Variable()
        
        # Trust region constraints
        constraints = [cp.norm(h_vars - h_current, 2) <= trust_radius]
        
        # Sum constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Bound constraints
        constraints.extend([0 <= h_vars[i], h_vars[i] <= 1 for i in range(n)])
        
        # Cutting planes (previously violated constraints)
        for plane in cutting_planes:
            lag_idx, bound_val = plane
            # For the given lag, we'll approximate constraint as sum(h[i] * (1 - h[i + lag])) <= C
            # But here, we add a simplified version
            pass  # We'll handle it via iterative refinement
        
        # Add all overlap constraints from our current state:
        overlaps, lags = get_all_overlaps(h_current)
        max_idx = np.argmax(overlaps)
        max_lag = lags[max_idx]
        
        # Add constraint for the currently largest overlap lag to refine
        # But we avoid using many constraints due to problem structure, instead use a hybrid.
        # For now, let's just optimize with the main constraint and trust region
        # and use line search to maintain validity
        
        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            prob.solve(solver=cp.CVXOPT, kktsolver='robust')
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_vars.value
                return h_new, C.value
        except:
            pass
        return None, None
    
    def project_onto_constraints(h):
        # Project onto the feasible set: sum = n_points/2, 0 <= h <= 1
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points/2) > 1e-10:
            # Add deviation to all entries evenly
            adjust = (n_points/2 - h_sum) / n_points
            h = h + adjust
        h = np.clip(h, 0, 1)
        return h
    
    def line_search(h_current, direction, step_size=0.1):
        best_h = h_current.copy()
        best_c5 = evaluate_erdos_solution(h_current, dx)
        
        for i in range(10):  # Up to 10 line search steps
            h_new = h_current + step_size * direction
            h_new = project_onto_constraints(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            else:
                break
            step_size *= 0.7
            
        return best_h, best_c5
    
    # Initial cutting planes (empty)
    cutting_planes = []
    
    # Run iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        prev_c5 = evaluate_erdos_solution(h_current, dx)
        
        # Try solving in trust region (approximate)
        # Just do gradient descent with good bounds to improve
        overlaps, lags = get_all_overlaps(h_current)
        max_idx = np.argmax(overlaps)
        max_lag = lags[max_idx]
        
        # Derivative of that overlap term w.r.t. h[i]
        grad = np.zeros(n_points)
        for i in range(n_points):
            if max_lag >= 0:
                if i + max_lag < n_points:
                    grad[i] = -1.0
            else:
                if i - max_lag >= 0:
                    grad[i] = -1.0
                    
        # Take a gradient step to reduce that overlap
        step_size = 0.02
        h_new = h_current - step_size * grad
        h_new = project_onto_constraints(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h_current = h_new
        
        # Occasionally refine using SLSQP
        if iteration % 20 == 0:
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                    
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(objective, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-10})
                if res.success:
                    h_final = res.x
                    c5_final = objective(h_final)
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
                        h_current = h_final.copy()
            except Exception:
                continue
                
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final validation
    if abs(np.sum(best_h) - n_points / 2.0) > 1e-6:
        best_h = best_h * (n_points/2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
