import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_coarse = 15
    n_fine = 35
    coarse_iterations = 500
    fine_iterations = 2000
    mutation_rate = 0.1
    step_size = 0.01
    tolerance = 1e-9
    
    # Initialize coarse grid
    h_coarse = np.ones(n_coarse) * 0.5
    
    # If initial construction exists, use it as coarse initialization
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        # Interpolate initial construction to coarse grid
        x_coarse = np.linspace(0, 2, n_coarse)
        x_initial = np.linspace(0, 2, len(h_initial))
        f = interp1d(x_initial, h_initial, kind='linear', fill_value='extrapolate')
        h_coarse = f(x_coarse)
        h_coarse = np.clip(h_coarse, 0, 1)
    
    # Normalize coarse grid
    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    best_coarse = h_coarse.copy()
    best_c5_coarse = evaluate_erdos_solution(h_coarse, 2.0 / n_coarse)
    
    # Coarse grid optimization
    for iteration in range(coarse_iterations):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute correlation
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_coarse - 1)
        
        # Compute gradient
        grad = np.zeros(n_coarse)
        for i in range(n_coarse):
            j = i + max_lag
            if 0 <= j < n_coarse:
                grad[i] = 1 - 2 * h_coarse[i] - h_coarse[j]
            else:
                grad[i] = 1 - 2 * h_coarse[i]
        
        # Gradient step
        h_coarse_new = h_coarse - step_size * grad
        h_coarse_new = np.clip(h_coarse_new, 0, 1)
        
        # Renormalize
        h_coarse_new = h_coarse_new * (n_coarse / 2) / np.sum(h_coarse_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_coarse_new, 2.0 / n_coarse)
        
        if c5_new < best_c5_coarse:
            best_c5_coarse = c5_new
            best_coarse = h_coarse_new.copy()
            h_coarse = h_coarse_new.copy()
        else:
            step_size *= 0.9
            if step_size < tolerance:
                break
    
    # Upsample to fine grid
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    f = interp1d(x_coarse, best_coarse, kind='linear', fill_value='extrapolate')
    h_fine = f(x_fine)
    h_fine = np.clip(h_fine, 0, 1)
    
    # Renormalize fine grid
    h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)
    
    best_fine = h_fine.copy()
    best_c5_fine = evaluate_erdos_solution(h_fine, 2.0 / n_fine)
    
    # Fine grid optimization with alternating scheme
    for iteration in range(fine_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Gradient descent phase
        corr = np.correlate(h_fine, 1 - h_fine, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_fine - 1)
        
        grad = np.zeros(n_fine)
        for i in range(n_fine):
            j = i + max_lag
            if 0 <= j < n_fine:
                grad[i] = 1 - 2 * h_fine[i] - h_fine[j]
            else:
                grad[i] = 1 - 2 * h_fine[i]
        
        h_fine_new = h_fine - step_size * grad
        h_fine_new = np.clip(h_fine_new, 0, 1)
        h_fine_new = h_fine_new * (n_fine / 2) / np.sum(h_fine_new)
        
        c5_new = evaluate_erdos_solution(h_fine_new, 2.0 / n_fine)
        
        # Mutation phase
        if c5_new >= best_c5_fine or iteration % 100 == 0:
            h_fine_mut = h_fine.copy()
            indices = np.random.choice(n_fine, size=int(mutation_rate * n_fine), replace=False)
            h_fine_mut[indices] = np.random.uniform(0, 1, len(indices))
            
            # Normalize
            h_fine_mut = h_fine_mut * (n_fine / 2) / np.sum(h_fine_mut)
            c5_mut = evaluate_erdos_solution(h_fine_mut, 2.0 / n_fine)
            
            if c5_mut < best_c5_fine:
                best_c5_fine = c5_mut
                best_fine = h_fine_mut.copy()
                h_fine = h_fine_mut.copy()
        else:
            if c5_new < best_c5_fine:
                best_c5_fine = c5_new
                best_fine = h_fine_new.copy()
                h_fine = h_fine_new.copy()
    
    # Final re-evaluation and return
    final_c5 = evaluate_erdos_solution(best_fine, 2.0 / n_fine)
    
    return (list(best_fine), final_c5, n_fine)
