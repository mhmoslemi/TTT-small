import numpy as np
from time import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_onto_constraints(h, n_points, dx):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Ensure sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        # Scale to meet sum constraint
        h = h * (n_points/2) / current_sum
        h = np.clip(h, 0, 1)
        
    return h

def line_search(h_current, gradient, dx, n_points, max_step=0.1):
    # Simple backtracking line search
    step_size = max_step
    best_step = 0.0
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    for _ in range(20):  # Max iterations for line search
        h_new = h_current - step_size * gradient
        h_new = project_onto_constraints(h_new, n_points, dx)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_step = step_size
        else:
            step_size *= 0.5
        if step_size < 1e-8:
            break
    
    return best_step

def run(seed=42, budget_s=600, **kwargs):

    h_current = h_start.copy()
    h_current = project_onto_constraints(h_current, n_points, dx)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Periodic perturbation settings
    perturb_interval = 500
    perturb_amount = 0.05
    
    iteration = 0
    step_size = 0.01

    while time() - start_time < budget_s * 0.95:
        # Compute all correlations
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Get top 5 lags (highest overlaps)
        top_lag_indices = np.argsort(corr)[-5:][::-1]
        top_lag_indices = [i - (n_points - 1) for i in top_lag_indices]
        
        # Compute combined gradient for top lags
        gradient = np.zeros(n_points)
        for lag in top_lag_indices:
            if lag >= 0:
                for i in range(n_points - lag):
                    gradient[i] += 1 - h_current[i + lag]
                    if i + lag < n_points:
                        gradient[i + lag] -= h_current[i]
            else:
                for i in range(n_points + lag):
                    gradient[i] += 1 - h_current[i - lag]
                    if i - lag >= 0:
                        gradient[i - lag] -= h_current[i]
        
        # Line search for step size
        ls_step = line_search(h_current, gradient, dx, n_points, step_size)
        h_new = h_current - ls_step * gradient
        h_new = project_onto_constraints(h_new, n_points, dx)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject step
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            step_size = min(0.1, step_size * 1.1)
        else:
            step_size *= 0.5
            if step_size < 1e-10:
                break
        
        h_current = h_new
        iteration += 1
        
        # Periodic perturbation to escape local minima
        if iteration % perturb_interval == 0 and iteration > 0:
            noise = np.random.normal(0, perturb_amount, n_points)
            h_current = h_current + noise
            h_current = project_onto_constraints(h_current, n_points, dx)
    
    return (list(best_h), best_c5, n_points)
