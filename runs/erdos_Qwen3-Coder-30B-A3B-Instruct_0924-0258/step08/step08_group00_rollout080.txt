import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a combination of variants
    # Variant 1: Coordinate descent with adaptive step size
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Gradient descent with backtracking line search and adaptive step sizes
    max_iter = 1000
    patience = 50
    no_improvement = 0
    base_step_size = 0.01
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate overlap for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            break
            
        # Compute gradient for this specific lag
        max_lag = max_lag_idx - (n_points - 1)
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 2 * (1 - h[j] - h[i]) / n_points

        # Backtracking line search
        step_size = base_step_size
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        iter_count = 0
        while iter_count < 10:
            if evaluate_erdos_solution(h_new, dx) < max_overlap:
                h = h_new.copy()
                break
            else:
                step_size *= 0.5
                h_new = h - step_size * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                iter_count += 1
        else:
            # If line search failed, reduce learning rate
            base_step_size *= 0.5
            step_size = base_step_size

    # Final refinement with trust-constr
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h, method='trust-constr', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'gtol': 1e-9, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Post-processing to refine the solution
    h_post = best_h.copy()

        # Normalize again to be sure
        h_post = h_post * (n_points/2) / np.sum(h_post)
        
        # Try adding local noise and optimizing further
        h_noisy = h_post + np.random.normal(0, 0.01, n_points)
        h_noisy = np.clip(h_noisy, 0, 1)
        h_noisy = h_noisy * (n_points/2) / np.sum(h_noisy)
        
        def obj_noisy(h):
            return evaluate_erdos_solution(h, dx)
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj_noisy, h_noisy, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        
        if res.success:
            h_refine = res.x
            c5_refine = obj_noisy(h_refine)
            if c5_refine < best_c5:
                best_h = h_refine
                best_c5 = c5_refine
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
