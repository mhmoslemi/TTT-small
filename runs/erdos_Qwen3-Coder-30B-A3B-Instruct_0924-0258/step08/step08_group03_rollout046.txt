import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Adaptive perturbation parameters
    epsilon_0 = 0.001
    beta = 0.99
    max_iter = 5000
    top_k = 100
    
    # Track best solution
    best_solution_history = []
    
    for t in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        # Get indices of top correlations (largest overlaps)
        top_k_indices = np.argsort(corr)[-top_k:]
        
        # Create CVXPY variables
        h_var = cp.Variable(n_points)
        
        # Objective with convex perturbation
        objective = 0
        for k_idx in top_k_indices:
            # Compute shift index
            shift = k_idx - (n_points - 1)
            # Simple quadratic penalty for this correlation
            if shift >= 0:
                # h[i] * (1 - h[i+shift]) for i = 0 to n_points-shift-1
                objective += cp.sum(cp.multiply(h_var[:-shift], 1 - h_var[shift:]))
            else:
                # h[i-shift] * (1 - h[i]) for i = 0 to n_points+shift-1
                objective += cp.sum(cp.multiply(h_var[-shift:], 1 - h_var[:-shift]))
        
        # Add convex perturbation term
        epsilon = epsilon_0 * (beta ** t)
        objective += epsilon * cp.sum_squares(h_var)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Solve optimization problem
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
            else:
                h_new = h.copy()  # fallback
        except:
            h_new = h.copy()  # fallback
            
        # Project to feasible region
        h_new = project_to_feasible(h_new, n_points, dx)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update best solution
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Update h
        h = h_new.copy()
        
        # Store for stagnation check
        best_solution_history.append(c5_new)
        if len(best_solution_history) > 100:
            best_solution_history.pop(0)
        
        # Early termination criteria
        if len(best_solution_history) >= 100:
            if max(best_solution_history[-50:]) - min(best_solution_history[-50:]) < 1e-8:
                if len(best_solution_history) >= 200:
                    break
    
    # Final local refinement
    if time() - start_time < budget_s * 0.9:
        # Generate random candidates
        candidates = []
        for _ in range(100):
            h_pert = best_h + np.random.normal(0, 0.001, n_points)
            h_pert = project_to_feasible(h_pert, n_points, dx)
            c5_test = evaluate_erdos_solution(h_pert, dx)
            candidates.append((c5_test, h_pert))
        
        if candidates:
            best_candidate = min(candidates, key=lambda x: x[0])
            if best_candidate[0] < best_c5:
                best_c5 = best_candidate[0]
                best_h = best_candidate[1]
    
    return (list(best_h), best_c5, n_points)
