import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search parameters
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    # Sample shifts for constraint checking
    shift_samples = 100
    shifts = np.random.choice(n_points, shift_samples, replace=False)
    
    def is_feasible(c5):
        # Define the optimization variables
        h_vars = cp.Variable(n_points)
        
        # Define constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Add overlap constraints
        for k in shifts:
            # For shift k, we compute sum h[i] * (1 - h[(i+k) % n])
            # Equivalent to sum h[i] - h[i] * h[(i+k) % n]
            # That is sum h[i] - sum h[i] * h[(i+k) % n]
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_vars[i] * (1 - h_vars[j])
            constraints.append(expr <= c5 * dx)
        
        # Create and solve the problem
        problem = cp.Problem(cp.Minimize(0), constraints)
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            return problem.status == cp.OPTIMAL
        except:
            return False
    
    # Binary search
    while time() - start_time < budget_s * 0.9 and high - low > tolerance:
        mid = (low + high) / 2
        if is_feasible(mid):
            high = mid
            best_c5 = mid
        else:
            low = mid
    
    # Local refinement with gradient descent
    def objective(h_vec):
        # Return negative of overlap (since we minimize)
        return -evaluate_erdos_solution(h_vec, dx)
    
    def constraint_fn(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    def constraint_fn_jac(h_vec):
        return np.ones_like(h_vec)
    
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_fn, 'jac': constraint_fn_jac}
    
    # Optimize with bounds and constraints
    result = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
    )
    
    if result.success:
        refined_h = result.x
        # Ensure it's within bounds
        refined_h = np.clip(refined_h, 0, 1)
        # Renormalize
        refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
        refined_c5 = evaluate_erdos_solution(refined_h, dx)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
    
    # Final check if we're under target
    if best_c5 < 0.3807:
        pass  # Already good enough
    
    return (list(best_h), best_c5, n_points)
