import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Parameters for convex optimization
    max_iterations = 1000
    K = 10  # Number of top overlaps to consider
    filter_bandwidth = 5
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        
        # Get indices of top K overlaps (excluding zero lag)
        top_k_indices = np.argpartition(-corr, K)[:K]
        top_k_indices = [i for i in top_k_indices if i != n_points - 1]  # Exclude zero lag
        if len(top_k_indices) < K:
            top_k_indices = np.argpartition(-corr, K)[:K]
            top_k_indices = [i for i in top_k_indices if i != n_points - 1][:K]
        
        # Create convex optimization problem
        h_var = cp.Variable(n_points)
        
        # Objective: Linear combination of top K overlaps with equal weights
        objective = cp.Minimize(cp.sum([cp.sum(h_var * (1 - h_var[(i + np.arange(n_points)) % n_points])) 
                                        for i in top_k_indices[:K]]))
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Solve optimization problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
            else:
                h_new = h
        except:
            h_new = h
            
        # Apply low-pass filter
        if len(h_new) >= filter_bandwidth:
            kernel = np.ones(filter_bandwidth) / filter_bandwidth
            h_filtered = np.convolve(h_new, kernel, mode='same')
        else:
            h_filtered = h_new
            
        # Renormalize to maintain constraint
        h_filtered = h_filtered * (n_points / 2) / np.sum(h_filtered)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_filtered, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_filtered.copy()
            h = h_filtered.copy()
            
            # Increase K if we're making good improvements
            if K < 20:
                K = min(K + 1, 20)
        else:
            # Apply a small perturbation if no improvement
            perturbation = np.random.normal(0, 0.01, n_points)
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
                h = h_perturbed.copy() 
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
