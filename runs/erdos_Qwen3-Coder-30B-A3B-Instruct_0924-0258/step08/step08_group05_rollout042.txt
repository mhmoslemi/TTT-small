import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    pop_size = 50

    fitness = []

    def create_feasible_h():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h

    if 'initial_h_values' in globals():
        h_norm = h_start * (n_points/2) / np.sum(h_start)
        population.append(h_norm)

    # Fill rest of population
    while len(population) < pop_size:
        h = create_feasible_h()
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Evolutionary loop
    best_fitness = min(fitness)
    best_individual = population[fitness.index(best_fitness)].copy()
    no_improvement = 0
    
    max_generations = 10000
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:

        # Tournament selection
        def tournament_select():
            k = 5
            indices = random.sample(range(pop_size), k)
            best_idx = min(indices, key=lambda i: fitness[i])
            return population[best_idx].copy()
        
        # Crossover
        def crossover(p1, p2):
            child = np.zeros(n_points)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = p1[i]
                else:
                    child[i] = p2[i]
            # Normalize
            child = child * (n_points/2) / np.sum(child)
            return child

        def mutate(h):
            child = h.copy()
            # Flip mutation
            if random.random() < 0.5:
                i, j = random.sample(range(n_points), 2)
                child[i] = max(0, min(1, child[i] + 0.1))
                child[j] = max(0, min(1, child[j] - 0.1))
            # Perturb mutation
            if random.random() < 0.5:
                for i in range(n_points):
                    if random.random() < 0.1:
                        child[i] *= np.random.uniform(0.9, 1.1)
            # Normalize
            child = child * (n_points/2) / np.sum(child)
            return child
        
        # Create new population
        new_population = [best_individual.copy()]
        new_fitness = [best_fitness]

            p1 = tournament_select()
            p2 = tournament_select()

            child = mutate(child)
            new_population.append(child)
            new_fitness.append(evaluate_erdos_solution(child, dx))

        fitness = new_fitness

        if current_best_fitness < best_fitness:

            best_individual = population[fitness.index(best_fitness)].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > 100:
            break
            
    return (list(best_individual), best_fitness, n_points)
