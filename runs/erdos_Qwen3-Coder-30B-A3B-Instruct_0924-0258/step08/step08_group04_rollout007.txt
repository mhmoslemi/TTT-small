import numpy as np
from time import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def compute_gradient(h, max_lags, n_points):
    """Compute combined gradient for top k lags"""
    grad = np.zeros(n_points)
    for k in max_lags:
        if k >= 0:
            # h[i] * (1 - h[i + k]) contributes to gradient
            for i in range(n_points - k):
                grad[i] += 1 - h[i + k]
                if i + k < n_points:
                    grad[i + k] -= h[i]
        else:
            # k < 0, so h[i] * (1 - h[i + k]) where i + k >= 0
            for i in range(-k, n_points):
                grad[i] += 1 - h[i + k]
                if i + k >= 0:
                    grad[i + k] -= h[i]
    return grad

def line_search(h_current, grad, dx, n_points, max_iter=20):
    """Backtracking line search to find step size"""
    alpha = 1.0
    step_size = 0.5
    
    # Evaluate current C5
    current_c5 = evaluate_erdos_solution(h_current, dx)
    
    for _ in range(max_iter):
        h_new = h_current - alpha * grad
        h_new = project_to_feasible(h_new, n_points)
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < current_c5:
            return alpha, h_new
        alpha *= step_size
    
    return alpha, h_new

    h_current = h_start.copy()
    h_current = project_to_feasible(h_current, n_points)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    max_iter = 10000
    iter_count = 0
    improvement_count = 0
    no_improvement_count = 0
    step_size = 0.05
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all correlations
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Get indices of top 5 lags by overlap
        top_k = min(5, len(corr))
        top_lag_indices = np.argsort(corr)[-top_k:][::-1]
        max_lags = [idx - (len(h_current) - 1) for idx in top_lag_indices]
        
        # Compute gradient
        grad = compute_gradient(h_current, max_lags, n_points)
        
        # Line search for step size
        alpha, h_new = line_search(h_current, grad, dx, n_points)
        
        # Project
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if it improves
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            improvement_count += 1
            no_improvement_count = 0
            # Increase step size
            step_size = min(1.0, step_size * 1.1)

            # Decrease step size
            step_size *= 0.5
            # Add some perturbation to escape local minima
            if no_improvement_count > 100:
