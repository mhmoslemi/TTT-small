import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_onto_feasible(h, n_points):
    """Project h onto feasible set: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # First clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Then adjust to satisfy sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Simple scaling approach
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.full_like(h, target_sum / n_points)
    
    # Ensure again that values are within [0,1]
    h = np.clip(h, 0, 1)
    
    return h

def compute_gradients(h, max_lags, n_points):
    """Compute combined gradient for multiple top lags"""
    gradient = np.zeros(n_points)
    for lag in max_lags:
        if lag >= 0:
            for i in range(n_points - lag):
                gradient[i] += 1 - h[i + lag]
                if i + lag < n_points:
                    gradient[i + lag] -= h[i]
        else:
            for i in range(n_points + lag):
                gradient[i] += 1 - h[i - lag]
                if i - lag >= 0:
                    gradient[i - lag] -= h[i]
    return gradient

def line_search(h_current, grad, n_points, dx, max_iter=20):
    """Backtracking line search"""
    alpha = 1.0
    c1 = 1e-4
    rho = 0.5
    
    current_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Compute directional derivative
    dir_deriv = np.dot(grad, grad)
    if abs(dir_deriv) < 1e-12:
        return 0.0
    
    for _ in range(max_iter):
        h_new = h_current - alpha * grad
        h_new = project_onto_feasible(h_new, n_points)
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 <= current_c5 + c1 * alpha * dir_deriv:
            return alpha
        alpha *= rho
    
    return alpha

    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Parameters
    max_lags_count = 5
    step_size = 0.01
    step_size_min = 1e-10
    perturbation_rate = 0.1
    perturbation_magnitude = 0.05
    max_no_improvement_iter = 2000
    no_improvement_count = 0
    
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95:
        # Compute all correlations
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Get top K lags by correlation
        top_lags = np.argsort(-corr)[:max_lags_count]
        
        # Compute combined gradient for top lags
        gradient = compute_gradients(h_current, top_lags, n_points)
        
        # Line search for optimal step
        alpha = line_search(h_current, gradient, n_points, dx)
        
        # Update step
        h_new = h_current - alpha * gradient

        h_new = project_onto_feasible(h_new, n_points)

        c5_new = evaluate_erdos_solution(h_new, dx)
