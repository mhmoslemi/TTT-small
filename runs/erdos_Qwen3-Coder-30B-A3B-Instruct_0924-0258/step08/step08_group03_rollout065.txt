import numpy as np
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search parameters
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    # Helper function for SDP check
    def check_feasible(c5_val):
        # Create variables
        h_vars = cp.Variable(n_points)
        Y = cp.Variable((n_points, n_points), symmetric=True)
        
        # Constraints
        constraints = [
            Y >> 0,  # Y is positive semidefinite
            cp.diag(Y) == h_vars,  # diagonal elements of Y equal h
            cp.sum(h_vars) == n_points / 2,  # normalization
        ]
        
        # Overlap constraints
        # Sample 100 random shifts
        shifts = np.random.randint(1, n_points, 100)
        for k in shifts:
            # Sum over valid indices
            expr = 0
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    expr += Y[i, j]
            constraints.append(cp.sum(h_vars) - expr <= c5_val * dx)
        
        # Objective (minimize C5)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=MOSEK, verbose=False)
            return prob.status == 'optimal'
        except:
            return False
    
    # Local optimization with gradient descent
    def local_optimize(h_in, max_iter=2000):
        h = h_in.copy()
        dx_here = 2.0 / len(h)
        step_size = 0.01
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute correlation
            corr = np.correlate(h, 1 - h, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (len(h) - 1)
            
            # Compute gradient
            grad = np.zeros(len(h))
            for j in range(len(h)):
                k = j + max_lag
                if 0 <= k < len(h):
                    grad[j] = 1 - 2 * h[j] - h[k]
                else:
                    grad[j] = 1 - 2 * h[j]
            
            # Update
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx_here)
            
            if c5_new < evaluate_erdos_solution(h, dx_here):
                h = h_new.copy()
                step_size *= 1.05
            else:
                step_size *= 0.5
                if step_size < 1e-8:
                    break
        
        return h
    
    # Binary search
    while time() - start_time < budget_s * 0.95 and high - low > tolerance:
        mid = (low + high) / 2
        if check_feasible(mid):
            high = mid
            # Try to refine
            h_proj = np.random.rand(n_points)
            h_proj = h_proj / np.sum(h_proj) * n_points / 2
            h_proj = np.clip(h_proj, 0, 1)
            h_proj = local_optimize(h_proj)
            
            c5_check = evaluate_erdos_solution(h_proj, dx)
            if c5_check < best_c5:
                best_c5 = c5_check
                best_h = h_proj.copy()
        else:
            low = mid
    
    # Final local optimization on best found
    best_h = local_optimize(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
