import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with gradient refinement
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    T = T_init
    max_iter = int(1e5)
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Annealing loop
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Perturb two random indices
        idx1, idx2 = random.sample(range(n_points), 2)
        epsilon = 0.1 * T
        
        # Adjust values
        delta1 = np.random.uniform(-epsilon, epsilon)
        delta2 = -delta1  # Keep sum constraint
        
        new_h = h.copy()
        new_h[idx1] += delta1
        new_h[idx2] += delta2
        
        # Clamp to [0,1]
        new_h = np.clip(new_h, 0, 1)
        
        # Maintain sum constraint
        current_sum = np.sum(new_h)
        new_h = new_h * (n_points/2) / current_sum
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject
        if new_c5 < best_c5:
            h = new_h
            best_c5 = new_c5
            best_h = new_h.copy()
        else:
            # Accept with probability
            delta_c5 = new_c5 - best_c5
            prob = np.exp(-delta_c5 / T)
            if random.random() < prob:
                h = new_h
        
        # Cool down
        T *= alpha
        if T < T_min:
            break
    
    # Local refinement with L-BFGS-B
    try:
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)
        
        # Define bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run optimization
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
