import numpy as np
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize bounds for binary search
    low = 0.38
    high = 0.38085
    best_c5 = 1.0
    best_h = h.copy()
    
    # Sample shifts for SDP constraints (100 random shifts)
    shifts = np.random.choice(range(1, n_points), size=100, replace=False)
    
    while time() - start_time < budget_s * 0.95 and high - low > 1e-5:
        mid = (low + high) / 2
        # SDP formulation
        Y = Semidef(n_points)
        h_var = Variable(n_points)
        
        # Constraints
        constraints = [diag(Y) == h_var,
                       sum(h_var) == n_points / 2,
                       h_var >= 0,
                       h_var <= 1]
        
        # Overlap constraints for sampled shifts
        for k in shifts:
            # For shift k: sum_i h[i] * (1 - h[i+k]) <= mid * dx
            # Which is: sum_i h[i] - sum_i h[i] * h[i+k] <= mid * dx
            # Which is: sum_i h[i] - sum_i Y[i, i+k] <= mid * dx
            constraints.append(sum(h_var) - sum(Y[i, (i+k) % n_points] for i in range(n_points)) <= mid * dx)
        
        # Objective: minimize mid (will be handled by binary search)
        prob = Problem(Maximize(0), constraints)
        
        try:
            prob.solve(solver=MOSEK, verbose=False)
            if prob.status == 'optimal':
                h_sol = h_var.value
                # Normalize
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                h_sol = np.clip(h_sol, 0, 1)
                c5_sol = evaluate_erdos_solution(h_sol, dx)
                
                if c5_sol < best_c5:
                    best_c5 = c5_sol
                    best_h = h_sol.copy()
                high = mid
            else:
                low = mid
        except Exception:
            low = mid
            
    # Local refinement with gradient descent
    h_refined = best_h.copy()
    step_size = 0.01
    tolerance = 1e-9
    max_iter = 2000
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        # Compute overlaps
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for j in range(n_points):
            k = j + max_lag
            if 0 <= k < n_points:
                grad[j] = 1 - 2 * h_refined[j] - h_refined[k]
            else:
                grad[j] = 1 - 2 * h_refined[j]
        
        # Update
        h_new = h_refined - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check improvement
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_refined = h_new.copy()
        else:
            step_size *= 0.5
            if step_size < tolerance:
                break
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
