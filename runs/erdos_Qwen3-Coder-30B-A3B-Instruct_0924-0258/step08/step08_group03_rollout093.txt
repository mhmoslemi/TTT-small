import numpy as np
from time import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search bounds
    low, high = 0.38, 0.38085
    tolerance = 1e-5
    
    def solve_sdp(c5_val):
        # Create variables
        h_vars = cp.Variable(n_points)
        Y = cp.Variable((n_points, n_points), symmetric=True)
        
        # Constraints
        constraints = [
            Y >> 0,  # Y is positive semidefinite
            cp.diag(Y) == h_vars,  # diagonal constraint
            cp.sum(h_vars) == n_points / 2,  # normalization
        ]
        
        # Add overlap constraints for sampled shifts
        shifts = np.random.choice(n_points, size=min(100, n_points), replace=False)
        for k in shifts:
            if k == 0:
                continue
            # We need sum(h[i] * (1 - h[i+k])) <= c5_val * dx
            # This is equivalent to sum(h[i]) - sum(Y[i, i+k]) <= c5_val * dx
            # In matrix form: sum(h_vars) - sum(Y[i, i+k]) <= c5_val * dx
            if k < n_points:
                overlap_sum = cp.sum(Y[np.arange(n_points-k), np.arange(k, n_points)])
                constraints.append(cp.sum(h_vars) - overlap_sum <= c5_val * dx)
            else:
                overlap_sum = cp.sum(Y[np.arange(n_points-(n_points-k)), np.arange(n_points-k, n_points)])
                constraints.append(cp.sum(h_vars) - overlap_sum <= c5_val * dx)
        
        # Objective (minimize C5)
        objective = cp.Minimize(c5_val)
        
        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_sol = h_vars.value
                h_clipped = np.clip(h_sol, 0, 1)
                h_normalized = h_clipped * (n_points / 2) / np.sum(h_clipped)
                return h_normalized
        except:
            pass
        return None
    
    def refine_solution(h_start, max_iter=500):
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_fun(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_fun}
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(objective, h_start, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': max_iter, 'ftol': 1e-8}, tol=1e-8)
        if result.success:
            h_refined = np.clip(result.x, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            return h_refined
        return h_start
    
    # Binary search loop
    while time() - start_time < budget_s * 0.9 and high - low > tolerance:
        mid = (low + high) / 2
        h_candidate = solve_sdp(mid)
        
        if h_candidate is not None:
            # Refine the candidate
            h_refined = refine_solution(h_candidate, 100)
            c5_candidate = evaluate_erdos_solution(h_refined, dx)
            
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_refined.copy()
            
            high = mid
        else:
            low = mid
            
        if time() - start_time > budget_s * 0.95:
            break
    
    # Final local optimization refinement
    if time() - start_time < budget_s * 0.98:
        final_h = refine_solution(best_h, 2000)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h
    
    return (list(best_h), best_c5, n_points)
