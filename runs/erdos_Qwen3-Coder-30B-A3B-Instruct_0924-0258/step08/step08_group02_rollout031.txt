import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter = int(1e5)
    
    # Perturbation step size
    epsilon = 0.1
    
    # SA main loop
    T = T_init
    h_current = h_start.copy()
    c5_current = best_c5
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Make a perturbation
        idx1, idx2 = random.sample(range(n_points), 2)
        delta = np.random.uniform(-epsilon, epsilon)
        
        # Adjust h[idx1] and h[idx2]
        h_new = h_current.copy()
        h_new[idx1] += delta
        h_new[idx2] -= delta
        
        # Keep values in [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain the sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Compute new overlap
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        delta_c5 = c5_new - c5_current
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h_current = h_new
            c5_current = c5_new
            
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
                
        # Cooling schedule
        T = max(T * alpha, T_min)
        
        # Adaptive epsilon
        if i % 1000 == 0 and i > 0:
            epsilon = max(epsilon * 0.95, 1e-4)
    
    # Local refinement with L-BFGS-B
    def objective(h):
        return np.max(np.correlate(h, 1 - h, mode='full') * dx)
    
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Ensure sum constraint in refinement
        h_refined = best_h.copy()
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        res = minimize(
            objective,
            h_refined,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            h_final = res.x
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
