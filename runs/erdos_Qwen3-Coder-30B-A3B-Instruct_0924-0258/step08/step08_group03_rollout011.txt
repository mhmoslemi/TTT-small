import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize parameters
    n_coarse = 15
    n_fine = 35
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # Initialize coarse grid
    h_coarse = np.ones(n_coarse) * 0.5
    
    # Initialize fine grid
    if 'initial_h_values' in globals():
        h_fine = np.array(initial_h_values)
    else:
        h_fine = np.ones(n_fine) * 0.5
    
    best_h_fine = h_fine.copy()
    best_c5 = evaluate_erdos_solution(h_fine, dx_fine)
    
    # Coarse optimization
    for iteration in range(500):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute overlaps
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_coarse - 1)
        
        # Compute gradient for the max lag
        grad = np.zeros(n_coarse)
        for i in range(n_coarse):
            j = i + max_lag
            if 0 <= j < n_coarse:
                grad[i] = 1 - 2 * h_coarse[i] - h_coarse[j]
            else:
                grad[i] = 1 - 2 * h_coarse[i]
        
        # Gradient descent step
        h_coarse_new = h_coarse - 0.01 * grad
        h_coarse_new = np.clip(h_coarse_new, 0, 1)
        
        # Renormalize to maintain constraint
        h_coarse_new = h_coarse_new * (n_coarse / 2) / np.sum(h_coarse_new)
        
        h_coarse = h_coarse_new
        
    # Upsample coarse to fine grid using linear interpolation
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    f = interp1d(x_coarse, h_coarse, kind='linear', fill_value='extrapolate')
    h_fine_upsampled = f(x_fine)
    
    # Renormalize
    h_fine_upsampled = h_fine_upsampled * (n_fine / 2) / np.sum(h_fine_upsampled)
    h_fine = h_fine_upsampled
    
    # Fine grid optimization
    for iteration in range(2000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Occasionally perform random mutation
        if iteration % 100 == 0:
            h_mutation = h_fine.copy()
            indices = np.random.choice(n_fine, size=int(0.1 * n_fine), replace=False)
            h_mutation[indices] += np.random.normal(0, 0.05, len(indices))
            h_mutation = np.clip(h_mutation, 0, 1)
            h_mutation = h_mutation * (n_fine / 2) / np.sum(h_mutation)
            c5_mutation = evaluate_erdos_solution(h_mutation, dx_fine)
            
            if c5_mutation < best_c5:
                best_c5 = c5_mutation
                best_h_fine = h_mutation.copy()
                h_fine = h_mutation.copy()
        else:
            # Compute overlaps
            corr = np.correlate(h_fine, 1 - h_fine, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (n_fine - 1)
            
            # Compute gradient for the max lag
            grad = np.zeros(n_fine)
            for i in range(n_fine):
                j = i + max_lag
                if 0 <= j < n_fine:
                    grad[i] = 1 - 2 * h_fine[i] - h_fine[j]
                else:
                    grad[i] = 1 - 2 * h_fine[i]
            
            # Gradient descent step
            h_fine_new = h_fine - 0.01 * grad
            h_fine_new = np.clip(h_fine_new, 0, 1)
            
            # Renormalize to maintain constraint
            h_fine_new = h_fine_new * (n_fine / 2) / np.sum(h_fine_new)
            
            c5_new = evaluate_erdos_solution(h_fine_new, dx_fine)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h_fine = h_fine_new.copy()
                h_fine = h_fine_new.copy()
    
    return (list(best_h_fine), best_c5, n_fine)
