import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach
    resolutions = [35, 70, 140, 280]
    current_n = n_points
    h = h_start.copy()
    h = h * (current_n/2) / np.sum(h)  # Ensure sum constraint
    
    # Simulated Annealing with cooling schedule
    def cooling_schedule(iteration, max_iter):
        return 1.0 / (1.0 + iteration / max_iter * 10)
    
    # Initialize temperature
    current_c5 = evaluate_erdos_solution(h, dx)
    temp = current_c5 * 0.1 if current_c5 > 0 else 1.0
    
    # Parameters for SA
    max_iter = 5000
    patience = 200
    no_improvement = 0
    max_temp_iter = 1000
    
    for resolution in resolutions:
        if resolution > current_n:
            # Increase resolution via interpolation
            h_new = np.interp(np.linspace(0, 1, resolution), 
                              np.linspace(0, 1, current_n), h)
            h = h_new
            current_n = resolution
            dx = 2.0 / current_n
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Get current evaluation
            current_c5 = evaluate_erdos_solution(h, dx)
            
            # Accept current solution if it's better or with probability
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                no_improvement = 0
            else:
                no_improvement += 1
                
            if no_improvement >= patience and iteration > max_temp_iter:
                break
                
            # Proposal: random perturbation
            h_proposal = h.copy()
            idx = np.random.randint(0, current_n)
            delta = np.random.normal(0, temp * 0.1)
            h_proposal[idx] += delta
            
            # Ensure constraints
            h_proposal = np.clip(h_proposal, 0, 1)
            h_proposal = h_proposal * (current_n/2) / np.sum(h_proposal)
            
            # Evaluate proposal
            proposal_c5 = evaluate_erdos_solution(h_proposal, dx)
            
            # Accept with probability based on difference and temperature
            if proposal_c5 < current_c5 or np.random.rand() < np.exp(-(proposal_c5 - current_c5) / (temp + 1e-8)):
                h = h_proposal
            
            # Update temperature
            temp = cooling_schedule(iteration, max_iter)
            
            if temp < 1e-8:
                break
    
    # Final optimization with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - current_n / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(current_n)]
        
        res = minimize(objective, h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, current_n)
