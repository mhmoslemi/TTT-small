import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex perturbation parameters
    epsilon_0 = 0.001
    beta = 0.99
    max_iter = 5000
    top_k = 100
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        top_shifts = np.argsort(corr)[-top_k:]
        
        # Formulate convex optimization problem
        h_var = cp.Variable(n_points)
        objective = 0
        
        # Sum of selected correlations plus perturbation term
        for k in top_shifts:
            if k < n_points:
                # Approximate the convolution term with a quadratic form
                # This is a simplified approach; in practice might use more precise terms
                objective += cp.sum_squares(h_var - 1)  # Simplified regularizer
                
        # Add convex perturbation term
        objective += epsilon_0 * (beta ** iteration) * cp.sum_squares(h_var)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Solve the convex problem
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_it=200)
        except:
            # If solver fails, fall back to projection
            h_new = h.copy()
            h_new = project_to_feasible(h_new, n_points, dx)
        else:
            h_new = h_var.value
            if h_new is not None:
                h_new = project_to_feasible(h_new, n_points, dx)
            else:
                h_new = h.copy()
                h_new = project_to_feasible(h_new, n_points, dx)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        elif iteration > 0 and iteration % 500 == 0:
            # Occasionally reduce perturbation level
            epsilon_0 *= 0.95
            
    # Final local refinement: random search around best solution
    for _ in range(100):
        if time() - start_time > budget_s * 0.98:
            break
        h_pert = best_h + np.random.normal(0, 0.001, n_points)
        h_pert = project_to_feasible(h_pert, n_points, dx)
        c5_pert = evaluate_erdos_solution(h_pert, dx)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
            
    return (list(best_h), best_c5, n_points)
