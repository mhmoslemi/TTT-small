import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    sum_h = np.sum(h_current)
    h_current = h_current * (n_points/2) / sum_h
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    max_iter = int(1e6)
    initial_T = 10.0
    cooling_rate = 0.9999
    min_T = 1e-6
    no_improvement_limit = 2000
    delta_range = 0.1
    
    T = initial_T
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor by swapping delta between two random indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(0, delta_range)
        
        # Ensure we don't exceed bounds
        delta = min(delta, h_current[i], 1 - h_current[j])
        
        # Create candidate solution
        h_candidate = h_current.copy()
        h_candidate[i] += delta
        h_candidate[j] -= delta
        
        # Adjust to maintain sum constraint
        sum_h = np.sum(h_candidate)
        adjustment = (n_points/2 - sum_h) / n_points
        h_candidate += adjustment
        
        # Clamp to bounds
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Evaluate new solution
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject based on SA criteria
        if c5_candidate < best_c5:
            h_current = h_candidate.copy()
            best_c5 = c5_candidate
            no_improvement_count = 0
        else:
            if T > min_T:
                accept_prob = math.exp(-(c5_candidate - best_c5) / T)
                if np.random.random() < accept_prob:
                    h_current = h_candidate.copy()
            no_improvement_count += 1
            
        # Update temperature
        T *= cooling_rate
        
        # Early stopping if no improvement for a while
        if no_improvement_count > no_improvement_limit:
            break
    
    # Final refinement using L-BFGS-B
    from scipy.optimize import minimize
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='L-BFGS-B', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
