import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Hybrid approach: convex relaxation followed by coordinate descent
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Set parameters for convex relaxation
    lambda_reg = 0.1
    alpha_reg = 0.01
    
    # Iterate for hybrid approach
    max_iterations = 100
    no_improvement_count = 0
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convex relaxation step
        try:
            # Create a relaxed problem using CVXPY
            h_relaxed = cp.Variable(n_points)
            objective = cp.Minimize(
                lambda_reg * cp.sum_squares(cp.conv matrix(h_relaxed, 1-h_relaxed)) +
                alpha_reg * cp.sum_squares(h_relaxed)
            )
            constraints = [
                cp.sum(h_relaxed) == n_points/2,
                h_relaxed >= 0,
                h_relaxed <= 1
            ]
            
            problem = cp.Problem(objective, constraints)
            problem.solve(solver=cp.ECOS, verbose=False)
            
            if problem.status == cp.OPTIMAL:
                h = h_relaxed.value
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
        except:
            # Fallback to simple heuristic if CVXPY fails
            pass
            
        # Coordinate descent step
        coords_to_optimize = np.random.choice(n_points, size=int(n_points*0.2), replace=False)
        
        for _ in range(50):  # Limited gradient steps
            if time() - start_time > budget_s * 0.95:
                break
                
            # Calculate overlaps for all lags
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            max_overlap = np.max(overlaps)
            l_max = np.argmax(overlaps)
            
            # Gradient computation for unstable coordinates
            gradient = np.zeros(n_points)
            for i in coords_to_optimize:
                j = (i + l_max) % n_points
                gradient[i] = -2 * dx * (1 - h[j] - h[i])
            
            # Update the selected coordinates
            step_size = 0.01
            for i in coords_to_optimize:
                h[i] -= step_size * gradient[i]
                h[i] = np.clip(h[i], 0, 1)
            
            # Renormalize to satisfy sum constraint
            h = h * (n_points/2) / np.sum(h)
            
            # Early stopping based on local improvement
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                if no_improvement_count > 50:
                    break
                    
        # Occasionally re-sample coordinates to escape local minima
        if iter % 5 == 0:
            coords_to_optimize = np.random.choice(n_points, size=int(n_points*0.2), replace=False)

    return (list(best_h), best_c5, n_points)
