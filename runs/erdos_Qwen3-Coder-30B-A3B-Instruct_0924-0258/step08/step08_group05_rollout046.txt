import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Use FFT-based approach for faster correlation computation
    def compute_overlap_fft(h):
        # Compute using FFT for efficiency
        H = np.fft.fft(h)
        G = 1 - h
        G_fft = np.fft.fft(G)
        # Cross-correlation in frequency domain (with proper normalization)
        corr_fft = H * np.conj(G_fft)
        corr = np.fft.ifft(corr_fft).real
        # Only take the part that corresponds to valid lags
        # The correlation for lags [0, 1, ..., 2*n_points-2] is returned
        # We want the middle part with lags [-n_points+1, ..., n_points-1]
        return corr
    
    def get_max_overlap_and_lag(h):
        corr = compute_overlap_fft(h)
        # The fft gives us correlation for lags from -n+1 to n-1
        # which corresponds to indices from 0 to 2n-2
        max_idx = np.argmax(corr)
        # Convert index to actual lag
        lag = max_idx - (n_points - 1)
        return corr[max_idx] * dx, lag
    
    # Optimizer parameters
    max_iter = 20000
    learning_rate = 0.01
    patience = 100
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Get maximum overlap and its lag
        max_overlap_val, max_lag = get_max_overlap_and_lag(h_current)
        
        # Compute gradient w.r.t h for this specific lag
        # The gradient is 1 for indices that contribute to the max overlap
        # but this is more complex in the FFT domain
        # Instead we compute the actual derivative for our step function technique
        gradient = np.zeros_like(h_current)
        h_shift = np.roll(h_current, -max_lag)
        if max_lag >= 0:
            # lags in correlation come from h[i] * (1 - h[i + max_lag])
            # So the gradient w.r.t h[i] term is: -1 if we are at i and h[i+max_lag] exists
            for i in range(len(h_current) - max_lag):
                gradient[i] = -1.0
        else:
            # Same thing but reversed
            for i in range(-max_lag, len(h_current)):
                gradient[i] = -1.0
                
        # Projection onto sum constraint
        sum_h = np.sum(h_current)
        projection = (n_points / 2 - sum_h) / len(h_current)
        
        # Update step
        h_new = h_current - learning_rate * gradient + learning_rate * projection
        
        # Clamp and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check for convergence
        change = np.sum(np.abs(h_new - h_current))
        if change < 1e-10:
            no_improvement_count += 1
            if no_improvement_count > patience:
                learning_rate *= 0.5
                no_improvement_count = 0
        else:
            no_improvement_count = 0
            
        h_current = h_new
        
        # Evaluate and update best
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
    
    # Final refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        # Random restarts at final stage
        best_final_h = best_h.copy()
        best_final_c5 = best_c5
        for r in range(10):  # Try 10 random restarts
            if time() - start_time > budget_s * 0.99:
                break
            h_rand = np.random.rand(n_points) * 0.2 + 0.4
            h_rand = h_rand * (n_points/2) / np.sum(h_rand)
            res = minimize(objective, h_rand, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success:
                c5_final = objective(res.x)
                if c5_final < best_final_c5:
                    best_final_c5 = c5_final
                    best_final_h = res.x
        if best_final_c5 < best_c5:
            best_h = best_final_h
            best_c5 = best_final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
