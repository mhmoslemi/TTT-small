import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex optimization with filtering approach
    max_iterations = 1000
    k_max = 20  # number of top overlaps to consider
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Step 1: Convex optimization relaxation
        # Focus on top k overlaps using a weighted sum
        corr = np.correlate(h, 1 - h, mode='full')
        top_k_indices = np.argsort(corr)[-k_max:][::-1]
        top_k_values = corr[top_k_indices]
        
        # Create convex optimization problem
        h_vars = cp.Variable(n_points)
        
        # Objective: minimize weighted sum of top overlaps
        objective = cp.Minimize(cp.sum(cp.multiply(top_k_values, 
                                                    [cp.norm(h_vars - np.roll(1-h, i), 1) 
                                                     for i in top_k_indices])))
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            h_new = np.array(h_vars.value).flatten()
        except:
            # Fallback to gradient descent if optimization fails
            grad = np.zeros(n_points)
            for i in range(n_points):
                for j in top_k_indices:
                    idx = (i + j - (n_points-1)) % n_points
                    if 0 <= idx < n_points:
                        grad[i] += (1 - 2*h[i] - h[idx] - h[i])
            
            step_size = 0.005
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Step 2: Apply low-pass filtering
        # Apply simple moving average filter
        kernel = np.ones(5) / 5
        h_filtered = np.convolve(h_new, kernel, mode='same')
        h_filtered = np.clip(h_filtered, 0, 1)
        
        # Renormalize
        h_filtered = h_filtered * (n_points / 2) / np.sum(h_filtered)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_filtered, dx)
        
        # Accept if better or if time is running out
        if c5_new < best_c5 or time() - start_time > budget_s * 0.9:
            best_c5 = c5_new
            best_h = h_filtered.copy()
            h = h_filtered.copy()
            
        # Update k for next iteration (dynamically)
        if k_max < 30:
            k_max = min(k_max + 1, 30)
    
    # Final refinement with local search if needed
    for _ in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Small random perturbation
        perturb = np.random.normal(0, 0.01, n_points)
        h_test = h + perturb
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        c5_test = evaluate_erdos_solution(h_test, dx)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()
            h = h_test.copy()
    
    return (list(best_h), best_c5, n_points)
