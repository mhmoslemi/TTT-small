import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # GA parameters
    population_size = 50
    generations = 0
    max_generations = 10000
    mutation_rate = 0.1
    crossover_rate = 0.8
    mutation_step = 0.05
    no_improvement_count = 0
    max_no_improvement = 200
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # scale to sum = n_points/2
        h = np.clip(h, 0, 1)
        return h
    
    if 'initial_h_values' in globals():
        population = [h_start.copy()]
        for _ in range(population_size - 1):
            population.append(create_individual())
    else:
        population = [create_individual() for _ in range(population_size)]
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    def select_parents(population, fitnesses):
        # Tournament selection
        selected = []
        for _ in range(len(population) // 2):
            tourn_size = 3
            tourn_indices = np.random.choice(len(population), tourn_size)
            tourn_fitnesses = [fitnesses[i] for i in tourn_indices]
            winner_idx = tourn_indices[np.argmin(tourn_fitnesses)]
            selected.append(population[winner_idx])
        return selected
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        alpha = np.random.uniform(0.3, 0.7)
        child1 = alpha * parent1 + (1 - alpha) * parent2
        child2 = (1 - alpha) * parent1 + alpha * parent2
        
        # Ensure feasibility
        child1 = np.clip(child1, 0, 1)
        child2 = np.clip(child2, 0, 1)
        
        # Fix sum constraint
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        
        return child1, child2
    
    def mutate(h):
        if np.random.rand() > mutation_rate:
            return h.copy()
        
        h_mut = h.copy()
        idx = np.random.randint(0, n_points)
        perturb = np.random.normal(0, mutation_step)
        h_mut[idx] += perturb
        
        # Clip and renormalize
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
        
        return h_mut
    
    def ga_step():
        nonlocal mutation_step, no_improvement_count
        
        fitnesses = [fitness(ind) for ind in population]
        
        # Select parents
        parents = select_parents(population, fitnesses)
        
        # Generate offspring
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                child1, child2 = crossover(parents[i], parents[i+1])
                offspring.extend([mutate(child1), mutate(child2)])
            else:
                offspring.append(mutate(parents[i]))
        
        # Combine with old population
        all_individuals = population + offspring
        all_fitnesses = fitnesses + [fitness(ind) for ind in offspring]
        
        # Keep best individuals
        sorted_indices = np.argsort(all_fitnesses)
        population[:] = [all_individuals[i] for i in sorted_indices[:population_size]]
        
        # Track best solution
        current_best_idx = np.argmin([fitness(ind) for ind in population])
        current_best = population[current_best_idx]
        current_c5 = fitness(current_best)
        
        if current_c5 < best_c5:
            best_h[:] = current_best
            best_c5 = current_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Update mutation step
        mutation_step *= 0.98
        
        return best_h, best_c5
    
    # Main GA loop
    while time() - start_time < budget_s * 0.85 and generations < max_generations:
        ga_step()
        generations += 1
        
        # Periodic local refinement
        if generations % 100 == 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 500, 'ftol': 1e-12})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h[:] = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
            
            # Check for timeout
            if time() - start_time > budget_s * 0.95:
                break
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 1000, 'ftol': 1e-14})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h[:] = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
