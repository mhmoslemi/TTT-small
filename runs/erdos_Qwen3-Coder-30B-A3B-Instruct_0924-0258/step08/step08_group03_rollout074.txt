import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex perturbation optimization
    max_iter = 5000
    epsilon_0 = 0.001
    beta = 0.99
    top_k = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Get the dominant overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        top_k_indices = np.argsort(corr)[-top_k:]
        
        # Convex optimization with perturbation
        h_var = cp.Variable(n_points)
        objective = 0
        
        # Form convex objective using top_k
        for k_idx in top_k_indices:
            if k_idx < n_points:
                # h * (1-h_k) for shift k
                shift = k_idx - (n_points - 1)
                term = 0
                for i in range(n_points):
                    j = i + shift
                    if 0 <= j < n_points:
                        term += h_var[i] * (1 - h_var[j])
                    else:
                        term += h_var[i]
                objective += term
                
        # Add regularizer
        epsilon = epsilon_0 * (beta ** iteration)
        objective += epsilon * cp.sum_squares(h_var)
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # Solve the convex problem
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.ECOS, abstol=1e-6, reltol=1e-6, feastol=1e-6)
            h_new = np.array(h_var.value).reshape(-1)
        except:
            h_new = h.copy()
            if iteration > 0 and iteration % 100 == 0:
                epsilon_0 *= 0.5
        
        # Projection step to maintain constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            
        # Early stopping condition
        if iteration > 0 and iteration % 500 == 0 and c5_new >= best_c5:
            epsilon_0 *= 0.5
            
    # Final refinement with random search
    best_h_final = best_h.copy()
    best_c5_final = best_c5
    for _ in range(100):
        if time() - start_time > budget_s * 0.99:
            break
        h_pert = best_h + np.random.normal(0, 0.001, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        c5_pert = evaluate_erdos_solution(h_pert, dx)
        if c5_pert < best_c5_final:
            best_c5_final = c5_pert
            best_h_final = h_pert.copy()
            
    return (list(best_h_final), best_c5_final, n_points)
