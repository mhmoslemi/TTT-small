import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # FFT-based optimization
    max_iter = 10000
    step_size = 0.01
    tolerance = 1e-9
    no_improvement_limit = 100
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Fast computation of all overlaps using FFT
        H = np.fft.fft(h_current)
        G = 1.0 - h_current
        G_fft = np.fft.fft(G)
        # Convolution in frequency domain: correlation
        corr_fft = H * np.conj(G_fft)
        overlap_values = np.fft.ifft(corr_fft).real
        # Adjust indices for proper lag alignment
        overlap_values = np.concatenate([overlap_values[-(n_points-1):], overlap_values[:n_points]])
        
        # Find the lag with maximum overlap
        max_lag_idx = np.argmax(overlap_values)
        max_lag = max_lag_idx - (n_points - 1) if max_lag_idx >= n_points else max_lag_idx - (n_points - 1) 
        max_overlap = overlap_values[max_lag_idx]
        
        # Compute gradient of overlap w.r.t. h for specific lag using FFT-based approach
        # We can compute this as the convolution of shifted h with 1-h (negative)
        # Gradient = 1-h  (for positive term in h[i+k] * h[i]), but inverse shift for lag
        # For lag k, gradient = [-1,1] to accommodate kernel waveforms
        # Let's think in terms of direct finite difference
        gradient = np.zeros_like(h_current)
        
        # Compute the gradient classically for this lag
        for i in range(len(h_current)):
            if i + max_lag < len(h_current) and max_lag >= 0:
                # h[i] and h[i+lag] contribute to overlap
                gradient[i] = -1
            elif i - max_lag >= 0 and max_lag < 0:
                gradient[i] = -1
                
        # Apply gradient step, careful to maintain constraint
        h_new = h_current - step_size * gradient
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Apply update
        h_current = h_new
        
        # Evaluate new solution
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > no_improvement_limit:
                step_size *= 0.5
                no_improvement_count = 0
            
    # Final refinement using SLSQP if after all iterations 
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
