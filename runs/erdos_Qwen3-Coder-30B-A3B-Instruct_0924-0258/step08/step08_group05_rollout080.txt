import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing approach
    current_energy = best_c5
    temperature = max(1e-3, 0.1 * current_energy)
    cooling_rate = 0.9995
    min_temperature = 1e-6
    max_no_improvement = 500
    no_improvement_count = 0
    max_iter = int(budget_s * 5000)  # Rough estimate
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor solution by swapping values
        i, j = np.random.randint(0, n_points, 2)
        if i == j:
            continue
            
        # Compute feasible delta
        max_delta = min(h_current[i], 1 - h_current[j])
        if max_delta <= 1e-10:
            continue
            
        delta = np.random.uniform(0, max_delta)
        
        # Create new solution
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
            
        # Enforce sum constraint
        sum_diff = np.sum(h_new) - n_points / 2
        if abs(sum_diff) > 1e-10:
            # Adjust one element to make sum correct
            idx = np.random.randint(0, n_points)
            h_new[idx] -= sum_diff
            if h_new[idx] < 0 or h_new[idx] > 1:
                continue
                
        # Calculate new energy
        new_energy = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on simulated annealing
        if new_energy < current_energy:
            h_current = h_new
            current_energy = new_energy
            if new_energy < best_c5:
                best_c5 = new_energy
                best_h = h_new.copy()
            no_improvement_count = 0
        else:
            # Probabilistically accept worse solutions
            if np.random.rand() < np.exp(-(new_energy - current_energy) / temperature):
                h_current = h_new
                current_energy = new_energy
            no_improvement_count += 1
            
        # Cooling schedule
        temperature *= cooling_rate
        if temperature < min_temperature:
            temperature = min_temperature
            
        # Early stopping if no improvement for many iterations
        if no_improvement_count > max_no_improvement:
            break
    
    # Final check to ensure sum constraint
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        scale = (n_points / 2) / sum_h
        best_h *= scale
        
    return (list(best_h), best_c5, n_points)
