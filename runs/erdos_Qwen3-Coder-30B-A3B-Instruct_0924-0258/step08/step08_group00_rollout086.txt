import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Modified optimization with improved heuristics
    max_iter = 5000
    patience = 100
    no_improvement = 0
    step_size = 0.01
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Warm-up phase with larger step sizes
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate overlap for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            break
            
        # Identify lag with maximum overlap
        max_lag = max_lag_idx - (n_points - 1)
        
        # Improved gradient computation with more accurate finite differences
        grad = np.zeros(n_points)
        for i in range(n_points):
            j1 = i + max_lag
            j2 = i - max_lag
            if 0 <= j1 < n_points:
                grad[i] += (1 - h[j1]) / n_points
            if 0 <= j2 < n_points:
                grad[i] += (1 - h[j2]) / n_points
                
        # Scale down gradient magnitude
        grad = grad * 0.5
        
        # Update with gradient descent
        h_new = h - step_size * grad
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check if the update improved the objective
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < max_overlap:
            h = h_new.copy()
            step_size = min(0.05, step_size * 1.1)  # Increase step size for rapid improvement
        else:
            step_size = max(1e-6, step_size * 0.5)  # Decrease step size for stability
    
    # Final refinement with trust-constr
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h, method='trust-constr', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'xtol': 1e-9, 'gtol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Ensure final constraints are met
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
