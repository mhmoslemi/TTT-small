import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    max_iter = int(1e6)
    initial_temp = 1.0
    cooling_rate = 0.9995
    min_temp = 1e-6
    no_improvement_limit = 500
    
    temp = initial_temp
    no_improvement_count = 0
    iteration = 0
    
    # Use the more efficient neighbor generation
    while iteration < max_iter and time() - start_time < budget_s * 0.95:
        # Generate neighbor by swapping values between two indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(0, 0.1)
        
        # Make sure we don't violate bounds
        delta = min(delta, h_current[i], 1 - h_current[j])
        
        if delta <= 0:
            iteration += 1
            continue
            
        # Create new candidate solution
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds are respected (should be true from above)
        h_new[i] = np.clip(h_new[i], 0, 1)
        h_new[j] = np.clip(h_new[j], 0, 1)
        
        # Re-normalize to satisfy sum constraint
        sum_h = np.sum(h_new)
        h_new = h_new * (n_points/2) / sum_h
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            if np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
                h_current = h_new.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
        # Cool down temperature
        temp = max(temp * cooling_rate, min_temp)
        
        # Reset temperature if no improvement for a while
        if no_improvement_count > no_improvement_limit:
            temp = initial_temp
            no_improvement_count = 0
            
        iteration += 1
            
    # Final refinement with trust-constrained optimization
    try:
        from scipy.optimize import minimize
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
