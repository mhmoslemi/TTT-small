import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Binary threshold optimization
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Binary search for better solutions
    for threshold in np.arange(1.0, 0.0, -0.1):
        if time() - start_time > budget_s * 0.6:
            break
            
        # Create binary approximation
        h_binary = np.zeros(n_points)
        # Sort values and use threshold to convert to binary
        sorted_vals = np.sort(h)[::-1]
        threshold_val = sorted_vals[int(n_points/2)]
        h_binary = (h >= threshold_val).astype(float)
        
        # Ensure sum constraint
        current_sum = np.sum(h_binary)
        if current_sum > n_points/2:
            # Need to set some 1s to 0s
            indices = np.argsort(h_binary)[::-1]
            for i in range(int(current_sum - n_points/2)):
                h_binary[indices[i]] = 0
        elif current_sum < n_points/2:
            # Need to set some 0s to 1s
            indices = np.argsort(h_binary)
            for i in range(int(n_points/2 - current_sum)):
                h_binary[indices[i]] = 1
        
        # Refine with optimization
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
                h_binary,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 10, 'ftol': 1e-7}
            )
            if res.success:
                h_refined = res.x
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined.copy()
        except Exception:
            pass
            
        # Also try greedy approach
        h_greedy = h_binary.copy()
        # Small local optimization for greedy solution
        for i in range(n_points):
            if time() - start_time > budget_s * 0.95:
                break
            h_test = h_greedy.copy()
            h_test[i] = 1 - h_test[i]
            c5_test = evaluate_erdos_solution(h_test, dx)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
    
    # Phase 2: Local gradient refinement
    if time() - start_time > budget_s * 0.95:
        return (list(best_h), best_c5, n_points)
    
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass

    # Ensure we maintain the sum constraint
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
