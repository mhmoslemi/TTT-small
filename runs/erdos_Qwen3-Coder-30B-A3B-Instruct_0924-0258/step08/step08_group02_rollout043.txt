import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convex relaxation approach
    n_iter = 0
    no_improve_count = 0
    max_no_improve = 50
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    while time() - start_time < budget_s * 0.95 and no_improve_count < max_no_improve:
        n_iter += 1
        # Convex relaxation step
        try:
            # Formulate and solve the relaxed problem
            h_relaxed = cp.Variable(n_points)
            constraints = [cp.sum(h_relaxed) == n_points/2, h_relaxed >= 0, h_relaxed <= 1]
            
            # Approximate the overlap using a convex relaxation
            # Minimize weighted sum of squared overlaps (non-convex, so we'll try a simple proxy)
            obj = cp.Minimize(cp.sum_squares(h_relaxed))
            prob = cp.Problem(obj, constraints)
            
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_relaxed.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                h = h_new.copy()
                
        except:
            pass
        
        # Coordinate descent refinement
        # Select 10% of coordinates to optimize
        indices_to_optimize = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        temp_h = h.copy()
        
        # Gradient descent for the selected coordinates
        for _ in range(50):  # 50 iterations of gradient descent
            if time() - start_time > budget_s * 0.95:
                break
                
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
                
            # Compute gradients for selected variables
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            l_max = np.argmax(overlaps)
            
            # Gradient for the max overlap lag
            gradient = np.zeros(n_points)
            for i in indices_to_optimize:
                j = (i + l_max) % n_points
                gradient[i] = -2 * dx * (1 - h[j] - h[i])
            
            # Update selected variables
            learning_rate = 0.01
            for i in indices_to_optimize:
                h[i] -= learning_rate * gradient[i]
                
            # Project back to [0, 1] and enforce sum constraint
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            
            # Early termination if improvement stalls
            if no_improve_count >= max_no_improve:
                break
                
        # Small random updates to escape local minima
        if np.random.rand() < 0.1:
            for i in range(n_points):
                if np.random.rand() < 0.05:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.01), 0, 1)
            h = h * (n_points/2) / np.sum(h)
            
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
