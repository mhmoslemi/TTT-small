import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    num_generations = 5000
    crossover_rate = 0.8
    mutation_rate = 0.1
    mutation_step = 0.05
    elite_size = population_size // 2
    
    # Initialize population
    def create_individual():
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    if 'initial_h_values' in globals():
        population = [h_start.copy()]
        for _ in range(population_size - 1):
            population.append(create_individual())
    else:
        population = [create_individual() for _ in range(population_size)]
    
    no_improvement_count = 0
    generation = 0
    
    while time() - start_time < budget_s * 0.95 and no_improvement_count < 200:
        # Evaluate fitness for all individuals
        fitness = []
        for individual in population:
            fit = evaluate_erdos_solution(individual, dx)
            fitness.append(fit)
        
        # Sort individuals by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update best if found new minimum
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_h = sorted_population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # selection (tournament)
        new_population = []
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = sorted_indices[random.randint(0, population_size//2)]
            parent2_idx = sorted_indices[random.randint(0, population_size//2)]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if random.random() < crossover_rate:
                alpha = random.uniform(0.3, 0.7)
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
            
            # Mutation
            if random.random() < mutation_rate:
                for i in range(n_points):
                    if random.random() < mutation_rate:
                        child[i] += random.gauss(0, mutation_step)
                
                # Normalize to maintain sum constraint
                child = child * (n_points / 2) / np.sum(child)
                child = np.clip(child, 0, 1)
            
            new_population.append(child)
        
        population = new_population
        mutation_step *= 0.98
        generation += 1
        
        if generation % 100 == 0:
            # Local refinement with SLSQP
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 100, 'ftol': 1e-12})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    return (list(best_h), best_c5, n_points)
