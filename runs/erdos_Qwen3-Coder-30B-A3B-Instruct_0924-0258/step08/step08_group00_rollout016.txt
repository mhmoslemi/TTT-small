import numpy as np
from scipy.optimize import minimize
from time import time
import random

def project(h, n_points):
    """Project h onto the feasible set: [0,1]^n and sum = n/2"""
    h_clipped = np.clip(h, 0, 1)
    if np.sum(h_clipped) == 0:
        return h_clipped
    h_scaled = h_clipped * (n_points / 2.0) / np.sum(h_clipped)
    return h_scaled

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def compute_gradient(h, dx, n_points):
    """Compute gradient for the current maximal overlap"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_lag = max_idx - (n_points - 1)
    
    # Create shifted version of (1-h) for this lag
    shifted = np.roll(1 - h, max_lag)
    
    # Gradient: partial derivative w.r.t. each component
    # For the maximizer lag, each term contributes as per d/dh_i [h_i * (1-h_{i+lag})] = 1 - 2*h_i - h_{i+lag} 
    # But using convolution properties, this turns out to be 2*(1-2*h)*shifted
    gradient = 2 * (1 - 2*h) * shifted
    return gradient

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = h_start.copy()
    h = project(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    max_iter = 1000
    patience = 50
    no_improvement = 0
    step_size = 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            break
            
        # Compute gradient
        try:
            grad = compute_gradient(h, dx, n_points)
        except:
            # Fallback to numerical gradient if needed
            h_plus = h + 1e-6
            h_plus = project(h_plus, n_points)
            c_plus = evaluate_erdos_solution(h_plus, dx)
            grad = (c_plus - max_overlap) / 1e-6
            
        # Update with gradient descent
        h_new = h - step_size * grad
        
        # Project back to feasible set
        h_new = project(h_new, n_points)
        
        # Simple line search to ensure improvement (cheap approximation)
        c_new = evaluate_erdos_solution(h_new, dx)
        if c_new < max_overlap:
            h = h_new
        else:
            # Reduce step size
            step_size *= 0.5
            if step_size < 1e-8:
                break

    # Final SLSQP refinement if there's time and improvement is possible
    if time() - start_time < budget_s * 0.9:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Use best solution as starting point
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass

    return (list(best_h), best_c5, n_points)
