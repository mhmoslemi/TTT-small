import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    temperature = 10.0
    cooling_rate = 0.995
    min_temperature = 1e-3
    max_iter = int(1e6)
    no_improvement_limit = 500
    no_improvement_count = 0
    
    # Normalize initial to exact constraint
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Make a move: swap small delta between two points
        i, j = np.random.randint(0, n_points, 2)
        while i == j:
            j = np.random.randint(0, n_points)
            
        delta = np.random.uniform(0.001, 0.1)
        # Ensure we don't violate bounds
        max_delta = min(h_current[i], 1 - h_current[j])
        delta = min(delta, max_delta)
        
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        # Re-normalize to constraint sum = n_points/2
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        # Accept or reject
        if c5_new < c5_current:
            h_current = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                no_improvement_count = 0
        else:
            # Probabilistically accept worse solution
            # Metropolis criterion
            if np.random.rand() < np.exp(-(c5_new - c5_current) / temperature):
                h_current = h_new
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
        # Cool down
        temperature *= cooling_rate
        
        # Early stopping criteria
        if temperature < min_temperature:
            break
        if no_improvement_count > no_improvement_limit:
            break
            
    # Final optimization with SLSQP if we have time
    if time() - start_time < budget_s * 0.9:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
