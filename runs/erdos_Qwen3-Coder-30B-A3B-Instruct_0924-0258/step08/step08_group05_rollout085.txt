import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a piecewise constant initial guess
        h_start = np.ones(n_points)
        h_start[:n_points//2] = 0.5
        h_start[n_points//2:] = 1.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Trust-region optimization with cutting planes
    max_iter = 5000
    trust_radius = 0.2
    min_trust_radius = 1e-6
    trust_radius_factor = 1.2
    
    def get_overlaps(h):
        corr = np.correlate(h, 1 - h, mode='full')
        lags = np.arange(len(corr)) - (len(h) - 1)
        return corr * dx, lags
    
    # Initial constraints
    overlaps, lags = get_overlaps(h_current)
    max_idx = np.argmax(overlaps)
    max_lag = lags[max_idx]
    
    # Create cutting-plane constraints iteratively
    cutting_planes = []
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Solve convex subproblem with trust region and cutting planes
        try:
            # Use cvxpy for convex optimization
            h_vars = cp.Variable(n_points)
            C = cp.Variable()
            
            # Objective
            objectives = []
            
            # Get current overlaps and their indices
            curr_overlaps, curr_lags = get_overlaps(h_current)
            max_idx = np.argmax(curr_overlaps)
            # Add constraint that max overlap is less than C
            objectives.append(C)
            
            # Build constraints: for each lag k, sum over valid indices
            constraints = [
                cp.sum(h_vars) == n_points / 2,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            for i, lag in enumerate(curr_lags):
                if lag == 0:
                    continue  # skip zero lag which isn't relevant here
                # Create a constraint for this lag
                valid_range = range(max(0, -lag), min(n_points, n_points - lag))
                if len(valid_range) > 0:
                    # sum(h[i] * (1 - h[i + lag])) <= C
                    overlap_sum = cp.sum(
                        cp.multiply(h_vars[valid_range], 
                                  1 - h_vars[valid_range[0] + lag : valid_range[-1] + lag + 1])
                    )
                    constraints.append(overlap_sum <= C)
            
            prob = cp.Problem(cp.Minimize(C), constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new_values = h_vars.value
                h_new = np.clip(h_new_values, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                
                c5_current = evaluate_erdos_solution(h_new, dx)
                if c5_current < best_c5:
                    best_c5 = c5_current
                    best_h = h_new.copy()
            
                # Update trust radius based on improvement
                if c5_current < best_c5 * 0.9999:  # significant improvement
                    trust_radius = min(trust_radius * trust_radius_factor, 1.0)
                elif c5_current > best_c5 * 1.001:
                    trust_radius = max(trust_radius / trust_radius_factor, min_trust_radius)
            
        except Exception:
            # Fallback to previous solution
            pass

        # Refinement with local optimization for final precision
        if iteration % 200 == 0 and time() - start_time < budget_s * 0.9:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 500, 'ftol': 1e-10})
                if res.success:
                    h_final = res.x
                    c5_final = objective(h_final)
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
            except Exception:
                pass
    
    # Final optimization validation
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
