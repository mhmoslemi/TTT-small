import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for GA
    population_size = 50
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 2
    max_no_improvement = 100
    no_improvement_count = 0
    generation = 0
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5 + 0.25
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5 + 0.25
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    def tournament_selection(pop, k=3):
        selected = []
        for _ in range(len(pop)):
            candidates = np.random.choice(len(pop), k)
            best = min(candidates, key=lambda i: evaluate_erdos_solution(pop[i], dx))
            selected.append(pop[best].copy())
        return selected
    
    def blend_crossover(parent1, parent2):
        child1, child2 = parent1.copy(), parent2.copy()
        if np.random.rand() < crossover_rate:
            alpha = np.random.rand()
            for i in range(n_points):
                child1[i] = alpha * parent1[i] + (1 - alpha) * parent2[i]
                child2[i] = (1 - alpha) * parent1[i] + alpha * parent2[i]
        return child1, child2
    
    def mutate(individual, mutation_rate):
        mutated = individual.copy()
        step_size = 0.1 * (1.0 - mutation_rate)
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                # Local perturbation within bounds
                delta = np.random.normal(0, step_size)
                mutated[i] += delta
        # Enforce constraints
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    eval_count = 0
    while time() - start_time < budget_s * 0.7:
        # Evaluate all individuals
        fitness = []
        for ind in population:
            fit = evaluate_erdos_solution(ind, dx)
            fitness.append(fit)
            if fit < best_c5:
                best_c5 = fit
                best_h = ind.copy()
                no_improvement_count = 0
            eval_count += 1
        
        # Rank individuals
        ranked_indices = np.argsort(fitness)
        new_population = [population[i] for i in ranked_indices[:elite_size]]
        
        # Tournament selection
        selected = tournament_selection(population)
        
        # Crossover and Mutation
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % len(selected)]
            child1, child2 = blend_crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:population_size]
        generation += 1
        
        if no_improvement_count >= max_no_improvement:
            break
            
        # Periodic SLSQP refinement
        if generation % 5 == 0 and generation > 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
                
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 100, 'ftol': 1e-10})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final refinement with all best solutions
    def objective(h):
        return evaluate_erdos_solution(h, dx)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try several random restarts
    tries = 0
    max_tries = 10
    while time() - start_time < budget_s and tries < max_tries:
        try:
            x0 = np.random.rand(n_points) * 0.5 + 0.25
            x0 = x0 * (n_points / 2) / np.sum(x0)
            
            res = minimize(objective, x0, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-12})
            if res.success:
                h_check = res.x
                c5_check = objective(h_check)
                if c5_check < best_c5:
                    best_c5 = c5_check
                    best_h = h_check
        except Exception:
            pass
        tries += 1

    return (list(best_h), best_c5, n_points)
