import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Start with initial solution
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    
    # Parameters
    max_iterations = 1000
    no_improve_limit = 50
    no_improve_count = 0
    improvement_threshold = 1e-6
    
    # Convex relaxation solver
    def solve_convex_relaxation(h_current, lambda_reg=0.1, alpha_reg=0.01):
        h_relaxed = cp.Variable(n_points)
        objective = cp.Minimize(
            lambda_reg * cp.sum_squares(cp.partial_sum(h_relaxed, 1-h_relaxed, mode='full')) + 
            alpha_reg * cp.sum_squares(h_relaxed)
        )
        constraints = [
            cp.sum(h_relaxed) == n_points/2,
            h_relaxed >= 0,
            h_relaxed <= 1
        ]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return np.array(h_relaxed.value)
        except:
            return h_current.copy()
    
    # Coordinate descent step
    def coordinate_descent_step(h_current, fraction_to_optimize=0.2, max_iter=50):
        h_new = h_current.copy()
        indices = np.random.permutation(n_points)
        optimize_indices = indices[:int(n_points * fraction_to_optimize)]
        
        def obj_and_grad(indices_to_optimize):
            h_temp = h_new.copy()
            h_temp[indices_to_optimize] = h_temp[indices_to_optimize]  # placeholder
            corr = np.correlate(h_temp, 1-h_temp, mode='full') * dx
            max_val = np.max(corr)
            return max_val, np.zeros_like(h_temp)
        
        # For each coordinate, compute approximate gradient
        for idx in optimize_indices:
            # Simple gradient estimation by perturbing
            temp_h = h_new.copy()
            steps = np.linspace(-0.05, 0.05, 11)
            values = []
            for step in steps:
                temp_h[idx] = max(0, min(1, h_new[idx] + step))
                corr = np.correlate(temp_h, 1-temp_h, mode='full') * dx
                values.append(np.max(corr))
            
            # Compute gradient
            grad = np.gradient(values, steps)
            best_step = steps[np.argmin(values)]
            h_new[idx] = max(0, min(1, h_new[idx] - 0.1 * best_step))
        
        # Project to feasible set
        h_new = h_new * (n_points/2) / np.sum(h_new)
        return h_new
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Try convex relaxation
        if iteration % 3 == 0:
            h_relaxed = solve_convex_relaxation(h)
            h = h_relaxed.copy()
        
        # Coordinate descent refinement
        h_refined = coordinate_descent_step(h, 0.2, 50)
        h = h_refined.copy()
        
        # Check constraint
        h = h * (n_points/2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count >= no_improve_limit:
            break
    
    # Final refinement using scipy
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        def objective(h_vec):
            return np.max(np.correlate(h_vec, 1-h_vec, mode='full') * dx)
        
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 50})
        if res.success:
            final_h = res.x
            final_h = final_h * (n_points/2) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
