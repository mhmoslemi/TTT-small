import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def repair(h, n_points):
    h = np.clip(h, 0, 1)
    if np.sum(h) == 0:
        return h
    h = h * (n_points/2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-start with Latin hypercube sampling
    n_initials = 50
    initial_guesses = []
    
    # Create Latin hypercube samples
    for _ in range(n_initials):
        sample = np.random.random(n_points)
        # Sort and spread out
        sorted_sample = np.sort(sample)
        # Distribute evenly
        for i in range(n_points):
            sample[i] = (i + sorted_sample[i]) / n_points
        # Normalize to sum to n_points/2
        sample = sample * (n_points/2) / np.sum(sample)
        # Clip to [0,1]
        sample = np.clip(sample, 0, 1)
        initial_guesses.append(sample)
    
    # Add the initial construction if available
    if 'initial_h_values' in globals():
        initial_guesses[0] = h_start.copy()
    
    # Run optimization for each initial guess
    for i, h0 in enumerate(initial_guesses):
        if time() - start_time > budget_s * 0.95:
            break
            
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h0, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                h_final = repair(h_final, n_points)
                c5_final = objective(h_final)
                
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
        except Exception:
            continue
    
    # Final refinement on the best solution
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            h_final = repair(h_final, n_points)
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
