import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    if np.sum(h_clipped) == 0:
        return h_clipped
    h_scaled = h_clipped * (n_points / 2.0) / np.sum(h_clipped)
    return h_scaled

def compute_gradient(h, n_points, dx):
    overlap = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(overlap)
    max_lag = max_idx - (n_points - 1)
    shifted = np.roll(1 - h, max_lag)
    gradient = 2 * (1 - 2 * h) * shifted
    return gradient

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Apply projection to satisfy constraints
    dx = 2.0 / n_points
    h = project(h_start, n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Alternating optimization parameters
    max_iter = 2000
    step_size = 0.1
    patience = 50
    no_improvement = 0
    min_step_size = 1e-8
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlap
        overlap = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(overlap)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient (just for checking)
        grad = compute_gradient(h, n_points, dx)
        
        # Update with gradient descent
        h_new = h - step_size * grad
        
        # Project to feasible set
        h_new = project(h_new, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Update current solution
        h = h_new.copy()
        
        # Adaptive step size
        if no_improvement >= patience:
            step_size *= 0.5
            no_improvement = 0
            if step_size < min_step_size:
                break
    
    # Final refinement with scipy optimize if time allows
    if time() - start_time < budget_s * 0.9:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 300, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
            
    return (list(best_h), best_c5, n_points)
