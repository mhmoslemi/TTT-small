import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    """Project h to satisfy integral constraint and bound constraints"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Use initial construction if available

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex perturbation parameters
    epsilon_0 = 0.001
    beta = 0.99
    max_iter = 5000
    top_k = 100
    
    # Optimization parameters
    solver = 'ECOS'
    
    # Convex perturbation optimization
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        dominant_shifts = np.argsort(corr)[-top_k:]
        
        # Create convex optimization problem
        h_var = cp.Variable(n_points)
        
        # Objective: minimize weighted sum of overlaps + regularization
        objective = 0
        epsilon = epsilon_0 * (beta ** iteration)
        
        for k in dominant_shifts:
            # This is a simplification - we'll directly optimize the correlation
            pass
        
        # Actual optimization: create a convex relaxation using CVXPY
        # Using the fact that we want to minimize max overlap  
        # So we minimize a convex regularization term plus overlap approximation
        
        # Build the optimization problem using convex relaxation
        h_constrained = h_var
        # Add constraint for integral
        constraints = [h_constrained >= 0, h_constrained <= 1, cp.sum(h_constrained) == n_points / 2]
        
        # Construct the overlap term - just one principal term for now
        # This is a proxy for minimizing the max correlation term
        objective_expr = 0
        overlap_sum = 0
        
        # Construct a good approximation to min max over all shifts
        # Iterate through the dominant overlaps we want to minimize
        for k in dominant_shifts:
            # Compute shifted version of h for correlation with (1-h) at shift k
            if k < 0:
                # Positive shift - so we compare h[i] to (1-h)[i-k]
                # This means shorter vector in the right side
                shifted_h = np.concatenate([h[-k:], np.zeros(-k)])
            else:
                # Negative shift - h[i+k] with (1-h)[i] 
                shifted_h = np.concatenate([np.zeros(k), h[:-k]])
                
            # Extract only the part that we care about for this shift
            # Then compute correlation with 1-h (total length: 2*n_points - 1)
            if k >= 0:
                corr_slice = 1 - h[:n_points-k]
                h_slice = h[k:]
            else:
                corr_slice = 1 - h[-k:]
                h_slice = h[:-k]
                
            # Collapse to single value  
            overlap_value = np.sum(h_slice * corr_slice)
            overlap_sum += overlap_value
            
        # Use soft minimization with regularization
        objective = cp.Minimize(overlap_sum + epsilon * cp.sum_squares(h_var))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
            h_new = h_var.value
            if h_new is not None:

                h_new = project_to_feasible(h_new, n_points, dx)

            h_new = h.copy()

            corr = np.correlate(h, 1 - h, mode='full')

            for i in range(n_points):

                if 0 <= j < n_points:
                    grad[i] = 1 - 2 * h[i] - h[j]
                else:
                    grad[i] = 1 - 2 * h[i]
            
            h_new = h - step_size * grad
            h_new = project_to_feasible(h_new, n_points, dx)

        # Early stopping on stagnation

    # Final local refinement with random search
    if time() - start_time < budget_s * 0.9:
        best_h_refined = best_h.copy()

            # Generate random perturbations
            perturbation = np.random.normal(0, 0.001, n_points)
            h_trial = best_h + perturbation
            h_trial = project_to_feasible(h_trial, n_points, dx)
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            
            if c5_trial < best_c5_refined:
                best_c5_refined = c5_trial
                best_h_refined = h_trial.copy()
                
        best_c5 = best_c5_refined
        best_h = best_h_refined
    
    return (list(best_h), best_c5, n_points)
