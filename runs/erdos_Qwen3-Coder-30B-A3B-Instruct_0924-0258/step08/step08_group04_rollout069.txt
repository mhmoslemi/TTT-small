import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # SDP approach
    h = cp.Variable(n_points, bounds=(0, 1))
    C5 = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h) == n_points/2]
    
    # For each possible lag k
    for k in range(-n_points+1, n_points):
        if k >= 0:
            # h[i] * (1 - h[i + k]) for valid indices
            expr = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
        else:
            # h[i] * (1 - h[i + k]) for valid indices
            expr = cp.sum(cp.multiply(h[-k:], 1 - h[:-k]))
        constraints.append(expr <= C5 * dx)
        
    # Objective: minimize C5
    prob = cp.Problem(cp.Minimize(C5), constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.MOSEK, time_limit=budget_s * 0.9)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_solution = h.value
        else:
            h_solution = h_start
    except:
        h_solution = h_start
    
    # Ensure feasibility
    h_solution = np.clip(h_solution, 0, 1)
    h_solution = h_solution * (n_points/2) / np.sum(h_solution)
    
    best_c5 = evaluate_erdos_solution(h_solution, dx)
    
    return (list(h_solution), best_c5, n_points)
