import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) * dx = 1"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex perturbation parameters
    epsilon_0 = 0.001
    beta = 0.99
    max_iter = 5000
    top_k = 100
    
    # Convex optimization setup
    h_var = cp.Variable(n_points)
    objective_expr = 0
    
    # Starting with a small perturbation
    epsilon = epsilon_0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Compute overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        top_shifts = np.argsort(corr)[-top_k:]
        
        # Form convexified objective
        # Empirically, we want to minimize: sum_{k} [h * (1 - h_k)] + epsilon * ||h||_2^2
        # But for simplicity, we'll cycle through these to scale the problem
        current_shift = top_shifts[iteration % len(top_shifts)]
        
        # Since CVXPY can't easily do convolution, we work directly with a simplified approach
        # Create a loss for the dominant overlaps
        loss = 0
        for i in range(n_points):
            j = i + current_shift
            if 0 <= j < n_points:
                # This is a simplified formulation. The actual implementation needs precision.
                # In practice, we solve this problem using a smooth surrogate.
                # A more principled approach would define the full correlation within CVXPY
                # but that introduces complexity and memory requirements.
                pass
            else:
                loss += h[i] * (1 - h[i])
        
        # Simplified direct convex optimization
        # Goal: minimize loss + epsilon * ||h||_2^2 with constraints
        # Define a quadratic loss based on identified worst-case overlaps
        def compute_loss_and_grad(h_vec):
            # Get the max overlap index
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (n_points - 1)
            
            # Now compute approximate gradient near the peak shift
            # Sensitivity: for a shift k, we consider where h at i and h at i+k appear in 
            # the correlation, so for overlap contribution h[i] * (1 - h[i+k])
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + max_lag
                if 0 <= j < n_points:
                    grad[i] = 1 - 2 * h_vec[i] - h_vec[j]
                else:
                    grad[i] = 1 - 2 * h_vec[i]
            
            # This is a bit of hack to create a tractable quadratic problem.
            # Instead, we step in the negative gradient direction.
            # For the actual term, compute the expected value.
            
            # Conservative approach:
            # We want to minimize the largest overlap.
            # So we model the overlap contribution using gradient descent on h.
            loss = -np.max(corr) # Trying to reduce the max overlap
            return loss, grad
        
        # But for efficiency, we'll reframe the problem differently 
        # and resort to a gradient based method using scipy
        
        # Use scipy for the actual optimization step
        def loss_func(x):
            h_test = x.copy()
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            corr = np.correlate(h_test, 1 - h_test, mode='full')
            return -np.max(corr)
        
        def jac_func(x):
            h_test = x.copy()
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            new_h = h_test.copy()
            
            corr = np.correlate(new_h, 1 - new_h, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (n_points - 1)
            grad = np.zeros(n_points)
            
            for i in range(n_points):
                if i + max_lag >= 0 and i + max_lag < n_points:
                    # This evaluates (1 - 2h[i] - h[i + max_lag])
                    grad[i] = 1 - 2 * new_h[i] - new_h[i + max_lag]
                else:
                    # Out of bounds
                    grad[i] = 1 - 2 * new_h[i]
            
            # Add gradient of the renormalization term
            sum_h = np.sum(new_h)
            grad += (1.0 / sum_h) * (sum_h - new_h * sum_h)
            return -grad  # We're minimizing the negative
            
        
        # Optimization step
        try:
            res = minimize(loss_func, h.copy(), method='L-BFGS-B', 
                           bounds=[(0, 1)]*n_points, 
                           jac=jac_func, options={'maxiter': 10})
            if res.success:
                h_new = res.x.copy()
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
            else:
                # Fall back to simple gradient descent
                corr = np.correlate(h, 1 - h, mode='full')
                max_lag_idx = np.argmax(corr)
                max_lag = max_lag_idx - (n_points - 1)
                grad = np.zeros(n_points)
                for i in range(n_points):
                    j = i + max_lag
                    if 0 <= j < n_points:
                        grad[i] = 1 - 2 * h[i] - h[j]
                    else:
                        grad[i] = 1 - 2 * h[i]
                
                h_new = h - 0.01 * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
        except:
            # Fallback to gradient descent
            corr = np.correlate(h, 1 - h, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + max_lag
                if 0 <= j < n_points:
                    grad[i] = 1 - 2 * h[i] - h[j]
                else:
                    grad[i] = 1 - 2 * h[i]
            
            h_new = h - 0.01 * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        
        # Adaptive decaying epsilon
        epsilon *= beta

    # Final refinement with local search
    if time() - start_time > budget_s * 0.95:
        pass
    else:
        ref_h = best_h.copy()
        ref_c5 = best_c5
        for _ in range(200):
            if time() - start_time > budget_s * 0.98:
                break
            pert = np.random.normal(0, 0.002, n_points)
            h_new = ref_h + pert
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < ref_c5:
                ref_c5 = c5_new
                ref_h = h_new.copy()
        best_h = ref_h
        best_c5 = ref_c5

    return (list(best_h), best_c5, n_points)
