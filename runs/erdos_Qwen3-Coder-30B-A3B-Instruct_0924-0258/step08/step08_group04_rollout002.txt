import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Coarse grid optimization
    n_coarse = 15
    dx_coarse = 2.0 / n_coarse
    population_size = 50
    population = []
    
    # Initialize population
    for _ in range(population_size):
        h = np.random.rand(n_coarse)
        h = h * (n_coarse/2) / np.sum(h)
        population.append(h)
    
    # Genetic algorithm on coarse grid
    for generation in range(100):
        if time() - start_time > budget_s * 0.8:
            break
            
        fitness = []
        for h in population:
            fitness.append(-evaluate_erdos_solution(h, dx_coarse))
        
        # Selection (tournament)
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(len(population), 3)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            # Crossover
            if np.random.rand() < 0.5:
                crossover_point = np.random.randint(1, n_coarse)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for h in [child1, child2]:
                for j in range(n_coarse):
                    if np.random.rand() < 0.2:
                        h[j] += np.random.normal(0, 0.05)
                h = np.clip(h, 0, 1)
                h = h * (n_coarse/2) / np.sum(h)
                
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
    
    # Find best coarse solution
    coarse_fitness = [evaluate_erdos_solution(h, dx_coarse) for h in population]
    best_coarse_idx = np.argmin(coarse_fitness)
    best_coarse = population[best_coarse_idx].copy()
    
    # Upscale to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, n_coarse), 
                       best_coarse)
    h_fine = h_fine * (n_points/2) / np.sum(h_fine)
    
    h_current = h_fine.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Gradient-based optimization on fine grid
    step_size = 0.01
    max_iter = 5000
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all correlations for all possible lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Get the maximum overlap lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Multi-lag gradient approach: focus on largest overlaps
        gradient = np.zeros(n_points)
        lag_indices = np.argsort(-corr)[:5]  # Focus on top 5 overlaps
        for lag_idx in lag_indices:
            lag = lag_idx - (len(h_current) - 1)
            if lag >= 0:
                for i in range(n_points - lag):
                    gradient[i] += 1 - h_current[i + lag]
                    if i + lag < n_points:
                        gradient[i + lag] -= h_current[i]
            else:
                for i in range(n_points + lag):
                    gradient[i] += 1 - h_current[i - lag]
                    if i - lag >= 0:
                        gradient[i - lag] -= h_current[i]
        
        # Gradient descent step
        h_new = h_current - step_size * gradient
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # If not improving significantly, reduce step size
        if c5_new >= evaluate_erdos_solution(h_current, dx) and iter_count % 100 == 0:
            step_size *= 0.9
            if step_size < 1e-15:
                break
        else:
            h_current = h_new
        
        iter_count += 1
    
    return (list(best_h), best_c5, n_points)
