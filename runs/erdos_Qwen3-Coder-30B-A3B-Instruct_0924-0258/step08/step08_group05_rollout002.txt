import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a piecewise constant function with alternating 0s and 1s
        h_start = np.zeros(n_points)
        for i in range(n_points):
            h_start[i] = 0.5 if i % 2 == 0 else 0.0
        h_start[0] = 1.0  # Make sure we have enough area
        h_start[1] = 0.0  # One at beginning, then alternating
        
    dx = 2.0 / n_points
    
    # Normalize h_start to have correct integral
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use SLSQP with different starting points
    np.random.seed(42)
    
    # Try multiple random restarts
    n_restarts = 5
    for restart in range(n_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate random starting point with proper sum constraint
        h_random = np.random.rand(n_points)
        h_random = h_random * (n_points/2) / np.sum(h_random)
        
        # Clip to [0,1] range
        h_random = np.clip(h_random, 0, 1)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_random, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()

            pass
        
        # Early exit if time's up

            break
                
    # Final refinement using more Iterations
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        # Use best solution so far as starting point
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
