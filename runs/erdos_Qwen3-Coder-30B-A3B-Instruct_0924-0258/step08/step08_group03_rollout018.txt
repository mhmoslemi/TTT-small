import numpy as np
from time import time
import cvxpy as cp
from scipy.linalg import svd
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 35
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search on C5 using SDP relaxation
    low, high = 0.38, 0.38085
    tolerance = 1e-5
    max_iterations = 100
    
    def sdp_feasible(C5):
        # Create SDP variables
        Y = cp.Variable((n_points, n_points), symmetric=True)
        h_var = cp.Variable(n_points)
        
        # SDP constraints
        constraints = [
            Y >> 0,  # Y is positive semidefinite
            cp.diag(Y) == h_var,  # Diagonal elements
            cp.sum(h_var) == n_points / 2,  # Sum constraint
        ]
        
        # Overlap constraints for sampled shifts
        shifts = np.random.choice(n_points, min(100, n_points), replace=False)
        for k in shifts:
            if k == 0:
                continue
            # Sum_i h_i * (1 - h_{i+k}) <= C5 * dx
            # This translates to sum_i h_i - sum_i h_i * h_{i+k} <= C5 * dx
            # Which is sum_i h_i - sum_i Y_{i, i+k} <= C5 * dx
            expr = cp.sum(h_var) - cp.sum([Y[i, (i+k) % n_points] for i in range(n_points)])
            constraints.append(expr <= C5 * dx)
        
        # Objective: minimize something (not used since this is feasibility)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            return prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]
        except:
            return False
    
    # Binary search for tightest C5
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
        mid = (low + high) / 2
        if sdp_feasible(mid):
            high = mid
            # Try to get a good feasible solution from SDP
            Y = cp.Variable((n_points, n_points), symmetric=True)
            h_var = cp.Variable(n_points)
            
            constraints = [
                Y >> 0,
                cp.diag(Y) == h_var,
                cp.sum(h_var) == n_points / 2,
            ]
            
            shifts = np.random.choice(n_points, min(100, n_points), replace=False)
            for k in shifts:
                if k == 0:
                    continue
                expr = cp.sum(h_var) - cp.sum([Y[i, (i+k) % n_points] for i in range(n_points)])
                constraints.append(expr <= mid * dx)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.MOSEK, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_sdp = h_var.value
                    if h_sdp is not None:
                        # Project back to valid h (non-negative and sum to n/2)
                        h_sdp = np.clip(h_sdp, 0, 1)
                        h_sdp = h_sdp * (n_points / 2) / np.sum(h_sdp)
                        c5_sdp = evaluate_erdos_solution(h_sdp, dx)
                        if c5_sdp < best_c5:
                            best_c5 = c5_sdp
                            best_h = h_sdp.copy()
            except:
                pass
        else:
            low = mid
    
    # Local refinement with gradient descent
    def objective(h_vals):
        h_vec = np.array(h_vals)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    def constraint_bounds(h_vals):
        return np.concatenate([h_vals, 1 - h_vals])
    
    # Use scipy minimize for refinement
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
    ]
    
    h_refined = best_h.copy()
    c5_refined = best_c5
    
    for i in range(5000):  # Limited iterations to avoid exceeding time
        if time() - start_time > budget_s * 0.98:
            break
            
        # Gradient descent step
        step_size = 0.001
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for j in range(n_points):
            k = j + max_lag
            if 0 <= k < n_points:
                grad[j] = 1 - 2 * h_refined[j] - h_refined[k]
            else:
                grad[j] = 1 - 2 * h_refined[j]
        
        h_new = h_refined - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < c5_refined:
            c5_refined = c5_new
            h_refined = h_new.copy()
        else:
            step_size *= 0.5
            
        if i % 200 == 0:
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
