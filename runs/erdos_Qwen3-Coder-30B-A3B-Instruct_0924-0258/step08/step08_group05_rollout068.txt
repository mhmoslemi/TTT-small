from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    if 'initial_h_values' in globals():

    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Tracking for early stopping
    last_improvement = 0
    generation = 0

        # Evaluate all individuals
        fitness_scores = [evaluate_erdos_solution(h, dx) for h in population]

        best_in_pop_idx = np.argmin(fitness_scores)
        if fitness_scores[best_in_pop_idx] < best_c5:

            best_h = population[best_in_pop_idx].copy()
            last_improvement = generation
            
        # Tournament selection (size 5)
        def tournament_select():
            selected_indices = np.random.choice(len(population), 5)
            selected_fitnesses = [fitness_scores[i] for i in selected_indices]
            winner_idx = selected_indices[np.argmin(selected_fitnesses)]
            return population[winner_idx].copy()

        new_population = [best_h.copy()]  # Elitism

            parent2 = tournament_select()

                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
