import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 5
    no_improvement_limit = 100
    slsqp_refine_interval = 100
    max_no_improvement = 200
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        population[0] = np.clip(population[0], 0, 1)
        population[0] = population[0] * (n_points / 2) / np.sum(population[0])
        
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    no_improvement_count = 0
    eval_count = 0
    
    # GA main loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            fitness = evaluate_erdos_solution(h, dx)
            fitness_scores.append(fitness)
            if fitness < best_c5:
                best_c5 = fitness
                best_h = h.copy()
                no_improvement_count = 0
                
        no_improvement_count += 1
        
        # Tournament selection
        def tournament_selection(tournament_size=3):
            tournament_indices = random.sample(range(population_size), tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            return population[winner_idx].copy()
        
        # Crossover (blend crossover) and mutation
        new_population = []
        
        # Elitism: keep the best individuals
        elite_indices = np.argsort(fitness_scores)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
            
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection()
            parent2 = tournament_selection()
            
            # Crossover
            if random.random() < crossover_rate:
                alpha = random.random()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
                
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child1[i] += random.gauss(0, 0.05)
                    child2[i] += random.gauss(0, 0.05)
            
            # Ensure constraints
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        population = new_population
        eval_count += population_size
        if eval_count % slsqp_refine_interval == 0:
            # Local refinement with SLSQP
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            constraint = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=constraint, 
                               options={'maxiter': 50, 'ftol': 1e-10})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
                
    return (list(best_h), best_c5, n_points)
