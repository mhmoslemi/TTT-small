from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    # Genetic Algorithm parameters
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 500
    stagnation_limit = 50
    elite_size = 2
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def create_population():
        pop = []
        if 'initial_h_values' in globals():
            pop.append(h_start.copy())
            for _ in range(population_size - 1):
                pop.append(create_individual())
        else:
            for _ in range(population_size):
                pop.append(create_individual())
        return pop
    
    # Fitness function
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    # Tournament selection
    def tournament_selection(pop, fitnesses, k=3):
        idxs = np.random.choice(len(pop), k)
        best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
        return pop[best_idx].copy()
    
    # Blend crossover
    def blend_crossover(parent1, parent2, alpha=0.5):
        child = np.zeros_like(parent1)
        for i in range(len(parent1)):
            if np.random.rand() < crossover_rate:
                lb = min(parent1[i], parent2[i]) - alpha * abs(parent1[i] - parent2[i])
                ub = max(parent1[i], parent2[i]) + alpha * abs(parent1[i] - parent2[i])
                child[i] = lb + np.random.rand() * (ub - lb)
            else:
                child[i] = parent1[i]
        # Ensure constraint
        child = child * (n_points / 2) / np.sum(child)
        return np.clip(child, 0, 1)
    
    # Mutation with adaptive rate
    def mutate(h, rate=0.1, step_size=0.05):
        new_h = h.copy()
        for i in range(len(new_h)):
            if np.random.rand() < rate:
                new_h[i] += np.random.normal(0, step_size)
        # Ensure constraint
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        return np.clip(new_h, 0, 1)
    
    # Main GA loop
    population = create_population()
    fitnesses = [fitness(ind) for ind in population]
    best_fitness_history = [min(fitnesses)]
    stagnant_count = 0
    
    generation = 0
    while time() - start_time < budget_s * 0.7 and generation < generations:
        # Evaluate current generation
        fitnesses = [fitness(ind) for ind in population]
        current_best = min(fitnesses)

            best_c5 = current_best
            best_h = population[np.argmin(fitnesses)].copy()
            stagnant_count = 0
        else:
            stagnant_count += 1

            break

        sorted_indices = np.argsort(fitnesses)
        new_population = [population[i] for i in sorted_indices[:elite_size]]
        
        # Generate offspring
        while len(new_population) < population_size:

            parent2 = tournament_selection(population, fitnesses)
            child = blend_crossover(parent1, parent2)

        generation += 1

        return evaluate_erdos_solution(h, dx)

                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
