import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Start with initial construction
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    current_c5 = evaluate_erdos_solution(h, dx)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Local improvement using gradient descent
    max_iter = 2000
    patience = 100
    no_improvement = 0
    step_size = 0.01
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate overlap for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            break
            
        # Identify lag with maximum overlap
        max_lag = max_lag_idx - (n_points - 1)  # Convert index to actual lag
        
        # Compute gradient w.r.t. to h for this specific lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 2 * (1 - h[j] - h[i]) / n_points
        
        # Update with gradient descent
        h_new = h - step_size * grad
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check if the update improved the objective
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
        else:
            # Reduce step size if no improvement
            step_size *= 0.5
            if step_size < 1e-8:
                break
    
    # Try different random restarts with different parameters
    n_restarts = 10
    for restart in range(n_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Start with perturbed version of best solution
        h_perturbed = best_h.copy() + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Local search from this point
        h_local = h_perturbed.copy()
        local_c5 = evaluate_erdos_solution(h_local, dx)
        local_patience = 0
        
        for local_iter in range(300):
            if time() - start_time > budget_s * 0.95:
                break
                
            corr = np.correlate(h_local, 1 - h_local, mode='full')
            max_lag_idx = np.argmax(corr)
            max_overlap = corr[max_lag_idx] * dx
            
            if max_overlap < local_c5:
                local_c5 = max_overlap
                local_patience = 0
            else:
                local_patience += 1
            if local_patience >= 30:
                break
                
            max_lag = max_lag_idx - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + max_lag
                if 0 <= j < n_points:
                    grad[i] = 2 * (1 - h_local[j] - h_local[i]) / n_points
            
            h_new = h_local - 0.005 * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            if evaluate_erdos_solution(h_new, dx) < local_c5:
                h_local = h_new
            else:
                break
                
        # Check if we found an improvement
        if local_c5 < best_c5:
            best_c5 = local_c5
            best_h = h_local.copy()
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
