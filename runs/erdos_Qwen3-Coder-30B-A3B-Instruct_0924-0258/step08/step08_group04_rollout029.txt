import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Multiresolution approach: coarse to fine
    coarse_n = 15
    fine_n = 35
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.5
    
    # Coarse grid optimization
    coarse_dx = 2.0 / coarse_n
    population = []
    for i in range(pop_size):
        h = np.random.rand(coarse_n) * 0.5 + 0.25  # balanced start
        h = h * (coarse_n/2) / np.sum(h)
        population.append(h.copy())
    
    # Gradient descent on coarse grid for a while
    coarse_iter = 0
    coarse_budget = 60
    while time() - start_time < start_time + coarse_budget and coarse_iter < 1000:
        # Evaluate all candidates
        scores = []
        for h in population:
            score = evaluate_erdos_solution(h, coarse_dx)
            scores.append(score)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(scores)
        top_candidates = [population[i] for i in sorted_indices[:pop_size//2]]
        
        # Keep best
        if scores[sorted_indices[0]] < best_c5:
            best_c5 = scores[sorted_indices[0]]
            best_h = top_candidates[0].copy()
        
        # Generate new population
        new_population = top_candidates.copy()
        
        # Crossover
        for i in range(0, len(top_candidates), 2):
            if i+1 < len(top_candidates) and np.random.rand() < crossover_rate:
                parent1 = top_candidates[i]
                parent2 = top_candidates[i+1]
                crossover_point = np.random.randint(1, coarse_n-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                new_population.extend([child1, child2])
            else:
                new_population.append(top_candidates[i])
        
        # Mutation
        for i in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                h = new_population[i]
                mutate_point = np.random.randint(0, coarse_n)
                h[mutate_point] += (np.random.rand() - 0.5) * 0.1
                h[mutate_point] = np.clip(h[mutate_point], 0, 1)
                # Rescale
                h = h * (coarse_n/2) / np.sum(h)
                new_population[i] = h
        
        # Limit population size
        population = new_population[:pop_size]
        coarse_iter += 1
    
    # Select best from coarse population
    coarse_scores = [evaluate_erdos_solution(h, coarse_dx) for h in population]
    best_coarse_idx = np.argmin(coarse_scores)
    best_coarse_h = population[best_coarse_idx]
    
    # Upscale to fine grid
    fine_h = np.interp(np.linspace(0, 2, fine_n), 
                       np.linspace(0, 2, coarse_n), 
                       best_coarse_h)
    fine_h = fine_h * (fine_n/2) / np.sum(fine_h)
    
    h_current = fine_h.copy()
    best_h = fine_h.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Gradient descent on fine grid
    step_size = 0.01
    iter_count = 0
    max_iter = 10000
    
    while time() - start_time < start_time + budget_s * 0.95 and iter_count < max_iter:
        # Compute all correlations
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_indices = np.argsort(corr)[-5:]  # top 5 largest overlaps
        
        # Compute combined gradient with multiple dominant lags
        gradient = np.zeros(fine_n)
        
        for lag_idx in max_lag_indices:
            lag = lag_idx - (fine_n - 1)
            if lag >= 0:
                for i in range(fine_n - lag):
                    gradient[i] += 1 - h_current[i + lag]
                    if i + lag < fine_n:
                        gradient[i + lag] -= h_current[i]
            else:
                for i in range(fine_n + lag):
                    gradient[i] += 1 - h_current[i - lag]
                    if i - lag >= 0:
                        gradient[i - lag] -= h_current[i]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 1e-10:
            gradient = gradient / grad_norm
            
        # Update
        h_new = h_current - step_size * gradient
        
        # Ensure feasibility and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (fine_n/2) / np.sum(h_new)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Check improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Adjust step size
        if c5_new >= evaluate_erdos_solution(h_current, dx):
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            h_current = h_new
            step_size *= 1.05
        
        iter_count += 1
    
    return (list(best_h), best_c5, fine_n)
