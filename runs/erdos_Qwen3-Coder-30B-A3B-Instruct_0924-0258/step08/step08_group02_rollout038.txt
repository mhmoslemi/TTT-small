import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Binary search initialization
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Set a few key parameters for the optimization process
    max_iterations = 500
    patience = 20
    patience_counter = 0
    best_c5 = float('inf')
    
    # Try different configurations - grid search on thresholding parameter
    threshold_values = np.linspace(0.01, 0.2, 20)
    
    for t in threshold_values:
        if time() - start_time > budget_s * 0.8:
            break
            
        h_temp = h.copy()
        current_c5 = evaluate_erdos_solution(h_temp, dx)
        
        # Sparse optimization step
        for iter in range(200):
            if time() - start_time > budget_s * 0.95:
                break
            
            overlaps = np.correlate(h_temp, 1 - h_temp, mode='full') * dx
            max_overlap = np.max(overlaps)
            l_max = np.argmax(overlaps)
            
            # Compute estimated gradient
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = (i + l_max) % n_points
                grad[i] = -2 * dx * (1 - h_temp[j] - h_temp[i])
            
            # Apply thresholding to update
            update = grad * 0.01
            update[np.abs(update) < t] = 0
            
            h_temp = h_temp - update
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points/2) / np.sum(h_temp)
            
            current_c5 = evaluate_erdos_solution(h_temp, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_temp.copy()
                patience_counter = 0
            else:
                patience_counter += 1
                
            if patience_counter > patience:
                break
                
    # Phase 2: Local refinement using L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    # Ensure sum constraint and bounds are still valid
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)

    return (list(best_h), best_c5, n_points)
