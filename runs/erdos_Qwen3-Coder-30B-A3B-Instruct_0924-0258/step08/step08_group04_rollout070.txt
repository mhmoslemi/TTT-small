import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = project_to_feasible(h_start, n_points)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    T0 = 0.05
    alpha = 0.999
    beta = 0.999
    T_min = 1e-6
    sigma = 0.05  # Initial perturbation scale
    
    T = T0
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and T > T_min:
        # Generate neighbor by perturbing with Gaussian noise
        delta_h = np.random.normal(0, sigma, n_points)
        h_new = h_current + delta_h
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        # Metropolis acceptance criterion
        delta_c5 = c5_new - c5_current
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h_current = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        
        # Cooling and adaptation
        T *= alpha
        sigma *= beta
        iter_count += 1
    
    return (list(best_h), best_c5, n_points)
