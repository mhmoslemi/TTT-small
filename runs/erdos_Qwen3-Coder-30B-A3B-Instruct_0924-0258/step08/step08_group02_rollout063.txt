import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Binary threshold search
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Faster approach: binary search for best threshold
    # Create binary-like approximation
    binary_h = np.zeros(n_points)
    mid = n_points // 2
    binary_h[:mid] = 1.0
    np.random.shuffle(binary_h)
    
    # Ensure sum constraint
    current_sum = np.sum(binary_h)
    scale = (n_points/2) / current_sum
    binary_h = binary_h * scale
    
    # Gradient-based enhancement with sparse updates
    for iter in range(1000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current state
        current_c5 = evaluate_erdos_solution(binary_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = binary_h.copy()
        
        # Compute overlaps for all lags
        overlaps = np.correlate(binary_h, 1 - binary_h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute approximate gradient
        # Using finite differences to estimate gradient direction
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            # Simple derivative approximation
            gradient[i] = -2 * dx * (1 - binary_h[j] - binary_h[i])
        
        # Sparse gradient update
        step_size = 0.01
        # Only update significant gradient directions
        threshold = np.percentile(np.abs(gradient), 50)
        update_mask = np.abs(gradient) > threshold
        
        # Apply update
        new_h = binary_h - step_size * gradient
        new_h = np.clip(new_h, 0, 1)
        
        # Re-scale to satisfy sum constraint
        new_sum = np.sum(new_h)
        new_h = new_h * (n_points/2) / new_sum
        
        # Keep the improvement
        binary_h = new_h
    
    # Phase 2: Local refinement using L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 200, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
