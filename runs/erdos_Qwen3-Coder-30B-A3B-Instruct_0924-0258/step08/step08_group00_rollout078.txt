import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()

    # Initialize from initial_h_values or start coarse with n_points=35
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)

    # Multi-resolution SA with local refinement
    resolutions = [35, 70, 140, 280]
    patience = 100
    no_improvement = 0

    for res in resolutions:
        # If current resolution is not the coarsest, upscale the previous solution
        if res > n_points:
            h = np.interp(np.linspace(0, 1, res), np.linspace(0, 1, n_points), best_h)
        else:
            h = best_h.copy()

        n_points = res
        dx = 2.0 / n_points
        h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint

        # SA parameters at this resolution
        initial_temp = 1.0
        cooling_rate = 0.95
        num_iter_per_temp = 500
        temp = initial_temp

        for iter_count in range(num_iter_per_temp * 1000):  # High upper limit for safety
            if time() - start_time > budget_s * 0.95:
                break

            # Make a random perturbation
            h_new = h.copy()
            idx = np.random.randint(0, n_points)
            perturb = np.random.normal(0, temp * 0.1)
            h_new[idx] += perturb
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)

            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            c5_old = evaluate_erdos_solution(h, dx)

            # Accept or reject
            delta = c5_new - c5_old
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improvement = 0
            else:
                no_improvement += 1

            if no_improvement >= patience:
                break

            # Cooling
            temp *= cooling_rate

        # Local refinement with SLSQP
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)

            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0

            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]

            res = minimize(objective, h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 500, 'ftol': 1e-9})

            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass

        if time() - start_time > budget_s * 0.95:
            break

    return (list(best_h), best_c5, n_points)
