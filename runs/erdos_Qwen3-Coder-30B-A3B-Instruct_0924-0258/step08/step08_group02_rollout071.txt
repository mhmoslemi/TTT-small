import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a simple binary-like pattern
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1.0
        # Add some randomness to avoid local minima
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Thresholding-based optimization with cooling
    threshold = 0.1
    alpha = 0.95
    max_iterations = 500
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for each point
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - best_h[j] - best_h[i])
        
        # Rank by gradient magnitude
        idx_sorted = np.argsort(np.abs(gradient))[::-1]
        
        # Apply thresholding to top 50%
        h_new = best_h.copy()
        for i in idx_sorted[:n_points//2]:
            if h_new[i] >= 1 - threshold:
                h_new[i] = 1
            elif h_new[i] <= threshold:
                h_new[i] = 0
            else:
                # Keep intermediate values but refine them
                pass
        
        # Maintain sum constraint
        current_sum = np.sum(h_new)
        scale_factor = (n_points / 2) / current_sum
        h_new = np.clip(h_new * scale_factor, 0, 1)
        
        # Rebalance excess if necessary
        current_sum = np.sum(h_new)
        if abs(current_sum - n_points / 2) > 1e-6:
            # Distribute the error to the points with highest gradient impact
            diff = current_sum - n_points / 2
            impact_idx = idx_sorted[:min(5, n_points)]
            for i in impact_idx:
                if abs(diff) < 1e-8:
                    break
                adjust = min(0.1, abs(diff))
                if diff > 0:
                    h_new[i] = max(0, h_new[i] - adjust)
                else:
                    h_new[i] = min(1, h_new[i] + adjust)
                diff -= adjust
        
        # Evaluate new solution
        current_c5 = evaluate_erdos_solution(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            
            # Adaptive threshold
            threshold = max(0.01, threshold * alpha)
        else:
            # Reduce threshold to explore more
            threshold = max(0.01, threshold * 0.99)
    
    # Coordinate-wise optimization
    def refine_coordinate(h, i):
        def obj(x):
            h_test = h.copy()
            h_test[i] = x
            h_test = h_test * (n_points/2) / np.sum(h_test)  # Ensure constraint
            return evaluate_erdos_solution(h_test, dx)
        # Golden section search for best x in [0,1]
        a, b = 0, 1
        res = minimize(obj, x0=(a+b)/2, bounds=[(0,1)], method='L-BFGS-B')
        return max(0, min(1, res.x[0]))
    
    # Refine top 20% by impact
    overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
    max_overlap = np.max(overlaps)
    l_max = np.argmax(overlaps)
    
    gradient = np.zeros(n_points)
    for i in range(n_points):
        j = (i + l_max) % n_points
        gradient[i] = -2 * dx * (1 - best_h[j] - best_h[i])
        
    idx_sorted = np.argsort(np.abs(gradient))[::-1]
    for i in idx_sorted[:n_points//5]:
        if time() - start_time > budget_s * 0.9:
            break
        new_value = refine_coordinate(best_h, i)
        best_h[i] = new_value
        
    # Normalize and finalize
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Final polishing with scipy
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
