import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Coarse-to-fine approach
    # First, work with a coarse grid to explore structure
    coarse_points = 15
    coarse_dx = 2.0 / coarse_points
    h_coarse = np.ones(coarse_points) * 0.5
    if 'initial_h_values' in globals():
        h_coarse = np.interp(np.linspace(0, 2, coarse_points), 
                             np.linspace(0, 2, len(initial_h_values)), 
                             initial_h_values)
    h_coarse = h_coarse * (coarse_points/2) / np.sum(h_coarse)
    
    # Genetic algorithm on coarse grid
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.5
    max_generations = 1000
    generations = 0
    
    population = []
    for _ in range(population_size):
        ind = np.random.rand(coarse_points)
        ind = ind * (coarse_points/2) / np.sum(ind)
        population.append(ind)
    
    # Add current solution as first individual
    population[0] = h_coarse.copy()
    
    while time() - start_time < budget_s * 0.1 and generations < max_generations:
        # Evaluate fitness (negative C5)
        fitnesses = []
        for ind in population:
            c5 = evaluate_erdos_solution(ind, coarse_dx)
            fitnesses.append(-c5)
        
        # Select top 50% (tournament selection)
        selected = []
        for _ in range(population_size // 2):
            idxs = np.random.choice(population_size, 3, replace=False)
            best_idx = idxs[np.argmax([fitnesses[i] for i in idxs])]
            selected.append(population[best_idx].copy())
        
        # Crossover and mutation
        new_pop = selected[:]
        
        while len(new_pop) < population_size:
            if np.random.rand() < crossover_rate and len(selected) >= 2:
                a, b = np.random.choice(len(selected), 2, replace=False)
                point = np.random.randint(1, coarse_points - 1)
                child = np.concatenate([selected[a][:point], selected[b][point:]])
                child = child * (coarse_points/2) / np.sum(child)
            else:
                child = selected[np.random.randint(len(selected))].copy()
                
            # Mutation
            if np.random.rand() < mutation_rate:
                mut_idx = np.random.randint(coarse_points)
                child[mut_idx] += np.random.normal(0, 0.05)
                child[mut_idx] = np.clip(child[mut_idx], 0, 1)
                child = child * (coarse_points/2) / np.sum(child)
            
            if len(new_pop) < population_size:
                new_pop.append(child)
        
        population = new_pop
        generations += 1
    
    # Find best in coarse population
    best_coarse = min(population, key=lambda x: evaluate_erdos_solution(x, coarse_dx))
    
    # Upscale to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, coarse_points), 
                       best_coarse)
    h_fine = h_fine * (n_points/2) / np.sum(h_fine)
    
    # Fine optimization
    h_current = h_fine
    step_size = 0.005
    max_iter = 10000
    iter_count = 0

    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all correlations
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Get the maximum overlap Lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)

        # Compute gradient for the max overlap lag
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_current[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= h_current[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_current[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_current[i]
        
        # Gradient descent step with clipped gradient magnitude
        grad_magnitude = np.linalg.norm(gradient)
        if grad_magnitude > 0:
            gradient = gradient / grad_magnitude
            
        h_new = h_current - step_size * gradient

        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # If not improving, reduce step size
        if c5_new >= evaluate_erdos_solution(h_current, dx):
            step_size *= 0.95
            if step_size < 1e-10:
                break
        else:
            h_current = h_new
            step_size *= 1.01  # Increase step size slightly
        
        iter_count += 1
        
    return (list(best_h), best_c5, n_points)
