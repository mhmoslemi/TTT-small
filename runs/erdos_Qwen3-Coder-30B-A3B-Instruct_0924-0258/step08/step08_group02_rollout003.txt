import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter = 100000
    
    # Initialize
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    current_c5 = evaluate_erdos_solution(h, dx)
    
    T = T_init
    iter_count = 0
    
    while T > T_min and time() - start_time < budget_s * 0.9:
        if iter_count >= max_iter:
            break
            
        # Perturb two indices
        i, j = np.random.choice(n_points, 2, replace=False)
        eps = np.random.uniform(-0.1, 0.1)
        
        # Save old values
        old_i, old_j = h[i], h[j]
        
        # Update values
        h[i] = old_i + eps
        h[j] = old_j - eps
        
        # Enforce bounds
        h[i] = np.clip(h[i], 0, 1)
        h[j] = np.clip(h[j], 0, 1)
        
        # Adjust to maintain sum
        diff = np.sum(h) - n_points / 2
        if abs(diff) > 1e-8:
            h[i] -= diff / 2
            h[j] += diff / 2
            h[i] = np.clip(h[i], 0, 1)
            h[j] = np.clip(h[j], 0, 1)
        
        # Compute new overlap
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        else:
            # Accept with probability
            prob = np.exp(-(new_c5 - current_c5) / T)
            if np.random.random() < prob:
                current_c5 = new_c5
            else:
                # Revert change
                h[i], h[j] = old_i, old_j
        
        # Cooling
        T *= alpha
        iter_count += 1
    
    # Local refinement using L-BFGS-B
    try:
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Normalize initial for refinement
        h_refine = best_h.copy()
        h_refine = h_refine * (n_points/2) / np.sum(h_refine)
        
        res = minimize(
            objective,
            h_refine,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            h_refine = res.x
            h_refine = h_refine * (n_points/2) / np.sum(h_refine)
            refined_c5 = evaluate_erdos_solution(h_refine, dx)
            if refined_c5 < best_c5:
                best_h = h_refine
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
