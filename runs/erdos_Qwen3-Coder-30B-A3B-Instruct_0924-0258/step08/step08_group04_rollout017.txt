import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Try SDP approach for better solution
    # We want to minimize C5 such that all overlaps are bounded above by C5*dx
    try:
        # Create CVXPY variables
        h_vars = cp.Variable(n_points, bounds=(0, 1))
        
        # Constraints for sum
        sum_constraint = cp.Equality(cp.sum(h_vars), n_points/2)
        
        # Precompute overlaps
        max_lag = n_points - 1
        constraints = []
        
        # For each possible lag k
        for k in range(-max_lag + 1, max_lag):
            # Find indices where i+k is valid
            if k >= 0:
                valid_indices = list(range(n_points - k))
                term = cp.sum(cp.multiply(h_vars[valid_indices], 1 - h_vars[valid_indices + k]))
            else:
                valid_indices = list(range(-k, n_points))
                term = cp.sum(cp.multiply(h_vars[valid_indices], 1 - h_vars[valid_indices - k]))
            
            # Add constraint: sum of h[i]*(1-h[i+k]) <= C5 * dx
            # We'll handle this when solving for C5
            constraints.append(term <= cp.Variable(1) * dx)
            
        # Create optimal C5 as a variable
        C5_var = cp.Variable(1, name='C5')
        
        # Add all constraints
        all_constraints = [sum_constraint] + constraints
        
        # Objective - minimize C5
        objective = cp.Minimize(C5_var)
        
        # Solve with early stopping
        prob = cp.Problem(objective, all_constraints)
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s - 5)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE] and not np.isnan(C5_var.value):
            h_sol = h_vars.value
            # Only consider solution if it's feasible
            if h_sol is not None and np.all(h_sol >= 0) and np.all(h_sol <= 1):
                h_sol = h_sol * (n_points/2) / np.sum(h_sol)
                c5_new = evaluate_erdos_solution(h_sol, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_sol.copy()
                    
    except Exception as e:
        pass  # Continue with previous approach if SDP fails

    # Continue with gradient descent-based approach to fine-tune
    step_size = 0.01
    max_iter = 10000
    
    while time() - start_time < budget_s * 0.95 and max_iter > 0:
        # Compute all correlations for all possible lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Get the maximum overlap lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Compute incremental gradient for the max overlap lag
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_current[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= h_current[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_current[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_current[i]
        
        # Gradient descent step
        h_new = h_current - step_size * gradient
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # If not improving, reduce step size
        if c5_new >= evaluate_erdos_solution(h_current, dx):
            step_size *= 0.8
            if step_size < 1e-10:
                break
        else:
            h_current = h_new
            step_size = min(0.05, step_size * 1.05)  # Increase step size slightly
        
        max_iter -= 1
    
    return (list(best_h), best_c5, n_points)
