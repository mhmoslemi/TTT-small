import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try multiple random restarts to avoid local minima
    num_restarts = 10
    for restart in range(num_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb the initial solution
        h = h_start.copy()
        # Add small random noise
        noise = np.random.normal(0, 0.02, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
        
        # Local optimization using L-BFGS-B
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final
        except Exception:
            pass
    
    # Final gradient-based refinement
    h = best_h.copy()
    step_size = 0.01
    max_iter = 1000
    patience = 30
    no_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            break
            
        # Compute gradient
        grad = np.zeros(n_points)
        max_lag = max_lag_idx - (n_points - 1)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 2 * (1 - h[j] - h[i]) / n_points
                
        # Update
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        if evaluate_erdos_solution(h_new, dx) < evaluate_erdos_solution(h, dx):
            h = h_new
        else:
            step_size *= 0.5
            if step_size < 1e-8:
                break
    
    # Post-processing: ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
