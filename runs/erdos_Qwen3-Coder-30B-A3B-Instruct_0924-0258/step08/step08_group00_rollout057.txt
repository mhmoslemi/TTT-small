import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Iterative local search with different strategies
    max_iter = 2000
    patience = 30
    no_improvement = 0
    step_size = 0.02
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Precompute correlators for reuse
    corr_buffer = np.zeros((n_points, n_points))
    for i in range(n_points):
        for j in range(n_points):
            k = j - i
            if 0 <= k < n_points:
                corr_buffer[i, j] = (1 - h[j]) * dx
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate overlap for all lags to find maximum
        corr_full = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr_full)
        max_overlap = corr_full[max_lag_idx] * dx
        max_lag = max_lag_idx - (n_points - 1)
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience or max_overlap < 0.3807:
            break
            
        # Initialize gradient
        grad = np.zeros(n_points)
        
        # Weighted gradient based on proximity to overlap maximum
        for i in range(n_points):
            # Consider region expansions around max_lag
            j = i + max_lag
            if 0 <= j < n_points:
                # Gradient components based on the kernel interaction
                grad[i] += 2 * (1 - h[j])
                # Add regularization for shape smoothness
                if i > 0:
                    grad[i] += 0.01 * (h[i] - h[i-1])
                if i < n_points - 1:
                    grad[i] += 0.01 * (h[i] - h[i+1])
        
        # Update with gradient descent with adaptive learning rate
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < max_overlap:
            h = h_new
            step_size = min(0.1, step_size * 1.02)  # Adaptive step growth
        else:
            step_size *= 0.7
            if step_size < 1e-8:
                break
    
    # Final refinement with constrained optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try multiple starting points for more thorough optimization
        best_final_h = h.copy()

            h_test = h_test * (n_points/2) / np.sum(h_test)
            
            res = minimize(objective, h_test, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum,

                h_final = res.x

                if c5_final < best_final_c5:
                    best_final_c5 = c5_final
                    best_final_h = h_final.copy()
        
        if best_final_c5 < best_c5:
            best_h = best_final_h
            best_c5 = best_final_c5
    
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
