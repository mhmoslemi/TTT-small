import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # PSO parameters
    num_particles = 75
    inertia = 0.8
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    particle_best = []
    particle_best_values = []
    
    # Create initial population
    for _ in range(num_particles):
        h = np.random.random(n_points) * 0.5 + 0.25  # Keep values in [0.25, 0.75]
        h = h * (n_points/2) / np.sum(h)  # Renormalize
        h = np.clip(h, 0, 1)  # Ensure bounds
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        particle_best.append(h.copy())
        particle_best_values.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    global_best_idx = np.argmin(particle_best_values)
    global_best = particle_best[global_best_idx].copy()
    global_best_value = particle_best_values[global_best_idx]
    
    # PSO loop
    max_iter = int(budget_s * 0.9 * 1000)  # Rough estimate
    iter_count = 0
    no_improvement_count = 0
    perturbation_rate = 0.01
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1
        
        # Update inertia
        current_inertia = inertia * (0.4 / inertia) ** (iter_count / max_iter)
        
        # Update each particle
        for i in range(num_particles):
            h = particles[i]
            vel = velocities[i]
            pbest = particle_best[i]
            
            # Update velocity
            r1, r2 = np.random.random(2)
            vel = (current_inertia * vel + 
                   cognitive_weight * r1 * (pbest - h) +
                   social_weight * r2 * (global_best - h))
            
            # Clip velocity
            vel = np.clip(vel, -max_velocity, max_velocity)
            
            # Update position
            h_new = h + vel
            
            # Apply boundary constraints
            h_new = np.clip(h_new, 0, 1)
            
            # Renormalize
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Update particle best
            if c5_new < particle_best_values[i]:
                particle_best[i] = h_new.copy()
                particle_best_values[i] = c5_new
                
                # Update global best
                if c5_new < global_best_value:
                    global_best = h_new.copy()
                    global_best_value = c5_new
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            
            # Update particle
            particles[i] = h_new.copy()
            velocities[i] = vel.copy()
            
        # Perturbation to avoid stagnation
        if iter_count % 1000 == 0 and np.random.random() < perturbation_rate:
            for i in range(num_particles):
                if np.random.random() < 0.2:  # 20% chance per particle
                    particles[i] = np.random.random(n_points) * 0.5 + 0.25
                    particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
                    particles[i] = np.clip(particles[i], 0, 1)
                    particle_best[i] = particles[i].copy()
                    particle_best_values[i] = evaluate_erdos_solution(particles[i], dx)
        
        # Early termination check
        if no_improvement_count > 5000:
            break
    
    # Final local refinement on best solution
    h_refined = global_best.copy()
    dx = 2.0 / n_points
    
    # Simple gradient descent refinement
    step_size = 0.01
    for _ in range(1000):
        if time() - start_time > budget_s * 0.99:
            break
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_refined) - 1)
        
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_refined[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= h_refined[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_refined[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_refined[i]
        
        # Update step
        h_new = h_refined - step_size * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < global_best_value:
            h_refined = h_new.copy()
            global_best_value = c5_new
        else:
            step_size *= 0.5
            if step_size < 1e-10:
                break
    
    return (list(h_refined), global_best_value, n_points)
