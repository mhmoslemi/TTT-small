import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Multi-resolution optimization
    # First, try coarse grid for global search
    coarse_points = 15
    coarse_dx = 2.0 / coarse_points
    h_coarse = np.ones(coarse_points) * 0.5
    h_coarse = h_coarse * (coarse_points/2) / np.sum(h_coarse)
    coarse_best_c5 = float('inf')
    coarse_best_h = h_coarse.copy()
    
    # Coarse optimization
    coarse_iter = 0
    coarse_budget = 60.0
    while time() - start_time < coarse_budget and coarse_iter < 500:
        # Genetic operators on coarse grid
        population_size = 50
        population = []
        for _ in range(population_size):
            h = np.random.rand(coarse_points) * 0.8 + 0.1  # keep in [0.1, 0.9]
            h = h * (coarse_points/2) / np.sum(h)
            population.append(h)
        
        for p in population:
            c5 = evaluate_erdos_solution(p, coarse_dx)
            if c5 < coarse_best_c5:
                coarse_best_c5 = c5
                coarse_best_h = p.copy()
        
        # Simple evolution
        population.sort(key=lambda h: evaluate_erdos_solution(h, coarse_dx))
        top_candidates = population[:10]
        
        # Crossover and mutation
        new_pop = []
        for i in range(population_size):
            parent1, parent2 = np.random.choice(top_candidates, 2, replace=False)
            # Blend crossover
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            # Mutation
            if np.random.rand() < 0.2:
                mutation_idx = np.random.randint(0, coarse_points)
                child[mutation_idx] = max(0.01, min(0.99, child[mutation_idx] + np.random.normal(0, 0.05)))
            child = child * (coarse_points/2) / np.sum(child)
            new_pop.append(child)
        population = new_pop
        coarse_iter += 1
    
    # Upscale and fine tune
    if coarse_best_c5 < best_c5:
        h_coarse = coarse_best_h.copy()
        h_current = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, coarse_points), h_coarse)
        h_current = h_current * (n_points/2) / np.sum(h_current)
        best_c5 = evaluate_erdos_solution(h_current, dx)
        best_h = h_current.copy()
    
    # Fine optimization
    step_size = 0.01
    max_iter = 10000
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        corr_indices = np.argsort(corr)[-5:][::-1]  # top 5 lags by highest overlap
        
        # Gradient descent on multiple max overlaps
        gradient = np.zeros(n_points)
        for lag_index in corr_indices:
            lag = lag_index - (len(h_current) - 1)
            if lag >= 0:
                for i in range(n_points - lag):
                    gradient[i] += 1 - h_current[i + lag]
                    if i + lag < n_points:
                        gradient[i + lag] -= h_current[i]
            else:
                for i in range(n_points + lag):
                    gradient[i] += 1 - h_current[i - lag]
                    if i - lag >= 0:
                        gradient[i - lag] -= h_current[i]
        
        h_new = h_current - step_size * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            step_size = min(0.02, step_size * 1.05)  # Increase step if improving
        else:
            step_size *= 0.5
            if step_size < 1e-10:
                break
                
        h_current = h_new
        iter_count += 1

    return (list(best_h), best_c5, n_points)
