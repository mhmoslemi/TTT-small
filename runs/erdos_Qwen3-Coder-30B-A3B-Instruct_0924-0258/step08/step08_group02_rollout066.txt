import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Hyperparameters
    max_iter = 500
    tol = 1e-6
    relax_weight = 0.1
    coord_frac = 0.2
    coord_steps = 50
    coord_batch = int(n_points * coord_frac)
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    no_improve_count = 0
    
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Convex relaxation step
        h_relaxed = h.copy()
        try:
            # Create the convex relaxation through variable substitution
            # x = h_i, we minimize sum_k (sum_i h_i(1-h_{i+k}))^2 + alpha*sum h_i^2
            # subject to sum h = n/2, 0 <= h <= 1
            # Solve using CVXPY
            
            # First isolate the correlation part
            # For simplicity, use small mini-batch coordinate descent
            
            # Sample a subset of indices for coordinate update
            indices = np.random.permutation(n_points)[:coord_batch]
            
            # Update for selected coordinates
            for _ in range(coord_steps):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Gradient estimation
                # We compute the gradient w.r.t. the selected indices
                overlaps = np.correlate(h, 1 - h, mode='full') * dx
                max_lag = np.argmax(overlaps)
                
                # Gradient for selected indices only
                grad = np.zeros_like(h)
                for i in indices:
                    # Need gradient of correlation term w.r.t. h[i]
                    j = (i + max_lag) % n_points
                    if j < len(h):
                        grad[i] = -2 * dx * (1 - h[j] - h[i])
                
                # Update
                h_new = h - 0.01 * grad
                # Project to [0, 1] bound
                h_new = np.clip(h_new, 0, 1)
                # Enforce sum
                h_new = h_new * (n_points/2) / np.sum(h_new)
                h = h_new.copy()
                
                # Update the best if improved
                current_c5 = evaluate_erdos_solution(h, dx)
                if current_c5 < best_c5 - tol:
                    best_c5 = current_c5
                    best_h = h.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1
                    
                if no_improve_count > 50:
                    break
                    
        except:
            # fallback to gradient based even if convex fails
            try:
                # Gradient based local optimization
                overlaps = np.correlate(h, 1 - h, mode='full') * dx
                max_lag = np.argmax(overlaps)
                
                # Gradient update
                grad = np.zeros_like(h)
                for i in range(n_points):
                    j = (i + max_lag) % n_points

                h = h - 0.01 * grad

                    no_improve_count = 0
                else:

                pass

            corr = np.correlate(h, 1 - h, mode='full')

                j = (i + max_lag) % len(h)

            objective,

            options={'maxiter': 100, 'ftol': 1e-9}

        c5_final = evaluate_erdos_solution(h_final, dx)
        
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    except:
        pass

    return (list(best_h), best_c5, n_points)
