import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    T0 = 0.05
    alpha = 0.999
    beta = 0.999
    T_min = 1e-6
    max_iter = 1000000
    
    T = T0
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and T > T_min and iter_count < max_iter:
        # Generate neighbor by perturbing with Gaussian noise
        delta = np.random.normal(0, 0.05 * dx, n_points)
        h_new = h_current + delta
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Calculate change in cost
        delta_c5 = c5_new - best_c5
        
        # Accept with probability based on SA criterion
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h_current = h_new.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        
        # Cooling and perturbation decay
        T *= alpha
        if iter_count % 100 == 0:
            # Slow down perturbation decay
            pass
        iter_count += 1
    
    return (list(best_h), best_c5, n_points)
