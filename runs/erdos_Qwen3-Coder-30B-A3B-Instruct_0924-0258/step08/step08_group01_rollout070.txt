import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale optimization
    max_scales = int(np.log2(n_points)) + 2
    scale = 1
    
    while scale <= max_scales and (time() - start_time) < budget_s * 0.95:
        current_n = min(2**scale, 1000)  # cap at reasonable size
        if current_n <= n_points:
            h_current = h_start.copy()
        else:
            # Interpolate/up-sample
            h_current = np.interp(np.linspace(0, 1, current_n), 
                                np.linspace(0, 1, n_points), h_start)
        
        # Add noise with decreasing magnitude
        noise_magnitude = 0.05 * (0.95 ** (scale - 1))
        keep_ratio = 0.7  # ratio to keep original values
        
        # Stochastic perturbation
        for _ in range(20):  # number of perturbation attempts
            if (time() - start_time) > budget_s * 0.95:
                break
                
            h_test = h_current.copy()
            indices = np.random.choice(current_n, 
                                     size=int(current_n * (1 - keep_ratio)), 
                                     replace=False)
            for i in indices:
                perturbation = noise_magnitude * (2 * np.random.random() - 1)
                h_test[i] += perturbation
                h_test[i] = np.clip(h_test[i], 0, 1)
            
            # Renormalize to maintain sum constraint
            h_test = h_test * (current_n / 2) / np.sum(h_test)
            
            c5_test = evaluate_erdos_solution(h_test, 2.0 / current_n)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
                noise_magnitude = 0.05 * (0.95 ** (scale - 1))  # reset noise
            else:
                noise_magnitude *= 0.9  # reduce noise
                
        # Local refinement with SQP
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / len(h))
        
        def constraint_sum(h):
            return np.sum(h) - len(h) / 2.0
        
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(len(best_h))]
            
            res = minimize(objective, best_h.copy(), method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 500, 'ftol': 1e-10})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
            
        scale += 1
        
    # Final refinement - but only if we're close to the time limit
    if (time() - start_time) < budget_s * 0.8:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, 2.0 / len(h))
            
            def constraint_sum(h):
                return np.sum(h) - len(h) / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(len(best_h))]
            
            res = minimize(objective, best_h.copy(), method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 1000, 'ftol': 1e-12})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, len(best_h))
