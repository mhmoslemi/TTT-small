import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm approach
    population_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    stagnation_limit = 100
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    # If we have initial solution, use it
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        population.append(create_individual())
    
    best_individual = best_h.copy()
    best_fitness = best_c5
    stagnation_count = 0
    
    # Fitness evaluation for entire population
    def evaluate_population(pop):
        fitnesses = []
        for h in pop:
            fit = evaluate_erdos_solution(h, dx)
            fitnesses.append(fit)
        return np.array(fitnesses)
    
    # Tournament selection
    def tournament_select(pop, fitnesses, k=3):
        idxs = np.random.choice(len(pop), k)
        best_idx = idxs[np.argmin(fitnesses[idxs])]
        return pop[best_idx].copy()
    
    # Blend crossover
    def blend_crossover(p1, p2, alpha=0.3):
        child = np.zeros_like(p1)
        for i in range(len(p1)):
            if np.random.rand() < crossover_rate:
                a, b = min(p1[i], p2[i]), max(p1[i], p2[i])
                range_ = (b - a) * alpha
                child[i] = np.clip(np.random.uniform(a - range_, b + range_), 0, 1)
            else:
                child[i] = p1[i]
        # Renormalize
        child = child * (n_points / 2) / np.sum(child)
        return np.clip(child, 0, 1)
    
    # Gaussian mutation
    def mutate(individual, mutation_strength):
        mutated = individual.copy()
        for i in range(len(mutated)):
            if np.random.rand() < mutation_rate:
                mutated[i] += np.random.normal(0, mutation_strength)
        # Renormalize
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return np.clip(mutated, 0, 1)
    
    # Main GA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = evaluate_population(population)
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > stagnation_limit:
            break
            
        # Selection, crossover, mutation
        new_population = [best_individual.copy()]
        while len(new_population) < population_size:
            p1 = tournament_select(population, fitnesses)
            p2 = tournament_select(population, fitnesses)
            child = blend_crossover(p1, p2)
            child = mutate(child, 0.05 * (1 - gen / generations))
            new_population.append(child)
        
        population = new_population
        
        # Occasionally refine best with SLSQP
        if gen % 50 == 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)

            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0

            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_individual, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 500, 'ftol': 1e-10})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_fitness:
                        best_individual = h_refined
                        best_fitness = c5_refined
            except Exception:
                pass

    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)

        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0

        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_individual, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            best_individual = res.x
            best_fitness = objective(best_individual)
    except Exception:
        pass
    
    return (list(best_individual), best_fitness, n_points)
