import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Iterative convex optimization with filtering
    max_iterations = 1000
    filter_bandwidth = 5
    top_k = 10
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Build convex optimization model for top K overlaps
        # Let's compute all overlaps and pick top k lags
        corr = np.correlate(h, 1 - h, mode='full')
        lag_indices = np.argsort(corr)[::-1][:top_k]
        
        # Define optimization variables
        h_var = cp.Variable(n_points)
        
        # Constraint: h in [0,1]
        constraints = [h_var >= 0, h_var <= 1]
        
        # Constraint: integral = 1
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Objective: maximize sum of top K overlaps
        objective = 0
        for i in range(top_k):
            lag_idx = lag_indices[i]
            lag = lag_idx - (n_points - 1)
            # We need to translate lags to matrix indexing for convolution
            # For each lag, we set up the corresponding overlap term
            # This is a complex constraint, so we'll linearly approximate it
            # by summing up products over valid pairs
            if lag >= 0:
                obj_terms = [h_var[j] * (1 - h_var[j + lag]) for j in range(n_points - lag)]
            else:
                obj_terms = [h_var[j - lag] * (1 - h_var[j]) for j in range(-lag, n_points)]
            objective += cp.sum(obj_terms)
        
        # Minimize the inverse of the max overlap
        # Can't directly minimize max overlap easily in CVX
        # But we can parameterize the most critical one
        # So we'll still do iterative gradient technique but convolve with fewer points
        
        # Let's switch to optimized approach using orthogonal projection
        # Define the problem: minimize max overlap
        # We'll do block coordinate descent
        # But simpler approach - only optimize the estimated max overlap
        
        # Better approach:
        # We introduce a variable for max overlap value
        # Minimize this value subject to overlap < value for all lag
        # But still not convex, so we linearize
        
        # For direct gradient-based refinement, keep the simple method but make 
        # it work better
        
        # Apply gradient-like descent direct approach but refined:
        # Compute all overlaps, find max, get contribution to gradient
        
        # Let's do optimized version: use a more scalable approach
        # we'll purposefully keep some entropy in the solution, but also be systematic
        
        corr = np.correlate(h, 1-h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient of the max overlap w.r.t h
        # this requires a bit of care
        h_grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                h_grad[i] = (1 - 2 * h[i]) - (1 - 2 * h[j])
            else:
                h_grad[i] = 1 - 2 * h[i]
        
        # Update selection (SGD style)
        step_size = 0.005 / (1.0 + iteration / 100.0)  # decay learning rate
        h_new = h - step_size * h_grad
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate and accept if better
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:

            pass
            
        # Apply smoothing filter periodically to prevent over-fitting to local maxima
        if iteration % 10 == 0:
            # Simple moving average smoothing with local bandwidth
            smoothed = np.convolve(h, np.ones(filter_bandwidth) / filter_bandwidth, mode='same')
            h = np.clip(smoothed, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Increase top_k for later iterations to explore more closely
        if iteration % 20 == 0 and top_k < 15:
            top_k += 1

    # Final evaluation and return
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
