import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        # Create a better initial construction
        h_start[:n_points//2] = 1.0
        h_start[n_points//2:] = 0.0
        # Add some smooth transition
        h_start = np.clip(h_start + np.random.normal(0, 0.05, n_points), 0, 1)
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Trust-region optimized approach
    max_iter = 5000
    trust_radius = 0.2
    min_trust_radius = 1e-6
    trust_radius_factor = 1.2
    
    # Compute one-time overlap structure
    full_corr = np.correlate(h_start, 1 - h_start, mode='full')
    lags = np.arange(len(full_corr)) - (len(h_start) - 1)
    max_overlaps = full_corr * dx
    
    # Helper to compute overlap for given h and lag
    def get_overlap_at_lag(h, lag):
        if lag >= 0:
            return np.sum(h[:-lag] * (1 - h[lag:])) * dx
        else:
            return np.sum(h[-lag:] * (1 - h[:-lag])) * dx
            
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Identify worst lag (highest overlap)
        max_idx = np.argmax(max_overlaps)
        target_lag = lags[max_idx]
        
        # Calculate gradient based on this lag
        grad = np.zeros_like(h_current)
        if target_lag >= 0:
            # h[i] contributes to overlap when multiplied by 1-h[i+lag]
            # gradient of overlap w.r.t. h[i] = -(1-h[i+lag])  if i < n-lag
            for i in range(len(h_current) - target_lag):
                grad[i] -= (1 - h_current[i + target_lag])
        else:
            # Similar logic
            for i in range(-target_lag, len(h_current)):
                grad[i] -= (1 - h_current[i + target_lag])
                
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
            
        # Apply trust region constraint
        step = -0.01 * grad  # Learning rate
        
        # Clip step to trust region
        step_norm = np.linalg.norm(step)
        if step_norm > trust_radius:
            step = step * trust_radius / step_norm
            
        # Compute new solution
        h_new = h_current + step
        
        # Project onto bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto sum constraint
        sum_h = np.sum(h_new)
        projection = (n_points/2 - sum_h) / len(h_new)
        h_new = h_new + projection
        
        # Normalize again
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            # Increase trust region
            trust_radius = min(trust_radius * trust_radius_factor, 0.5)
        else:
            # Decrease trust region
            trust_radius = max(trust_radius / trust_radius_factor, min_trust_radius)
            
        h_current = h_new.copy()
        
        # Every 200 iterations, try SLSQP refinement
        if iteration % 200 == 0 and iteration > 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
                
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 500, 'ftol': 1e-10})
                if res.success:
                    h_final = res.x
                    c5_final = objective(h_final)
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
            except Exception:
                pass
    
    # Final refinement using SLSQP with all constraints if time permits
    if time() - start_time < budget_s * 0.9:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
                
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
