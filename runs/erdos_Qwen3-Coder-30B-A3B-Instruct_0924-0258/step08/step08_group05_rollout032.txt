import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        # Ensure sum is exactly n_points/2
        h_start[0] = 0.5  # adjust one element to make sum correct
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    
    # Initialize best solution
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    initial_temp = 1.0
    cooling_rate = 0.9995
    min_temp = 1e-6
    max_iter = 1000000  # Will be limited by time
    
    # Temperature and iteration tracking
    T = initial_temp
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 2000
    
    while time() - start_time < budget_s * 0.95 and T > min_temp and no_improvement_count < max_no_improvement:
        # Generate new candidate solution by swapping values between two indices
        i, j = np.random.randint(0, n_points, 2)
        while i == j:
            j = np.random.randint(0, n_points)
            
        # Compute feasible delta
        max_delta = min(h_current[i], 1 - h_current[j])
        if max_delta <= 0:
            iteration += 1
            continue
            
        delta = np.random.uniform(0, max_delta)
        
        # Create new solution
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize to maintain sum constraint
        sum_h = np.sum(h_new)
        h_new = h_new * (n_points/2) / sum_h
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on simulated annealing criteria
        delta_e = c5_new - best_c5
        
        if delta_e < 0 or np.random.rand() < np.exp(-delta_e / T):
            h_current = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_current.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Cool down temperature
        T *= cooling_rate
        iteration += 1
        
        # Occasionally re-normalize to keep floating-point errors in check
        if iteration % 1000 == 0:
            sum_h = np.sum(h_current)
            h_current = h_current * (n_points/2) / sum_h
    
    # Final verification
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
