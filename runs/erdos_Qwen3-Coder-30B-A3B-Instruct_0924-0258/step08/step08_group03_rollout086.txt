import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region with sum = n_points/2 and bounds [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return np.clip(h, 0, 1)

    best_c5 = evaluate_erdos_solution(h_fine, dx)

    n_coarse = 15
    h_coarse = np.ones(n_coarse) * 0.5
    coarse_dx = 2.0 / n_coarse
    
    # Initialize coarse grid better
    if 'initial_h_values' in globals():

        coarse_x = np.linspace(0, 2, n_coarse)
        h_coarse = f_interp(coarse_x)
    else:
        h_coarse = np.ones(n_coarse) * 0.5
        
    # Coarse grid optimization
    for coarse_iter in range(500):
        if time() - start_time > budget_s * 0.8:

        # Compute overlaps for coarse h
        corr_coarse = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_lag_idx_coarse = np.argmax(corr_coarse)
        max_lag_coarse = max_lag_idx_coarse - (n_coarse - 1)
        
        # Compute gradient for coarse
        grad_coarse = np.zeros(n_coarse)
        for i in range(n_coarse):
            j = i + max_lag_coarse
            if 0 <= j < n_coarse:
                grad_coarse[i] = 1 - 2 * h_coarse[i] - h_coarse[j]
            else:
                grad_coarse[i] = 1 - 2 * h_coarse[i]
        
        # Gradient step
        h_coarse_new = h_coarse - 0.01 * grad_coarse
        h_coarse_new = project_feasible(h_coarse_new, n_coarse)
        
        # Evaluate coarse solution
        c5_coarse_new = evaluate_erdos_solution(h_coarse_new, coarse_dx)
        
        if c5_coarse_new < best_c5:
            best_c5 = c5_coarse_new
            # Update fine grid based on coarse
            f_interp = interp1d(np.linspace(0, 2, n_coarse), h_coarse_new, kind='linear', fill_value='extrapolate')
            x_fine = np.linspace(0, 2, n_points)
            h_fine = f_interp(x_fine)
            h_fine = project_feasible(h_fine, n_points)

    # Fine grid optimization with alternating gradient and mutation

                step_size *= 0.9
    
    return (list(best_h), best_c5, n_points)
