import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    alpha = 0.99
    T_min = 1e-5
    max_iter_sa = int(1e5)
    step_size = 0.1
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    T = T_init
    iter_count = 0
    
    # Simulated Annealing loop
    while T > T_min and iter_count < max_iter_sa and (time() - start_time) < budget_s * 0.9:
        iter_count += 1
        
        # Perturb two random indices
        i, j = np.random.randint(0, n_points, 2)
        delta = np.random.uniform(-step_size, step_size)
        
        # Store old values
        old_i, old_j = h[i], h[j]
        
        # Apply perturbation
        h[i] = old_i + delta
        h[j] = old_j - delta
        
        # Enforce bounds
        h[i] = np.clip(h[i], 0, 1)
        h[j] = np.clip(h[j], 0, 1)
        
        # Adjust to maintain sum
        sum_diff = np.sum(h) - n_points / 2
        h[0] -= sum_diff
        
        # Ensure bounds after adjustment
        h = np.clip(h, 0, 1)
        
        # Compute new C5
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Compute delta
        delta_c5 = new_c5 - best_c5
        
        # Accept or reject
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Revert changes
            h[i] = old_i
            h[j] = old_j
        
        # Update temperature
        T *= alpha
        
        # Adapt step size
        if iter_count % 1000 == 0:
            step_size = max(0.001, step_size * 0.99)
    
    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
