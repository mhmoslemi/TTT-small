import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use binary pattern to reduce overlap
        h_start = np.zeros(n_points)
        h_start[0:n_points//2] = 1.0
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Convex relaxation approach with McCormick envelopes
    max_iter = 5000
    tolerance = 1e-8
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Solve relaxed version
        try:
            # Set up the relaxed problem
            h_vars = cp.Variable(n_points)
            C = cp.Variable()
            
            # Sum constraint
            sum_constraint = cp.sum(h_vars) == n_points / 2
            
            # McCormick envelopes and overlap constraints
            constraints = [h_vars >= 0, h_vars <= 1, sum_constraint, C >= 0]
            
            # For each lag, add overlap constraints
            max_lag = n_points - 1
            for k in range(-max_lag, max_lag + 1):
                if k >= 0:
                    # h[i] * (1 - h[i + k])
                    overlap_sum = 0
                    for i in range(n_points - k):
                        # z[i][i + k] = h[i] * h[i + k]
                        z = cp.Variable()
                        # McCormick envelopes:
                        # h[i] + h[i+k] - 1 <= z <= h[i], z <= h[i+k], z >= 0
                        constraints.append(z <= h_vars[i])
                        constraints.append(z <= h_vars[i+k])
                        constraints.append(z >= 0)
                        constraints.append(z >= h_vars[i] + h_vars[i+k] - 1)
                        
                        overlap_sum += h_vars[i] - z
                
                else:
                    # h[i + |k|] * (1 - h[i])
                    overlap_sum = 0
                    k_abs = -k
                    for i in range(n_points - k_abs):
                        # z[i][i + k] = h[i] * h[i + k]
                        z = cp.Variable()
                        # McCormick envelopes
                        constraints.append(z <= h_vars[i])
                        constraints.append(z <= h_vars[i+k_abs])
                        constraints.append(z >= 0)
                        constraints.append(z >= h_vars[i] + h_vars[i+k_abs] - 1)
                        
                        overlap_sum += h_vars[i] - z
                        
                # Set up overlap constraint
                constraints.append(overlap_sum <= C)
            
            # Objective
            prob = cp.Problem(cp.Minimize(C), constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_relaxed = h_vars.value
                if h_relaxed is not None:
                    # Clamp values to [0,1]
                    h_new = np.clip(h_relaxed, 0, 1)
                    # Re-normalize
                    h_new = h_new * (n_points/2) / np.sum(h_new)
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        last_improvement = iteration
                        
        except Exception as e:
            pass
            
        # Local refinement with SLSQP
        if iteration % 5 == 0 and iteration > 0:
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                    
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                # Use best_h found so far as initial solution
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-12})
                if res.success:
                    h_final = res.x
                    c5_final = objective(h_final)
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
                        last_improvement = iteration
            except Exception:
                pass
                
        # Break early if no improvement for a long time
        if iteration - last_improvement > 500:
            break
    
    # Final refinement with SLSQP using direct optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
