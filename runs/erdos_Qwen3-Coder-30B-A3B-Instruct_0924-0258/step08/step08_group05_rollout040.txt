import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 35
    pop_size = 50
    max_generations = 10000
    mutation_rate = 0.1
    tournament_size = 5
    elite_size = 1
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h_start = h_start * (n_points/2) / np.sum(h_start)
    population.append(h_start.copy())
    
    # Fill rest of population
    for i in range(pop_size - 1):
        # Start with a simple construction
        h = np.zeros(n_points)
        # Randomly place 1s for even distribution
        if n_points % 2 == 0:
            # Even case: put 1s at alternating positions
            for j in range(n_points // 2):
                h[2*j] = 1.0
        else:
            # Odd case: put 1s at alternating positions, one middle element at 0.5
            for j in range((n_points - 1) // 2):
                h[2*j] = 1.0
            h[(n_points - 1) // 2] = 0.5
        # Random shuffling to introduce diversity
        np.random.shuffle(h)
        # Normalize sum to match constraint
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
    
    dx = 2.0 / n_points
    
    # Evaluation
    def evaluate_individual(h):
        return evaluate_erdos_solution(h, dx)
    
    # Tournament selection
    def tournament_select(pop, fitnesses):
        indices = np.random.choice(len(pop), size=tournament_size)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        return pop[best_idx].copy()
    
    # Crossover
    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(len(child)):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        # Renormalize sum
        child = child * (n_points/2) / np.sum(child)
        return child
    
    # Mutation
    def mutate(individual):
        child = individual.copy()
        # Flip mutation
        if np.random.rand() < 0.5:
            i1, i2 = np.random.choice(len(child), size=2, replace=False)
            child[i1] = max(0, min(1, child[i1] + 0.1))
            child[i2] = max(0, min(1, child[i2] - 0.1))
        # Perturb mutation
        else:
            num_to_perturb = int(len(child) * mutation_rate)
            indices = np.random.choice(len(child), size=num_to_perturb, replace=False)
            for i in indices:
                perturbation = np.random.uniform(0.9, 1.1)
                child[i] = max(0, min(1, child[i] * perturbation))
        # Renormalize
        child = child * (n_points/2) / np.sum(child)
        return child
    
    # Initial evaluation of population
    fitnesses = [evaluate_individual(ind) for ind in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Main evolution loop
    no_improvement_count = 0
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Store current best
        if sorted_fitnesses[0] < best_c5:
            best_c5 = sorted_fitnesses[0]
            best_h = sorted_pop[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Stop if no improvement for many generations
        if no_improvement_count >= 100:
            break
        
        # Create new population with elitism
        new_pop = sorted_pop[:elite_size].copy()
        new_fitnesses = sorted_fitnesses[:elite_size].copy()
        
        # Generate rest using tournament selection, crossover, and mutation
        while len(new_pop) < pop_size:
            parent1 = tournament_select(sorted_pop, sorted_fitnesses)
            parent2 = tournament_select(sorted_pop, sorted_fitnesses)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_pop.append(child)
            new_fitnesses.append(evaluate_individual(child))
            
        population = new_pop
        fitnesses = new_fitnesses
    
    # Final refinement using optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
