import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    population_size = 50
    n_points = 35
    dx = 2.0 / n_points
    max_generations_no_improvement = 100
    mutation_rate = 0.1
    
    # Initialize population
    population = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(n_points) * 0.5
    
    # Normalize initial solution
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    population.append(h_start.copy())
    
    # Fill rest of population
    for _ in range(population_size - 1):
        # Use heuristic: mix of uniform random and binary construction
        h = np.zeros(n_points)
        ones_count = n_points // 2
        h[:ones_count] = 1.0
        np.random.shuffle(h)
        # Add some randomness
        for i in range(n_points):
            if random.random() < 0.1:
                h[i] = random.random()
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fit = evaluate_erdos_solution(h, dx)
        fitness.append(fit)
    
    # Track best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    generation = 0
    no_improvement = 0
    
    # Main GA loop
    while time() - start_time < budget_s * 0.95 and no_improvement < max_generations_no_improvement:
        generation += 1
        
        # Tournament selection
        selected = []
        for _ in range(population_size):
            # Tournament size 5
            tournament_indices = np.random.choice(population_size, 5, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create new population through crossover and mutation
        new_population = []
        new_fitness = []
        
        # Elitism: keep best solution
        new_population.append(best_h.copy())
        new_fitness.append(best_c5)
        
        # Generate rest of population
        while len(new_population) < population_size:
            # Select two parents
            parent1 = selected[np.random.randint(0, population_size)]
            parent2 = selected[np.random.randint(0, population_size)]
            
            # Uniform crossover
            child = np.zeros(n_points)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    delta = (random.random() - 0.5) * 0.2
                    child[i] += delta
                    child[i] = max(0, min(1, child[i]))
            
            # Fix sum constraint
            child_sum = np.sum(child)
            if abs(child_sum - n_points / 2) > 1e-10:
                # Scale to meet constraint
                child = child * (n_points / 2) / child_sum
            
            # Add to new population
            new_population.append(child)
            new_fitness.append(evaluate_erdos_solution(child, dx))
        
        # Replace old population
        population = new_population
        fitness = new_fitness
        
        # Update best if improved
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
    
    # Final refinement using local search
    try:
        # Sample a few good individuals for refinement
        top_candidates = np.argsort(fitness)[:5]
        for candidate_idx in top_candidates:
            candidate = population[candidate_idx]
            # Simple hill climbing with small perturbations
            for _ in range(100):
                if time() - start_time > budget_s * 0.99:
                    break
                h_new = candidate.copy()
                # Perturb randomly
                idx1, idx2 = np.random.randint(0, n_points, 2)
                delta = (random.random() - 0.5) * 0.05
                h_new[idx1] += delta
                h_new[idx2] -= delta
                h_new[idx1] = max(0, min(1, h_new[idx1]))
                h_new[idx2] = max(0, min(1, h_new[idx2]))
                # Renormalize
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
