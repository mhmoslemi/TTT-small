import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = np.max(np.correlate(h_current, 1 - h_current, mode='full')) * dx
    
    # Precompute all lags
    lags = np.arange(-n_points + 1, n_points)
    
    # Initialize trust region parameters
    trust_radius = 0.2
    max_trust_radius = 0.5
    min_trust_radius = 1e-6
    
    # Trust region loop
    trust_iter = 0
    max_trust_iter = 300
    improvement_threshold = 1e-8
    
    while time() - start_time < budget_s * 0.7 and trust_iter < max_trust_iter:
        # Compute all overlaps
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        overlap_values = corr * dx
        max_lag_idx = np.argmax(overlap_values)
        max_lag = lags[max_lag_idx]
        
        # Perform local search in trust region using cutting planes
        h_new = h_current.copy()
        improvement = 0
        
        # Simple gradient-based search within trust region
        for _ in range(50):
            if time() - start_time > budget_s * 0.65:
                break
                
            # Compute gradient for max overlap lag
            grad = np.zeros_like(h_current)
            if max_lag >= 0:
                for i in range(len(h_current) - max_lag):
                    grad[i] -= 1.0
            else:
                for i in range(-max_lag, len(h_current)):
                    grad[i] -= 1.0
            
            # Gradient step with trust region constraint
            step = -0.01 * grad
            step_norm = np.linalg.norm(step)
            if step_norm > trust_radius:
                step = step * trust_radius / step_norm
            
            # Update solution
            h_candidate = h_current + step
            h_candidate = np.clip(h_candidate, 0, 1)
            sum_h = np.sum(h_candidate)
            h_candidate = h_candidate * (n_points/2) / sum_h
            
            # Evaluate improvement
            c5_candidate = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
            if c5_candidate < best_c5:
                h_current = h_candidate
                best_c5 = c5_candidate
                improvement = 1
            
            # Adjust trust region
            if improvement == 0:
                trust_radius = max(min_trust_radius, trust_radius * 0.8)
            else:
                trust_radius = min(max_trust_radius, trust_radius * 1.2)
                improvement = 0
                
        trust_iter += 1
        
        # Stop if improvement is minimal
        if trust_iter > 20 and abs(best_c5 - np.max(np.correlate(h_current, 1 - h_current, mode='full')) * dx) < improvement_threshold:
            break

    # Final refinement
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    try:
        res = minimize(objective, h_current, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    # Final validation
    h_final = np.clip(best_h, 0, 1)
    sum_h = np.sum(h_final)
    h_final = h_final * (n_points/2) / sum_h
    
    c5_final = np.max(np.correlate(h_final, 1 - h_final, mode='full')) * dx
    
    return (list(h_final), c5_final, n_points)
