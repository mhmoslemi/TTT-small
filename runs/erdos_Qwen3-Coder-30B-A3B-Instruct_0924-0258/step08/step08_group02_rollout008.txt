import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter_sa = 100000
    
    # Initial temperature and iteration counter
    T = T_init
    iteration = 0
    
    # Make a copy of the initial solution
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Keep track of best solution found during SA
    best_h_sa = h.copy()
    best_c5_sa = evaluate_erdos_solution(h, dx)
    
    # Annealing loop
    while T > T_min and time() - start_time < budget_s * 0.9:
        if iteration >= max_iter_sa:
            break
            
        # Perturb solution: select two random indices
        i, j = random.sample(range(n_points), 2)
        delta = np.random.uniform(-0.1, 0.1)
        
        # Adjust the values of h[i] and h[j]
        h_new = h.copy()
        h_new[i] = h[i] + delta
        h_new[j] = h[j] - delta
        
        # Enforce bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint: scale to match sum = n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new overlap
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject the change
        delta_c5 = c5_new - best_c5_sa
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new
            if c5_new < best_c5_sa:
                best_c5_sa = c5_new
                best_h_sa = h.copy()
        
        # Update temperature
        T *= alpha
        iteration += 1
        
        # Check if we've found a better solution so far
        if best_c5_sa < best_c5:
            best_c5 = best_c5_sa
            best_h = best_h_sa.copy()
    
    # Local refinement with L-BFGS-B
    try:
        def objective(h):
            return np.max(np.correlate(h, 1 - h, mode='full') * dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
