import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.999
    min_T = 1e-5
    max_iterations = int(1e6)
    no_improvement_count = 0
    max_no_improvement = 1000
    
    # Initial energy calculation
    current_energy = evaluate_erdos_solution(h_current, dx)
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor by swapping small delta between two indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(0.01, 0.1)
        
        # Ensure we don't violate bounds
        max_delta = min(delta, h_current[i], 1 - h_current[j])
        if max_delta <= 0:
            continue
            
        # Apply changes
        h_new = h_current.copy()
        h_new[i] += max_delta
        h_new[j] -= max_delta
        
        # Ensure bounds are respected (should be redundant but safety)
        h_new[i] = np.clip(h_new[i], 0, 1)
        h_new[j] = np.clip(h_new[j], 0, 1)
        
        # Renormalize to maintain sum constraint
        sum_h = np.sum(h_new)
        h_new = h_new * (n_points/2) / sum_h
        
        # Calculate new energy
        new_energy = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on simulated annealing criterion
        if new_energy < current_energy:
            h_current = h_new
            current_energy = new_energy
            if new_energy < best_c5:
                best_c5 = new_energy
                best_h = h_new.copy()
            no_improvement_count = 0
        else:
            # Accept with probability
            if np.random.random() < np.exp(-(new_energy - current_energy) / T):
                h_current = h_new
                current_energy = new_energy
            no_improvement_count += 1
            
        # Cooling schedule
        if T > min_T:
            T *= cooling_rate
            
        # Early stopping if no improvement for many iterations
        if no_improvement_count > max_no_improvement:
            break
            
    # Final refinement with local search
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Try one more local optimization if progress was made
    if time() - start_time < budget_s * 0.9:
        from scipy.optimize import minimize
        
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
