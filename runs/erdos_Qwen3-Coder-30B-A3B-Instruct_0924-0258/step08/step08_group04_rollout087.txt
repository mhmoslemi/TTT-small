import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # PSO parameters
    num_particles = 75
    inertia = 0.8
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    max_iter = int(budget_s * 0.9 * 1000)  # Rough estimate
    perturbation_rate = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    particle_best = []
    particle_best_fitness = []
    
    for _ in range(num_particles):
        h = h_current.copy()
        # Add small random perturbation in feasible region
        noise = np.random.normal(0, 0.05, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        particle_best.append(h.copy())
        particle_best_fitness.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    global_best_idx = np.argmin(particle_best_fitness)
    global_best = particle_best[global_best_idx].copy()
    global_best_fitness = particle_best_fitness[global_best_idx]
    
    # PSO iterations
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Inertia decay
        current_inertia = inertia * (1.0 - iter_count / max_iter)
        
        for i in range(num_particles):
            # Evaluate current particle
            fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if fitness < particle_best_fitness[i]:
                particle_best[i] = particles[i].copy()
                particle_best_fitness[i] = fitness
            
            # Update global best
            if fitness < global_best_fitness:
                global_best = particles[i].copy()
                global_best_fitness = fitness
            
            # Velocity update
            r1, r2 = np.random.random(2)
            cognitive = cognitive_weight * r1 * (particle_best[i] - particles[i])
            social = social_weight * r2 * (global_best - particles[i])
            velocities[i] = (current_inertia * velocities[i] + 
                             cognitive + social)
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible set
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
        # Periodic perturbation to escape local minima
        if iter_count % 1000 == 0 and iter_count > 0:
            for i in range(num_particles):
                if np.random.random() < perturbation_rate:
                    noise = np.random.normal(0, 0.05, n_points)
                    particles[i] = particles[i] + noise
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
    
    # Final refinement with local search
    best_h = global_best.copy()
    best_c5 = global_best_fitness
    
    # Try slight local search around the best solution
    for _ in range(1000):
        if time() - start_time > budget_s * 0.99:
            break
            
        # Small perturbation with step size adaptation
        h_new = best_h + np.random.normal(0, 0.01, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    return (list(best_h), best_c5, n_points)
