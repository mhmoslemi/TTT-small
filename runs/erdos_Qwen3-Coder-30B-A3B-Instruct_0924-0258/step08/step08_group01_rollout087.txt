import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def create_feasible_h(n_points):
    # Create random feasible h with sum = n_points/2
    h = np.random.rand(n_points)
    h = h * (n_points / 2) / np.sum(h)
    return np.clip(h, 0, 1)

def ga_search(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = create_feasible_h(n_points)
    
    # GA parameters
    population_size = 50
    generations = 5000
    crossover_rate = 0.8
    mutation_rate = 0.1
    mutation_step = 0.05
    slsqp_freq = 100
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        population.append(create_feasible_h(n_points))
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / n_points)
    
    no_improvement_count = 0
    max_no_improvement = 200
    
    # Precompute dx
    dx = 2.0 / n_points
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness of population
        fitnesses = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitnesses.append(c5)
            
        # Tournament selection
        new_population = []
        for _ in range(population_size // 2):
            # Select 2 parents
            idx1, idx2 = random.sample(range(population_size), 2)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Tournament
            if fitnesses[idx1] < fitnesses[idx2]:
                winner = parent1
            else:
                winner = parent2
                
            new_population.append(winner.copy())
            
        # Generate offspring
        offspring = []
        for _ in range(population_size // 2):
            if random.random() < crossover_rate:
                # Blend crossover
                parent1 = random.choice(new_population)
                parent2 = random.choice(new_population)
                alpha = np.random.uniform(0.3, 0.7)
                child = alpha * parent1 + (1 - alpha) * parent2
                # Apply mutation
                if random.random() < mutation_rate:
                    noise = np.random.normal(0, mutation_step, n_points)
                    child += noise
                    child = np.clip(child, 0, 1)
                    # Fix sum
                    child = child * (n_points / 2) / np.sum(child)
                offspring.append(child)
            else:
                # Clone
                parent = random.choice(new_population)
                if random.random() < mutation_rate:
                    noise = np.random.normal(0, mutation_step, n_points)
                    parent += noise
                    parent = np.clip(parent, 0, 1)
                    parent = parent * (n_points / 2) / np.sum(parent)
                offspring.append(parent.copy())
                
        # Replace population
        population = new_population + offspring
        
        # Track best solution
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
        # Periodic SLSQP refinement
        if gen % slsqp_freq == 0 and gen > 0:
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 500, 'ftol': 1e-12})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
                
        # Reset mutation rate periodically
        if gen % 50 == 0 and mutation_step > 0.001:
            mutation_step *= 0.98
            
        if no_improvement_count > max_no_improvement:
            break
    
    # Final local refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return ga_search(seed, budget_s)
