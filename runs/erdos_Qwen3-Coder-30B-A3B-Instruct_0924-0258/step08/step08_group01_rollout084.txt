import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    crossover_rate = 0.8
    mutation_rate = 0.1
    generations = 1000
    no_improvement_limit = 100
    refine_period = 100
    
    # Initialize population
    population = []
    fitnesses = []
    
    # Add the initial solution
    population.append(h_start.copy())
    fitnesses.append(evaluate_erdos_solution(h_start, dx))
    
    # Add random valid solutions
    for i in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
        fitnesses.append(evaluate_erdos_solution(h, dx))
    
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    no_improvement_count = 0
    generation = 0
    
    while time() - start_time < budget_s * 0.7 and generation < generations:
        # Evaluate current generation
        for i in range(population_size):
            if i == 0 or i >= len(fitnesses):
                fitnesses[i] = evaluate_erdos_solution(population[i], dx)
        
        # Track best solution in current generation
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Early termination check
        if no_improvement_count >= no_improvement_limit:
            break
            
        # Tournament selection
        def tournament_select():
            k = 3
            idxs = np.random.choice(population_size, k)
            best_idx_in_tourn = idxs[np.argmin([fitnesses[i] for i in idxs])]
            return population[best_idx_in_tourn].copy()
        
        # Create new population
        new_population = [best_h.copy()]  # Elitism
        
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < crossover_rate:
                parent1 = tournament_select()
                parent2 = tournament_select()
                
                # Blend crossover
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
                
                # Adjust to maintain constraints
                child_sum = np.sum(child)
                child = child * (n_points / 2) / child_sum
                child = np.clip(child, 0, 1)
                
                # Mutation
                if np.random.rand() < mutation_rate:
                    # Random mutation
                    mutate_idx = np.random.randint(0, n_points)
                    delta = np.random.normal(0, 0.05)
                    child[mutate_idx] += delta
                    # Re-adjust sum
                    child_sum = np.sum(child)
                    child = child * (n_points / 2) / child_sum
                    child = np.clip(child, 0, 1)
                
                new_population.append(child)
            else:
                # Copy best and mutate
                parent = tournament_select()
                child = parent.copy()
                if np.random.rand() < mutation_rate:
                    mutate_idx = np.random.randint(0, n_points)
                    delta = np.random.normal(0, 0.05)
                    child[mutate_idx] += delta
                    # Re-adjust sum
                    child_sum = np.sum(child)
                    child = child * (n_points / 2) / child_sum
                    child = np.clip(child, 0, 1)
                new_population.append(child)
        
        population = new_population
        
        if generation % refine_period == 0 and time() - start_time < budget_s * 0.9:
            # Local refinement
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            for i in range(min(5, len(population))):
                try:
                    res = minimize(objective, population[i], method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 100, 'ftol': 1e-10})
                    if res.success:
                        refined = res.x
                        refined_fitness = objective(refined)
                        if refined_fitness < fitnesses[i]:
                            population[i] = refined
                            fitnesses[i] = refined_fitness
                except Exception:
                    pass
        
        generation += 1
    
    # Final refinement of best solution
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
