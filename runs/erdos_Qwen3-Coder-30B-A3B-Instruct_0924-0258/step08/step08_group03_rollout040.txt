import numpy as np
from time import time
import cvxpy as cp
from scipy.linalg import svd

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def solve_sdp(h0, dx, C5_val, n_points):
    # SDP relaxation approach
    Y = cp.Variable((n_points, n_points), symmetric=True)
    h = cp.Variable(n_points)
    
    # Constraints
    constraints = [
        Y >> 0,  # Y is positive semidefinite
        cp.diag(Y) == h,  # diagonal of Y equals h
        cp.sum(h) == n_points / 2,  # normalization
    ]
    
    # Add overlap constraints for sampled shifts
    shifts = np.random.choice(n_points, size=min(100, n_points), replace=False)
    for k in shifts:
        if k == 0:
            continue
        # Sum over all valid indices (shifting by k)
        expr = 0
        for i in range(n_points):
            j = i + k
            if j < n_points:
                expr += Y[i, j]
        constraints.append(cp.sum(h) - expr <= C5_val * dx)
    
    # Objective (minimize C5)
    objective = cp.Minimize(C5_val)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_sol = h.value
            if h_sol is not None and np.all(h_sol >= 0) and np.all(h_sol <= 1):
                return h_sol
    except:
        pass
    return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

    else:

        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search on C5
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    while time() - start_time < budget_s * 0.98 and (high - low) > tolerance:
        mid = (low + high) / 2
        
        h_tmp = solve_sdp(h, dx, mid, n_points)
        if h_tmp is not None:
            # Project back and refine
            h_projected = np.clip(h_tmp, 0, 1)
            h_projected = h_projected * (n_points / 2) / np.sum(h_projected)
            c5_estimate = evaluate_erdos_solution(h_projected, dx)
            if c5_estimate < best_c5:
                best_c5 = c5_estimate
                best_h = h_projected.copy()
                h = h_projected.copy()
            # Tighten upper bound
            high = mid
        else:
            # Widen lower bound
            low = mid
    
    # Local refinement
    max_iterations = 1000
    step_size = 0.005
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.99:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient for the max lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h[i] - h[j]
            else:
                grad[i] = 1 - 2 * h[i]
        
        # Gradient descent step
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to maintain constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            step_size *= 0.99  # Reduce step size
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
