import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0<=h<=1 and sum(h)=n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def convex_perturbation_optimization(h, n_points, dx, max_iter=5000):
    epsilon_0 = 0.001
    beta = 0.99
    top_k = 100
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    for t in range(max_iter):
        # Compute overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        top_shifts = np.argsort(corr)[-top_k:]
        
        # Define optimization variables
        h_var = cp.Variable(n_points)
        
        # Objective: maximize correlation with convex perturbation
        objective = 0
        epsilon = epsilon_0 * (beta ** t)
        
        for k in top_shifts:
            # Compute shifted indices
            shifted_indices = np.arange(n_points)
            if k < 0:
                shifted_indices = np.arange(-k, n_points)
                h_k = np.concatenate([np.zeros(-k), h[:n_points+k]])
            elif k > 0:
                shifted_indices = np.arange(n_points-k)
                h_k = np.concatenate([h[k:], np.zeros(k)])
            else:
                h_k = h
            
            # Element-wise product for the correlation term
            integral_term = cp.sum(cp.multiply(h_var, 1 - h_k))
            objective += integral_term
        
        objective = cp.Minimize(-objective + epsilon * cp.sum_squares(h_var))
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=1000, verbose=False)
        except:
            # Fallback to gradient ascent if SCS fails
            break
            
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_new = h_var.value
        else:
            # If solving fails, just try projection
            h_new = h.copy()
            h_new = project_to_feasible(h_new, n_points)
        
        # Project to feasible region
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate and update if better
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Try convex perturbation approach
    try:
        h_opt, c5_opt = convex_perturbation_optimization(h.copy(), n_points, dx, 5000)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    except:
        pass
    
    # Local refinement with gradient descent
    max_iterations = 5000
    step_size = 0.01
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient for the max lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h[i] - h[j]
            else:
                grad[i] = 1 - 2 * h[i]
        
        # Gradient descent step
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to maintain constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            step_size *= 0.95
            if step_size < 1e-8:
                break
    
    # Final local perturbation
    for _ in range(100):
        if time() - start_time > budget_s * 0.95:
            break
        h_pert = h + np.random.normal(0, 0.001, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        c5_pert = evaluate_erdos_solution(h_pert, dx)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
    
    return (list(best_h), best_c5, n_points)
