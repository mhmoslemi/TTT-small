import numpy as np
from time import time
from cvxpy import *
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex perturbation parameters
    epsilon_0 = 0.001
    beta = 0.99
    max_iter = 5000
    top_k = 100
    
    # Track best solution
    best_c5_history = []
    
    # Convex optimization approach
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Get top k dominant correlations (for convexification)
        corr = np.correlate(h, 1 - h, mode='full')
        top_k_shifts = np.argsort(corr)[-top_k:] - (n_points - 1)
        
        # Create convexified optimization
        h_var = Variable(n_points)
        
        # Objective: minimize overlap + convex perturbation
        objective_terms = []
        for shift in top_k_shifts:
            # Build shifted h_k computed as convolution
            if shift >= 0:
                h_shifted = np.concatenate([np.zeros(shift), h[:-shift]])
            else:
                h_shifted = np.concatenate([h[-shift:], np.zeros(-shift)])
            
            # Compute dot product term
            if shift >= 0:
                terms = h_var[shift:] * (1 - h_shifted[shift:])
            else:
                terms = h_var[:shift] * (1 - h_shifted[:shift])
            
            objective_terms.append(-sum(terms))
            
        # Add convex perturbation
        objective_terms.append(epsilon_0 * (beta ** iteration) * sum_squares(h_var))
        
        # Total objective
        objective = Minimize(sum(objective_terms))
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            sum(h_var) == n_points / 2
        ]
        
        # Solve
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=CPLEX, verbose=False, time_limit=0.1)
            h_new = h_var.value
        except:
            h_new = h
            
        if h_new is None:
            continue
            
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Store for termination signal
        best_c5_history.append(best_c5)
        
        # Early termination: stagnation detection
        if len(best_c5_history) > 100:
            recent = best_c5_history[-100:]
            if abs(recent[-1] - recent[0]) < 1e-8:
                break
                
        # Adaptive parameters
        if iteration % 500 == 0:
            # Reduce solver tolerance
            pass
            
    # Final local refinement with random steps
    final_h = best_h.copy()
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    for _ in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor
        h_neighbor = final_h + np.random.uniform(-0.01, 0.01, n_points)
        h_neighbor = np.clip(h_neighbor, 0, 1)
        h_neighbor = h_neighbor * (n_points / 2) / np.sum(h_neighbor)
        
        c5_neighbor = evaluate_erdos_solution(h_neighbor, dx)
        if c5_neighbor < final_c5:
            final_h = h_neighbor.copy()
            final_c5 = c5_neighbor
            
    # Final validation check
    if not (np.all(final_h >= 0) and np.all(final_h <= 1)):
        final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
