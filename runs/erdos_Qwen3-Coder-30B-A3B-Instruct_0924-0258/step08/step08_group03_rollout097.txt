import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Renormalize
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    max_iter = 5000
    epsilon_0 = 0.001
    beta = 0.99
    top_k = 100
    
    # Convex optimization helper
    def solve_convex_subproblem(h_current, epsilon):
        h_var = cp.Variable(n_points)
        # Build the objective with convex perturbation
        # We want to minimize sum_{k in top_k} (h * (1 - h_k)) + epsilon * ||h||_2^2
        
        # This is approximated by a finite sum over dominant shifts (top_k)
        # For simplicity we'll use an average over recent dominant shifts
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        dominant_shifts = np.argsort(corr)[-top_k:]
        objective = 0
        for k in dominant_shifts:
            shift_idx = k - (n_points - 1)  # relative shift
            # Build correlation term using linear operations
            if shift_idx >= 0:
                # h[i] * (1 - h[i+shift_idx]) for valid indices
                h_shifted = np.concatenate([h_current[shift_idx:], np.zeros(shift_idx)])
            else:
                # h[i] * (1 - h[i-shift_idx]) with appropriate padding
                h_shifted = np.concatenate([np.zeros(-shift_idx), h_current[:n_points+shift_idx]])
            
            objective += cp.sum(cp.multiply(h_var, 1-h_shifted))
            
        objective += epsilon * cp.sum_squares(h_var)
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # Solve
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_new = h_var.value
            else:
                h_new = h_current
        except:
            h_new = h_current
            
        return h_new
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps and find dominant ones
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Adaptive epsilon
        epsilon = epsilon_0 * (beta ** iteration)
        
        # Do convex subproblem
        h_new = solve_convex_subproblem(h, epsilon)
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Occasionally perturb with random noise to escape local minima
            if iteration % 100 == 0:
                noise = np.random.normal(0, 0.001, n_points)
                h_new = h + noise
                h_new = project_to_feasible(h_new, n_points)
                c5_new_pert = evaluate_erdos_solution(h_new, dx)
                if c5_new_pert < best_c5:
                    best_c5 = c5_new_pert
                    best_h = h_new.copy()
                    h = h_new.copy()
        
        if iteration % 1000 == 0:
            # Wake-up periodically to see if we are making progress
            residual = time() - start_time
            if residual > budget_s * 0.9:
                break
    
    # Final local refinement
    best_trials = []
    for _ in range(100):
        noise = np.random.normal(0, 0.001, n_points)
        h_trial = h + noise
        h_trial = project_to_feasible(h_trial, n_points)
        c5_trial = evaluate_erdos_solution(h_trial, dx)
        best_trials.append((c5_trial, h_trial.copy()))
    
    if best_trials:
        c5_best, h_best = min(best_trials, key=lambda x: x[0])
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()
    
    return (list(best_h), best_c5, n_points)
