import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize with the starting solution
    h_current = h_start.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Define optimization variables for CVXPY
    h = cp.Variable(n_points, bounds=(0, 1))
    C5 = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h) == n_points / 2, C5 >= 0]
    
    # Add overlap constraints for all k
    for k in range(-n_points + 1, n_points):
        # Indices for h[i] and h[i+k]
        if k >= 0:
            i_indices = np.arange(n_points - k)
            k_indices = np.arange(k, n_points)
        else:
            i_indices = np.arange(-k, n_points)
            k_indices = np.arange(n_points + k)
        
        if len(i_indices) > 0:
            # h[i] * (1 - h[i+k]) constraint
            # <= (sum(h) - C5*dx) * dx  for maximum overlap bound
            corr_term = cp.sum(cp.multiply(h[i_indices], 1 - h[k_indices])) * dx
            # Add constraint that this term is less than or equal to C5*dx
            constraints.append(corr_term <= C5 * dx)
    
    # Define problem
    objective = cp.Minimize(C5)
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with time limit
    try:
        prob.solve(solver=cp.MOSEK, time_limit=budget_s * 0.95, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            if h_opt is not None and np.all(0 <= h_opt) and np.all(h_opt <= 1):
                h_normalized = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_val = evaluate_erdos_solution(h_normalized, dx)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_normalized.copy()
    except:
        pass
    
    # If SDP didn't give better result, fall back to using gradient descent with improvements
    if best_c5 >= 0.381:
        step_size = 0.01
        max_iter = 10000
        iter_count = 0
        h_current = best_h.copy()
        
        # More sophisticated heuristic for the gradient update
        # For n_points=35 and proper correlation computation, reduce exploration if needed
        while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
            # Compute all correlations for maximal overlap lag
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (len(h_current) - 1)
            
            # Compute detailed gradient
            gradient = np.zeros(n_points)
            if max_lag >= 0:
                for i in range(n_points - max_lag):
                    gradient[i] += 1 - h_current[i + max_lag]
                    if i + max_lag < n_points:
                        gradient[i + max_lag] -= h_current[i]
            else:
                for i in range(n_points + max_lag):
                    gradient[i] += 1 - h_current[i - max_lag]
                    if i - max_lag >= 0:
                        gradient[i - max_lag] -= h_current[i]
            
            # Gradient step with bounds
            h_new = h_current - step_size * gradient
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate new solution
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept step if improvement
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            else:
                step_size *= 0.99
                if step_size < 1e-10:
                    break
            
            h_current = h_new
            iter_count += 1
    
    return (list(best_h), best_c5, n_points)
