import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize grid parameters
    n_coarse = 15
    n_fine = 35
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # Initialize coarse grid
    h_coarse = np.ones(n_coarse) * 0.5
    
    # If initial_h_values exists, use it for coarse grid
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Interpolate to coarse grid
        x_fine = np.linspace(0, 2, n_fine)
        x_coarse = np.linspace(0, 2, n_coarse)
        f = interp1d(x_fine, h_init, kind='linear', fill_value='extrapolate')
        h_coarse = f(x_coarse)
    else:
        # Start with uniform distribution
        h_coarse = np.ones(n_coarse) * 0.5
    
    best_c5_coarse = float('inf')
    best_h_coarse = h_coarse.copy()
    
    # Coarse grid optimization
    max_iterations_coarse = 500
    step_size = 0.01
    
    for iteration in range(max_iterations_coarse):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute overlap
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_coarse - 1)
        
        # Compute gradient
        grad = np.zeros(n_coarse)
        for i in range(n_coarse):
            j = i + max_lag
            if 0 <= j < n_coarse:
                grad[i] = 1 - 2 * h_coarse[i] - h_coarse[j]
            else:
                grad[i] = 1 - 2 * h_coarse[i]
        
        # Update
        h_new = h_coarse - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = h_new * (n_coarse / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx_coarse)
        
        if c5_new < best_c5_coarse:
            best_c5_coarse = c5_new
            best_h_coarse = h_new.copy()
        
        h_coarse = h_new.copy()
        
    # Upsample to fine grid
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    f = interp1d(x_coarse, best_h_coarse, kind='linear', fill_value='extrapolate')
    h_fine = f(x_fine)
    
    # Ensure integral constraint
    h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)
    
    best_c5_fine = evaluate_erdos_solution(h_fine, dx_fine)
    best_h_fine = h_fine.copy()
    
    # Fine grid optimization - alternating phases
    max_iterations_fine = 2000
    mutation_rate = 0.10
    step_size_fine = 0.01
    
    for iteration in range(max_iterations_fine):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Random mutation phase
        if iteration % 2 == 0:
            # Mutate 10% of values
            n_mutate = int(mutation_rate * n_fine)
            indices = np.random.choice(n_fine, n_mutate, replace=False)
            h_fine[indices] += np.random.normal(0, 0.01, n_mutate)
            h_fine = np.clip(h_fine, 0, 1)
            h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)  # Renormalize
            
        # Gradient descent phase
        else:
            # Compute overlap
            corr = np.correlate(h_fine, 1 - h_fine, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (n_fine - 1)
            
            # Compute gradient
            grad = np.zeros(n_fine)
            for i in range(n_fine):
                j = i + max_lag
                if 0 <= j < n_fine:
                    grad[i] = 1 - 2 * h_fine[i] - h_fine[j]
                else:
                    grad[i] = 1 - 2 * h_fine[i]
            
            # Update
            h_fine_new = h_fine - step_size_fine * grad
            h_fine_new = np.clip(h_fine_new, 0, 1)
            
            # Renormalize
            h_fine_new = h_fine_new * (n_fine / 2) / np.sum(h_fine_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_fine_new, dx_fine)
            
            if c5_new < best_c5_fine:
                best_c5_fine = c5_new
                best_h_fine = h_fine_new.copy()
            
            h_fine = h_fine_new.copy()
    
    return (list(best_h_fine), best_c5_fine, n_fine)
