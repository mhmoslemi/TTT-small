import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_coarse = 15
    n_fine = 35
    coarse_iterations = 500
    fine_iterations = 2000
    mutation_rate = 0.1
    
    # Initialize coarse grid
    h_coarse = np.random.uniform(0, 1, n_coarse)
    h_coarse = project_feasible(h_coarse, n_coarse)
    
    dx_coarse = 2.0 / n_coarse
    best_coarse_c5 = float('inf')
    best_coarse_h = h_coarse.copy()
    
    # Coarse grid optimization
    for iteration in range(coarse_iterations):
        if time() - start_time > budget_s * 0.8:
            break
            
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_coarse - 1)
        
        grad = np.zeros(n_coarse)
        for i in range(n_coarse):
            j = i + max_lag
            if 0 <= j < n_coarse:
                grad[i] = 1 - 2 * h_coarse[i] - h_coarse[j]
            else:
                grad[i] = 1 - 2 * h_coarse[i]
        
        h_new = h_coarse - 0.01 * grad
        h_new = project_feasible(h_new, n_coarse)
        
        c5_new = evaluate_erdos_solution(h_new, dx_coarse)
        
        if c5_new < best_coarse_c5:
            best_coarse_c5 = c5_new
            best_coarse_h = h_new.copy()
            
        h_coarse = h_new
    
    # Upsample to fine grid
    coarse_x = np.linspace(0, 2, n_coarse)
    fine_x = np.linspace(0, 2, n_fine)
    
    f = interp1d(coarse_x, best_coarse_h, kind='linear', fill_value='extrapolate')
    h_fine = f(fine_x)
    h_fine = project_feasible(h_fine, n_fine)
    
    # Fine grid optimization
    dx_fine = 2.0 / n_fine
    best_fine_c5 = evaluate_erdos_solution(h_fine, dx_fine)
    best_fine_h = h_fine.copy()
    
    for iteration in range(fine_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Gradient-based update
        corr = np.correlate(h_fine, 1 - h_fine, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_fine - 1)
        
        grad = np.zeros(n_fine)
        for i in range(n_fine):
            j = i + max_lag
            if 0 <= j < n_fine:
                grad[i] = 1 - 2 * h_fine[i] - h_fine[j]
            else:
                grad[i] = 1 - 2 * h_fine[i]
        
        h_new = h_fine - 0.01 * grad
        h_new = project_feasible(h_new, n_fine)
        
        c5_new = evaluate_erdos_solution(h_new, dx_fine)
        
        if c5_new < best_fine_c5:
            best_fine_c5 = c5_new
            best_fine_h = h_new.copy()
            h_fine = h_new.copy()
        else:
            # Mutation phase
            mutation_indices = np.random.choice(n_fine, int(n_fine * mutation_rate), replace=False)
            h_new = h_fine.copy()
            h_new[mutation_indices] += np.random.normal(0, 0.05, len(mutation_indices))
            h_new = project_feasible(h_new, n_fine)
            
            c5_mutation = evaluate_erdos_solution(h_new, dx_fine)
            if c5_mutation < best_fine_c5:
                best_fine_c5 = c5_mutation
                best_fine_h = h_new.copy()
                h_fine = h_new.copy()
    
    final_c5 = evaluate_erdos_solution(best_fine_h, dx_fine)
    
    return (list(best_fine_h), final_c5, n_fine)
