import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search on C5
    low, high = 0.38, 0.38085
    tolerance = 1e-5
    
    # Sample shifts for constraint checking
    max_shifts = min(100, n_points)
    shifts = np.random.choice(n_points, max_shifts, replace=False)
    
    # Function to check feasibility for a given C5
    def check_feasible(c5_val):
        # Create variables
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Add overlap constraints
        for k in shifts:
            # Calculate the overlap integral for shift k
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h_vars[i] * (1 - h_vars[j])
            constraints.append(overlap_expr <= c5_val * dx)
        
        # Solve the feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            return prob.status == cp.OPTIMAL
        except:
            return False
    
    # Binary search
    while high - low > tolerance and time() - start_time < budget_s * 0.9:
        mid = (low + high) / 2
        if check_feasible(mid):
            high = mid
            best_c5 = mid
        else:
            low = mid
    
    # Refinement with gradient descent
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Convert h to a continuous vector for optimization
    h_init = best_h.copy()
    
    # Use scipy's minimize with constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Run for a fixed number of iterations
    max_iter = 500
    for _ in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        result = minimize(objective, h_init, method='SLSQP', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 10})
        if not result.success:
            break
        
        h_init = result.x
        current_c5 = objective(h_init)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_init.copy()
    
    # Final normalization
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
