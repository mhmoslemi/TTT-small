import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def repair(h, n_points):
    h = np.clip(h, 0, 1)
    if np.sum(h) == 0:
        return h
    h = h * (n_points/2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-start with Latin hypercube sampling
    n_starts = 50
    
    # Generate Latin hypercube samples
    samples = np.random.uniform(0, 1, (n_starts, n_points))
    for i in range(n_points):
        col = samples[:, i]
        sorted_indices = np.argsort(col)
        samples[sorted_indices, i] = np.linspace(0, 1, n_starts)
    
    # Add the initial solution as first start
    starts = [h_start.copy()]
    starts.extend([repair(s, n_points) for s in samples])
    
    if len(starts) > n_starts:
        starts = starts[:n_starts]
    
    # SLSQP optimization with multi-start
    for i, h0 in enumerate(starts):
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h0, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                h_final = repair(h_final, n_points)
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
        except Exception:
            pass
    
    # Final check if our initial solution was better
    h_start = repair(h_start, n_points)
    c5_start = evaluate_erdos_solution(h_start, dx)
    if c5_start < best_c5:
        best_c5 = c5_start
        best_h = h_start.copy()
    
    # Ensure the final solution is valid
    best_h = repair(best_h, n_points)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if abs(final_c5 - best_c5) > 1e-10:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
