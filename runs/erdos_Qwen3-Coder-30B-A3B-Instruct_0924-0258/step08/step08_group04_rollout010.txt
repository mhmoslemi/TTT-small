import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # PSO parameters
    num_particles = 75
    inertia_start = 0.8
    inertia_end = 0.4
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    perturbation_rate = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    particle_best = []
    particle_best_values = []
    
    for i in range(num_particles):
        if i == 0:
            h_init = h_start.copy()
        else:
            # Perturb the initial solution
            h_init = h_start + np.random.normal(0, 0.05, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points/2) / np.sum(h_init)
        
        particles.append(h_init)
        velocities.append(np.zeros(n_points))
        particle_best.append(h_init.copy())
        particle_best_values.append(evaluate_erdos_solution(h_init, dx))
    
    # Global best
    global_best_idx = np.argmin(particle_best_values)
    global_best = particle_best[global_best_idx].copy()
    global_best_value = particle_best_values[global_best_idx]
    
    # PSO iterations
    max_iter = int(1e6)
    iteration = 0
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iter:
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * (iteration / max_iter)
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive_weight * r1 * (particle_best[i] - particles[i]) + 
                           social_weight * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible set
            particles[i] = np.clip(particles[i], 0, 1)
            # Adjust sum to exactly n_points/2
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate fitness
            current_value = evaluate_erdos_solution(particles[i], dx)
            
            # Update particle best
            if current_value < particle_best_values[i]:
                particle_best[i] = particles[i].copy()
                particle_best_values[i] = current_value
            
            # Update global best
            if current_value < global_best_value:
                global_best = particles[i].copy()
                global_best_value = current_value
        
        # Periodic perturbation to escape local minima
        if iteration % 1000 == 0 and iteration > 0:
            for i in range(num_particles):
                if np.random.random() < perturbation_rate:
                    particles[i] += np.random.normal(0, 0.02, n_points)
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
        
        iteration += 1
    
    return (list(global_best), global_best_value, n_points)
