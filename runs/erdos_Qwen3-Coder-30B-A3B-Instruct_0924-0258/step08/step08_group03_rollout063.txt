import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_fine = np.array(initial_h_values)
        n_points = len(h_fine)
    else:
        n_points = 35
        h_fine = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_fine.copy()
    best_c5 = evaluate_erdos_solution(h_fine, dx)
    
    # Coarse grid optimization
    n_coarse = 15
    h_coarse = np.ones(n_coarse) * 0.5
    coarse_dx = 2.0 / n_coarse
    
    for iter_coarse in range(500):
        if time() - start_time > budget_s * 0.9:
            break
        
        # Evaluate overlaps
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_coarse - 1)
        
        # Compute gradient
        grad = np.zeros(n_coarse)
        for i in range(n_coarse):
            j = i + max_lag
            if 0 <= j < n_coarse:
                grad[i] = 1 - 2 * h_coarse[i] - h_coarse[j]
            else:
                grad[i] = 1 - 2 * h_coarse[i]
        
        # Gradient descent
        h_coarse_new = h_coarse - 0.01 * grad
        h_coarse_new = np.clip(h_coarse_new, 0, 1)
        
        # Renormalize to maintain integral constraint
        h_coarse_new = h_coarse_new * (n_coarse / 2) / np.sum(h_coarse_new)
        
        # Maybe update best coarse solution
        c5_coarse_new = evaluate_erdos_solution(h_coarse_new, coarse_dx)
        if c5_coarse_new < best_c5:
            best_c5 = c5_coarse_new
            # Upsample to fine grid
            x_coarse = np.linspace(0, 2, n_coarse)
            x_fine = np.linspace(0, 2, n_points)
            interpolator = interp1d(x_coarse, h_coarse_new, kind='linear', fill_value='extrapolate')
            h_fine = interpolator(x_fine)
            h_fine = np.clip(h_fine, 0, 1)
            h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
            best_h = h_fine.copy()
        
        h_coarse = h_coarse_new
    
    # Fine grid refinement with alternating descent and mutation
    for iter_fine in range(2000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Gradient descent step
        corr = np.correlate(h_fine, 1 - h_fine, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h_fine[i] - h_fine[j]
            else:
                grad[i] = 1 - 2 * h_fine[i]
        
        h_fine_new = h_fine - 0.01 * grad
        h_fine_new = np.clip(h_fine_new, 0, 1)
        h_fine_new = h_fine_new * (n_points / 2) / np.sum(h_fine_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_fine_new, dx)
        
        # Accept if better or random mutation
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_fine_new.copy()
            h_fine = h_fine_new.copy()
        else:
            # Try random mutation
            if np.random.rand() < 0.3:
                h_mut = h_fine.copy()
                # Mutate 10% of values
                n_mutate = max(1, n_points // 10)
                indices = np.random.choice(n_points, n_mutate, replace=False)
                h_mut[indices] = np.random.rand(n_mutate)
                h_mut = np.clip(h_mut, 0, 1)
                h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
                c5_mut = evaluate_erdos_solution(h_mut, dx)
                if c5_mut < best_c5:
                    best_c5 = c5_mut
                    best_h = h_mut.copy()
                    h_fine = h_mut.copy()
    
    return (list(best_h), best_c5, n_points)
