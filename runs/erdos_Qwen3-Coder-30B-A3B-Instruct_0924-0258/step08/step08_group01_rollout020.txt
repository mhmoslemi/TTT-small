import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        n_points = len(h_current)
    else:
        n_points = 35
        h_current = np.ones(n_points) * 0.5
    
    # Compute dx
    dx = 2.0 / n_points
    
    # Initialize best solution and fitness
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Multi-scale optimization with increasing resolution
    n_scales = int(np.log2(max(n_points, 16))) + 1
    scale = 0
    perturbation_magnitude = 0.05
    max_scale_iterations = 10
    
    while scale < n_scales and (time() - start_time) < budget_s * 0.95:
        current_n_points = max(16, 2**scale)
        
        # If current resolution is below minimum, upsample
        if current_n_points > n_points:
            # Upsample linearly
            h_upsampled = np.interp(np.linspace(0, 1, current_n_points), 
                                  np.linspace(0, 1, n_points), h_current)
            h_upsampled = h_upsampled * (current_n_points / 2) / np.sum(h_upsampled)
            h_current = h_upsampled.copy()
            n_points = current_n_points
        elif current_n_points < n_points:
            # Reduce resolution (not used in here as we want to increase resolution)

        dx = 2.0 / n_points

        # Local optimization with perturbations
        scale_iterations = 0
        while scale_iterations < max_scale_iterations and (time() - start_time) < budget_s * 0.95:
            # Small random perturbation
            h_perturbed = h_current.copy()
            for i in range(n_points):
                if np.random.random() < 0.3:  # 30% chance to perturb
                    pert = perturbation_magnitude * (2 * np.random.random() - 1)
                    h_perturbed[i] += pert
            
            # Ensure feasibility
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            # Evaluate
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            
            # Accept improvement
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
                h_current = h_perturbed.copy()
                perturbation_magnitude = 0.05  # Reset magnitude
            else:
                perturbation_magnitude *= 0.95  # Decay magnitude
            
            # Local SQP refinement
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            res = minimize(objective, h_current, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-12})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined.copy()
                    h_current = h_refined.copy()
                    
            scale_iterations += 1
            
        scale += 1
    
    # Final SLSQP refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
