import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex perturbation optimization
    max_iter = 5000
    epsilon_0 = 0.001
    beta = 0.99
    num_shifts = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find dominant shifts
        corr = np.correlate(h, 1 - h, mode='full')
        top_k = np.argsort(corr)[-num_shifts:]
        
        # Convex optimization with perturbation
        h_var = cp.Variable(n_points)
        objective = 0
        
        # Add convex perturbation term
        epsilon = epsilon_0 * (beta ** iteration)
        objective += epsilon * cp.norm(h_var, 2)**2
        
        # Add overlap terms for dominant shifts
        for k in top_k:
            # Compute the overlap integral for shift k
            # This requires some extra work to get the proper convolution expression
            # Simplified: we approximate cumulative sum
            shift_idx = k - (n_points - 1)
            # s[i] = h[i] * (1 - h[i+shift_idx]) for valid indices
            if shift_idx >= 0:
                # h[i] * (1 - h[i + shift_idx])
                overlapping_terms = 0
                for i in range(n_points - shift_idx):
                    overlapping_terms += h_var[i] * (1 - h_var[i + shift_idx])
            else:
                # h[i + (-shift_idx)] * (1 - h[i])
                overlapping_terms = 0
                for i in range(-shift_idx, n_points):
                    overlapping_terms += h_var[i] * (1 - h_var[i - shift_idx])
            objective += overlapping_terms
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Solve convex problem
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000)
            h_new = np.array(h_var.value).flatten()
        except:
            h_new = h.copy()
            
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
            
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Update h
        h = h_new.copy()
        
        if iteration % 500 == 0:
            if iteration > 0:
                # Refinement with local search using different random seeds
                for _ in range(5):
                    if time() - start_time > budget_s * 0.95:
                        break
                    h_local = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                    h_local = h_local * (n_points / 2) / np.sum(h_local)
                    c5_local = evaluate_erdos_solution(h_local, dx)
                    if c5_local < best_c5:
                        best_c5 = c5_local
                        best_h = h_local.copy()
    
    # Final local refinement
    for _ in range(100):
        if time() - start_time > budget_s * 0.99:
            break
        perturbation = np.random.normal(0, 0.001, n_points)
        h_pert = np.clip(h + perturbation, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        c5_pert = evaluate_erdos_solution(h_pert, dx)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
        
    return (list(best_h), best_c5, n_points)
