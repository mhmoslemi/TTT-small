import numpy as np
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search parameters
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    def solve_sdp(c5_val):
        # Define variables
        h_var = cp.Variable(n_points)
        Y = cp.Variable((n_points, n_points), symmetric=True)
        
        # Constraints
        constraints = [
            Y >> 0,  # Y is positive semidefinite
            cp.diag(Y) == h_var,  # Diagonal elements of Y equal h_var
            cp.sum(h_var) == n_points / 2,  # Sum constraint
        ]
        
        # Overlap constraints for sampled shifts
        shifts = np.random.choice(n_points, size=min(100, n_points), replace=False)
        for k in shifts:
            # Sum of h_i * (1 - h_{i+k}) <= c5_val * dx
            # This is equivalent to sum of h_i - sum of h_i * h_{i+k} <= c5_val * dx
            # sum of h_i * h_{i+k} is the sum of Y[i, i+k]
            # So: sum(h_i) - sum(Y[i, i+k]) <= c5_val * dx
            sum_h = cp.sum(h_var)
            sum_Y_k = cp.sum([Y[i, (i+k) % n_points] for i in range(n_points)])
            constraints.append(sum_h - sum_Y_k <= c5_val * dx)
        
        # Objective (minimize C5)
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                if h_sol is not None and np.all(h_sol >= 0) and np.all(h_sol <= 1):
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    return h_sol
        except:
            pass
        return None
    
    def project_and_refine(h_in, max_iter=1000):
        h_out = h_in.copy()
        step_size = 0.005
        for _ in range(max_iter):
            if time() - start_time > budget_s * 0.90:
                break
            corr = np.correlate(h_out, 1 - h_out, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (n_points - 1)
            
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + max_lag
                if 0 <= j < n_points:
                    grad[i] = 1 - 2 * h_out[i] - h_out[j]
                else:
                    grad[i] = 1 - 2 * h_out[i]
            
            h_new = h_out - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_out = h_new.copy()
            else:
                step_size *= 0.5
                if step_size < 1e-6:
                    break
        return h_out, best_c5, best_h
    
    # Binary search loop
    while time() - start_time < budget_s * 0.95 and high - low > tolerance:
        mid = (low + high) / 2
        h_candidate = solve_sdp(mid)
        
        if h_candidate is not None:
            h_projected, c5_candidate, _ = project_and_refine(h_candidate)
            # If this improved, update best solution
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_projected.copy()
            high = mid
        else:
            low = mid
    
    # Final optimization round
    h_final, final_c5, final_h = project_and_refine(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h
    
    return (list(best_h), best_c5, n_points)
