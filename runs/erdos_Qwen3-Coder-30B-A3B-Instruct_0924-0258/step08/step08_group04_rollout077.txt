import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # COARSE GRID OPTIMIZATION (n=15) - 60s
    coarse_n = 15
    coarse_dx = 2.0 / coarse_n
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.5
    
    # Initialize coarse population
    coarse_population = []
    for _ in range(population_size):
        h = np.random.rand(coarse_n)
        h = h * (coarse_n/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        coarse_population.append(h.copy())
    
    # Evaluate initial population
    coarse_fitness = []
    for h in coarse_population:
        c5 = evaluate_erdos_solution(h, coarse_dx)
        coarse_fitness.append(-c5)
    
    # Evolutionary optimization
    max_gen = 1000
    gen = 0
    
    while time() - start_time < budget_s * 0.2 and gen < max_gen:
        # Sort by fitness (ascending since we use -C5)
        sorted_indices = np.argsort(coarse_fitness)
        top_indices = sorted_indices[:population_size//2]
        
        # Selection - keep top half
        selected = [coarse_population[i] for i in top_indices]
        
        # Crossover and mutation
        new_population = selected.copy()
        
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate and len(selected) >= 2:
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                crossover_point = np.random.randint(1, coarse_n-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutation
                if np.random.rand() < mutation_rate:
                    idx = np.random.randint(0, coarse_n)
                    child1[idx] = np.random.rand()
                if np.random.rand() < mutation_rate:
                    idx = np.random.randint(0, coarse_n)
                    child2[idx] = np.random.rand()
                
                # Repair
                child1 = child1 * (coarse_n/2) / np.sum(child1)
                child1 = np.clip(child1, 0, 1)
                child2 = child2 * (coarse_n/2) / np.sum(child2)
                child2 = np.clip(child2, 0, 1)
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Mutation only
                parent = np.random.choice(selected)
                child = parent.copy()
                if np.random.rand() < mutation_rate:
                    idx = np.random.randint(0, coarse_n)
                    child[idx] = np.random.rand()
                child = child * (coarse_n/2) / np.sum(child)
                child = np.clip(child, 0, 1)
                new_population.append(child)
        
        coarse_population = new_population
        
        # Evaluate
        coarse_fitness = []
        for h in coarse_population:
            c5 = evaluate_erdos_solution(h, coarse_dx)
            coarse_fitness.append(-c5)
        
        # Track best
        best_idx = np.argmin(coarse_fitness)
        if -coarse_fitness[best_idx] < best_c5:
            best_c5 = -coarse_fitness[best_idx]
            best_h = coarse_population[best_idx].copy()
        
        gen += 1
    
    # UPSCALING TO FINE GRID (n=35) - 10s
    if time() - start_time < budget_s * 0.3:
        # Interpolate and refine
        coarse_h = best_h.copy()
        fine_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, coarse_n), coarse_h)
        fine_h = fine_h * (n_points/2) / np.sum(fine_h)
        fine_h = np.clip(fine_h, 0, 1)
        best_h = fine_h
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
        # Final gradient optimization on fine grid - 500s
        step_size = 0.01
        max_iter = 10000
        iter_count = 0
        
        while time() - start_time < budget_s and iter_count < max_iter:
            # Compute correlations and get the key lags
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (len(best_h) - 1)
            
            # Compute gradient for max overlap
            gradient = np.zeros(n_points)
            if max_lag >= 0:
                for i in range(n_points - max_lag):
                    gradient[i] += 1 - best_h[i + max_lag]
                    if i + max_lag < n_points:
                        gradient[i + max_lag] -= best_h[i]
            else:
                for i in range(n_points + max_lag):
                    gradient[i] += 1 - best_h[i - max_lag]
                    if i - max_lag >= 0:
                        gradient[i - max_lag] -= best_h[i]
            
            # Gradient descent step
            new_h = best_h - step_size * gradient
            
            # Ensure feasibility
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points/2) / np.sum(new_h)
            
            # Evaluate new C5 value
            new_c5 = evaluate_erdos_solution(new_h, dx)
            
            # Accept if improvement
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
            else:
                step_size *= 0.95
                if step_size < 1e-10:
                    break
            
            iter_count += 1
    
    return (list(best_h), best_c5, n_points)
