import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Renormalize
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize coarse grid (15 points)
    n_coarse = 15
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.ones(n_coarse) * 0.5
    
    best_c5_coarse = float('inf')
    best_h_coarse = h_coarse.copy()
    
    # Coarse grid optimization for 500 iterations
    step_size = 0.01
    for iteration in range(500):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute overlaps and find max
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_coarse - 1)
        
        # Compute gradient
        grad = np.zeros(n_coarse)
        for i in range(n_coarse):
            j = i + max_lag
            if 0 <= j < n_coarse:
                grad[i] = 1 - 2 * h_coarse[i] - h_coarse[j]
            else:
                grad[i] = 1 - 2 * h_coarse[i]
        
        # Gradient descent step
        h_new_coarse = h_coarse - step_size * grad
        h_new_coarse = project_feasible(h_new_coarse, n_coarse)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new_coarse, dx_coarse)
        
        # Keep best
        if c5_new < best_c5_coarse:
            best_c5_coarse = c5_new
            best_h_coarse = h_new_coarse.copy()
            h_coarse = h_new_coarse.copy()
        else:
            step_size *= 0.995
            if step_size < 1e-8:
                break
    
    # Upsample to fine grid (35 points)
    n_points = 35
    dx = 2.0 / n_points
    
    # Interpolate
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_points)
    
    # Create the interpolation function (linear)
    f = interp1d(x_coarse, best_h_coarse, kind='linear', fill_value='extrapolate')
    h_fine = f(x_fine)
    
    # Project to feasible
    h_fine = project_feasible(h_fine, n_points)
    
    # Optimization on fine grid
    best_c5_fine = evaluate_erdos_solution(h_fine, dx)
    best_h_fine = h_fine.copy()
    
    # Phase 1: gradient descent
    for phase in range(2):
        if time() - start_time > budget_s * 0.95:
            break
        for iteration in range(1000):
            if time() - start_time > budget_s * 0.95:
                break
            
            corr = np.correlate(h_fine, 1 - h_fine, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (n_points - 1)
            
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + max_lag
                if 0 <= j < n_points:
                    grad[i] = 1 - 2 * h_fine[i] - h_fine[j]
                else:
                    grad[i] = 1 - 2 * h_fine[i]
            
            step_size = 0.01 * (0.99 ** phase)
            h_new_fine = h_fine - step_size * grad
            h_new_fine = project_feasible(h_new_fine, n_points)
            
            c5_new = evaluate_erdos_solution(h_new_fine, dx)
            
            if c5_new < best_c5_fine:
                best_c5_fine = c5_new
                best_h_fine = h_new_fine.copy()
                h_fine = h_new_fine.copy()
            else:
                # Random mutation: 10% of the values
                mutate_indices = np.random.choice(n_points, int(0.1 * n_points), replace=False)
                h_new_fine[mutate_indices] = np.random.uniform(0, 1, len(mutate_indices))
                h_new_fine = project_feasible(h_new_fine, n_points)
                c5_new = evaluate_erdos_solution(h_new_fine, dx)
                if c5_new < best_c5_fine:
                    best_c5_fine = c5_new
                    best_h_fine = h_new_fine.copy()
                    h_fine = h_new_fine.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h_fine, dx)
    
    return (list(best_h_fine), final_c5, n_points)
