import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # GA parameters
    population_size = 50
    max_generations = 10000
    tournament_size = 5
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    population.append(h_start.copy())
    for _ in range(population_size - 1):
        h = np.random.random(n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Precompute lags for correlation
    lags = np.arange(-n_points + 1, n_points)
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness_scores.append(c5)
            
        # Track best
        min_fitness = min(fitness_scores)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitness_scores.index(min_fitness)].copy()
        
        # Selection (tournament)
        new_population = []
        
        # Elitism: keep best
        best_idx = np.argmin(fitness_scores)
        new_population.append(population[best_idx].copy())
        
        while len(new_population) < population_size:
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Select parent
            parent = population[winner_idx]
            
            # Crossover with probability
            if np.random.random() < crossover_rate and len(new_population) < population_size - 1:
                # Select second parent
                tournament_indices2 = np.random.choice(population_size, tournament_size)
                tournament_fitness2 = [fitness_scores[i] for i in tournament_indices2]
                parent2_idx = tournament_indices2[np.argmin(tournament_fitness2)]
                parent2 = population[parent2_idx]
                
                # Uniform crossover
                mask = np.random.random(n_points) < 0.5
                child = np.where(mask, parent, parent2)
                
                # Adjust sum constraint
                sum_h = np.sum(child)
                child = child * (n_points/2) / sum_h
            else:
                child = parent.copy()
                
            # Mutation
            if np.random.random() < mutation_rate:
                # Flip mutation on two random positions
                idx1, idx2 = np.random.choice(n_points, 2, replace=False)
                child[idx1] = np.clip(child[idx1] + 0.1, 0, 1)
                child[idx2] = np.clip(child[idx2] - 0.1, 0, 1)
                
                # Adjust sum constraint
                sum_h = np.sum(child)
                child = child * (n_points/2) / sum_h
            
            new_population.append(child)
        
        population = new_population[:]
    
    # Final refinement with SLSQP if we have time
    if time() - start_time < budget_s * 0.98:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
