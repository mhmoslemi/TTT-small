import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter = 100000
    epsilon = 0.1
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    current_c5 = evaluate_erdos_solution(h, dx)
    T = T_init
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Perturb two random indices
        idx1, idx2 = np.random.randint(0, n_points, 2)
        
        # Save old values
        old_h1, old_h2 = h[idx1], h[idx2]
        
        # Adjust
        delta = np.random.uniform(-epsilon, epsilon)
        h[idx1] += delta
        h[idx2] -= delta
        
        # Enforce bounds
        h = np.clip(h, 0, 1)
        
        # Rebalance to satisfy sum constraint
        total = np.sum(h)
        h = h * (n_points/2) / total
        
        # Compute new overlap
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Accept with probability
            prob = np.exp(-(new_c5 - current_c5) / T)
            if np.random.rand() < prob:
                current_c5 = new_c5
            else:
                # Revert
                h[idx1], h[idx2] = old_h1, old_h2
        
        # Cool down
        T = max(T * alpha, T_min)
    
    # Local refinement with L-BFGS-B
    try:
        def obj_and_grad(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_overlap = np.max(corr) * dx
            l_max = np.argmax(corr)
            
            # Simple gradient approximation
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = (i + l_max) % n_points
                grad[i] = -2 * dx * (1 - h[j] - h[i])
            
            return max_overlap, grad
        
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
