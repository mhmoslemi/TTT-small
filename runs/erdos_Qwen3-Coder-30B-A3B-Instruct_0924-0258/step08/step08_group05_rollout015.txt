import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h_normalized = h_start * (n_points/2) / np.sum(h_start)
        population.append(h_normalized.copy())
    
    # Fill with random feasible solutions
    for _ in range(population_size - len(population)):
        # Create random solution
        h = np.random.rand(n_points)
        # Make it feasible
        sum_h = np.sum(h)
        h = h * (n_points/2) / sum_h
        # Clamp to [0,1]
        h = np.clip(h, 0, 1)
        population.append(h)
    
    # Evolution parameters
    max_generations = 10000
    elite_size = 1
    mutation_rate = 0.1
    tournament_size = 5
    no_improvement_limit = 100
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    def tournament_selection(population, fitnesses, k):
        selected_indices = np.random.choice(len(population), k)
        selected_fitness = [fitnesses[i] for i in selected_indices]
        winner_idx = selected_indices[np.argmin(selected_fitness)]
        return population[winner_idx]
    
    def crossover(parent1, parent2):
        # Uniform crossover
        mask = np.random.rand(len(parent1)) > 0.5
        child = np.where(mask, parent1, parent2)
        
        # Normalize to meet sum constraint
        sum_child = np.sum(child)
        child = child * (n_points/2) / sum_child
        # Clamp to [0,1]
        child = np.clip(child, 0, 1)
        return child
    
    def mutate(h):
        # Flip mutation
        h_mut = h.copy()
        if np.random.rand() < 0.5:
            idx1, idx2 = np.random.choice(len(h), 2, replace=False)
            h_mut[idx1] = np.clip(h_mut[idx1] + 0.1, 0, 1)
            h_mut[idx2] = np.clip(h_mut[idx2] - 0.1, 0, 1)
            
        # Perturb mutation
        if np.random.rand() < 0.5:
            num_perturb = max(1, int(len(h) * 0.1))
            indices = np.random.choice(len(h), num_perturb, replace=False)
            for i in indices:
                factor = np.random.uniform(0.9, 1.1)
                h_mut[i] = np.clip(h_mut[i] * factor, 0, 1)
                
        # Normalize and ensure sum constraint
        sum_h = np.sum(h_mut)
        h_mut = h_mut * (n_points/2) / sum_h
        return h_mut
    
    # Evolution loop
    no_improvement = 0
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= no_improvement_limit:
            break
            
        # Create new population
        new_population = []
        
        # Keep elite
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            child = crossover(parent1, parent2)
            
            if np.random.rand() < mutation_rate:
                child = mutate(child)
                
            new_population.append(child)
            
        population = new_population[:population_size]
    
    # Final check with optimization
    try:
        from scipy.optimize import minimize
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
