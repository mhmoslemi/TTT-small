import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.5
        h_start[1] = 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    max_iter = 100000
    no_improvement_count = 0
    max_no_improvement = 5000
    
    # Track time
    time_budget = budget_s * 0.95
    
    for iteration in range(max_iter):
        if time() - start_time > time_budget:
            break
            
        # Generate neighbor by swapping delta between two indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(0.01, 0.1)
        
        # Ensure we don't violate bounds
        max_delta = min(delta, h_current[i], 1 - h_current[j])
        if max_delta <= 0:
            continue
            
        # Create new candidate
        h_new = h_current.copy()
        h_new[i] += max_delta
        h_new[j] -= max_delta
        
        # Ensure bounds
        h_new[i] = np.clip(h_new[i], 0, 1)
        h_new[j] = np.clip(h_new[j], 0, 1)
        
        # Adjust to maintain sum constraint
        sum_diff = np.sum(h_new) - n_points / 2
        if abs(sum_diff) > 1e-10:
            # Distribute adjustment to all elements
            adjustment = sum_diff / n_points
            h_new = np.clip(h_new - adjustment, 0, 1)
            
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Calculate energy difference
        c5_current = evaluate_erdos_solution(h_current, dx)
        delta_energy = c5_new - c5_current
        
        # Accept or reject based on SA probability
        if delta_energy < 0 or np.random.rand() < np.exp(-delta_energy / T):
            h_current = h_new
            c5_current = c5_new
            
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Cooling schedule
        T *= cooling_rate
        if T < min_temp:
            T = min_temp
            
        # Early stopping
        if no_improvement_count > max_no_improvement:
            break
    
    # Final optimization using scipy
    try:
        from scipy.optimize import minimize
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
