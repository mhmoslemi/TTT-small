import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Multi-resolution SA with local refinement
    resolutions = [35, 70, 140, 280]
    current_n = resolutions[0]
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_current = h_start.copy()
    else:
        h_current = np.ones(current_n) * 0.5
    
    h_current = h_current * (current_n/2) / np.sum(h_current)
    dx = 2.0 / current_n
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    def generate_neighbor(h, temp):
        h_new = h.copy()
        # Perturb a few random points
        num_pert = max(1, int(len(h) * 0.1))
        indices = np.random.choice(len(h), size=num_pert, replace=False)
        for i in indices:
            delta = np.random.normal(0, temp * 0.2)
            h_new[i] += delta
        # Project back to constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (current_n/2) / np.sum(h_new)
        return h_new
    
    def sa_step(h, temp):
        h_new = generate_neighbor(h, temp)
        c5_old = evaluate_erdos_solution(h, dx)
        c5_new = evaluate_erdos_solution(h_new, dx)
        delta = c5_new - c5_old
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            return h_new
        return h
    
    patience = 50
    no_improvement = 0
    temp = 0.1
    cooling_rate = 0.95
    min_temp = 1e-6
    
    for res_idx, n_points in enumerate(resolutions):
        if n_points > current_n:
            # Upscale previous solution
            h_upscaled = np.interp(np.linspace(0, 1, n_points), 
                                   np.linspace(0, 1, current_n), 
                                   h_current)
            h_current = h_upscaled
            dx = 2.0 / n_points
            current_n = n_points
        else:
            dx = 2.0 / current_n
        
        # SA at this resolution
        for iteration in range(2000):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_current = sa_step(h_current, temp)
            
            # Update best solution
            current_c5 = evaluate_erdos_solution(h_current, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                no_improvement = 0
            else:
                no_improvement += 1
                
            if no_improvement >= patience:
                break
                
            temp = max(min_temp, temp * cooling_rate)
            
        # Local refinement with SLSQP
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - current_n / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(current_n)]
            
            res = minimize(objective, h_current, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_current = res.x
                current_c5 = objective(h_current)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_current.copy()
        except Exception:
            pass
        
        if time() - start_time > budget_s * 0.95:
            break
    
    return (list(best_h), best_c5, current_n)
