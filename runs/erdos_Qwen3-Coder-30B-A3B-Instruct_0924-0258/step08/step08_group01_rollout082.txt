import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    crossover_rate = 0.8
    mutation_rate = 0.1
    max_generations = 1000
    stagnation_limit = 100
    refinement_interval = 100
    
    # Initialize population
    population = []
    fitnesses = []
    
    # Add initial solution
    population.append(h_start.copy())
    fitnesses.append(evaluate_erdos_solution(h_start, dx))
    
    # Fill population with random solutions
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        h_rand = np.clip(h_rand, 0, 1)
        population.append(h_rand)
        fitnesses.append(evaluate_erdos_solution(h_rand, dx))
    
    # Main GA loop
    generation = 0
    stagnation_count = 0
    best_fitness = min(fitnesses)
    
    while time() - start_time < budget_s * 0.7:
        # Evaluate all individuals
        for i in range(len(population)):
            if i == 0 or time() - start_time > budget_s * 0.7:
                break
            fitnesses[i] = evaluate_erdos_solution(population[i], dx)
        
        # Selection (tournament)
        selected = []
        for _ in range(population_size):
            idxs = np.random.choice(len(population), 3, replace=False)
            tournament = [fitnesses[i] for i in idxs]
            winner_idx = idxs[np.argmin(tournament)]
            selected.append(population[winner_idx].copy())
        
        # Crossover
        next_pop = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < len(selected) else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Blend crossover
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
                
            next_pop.extend([child1, child2])
        
        # Keep population size
        next_pop = next_pop[:population_size]
        
        # Mutation
        mutation_step = 0.05 * (0.95 ** generation)
        for i in range(len(next_pop)):
            if np.random.rand() < mutation_rate:
                # Add noise
                noise = np.random.normal(0, mutation_step, n_points)
                next_pop[i] += noise
                # Ensure feasibility
                next_pop[i] = np.clip(next_pop[i], 0, 1)
                # Renormalize
                next_pop[i] = next_pop[i] * (n_points / 2) / np.sum(next_pop[i])
        
        # Update population
        population = next_pop
        
        # Evaluate new population
        new_fitnesses = []
        for h in population:
            new_fitnesses.append(evaluate_erdos_solution(h, dx))
        
        fitnesses = new_fitnesses
        
        # Track best
        min_fitness = min(fitnesses)
        if min_fitness < best_fitness:
            best_fitness = min_fitness
            best_idx = np.argmin(fitnesses)
            best_h = population[best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > stagnation_limit:
            break
            
        generation += 1
        
        # Periodic refinement
        if generation % refinement_interval == 0:
            # Try to improve best with local optimization
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 500, 'ftol': 1e-12})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-14})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
