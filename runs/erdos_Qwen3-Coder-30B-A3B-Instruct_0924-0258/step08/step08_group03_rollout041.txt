import numpy as np
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search setup
    low = 0.38
    high = 0.38085
    tol = 1e-5
    max_iterations = 100
    
    # Sample shifts for SDP constraints
    shifts = np.random.randint(1, n_points, 100)
    
    # SDP solver
    solver = 'MOSEK'  # fastest available
    
    # Iteratively refine using binary search
    for _ in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        
        # SDP formulation
        h_var = cp.Variable(n_points)
        Y = cp.Variable((n_points, n_points), symmetric=True)
        
        # Constraints
        constraints = [
            Y >> 0,  # Positive semidefinite
            cp.diag(Y) == h_var,  # Diagonal constraint
            cp.sum(h_var) == n_points / 2,  # Measure constraint
        ]
        
        # Overlap constraints for sampled shifts
        for k in shifts:
            # Consider Y matrix elements for appropriate shift
            if k < n_points:
                # Sum of h_i * (1 - h_{i+k}) <= mid * dx
                # Needs to account for convolution structure
                shift_sum = cp.sum(cp.multiply(h_var, 1 - h_var[k:]))
                # But we're equivalently summing over valid indices
                # For simplicity, keep it approximate via matrix entries
                # Let's rebuild matching original correlation computation:
                # For i = 0 to n_points - 1, we want to add h[i]*(1-h[i+k]) 
                # for valid i+k
                subsum = 0
                for i in range(n_points - k):
                    subsum += h_var[i] * (1 - h_var[i + k])
                constraints.append(subsum <= mid * dx)
        
        # Objective - minimize mid (but just feasible check)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                # Extract solution
                h_sdp = h_var.value
                if h_sdp is not None:
                    h_sdp = np.clip(h_sdp, 0, 1)
                    h_sdp = h_sdp * (n_points / 2) / np.sum(h_sdp)
                    
                    # Local refinement
                    c5_sdp = evaluate_erdos_solution(h_sdp, dx)
                    if c5_sdp < best_c5:
                        best_c5 = c5_sdp
                        best_h = h_sdp.copy()
                    # Refine further with gradient descent if close
                    if c5_sdp < mid:
                        h_refined = h_sdp.copy()
                        for _ in range(1000):
                            if time() - start_time > budget_s * 0.9:
                                break
                            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
                            max_lag_idx = np.argmax(corr)
                            max_lag = max_lag_idx - (n_points - 1)
                            
                            grad = np.zeros(n_points)
                            for i in range(n_points):
                                j = i + max_lag
                                if 0 <= j < n_points:
                                    grad[i] = 1 - 2 * h_refined[i] - h_refined[j]
                                else:
                                    grad[i] = 1 - 2 * h_refined[i]
                            
                            h_new = h_refined - 0.005 * grad
                            h_new = np.clip(h_new, 0, 1)
                            h_new = h_new * (n_points / 2) / np.sum(h_new)
                            
                            c5_new = evaluate_erdos_solution(h_new, dx)
                            if c5_new < best_c5:
                                best_c5 = c5_new
                                best_h = h_new.copy()
                            h_refined = h_new.copy()
                            
                    high = min(high, mid)
                else:
                    low = mid
            else:
                low = mid
        except:
            low = mid
            
        if abs(high - low) < tol:
            break
            
    # Final pass with local optimization
    h_local = best_h.copy()
    for _ in range(1000):
        if time() - start_time > budget_s * 0.95:
            break
        corr = np.correlate(h_local, 1 - h_local, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h_local[i] - h_local[j]
            else:
                grad[i] = 1 - 2 * h_local[i]
        
        h_new = h_local - 0.005 * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_local = h_new.copy()
    
    return (list(best_h), best_c5, n_points)
