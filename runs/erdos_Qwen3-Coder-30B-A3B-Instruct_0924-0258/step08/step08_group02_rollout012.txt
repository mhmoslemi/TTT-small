import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Binary initialization
    h_binary = np.zeros(n_points)
    ones_count = n_points // 2
    h_binary[:ones_count] = 1.0
    np.random.shuffle(h_binary)
    
    # Phase 2: Local search with binary-mixed optimization
    max_iterations = 500
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps
        overlaps = np.correlate(h_binary, 1 - h_binary, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for the maximizer lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h_binary[j] - h_binary[i])
        
        # Update with threshold-based approach
        threshold = 0.05
        update_mask = np.abs(gradient) > threshold
        h_updated = h_binary.copy()
        
        for i in range(n_points):
            if update_mask[i]:
                if gradient[i] > 0:
                    h_updated[i] = min(1.0, h_updated[i] + 0.02)
                else:
                    h_updated[i] = max(0.0, h_updated[i] - 0.02)
        
        # Normalize to sum constraint
        current_sum = np.sum(h_updated)
        if current_sum > 0:
            h_updated = h_updated * (n_points/2) / current_sum
            
        # Enforce bounds
        h_updated = np.clip(h_updated, 0, 1)
        
        # Track best solution
        current_c5 = evaluate_erdos_solution(h_updated, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_updated.copy()
            
        h_binary = h_updated.copy()
        
        # Adaptive step size
        if iter % 20 == 0:
            if best_c5 < 0.381:
                # More aggressive refinement
                h_binary = np.clip(h_binary + np.random.normal(0, 0.005, n_points), 0, 1)
            else:
                # Start from random binary
                h_binary = np.zeros(n_points)
                ones_count = n_points // 2
                h_binary[:ones_count] = 1.0
                np.random.shuffle(h_binary)
    
    # Phase 3: Fine-tune with L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    # Final pass: keep binary solution if it's better
    final_h = best_h.copy()
    if best_c5 < 0.381:
        # Post-process to create binary-like solution
        # Sort values and set median cut to get approximately n/2 ones
        sorted_indices = np.argsort(final_h)
        num_ones = n_points // 2
        final_h_binary = np.zeros(n_points)
        final_h_binary[sorted_indices[-num_ones:]] = 1.0
        final_h = final_h_binary
    
    return (list(final_h), best_c5, n_points)
