import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach with simulated annealing
    resolutions = [35, 70, 140, 280]
    patience = 100
    no_improvement = 0
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to current resolution if needed
        if res > n_points:
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            h_interp = np.interp(x_new, x_old, best_h)
            h_current = h_interp
        else:
            h_current = best_h[:res]
        
        n_points = res
        dx = 2.0 / n_points
        
        # Simulated Annealing parameters
        temp = best_c5 * 0.1
        cooling_rate = 0.995
        max_iter = 1000 if res == 35 else 500
        
        h = h_current.copy()
        h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate neighbor solution
            h_new = h.copy()
            # Random perturbation
            idx = np.random.randint(0, n_points)
            perturbation = np.random.normal(0, temp * 0.5)
            h_new[idx] += perturbation
            
            # Ensure bounds and sum constraint
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate overlap
            c5_new = evaluate_erdos_solution(h_new, dx)
            c5_current = evaluate_erdos_solution(h, dx)
            
            # Metropolis criterion
            if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / temp):
                h = h_new
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
                    no_improvement = 0
                else:
                    no_improvement += 1
            else:
                no_improvement += 1
                
            # Cooling schedule
            temp *= cooling_rate
            
            if no_improvement >= patience:
                break
                
        # Local refinement with SLSQP
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res_slsqp = minimize(objective, h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-9})
            
            if res_slsqp.success:
                h_refined = res_slsqp.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
            
        # Early termination if time is up
        if time() - start_time > budget_s * 0.95:
            break
    
    return (list(best_h), best_c5, n_points)
