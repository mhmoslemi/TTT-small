import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_onto_feasible_set(h, target_sum):
    # Clip to [0,1] first
    h = np.clip(h, 0, 1)
    
    # Then adjust to meet sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Simple scaling method with clipping
    scale_factor = target_sum / current_sum
    h_scaled = h * scale_factor
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled

def compute_gradient(h, max_lags, n_points, dx):
    grad = np.zeros(n_points)
    for lag in max_lags:
        if lag >= 0:
            # h[i] contributes to overlap with h[i+lag]
            for i in range(n_points - lag):
                grad[i] += 1 - h[i + lag]
                grad[i + lag] -= h[i]
        else:
            # lag is negative, h[i+lag] contributes to overlap with h[i]
            for i in range(-lag, n_points):
                grad[i] += 1 - h[i + lag]
                grad[i + lag] -= h[i]
    return grad * dx

def line_search(h, grad, max_lags, n_points, dx, c5_prev, max_iter=10):
    c = 1e-4
    alpha = 1.0
    step_size = 1.0
    
    for _ in range(max_iter):
        h_new = h - step_size * grad
        h_new = project_onto_feasible_set(h_new, n_points/2)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Armijo condition: f(x + alpha*p) <= f(x) + c*alpha*grad^T*p
        if c5_new <= c5_prev + c * step_size * np.dot(grad, -grad):
            return step_size, h_new, c5_new
        step_size *= 0.5
    
    return step_size, h, c5_prev

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = project_onto_feasible_set(h_current, n_points/2)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    max_iter = 50000
    iter_count = 0
    step_size = 0.1
    no_improve_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all correlations
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Get top 5 lags
        top_k = 5
        max_lag_indices = np.argpartition(corr, -top_k)[-top_k:]
        max_lags = [i - (n_points - 1) for i in max_lag_indices]
        # Compute combined gradient
        gradient = compute_gradient(h_current, max_lags, n_points, dx)
        
        # Line search for optimal step size
        step_size, h_new, c5_new = line_search(
            h_current, gradient, max_lags, n_points, dx, 
            evaluate_erdos_solution(h_current, dx)
        )
        
        # Project new h onto feasible set
        h_new = project_onto_feasible_set(h_new, n_points/2)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject the step
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            # Perturb occasionally to escape local minima
            if no_improve_count % 200 == 0:

        h_current = h_new
        iter_count += 1
