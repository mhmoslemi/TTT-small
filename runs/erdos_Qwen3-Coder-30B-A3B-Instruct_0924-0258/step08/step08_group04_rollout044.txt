import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # SDP approach
    # Variables
    h = cp.Variable(n_points, bounds=(0, 1))
    
    # Constraint: sum = n_points / 2
    sum_constraint = cp.sum(h) == n_points / 2
    
    # For each k compute max overlap constraints
    # Overlap for lag k: sum over i of h[i]*(1-h[i+k])  
    # = sum h[i] - sum h[i]*h[i+k]
    # We add constraint that this is <= C5*dx
    # so (sum h[i] - sum h[i]*h[i+k]) * dx <= C5 * dx
    # so sum h[i] - sum h[i]*h[i+k] <= C5
    
    # SDP approach: relax h[i]*h[i+k] to t[i,k]
    # minimize C5
    # such that sum h[i] - sum t[i,k] <= C5 for all k
    # and t[i,k] >= h[i] + h[i+k] - 1 (valid relaxation)
    # and t[i,k] <= h[i] (valid relaxation)
    # and t[i,k] <= h[i+k] (valid relaxation)
    
    # We'll try a simpler approach: formulate specific constraints
    # Place the problem into CVXPY framework with MOSEK solver
    
    # Set up problem with few constraints at first to see if solver works
    max_lag = n_points - 1
    # Create constraints for lags 0 to max_lag
    # But actually since H is periodic with period 2, we care about lags
    # in [-n_points+1, n_points-1]
    
    C5 = cp.Variable()
    
    # Create constraints: C5 >= sum(h[i](1-h[i+k])) * dx for all valid k
    constraints = []
    constraints.append(sum_constraint)
    
    # Only evaluate for relevant lags
    for k in range(-n_points + 1, n_points):
        # For lag k, compute sum over valid i of h[i]*(1-h[i+k])
        # For i-th element, it contributes to lag k if there exists (i+k)th term
        # The contributing terms are from i=0 to (n_points - abs(k))
        # i must also be within [0, n_points-1] and (i+k) must be in [0, n_points-1]
        # So i ranges from max(0, -k) to min(n_points-1, n_points-1-k)
        if k >= 0:
            valid_range = range(max(0, k), min(n_points, n_points - k))
        else:
            # k < 0 means (i+k) must be in [0, n_points-1]  
            # So i must be in [-k, n_points-1+k]
            valid_range = range(max(0, -k), min(n_points, n_points - k))
        
        # Compute overlap = sum h[i] * (1 - h[i + k]) = sum h[i] - sum h[i] * h[i+k]
        # We have sum h[i] = n_points/2 fixed
        # So overlap = n_points/2 - sum h[i] * h[i+k]
        # Constraint becomes: n_points/2 - sum h[i] * h[i+k] <= C5 * dx
        # or: sum h[i] * h[i+k] >= (n_points/2 - C5*dx)
        
        # Use relaxation: max sum h[i] * h[i+k] <= (sum h[i] +	sum h[i+k])/2  
        # sum h[i] * h[i+k] <= 0.5*(sum h[i] + sum h[i+k]) valid for [0,1]
        # But that would be quite loose, so better to formulate specific constraints.
        
        # Exact constraints using SDP would be more involved.
        # Let's take a simpler approach: sample constraints to cover the space.
        # points are at i indices. Valid indices for lag k 
        term_sum = 0
        for i in valid_range:
            if k >= 0:  
                term_sum += h[i] * h[i + k]
            else:
                # k < 0, so i+k >= 0 and i must be such that i+k is valid
                term_sum += h[i] * h[i + k]
        # This is the sum for that lag
        # So overlap = sum h[i] - term_sum = n_points/2 - term_sum
        # So we want: n_points/2 - term_sum <= C5 * dx
        # i.e., C5 >= (n_points/2 - term_sum)/dx
        # to minimize C5: minimize (n_points/2 - term_sum)/dx
        
        # But CVXPY does this by minimizing variable subject to constraints
        # So we want: n_points/2 - term_sum <= C5 * dx
        # i.e., term_sum >= (n_points/2 - C5 * dx) 
        # But want relation in opposite direction
        # try: n_points/2 - term_sum <= C5 * dx -> C5 >= (n_points/2 - term_sum)/dx
        # also: C5 >= 0, so we maximize the RHS
        constraints.append(term_sum + C5 * dx >= n_points / 2)
    
    # Dummy objective to minimize C5
    objective = cp.Minimize(C5)
    
    # Create SDP problem
    prob = cp.Problem(objective, constraints)
    
    # Attempt solving SDP
    solver_timeout = budget_s - 10
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=solver_timeout)
        # Check solution
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value
            if h_sol is not None and np.all(0 <= h_sol) and np.all(h_sol <= 1):
                # Renormalize
                renorm_factor = (n_points/2) / np.sum(h_sol)
                h_sol = h_sol * renorm_factor 
                c5_sol = C5.value 
                if isinstance(c5_sol, (int, float)) and not np.isnan(c5_sol) and not np.isinf(c5_sol):
                    if c5_sol < best_c5:
                        best_c5 = c5_sol
                        best_h = h_sol.copy()
    except Exception:
        pass

    # Try another optimization approach if needed
    # start with more optimized version of gradient descent
    if time() - start_time > budget_s * 0.95:
        return (list(best_h), best_c5, n_points)
        
    # Try optimization with gradient descent again but different
    # hyperparameters and carefully chosen refine steps
    
    # Since we have good SDP solution lets try it as starting point
    h_current = best_h.copy() if np.sum(best_h) > 0 else h_start.copy() 
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Decrease step size
    step_size = 0.005
    max_iter = 5000
    iter_count = 0
    
    # Keep track of improvements
    last_c5 = evaluate_erdos_solution(h_current, dx)
    improvement_threshold = 1e-5
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute correlation
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Compute approximate gradient
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_current[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= h_current[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_current[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_current[i]
        
        # Apply gradient step
        h_new = h_current - step_size * gradient
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        elif c5_new >= last_c5 and step_size > 1e-6:
            step_size *= 0.8
        else:
            h_current = h_new
            if step_size < 0.01:
                step_size *= 1.03
        last_c5 = c5_new
        iter_count += 1

    return (list(best_h), best_c5, n_points)
