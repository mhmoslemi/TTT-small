import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Global search using simulated annealing with restarts
    max_iter = 10000
    max_restarts = 10
    patience = 100
    
    for restart in range(max_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        h = h_start.copy()
        h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
        
        current_c5 = evaluate_erdos_solution(h, dx)
        temp = 0.1
        no_improvement = 0
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb randomly
            h_new = h.copy()
            # Modify some elements
            n_mod = max(1, n_points // 10)
            indices = np.random.choice(n_points, n_mod, replace=False)
            for i in indices:
                h_new[i] += np.random.normal(0, 0.05)
                h_new[i] = np.clip(h_new[i], 0, 1)
            
            h_new = h_new * (n_points/2) / np.sum(h_new)  # Ensure sum constraint
            
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject based on simulated annealing
            if new_c5 < current_c5 or np.random.random() < np.exp(-(new_c5 - current_c5) / temp):
                h = h_new
                current_c5 = new_c5
                no_improvement = 0
            else:
                no_improvement += 1
                
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            if no_improvement >= patience:
                break
                
            temp *= 0.995
        
        # Local refinement
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
        
        if time() - start_time > budget_s * 0.95:
            break
    
    return (list(best_h), best_c5, n_points)
