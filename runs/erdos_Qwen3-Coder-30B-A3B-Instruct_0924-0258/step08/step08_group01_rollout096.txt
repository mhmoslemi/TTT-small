import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale optimization
    max_scales = int(np.log2(n_points)) + 2
    scale = 1
    h_current = h_start.copy()
    perturb_mag = 0.05
    no_improvement_count = 0
    max_no_improvement = 20
    
    while scale <= max_scales and (time() - start_time) < budget_s * 0.95:
        current_n = 2 ** scale
        if current_n > 1000:
            break
            
        # Upsample the current solution
        h_upsampled = np.interp(np.linspace(0, 1, current_n), 
                               np.linspace(0, 1, len(h_current)), 
                               h_current)
        
        # Perturb the upsampled solution
        h_perturbed = h_upsampled + perturb_mag * (2 * np.random.random(current_n) - 1)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (current_n / 2) / np.sum(h_perturbed)
        
        # Evaluate
        c5_perturbed = evaluate_erdos_solution(h_perturbed, 2.0 / current_n)
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
            no_improvement_count = 0
            perturb_mag = 0.05  # Reset perturbation magnitude
        else:
            no_improvement_count += 1
            perturb_mag *= 0.95  # Reduce perturbation magnitude
            
        # SQP refinement
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / current_n)
            
        def constraint_sum(h):
            return np.sum(h) - current_n / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(current_n)]
        
        try:
            res = minimize(objective, h_perturbed, method='SLSQP',
                          bounds=bounds, constraints=cons_sum,
                          options={'maxiter': 50, 'ftol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined.copy()
        except Exception:
            pass
            
        if no_improvement_count >= max_no_improvement:
            break
            
        scale += 1
    
    # Final local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
        
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP',
                      bounds=bounds, constraints=cons_sum,
                      options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
