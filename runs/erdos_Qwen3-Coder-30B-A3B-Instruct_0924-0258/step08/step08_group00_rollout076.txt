import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution simulated annealing with local refinement
    resolutions = [35, 70, 140, 280]
    current_h = h_start.copy()
    current_n = n_points
    
    patience = 50
    no_improvement = 0
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Upscale if needed
        if res > current_n:
            new_h = np.interp(np.linspace(0, 1, res), np.linspace(0, 1, current_n), current_h)
            current_h = new_h
            current_n = res
            dx = 2.0 / current_n
        
        # Annealing parameters
        temp = best_c5 * 0.5
        cooling_rate = 0.98
        max_iter = 500 if res == resolutions[0] else 200
        
        h = current_h.copy()
        h = h * (current_n/2) / np.sum(h)  # Ensure sum constraint
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb h
            h_new = h.copy()
            idx = random.randint(0, current_n - 1)
            delta = np.random.normal(0, temp * 0.1)
            h_new[idx] += delta
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (current_n/2) / np.sum(h_new)
            
            # Compute new objective
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                no_improvement = 0
                if res == resolutions[-1]:  # Last resolution, do refinement
                    # Local refinement with SLSQP
                    try:
                        def objective(h):
                            return evaluate_erdos_solution(h, dx)
                        
                        def constraint_sum(h):
                            return np.sum(h) - current_n / 2.0
                        
                        cons_sum = {'type': 'eq', 'fun': constraint_sum}
                        bounds = [(0, 1) for _ in range(current_n)]
                        
                        res_min = minimize(objective, h, method='SLSQP', 
                                           bounds=bounds, constraints=cons_sum, 
                                           options={'maxiter': 100, 'ftol': 1e-9})
                        
                        if res_min.success:
                            h_final = res_min.x
                            c5_final = objective(h_final)
                            if c5_final < best_c5:
                                best_h = h_final
                                best_c5 = c5_final
                    except Exception:
                        pass
            else:
                # Accept with probability based on temperature
                if random.random() < np.exp(-(c5_new - best_c5) / (temp + 1e-10)):
                    h = h_new
                no_improvement += 1
                
            temp *= cooling_rate
            
            if no_improvement >= patience:
                break
                
        current_h = h
        
    # Final polishing if needed
    if time() - start_time < budget_s * 0.95:
        try:
            h_final = current_h.copy()
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - current_n / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(current_n)]
            
            res = minimize(objective, h_final, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, current_n)
