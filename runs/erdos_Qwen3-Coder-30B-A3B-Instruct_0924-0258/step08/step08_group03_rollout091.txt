import numpy as np
from time import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # SDP-based approach for tightening the bound
    def solve_sdp(c5_bound):
        # SDP formulation
        Y = cp.Variable((n_points, n_points), symmetric=True)
        h_vec = cp.Variable(n_points)
        
        # Objective: minimize C5 (we're checking feasibility)
        constraints = [
            Y >> 0,  # Y is PSD
            cp.diag(Y) == h_vec,  # diagonal constraint
            cp.sum(h_vec) == n_points / 2,  # normalization constraint
        ]
        
        # Overlap constraints
        for k in range(n_points):
            if k == 0:
                # For k=0, we have sum(h * (1-h)) 
                overlap = cp.sum(cp.multiply(h_vec, 1 - h_vec))
            else: 
                # For k>0, we compute the shifted overlap
                # sum_i h_i (1 - h_{i+k}) = sum h_i - sum h_i h_{i+k}
                sum_h = cp.sum(h_vec) 
                sum_hh = cp.sum(cp.multiply(h_vec, cp.roll(h_vec, k)))
                overlap = sum_h - sum_hh
            constraints.append(overlap <= c5_bound * dx)
            
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return h_vec.value, True
        except:
            pass
        return None, False
    
    # Local refinement
    def refine_with_gradient(h_initial, c5_target):
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
            
        def constraint_fun(h):
            return np.sum(h) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint_fun}
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(objective, h_initial, method='SLSQP', 
                         bounds=bounds, constraints=cons, 
                         options={'maxiter': 500, 'ftol': 1e-9})
        return result.x
    
    # Start with binary search over the target C5
    low = 0.38
    high = 0.38085
    # Since solution time is bounded (~300s is max assumed to be used in binary search)
    # perform a softer binary search
    
    iteration = 0
    while time() - start_time < budget_s * 0.9:
        iteration += 1
        mid = (low + high) / 2.0
        h_sol, feasible = solve_sdp(mid)
        
        if feasible and h_sol is not None:
            # Refine and validate
            h_final = np.clip(h_sol, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            c5_refined = evaluate_erdos_solution(h_final, dx)
            
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_final.copy()
                
            high = mid
        else:
            low = mid
            
        # Exit if tolerance met
        if (high - low) < 1e-6:
            break
    
    # Final refinement
    # Local optimization to possibly improve further
    if time() - start_time < budget_s * 0.99:
        local_h = refine_with_gradient(best_h, best_c5)
        local_c5 = evaluate_erdos_solution(local_h, dx)
        if local_c5 < best_c5:
            best_c5 = local_c5
            best_h = local_h.copy()
    
    return (list(best_h), best_c5, n_points)
