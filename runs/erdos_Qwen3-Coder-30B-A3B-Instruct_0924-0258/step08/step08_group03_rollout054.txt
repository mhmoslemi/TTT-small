import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Parameters
    max_iter = 1000
    k_max = 20  # Top k overlaps to consider in convex optimization
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Find top k overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        top_k_indices = np.argsort(corr)[-k_max:]
        
        # Create convex optimization problem
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Objective: minimize sum of top k overlaps
        objective = 0
        for idx in top_k_indices:
            # Compute the correlation contribution at this lag
            lag = idx - (n_points - 1)
            # This is a simplified approximation for convex optimization
            # Using standard linearization approach
            pass  # We directly use the gradient to speed up
        
        # Gradient computation and optimization step
        grad = np.zeros(n_points)
        for i in range(n_points):
            for k in top_k_indices:
                lag = k - (n_points - 1)
                j = i + lag
                if 0 <= j < n_points:
                    grad[i] += (h[j] - 1)  # gradient of overlapping term with respect to h[i]
                else:
                    grad[i] += -1  # penalty for out of bounds
        
        # Optimization step
        step_size = 0.001
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Apply low-pass filtering to smooth
        if iteration % 10 == 0:
            kernel = np.ones(5) / 5.0
            h_new = np.convolve(h_new, kernel, mode='same')
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        
        # Increase top-k gradually
        if iteration % 20 == 0 and k_max < 50:
            k_max = min(k_max + 1, 50)
    
    # Final refinement
    for _ in range(10):
        if time() - start_time > budget_s * 0.95:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Gradient for this lag only (simplified for efficiency)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h[i] - h[j]
            else:
                grad[i] = 1 - 2 * h[i]
        
        step_size = 0.001 * (1.0 - (time() - start_time) / budget_s)

    if final_c5 < best_c5:

    return (list(best_h), best_c5, n_points)
