import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    # Project h to feasible set ensuring 0 <= h[i] <= 1 and sum(h) = n_points/2
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to hit the exact sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        # Simple scaling approach
        if current_sum > 0:
            h = h * (n_points/2) / current_sum
        else:
            h = np.ones_like(h) * (n_points/2) / n_points
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Line search parameters
    max_line_search = 10
    backtracking_factor = 0.5
    min_step_size = 1e-10
    max_no_improvement = 2000
    no_improvement_count = 0
    perturbation_interval = 500
    perturbation_magnitude = 0.05
    k_lags = 5  # Top k lags to consider
    
    step_size = 0.01
    
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95:
        # Compute all correlations for all possible lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Get indices and values of top K lags by overlap
        lag_indices = np.arange(len(corr)) - (len(h_current) - 1)
        sorted_indices = np.argsort(corr)[::-1][:k_lags]
        top_lags = lag_indices[sorted_indices]
        
        # Compute combined gradient for top lags
        gradient = np.zeros(n_points)
        for lag in top_lags:
            if lag >= 0:
                # For h[i] term in overlap with h[i+lag] 
                # gradient wrt h[i] += (1 - h[i+lag])
                # gradient wrt h[i+lag] -= h[i]
                for i in range(n_points - lag):
                    gradient[i] += 1 - h_current[i + lag]
                    gradient[i + lag] -= h_current[i]
            else:
                # lag < 0
                # gradient wrt h[i] -= h[i-lag] 
                # gradient wrt h[i-lag] += (1 - h[i])
                for i in range(-lag, n_points):
                    gradient[i] += 1 - h_current[i + lag]
                    gradient[i + lag] -= h_current[i]
        
        # Line search to find the optimal step size
        success = False
        current_c5 = evaluate_erdos_solution(h_current, dx)
        for _ in range(max_line_search):
            h_new = h_current - step_size * gradient
            h_new = project_to_feasible(h_new, n_points, dx)
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < current_c5:
                h_current = h_new
                no_improvement_count = 0
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_current.copy()
                step_size *= 1.05  # Increase step size
                success = True
                break
            else:
                step_size *= backtracking_factor
        
        if not success:
            step_size *= backtracking_factor
            if step_size < min_step_size:
                break
        
        iter_count += 1
        no_improvement_count += 1
        
        # Periodically perturb to escape local minima
        if iter_count % perturbation_interval == 0 and no_improvement_count > 100:
            perturbation = np.random.normal(0, perturbation_magnitude, n_points)
            h_current += perturbation
            h_current = project_to_feasible(h_current, n_points, dx)
            no_improvement_count = 0
            # Reset step size
            step_size = 0.01
    
    return (list(best_h), best_c5, n_points)
