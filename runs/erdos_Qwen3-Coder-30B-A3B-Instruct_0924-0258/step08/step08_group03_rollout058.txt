import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search bounds
    low, high = 0.38, 0.38085
    tolerance = 1e-5
    
    # Sample shifts for feasibility check - 100 random shifts
    shifts = np.random.choice(n_points, size=100, replace=False)
    
    # Binary search loop
    while high - low > tolerance and time() - start_time < budget_s * 0.9:
        mid = (low + high) / 2
        
        # Formulate CVXPY feasibility problem
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Add overlap constraints
        for k in shifts:
            # Calculate the correlation
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h_var[i] * (1 - h_var[j])
            constraints.append(overlap_expr <= mid * dx)
        
        # Dummy objective (feasibility problem)
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.SOLVED:
                # Accept this solution and try to improve
                h_new = h_var.value
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                high = mid
            else:
                low = mid
        except:
            low = mid
            
        if time() - start_time > budget_s * 0.9:
            break
    
    # Local refinement with gradient descent
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def gradient(h_vec):
        # Simple finite difference gradient
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        base_val = objective(h_vec)
        for i in range(len(h_vec)):
            h_test = h_vec.copy()
            h_test[i] += eps
            # Renormalize
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            grad[i] = (objective(h_test) - base_val) / eps
        return grad
    
    # Golden section line search setup
    def line_search(h_vec, d_vec, alpha_max=1.0):
        def phi(alpha):
            h_new = h_vec + alpha * d_vec
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            return objective(h_new)
        
        # Golden ratio method
        gr = 1.618033988749895
        a, b = 0.0, alpha_max
        resphi = 2.0 - gr
        
        x1 = resphi * a + gr * b
        x2 = gr * a + resphi * b
        
        f1 = phi(x1)
        f2 = phi(x2)
        
        while abs(b - a) > 1e-8:
            if f1 < f2:
                b = x2
                x2 = x1
                f2 = f1
                x1 = resphi * a + gr * b
                f1 = phi(x1)
            else:
                a = x1
                x1 = x2
                f1 = f2
                x2 = gr * a + resphi * b
                f2 = phi(x2)
        
        return (a + b) / 2
    
    # Refinement iterations
    h_refine = best_h.copy()
    max_refine = 500
    step_size = 0.1
    
    for i in range(max_refine):
        if time() - start_time > budget_s * 0.95:
            break
        
        grad = gradient(h_refine)
        d = -grad
        alpha = line_search(h_refine, d)
        h_new = h_refine + alpha * d
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        h_refine = h_new
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
