import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary initialization
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    
    # Phase 1: Binary search optimization with thresholding
    n_iterations = 150
    for iter in range(n_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for maximum overlap
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Apply threshold-based updates
        threshold = 0.01 + (0.1 - 0.01) * (1 - iter / n_iterations)
        mask = np.abs(gradient) > threshold
        
        # Update h using gradient
        h_new = h - 0.05 * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check if update improves solution
        current_c5 = evaluate_erdos_solution(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        
        h = h_new.copy()
    
    # Phase 2: Local refinement with L-BFGS-B and binary projection
    try:
        def objective(h):
            return np.max(np.correlate(h, 1 - h, mode='full') * dx)
        
        # Ensure sum constraint
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Refinement pass
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
        # Binary projection
        binary_h = np.round(best_h)
        binary_h = binary_h * (n_points/2) / np.sum(binary_h)
        binary_c5 = evaluate_erdos_solution(binary_h, dx)
        if binary_c5 < best_c5:
            best_h = binary_h
            best_c5 = binary_c5
            
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
