import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm Parameters
    population_size = 50
    generations = 500
    crossover_rate = 0.8
    mutation_rate = 0.1
    elitism_rate = 0.1
    max_no_improvement = 100
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    if 'initial_h_values' in globals():
        population = [h_start.copy()]
        while len(population) < population_size:
            population.append(create_individual())
    else:
        population = [create_individual() for _ in range(population_size)]
    
    # Elitism
    def select_parents(population, fitnesses):
        tournament_size = 3
        selected = []
        for _ in range(population_size):
            tournament = random.sample(list(zip(population, fitnesses)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0].copy())
        return selected
    
    # Blend crossover
    def blend_crossover(parent1, parent2, alpha=0.5):
        offspring1 = np.zeros_like(parent1)
        offspring2 = np.zeros_like(parent1)
        for i in range(len(parent1)):
            gamma = (1 + 2 * alpha) * random.random() - alpha
            offspring1[i] = parent1[i] + gamma * (parent2[i] - parent1[i])
            offspring2[i] = parent2[i] + gamma * (parent1[i] - parent2[i])
        return np.clip(offspring1, 0, 1), np.clip(offspring2, 0, 1)
    
    # Mutation with normalization
    def mutate(individual, mutation_rate, generation, max_generations):
        step_size = 0.05 * (1 - generation / max_generations)
        for i in range(len(individual)):
            if random.random() < mutation_rate:
                individual[i] += np.random.normal(0, step_size)
        # Normalize
        individual = individual * (n_points / 2) / np.sum(individual)
        return np.clip(individual, 0, 1)
    
    no_improvement_count = 0
    best_fitness_history = []
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate fitness
        fitnesses = []
        for individual in population:
            fitness = evaluate_erdos_solution(individual, dx)
            fitnesses.append(fitness)
        
        # Track best
        min_fitness = min(fitnesses)
        best_fitness_history.append(min_fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[np.argmin(fitnesses)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            break
        
        # Elitism
        elite_count = int(elitism_rate * population_size)
        elite_indices = np.argsort(fitnesses)[:elite_count]
        elite = [population[i].copy() for i in elite_indices]
        
        # Selection
        parents = select_parents(population, fitnesses)
        
        # Crossover and mutation
        new_population = elite[:]
        while len(new_population) < population_size:
            p1, p2 = random.sample(parents, 2)
            if random.random() < crossover_rate:
                o1, o2 = blend_crossover(p1, p2)
            else:
                o1, o2 = p1.copy(), p2.copy()
            
            o1 = mutate(o1, mutation_rate, gen, generations)
            o2 = mutate(o2, mutation_rate, gen, generations)
            
            new_population.append(o1)
            if len(new_population) < population_size:
                new_population.append(o2)
        
        population = new_population[:population_size]
    
    # Final local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
