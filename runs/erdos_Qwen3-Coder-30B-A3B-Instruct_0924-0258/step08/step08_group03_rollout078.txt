import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Parameters for optimization
    max_iterations = 5000
    K = 10  # Number of top overlaps to consider
    filter_bandwidth = 5
    iter_count = 0
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Get the top K overlapping indices
        corr = np.correlate(h, 1 - h, mode='full')
        top_k_indices = np.argsort(corr)[-K:][::-1]
        
        # Create convex optimization problem
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Minimize sum of top K overlaps
        objective = 0
        for idx in top_k_indices:
            if idx < 2 * n_points - 1:
                # Compute the corresponding overlap
                lag = idx - (n_points - 1)
                # Create shifted version of h_var
                overlap_sum = 0
                for i in range(n_points):
                    j = i + lag
                    if 0 <= j < n_points:
                        overlap_sum += h_var[i] * (1 - h_var[j])
                objective += overlap_sum
        
        # We're minimizing the sum of overlaps (not the max)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                
                # Renormalize to satisfy constraint
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Apply low-pass filtering to smooth
                h_filtered = np.zeros(n_points)
                for i in range(n_points):
                    total = 0.0
                    count = 0
                    for j in range(max(0, i - filter_bandwidth), min(n_points, i + filter_bandwidth + 1)):
                        weight = 1.0 / (1.0 + abs(i - j))
                        total += h_new[j] * weight
                        count += weight
                    h_filtered[i] = total / count if count > 0 else 0.0
                
                # Evaluate new solution
                c5_new = evaluate_erdos_solution(h_filtered, dx)
                
                # Accept if better
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_filtered.copy()
                    h = h_filtered.copy()
                    
                    # Increase K if we're making progress
                    if K < 25:
                        K = min(K + 1, 25)
                else:
                    # Reduce K if no progress
                    K = max(K - 1, 5)
        except Exception:
            pass
            
        iter_count += 1
        if iter_count % 100 == 0:
            if K > 10:
                K = max(K - 1, 10)
    
    # Final re-evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
