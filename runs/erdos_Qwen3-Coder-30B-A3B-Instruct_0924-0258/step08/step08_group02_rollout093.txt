import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize with normalized h_start
    h_start = h_start * (n_points/2) / np.sum(h_start)
    h = h_start.copy()
    
    # Parameters for hybrid approach
    max_iter = 200
    no_improve_iter = 0
    max_no_improve = 50
    
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convex relaxation step using CVXPY
        try:
            h_relaxed = np.zeros(n_points)
            
            # Define the convex problem for relaxation
            h_vars = cp.Variable(n_points)
            objective = cp.Minimize(
                cp.sum_squares(cp.vstack([
                    cp.sum(h_vars[i] * (1 - h_vars[(i + k) % n_points]) 
                           for i in range(n_points)]
                          ) for k in range(n_points)
                )) + 0.01 * cp.sum_squares(h_vars)
            )
            constraints = [
                cp.sum(h_vars) == n_points/2,
                h_vars >= 0,
                h_vars <= 1
            ]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_relaxed = h_vars.value
                h_relaxed = np.clip(h_relaxed, 0, 1)
                # Ensure sum constraint
                h_relaxed = h_relaxed * (n_points/2) / np.sum(h_relaxed)
        except:
            h_relaxed = h.copy()
        
        # Coordinate descent step on a subset of variables
        h_new = h.copy()
        indices = np.random.permutation(n_points)[:n_points//5]  # Optimize 20%
        
        # Create a small subproblem for the selected indices
        for i in range(n_points):
            if i not in indices:
                continue
            
            # Compute current overlap contributions
            def obj_val(h_vec):
                return np.max(np.correlate(h_vec, 1 - h_vec, mode='full')) * dx
            
            def grad(h_vec):
                overlaps = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
                max_idx = np.argmax(overlaps)
                grad_vec = np.zeros_like(h_vec)
                for j in range(n_points):
                    k = (j + max_idx) % n_points
                    grad_vec[j] = -2 * dx * (1 - h_vec[k] - h_vec[j])
                return grad_vec
            
            # For the selected index, do line search step
            if len(indices) > 1:
                # Simple gradient decent with small step
                h_test = h.copy()
                step = 0.001
                for _ in range(50):
                    g = grad(h_test)
                    h_test[i] -= step * g[i]
                    h_test = np.clip(h_test, 0, 1)
                    h_test = h_test * (n_points/2) / np.sum(h_test)
                h_new[i] = h_test[i]
        
        h = h_new.copy()
        
        # Ensure sum constraint still holds
        h = h * (n_points/2) / np.sum(h)
        
        # Evaluate and track best solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_iter = 0
        else:
            no_improve_iter += 1
            
        if no_improve_iter > max_no_improve:
            # Try random perturbation
            h = h_start.copy()
            pert = np.random.normal(0, 0.01, n_points)
            h += pert
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            no_improve_iter = 0
            
        # Local L-BFGS optimization
        if iter % 5 == 0:
            try:
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(
                    lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
                    h.copy(),
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 10, 'ftol': 1e-9}
                )
                if res.success:
                    h_refine = res.x
                    h_refine = h_refine * (n_points/2) / np.sum(h_refine)
                    c5_refine = evaluate_erdos_solution(h_refine, dx)
                    if c5_refine < best_c5:
                        best_c5 = c5_refine
                        best_h = h_refine
            except:
                pass

    return (list(best_h), best_c5, n_points)
