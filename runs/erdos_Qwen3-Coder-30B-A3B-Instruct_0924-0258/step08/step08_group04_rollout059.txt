import numpy as np
from time import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_onto_feasible(h, n_points, dx):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if abs(sum_h - n_points/2) < 1e-10:
        return h
    # Use a simple projection method that maintains the sum constraint
    # First clip to [0,1], then scale appropriately
    h_scaled = h * (n_points/2) / sum_h
    return np.clip(h_scaled, 0, 1)

def line_search(h_current, gradient, dx, n_points, max_iter=20):
    # Simple backtracking line search
    alpha = 1.0
    step_size = 0.1
    c1 = 1e-4
    max_alpha = 10.0
    
    for _ in range(max_iter):
        h_new = h_current - alpha * gradient
        h_new = project_onto_feasible(h_new, n_points, dx)
        new_c5 = evaluate_erdos_solution(h_new, dx)
        old_c5 = evaluate_erdos_solution(h_current, dx)
        if new_c5 < old_c5:
            return h_new, alpha
        alpha *= step_size
        if alpha < 1e-8:
            break
    return h_current, alpha

def get_top_k_lags(h, dx, k=5):
    corr = np.correlate(h, 1 - h, mode='full')
    lags = np.arange(-len(h) + 1, len(h))
    corr_lags = list(zip(corr, lags))
    corr_lags.sort(reverse=True)
    top_k = corr_lags[:k]
    return [lag for _, lag in top_k]

def compute_combined_gradient(h, top_lags, n_points):
    gradient = np.zeros(n_points)
    for lag in top_lags:
        if lag >= 0:
            for i in range(n_points - lag):
                gradient[i] += 1 - h[i + lag]
                if i + lag < n_points:
                    gradient[i + lag] -= h[i]
        else:
            for i in range(n_points + lag):
                gradient[i] += 1 - h[i - lag]
                if i - lag >= 0:
                    gradient[i - lag] -= h[i]
    return gradient

def perturb_h(h, rate=0.1, magnitude=0.05):
    # Add small random perturbations
    perturbation = np.random.uniform(-magnitude, magnitude, len(h))
    h_perturbed = h + rate * perturbation
    return np.clip(h_perturbed, 0, 1)
