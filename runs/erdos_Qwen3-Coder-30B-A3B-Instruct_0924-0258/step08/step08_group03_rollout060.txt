import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_coarse = 15
    n_fine = 35
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # Initialize coarse grid
    h_coarse = np.ones(n_coarse) * 0.5
    
    # Initialize fine grid - use initial_h_values if available
    if 'initial_h_values' in globals():
        h_fine = np.array(initial_h_values)
    else:
        h_fine = np.ones(n_fine) * 0.5
    
    best_c5 = float('inf')
    best_h_fine = h_fine.copy()
    
    # Coarse grid optimization
    for iteration in range(500):
        if time() - start_time > budget_s * 0.7:
            break
            
        # Compute overlaps
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_coarse - 1)
        
        # Compute gradient estimate
        grad = np.zeros(n_coarse)
        for i in range(n_coarse):
            j = i + max_lag
            if 0 <= j < n_coarse:
                grad[i] = 1 - 2 * h_coarse[i] - h_coarse[j]
            else:
                grad[i] = 1 - 2 * h_coarse[i]
        
        # Gradient descent update
        h_coarse_new = h_coarse - 0.01 * grad
        h_coarse_new = np.clip(h_coarse_new, 0, 1)
        h_coarse_new = h_coarse_new * (n_coarse / 2) / np.sum(h_coarse_new)
        
        # Evaluate
        c5 = evaluate_erdos_solution(h_coarse_new, dx_coarse)
        if c5 < best_c5:
            best_c5 = c5
            h_coarse = h_coarse_new.copy()
    
    # Upsample coarse to fine
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    f = interp1d(x_coarse, h_coarse, kind='linear', fill_value='extrapolate')
    h_fine = f(x_fine)
    
    # Renormalize fine grid
    h_fine = project_to_feasible(h_fine, n_fine)
    
    # Fine grid optimization
    for iteration in range(2000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random mutation (10% of the values)
        if iteration % 2 == 0:
            num_mutate = int(0.1 * n_fine)
            indices = np.random.choice(n_fine, num_mutate, replace=False)
            h_fine[indices] = np.random.uniform(0, 1, num_mutate)
        
        # Gradient step
        else:
            corr = np.correlate(h_fine, 1 - h_fine, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (n_fine - 1)
            
            grad = np.zeros(n_fine)
            for i in range(n_fine):
                j = i + max_lag
                if 0 <= j < n_fine:
                    grad[i] = 1 - 2 * h_fine[i] - h_fine[j]
                else:
                    grad[i] = 1 - 2 * h_fine[i]
            
            # Adaptive step size
            step_size = 0.01 if iteration < 1000 else 0.005
            h_fine = h_fine - step_size * grad
            h_fine = np.clip(h_fine, 0, 1)
            
        # Renormalize
        h_fine = project_to_feasible(h_fine, n_fine)
        
        # Evaluate
        c5 = evaluate_erdos_solution(h_fine, dx_fine)
        if c5 < best_c5:
            best_c5 = c5
            best_h_fine = h_fine.copy()
    
    final_c5 = evaluate_erdos_solution(best_h_fine, dx_fine)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h_fine), best_c5, n_fine)
