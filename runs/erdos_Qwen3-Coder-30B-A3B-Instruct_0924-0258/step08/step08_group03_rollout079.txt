import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization parameters
    max_iterations = 5000
    filter_bandwidth = 5
    k_top = 10
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convex optimization step
        # Create variables and constraints
        h_var = cp.Variable(n_points)
        
        # Constraint: 0 <= h[i] <= 1
        constraints = [h_var >= 0, h_var <= 1]
        
        # Constraint: integral = 1
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Approximate objective using top k overlaps
        # First compute current overlaps
        corr_current = np.correlate(h, 1 - h, mode='full')
        top_k_indices = np.argsort(corr_current)[-k_top:][::-1]
        
        # Create linear approximation to minimize sum of top overlaps
        objective_terms = []
        for idx in top_k_indices:
            lag = idx - (n_points - 1)
            if lag >= 0:
                # h[i] * (1 - h[i + lag]) 
                expr = cp.sum(cp.multiply(h_var, cp.roll(1 - h_var, -lag)))
            else:
                # h[i] * (1 - h[i - lag]) 
                expr = cp.sum(cp.multiply(h_var, cp.roll(1 - h_var, -lag)))
            objective_terms.append(expr)
        
        objective = cp.Minimize(cp.sum(objective_terms))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    # Apply low-pass filter
                    h_filtered = np.convolve(h_new, np.ones(filter_bandwidth)/filter_bandwidth, mode='same')
                    h_new = h_filtered
                    
                    # Renormalize
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    
                    # Clip to bounds
                    h_new = np.clip(h_new, 0, 1)
                    
                    # Evaluate
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
                        
                        # Increase k for next iteration if we're improving
                        k_top = min(k_top + 1, 25)
                        
        except:
            pass
        
        # Apply smoothing filter to current h
        if iteration % 2 == 0:
            h_smooth = np.convolve(h, np.ones(filter_bandwidth)/filter_bandwidth, mode='same')
            h = h_smooth.copy()
    
    # Final refinement
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (list(best_h), best_c5, n_points)
