import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    population_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    mutation_step = 0.05
    slsqp_freq = 100
    max_no_improvement = 200
    no_improvement_count = 0
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.2 + 0.4
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.2 + 0.4
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # GA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitnesses = []
        for h in population:
            f = evaluate_erdos_solution(h, dx)
            fitnesses.append(f)
        
        # Find best in current generation
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitnesses.index(min_fitness)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
        
        # Tournament selection
        selected = []
        for _ in range(population_size // 2):
            idx1, idx2 = np.random.choice(population_size, 2, replace=False)
            if fitnesses[idx1] < fitnesses[idx2]:
                selected.append(population[idx1])
            else:
                selected.append(population[idx2])
        
        # Create offspring
        offspring = []
        for i in range(0, len(selected), 2):
            p1 = selected[i]
            p2 = selected[(i + 1) % len(selected)]
            
            if np.random.rand() < crossover_rate:
                alpha = np.random.uniform(0.3, 0.7)
                child1 = alpha * p1 + (1 - alpha) * p2
                child2 = (1 - alpha) * p1 + alpha * p2
            else:
                child1 = p1.copy()
                child2 = p2.copy()
            
            # Apply mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_step, n_points)
                child1 += noise
                child1 = np.clip(child1, 0, 1)
                # Renormalize
                child1 = child1 * (n_points / 2) / np.sum(child1)
            
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_step, n_points)
                child2 += noise
                child2 = np.clip(child2, 0, 1)
                # Renormalize
                child2 = child2 * (n_points / 2) / np.sum(child2)
            
            offspring.append(child1)
            offspring.append(child2)
        
        # Replace population with top individuals
        all_individuals = population + offspring
        all_fitnesses = fitnesses + [evaluate_erdos_solution(h, dx) for h in offspring]
        sorted_indices = np.argsort(all_fitnesses)
        population = [all_individuals[i] for i in sorted_indices[:population_size]]
        
        # Decrease mutation rate
        mutation_step *= 0.98
        
        # Refine best solution locally
        if gen % slsqp_freq == 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 100, 'ftol': 1e-12})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
