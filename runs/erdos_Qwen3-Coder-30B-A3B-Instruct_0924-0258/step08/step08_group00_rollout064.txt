import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm with local refinement
    pop_size = 100
    generations = budget_s // 10  # Rough estimate
    patience = 50
    no_improvement = 0
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points) * 0.5 + 0.25  # Controlled variation
        h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
        h = np.clip(h, 0, 1)
        population.append(h)
    
    if 'initial_h_values' in globals():
        population[0] = h_start.copy()  # Use as first member if present
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
            
        # Sort by fitness
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]
        fitness = [fitness[i] for i in sorted_idx]
        
        # Keep best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            break
            
        # Selection (top 30%)
        elite_size = max(10, pop_size // 3)
        elite = population[:elite_size]
        
        # Create new population
        new_pop = elite.copy()
        
        # Crossover and mutation
        while len(new_pop) < pop_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(elite_size, 2, replace=False)
            parent1, parent2 = elite[idx1], elite[idx2]
            
            # Crossover (uniform)
            mask = np.random.rand(n_points) > 0.5
            child = np.where(mask, parent1, parent2)
            
            # Mutation (Gaussian noise)
            noise = np.random.normal(0, 0.02, n_points)
            child += noise
            
            # Ensure feasibility and constraint
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_pop.append(child)
            
        population = new_pop[:pop_size]
        
        # Local refinement on best individual using SLSQP
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, population[0], method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-6})
            
            if res.success:
                refined_h = res.x
                refined_fitness = objective(refined_h)
                if refined_fitness < best_c5:
                    best_c5 = refined_fitness
                    best_h = refined_h.copy()
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
