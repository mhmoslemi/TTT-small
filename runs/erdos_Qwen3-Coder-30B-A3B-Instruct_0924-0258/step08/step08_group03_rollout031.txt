import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search bounds
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    def check_feasible(c5_target):
        # Create variables
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Sample a subset of shifts to add constraints (100 randomly selected)
        shift_indices = np.random.choice(n_points, size=min(100, n_points), replace=False)
        
        # For each shift, add constraint
        for k in shift_indices:
            # Compute overlap for shift k
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h_var[i] * (1 - h_var[j])
            constraints.append(overlap_expr <= c5_target * dx)
        
        # Problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            return prob.status == 'optimal'
        except:
            return False
    
    # Binary search
    while time() - start_time < budget_s * 0.8 and high - low > tolerance:
        mid = (low + high) / 2
        if check_feasible(mid):
            high = mid
            # Update best solution
            # Re-solve for this C5 to get a new solution
            h_var = cp.Variable(n_points)
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points / 2
            ]
            # Add sample constraints
            shift_indices = np.random.choice(n_points, size=min(100, n_points), replace=False)
            for k in shift_indices:
                overlap_expr = 0
                for i in range(n_points):
                    j = (i + k) % n_points
                    overlap_expr += h_var[i] * (1 - h_var[j])
                constraints.append(overlap_expr <= mid * dx)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.CVXOPT, verbose=False)
                if prob.status == 'optimal':
                    new_h = h_var.value
                    if new_h is not None:
                        new_h = np.clip(new_h, 0, 1)
                        new_h = new_h * (n_points / 2) / np.sum(new_h) # Renormalize
                        new_c5 = evaluate_erdos_solution(new_h, dx)
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = new_h.copy()
            except:
                pass
        else:
            low = mid
    
    # Local refinement around best solution
    # Using gradient-based method on best found solution
    def obj(h_flat):
        h = h_flat.reshape(-1)
        return evaluate_erdos_solution(h, dx)
    
    def grad(h_flat):
        h = h_flat.reshape(-1)
        # Approximate gradient with finite differences
        eps = 1e-4
        grad_vec = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad_vec[i] = (obj(h_plus) - obj(h_minus)) / (2 * eps)
        return grad_vec
    
    # Refine best solution
    final_h = best_h.copy()
    for _ in range(500):
        if time() - start_time > budget_s * 0.95:
            break
        g = grad(final_h)
        final_h = final_h - 0.1 * g
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    best_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), best_c5, n_points)
