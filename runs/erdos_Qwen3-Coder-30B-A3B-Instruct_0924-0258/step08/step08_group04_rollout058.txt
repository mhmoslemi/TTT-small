import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    h_proj = np.clip(h, 0, 1)
    sum_h = np.sum(h_proj)
    if abs(sum_h - n_points/2) < 1e-10:
        return h_proj
    # Scale to meet constraint
    if sum_h > 0:
        h_proj = h_proj * (n_points/2) / sum_h
    else:
        h_proj = np.ones(n_points) * (n_points/2) / n_points
    return h_proj

def compute_gradient(h, max_lags, n_points):
    grad = np.zeros(n_points)
    for k in max_lags:
        if k >= 0:
            for i in range(n_points - k):
                grad[i] += 1 - h[i + k]
                if i + k < n_points:
                    grad[i + k] -= h[i]
        else:
            for i in range(n_points + k):
                grad[i] += 1 - h[i - k]
                if i - k >= 0:
                    grad[i - k] -= h[i]
    return grad

def line_search(h, grad, dx, n_points, max_iter=20):
    alpha = 1.0
    step_size = 0.1
    c1 = 1e-4
    c2 = 0.9
    h_current = h.copy()
    c5_current = evaluate_erdos_solution(h_current, dx)
    
    for i in range(max_iter):
        h_new = h_current - alpha * grad
        h_new = project_to_feasible(h_new, n_points, dx)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < c5_current:
            return alpha, h_new
        else:
            alpha *= step_size
    
    return alpha, h_new
