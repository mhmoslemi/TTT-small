import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    # Initialize best solution
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Trust region parameters
    trust_radius = 0.2
    max_trust_radius = 0.5
    min_trust_radius = 1e-6
    
    # Iteration parameters
    max_iter = 5000
    patience = 100
    no_improvement_count = 0
    
    h_current = h_start.copy()
    
    # Prepare for cutting plane constraints
    overlap_constraints = []
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find current maximum overlap and lag
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        overlap_values = corr * dx
        max_idx = np.argmax(overlap_values)
        max_overlap = overlap_values[max_idx]
        
        # Check if this is better than best so far
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h_current.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > patience:
            break
            
        # Build constraint matrix for trust region
        num_constraints = len(overlap_constraints)
        
        # Define optimization problem
        h_vars = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = []
        
        # Sum constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Bound constraints
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        
        # Trust region constraint
        trust_violation = cp.sum_squares(h_vars - h_current)
        constraints.append(trust_violation <= trust_radius**2)
        
        # Overlap constraints
        for k in range(len(overlap_constraints)):
            lag, overlap_val = overlap_constraints[k]
            # Add constraint for this lag
            if lag >= 0:
                expr = cp.sum(cp.multiply(h_vars[:n_points-lag], 1-h_vars[lag:]))
            else:
                expr = cp.sum(cp.multiply(h_vars[-lag:], 1-h_vars[:n_points+lag]))
            constraints.append(expr <= C)
            
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                new_c5 = prob.value
                # Update trust region
                if new_c5 < best_c5:
                    trust_radius = min(max_trust_radius, trust_radius * 1.2)
                else:
                    trust_radius = max(min_trust_radius, trust_radius * 0.8)
                h_current = np.clip(h_new, 0, 1)
            else:
                # Fall back to gradient descent with projection
                # Compute gradient of overlap with respect to h and update
                gradient = np.zeros_like(h_current)
                # Here we take a simpler approach and try gradient descent
                current_overlap_lag = np.argmax(overlap_values)
                step_size = 0.01
                h_new = h_current - step_size * gradient
                
                # Re-normalize and bound
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                h_current = h_new
                
        except Exception as e:
            # Fallback to simple gradient descent
            step_size = 0.01
            
            # Approximate gradient direction
            gradient = np.zeros_like(h_current)
            current_max_lag = np.argmax(overlap_values) - (len(h_current) - 1)
            # Simple idea: look at effect from one representative lag
            # Just project on sum constraint for now
            h_new = h_current.copy()
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Compute overlap for new solution
            new_overlap = evaluate_erdos_solution(h_new, dx)
            if new_overlap < best_c5:
                h_current = h_new.copy()
                best_c5 = new_overlap
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            if no_improvement_count > patience:
                break
    
    # Final SLSQP refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
