import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def solve_convex_relaxation(h_init):
        h = cp.Variable(n_points)
        constraints = [cp.sum(h) == n_points/2, h >= 0, h <= 1]
        
        # Approximate the objective using a quadratic relaxation
        # For all lags k, we want to minimize sum(h_i * (1 - h_{i+k}))
        objectives = []
        for k in range(n_points):
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h[i] * (1 - h[j])
            objectives.append(expr)
        
        # Minimize maximum of all such expressions
        max_expr = cp.Variable()
        for expr in objectives:
            constraints.append(max_expr >= expr)
        
        prob = cp.Problem(cp.Minimize(max_expr), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                return np.array(h.value).reshape(-1)
        except:
            pass
        return h_init
    
    def coordinate_descent_step(h, fixed_indices, optimize_indices, n_iter=50):
        h = h.copy()
        for _ in range(n_iter):
            for i in optimize_indices:
                # Compute gradient and update
                grad = 0
                for k in range(n_points):
                    j = (i + k) % n_points
                    grad += (1 - h[j] - h[i]) * dx
                h[i] -= 0.1 * grad
            h = project_h(h)
        return h
    
    # Main optimization loop
    for iteration in range(2000):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Convex relaxation step
        h_relaxed = solve_convex_relaxation(best_h)
        if h_relaxed is not None:
            best_h = project_h(h_relaxed)
        
        # Coordinate descent step
        fixed_indices = np.random.choice(n_points, size=int(0.8*n_points), replace=False)
        optimize_indices = np.setdiff1d(np.arange(n_points), fixed_indices)
        
        h_new = coordinate_descent_step(best_h, fixed_indices, optimize_indices, 50)
        h_new = project_h(h_new)
        
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
        
        # Adaptive stopping
        if iteration > 0 and iteration % 100 == 0:
            if iteration > 1000 and best_c5 > 0.38085:
                break
            elif iteration > 500 and best_c5 > 0.381:
                break
    
    # Final refinement with L-BFGS-B
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda h: objective(h),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success and res.fun < best_c5:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            best_c5 = evaluate_erdos_solution(refined_h, dx)
            best_h = refined_h
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
