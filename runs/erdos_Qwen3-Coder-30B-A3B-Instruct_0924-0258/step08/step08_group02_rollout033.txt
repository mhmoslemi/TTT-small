import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with adaptive cooling
    T = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter = 100000
    epsilon = 0.1
    iter_count = 0
    
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    while T > T_min and iter_count < max_iter and time() - start_time < budget_s * 0.9:
        # Perturb two random indices
        i, j = random.randint(0, n_points-1), random.randint(0, n_points-1)
        delta = random.uniform(-epsilon, epsilon)
        
        # Adjust values while maintaining sum
        if i != j:
            h_new = h_current.copy()
            diff = delta * (1.0 - np.sum(h_new) / (n_points/2))
            
            h_new[i] += delta
            h_new[j] -= diff
            
            # Ensure bounds
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            delta_c5 = c5_new - best_c5
            if delta_c5 < 0 or random.random() < np.exp(-delta_c5 / T):
                h_current = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_current.copy()
        
        # Cool temperature
        T *= alpha
        iter_count += 1
    
    # Local refinement with L-BFGS-B
    if time() - start_time < budget_s * 0.95:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        try:
            res = minimize(
                objective,
                best_h.copy(),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 50, 'ftol': 1e-9}
            )
            
            if res.success:
                refined_h = res.x
                refined_h = refined_h * (n_points/2) / np.sum(refined_h)
                refined_c5 = evaluate_erdos_solution(refined_h, dx)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
