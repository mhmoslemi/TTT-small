import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    generations = 500
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 2
    stagnation_limit = 50
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        pop_size -= 1
    
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    # Helper functions
    def feasible_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        alpha = np.random.rand()
        child1 = alpha * parent1 + (1 - alpha) * parent2
        child2 = (1 - alpha) * parent1 + alpha * parent2
        return feasible_h(child1), feasible_h(child2)
    
    def mutate(h, generation):
        mutated = h.copy()
        mutation_step = 0.05 * (0.95 ** generation)
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                mutated[i] += np.random.normal(0, mutation_step)
        return feasible_h(mutated)
    
    def tournament_selection(pop, fitnesses, k=3):
        selected_idx = np.random.choice(len(pop), size=k)
        selected_fitness = [fitnesses[i] for i in selected_idx]
        winner = selected_idx[np.argmin(selected_fitness)]
        return pop[winner]
    
    def evaluate_population(population):
        fitnesses = []
        for h in population:
            fitness = evaluate_erdos_solution(h, dx)
            fitnesses.append(fitness)
        return fitnesses
    
    # Main GA loop
    no_improvement = 0
    for generation in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitnesses = evaluate_population(population)
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1

            break

        sorted_indices = np.argsort(fitnesses)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]

        new_population = sorted_pop[:elite_size]
        
        # Selection and reproduction

            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, generation)
            child2 = mutate(child2, generation)

        population = new_population[:pop_size]

        if generation % 50 == 0 and time() - start_time < budget_s * 0.95:

                return evaluate_erdos_solution(h, dx)

                return np.sum(h) - n_points / 2.0

                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    # Final SLSQP refinement on best solution
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
