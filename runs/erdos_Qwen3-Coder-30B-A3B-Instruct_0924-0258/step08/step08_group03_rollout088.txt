import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_coarse = 15
    n_fine = 35
    coarse_iter = 500
    fine_iter = 2000
    mutation_rate = 0.1
    
    # Initialize coarse grid
    h_coarse = np.ones(n_coarse) * 0.5
    
    # Initialize fine grid
    if 'initial_h_values' in globals():
        h_fine = np.array(initial_h_values)
    else:
        h_fine = np.ones(n_fine) * 0.5
    
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # Best solutions
    best_coarse = h_coarse.copy()
    best_fine = h_fine.copy()
    best_c5_coarse = evaluate_erdos_solution(h_coarse, dx_coarse)
    best_c5_fine = evaluate_erdos_solution(h_fine, dx_fine)
    
    # Coarse optimization
    for i in range(coarse_iter):
        if time() - start_time > budget_s * 0.7:
            break
            
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_coarse - 1)
        
        grad = np.zeros(n_coarse)
        for j in range(n_coarse):
            k = j + max_lag
            if 0 <= k < n_coarse:
                grad[j] = 1 - 2 * h_coarse[j] - h_coarse[k]
            else:
                grad[j] = 1 - 2 * h_coarse[j]
        
        h_coarse_new = h_coarse - 0.01 * grad
        h_coarse_new = np.clip(h_coarse_new, 0, 1)
        h_coarse_new = h_coarse_new * (n_coarse / 2) / np.sum(h_coarse_new)
        
        c5_new = evaluate_erdos_solution(h_coarse_new, dx_coarse)
        if c5_new < best_c5_coarse:
            best_c5_coarse = c5_new
            best_coarse = h_coarse_new.copy()
            
        h_coarse = h_coarse_new.copy()
    
    # Upsample coarse to fine grid
    f = interp1d(np.linspace(0, 2, n_coarse), best_coarse, kind='linear', fill_value='extrapolate')
    h_fine = f(np.linspace(0, 2, n_fine))
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)
    
    # Fine grid optimization
    for i in range(fine_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        corr = np.correlate(h_fine, 1 - h_fine, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_fine - 1)
        
        grad = np.zeros(n_fine)
        for j in range(n_fine):
            k = j + max_lag
            if 0 <= k < n_fine:
                grad[j] = 1 - 2 * h_fine[j] - h_fine[k]
            else:
                grad[j] = 1 - 2 * h_fine[j]
        
        # Gradient descent step
        h_fine_new = h_fine - 0.01 * grad
        h_fine_new = np.clip(h_fine_new, 0, 1)
        h_fine_new = h_fine_new * (n_fine / 2) / np.sum(h_fine_new)
        
        c5_new = evaluate_erdos_solution(h_fine_new, dx_fine)
        if c5_new < best_c5_fine:
            best_c5_fine = c5_new
            best_fine = h_fine_new.copy()
            
        # Occasionally perturb
        if i % 100 == 0:
            perturb_idx = np.random.choice(n_fine, size=int(n_fine * mutation_rate), replace=False)
            h_fine_new[perturb_idx] = np.random.uniform(0, 1, len(perturb_idx))
            h_fine_new = h_fine_new * (n_fine / 2) / np.sum(h_fine_new)
            c5_new = evaluate_erdos_solution(h_fine_new, dx_fine)
            if c5_new < best_c5_fine:
                best_c5_fine = c5_new
                best_fine = h_fine_new.copy()
                
        h_fine = h_fine_new.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_fine, dx_fine)
    
    return (list(best_fine), final_c5, n_fine)
