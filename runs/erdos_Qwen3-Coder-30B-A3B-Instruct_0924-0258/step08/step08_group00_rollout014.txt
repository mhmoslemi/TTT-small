import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # More sophisticated optimization approach: hybrid with restarts
    max_iter = 5000
    patience = 100
    no_improvement = 0
    step_size = 0.01
    
    # Try multiple restarts with different strategies
    for restart in range(5):
        if time() - start_time > budget_s * 0.90:
            break
            
        h = h_start.copy()
        h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Evaluate overlap for all lags
            corr = np.correlate(h, 1 - h, mode='full')
            max_lag_idx = np.argmax(corr)
            max_overlap = corr[max_lag_idx] * dx
            
            if max_overlap < best_c5:
                best_c5 = max_overlap
                best_h = h.copy()
                no_improvement = 0
            else:
                no_improvement += 1
                
            if no_improvement >= patience:
                break
                
            # Identify lag with maximum overlap
            max_lag = max_lag_idx - (n_points - 1)
            
            # Compute gradient
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + max_lag
                if 0 <= j < n_points:
                    grad[i] = 2 * (1 - h[j] - h[i]) / n_points
            
            # Update with gradient descent (adaptive learning rate)
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Accept improvement or reduce step size
            new_c5 = evaluate_erdos_solution(h_new, dx)
            if new_c5 < max_overlap:
                h = h_new
            else:
                step_size *= 0.9
                if step_size < 1e-8:
                    break
                    
        # Try random perturbations
        h_perturbed = h + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
    
    # Final global refinement using scipy
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
