import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available (n=35)
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Multiresolution strategy: coarse grid to explore, fine grid to refine
    coarse_points = 15
    coarse_dx = 2.0 / coarse_points
    
    # Generate population of coarse solutions
    def generate_population(size, n_coarse):
        population = []
        for _ in range(size):
            h = np.random.rand(n_coarse)
            h = h * (coarse_points/2) / np.sum(h)
            population.append(h)
        return population
    
    # Evaluate and rank population
    def evaluate_population(population, dx_coarse):
        scores = [evaluate_erdos_solution(h, dx_coarse) for h in population]
        return scores
    
    # Select top candidates
    def selection(population, scores, num_select):
        idx = np.argsort(scores)[:num_select]
        return [population[i] for i in idx]
    
    # Crossover two parents
    def crossover(parent1, parent2, rate=0.5):
        if np.random.rand() > rate:
            return parent1.copy(), parent2.copy()
        child1, child2 = parent1.copy(), parent2.copy()
        crossover_point = np.random.randint(1, len(parent1))
        child1[crossover_point:] = parent2[crossover_point:]
        child2[crossover_point:] = parent1[crossover_point:]
        return child1, child2
    
    # Mutate a solution
    def mutate(h, mutation_rate=0.2):
        h_mut = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h_mut[i] += np.random.normal(0, 0.05)
        # Clamp values
        h_mut = np.clip(h_mut, 0, 1)
        # Renormalize to correct sum
        h_mut = h_mut * (coarse_points/2) / np.sum(h_mut)
        return h_mut
    
    # Upscale coarse solution to fine grid
    def upscale_to_fine(coarse_h):
        coarse_x = np.linspace(0, 2, len(coarse_h))
        fine_x = np.linspace(0, 2, n_points)
        fine_h = np.interp(fine_x, coarse_x, coarse_h)
        # Adjust to sum constraint
        fine_h = fine_h * (n_points/2) / np.sum(fine_h)
        return fine_h
    
    # Coarse-grid optimization
    pop_size = 50
    num_select = 20
    max_coarse_iter = 100
    
    coarse_pop = generate_population(pop_size, coarse_points)
    coarse_best_c5 = float('inf')
    coarse_best_h = None
    
    for i in range(max_coarse_iter):
        if time() - start_time > budget_s * 0.8:
            break
        scores = evaluate_population(coarse_pop, coarse_dx)
        
        # Track best
        min_score_idx = np.argmin(scores)
        if scores[min_score_idx] < coarse_best_c5:
            coarse_best_c5 = scores[min_score_idx]
            coarse_best_h = coarse_pop[min_score_idx].copy()
        
        # Selection
        selected = selection(coarse_pop, scores, num_select)
        
        # Generate new population
        new_pop = selected.copy()
        while len(new_pop) < pop_size:
            # Crossover
            p1, p2 = np.random.choice(selected, 2, replace=False)
            c1, c2 = crossover(p1, p2)
            
            # Mutate
            c1 = mutate(c1)
            c2 = mutate(c2)
            
            new_pop.extend([c1, c2])
        
        coarse_pop = new_pop[:pop_size]
    
    # Upscale best coarse solution to full resolution
    if coarse_best_h is not None:
        h_current = upscale_to_fine(coarse_best_h)
    
    # Refine with gradient descent
    step_size = 0.01
    max_refine_iter = 10000
    
    for i in range(max_refine_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all correlations for all possible lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Get the maximum overlap lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Compute gradient for the max overlap
        gradient = np.zeros(n_points)
        
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_current[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= h_current[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_current[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_current[i]
        
        # Gradient descent step
        h_new = h_current - step_size * gradient
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # If not improving, reduce step size
        if c5_new >= evaluate_erdos_solution(h_current, dx):
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            h_current = h_new
            step_size *= 1.05  # Increase step size slightly
    
    return (list(best_h), best_c5, n_points)
