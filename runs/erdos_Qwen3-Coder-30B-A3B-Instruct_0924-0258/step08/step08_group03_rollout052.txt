import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex optimization + filtering approach
    max_iterations = 2000
    K = 10  # Number of top overlaps to consider
    filter_bandwidth = 5
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        top_k_indices = np.argsort(corr)[-K:][::-1]
        
        # Construct linear approximation of the maximum overlap
        # Use a convex optimization to minimize these overlaps
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Approximate maximum with sum of top overlaps
        objective_terms = []
        for idx in top_k_indices:
            # Create the shifted version of h_var for this lag
            lag = idx - (n_points - 1)
            if lag >= 0:
                shifted = h_var[lag:]
                original = h_var[:len(shifted)]
            else:
                shifted = h_var[:lag]
                original = h_var[-lag:]
            
            # Contribution of this overlap
            term = cp.sum(cp.multiply(original, 1 - shifted))
            objective_terms.append(term)
        
        # Minimize the sum of top overlaps (possibly weighted)
        objective = cp.Minimize(cp.sum(objective_terms) / len(objective_terms))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            # Fallback to gradient descent if optimization fails
            h_new = h.copy()
            step_size = 0.01
            
            # Compute gradient for max overlap
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + max_lag
                if 0 <= j < n_points:
                    grad[i] = 1 - 2 * h[i] - h[j]
                else:
                    grad[i] = 1 - 2 * h[i]
            
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            c5_new = evaluate_erdos_solution(h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            continue
        
        # Extract the solution
        h_new = h_var.value
        if h_new is None:
            # Fallback to gradient descent
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + max_lag
                if 0 <= j < n_points:
                    grad[i] = 1 - 2 * h[i] - h[j]
                else:
                    grad[i] = 1 - 2 * h[i]
            
            h_new = h - 0.01 * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Apply low-pass filtering
        if filter_bandwidth > 0:
            h_filtered = np.zeros(n_points)
            for i in range(n_points):
                total = 0
                count = 0
                for j in range(max(0, i - filter_bandwidth), min(n_points, i + filter_bandwidth + 1)):
                    weight = 1.0 / (abs(i - j) + 1)
                    total += h_new[j] * weight
                    count += weight
                h_filtered[i] = total / count
            h_new = h_filtered
        
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            
            # Occasionally increase K to focus on more overlaps
            if iteration % 50 == 0 and K < 20:
                K = min(K + 1, 20)
                
            # Occasionally adjust filter bandwidth (softer filter)
            if iteration % 100 == 0 and filter_bandwidth > 1:
                filter_bandwidth = max(1, filter_bandwidth - 1)
        else:
            # Reduce K or escalate search
            if K > 5 and iteration % 100 == 0:
                K = max(5, K - 1)
                
    return (list(best_h), best_c5, n_points)
