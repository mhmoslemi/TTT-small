import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convex relaxation approach
    max_iter = 100
    no_improve_iter = 0
    max_no_improve = 50
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Relaxation parameters
    lambda_reg = 0.1
    alpha_reg = 0.01
    
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convex relaxation step
        try:
            # Create variables for convex optimization
            h_var = cp.Variable(n_points)
            
            # Objective: minimize sum of squared overlaps + regularization
            overlaps = []
            objective = 0
            
            # Compute all overlaps for current h
            corr = np.correlate(h, 1 - h, mode='full')
            max_overlap = np.max(corr)
            
            # For each lag k, compute overlap
            for k in range(2 * n_points - 1):
                overlap = 0
                for i in range(n_points):
                    j = (i + k) % n_points
                    overlap += h_var[i] * (1 - h_var[j])
                overlap *= dx
                overlaps.append(overlap)
            
            # Objective function - sum of squared overlaps + regularization
            objective = cp.sum_squares(overlaps) + alpha_reg * cp.sum_squares(h_var)
            
            # Constraints
            constraints = [cp.sum(h_var) == n_points/2,
                           h_var >= 0,
                           h_var <= 1]
            
            # Formulate and solve problem
            prob = cp.Problem(cp.Minimize(objective), constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_relaxed = h_var.value
                h_relaxed = np.clip(h_relaxed, 0, 1)
                h_relaxed = h_relaxed * (n_points/2) / np.sum(h_relaxed)
                h = h_relaxed
        except:
            pass
        
        # Coordinate descent refinement
        indices = np.arange(n_points)
        np.random.shuffle(indices)
        
        # Select 20% to optimize
        selected = indices[:int(0.2 * n_points)]
        
        # Gradient descent on selected coordinates
        for _ in range(50):
            if time() - start_time > budget_s * 0.95:
                break
                
            grad = np.zeros(n_points)
            for i in range(n_points):
                for k in range(n_points):
                    j = (i + k) % n_points
                    grad[i] += -2 * dx * (1 - h[j] - h[i])
                    
            h_new = h - 0.001 * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            h = h_new
            
            # Early stopping if time is almost up
            if time() - start_time > budget_s * 0.95:
                break
                
        # Evaluate and track
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_iter = 0
        else:
            no_improve_iter += 1
            
        if no_improve_iter >= max_no_improve:
            # Random perturbation if no improvement for a long time
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points/2) / np.sum(h)
            no_improve_iter = 0
            
        if time() - start_time > budget_s * 0.95:
            break
    
    # Final refinement with scipy if needed
    try:
        from scipy.optimize import minimize
        
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)
            
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
