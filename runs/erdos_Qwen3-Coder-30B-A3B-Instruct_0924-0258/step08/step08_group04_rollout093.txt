import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # SDP approach
    try:
        # We want to minimize C5 subject to:
        # C5 >= sum_i h[i] * (1 - h[i+k]) * dx for all k
        # This corresponds to:
        # sum_i h[i] - sum_i h[i] * h[i+k] <= C5 * dx for all k
        
        # Let's try to set up convex optimization
        h_var = cp.Variable(n_points, name='h')
        
        # Constraint: sum(h) = n_points / 2
        sum_constraint = cp.Equality(cp.sum(h_var), n_points / 2)
        
        # Constraint: 0 <= h <= 1
        bound_constraints = [h_var >= 0, h_var <= 1]
        
        # We want to minimize C5 such that:
        # sum_i h[i] * (1 - h[i + k]) <= C5 * dx for all valid k
        # which means: sum_i h[i] - sum_i h[i] * h[i + k] <= C5 * dx
        # Rearranging: sum_i h[i] * h[i + k] >= sum_i h[i] - C5 * dx
        
        # For efficiency, rather than adding all constraints, we'll iterate
        # and add constraint for the worst case
        
        # Use a candidate best solution to start
        h_sdp = h_current.copy()
        
        # Try solving SDP approach with most recent good solution as initialization
        # We try a narrower search with fewer iterations but MP-based approach
        problem = cp.Problem(
            cp.Minimize(1.0),
            [sum_constraint] + bound_constraints
        )
        
        # Use just a reasonable number of iterations since we have a good starting point.
        # Allow MOSEK solver with CVXPY to be efficient:
        # We'll solve to get a good initial value, then move to local optimization if needed.
        try:
            problem.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s*0.9)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # The solver found a solution, but since we're minimizing 1.0 we should ignore the objective.
                # Extract the solution
                h_sdp = h_var.value
                if h_sdp is not None:
                    # Ensure values are within bounds and sum is correct
                    h_sdp = np.clip(h_sdp, 0, 1)
                    h_sdp = h_sdp * (n_points/2) / np.sum(h_sdp)
                    
                    # Check if better
                    c5_sdp = evaluate_erdos_solution(h_sdp, dx)
                    if c5_sdp < best_c5:
                        best_c5 = c5_sdp
                        best_h = h_sdp.copy()
        except:
            # Fall back to original optimization method if SDP fails
            pass
            
    except Exception as e:
        # Best to fall back to gradient-based method on error
        pass
    
    # Perform some gradient-based polishing with reduced epsilon to explore finer space
    if time() - start_time < budget_s * 0.85:
        h_current = best_h.copy()
        step_size = 0.005
        max_iter = 2000
        iter_count = 0
        
        while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
            # Compute all correlations for all possible lags
            overlap = np.correlate(h_current, 1 - h_current, mode='full')
            # Get the maximum correlation index (i.e. lag that gives max overlap)
            max_lag_idx = np.argmax(overlap)
            max_lag = max_lag_idx - (len(h_current) - 1)
            
            # Compute gradient of max overlap lag  
            gradient = np.zeros(n_points)
            if max_lag >= 0:
                for i in range(n_points - max_lag):
                    gradient[i] += 1 - h_current[i + max_lag]
                    if i + max_lag < n_points:
                        gradient[i + max_lag] -= h_current[i]
            else:
                for i in range(n_points + max_lag):
                    gradient[i] += 1 - h_current[i - max_lag]
                    if i - max_lag >= 0:
                        gradient[i - max_lag] -= h_current[i]
            
            # Gradient descent step
            h_new = h_current - step_size * gradient
            
            # Ensure feasibility
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate new C5 value
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept step if an improvement
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            # Adaptive step size adjustments
            if c5_new >= evaluate_erdos_solution(h_current, dx):
                step_size *= 0.8
                if step_size < 1e-6:
                    break
            else:
                h_current = h_new.copy()
                step_size = min(0.01, step_size * 1.05)
            
            iter_count += 1
    
    return (list(best_h), best_c5, n_points)
