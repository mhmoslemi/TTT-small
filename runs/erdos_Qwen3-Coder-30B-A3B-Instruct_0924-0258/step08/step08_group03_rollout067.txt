import numpy as np
from time import time
import cvxpy as cp
from scipy.linalg import svd

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # SDP-based optimization
    n_iter = 0
    low = 0.38
    high = 0.38085
    tol = 1e-5
    
    # Precompute all shifts we'll use
    shifts = np.arange(-n_points + 1, n_points)
    
    while time() - start_time < budget_s * 0.95 and (high - low) > tol:
        mid = (low + high) / 2
        
        # SDP relaxation approach to check if C5 <= mid is possible
        Y = cp.Variable((n_points, n_points), symmetric=True)
        h_var = cp.Variable(n_points)
        
        constraints = [
            Y >> 0,  # Y is positive semidefinite
            cp.diag(Y) == h_var,  # Diagonal constraint
            cp.sum(h_var) == n_points / 2,  # Normalization
        ]
        
        # Add overlap constraints for all shifts
        for k in shifts:
            if k >= 0:
                # For k >= 0: h_i * (1 - h_{i+k})
                sum_term = 0
                for i in range(n_points - k):
                    sum_term += h_var[i] - Y[i, i+k]
                constraints.append(sum_term <= mid * dx)
            else:
                # For k < 0: h_{i+k} * (1 - h_i) 
                sum_term = 0
                for i in range(-k, n_points):
                    sum_term += h_var[i] - Y[i+k, i]
                constraints.append(sum_term <= mid * dx)
        
        # Minimize slack (dummy objective)
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                # Extract solution
                h_sol = h_var.value
                Y_sol = Y.value
                
                # Project Y to nearest feasible h
                _, _, V = svd(Y_sol)
                h_proj = np.maximum(0, np.minimum(1, V[:, 0]**2))
                
                # Normalize
                h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
                
                # Evaluate
                c5_proj = evaluate_erdos_solution(h_proj, dx)
                if c5_proj < best_c5:
                    best_c5 = c5_proj
                    best_h = h_proj.copy()
                
                # Refine locally
                for _ in range(500):
                    corr = np.correlate(h_proj, 1 - h_proj, mode='full')
                    max_lag_idx = np.argmax(corr)
                    max_lag = max_lag_idx - (n_points - 1)
                    
                    grad = np.zeros(n_points)
                    for i in range(n_points):
                        j = i + max_lag
                        if 0 <= j < n_points:
                            grad[i] = 1 - 2 * h_proj[i] - h_proj[j]
                        else:
                            grad[i] = 1 - 2 * h_proj[i]
                    
                    h_new = h_proj - 0.005 * grad
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    if c5_new < c5_proj:
                        h_proj = h_new.copy()
                        c5_proj = c5_new
                    else:
                        break
                
                if c5_proj < best_c5:
                    best_c5 = c5_proj
                    best_h = h_proj.copy()
                
                high = mid
            else:
                low = mid
        except:
            low = mid
            
        n_iter += 1
    
    # Local refinement
    h_refined = best_h.copy()
    for _ in range(2000):
        if time() - start_time > budget_s * 0.95:
            break
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h_refined[i] - h_refined[j]
            else:
                grad[i] = 1 - 2 * h_refined[i]
        
        h_new = h_refined - 0.002 * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_refined = h_new.copy()
        else:
            break
    
    return (list(best_h), best_c5, n_points)
