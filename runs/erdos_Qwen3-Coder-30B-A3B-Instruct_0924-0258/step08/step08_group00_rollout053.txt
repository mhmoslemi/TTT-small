import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution SA with refinement
    resolutions = [35, 70, 140, 280]
    current_h = h_start.copy()
    current_n = n_points
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > current_n:
            # Interpolate to higher resolution
            old_x = np.linspace(0, 2, current_n)
            new_x = np.linspace(0, 2, res)
            current_h = np.interp(new_x, old_x, current_h)
            current_n = res
            dx = 2.0 / current_n
            
        # Simple simulated annealing at this resolution
        temp = best_c5 * 0.1  # Initial temperature
        cooling_rate = 0.999
        max_iter = 5000 if res == 35 else 2000
        patience = 200
        no_improvement = 0
        
        h = current_h.copy()
        h = h * (current_n/2) / np.sum(h)  # Ensure sum constraint
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb a random point
            idx = random.randint(0, current_n - 1)
            delta = random.gauss(0, temp * 0.1)
            h_new = h.copy()
            h_new[idx] += delta
            h_new[idx] = np.clip(h_new[idx], 0, 1)
            h_new = h_new * (current_n/2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            if c5_new < best_c5:
                h = h_new
                current_h = h_new
                best_c5 = c5_new
                best_h = h_new.copy()
                no_improvement = 0
            else:
                # Accept with probability
                if random.random() < np.exp(-(c5_new - best_c5) / temp):
                    h = h_new
                    current_h = h_new
                no_improvement += 1
                
            if no_improvement >= patience:
                break
                
            temp *= cooling_rate
            
        # Local refinement with SLSQP
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - current_n / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(current_n)]
            
            res = minimize(objective, h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, current_n)
