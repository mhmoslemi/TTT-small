import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    # Ensure sum(h_start) = n_points / 2
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # PSO parameters
    num_particles = 75
    inertia_start = 0.8
    inertia_end = 0.4
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    perturbation_rate = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    
    for _ in range(num_particles):
        h = h_start.copy()
        # Add small random perturbation
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        # Renormalize
        h = h * (n_points/2) / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
    
    # Track best solutions
    particle_best = [p.copy() for p in particles]
    particle_best_fitness = [evaluate_erdos_solution(p, dx) for p in particles]
    global_best_idx = np.argmin(particle_best_fitness)
    global_best = particle_best[global_best_idx].copy()
    global_best_fitness = particle_best_fitness[global_best_idx]
    
    max_iter = int(1e6)
    iteration = 0
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iter:
        # Update inertia linearly
        inertia = inertia_start - (inertia_start - inertia_end) * (iteration / max_iter)
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_weight * r1 * (particle_best[i] - particles[i])
            social = social_weight * r2 * (global_best - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Renormalize
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate fitness
            fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update particle best
            if fitness < particle_best_fitness[i]:
                particle_best[i] = particles[i].copy()
                particle_best_fitness[i] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = fitness
        
        # Apply perturbation occasionally
        if iteration % 1000 == 0 and iteration > 0:
            for i in range(num_particles):
                if np.random.rand() < perturbation_rate:
                    particles[i] += np.random.normal(0, 0.05, n_points)
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
        
        iteration += 1
    
    # Final refinement with local search
    best_h = global_best.copy()
    best_c5 = global_best_fitness
    dx = 2.0 / n_points
    
    # Try some local gradient steps
    step_size = 0.001
    for _ in range(500):
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(best_h) - 1)
        
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - best_h[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= best_h[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - best_h[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= best_h[i]
        
        h_new = best_h - step_size * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        else:
            step_size *= 0.98
            if step_size < 1e-6:
                break
    
    return (list(best_h), best_c5, n_points)
