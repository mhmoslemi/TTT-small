import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    
    # Normalize to satisfy constraint
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Use convex relaxation approach with McCormick envelopes
    # Variables for h and z = h * h
    h_vars = cp.Variable(n_points)
    z_vars = cp.Variable((n_points, n_points))
    C = cp.Variable()
    
    # McCormick envelopes for z[i,j] = h[i] * h[j]
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h_vars) == n_points / 2)
    
    # McCormick constraints
    for i in range(n_points):
        for j in range(n_points):
            if i == j:
                constraints.append(z_vars[i, j] == h_vars[i])
            else:
                constraints.append(z_vars[i, j] >= h_vars[i] + h_vars[j] - 1)
                constraints.append(z_vars[i, j] <= h_vars[i])
                constraints.append(z_vars[i, j] <= h_vars[j])
                constraints.append(z_vars[i, j] >= 0)
    
    # Add overlap constraints for all lags
    max_lag = n_points - 1
    for lag in range(-max_lag, max_lag + 1):
        if lag >= 0:
            # h[i] * (1 - h[i + lag]) = h[i] - z[i][i + lag]
            expr = 0
            for i in range(n_points - lag):
                expr += h_vars[i] - z_vars[i, i + lag]
        else:
            # h[i] * (1 - h[i - lag]) = h[i] - z[i][i - lag]
            expr = 0
            for i in range(-lag, n_points):
                expr += h_vars[i] - z_vars[i, i - lag]
        
        constraints.append(expr <= C)
    
    # Solve the convex relaxation
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_relaxed = h_vars.value
            h_refined = np.clip(h_relaxed, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
    except:
        pass
    
    # Local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        h_refined = h_current
        max_iter = 5000
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.90:
                break
            
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-12})
            if res.success:
                h_refined = res.x
                c5_new = objective(h_refined)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_refined.copy()
                    
                # Small random perturbation to escape local minima
                h_perturb = h_refined + np.random.normal(0, 0.001, n_points)
                h_perturb = np.clip(h_perturb, 0, 1)
                h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
                c5_perturb = objective(h_perturb)
                if c5_perturb < best_c5:
                    best_c5 = c5_perturb
                    best_h = h_perturb.copy()
                    
                # Keep current h_refined for next iteration
                h_refined = h_refined + 0.01 * (best_h - h_refined)
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            else:
                break
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
