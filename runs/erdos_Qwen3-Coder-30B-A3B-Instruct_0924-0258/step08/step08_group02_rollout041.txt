import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for convex relaxation
    lambda_reg = 0.1
    alpha_reg = 0.01
    coord_fraction = 0.2
    grad_steps = 50
    
    # Main iterative refinement loop
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time() - start_time < budget_s * 0.95:
        # Convex relaxation step
        h_relaxed = h_start.copy()
        try:
            # Formulate convex optimization problem using CVXPY
            h_var = cp.Variable(n_points)
            h_constrained = h_var * (n_points/2) / cp.sum(h_var)
            
            # Objective: minimize a relaxed version of the problem
            # We use a quadratic penalty to ensure balance among overlaps
            overlaps_sum = 0
            for k in range(n_points):
                overlap = cp.sum(cp.multiply(h_constrained, 
                                           cp.roll(1 - h_constrained, k)))
                overlaps_sum += overlap**2
            
            # Regularization terms
            regularization = alpha_reg * cp.sum_squares(h_var) + \
                            lambda_reg * overlaps_sum
            
            # Constraints
            constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points/2]
            
            # Solve
            prob = cp.Problem(cp.Minimize(regularization), constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_relaxed = h_var.value
                h_relaxed = np.clip(h_relaxed, 0, 1)
                h_relaxed = h_relaxed * (n_points/2) / np.sum(h_relaxed)
        except:
            pass
        
        # Coordinate descent step
        h_updated = h_relaxed.copy()
        coord_indices = np.random.permutation(n_points)[:int(n_points * coord_fraction)]
        
        for _ in range(grad_steps):
            if time() - start_time > budget_s * 0.95:
                break
            for idx in coord_indices:
                # Compute current overlap impact of changing this point
                old_val = h_updated[idx]
                old_overlap = 0
                for k in range(n_points):
                    j = (idx + k) % n_points
                    old_overlap += old_val * (1 - h_updated[j])
                
                # Compute gradient (approximate)
                grad = 0
                for k in range(n_points):
                    j = (idx + k) % n_points
                    grad += -2 * (1 - h_updated[j] - old_val)
                
                # Update
                new_val = old_val - 0.001 * grad
                h_updated[idx] = np.clip(new_val, 0, 1)
            
            # Enforce sum constraint
            h_updated = h_updated * (n_points/2) / np.sum(h_updated)
        
        # Evaluate and update best solution
        current_c5 = evaluate_erdos_solution(h_updated, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_updated.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early stopping
        if no_improvement_count >= max_no_improvement:
            break
            
        h_start = h_updated.copy()
    
    return (list(best_h), best_c5, n_points)
