import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution SA with refinement
    resolutions = [35, 70, 140, 280]
    current_h = h_start.copy()
    
    # Simulated Annealing parameters
    initial_temp = 0.1
    cooling_rate = 0.95
    min_temp = 1e-6
    max_iter_per_temp = 100
    patience = 50
    no_improvement = 0
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            current_h = np.interp(x_new, x_orig, current_h)
            n_points = res
            dx = 2.0 / n_points
        
        # Initialize temperature
        temp = initial_temp
        h_current = current_h.copy()
        
        # Ensure correct sum
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Store best in this resolution
        best_in_res = h_current.copy()
        best_c5_in_res = evaluate_erdos_solution(h_current, dx)
        
        iter_counter = 0
        while temp > min_temp and time() - start_time < budget_s * 0.95:
            for _ in range(max_iter_per_temp):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Propose new solution: random perturbation
                h_new = h_current.copy()
                idx = np.random.randint(0, n_points)
                delta = np.random.normal(0, temp * 0.1)
                h_new[idx] += delta
                h_new[idx] = np.clip(h_new[idx], 0, 1)
                
                # Renormalize
                h_new = h_new * (n_points/2) / np.sum(h_new)
                
                # Evaluate
                c5_new = evaluate_erdos_solution(h_new, dx)
                c5_current = evaluate_erdos_solution(h_current, dx)
                
                # Accept or reject
                if c5_new < c5_current:
                    h_current = h_new
                    if c5_new < best_c5_in_res:
                        best_c5_in_res = c5_new
                        best_in_res = h_new.copy()
                        no_improvement = 0
                    else:
                        no_improvement += 1
                else:
                    # Accept with probability
                    if np.random.rand() < np.exp(-(c5_new - c5_current) / temp):
                        h_current = h_new
                        no_improvement += 1
                    else:
                        no_improvement += 1
                
                iter_counter += 1
                if no_improvement >= patience:
                    break
                    
            if no_improvement >= patience:
                break
            temp *= cooling_rate
            
        current_h = best_in_res.copy()
        if best_c5_in_res < best_c5:
            best_c5 = best_c5_in_res
            best_h = current_h.copy()
    
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, current_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
