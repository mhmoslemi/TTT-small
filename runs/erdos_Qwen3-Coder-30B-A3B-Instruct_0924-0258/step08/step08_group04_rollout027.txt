import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Coarse grid optimization (n=15) for global exploration
    coarse_points = 15
    coarse_dx = 2.0 / coarse_points
    coarse_h = np.ones(coarse_points) * 0.5
    coarse_h = coarse_h * (coarse_points/2) / np.sum(coarse_h)
    
    # Genetic algorithm on coarse grid
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.5
    generations = 1000
    
    population = []
    for _ in range(pop_size):
        h = np.random.rand(coarse_points)
        h = h * (coarse_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, coarse_dx)
            fitness.append(-c5)  # Negative because we want min
        
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            idx1, idx2 = random.sample(range(pop_size), 2)
            if fitness[idx1] > fitness[idx2]:
                selected.append(population[idx1].copy())
            else:
                selected.append(population[idx2].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1, parent2 = selected[i], selected[i+1]
            if random.random() < crossover_rate:
                # Blend crossover
                alpha = random.random()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(coarse_points):
                if random.random() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                    child2[j] += np.random.normal(0, 0.05)
            
            child1 = np.clip(child1, 0, 1)
            child2 = np.clip(child2, 0, 1)
            child1 = child1 * (coarse_points/2) / np.sum(child1)
            child2 = child2 * (coarse_points/2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
    
    # Select best coarse solution and upscale
    best_coarse = None
    best_coarse_c5 = float('inf')
    for h in population:
        c5 = evaluate_erdos_solution(h, coarse_dx)
        if c5 < best_coarse_c5:
            best_coarse_c5 = c5
            best_coarse = h.copy()
    
    # Upscale to fine grid (n=35)
    fine_h = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, coarse_points), 
                       best_coarse)
    fine_h = fine_h * (n_points/2) / np.sum(fine_h)
    fine_h = np.clip(fine_h, 0, 1)
    
    h_current = fine_h
    cur_c5 = evaluate_erdos_solution(h_current, dx)
    
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = h_current.copy()
    
    # Fine-tune with gradient descent
    step_size = 0.01
    max_iter = 10000
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_current[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= h_current[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_current[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_current[i]
        
        h_new = h_current - step_size * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        if c5_new >= evaluate_erdos_solution(h_current, dx):
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            h_current = h_new.copy()
            step_size *= 1.05
            
        iter_count += 1
    
    return (list(best_h), best_c5, n_points)
