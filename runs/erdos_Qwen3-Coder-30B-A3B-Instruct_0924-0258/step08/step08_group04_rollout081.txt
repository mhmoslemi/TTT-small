import numpy as np
from time import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_onto_constraints(h, n_points):
    """Project h onto the feasible set: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Adjust to meet sum constraint using scaling
    sum_h = np.sum(h)
    if abs(sum_h) > 1e-10:
        h = h * (n_points / 2) / sum_h
    else:
        h = np.ones_like(h) * (n_points / 2) / n_points
    
    return h

def compute_gradient(h, max_lags, n_points):
    """Compute gradient based on top lags"""
    grad = np.zeros(n_points)
    for k in max_lags:
        if k >= 0:
            # h[i] contributes to lag k: h[i] * (1 - h[i+k])
            for i in range(n_points - k):
                grad[i] += 1 - h[i + k]
                if i + k < n_points:
                    grad[i + k] -= h[i]
        else:
            # h[i] contributes to negative lag k: h[i] * (1 - h[i+k])
            for i in range(n_points + k):
                grad[i] += 1 - h[i - k]
                if i - k >= 0:
                    grad[i - k] -= h[i]
    return grad

def line_search(h, grad, c5_func, dx, alpha_max=1.0):
    """Perform backtracking line search"""
    alpha = alpha_max
    c5_current = c5_func(h)
    c5_prev = c5_current
    
    # Try decreasing step sizes until we find an improvement
    while alpha > 1e-10:
        h_new = h - alpha * grad
        h_new = project_onto_constraints(h_new, len(h))
        c5_new = c5_func(h_new)
        
        if c5_new < c5_prev:
            return alpha, h_new
        else:
            alpha *= 0.5
            c5_prev = c5_new
            
    return alpha, h

    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    step_size = 0.01
    iter_count = 0
    stagnation_count = 0
    max_stagnation = 2000
    
    # Gradient descent
    while time() - start_time < budget_s * 0.95 and stagnation_count < max_stagnation:
        # Compute all correlations for all possible lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Get top K lags
        max_lags_idx = np.argsort(corr)[-5:][::-1]
        max_lags = [i - (len(h_current) - 1) for i in max_lags_idx]
        
        # Compute gradient
