import numpy as np
from time import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # SDP-based approach to improve the solution
    # Binary search on C5 using SDP
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    max_iterations = 100
    
    # Sample enough shifts for robust SDP constraints
    shifts = np.arange(1, n_points)
    np.random.shuffle(shifts)
    shift_sample = shifts[:100]
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        
        # SDP relaxation
        h_var = cp.Variable(n_points)
        Y = cp.Variable((n_points, n_points), symmetric=True)
        
        constraints = [
            Y >> 0,  # Y is positive semidefinite
            cp.diag(Y) == h_var,  # Diagonal elements of Y equal h_var
            cp.sum(h_var) == n_points / 2,  # Sum constraint
        ]
        
        # Add overlap constraints
        for k in shift_sample:
            # Sum of h[i] * (1 - h[i+k]) over all valid i
            expr = 0
            for i in range(n_points):
                if i + k < n_points:
                    expr += h_var[i] - Y[i, i+k]
            constraints.append(expr <= mid * dx)
        
        objective = cp.Minimize(0)  # Dummy objective
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sdp = h_var.value
                # Projection: ensure feasible region
                h_sdp = np.clip(h_sdp, 0, 1)
                h_sdp = h_sdp * (n_points / 2) / np.sum(h_sdp)
                c5_sdp = evaluate_erdos_solution(h_sdp, dx)
                
                if c5_sdp < best_c5:
                    best_c5 = c5_sdp
                    best_h = h_sdp.copy()
                    
                high = mid
            else:
                low = mid
        except Exception:
            low = mid

    # Local refinement
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    def gradient(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h[i] - h[j]
            else:
                grad[i] = 1 - 2 * h[i]
        return grad

    # Apply optimization on best solution so far
    h_refine = best_h.copy()
    h_flat = h_refine.copy()
    
    for i in range(1000):
        if time() - start_time > budget_s * 0.95:
            break
        h_flat = h_flat - 0.01 * gradient(h_flat)
        h_flat = np.clip(h_flat, 0, 1)
        h_flat = h_flat * (n_points / 2) / np.sum(h_flat)
        
        c5_trial = evaluate_erdos_solution(h_flat, dx)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_flat.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
