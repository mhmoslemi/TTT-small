import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_onto_feasible(h, n_points):
    """Project h onto feasible set: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h_proj = np.clip(h, 0, 1)
    current_sum = np.sum(h_proj)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) < 1e-10:
        return h_proj
    
    # Scale to meet sum constraint
    if current_sum > 0:
        h_proj = h_proj * target_sum / current_sum
        h_proj = np.clip(h_proj, 0, 1)
    
    # If still not exactly the right sum, adjust carefully
    current_sum = np.sum(h_proj)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Adjust the largest elements
        indices = np.argsort(h_proj)
        for i in range(min(len(indices), int(abs(diff)*100))):
            idx = indices[-(i+1)]
            if h_proj[idx] < 1 and diff > 0:
                add = min(diff, 1 - h_proj[idx])
                h_proj[idx] += add
                diff -= add
            elif h_proj[idx] > 0 and diff < 0:
                sub = min(-diff, h_proj[idx])
                h_proj[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    
    return h_proj

def line_search(h, grad, dx, step_size, n_points):
    """Line search to find a good step size"""
    for _ in range(10):  # Max iterations in line search
        h_new = h - step_size * grad
        h_new = project_onto_feasible(h_new, n_points)
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < evaluate_erdos_solution(h, dx):
            return h_new, c5_new
        step_size *= 0.5
    return h, evaluate_erdos_solution(h, dx)

    # Parameters for optimization
    max_iter = 50000
    step_size = 0.01
    K_top_lags = 5  # Top K lags to consider
    
    iter_count = 0
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all correlations for all possible lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        
        # Get the top K lags by overlap and compute combined gradient
        max_lag_indices = np.argsort(corr)[-K_top_lags:][::-1]
        gradient = np.zeros(n_points)
        k_lags = []
        
        for idx in max_lag_indices:
            lag = idx - (len(h_current) - 1)
            k_lags.append(lag)
            
            # Update gradient for this lag
            if lag >= 0:

                    gradient[i] += 1 - h_current[i + lag]
                    if i + lag < n_points:

                    if i - lag >= 0:
