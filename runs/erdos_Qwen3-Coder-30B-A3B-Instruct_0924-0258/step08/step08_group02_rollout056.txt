import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing for global search
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    T = T_init
    max_iter = int(1e5)
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Perturbation parameter
    epsilon = 0.1
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Make perturbation
        h_new = h.copy()
        idx1, idx2 = np.random.randint(0, n_points, 2)
        delta = np.random.uniform(-epsilon, epsilon)
        
        # Apply perturbation
        h_new[idx1] += delta
        h_new[idx2] -= delta
        
        # Enforce bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Compute overlaps
        old_overlap = evaluate_erdos_solution(h, dx)
        new_overlap = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if new_overlap < old_overlap:
            h = h_new.copy()
            if new_overlap < best_c5:
                best_c5 = new_overlap
                best_h = h.copy()
        else:
            # Accept with probability
            prob = np.exp(-(new_overlap - old_overlap) / T)
            if np.random.rand() < prob:
                h = h_new.copy()
                if new_overlap < best_c5:
                    best_c5 = new_overlap
                    best_h = h.copy()
        
        # Cool down
        T = max(T * alpha, T_min)
    
    # Local refinement using L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        # Normalize for optimization
        h_normalized = best_h * (n_points / 2) / np.sum(best_h)
        
        res = minimize(
            obj,
            h_normalized,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    # Final validation
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
