import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.5  # Make it sum to n_points/2
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    max_iter = int(1e6)
    no_improvement_limit = 5000
    no_improvement_count = 0
    
    # Main SA loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate new candidate by swapping values between two indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(0, 0.1)
        
        # Ensure we respect bounds
        max_delta = min(delta, h_current[i], 1 - h_current[j])
        if max_delta <= 0:
            continue
            
        # Apply change
        h_new = h_current.copy()
        h_new[i] += max_delta
        h_new[j] -= max_delta
        
        # Ensure bounds
        h_new[i] = np.clip(h_new[i], 0, 1)
        h_new[j] = np.clip(h_new[j], 0, 1)
        
        # Renormalize to maintain sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5:
            h_current = h_new
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            # SA acceptance probability
            if np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
                h_current = h_new
            no_improvement_count += 1
            
        # Cooling schedule
        temp *= cooling_rate
        
        # Stop if temperature is too low
        if temp < min_temp or no_improvement_count > no_improvement_limit:
            break
    
    # Final refinement with local search
    try:
        # Try to refine using gradient descent on best solution
        h_refined = best_h.copy()
        step_size = 0.001
        for _ in range(1000):
            # Compute overlap values
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            lag_indices = np.arange(len(corr))
            lags = lag_indices - (len(h_refined) - 1)
            overlap_values = corr * dx
            
            # Find maximum
            max_idx = np.argmax(overlap_values)
            max_lag = lags[max_idx]
            
            # Gradient
            gradient = np.zeros_like(h_refined)
            for i in range(len(h_refined)):
                if max_lag >= 0:
                    if i + max_lag < len(h_refined):
                        gradient[i] = -1
                else:
                    if i - max_lag >= 0:
                        gradient[i] = -1
                        
            # Project onto constraints
            sum_h = np.sum(h_refined)
            projection = (n_points/2 - sum_h) / len(h_refined)
            
            # Update
            h_refined = h_refined - step_size * gradient + step_size * projection
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            
            # Check for convergence
            if np.sum(np.abs(h_refined - best_h)) < 1e-6:
                break
                
        refined_c5 = evaluate_erdos_solution(h_refined, dx)
        if refined_c5 < best_c5:
            best_h = h_refined
            best_c5 = refined_c5
            
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
