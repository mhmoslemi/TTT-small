import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 2
    stagnation_limit = 100
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        population[0] = h_start.copy()
        for i in range(1, population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)
    else:
        for i in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)
            
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    # GA main loop
    no_improvement_count = 0
    generation = 0
    
    while time() - start_time < budget_s * 0.7 and generation < generations:
        # Evaluate fitness of all individuals
        fitness_values = [fitness(individual) for individual in population]
        sorted_indices = np.argsort(fitness_values)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness_values[i] for i in sorted_indices]
        
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_h = sorted_population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= stagnation_limit:
            break
            
        # Select parents via tournament selection
        def tournament_selection(k=3):
            tournament_indices = np.random.choice(population_size, k)
            tournament_fitness = [fitness_values[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]
            return population[winner_index]
        
        # Create new population
        new_population = sorted_population[:elite_size]
        
        while len(new_population) < population_size:
            # Crossover
            parent1 = tournament_selection()
            parent2 = tournament_selection()
            
            if random.random() < crossover_rate:
                # Blend crossover
                alpha = random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
                
            # Mutation
            if random.random() < mutation_rate:
                # Add small random change
                mutation_strength = 0.05 * (1.0 - generation / generations)
                for i in range(n_points):
                    if random.random() < 0.1:
                        child[i] += random.gauss(0, mutation_strength)
                # Ensure constraints
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                
            new_population.append(child)
            
        population = new_population
        generation += 1
    
    # Final local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
