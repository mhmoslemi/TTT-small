import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.999
    min_T = 1e-6
    max_iter = 100000
    no_improvement_count = 0
    max_no_improvement = 5000
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor by swapping delta between two indices
        i, j = np.random.randint(0, n_points, 2)
        if i == j:
            j = (j + 1) % n_points
            
        # Compute feasible delta
        delta_max = min(h_current[i], 1 - h_current[j])
        if delta_max <= 0:
            continue
            
        delta = np.random.uniform(0, delta_max)
        
        # Create new candidate
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds are respected
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize to satisfy sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Metropolis criterion
        delta_E = c5_new - best_c5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h_current = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_current.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Cooling schedule
        T *= cooling_rate
        if T < min_T:
            T = min_T
            
        if no_improvement_count > max_no_improvement:
            # Reset with new random values
            h_current = np.random.rand(n_points)
            h_current = h_current * (n_points/2) / np.sum(h_current)
            no_improvement_count = 0
    
    # Final refinement using SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
