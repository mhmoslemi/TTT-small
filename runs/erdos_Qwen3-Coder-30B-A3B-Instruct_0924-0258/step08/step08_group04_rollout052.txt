import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Multiresolution approach
    # Start with coarse resolution
    n_coarse = 15
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.array([np.mean(h_current[i*n_points//n_coarse:(i+1)*n_points//n_coarse]) 
                        for i in range(n_coarse)])
    h_coarse = h_coarse * (n_coarse/2) / np.sum(h_coarse)
    
    # Population-based optimization on coarse grid
    population_size = 50
    population = []
    for _ in range(population_size):
        h_rand = np.random.rand(n_coarse)
        h_rand = h_rand * (n_coarse/2) / np.sum(h_rand)
        h_rand = np.clip(h_rand, 0, 1)
        population.append(h_rand)
    
    # Evaluate initial population
    fitnesses = []
    for h in population:
        fitness = -evaluate_erdos_solution(h, dx_coarse)
        fitnesses.append(fitness)
    
    # Genetic optimization on coarse grid
    for gen in range(100):
        if time() - start_time > budget_s * 0.4:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        top_indices = sorted_indices[:population_size//2]
        
        # Selection
        selected = [population[i] for i in top_indices]
        
        # Crossover and mutation
        new_population = selected.copy()
        for i in range(population_size - len(selected)):
            if np.random.rand() < 0.5 and len(selected) > 1:
                p1, p2 = np.random.choice(selected, 2, replace=False)
                crossover_point = np.random.randint(1, n_coarse)
                child = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
            else:
                child = selected[np.random.randint(0, len(selected))].copy()
                # Mutation
                mut_point = np.random.randint(n_coarse)
                child[mut_point] += np.random.normal(0, 0.05)
                child[mut_point] = max(0, min(1, child[mut_point]))
            
            # Rescale and clip
            child = child * (n_coarse/2) / np.sum(child)
            child = np.clip(child, 0, 1)
            new_population.append(child)
        
        # Evaluate new population
        population = new_population
        fitnesses = []
        for h in population:
            fitness = -evaluate_erdos_solution(h, dx_coarse)
            fitnesses.append(fitness)
    
    # Select best from coarse
    best_coarse_idx = np.argmin(fitnesses)
    best_coarse = population[best_coarse_idx]
    
    # Upscale to fine grid
    h_fine = np.array([np.mean(best_coarse[i*n_coarse//n_points:(i+1)*n_coarse//n_points]) 
                      for i in range(n_points)])
    h_fine = h_fine * (n_points/2) / np.sum(h_fine)
    h_current = np.clip(h_fine, 0, 1)
    
    max_iter = 10000
    iter_count = 0
    step_size = 0.01
    
    # Gradient descent on fine grid
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lags = np.argsort(corr)[-5:]  # Top 5 lags
        max_lags = [lag - (len(h_current) - 1) for lag in max_lags]
        
        gradient = np.zeros(n_points)
        for max_lag in max_lags:
            if max_lag >= 0:
                for i in range(n_points - max_lag):
                    gradient[i] += 1 - h_current[i + max_lag]
                    if i + max_lag < n_points:
                        gradient[i + max_lag] -= h_current[i]
            else:
                for i in range(n_points + max_lag):
                    gradient[i] += 1 - h_current[i - max_lag]
                    if i - max_lag >= 0:
                        gradient[i - max_lag] -= h_current[i]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm
        
        h_new = h_current - step_size * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        if c5_new >= evaluate_erdos_solution(h_current, dx):
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            h_current = h_new
            step_size *= 1.05
        
        iter_count += 1
    
    return (list(best_h), best_c5, n_points)
