import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    population_size = 50
    n_points = 35
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        pop = [initial_h.copy()]
        for _ in range(population_size - 1):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            pop.append(h)
    else:
        pop = []
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            pop.append(h)
    
    # Store best solution
    best_h = None
    best_c5 = float('inf')
    no_improvement = 0
    max_no_improvement = 200
    
    # GA parameters
    crossover_rate = 0.8
    mutation_rate = 0.1
    mutation_step = 0.05
    
    def mutate(h, step):
        h_mut = h.copy()
        for i in range(n_points):
            if np.random.random() < mutation_rate:
                h_mut[i] += np.random.normal(0, step)
        # Enforce constraints
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
        return h_mut
    
    def crossover(h1, h2):
        if np.random.random() > crossover_rate:
            return h1.copy(), h2.copy()
        
        alpha = np.random.uniform(0.3, 0.7)
        h_child1 = alpha * h1 + (1 - alpha) * h2
        h_child2 = (1 - alpha) * h1 + alpha * h2
        
        # Enforce constraints
        h_child1 = np.clip(h_child1, 0, 1)
        h_child1 = h_child1 * (n_points / 2) / np.sum(h_child1)
        h_child2 = np.clip(h_child2, 0, 1)
        h_child2 = h_child2 * (n_points / 2) / np.sum(h_child2)
        
        return h_child1, h_child2
    
    gen = 0
    while time() - start_time < budget_s * 0.95:
        # Evaluate all population members
        fitnesses = [evaluate_erdos_solution(h, dx) for h in pop]
        
        # Update best
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = pop[min_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= max_no_improvement:
            break

        # Tournament selection
        parent_indices = []
        tournament_size = 3
        for _ in range(population_size):
            candidates = np.random.choice(population_size, tournament_size)
            best_candidate = candidates[np.argmin([fitnesses[i] for i in candidates])]
            parent_indices.append(best_candidate)
        
        # Generate new population
        new_pop = []
        
        # Add best individuals to new population
        top_indices = np.argsort(fitnesses)[:population_size//2]
        for i in top_indices:
            new_pop.append(pop[i].copy())
        
        # Create offspring
        while len(new_pop) < population_size:
            parents = np.random.choice(population_size, 2)
            h1, h2 = pop[parents[0]], pop[parents[1]]
            child1, child2 = crossover(h1, h2)
            child1 = mutate(child1, mutation_step)
            child2 = mutate(child2, mutation_step)
            new_pop.append(child1)
            if len(new_pop) < population_size:
                new_pop.append(child2)
        
        pop = new_pop[:population_size]
        mutation_step *= 0.98
        gen += 1
        
        # Periodically refine with SLSQP
        if gen % 100 == 0 and best_h is not None:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-10})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final SLSQP refinement
    if best_h is not None:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    # Ensure final result is valid
    if best_h is None:
        # Fallback
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
        else:
            best_h = np.ones(n_points) * 0.5
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
    # Double check validity
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "Values out of bounds"
    assert abs(np.sum(best_h) - n_points / 2.0) < 1e-10, "Sum constraint violated"
    
    return (list(best_h), best_c5, n_points)
