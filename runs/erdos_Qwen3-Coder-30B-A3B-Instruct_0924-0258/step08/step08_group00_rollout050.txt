import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initialize from initial_h_values or use default
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach
    resolutions = [35, 70, 140, 280]
    patience = 50
    no_improvement = 0
    temp_start = 1.0
    cooling_rate = 0.995
    
    # For each resolution level
    for res in resolutions:
        if res == 35:
            h = h_start.copy()
        else:
            # Interpolate to higher resolution
            h_coarse = h.copy()
            h = np.interp(np.linspace(0, 1, res), np.linspace(0, 1, len(h_coarse)), h_coarse)
            # Ensure sum constraint
            h = h * (res/2) / np.sum(h)
        
        n_points = res
        dx = 2.0 / n_points
        current_temp = temp_start
        
        for iteration in range(1000):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate neighbor solution
            h_new = h.copy()
            idx = random.randint(0, n_points - 1)
            perturb = random.uniform(-current_temp * 0.5, current_temp * 0.5)
            h_new[idx] += perturb
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Compute objective values
            c5_new = evaluate_erdos_solution(h_new, dx)
            c5_old = evaluate_erdos_solution(h, dx)
            
            # Accept or reject
            if c5_new < c5_old or random.random() < np.exp(-(c5_new - c5_old) / max(current_temp, 1e-10)):
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improvement = 0
                else:
                    no_improvement += 1
            else:
                no_improvement += 1
                
            if no_improvement >= patience:
                break
                
            # Cooling schedule
            current_temp *= cooling_rate
            if current_temp < 1e-6:
                break
                
        # Final SLSQP refinement
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
