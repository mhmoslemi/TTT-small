import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search bounds
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    # Sample shifts for feasibility problem
    num_shifts = 100
    
    while time() - start_time < budget_s * 0.95 and high - low > tolerance:
        mid = (low + high) / 2.0
        
        # Formulate CVXPY feasibility problem
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2.0
        ]
        
        # Sample shifts
        shifts = np.random.choice(n_points, num_shifts, replace=False)
        
        # Add overlap constraints
        for k in shifts:
            # Compute overlap for shift k
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h_var[i] * (1 - h_var[j])
            constraints.append(overlap <= mid * dx)
        
        # Create and solve problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                if h_sol is not None:
                    h_sol = np.maximum(0, np.minimum(1, h_sol))
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    c5_new = evaluate_erdos_solution(h_sol, dx)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_sol.copy()
                    high = mid
                else:
                    low = mid
            else:
                low = mid
        except:
            low = mid
            
    # Local refinement with gradient descent if needed
    h_refine = best_h.copy()
    dx_refine = 2.0 / n_points
    step_size = 0.1
    for _ in range(500):
        if time() - start_time > budget_s * 0.98:
            break
        corr = np.correlate(h_refine, 1 - h_refine, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h_refine[i] - h_refine[j]
            else:
                grad[i] = 1 - 2 * h_refine[i]
        
        h_new = h_refine - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx_refine)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_refine = h_new.copy()
        else:
            step_size *= 0.5
    
    return (list(best_h), best_c5, n_points)
