from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

        h_temp = np.random.random(n_points)

        return evaluate_erdos_solution(h, dx)

    def tournament_selection(pop, k=5):
        selected = []
        for _ in range(len(pop)):
            tournament = random.sample(pop, min(k, len(pop)))
            winner = min(tournament, key=fitness)
            selected.append(winner.copy())
        return selected

    def crossover(parent1, parent2):
        mask = np.random.random(len(parent1)) > 0.5
        child = np.where(mask, parent1, parent2)
        return child
    
    # Mutation: flip and perturb
    def mutate(h, rate=0.1):
        h_mut = h.copy()
        # Flip mutation
        if random.random() < 0.5:
            indices = random.sample(range(len(h)), max(1, int(len(h) * rate)))
            for idx in indices:
                delta = random.choice([-0.1, 0.1])
                h_mut[idx] = max(0, min(1, h_mut[idx] + delta))

        else:
            indices = random.sample(range(len(h)), max(1, int(len(h) * rate)))

                factor = random.uniform(0.9, 1.1)
                h_mut[idx] = max(0, min(1, h_mut[idx] * factor))
        return h_mut

    def enforce_sum_constraint(h):
        h = h * (n_points/2) / np.sum(h)
        return np.clip(h, 0, 1)

        except Exception:
