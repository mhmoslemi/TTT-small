import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary optimization with threshold-based updates
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Phase 1: Binary threshold search
    for phase in range(3):
        if time() - start_time > budget_s * 0.8:
            break
        
        # Generate binary approximation
        h_binary = np.zeros(n_points)
        indices = np.argsort(h)[::-1][:n_points//2]
        h_binary[indices] = 1.0
        
        # Refine with threshold-based gradient
        for iter in range(100):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Calculate overlaps
            overlaps = np.correlate(h_binary, 1 - h_binary, mode='full') * dx
            max_overlap = np.max(overlaps)
            l_max = np.argmax(overlaps)
            
            # Simple gradient ascent
            gradient = np.zeros(n_points)
            for i in range(n_points):
                j = (i + l_max) % n_points
                gradient[i] = -2 * dx * (1 - h_binary[j] - h_binary[i])
            
            # Update with threshold
            threshold = 0.05 * np.abs(gradient).max()
            update = np.where(np.abs(gradient) > threshold, np.sign(gradient) * 0.01, 0)
            
            h_new = h_binary + update
            h_new = np.clip(h_new, 0, 1)
            
            # Project to valid sum
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Toggle binary based on proximity to 0 or 1
            h_binary = np.where(h_new > 0.5, 1.0, 0.0)
            h_binary = h_binary * (n_points/2) / np.sum(h_binary)
            
            current_c5 = evaluate_erdos_solution(h_binary, dx)
            if current_c5 < best_c5:
                best_h = h_binary.copy()
                best_c5 = current_c5
    
    # Phase 2: Local refinement with L-BFGS
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint(h):
            return np.sum(h) - n_points / 2
            
        # Refine with scipy's L-BFGS-B
        res = minimize(
            objective,
            best_h.copy(),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
