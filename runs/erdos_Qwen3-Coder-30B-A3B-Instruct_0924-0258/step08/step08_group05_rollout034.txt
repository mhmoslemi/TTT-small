import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.4
        h_start[1] = 0.6
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    T = 0.1
    cooling_rate = 0.999
    min_temp = 1e-6
    max_iter = int(1e6)
    no_improvement_limit = 500
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor solution by swapping delta between two indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(0.001, 0.05)
        
        # Ensure we can make the swap without violating constraints
        max_delta = min(h_current[i], 1 - h_current[j])
        if max_delta < delta:
            delta = max_delta
            
        if delta <= 0:
            continue
            
        # Create new candidate solution
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds are respected
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy sum constraint
        sum_h = np.sum(h_new)
        h_new = h_new * (n_points/2) / sum_h
        
        # Calculate energies
        c5_old = evaluate_erdos_solution(h_current, dx)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject the new solution
        if c5_new < c5_old:
            h_current = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                no_improvement_count = 0
        else:
            # Accept with probability based on temperature
            if np.random.rand() < np.exp(-(c5_new - c5_old) / T):
                h_current = h_new
            no_improvement_count += 1
        
        # Update temperature
        T *= cooling_rate
        if T < min_temp:
            T = min_temp
            
        # Early stopping if no improvement for a while
        if no_improvement_count > no_improvement_limit:
            break
            
    # Refinement with local search
    if time() - start_time < budget_s * 0.90:
        # Perform gradient-based refinement on the best solution
        try:
            step_size = 0.01
            tolerance = 1e-6
            for _ in range(1000):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                corr = np.correlate(best_h, 1 - best_h, mode='full')
                lag_indices = np.arange(len(corr))
                lags = lag_indices - (len(best_h) - 1)
                overlap_values = corr * dx
                max_idx = np.argmax(overlap_values)
                max_lag = lags[max_idx]
                
                # Compute gradient - simplified approach
                gradient = np.zeros_like(best_h)
                for i in range(len(best_h)):
                    if max_lag >= 0:
                        if i + max_lag < len(best_h):
                            gradient[i] = -1.0
                    else:
                        if i - max_lag >= 0:
                            gradient[i] = -1.0
                
                # Gradient descent step
                h_new = best_h - step_size * gradient
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                else:
                    break
                    
        except Exception:
            pass
            
    return (list(best_h), best_c5, n_points)
