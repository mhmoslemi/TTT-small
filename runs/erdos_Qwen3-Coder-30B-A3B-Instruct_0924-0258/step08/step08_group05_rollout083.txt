import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T = T_init
    cooling_rate = 0.999
    min_temp = 1e-6
    max_iter = int(1e6)
    no_improvement_limit = 5000
    no_improvement_count = 0
    
    # Initialize temperature based on initial variance
    initial_energy = evaluate_erdos_solution(h_current, dx)
    T = initial_energy * 0.1 if initial_energy > 0 else 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor by swapping small delta between two random indices
        i, j = np.random.randint(0, n_points, 2)
        while i == j:
            j = np.random.randint(0, n_points)
            
        # Compute allowable delta
        max_delta = min(h_current[i], 1 - h_current[j])
        if max_delta <= 0:
            continue
            
        delta = np.random.uniform(0, max_delta)
        
        # Create new candidate
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
            
        # Apply sum constraint
        sum_diff = np.sum(h_new) - n_points / 2
        if abs(sum_diff) > 1e-10:
            # Adjust one more element to maintain sum
            k = np.random.randint(0, n_points)
            while k == i or k == j:
                k = np.random.randint(0, n_points)
            adjustment = sum_diff / (n_points - 2)  # Distribute evenly to others, not in this format but adjust one
            h_new[k] -= adjustment
            # Ensure bounds still hold
            if h_new[k] < 0 or h_new[k] > 1:
                continue
                
        # Calculate energies
        energy_current = evaluate_erdos_solution(h_current, dx)
        energy_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject the move
        if energy_new < energy_current:
            h_current = h_new
            if energy_new < best_c5:
                best_c5 = energy_new
                best_h = h_new.copy()
            no_improvement_count = 0
        else:
            # Possibly accept the worse solution
            if np.random.rand() < np.exp(-(energy_new - energy_current) / T):
                h_current = h_new
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
        # Cool down
        T *= cooling_rate
        if T < min_temp:
            T = min_temp
            
        # Check for early termination
        if no_improvement_count > no_improvement_limit:
            break
    
    # Final refinement with optimization
    try:
        from scipy.optimize import minimize
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
