import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adam optimizer with constraints
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    alpha = 0.01
    
    h_current = h_start.copy()
    last_improvement = 0
    improvement_threshold = 1e-4
    
    def compute_gradient(h, max_lag):
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_lag) % n_points
            grad[i] += 2 * (1 - h[j] - h[i])
        return grad * dx

    iteration = 0
    while time() - start_time < budget_s * 0.90:
        # Evaluate current solution
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5 - improvement_threshold:
            best_c5 = c5_current
            best_h = h_current.copy()
            last_improvement = iteration
            
        # Compute overlap correlation and identify max lag
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        max_lag = max_idx - (len(h_current) - 1)
        
        # Compute gradient
        grad = compute_gradient(h_current, max_lag)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h_current - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Projection to feasible set
        h_new = np.clip(h_new, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h_new)
        h_new = h_new * scale_factor

        # Update current solution
        h_current = h_new

        iteration += 1
        
        # Restart Adam if no improvement for a while
        if iteration - last_improvement > 100:
            m.fill(0)
            v.fill(0)
            t = 0
            alpha *= 0.5
            if alpha < 1e-6:
                break
    
    # Final refinement with SLSQP if acceptable
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
                
    return (list(best_h), best_c5, n_points)
