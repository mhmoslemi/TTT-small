import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter_sa = 100000
    epsilon = 0.1
    
    T = T_init
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    c5_current = evaluate_erdos_solution(h_current, dx)
    
    # Track best solution
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Simulated Annealing loop
    for i in range(max_iter_sa):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Perturbation: randomly select two indices
        i1 = np.random.randint(0, n_points)
        i2 = np.random.randint(0, n_points)
        
        # Create a candidate solution by perturbing two values
        h_candidate = h_current.copy()
        
        # Ensure we don't make the same index twice
        delta = np.random.uniform(-epsilon, epsilon)
        if abs(delta) < 1e-10:
            delta = epsilon / 2 if np.random.random() < 0.5 else -epsilon / 2
        
        delta1 = delta
        delta2 = -delta1  # To maintain sum constraint
        
        # Apply perturbations
        h_candidate[i1] += delta1
        h_candidate[i2] += delta2
        
        # Clip to [0, 1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Renormalize to maintain sum = n_points/2
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        
        # Compute overlap for candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Acceptance criterion
        if c5_candidate < c5_current:
            h_current = h_candidate
            c5_current = c5_candidate
            
            # Update best if better
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
        else:
            # Accept with probability
            prob = np.exp(-(c5_candidate - c5_current) / T)
            if np.random.random() < prob:
                h_current = h_candidate
                c5_current = c5_candidate
                
        # Decrease temperature
        T = max(T_min, T * alpha)
    
    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return np.max(np.correlate(h, 1 - h, mode='full') * dx)
            
        # Ensure constraints
        h_refined = best_h.copy() * (n_points/2) / np.sum(best_h)
        
        res = minimize(
            objective,
            h_refined,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if res.success:
            h_final = res.x
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
                
    except Exception:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
