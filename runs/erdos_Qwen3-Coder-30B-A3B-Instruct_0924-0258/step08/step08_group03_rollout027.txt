import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Use CVXPY for convex optimization
    def optimize_with_cvxpy(h_vals, k_top=10, max_iter=10):
        n = len(h_vals)
        h = cp.Variable(n)
        constraints = [h >= 0, h <= 1]
        
        # Compute all correlations and get top k lags
        corr_vals = np.correlate(h_vals, 1 - h_vals, mode='full')
        top_lags = np.argsort(corr_vals)[-k_top:][::-1]  # Get indices of top k
        top_lags = [lag - (n-1) for lag in top_lags]
        
        # Compute sum of top overlaps
        objective_terms = []
        for lag in top_lags:
            if lag >= 0:
                term = cp.sum(cp.multiply(h[:n-lag], 1 - h_vals[lag:])) * dx
            else:
                term = cp.sum(cp.multiply(h_vals[:n+lag], 1 - h[-lag:])) * dx
            objective_terms.append(term)
        
        objective = cp.Minimize(cp.sum(objective_terms))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if h.value is not None:
            new_h = np.clip(h.value, 0, 1)
            # Renormalize
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            return new_h
        else:
            return h_vals.copy()
    
    # Filtering function
    def apply_filter(h_vals, filter_width=5):
        if filter_width >= len(h_vals):
            return h_vals.copy()
        # Create simple moving average filter
        filt = np.ones(filter_width) / filter_width
        h_filtered = np.convolve(h_vals, filt, mode='same')
        # Project back to [0,1]
        h_filtered = np.clip(h_filtered, 0, 1)
        # Renormalize
        h_filtered = h_filtered * (n_points / 2) / np.sum(h_filtered)
        return h_filtered

    # Main loop
    max_iterations = 2000
    k_top = 10
    filter_width = 5
    step = 0
    
    while step < max_iterations and (time() - start_time) < budget_s * 0.95:
        # Optimize convex relaxation
        h_opt = optimize_with_cvxpy(h, k_top=k_top, max_iter=5)
        
        # Apply filtering
        h_filtered = apply_filter(h_opt, filter_width=filter_width)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_filtered, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_filtered.copy()
        
        h = h_filtered.copy()
        
        # Increase k_top gradually to capture more overlaps
        k_top = min(20, k_top + 1)
        # Increase filter width slightly
        filter_width = min(10, filter_width + 1)
        
        step += 1
    
    return (list(best_h), best_c5, n_points)
