import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # PSO parameters
    num_particles = 75
    inertia = 0.8
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    perturbation_rate = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    particle_best = []
    particle_best_fitness = []
    
    # Generate initial feasible particles
    for _ in range(num_particles):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        particle_best.append(h.copy())
        particle_best_fitness.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    global_best_idx = np.argmin(particle_best_fitness)
    global_best = particle_best[global_best_idx].copy()
    global_best_fitness = particle_best_fitness[global_best_idx]
    
    max_iter = int(1e6)
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 5000
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # PSO iteration
        for i in range(num_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_weight * r1 * (particle_best[i] - h)
            social = social_weight * r2 * (global_best - h)
            v = inertia * v + cognitive + social
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            
            # Update position
            h_new = h + v
            
            # Project to feasible set
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate fitness
            fitness = evaluate_erdos_solution(h_new, dx)
            
            # Update particle best
            if fitness < particle_best_fitness[i]:
                particle_best[i] = h_new.copy()
                particle_best_fitness[i] = fitness
            
            # Update global best
            if fitness < global_best_fitness:
                global_best = h_new.copy()
                global_best_fitness = fitness
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            
            # Store new values
            particles[i] = h_new
            velocities[i] = v
        
        # Adapt inertia
        if iter_count % 100 == 0:
            inertia = max(0.4, 0.8 - 0.4 * (iter_count / 1000))
        
        # Apply periodic perturbation
        if iter_count % 1000 == 0 and np.random.rand() < perturbation_rate:
            for i in range(num_particles):
                # Randomly perturb a fraction of particles
                if np.random.rand() < 0.3:
                    h = particles[i]
                    idx = np.random.randint(0, n_points)
                    h[idx] = np.random.uniform(0, 1)
                    # Renormalize
                    h = h * (n_points/2) / np.sum(h)
                    particles[i] = h
        
        iter_count += 1
        
        if no_improvement_count > max_no_improvement:
            break
    
    # Final refinement with gradient descent
    h_refined = global_best.copy()
    step_size = 0.01
    for _ in range(500):
        if time() - start_time > budget_s * 0.99:
            break
        # Compute correlations and find max lag
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_refined) - 1)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_refined[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= h_refined[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_refined[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_refined[i]
        
        # Gradient descent step
        h_new = h_refined - step_size * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < evaluate_erdos_solution(h_refined, dx):
            h_refined = h_new
            step_size *= 1.05
        else:
            step_size *= 0.5
            if step_size < 1e-10:
                break
    
    final_c5 = evaluate_erdos_solution(h_refined, dx)
    
    return (list(h_refined), final_c5, n_points)
