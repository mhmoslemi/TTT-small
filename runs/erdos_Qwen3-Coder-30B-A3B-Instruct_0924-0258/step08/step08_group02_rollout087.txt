import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Ensure the sum constraint
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Binary search with threshold
    max_iterations = 500
    threshold_schedule = np.linspace(1.0, 0.01, 50)
    
    h = h_start.copy()
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Threshold-based update
        threshold = threshold_schedule[min(iter // 10, len(threshold_schedule) - 1)]
        update_mask = np.abs(gradient) > threshold
        
        # Update h values
        h_update = h.copy()
        for i in range(n_points):
            if update_mask[i]:
                if gradient[i] > 0:
                    h_update[i] = min(1.0, h[i] + 0.01)
                else:
                    h_update[i] = max(0.0, h[i] - 0.01)
        
        # Enforce bounds and sum constraint
        h_update = np.clip(h_update, 0, 1)
        h_update = h_update * (n_points / 2) / np.sum(h_update)
        
        # Track best solution
        current_c5 = evaluate_erdos_solution(h_update, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_update.copy()
        
        h = h_update.copy()
    
    # Local refinement with L-BFGS-B using a tighter tolerance
    try:
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception as e:
        pass
        
    return (list(best_h), best_c5, n_points)
