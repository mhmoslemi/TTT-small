import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # PSO parameters
    num_particles = 75
    inertia_start = 0.8
    inertia_end = 0.4
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    perturbation_rate = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    particle_best = []
    particle_best_c5 = []
    
    for _ in range(num_particles):
        h = np.random.rand(n_points) * 0.5 + 0.25  # Random feasible solution
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        particle_best.append(h.copy())
        particle_best_c5.append(evaluate_erdos_solution(h, dx))
        
    # Global best
    global_best_idx = np.argmin(particle_best_c5)
    global_best = particle_best[global_best_idx].copy()
    global_best_c5 = particle_best_c5[global_best_idx]
    
    # PSO iteration
    max_iter = int(1e6)
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_count / max_iter)
        
        for i in range(num_particles):
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            cognitive_comp = cognitive_weight * r1 * (particle_best[i] - particles[i])
            social_comp = social_weight * r2 * (global_best - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive_comp + social_comp
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible set
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate fitness
            c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update particle best
            if c5 < particle_best_c5[i]:
                particle_best[i] = particles[i].copy()
                particle_best_c5[i] = c5
            
            # Update global best
            if c5 < global_best_c5:
                global_best = particles[i].copy()
                global_best_c5 = c5
                
        # Perturbation to escape local minima
        if iter_count % 1000 == 0 and iter_count > 0:
            for i in range(num_particles):
                if np.random.rand() < perturbation_rate:
                    particles[i] = np.random.rand(n_points) * 0.5 + 0.25
                    particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
        
        iter_count += 1
                
    # Final refinement using local search
    refined_h = global_best.copy()
    
    # Perform local gradient descent on best solution
    step_size = 0.005
    refinements = 1000
    for _ in range(refinements):
        if time() - start_time > budget_s * 0.95:
            break
            
        corr = np.correlate(refined_h, 1 - refined_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(refined_h) - 1)
        
        gradient = np.zeros(n_points)
        
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - refined_h[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= refined_h[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - refined_h[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= refined_h[i]
        
        # Gradient descent step
        h_new = refined_h - step_size * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < evaluate_erdos_solution(refined_h, dx):
            refined_h = h_new
            
    # Final evaluation
    final_c5 = evaluate_erdos_solution(refined_h, dx)
    
    return (list(refined_h), final_c5, n_points)
