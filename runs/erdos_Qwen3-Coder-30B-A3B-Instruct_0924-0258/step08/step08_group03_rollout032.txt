import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search parameters
    low, high = 0.38, 0.38085
    tolerance = 1e-5
    
    # Sample shifts for constraints - avoid full correlation for efficiency
    n_shifts = 100
    shift_indices = np.random.choice(n_points, size=n_shifts, replace=False) 
    
    while time() - start_time < budget_s * 0.9 and high - low > tolerance:
        mid = (low + high) / 2
        
        # Formulate convex feasibility problem
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Add overlap constraints for sampled shifts
        for k in shift_indices:
            # Compute overlap for shift k
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h_var[i] * (1 - h_var[j])
            constraints.append(overlap_expr <= mid * dx)
        
        # Solve convex feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                # Check real correlation (full convolution)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                high = mid
            else:
                low = mid
        except:
            low = mid
            
    # Local refinement of best solution
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use scipy optimization for final refinement
    refined = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
    
    if refined.success:
        final_h = refined.x
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h
    
    return (list(best_h), best_c5, n_points)
