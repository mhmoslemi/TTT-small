import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_population(n_points, population_size, initial_h_values=None):
    population = []
    
    if initial_h_values is not None:
        h_start = np.array(initial_h_values)
        h_start = h_start * (n_points/2) / np.sum(h_start)
        population.append(h_start.copy())
    
    while len(population) < population_size:
        h = np.random.random(n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    selected_indices = np.random.choice(len(population), tournament_size)
    best_idx = selected_indices[np.argmin([fitnesses[i] for i in selected_indices])]
    return population[best_idx]

def uniform_crossover(parent1, parent2):
    mask = np.random.random(len(parent1)) > 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(h, mutation_rate=0.1, amplitude=0.1):
    h = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            # Flip mutation
            if random.random() < 0.5:
                h[i] = min(1.0, h[i] + amplitude)
            else:
                h[i] = max(0.0, h[i] - amplitude)
    
    # Apply multiplicative perturbation to 10% of elements
    n_perturb = max(1, int(len(h) * 0.1))
    indices = np.random.choice(len(h), n_perturb, replace=False)
    for i in indices:
        factor = random.uniform(0.9, 1.1)
        h[i] = np.clip(h[i] * factor, 0.0, 1.0)
    
    return h

def adjust_to_constraint(h, n_points):
    """Adjust h to satisfy sum constraint while preserving values in [0,1]"""
    target_sum = n_points / 2
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Simple scaling approach
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.full_like(h, 1.0 / n_points * target_sum)  # uniform distribution
    
    # Clamp to [0,1]
    h = np.clip(h, 0.0, 1.0)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    population_size = 50
    max_generations = 10000
    no_improvement_limit = 100
    
    # Initialize population
    population = initialize_population(n_points, population_size, initial_h_values)
    
    # Evaluate fitness for all individuals
    fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
    
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    no_improvement_count = 0
    
    # Start evolution
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create new generation
        new_population = [best_h.copy()]  # Elitism
        
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child = uniform_crossover(parent1, parent2)
            
            # Mutation
            child = mutate(child)
            
            # Ensure constraint
            child = adjust_to_constraint(child, n_points)
            
            new_population.append(child)
            
        # Replace old population
        population = new_population
        
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = current_best_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= no_improvement_limit:
            break
    
    # Final re-evaluation to ensure quality
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
