import numpy as np
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Set up binary search bounds
    low = 0.38
    high = 0.38085
    
    # Sample shifts for SDP constraints
    shift_samples = np.random.choice(range(1, n_points), size=min(100, n_points-1), replace=False)
    
    def solve_sdp_with_shifts(c5_val, h_guess):
        # Variable definitions  
        h = cp.Variable(n_points)
        Y = cp.Variable((n_points, n_points), symmetric=True)
        
        # Constraints
        constraints = [
            Y >> 0,  # Y is positive semidefinite
            cp.diag(Y) == h,  # Diagonal elements of Y match h
            cp.sum(h) == n_points / 2,  # Normalization constraint
        ]
        
        # Overlap constraints for sampled shifts
        for k in shift_samples:
            # Sum for shift k
            shift_sum = 0
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    shift_sum += Y[i, j]
            constraints.append(cp.sum(h) - shift_sum <= c5_val * dx)
        
        # Objective: minimize C_5 (we're doing feasibility check, not minimizing)
        objective = cp.Minimize(0)
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=MOSEK, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                return h.value
        except:
            pass
        return None
    
    # Binary search
    while time() - start_time < budget_s * 0.95 and high - low > 1e-5:
        mid = (low + high) / 2
        h_result = solve_sdp_with_shifts(mid, h)
        if h_result is not None:
            # Project to [0,1] and renormalize
            h_proj = np.clip(h_result, 0, 1)
            h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
            
            # Local refinement
            corr = np.correlate(h_proj, 1 - h_proj, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (n_points - 1)
            
            # Gradient descent
            step_size = 0.001
            for _ in range(1000):
                grad = np.zeros(n_points)
                for i in range(n_points):
                    j = i + max_lag
                    if 0 <= j < n_points:
                        grad[i] = 1 - 2 * h_proj[i] - h_proj[j]
                    else:
                        grad[i] = 1 - 2 * h_proj[i]
                
                h_new = h_proj - step_size * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    
                if c5_new < mid:
                    h_proj = h_new.copy()
                else:
                    break
            
            high = mid
        else:
            low = mid
    
    # Final local optimization on best solution
    for _ in range(1000):
        if time() - start_time > budget_s * 0.95:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * best_h[i] - best_h[j]
            else:
                grad[i] = 1 - 2 * best_h[i]
        
        h_new = best_h - 0.001 * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        else:
            break
    
    return (list(best_h), best_c5, n_points)
