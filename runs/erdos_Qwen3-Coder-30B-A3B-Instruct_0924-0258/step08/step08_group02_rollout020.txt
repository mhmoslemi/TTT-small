import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Thresholding-based optimization
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    threshold = 0.1
    cooling_rate = 0.95
    min_threshold = 0.01
    max_iterations = 500
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for each coordinate
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Rank coordinates by gradient magnitude
        idx_sorted = np.argsort(np.abs(gradient))[::-1]
        
        # Threshold based on gradient magnitude
        top_indices = idx_sorted[:max(1, n_points // 2)]
        
        # Update h values based on threshold
        h_new = h.copy()
        for i in top_indices:
            if h[i] >= 1 - threshold:
                h_new[i] = 1.0
            elif h[i] <= threshold:
                h_new[i] = 0.0
        
        # Ensure sum constraint
        current_sum = np.sum(h_new)
        if abs(current_sum - n_points/2) > 1e-6:
            # Distribute surplus or deficit to most impacted coordinates
            diff = n_points/2 - current_sum
            sorted_idx = np.argsort(np.abs(gradient))[::-1][:n_points//4]
            if len(sorted_idx) > 0:
                # Distribute evenly among top gradient coordinates
                for i in sorted_idx:
                    if abs(diff) < 1e-8:
                        break
                    delta = min(1.0 - h_new[i], max(-h_new[i], diff))
                    h_new[i] += delta
                    diff -= delta

        h = h_new
        h = np.clip(h, 0, 1)
        
        # Re-normalize if needed
        h = h * (n_points/2) / np.sum(h)

        # Evaluate
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Cool threshold
        threshold = max(min_threshold, threshold * cooling_rate)
        
    # Coordinate-wise refinement
    for _ in range(100):
        if time() - start_time > budget_s * 0.98:
            break
            
        improved = False
        # Optimize each point individually
        for i in range(n_points):
            if time() - start_time > budget_s * 0.98:
                break
                
            # Compute overlaps using current solution except h[i]
            def obj(val):
                h_test = best_h.copy()
                h_test[i] = val
                h_test = h_test * (n_points/2) / np.sum(h_test)
                overlaps = np.correlate(h_test, 1 - h_test, mode='full') * dx
                return np.max(overlaps)
            
            # Golden section search in [0,1]
            phi = (1 + np.sqrt(5)) / 2
            resphi = 2 - phi
            
            a, b = 0.0, 1.0
            v1 = a + resphi * (b - a)
            v2 = b - resphi * (b - a)
            
            f1 = obj(v1)
            f2 = obj(v2)
            
            while abs(b - a) > 1e-6:
                if f1 < f2:
                    b = v2
                    v2 = v1
                    f2 = f1
                    v1 = a + resphi * (b - a)
                    f1 = obj(v1)
                else:
                    a = v1
                    v1 = v2
                    f1 = f2
                    v2 = b - resphi * (b - a)
                    f2 = obj(v2)
                    
            new_val = (a + b) / 2
            best_h[i] = new_val
        
        # Re-normalize and check improvement
        best_h = best_h * (n_points/2) / np.sum(best_h)
        improved_c5 = evaluate_erdos_solution(best_h, dx)
        if improved_c5 < best_c5:
            best_c5 = improved_c5
            improved = True
        else:
            # If no improvement, break early
            break
    
    # Final L-BFGS refinement
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def obj(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)
        
        res = minimize(
            obj,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
