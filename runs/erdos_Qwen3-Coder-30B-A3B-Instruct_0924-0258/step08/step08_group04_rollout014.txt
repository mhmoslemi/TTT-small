import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points / 2 while keeping bounds [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) < 1e-10:
        return h
    if current_sum > 0:
        h = h * (n_points/2) / current_sum
    else:
        h = np.ones_like(h) * 0.5
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        n_points = len(h_current)
    else:
        n_points = 35
        h_current = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = project_to_feasible(h_current, n_points)
    
    # PSO parameters
    num_particles = 75
    inertia_start = 0.8
    inertia_end = 0.4
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    for i in range(num_particles):
        if i == 0:
            h = h_current.copy()
        else:
            # Add small random perturbation to initial
            h = h_current + np.random.normal(0, 0.02, n_points)
        h = project_to_feasible(h, n_points)
        particles.append(h)
        velocities.append(np.zeros(n_points))
    
    # Track best solutions
    particle_best = [p.copy() for p in particles]
    particle_best_fitness = [evaluate_erdos_solution(p, dx) for p in particles]
    global_best_idx = np.argmin(particle_best_fitness)
    global_best = particle_best[global_best_idx].copy()
    global_best_fitness = particle_best_fitness[global_best_idx]
    
    max_iter = 100000
    iter_count = 0
    
    # PSO loop
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_count / max_iter)
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            
            # Cognitive component
            cognitive = cognitive_weight * r1 * (particle_best[i] - particles[i])
            
            # Social component
            social = social_weight * r2 * (global_best - particles[i])
            
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible set
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate fitness
            fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update particle best
            if fitness < particle_best_fitness[i]:
                particle_best[i] = particles[i].copy()
                particle_best_fitness[i] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = fitness
                    
        iter_count += 1
    
    # Final refinement with local search
    h_final = global_best.copy()
    dx_final = 2.0 / n_points
    
    # Local descent search
    step_size = 0.005
    max_local_iter = 5000
    local_iter = 0
    
    while time() - start_time < budget_s * 0.98 and local_iter < max_local_iter:
        corr = np.correlate(h_final, 1 - h_final, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_final) - 1)
        
        # Compute gradient for overlapping integral at the max lag
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_final[i + max_lag]
                gradient[i + max_lag] -= h_final[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_final[i - max_lag]
                gradient[i - max_lag] -= h_final[i]
        
        # Gradient descent step
        h_next = h_final - step_size * gradient
        h_next = project_to_feasible(h_next, n_points)
        
        # Accept if improvement
        c5_new = evaluate_erdos_solution(h_next, dx_final)
        if c5_new < evaluate_erdos_solution(h_final, dx_final):
            h_final = h_next
            step_size *= 1.05
        else:
            step_size *= 0.5  # Reduce step size
            if step_size < 1e-10:
                break
                
        local_iter += 1
    
    # Final evaluation
    final_value = evaluate_erdos_solution(h_final, dx_final)
    
    return (list(h_final), final_value, n_points)
