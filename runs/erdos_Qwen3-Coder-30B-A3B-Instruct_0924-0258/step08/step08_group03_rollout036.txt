import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_coarse = 15
    n_fine = 35
    coarse_iterations = 500
    fine_iterations = 2000
    mutation_rate = 0.1
    
    # Initialize coarse grid
    h_coarse = np.ones(n_coarse) * 0.5
    
    # If initial construction is available, use it
    if 'initial_h_values' in globals():
        h_fine = np.array(initial_h_values)
    else:
        h_fine = np.ones(n_fine) * 0.5
    
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    best_c5 = float('inf')
    best_h_fine = h_fine.copy()
    
    # Coarse grid optimization
    for iteration in range(coarse_iterations):
        if time() - start_time > budget_s * 0.8:
            break
        
        # Compute overlaps for coarse grid
        corr_coarse = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_lag_idx = np.argmax(corr_coarse)
        max_lag = max_lag_idx - (n_coarse - 1)
        
        # Compute gradient for the max lag
        grad = np.zeros(n_coarse)
        for i in range(n_coarse):
            j = i + max_lag
            if 0 <= j < n_coarse:
                grad[i] = 1 - 2 * h_coarse[i] - h_coarse[j]
            else:
                grad[i] = 1 - 2 * h_coarse[i]
        
        # Gradient descent step
        h_new = h_coarse - 0.01 * grad
        h_new = project_to_feasible(h_new, n_coarse)
        
        # Update coarse solution
        h_coarse = h_new
        
    # Upsample to fine grid using linear interpolation
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    f_interp = interp1d(x_coarse, h_coarse, kind='linear', fill_value='extrapolate')
    h_fine = f_interp(x_fine)
    
    # Ensure length match
    if len(h_fine) != n_fine:
        h_fine = np.resize(h_fine, n_fine)
        
    # Renormalize fine grid
    h_fine = project_to_feasible(h_fine, n_fine)
    
    # Fine grid optimization alternating between gradient descent and mutation
    for iteration in range(fine_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for fine grid
        corr_fine = np.correlate(h_fine, 1 - h_fine, mode='full')
        max_lag_idx = np.argmax(corr_fine)
        max_lag = max_lag_idx - (n_fine - 1)
        
        # Compute gradient for the max lag
        grad = np.zeros(n_fine)
        for i in range(n_fine):
            j = i + max_lag
            if 0 <= j < n_fine:
                grad[i] = 1 - 2 * h_fine[i] - h_fine[j]
            else:
                grad[i] = 1 - 2 * h_fine[i]
        
        # Mutate some fraction randomly
        if iteration % 2 == 0:
            # Gradient descent step
            h_new = h_fine - 0.01 * grad
        else:
            # Random mutation
            h_new = h_fine.copy()
            indices = np.random.choice(n_fine, size=int(n_fine * mutation_rate), replace=False)
            for idx in indices:
                h_new[idx] = np.random.uniform(0, 1)
                
        h_new = project_to_feasible(h_new, n_fine)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx_fine)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h_fine = h_new.copy()
            h_fine = h_new.copy()
    
    return (list(best_h_fine), best_c5, n_fine)
