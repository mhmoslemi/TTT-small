import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a structured starting point
        n_points = 35
        h_start = np.zeros(n_points)
        # Place 0s and 1s in alternating pattern to reduce overlap
        for i in range(n_points):
            if i % 2 == 0:
                h_start[i] = 0.0
            else:
                h_start[i] = 1.0
        # Normalize
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Stage 1: Convex relaxation
    # For large problems, skip convex relaxation and proceed with direct optimization
    
    # Stage 2: Direct gradient-based local search with improved strategy
    max_iter = 5000
    step_size = 0.01
    no_improvement_count = 0
    no_improvement_limit = 200
    tolerance = 1e-9
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        lag_indices = np.arange(len(corr))
        lags = lag_indices - (len(h_current) - 1)
        overlap_values = corr * dx
        
        # Find the (lag, max_overlap) with maximum overlap
        max_idx = np.argmax(overlap_values)
        max_lag = lags[max_idx]
        max_overlap = overlap_values[max_idx]
        
        # Compute gradient for the specific lag achieving maximum overlap
        gradient = np.zeros_like(h_current)
        for i in range(len(h_current)):
            if max_lag >= 0:
                if i + max_lag < len(h_current):
                    gradient[i] = -1.0  # Contribution of h[i] to the overlap term h[i](1-h[i+lag])
            else:
                if i - max_lag >= 0:
                    gradient[i] = -1.0
        
        # Project onto constraint set: sum(h) = n_points / 2
        sum_h = np.sum(h_current)
        projection = (n_points / 2 - sum_h) / len(h_current)
        
        # Adaptive step size with backtracking
        h_new = h_current - step_size * gradient + step_size * projection
        
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check convergence
        change = np.sum(np.abs(h_new - h_current))
        if change < tolerance:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            
        if no_improvement_count > no_improvement_limit:
            # Reduce step size on stagnation
            step_size *= 0.5
            no_improvement_count = 0
            
        h_current = h_new
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            no_improvement_count = 0  # Reset when improvement is found
    
    # Refinement with SLSQP using the best solution as initial guess
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
