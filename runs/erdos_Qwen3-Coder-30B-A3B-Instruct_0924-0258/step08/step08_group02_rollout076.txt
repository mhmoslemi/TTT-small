import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter_sa = 100000
    epsilon = 0.1
    
    # Perturb two random indices while maintaining sum constraint
    def perturb(h, epsilon):
        h_new = h.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-epsilon, epsilon)
        h_new[i] += delta
        h_new[j] -= delta
        # Ensure bounds and sum constraint
        h_new = np.clip(h_new, 0, 1)
        # Rebalance to maintain sum
        total = np.sum(h_new)
        h_new = h_new * (n_points/2) / total
        return h_new
    
    # Simulated Annealing loop
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    T = T_init
    iter_count = 0
    
    while T > T_min and (time() - start_time) < budget_s * 0.95:
        if iter_count >= max_iter_sa:
            break
            
        # Make a perturbation
        h_new = perturb(h, epsilon)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Accept with probability
            delta = c5_new - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_new
        
        # Update T and iteration counter
        T *= alpha
        iter_count += 1
    
    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return np.max(np.correlate(h, 1 - h, mode='full') * dx)
        
        # Normalize the best solution before refinement
        h_normalized = best_h * (n_points/2) / np.sum(best_h)
        res = minimize(
            objective,
            h_normalized,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
