import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible set: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = project_to_feasible(h_start.copy(), n_points)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Annealing parameters
    T0 = 0.05
    alpha = 0.999
    beta = 0.999
    T_min = 1e-6
    
    T = T0
    sigma = 0.05 * dx
    
    # Main simulated annealing loop
    while time() - start_time < budget_s * 0.95 and T > T_min:
        # Generate neighbor by perturbing h
        delta = np.random.normal(0, sigma, n_points)
        h_new = h_current + delta
        
        # Project to feasible set
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Metropolis criterion
        delta_c5 = c5_new - best_c5
        if delta_c5 < 0 or np.random.random() < np.exp(-delta_c5 / T):
            h_current = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        
        # Update temperature and perturbation size
        T *= alpha
        sigma *= beta
    
    return (list(best_h), best_c5, n_points)
