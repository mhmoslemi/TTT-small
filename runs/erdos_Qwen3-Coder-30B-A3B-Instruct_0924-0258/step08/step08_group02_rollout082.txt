import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_sa_iter = 100000
    epsilon = 0.1
    
    # Initialize SA
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    T = T_init
    sa_iter = 0
    
    while T > T_min and sa_iter < max_sa_iter:
        if time() - start_time > budget_s * 0.9:
            break
            
        # Perturb two random indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-epsilon, epsilon)
        
        # Calculate new values
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        sum_diff = np.sum(h_new) - n_points/2
        h_new[0] -= sum_diff
        
        # Ensure bounds again
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5:
            h_current = h_new.copy()
            best_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
        else:
            # Accept with probability
            prob = np.exp(-(c5_new - best_c5) / T)
            if np.random.rand() < prob:
                h_current = h_new.copy()
        
        T *= alpha
        sa_iter += 1
    
    # Local refinement with L-BFGS
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
