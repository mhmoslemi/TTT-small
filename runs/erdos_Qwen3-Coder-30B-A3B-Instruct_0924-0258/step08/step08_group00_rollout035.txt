import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Start with a step function with 1s in the first half, 0s in the second
        n_points = 35
        h_start = np.ones(n_points)
        h_start[n_points//2:] = 0
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a hybrid approach: simulated annealing with local refinement
    current_h = h_start.copy()
    current_c5 = best_c5
    temperature = 0.1
    max_iter = 10000
    patience = 100
    no_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb current solution
        h_new = current_h.copy()
        
        # Randomly choose a point to modify
        i = np.random.randint(0, n_points)
        # Small random perturbation within bounds
        delta = np.random.normal(0, 0.05 * temperature)
        h_new[i] += delta
        
        # Ensure h_new stays within [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy the integral constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < current_c5:
            current_h = h_new
            current_c5 = c5_new
            no_improvement = 0
        else:
            # Accept with probability based on temperature
            delta_c5 = c5_new - current_c5
            if np.random.rand() < np.exp(-delta_c5 / temperature):
                current_h = h_new
                current_c5 = c5_new
                no_improvement = 0
            else:
                no_improvement += 1
                
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

        # Cooling schedule
        if iteration % 50 == 0:
            temperature *= 0.99
            
        if no_improvement >= patience:
            break
    
    # Final local optimization for the best solution found
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use the best solution as starting point for SLSQP
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
