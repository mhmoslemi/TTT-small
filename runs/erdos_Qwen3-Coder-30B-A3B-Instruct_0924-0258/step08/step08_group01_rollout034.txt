import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm
    population_size = 50
    max_generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    no_improvement_limit = 100
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # Normalize sum
        h = np.clip(h, 0, 1)  # Clip to [0,1]
        return h
    
    def create_feasible_individual():
        # Make sure sum = n_points/2 and all values in [0,1]
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        # Adjust to exactly match sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Create initial population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        population.append(create_feasible_individual())
    
    # Genetic operators
    def blend_crossover(parent1, parent2):
        alpha = np.random.rand()
        child = alpha * parent1 + (1 - alpha) * parent2
        child = np.clip(child, 0, 1)
        return child * (n_points / 2) / np.sum(child)
    
    def mutate(individual, generation):
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                delta = np.random.normal(0, 0.05 * (1 - generation / max_generations))
                mutated[i] += delta
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    # Evaluation function
    def evaluate_individual(individual):
        return evaluate_erdos_solution(individual, dx)
    
    # Main GA loop
    no_improvement_count = 0
    generation = 0
    best_fitness = best_c5
    
    while time() - start_time < budget_s * 0.8 and generation < max_generations:
        # Evaluate population
        fitnesses = [evaluate_individual(ind) for ind in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > no_improvement_limit:
            break
            
        # Selection with tournament
        def tournament_select():
            k = 3
            tournament = random.sample(list(zip(population, fitnesses)), k)
            return min(tournament, key=lambda x: x[1])[0]
            
        # Create new population
        new_population = [best_h.copy()]
        for _ in range(population_size - 1):
            parent1 = tournament_select()
            parent2 = tournament_select()
            
            if np.random.rand() < crossover_rate:
                child = blend_crossover(parent1, parent2)
            else:
                child = parent1.copy()
                
            child = mutate(child, generation)
            new_population.append(child)
            
        population = new_population
        generation += 1
    
    # Final refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
