import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a better construction
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1.0
        # Shuffle to avoid structured initializations
        np.random.shuffle(h_start)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Threshold-based optimization
    threshold = 0.1
    cooling_rate = 0.95
    max_iterations = 500
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient approximation for each point
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - best_h[j] - best_h[i])
        
        # Thresholding heuristic: set points with large gradient to 0 or 1
        # Find top points by absolute gradient values
        idxs = np.argsort(np.abs(gradient))[::-1][:n_points//2] 
        new_h = best_h.copy()
        
        # Apply thresholding
        for i in idxs:
            if new_h[i] >= 1 - threshold:
                new_h[i] = 1.0
            elif new_h[i] <= threshold:
                new_h[i] = 0.0
        
        # Normalize to satisfy sum constraint
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Check if it's still feasible (ensure all values in [0,1])
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate new solution
        current_c5 = evaluate_erdos_solution(new_h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_h.copy()
            # Adaptive threshold: increase learning rate
            threshold = min(0.1, threshold * 1.1)
        else:
            # Cooling down the threshold to refine
            threshold = max(0.01, threshold * cooling_rate)
            
        # Occasionally, do local search using golden section
        if iter % 10 == 0 and time() - start_time < budget_s * 0.9:
            try:
                # Perform local optimization on lucky points
                # Create a small neighborhood to optimize
                local_h = best_h.copy()
                idx_opt = np.where(np.abs(gradient) > np.percentile(np.abs(gradient), 70))[0]
                
                if len(idx_opt) > 0:
                    # Randomly select one index to optimize
                    idx = np.random.choice(idx_opt)
                    # Optimize this single element using golden section search
                    def objective(x):
                        local_h[idx] = x
                        return np.max(np.correlate(local_h, 1 - local_h, mode='full') * dx)
                    
                    # Golden section search on [0,1]
                    def golden_section_search(f, a, b, tol=1e-6):
                        phi = (1 + 5**0.5) / 2
                        resphi = 2 - phi
                        
                        # Make sure a < b
                        if a > b:
                            a, b = b, a
                        
                        # Initial points
                        x1 = a + resphi * (b - a)
                        x2 = b - resphi * (b - a)
                        
                        f1 = f(x1)
                        f2 = f(x2)
                        
                        while abs(b - a) > tol:
                            if f1 < f2:
                                b = x2
                                x2 = x1
                                f2 = f1
                                x1 = a + resphi * (b - a)
                                f1 = f(x1)
                            else:
                                a = x1
                                x1 = x2
                                f1 = f2
                                x2 = b - resphi * (b - a)
                                f2 = f(x2)
                        
                        return (a + b) / 2
                    
                    # Update that coordinate
                    opt_val = golden_section_search(objective, 0, 1)
                    local_h[idx] = opt_val
                    
                    # Check improved validity
                    local_h = local_h * (n_points/2) / np.sum(local_h)
                    local_h = np.clip(local_h, 0, 1)
                    local_h = local_h * (n_points/2) / np.sum(local_h)
                    local_c5 = evaluate_erdos_solution(local_h, dx)
                    if local_c5 < best_c5:
                        best_c5 = local_c5
                        best_h = local_h.copy()
            except:
                continue
    
    # Final refinement using L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
