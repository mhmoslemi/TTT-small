import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def solve_convex_perturbation(h, n_points, dx, epsilon):
    """Solve convex perturbed optimization using CVXPY"""
    # Define variables
    h_var = cp.Variable(n_points)
    
    # Objective: minimize sum of overlaps plus convex perturbation
    # For efficiency, we'll approximate the sum of top 100 overlaps
    
    # Compute correlations manually for 100 largest shifts
    corr = np.correlate(h, 1 - h, mode='full')
    top_k = np.argsort(corr)[-100:]  # 100 most significant correlations
    
    obj = 0
    for k in top_k:
        # Get shifted indices for correlation at shift k
        shift = k - (n_points - 1)
        if shift >= 0:
            # negative shift: (1-h shifted left by k)
            obj += cp.sum(cp.multiply(h_var[:n_points-shift], (1 - h_var[shift:])))
        else:
            # positive shift: (1-h shifted right by k)
            obj += cp.sum(cp.multiply(h_var[-shift:], (1 - h_var[:-shift])))
    
    # Add convex regularization
    obj += epsilon * cp.sum_squares(h_var)
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    
    # Solve
    prob = cp.Problem(cp.Minimize(obj), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
        return h_var.value
    except:
        return h.copy()  # Return original if solver fails
    
def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    else:

        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex perturbation parameters
    epsilon_0 = 0.001
    beta = 0.99
    max_iter = 5000
    epsilon = epsilon_0
    
    # Main optimization loop
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Solve convex perturbed optimization
        h_new = solve_convex_perturbation(h, n_points, dx, epsilon)
        
        # Project to feasible region
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Update h with new solution
        h = h_new.copy()
        
        # Update epsilon
        epsilon *= beta
        
        # Early stopping based on stagnation
        if it > 0 and it % 100 == 0 and abs(c5_new - best_c5) < 1e-8:
            break

    # Final local refinement using gradient-free search
    best_h = project_to_feasible(best_h, n_points)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Add small random perturbations to explore neighborhood
    num_refine = 100
    for _ in range(num_refine):
        if time() - start_time > budget_s * 0.98:
            break
        h_pert = best_h + np.random.normal(0, 0.001, n_points)
        h_pert = project_to_feasible(h_pert, n_points)
        c5_pert = evaluate_erdos_solution(h_pert, dx)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
