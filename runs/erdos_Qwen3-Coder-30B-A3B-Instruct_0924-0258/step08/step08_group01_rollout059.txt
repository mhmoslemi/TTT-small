import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 2
    stagnation_limit = 100
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        for _ in range(population_size - 1):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Evaluation
    fitness = []
    for h in population:
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Main GA loop
    no_improvement_count = 0
    gen = 0
    
    while time() - start_time < budget_s * 0.95 and no_improvement_count < stagnation_limit:
        # Select parents (tournament selection with size 3)
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, 3)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Crossover (blend crossover)
        new_population = []
        for i in range(0, population_size, 2):
            if np.random.rand() < crossover_rate and i+1 < population_size:
                parent1 = selected[i]
                parent2 = selected[i+1]
                
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
                
                # Ensure constraints
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points / 2) / np.sum(child1)
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
                
                new_population.extend([child1, child2])
            else:
                new_population.extend([selected[i], selected[i+1]])
        
        # Mutation
        for i in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                h = new_population[i]
                mut_strength = 0.05 * (0.95 ** gen)
                for j in range(n_points):
                    if np.random.rand() < 0.1:  # 10% of genes mutated
                        delta = np.random.normal(0, mut_strength)
                        h[j] += delta
                # Re-normalize to sum constraint
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                new_population[i] = h
        
        # Replace population with elite + new population
        elite_idx = np.argsort(fitness)[:elite_size]
        for i in range(elite_size):
            new_population[i] = population[elite_idx[i]]
            
        population = new_population
        
        # Evaluate new population
        fitness = []
        for h in population:
            fitness.append(evaluate_erdos_solution(h, dx))
        
        # Find best in current generation
        min_fitness = min(fitness)
        best_idx = fitness.index(min_fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        gen += 1
        
        # Occasionally refine the best solution with SLSQP
        if gen % 50 == 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 1000, 'ftol': 1e-12})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
