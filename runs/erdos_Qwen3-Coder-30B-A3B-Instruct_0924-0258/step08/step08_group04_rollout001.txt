import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Multiresolution approach: coarse grid first
    coarse_n = 15
    coarse_dx = 2.0 / coarse_n
    coarse_h = np.linspace(0, 1, coarse_n)
    coarse_h = coarse_h * (coarse_n/2) / np.sum(coarse_h)
    
    # Genetic algorithm on coarse grid
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.5
    generations = 300
    pop = [np.random.rand(coarse_n) for _ in range(population_size)]
    for i in range(population_size):
        pop[i] = pop[i] * (coarse_n/2) / np.sum(pop[i])
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate fitness
        fitness = []
        for individual in pop:
            # Upscale to fine grid
            fine_individual = np.interp(np.linspace(0, 2, n_points), 
                                      np.linspace(0, 2, coarse_n), 
                                      individual)
            fine_individual = fine_individual * (n_points/2) / np.sum(fine_individual)
            c5 = evaluate_erdos_solution(fine_individual, dx)
            fitness.append(-c5)
        
        # Selection
        sorted_indices = np.argsort(fitness)[::-1]
        elite = [pop[i] for i in sorted_indices[:population_size//2]]
        
        # Crossover and mutation
        new_pop = elite.copy()
        while len(new_pop) < population_size:
            if np.random.rand() < crossover_rate and len(elite) >= 2:
                parent1, parent2 = np.random.choice(elite, size=2, replace=False)
                crossover_point = np.random.randint(1, coarse_n-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                parent = np.random.choice(elite)
                child = parent.copy()
            
            # Mutation
            for i in range(coarse_n):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.normal(0, 0.05)
                    child[i] = np.clip(child[i], 0, 1)
            
            # Normalize
            child = child * (coarse_n/2) / np.sum(child)
            new_pop.append(child)
        
        pop = new_pop[:population_size]
    
    # Find best coarse solution and upscale
    coarse_fitness = []
    for individual in pop:
        fine_individual = np.interp(np.linspace(0, 2, n_points), 
                                  np.linspace(0, 2, coarse_n), 
                                  individual)
        fine_individual = fine_individual * (n_points/2) / np.sum(fine_individual)
        c5 = evaluate_erdos_solution(fine_individual, dx)
        coarse_fitness.append(c5)
    
    best_coarse_idx = np.argmin(coarse_fitness)
    best_coarse = pop[best_coarse_idx]
    best_fine = np.interp(np.linspace(0, 2, n_points), 
                         np.linspace(0, 2, coarse_n), 
                         best_coarse)
    best_fine = best_fine * (n_points/2) / np.sum(best_fine)
    
    # Gradient-based refinement
    h_current = best_fine.copy()
    step_size = 0.01
    iter_count = 0
    max_iter = 10000
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_current[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= h_current[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_current[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_current[i]
        
        h_new = h_current - step_size * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        if c5_new >= evaluate_erdos_solution(h_current, dx):
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            h_current = h_new
            step_size *= 1.05
        
        iter_count += 1
    
    return (list(best_h), best_c5, n_points)
