import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points / 2 and 0 <= h[i] <= 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-10:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_start = project_to_feasible(h_start, n_points)
    
    # PSO parameters
    num_particles = 75
    inertia_start = 0.8
    inertia_end = 0.4
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    particle_best = []
    particle_best_values = []
    
    for _ in range(num_particles):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        particle_best.append(h.copy())
        particle_best_values.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    global_best_idx = np.argmin(particle_best_values)
    global_best = particle_best[global_best_idx].copy()
    global_best_value = particle_best_values[global_best_idx]
    
    max_iter = int(1e6)
    time_budget = budget_s * 0.95
    iter_count = 0
    
    while time() - start_time < time_budget and iter_count < max_iter:
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_count / max_iter)
        
        for p in range(num_particles):
            # Velocity update
            r1, r2 = np.random.rand(2)
            velocities[p] = (inertia * velocities[p] + 
                             cognitive_weight * r1 * (particle_best[p] - particles[p]) +
                             social_weight * r2 * (global_best - particles[p]))
            
            # Clip velocities
            velocities[p] = np.clip(velocities[p], -max_velocity, max_velocity)
            
            # Update positions
            particles[p] += velocities[p]
            
            # Projection to feasible set
            particles[p] = project_to_feasible(particles[p], n_points)
            
            # Evaluate fitness
            current_value = evaluate_erdos_solution(particles[p], dx)
            
            # Update particle best
            if current_value < particle_best_values[p]:
                particle_best[p] = particles[p].copy()
                particle_best_values[p] = current_value
                
                # Update global best
                if current_value < global_best_value:
                    global_best = particles[p].copy()
                    global_best_value = current_value
        
        iter_count += 1
    
    # Final refinement with local optimization
    h_final = global_best.copy()
    current_c5 = global_best_value
    step_size = 0.01
    
    for _ in range(1000):
        if time() - start_time > time_budget:
            break
            
        # Compute all correlations 
        corr = np.correlate(h_final, 1 - h_final, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_final) - 1)
        
        # Compute gradient for max overlap lag
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_final[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= h_final[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_final[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_final[i]
        
        # Gradient descent step
        h_new = h_final - step_size * gradient
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < current_c5:
            h_final = h_new.copy()
            current_c5 = c5_new
            step_size *= 1.05
        else:
            step_size *= 0.5
            if step_size < 1e-10:
                break
    
    return (list(h_final), current_c5, n_points)
