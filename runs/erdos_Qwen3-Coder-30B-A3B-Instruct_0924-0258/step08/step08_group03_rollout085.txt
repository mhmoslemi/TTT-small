import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_fine)
    else:
        n_points = 35
        h_fine = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_fine.copy()
    best_c5 = evaluate_erdos_solution(h_fine, dx)
    
    # Coarse grid optimization (15 points)
    n_coarse = 15
    h_coarse = np.ones(n_coarse) * 0.5
    coarse_dx = 2.0 / n_coarse
    fine_to_coarse_indices = np.linspace(0, n_points - 1, n_coarse, dtype=int)
    
    # Initial coarse optimization
    for it in range(500):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Interpolate coarse to fine for evaluation
        f_interp = interp1d(np.linspace(0, 2, n_coarse), h_coarse, kind='linear', fill_value='extrapolate')
        h_temp = f_interp(np.linspace(0, 2, n_points))
        h_temp = project_to_feasible(h_temp, n_points)
        
        # Compute overlaps
        corr = np.correlate(h_temp, 1 - h_temp, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h_temp[i] - h_temp[j]
            else:
                grad[i] = 1 - 2 * h_temp[i]
        
        # Gradient descent
        h_new = h_temp - 0.01 * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Update coarse grid
        h_new_coarse = h_new[fine_to_coarse_indices]
        h_new_coarse = project_to_feasible(h_new_coarse, n_coarse)
        h_coarse = h_new_coarse
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
    
    # Upsample coarse to fine
    f_interp = interp1d(np.linspace(0, 2, n_coarse), h_coarse, kind='linear', fill_value='extrapolate')
    h_fine = f_interp(np.linspace(0, 2, n_points))
    h_fine = project_to_feasible(h_fine, n_points)
    
    # Fine grid optimization with alternating steps
    step_size = 0.01
    for iteration in range(2000):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Compute overlaps
        corr = np.correlate(h_fine, 1 - h_fine, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Gradient descent step
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h_fine[i] - h_fine[j]
            else:
                grad[i] = 1 - 2 * h_fine[i]
        
        h_new = h_fine - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            h_fine = h_new.copy()

            mutation_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            h_mut = h_fine.copy()

            h_mut = project_to_feasible(h_mut, n_points)
            
            new_c5_mut = evaluate_erdos_solution(h_mut, dx)
            if new_c5_mut < best_c5:
                best_c5 = new_c5_mut
                best_h = h_mut.copy()
                h_fine = h_mut.copy()
                
            step_size *= 0.995
    
    return (list(best_h), best_c5, n_points)
