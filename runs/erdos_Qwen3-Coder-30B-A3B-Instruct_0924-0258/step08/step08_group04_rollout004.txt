import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Coarse optimization on smaller grid
    coarse_n = 15
    fine_n = 35
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.5
    max_gen = 1000
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        # Rescale to correct integral
        h0 = h0 * (coarse_n/2) / np.sum(h0)
        population.append(h0.copy())
    else:
        h0 = np.ones(coarse_n) * 0.5
        h0 = h0 * (coarse_n/2) / np.sum(h0)
        population.append(h0.copy())
    
    # Generate initial population
    for _ in range(population_size - 1):
        h = np.random.rand(coarse_n)
        h = h * (coarse_n/2) / np.sum(h)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0/coarse_n)
    
    gen = 0
    no_improvement = 0
    max_no_improvement = 1000
    
    # Genetic algorithm on coarse grid
    while time() - start_time < budget_s * 0.5 and gen < max_gen:
        # Evaluate fitness (negative C5)
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, 2.0/coarse_n)
            fitness.append(-c5)
        
        # Find best solution
        best_idx = np.argmax(fitness)
        if fitness[best_idx] < -best_c5:
            best_c5 = -fitness[best_idx]
            best_h = population[best_idx].copy()
        
        # Selection (tournament)
        new_population = []
        for _ in range(population_size):
            tournament = random.sample(list(zip(population, fitness)), 3)
            winner = max(tournament, key=lambda x: x[1])
            new_population.append(winner[0].copy())
        
        # Crossover and mutation
        for i in range(0, population_size, 2):
            if random.random() < crossover_rate and i+1 < population_size:
                # Blend crossover
                parent1, parent2 = new_population[i], new_population[i+1]
                alpha = random.random()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
                
                # Repair
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (coarse_n/2) / np.sum(child1)
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (coarse_n/2) / np.sum(child2)
                
                new_population[i] = child1
                new_population[i+1] = child2
            
            # Mutation
            for j in range(i, min(i+2, population_size)):
                if random.random() < mutation_rate:
                    h = new_population[j]
                    idx = random.randint(0, coarse_n-1)
                    h[idx] += random.gauss(0, 0.02)
                    # Repair
                    h = np.clip(h, 0, 1)
                    h = h * (coarse_n/2) / np.sum(h)
                    new_population[j] = h
        
        population = new_population
        gen += 1
        
        # Early exit if no improvement
        if gen > 100 and no_improvement > max_no_improvement:
            break
    
    # Upscale to fine grid
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = h_start * (fine_n/2) / np.sum(h_start)
        h_current = h_start.copy()
    else:
        # Interpolate from best coarse solution
        coarse_h = best_h
        coarse_x = np.linspace(0, 2, coarse_n)
        fine_x = np.linspace(0, 2, fine_n)
        h_current = np.interp(fine_x, coarse_x, coarse_h)
        h_current = h_current * (fine_n/2) / np.sum(h_current)
    
    # Gradient optimization on fine grid
    dx = 2.0 / fine_n
    step_size = 0.01
    max_iter = 10000
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        gradient = np.zeros(fine_n)
        if max_lag >= 0:
            for i in range(fine_n - max_lag):
                gradient[i] += 1 - h_current[i + max_lag]
                if i + max_lag < fine_n:
                    gradient[i + max_lag] -= h_current[i]
        else:
            for i in range(fine_n + max_lag):
                gradient[i] += 1 - h_current[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_current[i]
        
        h_new = h_current - step_size * gradient
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (fine_n/2) / np.sum(h_new)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # If not improving, reduce step size
        if c5_new >= evaluate_erdos_solution(h_current, dx):
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            h_current = h_new
            step_size *= 1.05  # Increase step size slightly
        
        iter_count += 1
    
    return (list(best_h), best_c5, fine_n)
