def project_to_feasible(Y, n):
    # Truncated SVD to find closest feasible h
    U, s, Vt = svd(Y)
    s = np.maximum(s, 0)  # Ensure non-negative
    Y_proj = U @ np.diag(s) @ Vt
    h = np.sqrt(np.diag(Y_proj))
    h = np.clip(h, 0, 1)
    # Renormalize
    h = h * (n / 2) / np.sum(h)
    return h
