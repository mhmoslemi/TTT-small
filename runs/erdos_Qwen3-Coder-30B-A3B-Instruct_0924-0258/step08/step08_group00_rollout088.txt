import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Hybrid approach: iterative local search with random restarts
    max_iter = 2000
    patience = 50
    no_improvement = 0
    step_size = 0.01
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate overlap for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            # Random restart
            h = np.random.rand(n_points) * 0.5
            h = h * (n_points/2) / np.sum(h)
            no_improvement = 0
            step_size = 0.01
            continue
            
        # Identify lag with maximum overlap
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                # We're using a simplified gradient estimation
                # For more precise results we'd compute the full derivative
                # but for this optimization, we'll proceed with a numeric approach
                pass
        
        # Compute numerical gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            val_plus = evaluate_erdos_solution(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            val_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (val_plus - val_minus) / (2 * eps)
        
        # Update with gradient descent
        h_new = h - step_size * grad
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Adaptive step size based on improvement
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < max_overlap:
            h = h_new
            step_size *= 1.1
        else:
            step_size *= 0.5
            if step_size < 1e-8:
                step_size = 1e-8
                
    # Final refinement with trust-constr
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='trust-constr', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'gtol': 1e-9, 'xtol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
