import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Use initial construction if available

        h_start[:n_points//2] = 1.0
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Thresholding-based optimization with adaptive cooling
    threshold = 0.1
    cooling_factor = 0.95
    max_iterations = 500
    refinement_iterations = 100
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradients
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - best_h[j] - best_h[i])
        
        # Rank coordinates by gradient magnitude
        idx_sorted = np.argsort(np.abs(gradient))[::-1]
        top_indices = idx_sorted[:max(1, n_points // 2)]
        
        # Apply thresholding
        h_new = best_h.copy()
        for i in top_indices:
            if h_new[i] >= 1 - threshold:
                h_new[i] = 1.0
            elif h_new[i] <= threshold:
                h_new[i] = 0.0
        
        # Maintain sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Update if improved
        current_c5 = evaluate_erdos_solution(h_new, dx)
        if current_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = current_c5
        
        # Cool threshold
        threshold = max(0.01, threshold * cooling_factor)

    overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
    l_max = np.argmax(overlaps)
    gradient = np.zeros(n_points)
    for i in range(n_points):
        j = (i + l_max) % n_points
        gradient[i] = -2 * dx * (1 - best_h[j] - best_h[i])
    
    idx_sorted = np.argsort(np.abs(gradient))[::-1]
    top_indices = idx_sorted[:max(1, n_points // 5)]
    
    for i in top_indices:
        if time() - start_time > budget_s * 0.95:
            break
        # Golden section search on h[i] fixing everything else
        orig_val = best_h[i]
        
        def obj(x):
            best_h[i] = x

            overlap = evaluate_erdos_solution(best_h, dx)

        x1 = resphi * a + (1 - resphi) * b

        f1 = obj(x1)

                break

                x1 = (1 - resphi) * a + resphi * b

        best_h[i] = new_val

        current_c5 = evaluate_erdos_solution(best_h, dx)
        if current_c5 < best_c5:

    try:
        bounds = [(0, 1) for _ in range(n_points)]

            return evaluate_erdos_solution(h, dx)
        
        # Normalize to satisfy sum constraint
        best_h = best_h * (n_points/2) / np.sum(best_h)
        
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 50, 'ftol': 1e-9})
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
