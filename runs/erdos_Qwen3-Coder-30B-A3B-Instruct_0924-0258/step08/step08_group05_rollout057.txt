import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.5
        h_start[1] = 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.995
    min_T = 1e-6
    max_iter = 100000
    no_improvement_limit = 5000
    no_improvement_count = 0
    
    # Initial energy
    E_current = evaluate_erdos_solution(h_current, dx)
    
    # Main SA loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor by swapping delta between two positions
        i, j = np.random.randint(0, n_points, 2)
        while i == j:
            j = np.random.randint(0, n_points)
            
        # Compute feasible delta
        delta_max = min(h_current[i], 1 - h_current[j])
        if delta_max <= 0:
            continue
            
        delta = np.random.uniform(0, delta_max)
        
        # Create new solution
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        E_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if E_new < E_current or np.random.rand() < np.exp(-(E_new - E_current) / T):
            h_current = h_new
            E_current = E_new
            
            if E_current < best_c5:
                best_c5 = E_current
                best_h = h_current.copy()
                no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Annealing
        T *= cooling_rate
        if T < min_T:
            T = min_T
            
        # Early termination check
        if no_improvement_count > no_improvement_limit:
            break
    
    # Final refinement with local search
    for _ in range(1000):
        if time() - start_time > budget_s * 0.99:
            break
            
        # Small perturbation
        i = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.05, 0.05)
        h_trial = best_h.copy()
        h_trial[i] += delta
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        
        if evaluate_erdos_solution(h_trial, dx) < best_c5:
            best_h = h_trial.copy()
            best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
