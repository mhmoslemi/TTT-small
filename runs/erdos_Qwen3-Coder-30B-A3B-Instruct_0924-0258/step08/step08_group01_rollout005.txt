import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    crossover_rate = 0.8
    mutation_rate = 0.1
    stagnation_limit = 100
    refinement_interval = 100
    
    # Initialize population
    population = []
    fitness = []
    
    # Add initial solution
    population.append(h_start.copy())
    fit = evaluate_erdos_solution(h_start, dx)
    fitness.append(fit)
    
    # Add random solutions
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
        fit = evaluate_erdos_solution(h, dx)
        fitness.append(fit)
    
    best_in_gen = float('inf')
    stagnation_count = 0
    
    eval_count = population_size
    
    while time() - start_time < budget_s * 0.9:
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count >= stagnation_limit:
            break
        
        # Tournament selection
        def tournament_selection(k=3):
            idxs = np.random.choice(population_size, k)
            best_idx = idxs[np.argmin([fitness[i] for i in idxs])]
            return population[best_idx]
        
        new_population = [population[0].copy()]  # Elitism
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection()
            parent2 = tournament_selection()
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mu = 0.05 * np.exp(-0.01 * eval_count)
                for i in range(n_points):
                    if np.random.rand() < 0.1:  # 10% chance to mutate each element
                        child[i] += np.random.normal(0, mu)
                
                # Ensure feasibility
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
            eval_count += 1
            
            # Periodic SLSQP refinement
            if eval_count % refinement_interval == 0 and len(new_population) > 1:
                try:
                    def obj(h):
                        return evaluate_erdos_solution(h, dx)
                    
                    def cons_sum(h):
                        return np.sum(h) - n_points / 2.0
                    
                    res = minimize(obj, new_population[-1], method='SLSQP',
                                   bounds=[(0, 1) for _ in range(n_points)],
                                   constraints={'type': 'eq', 'fun': cons_sum},
                                   options={'maxiter': 50, 'ftol': 1e-8})
                    if res.success:
                        refined = res.x
                        new_population[-1] = refined
                except:
                    pass
        
        # Update population and fitness
        population = new_population[:population_size]
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Final local refinement with SLSQP on best solution
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
