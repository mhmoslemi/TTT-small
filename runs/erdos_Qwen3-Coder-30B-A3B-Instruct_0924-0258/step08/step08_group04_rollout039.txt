import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # PSO parameters
    num_particles = 75
    inertia = 0.8
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    perturbation_rate = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    particle_best = []
    particle_best_fitness = []
    
    # Create initial population
    for _ in range(num_particles):
        h_particle = np.random.rand(n_points) * 0.5 + 0.25  # Random feasible h in [0.25, 0.75]
        h_particle = h_particle * (n_points/2) / np.sum(h_particle)  # Renormalize
        h_particle = np.clip(h_particle, 0, 1)
        particles.append(h_particle.copy())
        velocities.append(np.zeros(n_points))
        particle_best.append(h_particle.copy())
        particle_best_fitness.append(evaluate_erdos_solution(h_particle, dx))
    
    # Global best initialization
    global_best_idx = np.argmin(particle_best_fitness)
    global_best = particle_best[global_best_idx].copy()
    global_best_fitness = particle_best_fitness[global_best_idx]
    
    max_iter = int(5e5)  # Sufficient for budget
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Update inertia
        current_inertia = inertia * (0.4 / inertia) ** (iter_count / max_iter)
        
        # Update each particle
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_update = (current_inertia * velocities[i] +
                         cognitive_weight * r1 * (particle_best[i] - particles[i]) +
                         social_weight * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(vel_update, -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible set
            particles[i] = np.clip(particles[i], 0, 1)
            # Ensure sum matches constraint
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate fitness
            fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if fitness < particle_best_fitness[i]:
                particle_best[i] = particles[i].copy()
                particle_best_fitness[i] = fitness
            
            # Update global best
            if fitness < global_best_fitness:
                global_best = particles[i].copy()
                global_best_fitness = fitness
        
        # Periodically perturb particles to escape local minima
        if iter_count % 1000 == 0 and iter_count > 0:
            for i in range(num_particles):
                if np.random.rand() < perturbation_rate:
                    # Add small random perturbation
                    perturbation = np.random.normal(0, 0.01, n_points)
                    particles[i] += perturbation
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
        
        iter_count += 1
    
    # Final refinement using gradient step on best solution
    best_h = global_best.copy()
    best_c5 = global_best_fitness
    
    # Try a few local gradient steps
    step_size = 0.005
    for _ in range(500):
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(best_h) - 1)
        
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - best_h[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= best_h[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - best_h[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= best_h[i]
        
        # Gradient descent step
        new_h = best_h - step_size * gradient
        
        # Ensure feasibility
        new_h = np.clip(new_h, 0, 1)
        # Renormalize
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate new C5 value
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
        else:
            step_size *= 0.995
            if step_size < 1e-8:
                break
    
    return (list(best_h), best_c5, n_points)
