import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create binary-like initial solution
        h_start = np.ones(n_points)
        h_start[:n_points//2] = 0
        # Adjust to meet constraint sum(h) = n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    initial_temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    max_iter = 100000
    no_improvement_limit = 1000
    temp = initial_temp
    
    # Perturbation parameters
    max_delta = 0.1
    
    iter_count = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95 and temp > min_temp and no_improvement_count < no_improvement_limit:
        iter_count += 1
        
        # Generate neighbor by perturbing two random indices
        i, j = random.sample(range(n_points), 2)
        delta = random.uniform(0, max_delta)
        
        # Ensure we don't exceed bounds
        if h_current[i] + delta > 1:
            delta = 1 - h_current[i]
        if h_current[j] - delta < 0:
            delta = h_current[j]
            
        if delta <= 0:
            continue
            
        # Create new solution
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Maintain sum constraint
        sum_diff = np.sum(h_new) - n_points / 2
        if abs(sum_diff) > 1e-10:
            # Add correction to first element
            h_new[0] -= sum_diff
            
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        # Re-normalize sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate energies
        c5_current = evaluate_erdos_solution(h_current, dx)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject new solution
        if c5_new < c5_current:
            h_current = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                no_improvement_count = 0
        else:
            prob = np.exp(-(c5_new - c5_current) / temp)
            if random.random() < prob:
                h_current = h_new
                
        # Cooling schedule
        temp *= cooling_rate
        
        no_improvement_count += 1
        
    # Final refinement with local search
    # Try a few gradient-based steps to improve the current solution
    for _ in range(100):
        if time() - start_time > budget_s * 0.98:
            break
            
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        lag_indices = np.arange(len(corr))
        lags = lag_indices - (len(h_current) - 1)
        overlap_values = corr * dx
        
        max_idx = np.argmax(overlap_values)
        max_lag = lags[max_idx]
        
        # Approximate gradient using finite differences
        h_new = h_current.copy()
        step = 0.001
        
        # Slightly perturb to estimate gradient
        for i in range(len(h_current)):
            if max_lag >= 0:
                if i + max_lag < len(h_current):
                    # Gradient with respect to h[i]
                    h_new[i] = h_current[i] + step
                    c5_plus = evaluate_erdos_solution(h_new, dx)
                    h_new[i] = h_current[i] - step
                    c5_minus = evaluate_erdos_solution(h_new, dx)
                    h_new[i] = h_current[i]  # restore
                    grad = (c5_plus - c5_minus) / (2 * step)
                    
                    # Update
                    h_new[i] = h_current[i] - 0.1 * grad
            else:
                if i - max_lag >= 0:
                    h_new[i] = h_current[i] + step
                    c5_plus = evaluate_erdos_solution(h_new, dx)
                    h_new[i] = h_current[i] - step
                    c5_minus = evaluate_erdos_solution(h_new, dx)
                    h_new[i] = h_current[i]
                    grad = (c5_plus - c5_minus) / (2 * step)
                    h_new[i] = h_current[i] - 0.1 * grad
            
            # Clamp and re-normalize
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    return (list(best_h), best_c5, n_points)
