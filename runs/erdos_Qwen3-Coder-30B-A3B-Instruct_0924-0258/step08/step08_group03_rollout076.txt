import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex optimization with filtering approach
    max_iterations = 1000
    K = 10  # Number of top overlaps to consider
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Get top K overlaps (indices and values)
        corr = np.correlate(h, 1 - h, mode='full')
        top_k_indices = np.argpartition(corr, -K)[-K:]
        top_k_values = corr[top_k_indices]
        
        # Convex optimization
        h_var = cp.Variable(n_points)
        
        # Objective: minimize weighted sum of top overlaps
        objective = cp.Minimize(cp.sum(cp.multiply(top_k_values, 
            [cp.sum(cp.multiply(h_var, 
                cp.convolution(1-h_var, np.ones(1), mode='valid')[i: i+n_points])) 
                for i in range(n_points)])))
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Solve problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=100)
            h_new = np.array(h_var.value).flatten()
        except:
            # Fall back to gradient descent
            h_new = h.copy()
            grad = np.zeros(n_points)
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (n_points - 1)
            for i in range(n_points):
                j = i + max_lag
                if 0 <= j < n_points:
                    grad[i] = 1 - 2 * h[i] - h[j]
                else:
                    grad[i] = 1 - 2 * h[i]
            
            step_size = 0.01 * (1.0 - iteration / max_iterations)
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Apply low-pass filter to smooth
        kernel_size = min(5, n_points // 2)
        if kernel_size > 1:
            kernel = np.ones(kernel_size) / kernel_size
            h_filtered = np.convolve(h_new, kernel, mode='same')
            # Normalize
            h_filtered = h_filtered * (n_points / 2) / np.sum(h_filtered)
            h_new = h_filtered
            
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            
            # Increase K for next iteration if needed
            if K < 20 and iteration % 50 == 0:
                K = min(K + 1, 20)
                
        # Early stopping
        if iteration > 0 and iteration % 100 == 0 and c5_new >= best_c5:
            if K > 5 and iteration > 200:
                K = max(K - 1, 5)
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
