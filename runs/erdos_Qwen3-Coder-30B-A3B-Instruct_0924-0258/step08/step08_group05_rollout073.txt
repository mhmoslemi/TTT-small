import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Use FFT-based correlation for faster evaluation
    def fast_correlation(h):
        # Use FFT for correlation
        H = np.fft.fft(h)
        G = 1 - h
        G_fft = np.fft.fft(G)
        # Cross-correlation in frequency domain
        corr = np.fft.ifft(H * np.conj(G_fft))
        return np.real(corr)
    
    # Gradient descent with FFT-based correlation
    max_iter = 10000
    step_size = 0.01
    tolerance = 1e-9
    no_improvement_count = 0
    no_improvement_limit = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Fast correlation computation
        corr = fast_correlation(h_current)
        overlap_values = corr * dx
        
        # Find the lag with maximum overlap
        max_idx = np.argmax(overlap_values)
        max_lag = max_idx - (len(h_current) - 1)
        
        # Compute gradient for the identified lag
        gradient = np.zeros_like(h_current)
        for i in range(len(h_current)):
            if max_lag >= 0:
                if i + max_lag < len(h_current):
                    gradient[i] = -1.0
            else:
                if i - max_lag >= 0:
                    gradient[i] = -1.0
        
        # Apply gradient descent step
        h_new = h_current - step_size * gradient
        
        # Project onto feasible set: keep values between 0 and 1 and maintain sum constraint
        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)
        if abs(sum_h - n_points/2) > 1e-10:
            projection = (n_points/2 - sum_h) / len(h_new)
            h_new = h_new + projection
            h_new = np.clip(h_new, 0, 1)
        
        # Compute the change in h
        change = np.sum(np.abs(h_new - h_current))
        if change < tolerance:
            no_improvement_count += 1
            if no_improvement_count > no_improvement_limit:
                step_size *= 0.5
                no_improvement_count = 0
        else:
            no_improvement_count = 0
            
        # Update current solution
        h_current = h_new
        
        # Evaluate new solution
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
    
    # Final refinement using SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
