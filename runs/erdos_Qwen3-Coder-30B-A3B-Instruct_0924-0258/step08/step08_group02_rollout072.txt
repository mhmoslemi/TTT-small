import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a binary-like construction
        h_start = np.ones(n_points)
        h_start[:n_points//2] = 0
        # Add some variation to avoid local minima
        h_start[0] = 0.5
        h_start[-1] = 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Threshold-based optimization
    current_h = h_start.copy()
    current_h = current_h * (n_points/2) / np.sum(current_h)
    
    # Use a cooling schedule for threshold
    threshold = 0.1
    cooling_factor = 0.95
    min_threshold = 0.01
    
    # Optimization loop
    iteration = 0
    max_iterations = 500
    
    while iteration < max_iterations and time() - start_time < budget_s * 0.95:
        # Compute overlaps for all lags
        overlaps = np.correlate(current_h, 1 - current_h, mode='full') * dx
        max_overlap = np.max(overlaps)
        lag_max = np.argmax(overlaps)
        
        # Compute gradient (approximate)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + lag_max) % n_points
            gradient[i] = -2 * dx * (1 - current_h[j] - current_h[i])
        
        # Thresholding: set extreme values to 0 or 1
        # Use top 50% of gradient by magnitude
        grad_magnitudes = np.abs(gradient)
        threshold_idx = int(n_points * 0.5)
        top_indices = np.argpartition(grad_magnitudes, -threshold_idx)[-threshold_idx:]
        
        # Apply thresholding with binary assignment
        new_h = current_h.copy()
        for i in range(n_points):
            if i in top_indices:
                if current_h[i] > 1 - threshold:
                    new_h[i] = 1.0
                elif current_h[i] < threshold:
                    new_h[i] = 0.0
                else:
                    # Interpolate if within threshold range
                    # Use gradient sign to guide interpolation
                    if gradient[i] > 0:
                        new_h[i] = min(1.0, current_h[i] + 0.1)
                    else:
                        new_h[i] = max(0.0, current_h[i] - 0.1)
        
        # Maintain sum constraint
        total = np.sum(new_h)
        if abs(total - n_points/2) > 1e-6:
            # Distribute the difference
            diff = n_points/2 - total
            # Distribute proportionally to gradient magnitude
            # or simply to smallest values to maximize change
            sorted_indices = np.argsort(new_h)
            for i in range(int(abs(diff))):  # only adjust necessary amount
                if diff > 0:
                    new_h[sorted_indices[i]] = min(1.0, new_h[sorted_indices[i]] + 1e-3)
                elif diff < 0:
                    new_h[sorted_indices[-(i+1)]] = max(0.0, new_h[sorted_indices[-(i+1)]] - 1e-3)
        
        # Final normalization
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Clip to feasible bounds
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Check if this improves the solution
        current_c5 = evaluate_erdos_solution(new_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_h.copy()
        
        current_h = new_h
        threshold = max(min_threshold, threshold * cooling_factor)
        iteration += 1
    
    # Coordinate-wise refinement: optimize 50% of coordinates with highest gradients
    for _ in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        overlaps = np.correlate(current_h, 1 - current_h, mode='full') * dx
        lag_max = np.argmax(overlaps)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + lag_max) % n_points
            gradient[i] = -2 * dx * (1 - current_h[j] - current_h[i])
        
        # Optimize top 50%
        grad_magnitudes = np.abs(gradient)
        threshold_idx = int(n_points * 0.5)
        top_indices = np.argpartition(grad_magnitudes, -threshold_idx)[-threshold_idx:]
        
        # Optimize each of these coordinates
        for idx in top_indices:
            orig = current_h[idx]
            best_val = orig
            best_score = evaluate_erdos_solution(current_h, dx)
            
            # Golden-section search from 0 to 1
            # Alternatively use a simple line search for efficiency
            for test_val in [0.0, 0.25, 0.5, 0.75, 1.0]:
                if test_val == 0.0 or test_val == 1.0:
                    test_h = current_h.copy()
                    test_h[idx] = test_val
                    test_h = test_h * (n_points/2) / np.sum(test_h)
                    score = evaluate_erdos_solution(test_h, dx)
                    if score < best_score:
                        best_score = score
                        best_val = test_val
                else:
                    test_h = current_h.copy()
                    test_h[idx] = test_val
                    # Maintain sum constraint by adjusting other values
                    diff = test_val - orig
                    other_indices = [i for i in range(n_points) if i != idx]
                    adjust_amount = diff / len(other_indices)
                    for i in other_indices:
                        test_h[i] -= adjust_amount
                    test_h = np.clip(test_h, 0, 1)
                    test_h = test_h * (n_points/2) / np.sum(test_h)
                    score = evaluate_erdos_solution(test_h, dx)
                    if score < best_score:
                        best_score = score
                        best_val = test_val
            
            # Apply the best value
            current_h[idx] = best_val
            current_h = current_h * (n_points/2) / np.sum(current_h)
            
        # Update best solution if improved
        current_c5 = evaluate_erdos_solution(current_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
    
    # Final L-BFGS refinement
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    # Final normalization and clipping
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
