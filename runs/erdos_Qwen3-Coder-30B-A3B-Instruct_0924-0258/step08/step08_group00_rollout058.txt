import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try with different initializations
    initializations = [h_start]
    
    # Add some random variations
    for _ in range(3):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        initializations.append(h_rand)
    
    # Try grid-based constructions
    grid_points = [10, 20, 30, 35]
    for gp in grid_points:
        if gp <= n_points:
            h_grid = np.zeros(n_points)
            # Create a piecewise constant function
            segments = gp
            segment_size = n_points // segments
            for i in range(segments):
                start_idx = i * segment_size
                end_idx = min((i + 1) * segment_size, n_points)
                # Alternate between 0 and 1
                h_grid[start_idx:end_idx] = 1.0 if i % 2 == 0 else 0.0
            h_grid = h_grid * (n_points/2) / np.sum(h_grid)
            initializations.append(h_grid)
    
    # Try constructive approach
    for h_init in initializations:
        h = h_init.copy()
        h = h * (n_points/2) / np.sum(h)
        
        # Optimization using Nelder-Mead
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use simpler optimization first to get close to good solution
        try:
            res = minimize(objective, h, method='Nelder-Mead', 
                           options={'maxiter': 200, 'fatol': 1e-8})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
        
        # Then use more advanced methods if time allows
        if time() - start_time > budget_s * 0.7:
            break
            
        try:
            res = minimize(objective, h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 300, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    # Final refinement
    if time() - start_time < budget_s * 0.95:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP',

                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    # Ensure final solution meets bounds
    best_h = np.clip(best_h, 0, 1)

    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
