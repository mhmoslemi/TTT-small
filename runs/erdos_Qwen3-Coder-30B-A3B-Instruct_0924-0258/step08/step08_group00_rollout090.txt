import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h onto feasible set: [0,1]^n and sum = n/2"""
    h_clipped = np.clip(h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    h_scaled = h_clipped * (n_points / 2.0) / h_sum
    return h_scaled

def compute_gradient(h, max_lag, n_points):
    """Compute gradient of overlap at the max lag"""
    shifted = np.roll(1 - h, max_lag)
    gradient = 2 * (1 - 2 * h) * shifted
    return gradient

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction or create starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = project(h_start, n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    max_iter = 1000
    patience = 50
    no_improvement = 0
    step_size = 0.1
    min_step_size = 1e-8
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlap
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient
        grad = compute_gradient(h, max_lag, n_points)
        
        # Update
        h_new = h - step_size * grad
        h_new = project(h_new, n_points)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement >= patience:
                step_size *= 0.5
                if step_size < min_step_size:
                    break
                no_improvement = 0
                # Try small step in same direction
                h_new = h - step_size * grad
                h_new = project(h_new, n_points)
                new_c5 = evaluate_erdos_solution(h_new, dx)
                if new_c5 >= best_c5:
                    break
        
        h = h_new
        
    return (list(best_h), best_c5, n_points)
