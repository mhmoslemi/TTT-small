import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different starting points
    candidates = [h_start]
    
    # Add some random perturbations to the initial solution
    for i in range(5):
        h_pert = h_start.copy()
        perturb = np.random.normal(0, 0.05, n_points)
        h_pert += perturb
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        candidates.append(h_pert)
    
    # Try a few different optimization strategies
    
    # 1. Gradient-based with multiple restarts
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    
    max_iter = 500
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Identify lag with maximum overlap
        max_lag = max_lag_idx - (n_points - 1)  # Convert index to actual lag
        
        # Compute gradient w.r.t. to h for this specific lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 2 * (1 - h[j] - h[i]) / n_points
        
        # Update with gradient descent
        h_new = h - 0.01 * grad
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        h = h_new
    
    # 2. Global optimization with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        for h_start in candidates:
            res = minimize(objective, h_start, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 300, 'ftol': 1e-8})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final
    except Exception:
        pass
    
    # 3. Local search with simulated annealing style
    h = best_h.copy()
    temperature = 0.1
    cooling_rate = 0.99
    iterations_per_temp = 100
    
    for _ in range(10000):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Perturb h slightly
        h_new = h.copy()
        idx = random.randint(0, n_points - 1)
        delta = random.uniform(-0.1, 0.1)
        h_new[idx] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        old_overlap = evaluate_erdos_solution(h, dx)
        new_overlap = evaluate_erdos_solution(h_new, dx)
        
        if new_overlap < old_overlap or random.random() < np.exp((old_overlap - new_overlap) / temperature):
            h = h_new
        
        temperature *= cooling_rate
        if temperature < 1e-6:
            temperature = 1e-6
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    return (list(best_h), best_c5, n_points)
