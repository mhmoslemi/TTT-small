import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search parameters
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    # Sampling for feasible checks
    shift_samples = 100
    
    def check_feasible(c):
        # Sample shifts
        shifts = np.random.choice(n_points, shift_samples, replace=False)
        
        # CVXPY problem
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Overlap constraints
        for k in shifts:
            # Compute overlap
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h_var[i] * (1 - h_var[j])
            constraints.append(overlap <= c * dx)
        
        # Objective (just feasibility)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INACCURATE
        except:
            return False
    
    # Binary search
    while time() - start_time < budget_s * 0.9 and high - low > tolerance:
        mid = (low + high) / 2
        if check_feasible(mid):
            high = mid
            # Refine h to better satisfy this bound
            h_new = h.copy()
            def objective(x):
                return evaluate_erdos_solution(x, dx)
            
            def constraint_func(x):
                # Ensure normalization
                return np.sum(x) - n_points / 2
            
            cons = {'type': 'eq', 'fun': constraint_func}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Local optimization
            result = minimize(objective, h_new, method='SLSQP', 
                            bounds=bounds, constraints=cons, 
                            options={'maxiter': 500, 'ftol': 1e-6})
            
            if result.success:
                h_new = result.x
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        else:
            low = mid
    
    # Final refinement
    h_final = best_h.copy()
    
    # Gradient descent refinement with limited iterations
    max_iter = 1000
    step_size = 0.1
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlations
        corr = np.correlate(h_final, 1 - h_final, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient for the worst case
        grad = np.zeros(n_points)
        for idx in range(n_points):
            j = idx + max_lag
            if 0 <= j < n_points:
                grad[idx] = 1 - 2 * h_final[idx] - h_final[j]
            else:
                grad[idx] = 1 - 2 * h_final[idx]
        
        # Update
        h_new = h_final - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_final = h_new.copy()
        else:
            step_size *= 0.9
            if step_size < 1e-6:
                break
    
    return (list(best_h), best_c5, n_points)
