import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    T = T_init
    max_iterations = 100000
    epsilon = 0.1
    
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    c5_current = evaluate_erdos_solution(h_current, dx)
    
    if c5_current < best_c5:
        best_c5 = c5_current
        best_h = h_current.copy()
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb two random indices
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Store old values
        old_i = h_current[i]
        old_j = h_current[j]
        
        # Modify values
        delta_i = np.random.uniform(-epsilon, epsilon)
        delta_j = -delta_i
        
        # Ensure constraints
        new_i = old_i + delta_i
        new_j = old_j + delta_j
        
        # Clip to bounds
        new_i = np.clip(new_i, 0, 1)
        new_j = np.clip(new_j, 0, 1)
        
        # Adjust other values to preserve sum constraint
        diff = (new_i - old_i) + (new_j - old_j)
        if abs(diff) > 1e-10:
            # Find another index to adjust
            k = np.random.randint(0, n_points)
            while k == i or k == j:
                k = np.random.randint(0, n_points)
            h_current[k] = np.clip(h_current[k] - diff, 0, 1)
            
        # Update h_current
        h_current[i] = new_i
        h_current[j] = new_j
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_current, dx)
        
        # Accept or reject
        if c5_new < c5_current:
            c5_current = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_current.copy()
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(c5_new - c5_current) / T):
                c5_current = c5_new
            else:
                # Revert change
                h_current[i] = old_i
                h_current[j] = old_j
                if abs(diff) > 1e-10:
                    h_current[k] = np.clip(h_current[k] + diff, 0, 1)
        
        # Cooling schedule
        T *= alpha
        if T < T_min:
            break
    
    # Local refinement with L-BFGS-B
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception as e:
        pass
    
    return (list(best_h), best_c5, n_points)
