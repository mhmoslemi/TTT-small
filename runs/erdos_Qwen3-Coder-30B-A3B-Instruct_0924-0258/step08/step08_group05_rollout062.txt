import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Trust-region method with cutting planes
    max_iter = 1000
    trust_radius = 0.2
    min_trust_radius = 1e-6
    max_trust_radius = 1.0
    improvement_threshold = 1e-8
    no_improvement_count = 0
    max_no_improvement = 50
    
    # Build initial constraints
    corr = np.correlate(h_current, 1 - h_current, mode='full')
    lags = np.arange(len(corr)) - (len(h_current) - 1)
    overlap_values = corr * dx
    
    max_idx = np.argmax(overlap_values)
    max_lag = lags[max_idx]
    
    # Convex relaxation approach with cutting planes
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Define optimization variables
        h_vars = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(h_vars) == n_points / 2.0]
        constraints += [h_vars >= 0]
        constraints += [h_vars <= 1]
        
        # Add previous best constraints
        for i in range(len(h_current)):
            idx = i + max_lag
            if 0 <= idx < len(h_current):
                constraints += [h_vars[i] * (1 - h_vars[idx]) <= C]
                
        # Convex relaxation (approximate with linearization around current point)
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_vars.value
                if h_new is not None:
                    h_new = np.clip(h_new, 0, 1)
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        trust_radius = min(max_trust_radius, trust_radius * 1.2)
                        no_improvement_count = 0
                    else:
                        no_improvement_count += 1
                        trust_radius = max(min_trust_radius, trust_radius * 0.8)
                else:
                    trust_radius = max(min_trust_radius, trust_radius * 0.8)
            else:
                trust_radius = max(min_trust_radius, trust_radius * 0.8)
        except:
            trust_radius = max(min_trust_radius, trust_radius * 0.8)
            
        if no_improvement_count > max_no_improvement:
            break
    
    # Final local refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
