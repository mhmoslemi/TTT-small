import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_fine = np.array(initial_h_values)
        n_points = len(h_fine)
    else:
        n_points = 35
        h_fine = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_fine.copy()
    best_c5 = evaluate_erdos_solution(h_fine, dx)
    
    # Coarse grid optimization
    n_coarse = 15
    h_coarse = np.ones(n_coarse) * 0.5
    coarse_dx = 2.0 / n_coarse
    
    for iter_coarse in range(500):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate overlaps
        corr_coarse = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_lag_idx = np.argmax(corr_coarse)
        max_lag = max_lag_idx - (n_coarse - 1)
        
        # Gradient estimation
        grad_coarse = np.zeros(n_coarse)
        for i in range(n_coarse):
            j = i + max_lag
            if 0 <= j < n_coarse:
                grad_coarse[i] = 1 - 2 * h_coarse[i] - h_coarse[j]
            else:
                grad_coarse[i] = 1 - 2 * h_coarse[i]
        
        # Descent with step size 0.01
        h_coarse_new = h_coarse - 0.01 * grad_coarse
        h_coarse_new = np.clip(h_coarse_new, 0, 1)
        h_coarse_new = h_coarse_new * (n_coarse / 2) / np.sum(h_coarse_new)
        
        # Update to coarse
        h_coarse = h_coarse_new.copy()
        
        # Interpolate to fine grid
        x_coarse = np.linspace(0, 2, n_coarse)
        x_fine = np.linspace(0, 2, n_points)
        h_fine = interp1d(x_coarse, h_coarse, kind='linear', fill_value='extrapolate')(x_fine)
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        # Evaluate and update best
        c5_new = evaluate_erdos_solution(h_fine, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_fine.copy()
    
    # Fine grid optimization with hybrid approach
    # Alternate gradient descent and random mutation
    fine_iterations = 2000
    mutation_rate = 0.10
    step_size = 0.01
    
    h_now = best_h.copy()
    
    for iter_fine in range(fine_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Every 50 iterations, perform mutation
        if iter_fine % 50 == 0:
            mutated_indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
            h_now[mutated_indices] += np.random.normal(0, 0.02, len(mutated_indices))
            h_now = np.clip(h_now, 0, 1)
            h_now = h_now * (n_points / 2) / np.sum(h_now)
            
            current_c5 = evaluate_erdos_solution(h_now, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_now.copy()
                
        # Compute max overlap and gradient
        corr = np.correlate(h_now, 1 - h_now, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h_now[i] - h_now[j]
            else:
                grad[i] = 1 - 2 * h_now[i]
        
        # Gradient descent step
        h_new = h_now - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Accept improvement or reduce step size
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            h_now = h_new.copy()
            best_c5 = c5_new
        else:
            step_size *= 0.95
            if step_size < 1e-8:
                step_size = 1e-6
            
    return (list(best_h), best_c5, n_points)
