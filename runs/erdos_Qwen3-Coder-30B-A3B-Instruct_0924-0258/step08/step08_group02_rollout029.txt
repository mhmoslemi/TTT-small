import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iterations = 100000
    
    # Perturbation step size
    epsilon = 0.1
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    T = T_init
    iteration = 0
    
    while T > T_min and iteration < max_iterations:
        if time() - start_time > budget_s * 0.9:
            break
            
        # Generate neighbor by perturbing two random points
        i, j = np.random.randint(0, n_points, 2)
        
        # Create a copy and perturb
        h_new = h.copy()
        
        # Adjust two points to maintain sum constraint
        delta = np.random.uniform(-epsilon, epsilon)
        h_new[i] += delta
        h_new[j] -= delta
        
        # Enforce bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Compute overlap
        old_overlap = evaluate_erdos_solution(h, dx)
        new_overlap = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if new_overlap < old_overlap:
            h = h_new.copy()
            if new_overlap < best_c5:
                best_c5 = new_overlap
                best_h = h_new.copy()
        else:
            # Accept with probability
            prob = np.exp(-(new_overlap - old_overlap) / T)
            if np.random.rand() < prob:
                h = h_new.copy()
        
        # Cooling schedule
        T *= alpha
        iteration += 1
    
    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        h_refined = best_h.copy()
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        # Define objective with sum constraint
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        # Use scipy minimize with L-BFGS-B
        res = minimize(
            objective,
            h_refined,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
