import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Start with a simple structure: one high peak and one low region
        n_points = 35
        h_start = np.zeros(n_points)
        # Create two regions, one high (0.7) and one low (0.3)
        h_start[:n_points//2] = 0.7
        h_start[n_points//2:] = 0.3
    
    dx = 2.0 / n_points
    
    # Normalize to satisfy sum constraint
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try local optimization with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try SLSQP on the initial construction
    try:
        res = minimize(objective, h_start, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 2000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    # Try gradient-free method - random search with local descent
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Let's also try a convex relaxation approach
    if n_points <= 40:
        try:
            # Use CVXPY for relaxed optimization (if feasible for small n)
            h_var = cp.Variable(n_points)
            C = cp.Variable()
            
            # Create constraints
            constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
            
            # Overlap constraints for all possible lags
            for k in range(-n_points + 1, n_points):
                if k >= 0:
                    lag_indices = range(n_points - k)
                    sum_term = cp.sum(cp.multiply(h_var[lag_indices], 1 - h_var[lag_indices + k]))
                else:
                    lag_indices = range(n_points + k)
                    sum_term = cp.sum(cp.multiply(h_var[lag_indices - k], 1 - h_var[lag_indices]))
                constraints.append(sum_term <= C)
            
            # Objective
            objective = cp.Minimize(C)
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                relaxed_h = h_var.value
                relaxed_c5 = prob.value
                if relaxed_c5 < current_c5:
                    # Refine with local search from this solution
                    try:
                        refined_res = minimize(objective, relaxed_h, method='SLSQP', bounds=bounds,
                                              constraints=cons_sum, options={'maxiter': 500, 'ftol': 1e-10})
                        if refined_res.success:
                            refined_h = refined_res.x
                            refined_c5 = objective(refined_h)
                            if refined_c5 < current_c5:
                                current_h = refined_h
                                current_c5 = refined_c5
                    except Exception:
                        current_h = relaxed_h
                        current_c5 = relaxed_c5
        except Exception:
            pass
    
    # Final pass with local descent starting from best solution
    try:
        res = minimize(objective, current_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 2000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < current_c5:
                current_h = h_final
                current_c5 = c5_final
    except Exception:
        pass

    # Final validation
    final_c5 = evaluate_erdos_solution(current_h, dx)
    
    return (list(current_h), final_c5, n_points)
