import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h = h * (n_points/2) / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    best_c5_time = time()
    
    # Conservative optimization loop
    max_iter = 50000
    no_improvement_limit = 500
    no_improvement_count = 0
    last_best = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation in valid range
        h_new = h.copy()
        i = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.05, 0.05)
        h_new[i] += delta
        h_new[i] = np.clip(h_new[i], 0, 1)
        
        # Renormalize to satisfy sum constraint
        sum_h = np.sum(h_new)
        h_new = h_new * (n_points/2) / sum_h
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_best = best_c5
            no_improvement_count = 0
            best_c5_time = time()
        else:
            no_improvement_count += 1
            
        # Early stopping heuristic
        if no_improvement_count >= no_improvement_limit:
            break
        h = h_new.copy()
    
    # Final refinement using SLSQP if time allows
    if time() - start_time < budget_s * 0.9:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
