import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate diverse initial guesses using Latin Hypercube Sampling
    n_initial_guesses = 50
    initial_guesses = []
    
    # Add the initial construction if available
    if 'initial_h_values' in globals():
        initial_guesses.append(h_start.copy())
    
    # Generate LHS samples
    for _ in range(n_initial_guesses - len(initial_guesses)):
        # Generate samples uniformly in [0,1] using LHS for better coverage
        sample = np.random.uniform(0, 1, n_points)
        # Normalize to satisfy sum constraint
        sample = sample * (n_points / 2) / np.sum(sample)
        initial_guesses.append(sample)
    
    # Refine each guess with SLSQP
    for i, h0 in enumerate(initial_guesses):
        if time() - start_time > budget_s * 0.95:
            break
            
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h0, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                # Repair to maintain feasibility
                h_final = np.clip(h_final, 0, 1)
                if np.sum(h_final) > 0:
                    h_final = h_final * (n_points/2) / np.sum(h_final)
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
        except Exception:
            pass
    
    # Final check with current best and return
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
