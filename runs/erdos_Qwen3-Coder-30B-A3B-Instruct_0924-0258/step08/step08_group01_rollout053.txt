import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    no_improvement_limit = 50
    refine_every = 100
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # Normalize to sum to n_points/2
        return np.clip(h, 0, 1)
    
    population = [h_start.copy()] + [create_individual() for _ in range(population_size-1)]
    
    def fitness(individual):
        return evaluate_erdos_solution(individual, dx)
    
    def tournament_selection(population, k=3):
        tournament = random.sample(population, min(k, len(population)))
        return min(tournament, key=fitness)
    
    def blend_crossover(parent1, parent2, alpha=0.5):
        # Blend crossover
        child1 = alpha * parent1 + (1 - alpha) * parent2
        child2 = (1 - alpha) * parent1 + alpha * parent2
        
        # Ensure sum constraint and clip to [0,1]
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        
        child1 = np.clip(child1, 0, 1)
        child2 = np.clip(child2, 0, 1)
        
        return child1, child2
    
    def mutate(individual, rate=0.1, step_size=0.05):
        mutated = individual.copy()
        for i in range(len(mutated)):
            if random.random() < rate:
                # Add small random change
                delta = random.gauss(0, step_size)
                mutated[i] += delta
        # Renormalize
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        mutated = np.clip(mutated, 0, 1)
        return mutated
    
    no_improvement_count = 0
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness_scores = [fitness(ind) for ind in population]
        
        # Track best
        min_fitness = min(fitness_scores)
        best_idx = np.argmin(fitness_scores)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= no_improvement_limit:
            break
            
        # Selection and reproduction
        new_population = [population[best_idx].copy()]  # Elitism
        
        while len(new_population) < population_size:
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            
            if random.random() < crossover_rate:
                child1, child2 = blend_crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
        
        # Trim to exact size
        population = new_population[:population_size]
    
    # Final refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
