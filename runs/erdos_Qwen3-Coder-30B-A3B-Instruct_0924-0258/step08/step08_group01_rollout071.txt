import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale optimization
    n_scales = int(np.log2(max(n_points, 16))) + 1
    scale = 1
    h_current = h_start.copy()
    
    while scale <= n_scales and time() - start_time < budget_s * 0.95:
        # Doubling the resolution
        new_n_points = 2 ** scale
        if new_n_points > 1000:
            break
            
        # Upsample the current solution
        h_upsampled = np.interp(np.linspace(0, 2, new_n_points), 
                               np.linspace(0, 2, len(h_current)), 
                               h_current)
        
        # Apply perturbations and refinement
        mu = 0.05
        no_improvement = 0
        max_no_improvement = 10
        
        for _ in range(10):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb the solution
            h_perturbed = h_upsampled + mu * (2 * np.random.rand(new_n_points) - 1)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            
            # Adjust to meet sum constraint
            h_perturbed = h_perturbed * (new_n_points / 2) / np.sum(h_perturbed)
            
            # Evaluate
            c5_perturbed = evaluate_erdos_solution(h_perturbed, 2.0 / new_n_points)
            
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
                h_current = h_perturbed.copy()
                no_improvement = 0
                mu = 0.05  # reset perturbation magnitude
            else:
                no_improvement += 1
                mu *= 0.95  # reduce perturbation magnitude
                
            if no_improvement >= max_no_improvement:
                break
        
        # SQP refinement
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / new_n_points)
        
        def constraint_sum(h):
            return np.sum(h) - new_n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(new_n_points)]
        
        try:
            res = minimize(objective, h_current, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 100, 'ftol': 1e-12})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
            
        scale += 1
        
    # Final refinement
    try:
        n_points_final = len(best_h)
        dx_final = 2.0 / n_points_final
        
        def objective(h):
            return evaluate_erdos_solution(h, dx_final)
        
        def constraint_sum(h):
            return np.sum(h) - n_points_final / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points_final)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons_sum, 
                      options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, len(best_h))
