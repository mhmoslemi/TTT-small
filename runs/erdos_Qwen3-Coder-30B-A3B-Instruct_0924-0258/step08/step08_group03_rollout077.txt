import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Parameters
    max_iterations = 10000
    k_top = 10  # Number of top overlaps to consider
    filter_width = 5  # Width of moving average filter
    
    # Iterative optimization
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convex relaxation: minimize a linear combination of top k overlaps
        # First, compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        # Get indices of top k largest values
        top_k_indices = np.argsort(corr)[-k_top:]
        
        # Create variables for cvxpy problem
        h_vars = cp.Variable(n_points)
        
        # Objective function: sum of top k correlations
        objective = cp.Minimize(cp.sum([cp.multiply(h_vars[i], 1 - h_vars[i + k - n_points + 1]) 
                                        for i, k in enumerate(top_k_indices)]))
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Solve problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == 'optimal':
                h_new = h_vars.value
            else:
                h_new = h.copy()
        except:
            h_new = h.copy()
        
        # Apply low-pass filter
        filtered_h = np.convolve(h_new, np.ones(filter_width) / filter_width, mode='same')
        h_new = filtered_h
        
        # Renormalize to satisfy constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            
            # Increase k_top gradually to focus more on significant overlaps
            if k_top < 30:
                k_top = min(k_top + 1, 30)
            
            # Reduce filter bandwidth to maintain smoothness
            filter_width = max(2, filter_width - 1)
        else:
            # Reduce k_top if no improvement made
            if k_top > 5:
                k_top = max(5, k_top - 1)
                
        # Early stopping if no improvement over recent iterations
        if iteration > 100 and iteration % 100 == 0:
            if c5_new >= best_c5:
                break
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
