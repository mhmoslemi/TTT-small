import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Trust Region with Cutting Planes
    max_iter = 5000
    delta = 0.2
    tolerance = 1e-8
    improvement_threshold = 1e-6
    no_improvement_count = 0
    max_no_improvement = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            # Adaptive perturbation
            h_current = best_h.copy()
            perturbation = np.random.normal(0, 0.01, n_points)
            h_current += perturbation
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
            
        # Solve convex approximation using CVXPY
        h_vars = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(h_vars) == n_points / 2.0]
        constraints += [0 <= h_vars[i] for i in range(n_points)]
        constraints += [h_vars[i] <= 1 for i in range(n_points)]
        
        # Add cutting planes (most violated constraints)
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        lag_indices = np.arange(len(corr))
        lags = lag_indices - (len(h_current) - 1)
        overlap_values = corr * dx
        
        # Add constraints for top 10% most violated overlaps
        most_violated = np.argsort(overlap_values)[-int(0.1 * len(overlap_values)):]
        
        for idx in most_violated:
            if overlap_values[idx] > C:
                lag = lags[idx]
                if lag >= 0:
                    # sum_{i=0}^{n_points-lag-1} h[i](1 - h[i+lag])
                    expr = cp.sum(cp.multiply(h_vars[:n_points-lag], 1 - h_vars[lag:]))
                else:
                    # sum_{i=-lag}^{n_points-1} h[i](1 - h[i-lag])
                    expr = cp.sum(cp.multiply(h_vars[-lag:], 1 - h_vars[:-lag]))
                constraints.append(expr <= C)
        
        # Objective: minimize C
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_vars.value
                if h_new is not None:
                    # Project onto feasible region
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points/2) / np.sum(h_new)
                    
                    # Update trust region if we made progress
                    if np.sum(np.abs(h_new - h_current)) > tolerance:
                        h_current = h_new
                        delta = min(1.2 * delta, 0.5)  # Increase trust region
                    else:
                        delta = max(0.8 * delta, 1e-6)  # Shrink trust region
                else:
                    # Fallback to simple update
                    h_current = np.clip(h_current, 0, 1)
                    h_current = h_current * (n_points/2) / np.sum(h_current)
            else:
                # Fallback if optimization fails
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points/2) / np.sum(h_current)
        except Exception:
            # Fallback
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
            
    # Final refinement using SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        # Perform final optimization with all constraints
        # with large number of overlap constraints
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
