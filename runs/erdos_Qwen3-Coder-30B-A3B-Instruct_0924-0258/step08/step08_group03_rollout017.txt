```python
import numpy as np
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search parameters
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    # Sample shifts for SDP constraint
    shifts = np.random.choice(range(1, n_points), size=min(100, n_points-1), replace=False)
    
    while time() - start_time < budget_s * 0.95 and high - low > tolerance:
        mid = (low + high) / 2
        
        # Formulate SDP
        h_var = cp.Variable(n_points)
        Y = cp.Variable((n_points, n_points), symmetric=True)
        
        constraints = [
            Y >> 0,  # Y is positive semidefinite
            cp.diag(Y) == h_var,  # diagonal constraint
            cp.sum(h_var) == n_points / 2,  # normalization constraint
        ]
        
        # Add overlap constraints for sampled shifts
        for k in shifts:
            # Convolution constraint: sum_i h_i * (1 - h_{i+k}) <= mid * dx
            # This becomes: sum_i h_i - sum_i Y[i, i+k] <= mid * dx
            constraint = cp.sum(h_var) - cp.sum(Y[range(n_points-k), range(k, n_points)]) <= mid * dx
            constraints.append(constraint)
        
        # Minimize C5 (i.e., this is feasibility check)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=CP_SOLVER, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                # Get solution
                h_sol = h_var.value
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    c5_sol = evaluate_erdos_solution(h_sol, dx)
                    if c5_sol < best_c5:
                        best_h = h_sol.copy()
                        best_c5 = c5_sol
                    high = mid
                else:
                    low = mid
            else:
                low = mid
        except:
            low = mid

    # Local refinement with gradient descent
    h_refined = best_h.copy()
    step_size = 0.01
    tolerance = 1e-9
    max_iter = 1000

    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for j in range(n_points):
            k = j + max_lag
            if 0 <= k < n_points:
                grad[j] = 1 - 2 * h_refined[j] - h_refined[k]
            else:
                grad[j] = 1 - 2 * h_refined[j]
        
        h_new = h_refined - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_refined = h_new.copy()
        else:
            step_size *= 0.5
            if step_size < tolerance:
                break

    return (list(best_h), best_c5, n_points)
```
