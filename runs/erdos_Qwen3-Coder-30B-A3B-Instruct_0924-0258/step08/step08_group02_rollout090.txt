import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    n_points = 35
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        # Create initial step function
        h_current = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint sum(h) * dx = 1
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Hyperparameters
    max_iter = 500
    lambda_reg = 0.1
    alpha_reg = 0.01
    coord_pct = 0.2
    max_inner_iter = 50
    no_improve_iter = 0
    max_no_improve = 50
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convex relaxation step
        try:
            # Optimization variables
            h_vars = cp.Variable(n_points)
            
            # Objective: relaxed sum of squares of overlaps
            obj = cp.Minimize(
                lambda_reg * cp.sum_squares(cp.matmul(cp.eye(n_points), h_vars) - 
                                          cp.roll(h_vars, 1)) + 
                alpha_reg * cp.quad_form(h_vars, cp.eye(n_points))
            )
            
            # Constraints
            constraints = [
                cp.sum(h_vars) == n_points/2,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Problem
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_relaxed = h_vars.value
                # Normalize
                h_relaxed = h_relaxed * (n_points/2) / np.sum(h_relaxed)
            else:
                h_relaxed = h_current.copy()
        except:
            h_relaxed = h_current.copy()
        
        # Coordinate descent step
        h_new = h_relaxed.copy()
        indices_to_update = np.random.permutation(n_points)[:int(n_points * coord_pct)]
        old_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Gradient-based update for selected coordinates
        for _ in range(max_inner_iter):
            overlaps = np.correlate(h_new, 1 - h_new, mode='full') * dx
            max_k = np.argmax(overlaps)
            
            # Compute gradient for selected indices
            gradient = np.zeros(n_points)
            for idx in indices_to_update:
                # Derivative of overlap with respect to h[idx] at max_k
                # We consider contribution from h[idx] and h[idx - max_k]
                j1 = (idx - max_k) % n_points
                if 0 <= j1 < n_points:
                    gradient[idx] += -2 * dx * (1 - h_new[j1])
                # The other term comes from h[idx + max_k] 
                # (but h[idx] is not directly in h[idx+k] for large k)
                # Since h[idx] appears in overlap with h[idx+k] 
                # for the term where k was the shift...
                # We need to be more careful here...

            # Build explicit gradient for all points
            gradient = np.zeros(n_points)
            # For the maximum lags (determined by max_k), do:
            # d/dh[i] (h[i] * (1 - h[i + max_k])) = 1 - h[i + max_k] - h[i]
            for i in range(n_points):
                j = (i + max_k) % n_points
                gradient[i] = -2 * dx * (1 - h_new[j] - h_new[i])
            
            # Update selected coordinates with gradient descent
            step_size = 0.01
            for idx in indices_to_update:
                h_new[idx] = h_new[idx] - step_size * gradient[idx]
                h_new[idx] = np.clip(h_new[idx], 0, 1)
            
            # Normalize
            h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Enforce constraints explicitly
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improve_iter = 0
        else:
            no_improve_iter += 1
            
        if no_improve_iter >= max_no_improve:
            break
        
        h_current = h_new.copy()
    
    # Final L-BFGS refinement
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return -np.max(overlaps)
        
        # Normalize initial
        h_init = best_h.copy()
        h_init = h_init * (n_points/2) / np.sum(h_init)
        
        # Optimize
        res = minimize(objective, h_init, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 100, 'ftol': 1e-9})
        
        if res.success:
            refined_h = res.x.copy()
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass

    return (list(best_h), best_c5, n_points)
