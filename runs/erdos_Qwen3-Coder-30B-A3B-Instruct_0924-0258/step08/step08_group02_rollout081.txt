import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    T = T_init
    max_iter = int(1e5)
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    
    epsilon = 0.1
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Perturb two random indices
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-epsilon, epsilon)
        
        # Adjust h[idx1] and h[idx2]
        old_h1 = h[idx1]
        old_h2 = h[idx2]
        new_h1 = h[idx1] + delta
        new_h2 = h[idx2] - delta
        
        # Bound adjustments to stay within [0,1]
        if new_h1 < 0:
            new_h1 = 0
            new_h2 = old_h2 + (old_h1 - h[idx1])
        elif new_h1 > 1:
            new_h1 = 1
            new_h2 = old_h2 + (old_h1 - h[idx1])
            
        if new_h2 < 0:
            new_h2 = 0
            new_h1 = old_h1 + (old_h2 - h[idx2])
        elif new_h2 > 1:
            new_h2 = 1
            new_h1 = old_h1 + (old_h2 - h[idx2])
        
        # Update h
        h[idx1] = new_h1
        h[idx2] = new_h2
        
        # Maintain sum constraint
        total_diff = np.sum(h) - (n_points / 2)
        if abs(total_diff) > 1e-10:
            # Adjust the first element
            h[0] = h[0] - total_diff
            h[0] = np.clip(h[0], 0, 1)
        
        # Compute C5
        current_c5 = evaluate_erdos_solution(h, dx)
        
        # Accept or reject
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        else:
            # Accept with probability
            delta_c5 = current_c5 - best_c5
            if np.random.rand() < np.exp(-delta_c5 / T):
                # Accept the worse solution
                pass
            else:
                # Reject, restore the old h
                h[idx1] = old_h1
                h[idx2] = old_h2
                
        # Cool down
        T = T * alpha
        if T < T_min:
            T = T_min
    
    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
