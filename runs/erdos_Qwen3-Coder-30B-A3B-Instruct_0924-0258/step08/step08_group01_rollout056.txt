import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 2
    no_improvement_limit = 100
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h_random = np.random.rand(n_points)
        h_random = h_random * (n_points / 2) / np.sum(h_random)
        h_random = np.clip(h_random, 0, 1)
        population.append(h_random)
    
    # Create initial fitness list
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    best_fitness = min(fitness)
    best_individual = population[fitness.index(best_fitness)]
    
    # For tracking improvements
    no_improvement_count = 0
    
    # Main GA loop
    gen = 0
    while time() - start_time < budget_s * 0.7 and gen < generations:
        # Evaluate current generation
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best if necessary
        current_best_fitness = min(fitness)
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = population[fitness.index(current_best_fitness)]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= no_improvement_limit:
            break
            
        # Tournament selection - 3 individuals
        def tournament_select():
            k = 3
            candidates = np.random.choice(len(population), k)
            best_candidate = candidates[np.argmin([fitness[i] for i in candidates])]
            return population[best_candidate].copy()
            
        # Create new population via selection, crossover, and mutation
        new_population = []
        
        # Elitism: keep the best individuals
        elites = [population[i] for i in np.argsort(fitness)[:elite_size]]
        new_population.extend(elites)
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_select()
            parent2 = tournament_select()
            
            # Crossover with blend method
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = parent1 * (1.0 - alpha) + parent2 * alpha
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Add small noise
                noise = np.random.normal(0, 0.02, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                
            # Preserve sum constraint
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
        # Replace population
        population = new_population[:population_size]
        
        gen += 1
    
    # Final refinement with SLSQP on best individual
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_individual, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_fitness:
                best_individual = h_refined
                best_fitness = c5_refined
    except Exception:
        pass
    
    return (list(best_individual), best_fitness, n_points)
