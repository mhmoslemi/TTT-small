import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize trust region with a reasonable radius
    trust_radius = 0.2
    
    # Helper for computing all overlaps
    def compute_all_overlaps(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        lags = np.arange(len(corr)) - (len(h_vals) - 1)
        overlap_values = corr * dx
        return overlap_values, lags
    
    # Initialize optimization variables
    h_current = h_start.copy()
    current_c5 = best_c5
    
    max_trust_iter = 500
    
    for iter_count in range(max_trust_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update best if necessary
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            
        # Get all overlap values and lags
        overlap_vals, lags = compute_all_overlaps(h_current)
        max_idx = np.argmax(overlap_vals)
        max_lag = lags[max_idx]
        
        # Compute gradients approximated by finite differences
        epsilon = 1e-5
        grad_h = np.zeros_like(h_current)
        
        for i in range(len(h_current)):
            if i + max_lag < len(h_current) and max_lag >= 0:
                grad_h[i] += -(1 - h_current[i + max_lag])
            elif i - max_lag >= 0 and max_lag < 0:
                grad_h[i] += -(1 - h_current[i - max_lag])
                
        # Normalize gradient
        grad_norm = np.linalg.norm(grad_h)
        if grad_norm > 0:
            grad_h /= grad_norm
            
        # Compute new candidate by taking step in gradient direction
        h_new = h_current - 0.01 * grad_h
        
        # Project to trust region
        diff = h_new - h_current
        diff_norm = np.linalg.norm(diff)
        if diff_norm > trust_radius:
            h_new = h_current + trust_radius * diff / diff_norm
            
        # Enforce bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Re-center to maintain sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new candidate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if beneficial
        if new_c5 < current_c5:
            h_current = h_new
            current_c5 = new_c5
            trust_radius = min(1.2 * trust_radius, 0.5)
        else:
            trust_radius = max(0.8 * trust_radius, 1e-6)
    
    # Do final refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
