import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # SDP approach with CVXPY
    # We want to minimize C5 subject to:
    #   C5 >= sum_{i} h[i] * (1 - h[i + k]) * dx for all valid k
    #   0 <= h[i] <= 1
    #   sum(h) = n_points / 2
    
    # Formulate using CVXPY with MOSEK
    h = cp.Variable(n_points, name='h')
    C5 = cp.Variable(name='C5')
    
    # Setup constraints
    constraints = []
    constraints.append(cp.sum(h) == n_points/2)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    constraints.append(C5 >= 0)

    # Add overlap constraints
    max_lag = n_points - 1
    for k in range(-max_lag, max_lag + 1):
        if k >= 0:
            valid_indices_i = list(range(0, n_points - k))
            valid_indices_j = list(range(k, n_points))
        else:
            valid_indices_i = list(range(-k, n_points))
            valid_indices_j = list(range(0, n_points + k))
        
        if len(valid_indices_i) == 0 or len(valid_indices_j) == 0:
            continue
            
        # Overlap constraint: sum h[i] (1 - h[i+k]) dx <= C5
        overlap_expr = cp.sum(cp.multiply(h[valid_indices_i], 1 - h[valid_indices_j])) * dx
        # The sign was flipped, we want to bound the correlation not make it >=
        constraints.append(overlap_expr <= C5)
    
    # Objective
    objective = cp.Minimize(C5)
    prob = cp.Problem(objective, constraints)
    
    # Attempt solve
    try:
        prob.solve(solver=cp.MOSEK, time_limit=budget_s - 10)
    except:
        pass  # Fall back to initial solution
    
    # Extract the solution
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_sol = h.value
        if h_sol is not None:
            h_sol = np.clip(h_sol, 0, 1)
            h_sol = h_sol * (n_points/2) / np.sum(h_sol)
            c5_sol = prob.value
        else:
            h_sol = h_start
            c5_sol = evaluate_erdos_solution(h_start, dx)
    else:
        h_sol = h_start
        c5_sol = evaluate_erdos_solution(h_start, dx)
            
    # Final validation
    if not (0 <= np.all(h_sol) <= 1) or not abs(np.sum(h_sol) - n_points/2) < 1e-6:
        h_sol = h_start
        c5_sol = evaluate_erdos_solution(h_start, dx)
    
    return (list(h_sol), float(c5_sol), n_points)
