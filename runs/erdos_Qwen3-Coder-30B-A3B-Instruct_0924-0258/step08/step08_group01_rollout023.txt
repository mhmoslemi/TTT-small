import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Setup
    max_scales = 6
    scale_iterations = 10
    perturbation_decay = 0.95
    perturbation_initial = 0.05
    time_budget_per_scale = 0.2 * budget_s / max_scales
    best_h = None
    best_c5 = float('inf')
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        best_h = h_start.copy()
        dx = 2.0 / n_points
        best_c5 = evaluate_erdos_solution(h_start, dx)
    else:
        n_points = 32
        h_start = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        best_h = h_start.copy()
        best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale approach
    current_h = best_h.copy()
    scale = 0
    perturbation = perturbation_initial
    
    while scale < max_scales and time() - start_time < budget_s * 0.95:
        current_time = time()
        target_points = min(2**(scale + 1), 1000)  # Cap at 1000 points
        
        # Upsample current solution
        if target_points > len(current_h):
            # Linear interpolation for upsampling
            old_x = np.linspace(0, 2, len(current_h))
            new_x = np.linspace(0, 2, target_points)
            current_h = np.interp(new_x, old_x, current_h)
            current_h = current_h * (target_points / 2) / np.sum(current_h)  # Scale to sum constraint
        
        # Refine at current scale
        for iteration in range(scale_iterations):
            if time() - current_time > time_budget_per_scale:
                break
                
            # Create perturbed version
            h_perturbed = current_h.copy()
            for i in range(len(h_perturbed)):
                h_perturbed[i] += perturbation * (2 * np.random.random() - 1)
            
            # Enforce bounds
            h_perturbed = np.clip(h_perturbed, 0, 1)
            
            # Adjust to meet sum constraint
            h_perturbed = h_perturbed * (target_points / 2) / np.sum(h_perturbed)
            
            # Evaluate
            c5_perturbed = evaluate_erdos_solution(h_perturbed, 2.0 / target_points)
            
            if c5_perturbed < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_perturbed
                current_h = h_perturbed.copy()
                perturbation = perturbation_initial  # Reset perturbation after improvement
            else:
                perturbation *= perturbation_decay
        
        # Local SQP refinement
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / target_points)
        
        def constraint_sum(h):
            return np.sum(h) - target_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(target_points)]
        
        try:
            # Refine the best solution found at this scale
            res = minimize(objective, current_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-12})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    current_h = h_refined.copy()
        except Exception:
            pass
        
        scale += 1
    
    # Final refinement
    if best_h is not None:
        try:
            n_points = len(best_h)
            dx = 2.0 / n_points
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-12})
                           
            if res.success:
                best_h = res.x
                best_c5 = objective(best_h)
        except Exception:
            pass
    
    if best_h is None:
        # Fallback to initial construction if no better solution found
        best_h = h_start
        best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (len(best_h) / 2) / np.sum(best_h)
    dx = 2.0 / len(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, len(best_h))
