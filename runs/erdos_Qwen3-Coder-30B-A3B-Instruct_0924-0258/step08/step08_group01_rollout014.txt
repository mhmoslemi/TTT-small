import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    num_generations = 10000
    crossover_rate = 0.8
    mutation_rate = 0.1
    mutation_step = 0.05
    elite_size = population_size // 2
    
    # Initialize population
    def create_individual():
        h = np.random.random(n_points) * 0.1
        h = np.clip(h, 0, 1)
        # Adjust to match sum constraint
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        return h
    
    def create_feasible_individual():
        # Start with initial solution if available, otherwise random
        if 'initial_h_values' in globals():
            h = h_start.copy()
            # Add some variation
            h += np.random.normal(0, 0.01, n_points)
        else:
            h = create_individual()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        return h
    
    population = [create_feasible_individual() for _ in range(population_size)]
    
    # GA loop
    no_improvement_count = 0
    best_in_generation = float('inf')
    
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            f = evaluate_erdos_solution(h, dx)
            fitness.append(f)
        
        # Track best in this generation
        current_best = min(fitness)
        if current_best < best_in_generation:
            best_in_generation = current_best
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early stopping if no progress
        if no_improvement_count > 200:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update global best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Selection (tournament)
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            def tournament_select():
                k = 3
                candidates = np.random.choice(population_size, size=k, replace=False)
                best_idx = candidates[np.argmin([fitness[i] for i in candidates])]
                return population[best_idx].copy()
            
            parent1 = tournament_select()
            parent2 = tournament_select()
            
            # Crossover
            if random.random() < crossover_rate:
                alpha = random.uniform(0.3, 0.7)
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
            
            # Mutation
            if random.random() < mutation_rate:
                for i in range(n_points):
                    if random.random() < 0.1:  # 10% chance to mutate each element
                        child[i] += np.random.normal(0, mutation_step)
                
                # Clamp values and adjust sum
                child = np.clip(child, 0, 1)
                sum_child = np.sum(child)
                child = child * (n_points / 2) / sum_child
            
            new_population.append(child)
        
        # Keep only population_size members
        population = new_population[:population_size]
        
        # Adapt mutation step
        if generation % 50 == 0 and mutation_step > 0.001:
            mutation_step *= 0.98
    
    # Final local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
