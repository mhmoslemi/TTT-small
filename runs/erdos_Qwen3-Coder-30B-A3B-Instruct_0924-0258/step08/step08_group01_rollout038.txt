import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    generations = 0
    no_improvement_count = 0
    max_no_improvement = 200
    mutation_rate = 0.1
    crossover_rate = 0.8
    mutation_step = 0.05
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.5
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.5
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)
    
    while time() - start_time < budget_s * 0.95:
        # Evaluate fitness
        fitness = []
        for h in population:
            f = evaluate_erdos_solution(h, dx)
            fitness.append(f)
        
        # Selection - tournament
        new_population = []
        for _ in range(population_size // 2):
            # Tournament selection
            idx1 = np.random.randint(0, population_size)
            idx2 = np.random.randint(0, population_size)
            if fitness[idx1] < fitness[idx2]:
                parent1 = population[idx1]
            else:
                parent1 = population[idx2]
                
            idx3 = np.random.randint(0, population_size)
            idx4 = np.random.randint(0, population_size)
            if fitness[idx3] < fitness[idx4]:
                parent2 = population[idx3]
            else:
                parent2 = population[idx4]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand() * 0.4 + 0.3  # 0.3 to 0.7
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, mutation_step)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, mutation_step)
            
            # Enforce constraints
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.append(child1)
            new_population.append(child2)
        
        # Keep best individuals
        all_pop = population + new_population
        all_fitness = fitness + [evaluate_erdos_solution(h, dx) for h in new_population]
        sorted_indices = np.argsort(all_fitness)
        population = [all_pop[i] for i in sorted_indices[:population_size]]
        
        # Update best solution
        current_best = population[0]
        current_c5 = evaluate_erdos_solution(current_best, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        generations += 1
        mutation_step *= 0.98
        
        # SLSQP refinement every 100 generations
        if generations % 100 == 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 1000, 'ftol': 1e-12})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        
        if no_improvement_count >= max_no_improvement:
            break
    
    return (list(best_h), best_c5, n_points)
