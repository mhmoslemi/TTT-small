import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    pop_size = 50
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_copy = h_start.copy()
        h_copy = h_copy * (n_points/2) / np.sum(h_copy)
        population.append(h_copy)
    else:
        population.append(h_start * (n_points/2) / np.sum(h_start))
    
    # Fill population with diverse solutions
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Initialize best solution
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # GA parameters
    tournament_size = 5
    mutation_rate = 0.1
    crossover_rate = 0.8
    no_improvement_limit = 100
    no_improvement_count = 0
    
    def tournament_select(pop, fitnesses):
        k = tournament_size
        tournament_indices = np.random.choice(len(pop), k)
        best_idx = tournament_indices[np.argmin([fitnesses[i] for i in tournament_indices])]
        return pop[best_idx].copy()
    
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        mask = np.random.rand(len(parent1)) > 0.5
        child1 = np.where(mask, parent1, parent2)
        child2 = np.where(mask, parent2, parent1)
        return child1, child2
    
    def mutate(h):
        h_mut = h.copy()
        # Flip mutation
        if random.random() < mutation_rate:
            i1, i2 = np.random.choice(len(h), 2, replace=False)
            h_mut[i1] = max(0, min(1, h_mut[i1] + 0.1))
            h_mut[i2] = max(0, min(1, h_mut[i2] - 0.1))
        # Perturb mutation
        if random.random() < mutation_rate:
            indices = np.random.choice(len(h), int(0.1 * len(h)), replace=False)
            for i in indices:
                h_mut[i] *= random.uniform(0.9, 1.1)
        # Normalize
        h_mut = h_mut * (n_points/2) / np.sum(h_mut)
        return h_mut
    
    generation = 0
    while time() - start_time < budget_s * 0.95:
        generation += 1
        
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= no_improvement_limit:
            break
        
        # Create new population
        new_population = [best_h.copy()]  # Elitism
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_select(population, fitnesses)
            parent2 = tournament_select(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:pop_size]
    
    # Final evaluation
    final_fitness = evaluate_erdos_solution(best_h, dx)
    if final_fitness < best_c5:
        best_c5 = final_fitness
    
    return (list(best_h), best_c5, n_points)
