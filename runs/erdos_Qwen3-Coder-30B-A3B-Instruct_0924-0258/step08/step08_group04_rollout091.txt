import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # SDP-based optimization
    h = cp.Variable(n_points, name="h")
    
    # Objective: minimize C5
    C5 = cp.Variable(name="C5")
    
    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == n_points/2, C5 >= 0]
    
    # Compute overlap constraints for all lags
    # We need to iterate through all potential lags k
    # For a shift k, we want sum_{i: i+k in range} h[i] * (1 - h[i+k]) <= C5 * dx
    for k in range(1 - n_points, n_points):
        if k >= 0:
            # h[i] * (1 - h[i+k]) for valid i's
            i_indices = np.arange(n_points - k)
            if len(i_indices) > 0:
                overlap = cp.sum(cp.multiply(h[i_indices], 1 - h[i_indices + k]))
                constraints.append(overlap <= C5 * dx)
        else:
            # h[i] * (1 - h[i+k]) for valid i's
            i_indices = np.arange(-k, n_points)
            if len(i_indices) > 0:
                overlap = cp.sum(cp.multiply(h[i_indices], 1 - h[i_indices + k]))
                constraints.append(overlap <= C5 * dx)
    
    # Create and solve problem
    objective = cp.Minimize(C5)
    prob = cp.Problem(objective, constraints)
    
    # Use MOSEK solver
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s * 0.95)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_solution = h.value
            if h_solution is not None:
                # Ensure feasibility
                h_solution = np.clip(h_solution, 0, 1)
                h_solution = h_solution * (n_points/2) / np.sum(h_solution)
                c5_bound = evaluate_erdos_solution(h_solution, dx)
                return (list(h_solution), c5_bound, n_points)
    except Exception:
        pass
    
    # Fallback to initial solution if SDP fails
    best_c5 = evaluate_erdos_solution(h_start, dx)
    return (list(h_start), best_c5, n_points)
