import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale optimization with scale doubling
    scale = 0
    max_scales = int(np.log2(max(n_points, 16))) + 1
    current_h = h_start.copy()
    current_n = n_points
    perturbation_magnitude = 0.05
    
    while scale < max_scales and (time() - start_time) < budget_s * 0.95:
        # Double the resolution
        new_n = current_n * 2
        new_h = np.zeros(new_n)
        
        # Upsample: linear interpolation
        for i in range(current_n):
            new_h[2*i] = current_h[i]
            new_h[2*i + 1] = current_h[i]
        
        # Add noise with constrained magnitude
        for i in range(new_n):
            delta = perturbation_magnitude * (2 * np.random.random() - 1)
            new_h[i] += delta
            
        # Clip and normalize
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (new_n / 2) / np.sum(new_h)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, 2.0 / new_n)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            current_h = new_h.copy()
            current_n = new_n
            perturbation_magnitude = 0.05  # reset
        else:
            perturbation_magnitude *= 0.95  # decay
            
        # Local refinement
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / new_n)
        
        def constraint_sum(h):
            return np.sum(h) - new_n / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(new_n)]
        
        try:
            res = minimize(objective, current_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-10})
            if res.success:
                refined_h = res.x
                refined_c5 = objective(refined_h)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
        except Exception:
            pass
            
        scale += 1
        
    # Final local SLSQP refinement on the best solution found
    def final_objective(h):
        return evaluate_erdos_solution(h, 2.0 / len(best_h))
    
    def final_constraint_sum(h):
        return np.sum(h) - len(best_h) / 2.0
    
    cons_sum = {'type': 'eq', 'fun': final_constraint_sum}
    bounds = [(0, 1) for _ in range(len(best_h))]
    
    try:
        res = minimize(final_objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            best_h = res.x
            best_c5 = final_objective(best_h)
    except Exception:
        pass
    
    return (list(best_h), best_c5, len(best_h))
