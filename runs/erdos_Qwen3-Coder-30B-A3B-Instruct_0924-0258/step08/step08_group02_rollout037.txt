import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Binary search with threshold-based update
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Threshold-based optimization
    threshold_levels = np.linspace(1.0, 0.01, 20)
    
    for threshold in threshold_levels:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for the maximum overlap lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Update based on threshold
        update_mask = np.abs(gradient) > threshold
        h_new = h.copy()
        
        # Apply changes with projection
        for i in range(n_points):
            if update_mask[i]:
                if gradient[i] < 0:
                    h_new[i] = min(1.0, h[i] - 0.005)
                else:
                    h_new[i] = max(0.0, h[i] + 0.005)
        
        # Enforce sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        
        # Track best solution
        current_c5 = evaluate_erdos_solution(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        h = h_new.copy()
    
    # Phase 2: L-BFGS refinement on best solution
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    # Phase 3: Project back to binary if beneficial
    if n_points <= 50:
        binary_h = np.round(best_h).clip(0, 1)
        binary_h = binary_h * (n_points/2) / np.sum(binary_h)
        binary_c5 = evaluate_erdos_solution(binary_h, dx)
        if binary_c5 < best_c5:
            best_h = binary_h
            best_c5 = binary_c5
    
    return (list(best_h), best_c5, n_points)
