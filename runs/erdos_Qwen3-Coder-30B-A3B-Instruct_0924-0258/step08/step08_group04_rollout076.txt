import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Set up genetic algorithm on coarse grid
    coarse_points = 15
    coarse_dx = 2.0 / coarse_points
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.5
    
    # Sample initial population on coarse grid
    coarse_population = []
    for i in range(pop_size):
        h = np.random.rand(coarse_points)
        h = h * (coarse_points/2) / np.sum(h)
        coarse_population.append(h)
    
    # Keep track of best coarse solution
    best_coarse_h = None
    best_coarse_c5 = float('inf')
    
    # Optimization loop for coarse grid
    coarse_budget = 60
    coarse_start = time()
    
    while time() - coarse_start < coarse_budget:
        # Evaluate population
        fitness = []
        for h in coarse_population:
            c5 = evaluate_erdos_solution(h, coarse_dx)
            fitness.append(-c5)
        
        # Selection (tournament)
        selected = []
        for i in range(pop_size):
            idx1, idx2 = np.random.choice(pop_size, 2, replace=False)
            if fitness[idx1] > fitness[idx2]:
                selected.append(coarse_population[idx1].copy())
            else:
                selected.append(coarse_population[idx2].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < pop_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Blend crossover
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(len(child1)):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                    child1[j] = max(0, min(1, child1[j]))
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.05)
                    child2[j] = max(0, min(1, child2[j]))
            
            # Rescale to sum == coarse_points/2
            child1 = child1 * (coarse_points/2) / np.sum(child1)
            child2 = child2 * (coarse_points/2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        coarse_population = new_population
        
        # Update best coarse solution
        for h in coarse_population:
            c5 = evaluate_erdos_solution(h, coarse_dx)
            if c5 < best_coarse_c5:
                best_coarse_c5 = c5
                best_coarse_h = h.copy()
    
    # Upscale best coarse to fine grid
    if best_coarse_h is not None:
        fine_h = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, coarse_points), 
                           best_coarse_h)
        # Rescale to correct sum
        fine_h = fine_h * (n_points/2) / np.sum(fine_h)
    else:
        fine_h = h_current.copy()
    
    h_current = fine_h
    
    # Fine grid optimization with gradient descent
    step_size = 0.005
    max_iter = 10000
    
    iter_count = 0
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all correlations
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Build gradient WRT to max overlap lag
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_current[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= h_current[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_current[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_current[i]
        
        # Apply gradient step with careful normalization
        h_new = h_current - step_size * gradient
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Adjust step size
        if c5_new >= evaluate_erdos_solution(h_current, dx):
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            h_current = h_new
            step_size *= 1.05
        
        iter_count += 1
    
    return (list(best_h), best_c5, n_points)
