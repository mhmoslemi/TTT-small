import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try several different strategies with restarts
    strategies = [
        "random_restart",
        "gradient_descent",
        "pattern_based"
    ]
    
    n_restarts = 5
    strategy_weights = [0.4, 0.3, 0.3]
    
    for restart in range(n_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        strategy = np.random.choice(strategies, p=strategy_weights)
        h = h_start.copy()
        h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
        
        if strategy == "random_restart":
            # Random perturbation with slight changes
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            
            # Local optimization
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(objective, h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-8})
                
                if res.success:
                    h = res.x
            except Exception:
                pass
                
        elif strategy == "gradient_descent":
            # Gradient descent with adaptive step size
            max_iter = 1000
            step_size = 0.01
            
            for iter in range(max_iter):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                corr = np.correlate(h, 1 - h, mode='full')
                max_lag_idx = np.argmax(corr)
                
                # Compute gradient for this specific lag
                grad = np.zeros(n_points)
                max_lag = max_lag_idx - (n_points - 1)
                for i in range(n_points):
                    j = i + max_lag
                    if 0 <= j < n_points:
                        grad[i] = 2 * (1 - h[j] - h[i]) / n_points
                
                h_new = h - step_size * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                
                # Check improvement
                old_val = evaluate_erdos_solution(h, dx)
                new_val = evaluate_erdos_solution(h_new, dx)
                
                if new_val < old_val:
                    h = h_new
                else:
                    step_size *= 0.9
                    if step_size < 1e-8:
                        break
                        
        elif strategy == "pattern_based":
            # Perturb based on a specific pattern
            h_pattern = h_start.copy()
            # Modify by scaling and adding noise
            scale_factor = 0.9 + 0.2 * np.random.rand()
            h_pattern = h_pattern * scale_factor
            h_pattern += np.random.normal(0, 0.02, n_points)
            h_pattern = np.clip(h_pattern, 0, 1)
            h_pattern = h_pattern * (n_points/2) / np.sum(h_pattern)
            h = h_pattern.copy()
        
        # Evaluate and update best solution
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final optimization with L-BFGS-B
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='L-BFGS-B', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 300})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 200, 'ftol': 1e-10})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
