import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution search
    resolutions = [35, 70, 140, 280]
    patience = 100
    no_improvement = 0
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to new resolution if needed
        if res > n_points:
            h = np.interp(np.linspace(0, 2, res), 
                         np.linspace(0, 2, n_points), 
                         h_start)
            n_points = res
        else:
            h = h_start.copy()
        
        dx = 2.0 / n_points
        
        # Simulated Annealing
        temp = best_c5 * 0.1
        cooling_rate = 0.95
        max_iter = 1000 if res == 35 else 500
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate neighbor by perturbing a random point
            h_new = h.copy()
            idx = random.randint(0, n_points - 1)
            delta = random.gauss(0, temp)
            h_new[idx] += delta
            h_new[idx] = np.clip(h_new[idx], 0, 1)
            
            # Maintain sum constraint
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                no_improvement = 0
            else:
                no_improvement += 1
                
            # Accept worse solution with probability
            if c5_new > best_c5:
                delta_e = c5_new - best_c5
                if random.random() > np.exp(-delta_e / temp):
                    # Reject the change
                    h_new = h.copy()
                    c5_new = evaluate_erdos_solution(h_new, dx)
            
            h = h_new.copy()
            temp *= cooling_rate
            
            if no_improvement >= patience:
                break
                
        h_start = best_h.copy()
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
