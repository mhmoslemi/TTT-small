import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    # GA parameters
    population_size = 50
    crossover_rate = 0.8
    mutation_rate = 0.1
    no_improvement_limit = 100
    refinement_interval = 100
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        n_points = len(h_start)
    else:
        n_points = 35
        population.append(np.ones(n_points) * 0.5)
    
    # Fill population with random valid individuals
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    no_improvement = 0
    
    eval_count = 0
    
    while time() - start_time < budget_s * 0.7:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Selection (tournament)
        selected = []
        for _ in range(population_size):
            i1, i2 = np.random.randint(0, population_size, 2)
            if fitness[i1] < fitness[i2]:
                selected.append(population[i1].copy())
            else:
                selected.append(population[i2].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Blend crossover with alpha = 0.5
                alpha = 0.5
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Random mutation
                idx = np.random.randint(0, n_points)
                child1[idx] = np.clip(child1[idx] + np.random.normal(0, 0.05), 0, 1)
                # Ensure sum constraint
                child1 = child1 * (n_points / 2) / np.sum(child1)
            
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(0, n_points)
                child2[idx] = np.clip(child2[idx] + np.random.normal(0, 0.05), 0, 1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        # Keep population size
        population = new_population[:population_size]
        
        # Track best
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement = 0
            else:
                no_improvement += 1

        # Periodic refinement with SLSQP
        if eval_count % refinement_interval == 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP',

                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:

    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Ensure the result meets constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
