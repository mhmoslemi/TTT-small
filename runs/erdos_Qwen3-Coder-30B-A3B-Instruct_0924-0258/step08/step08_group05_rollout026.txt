import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        h_start[::2] = 1.0
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Use a direct local optimization approach with better initialization
    max_iter = 5000
    
    # Start with a binary pattern to help avoid local minima
    h_current = h_start.copy()
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    bounds = [(0, 1) for _ in range(n_points)]

            break

        try:
            res = minimize(objective, h_current, method='SLSQP',

                h_new = res.x
                h_new = np.clip(h_new, 0, 1)

                h_new = h_new * (n_points/2) / np.sum(h_new)

                    best_h = h_new.copy()
                    best_c5 = c5_new
                h_current = h_new

                        if i - max_lag >= 0:
                            grad[i] += -1.0

                h_new = h_current - step_size * grad

            corr = np.correlate(h_current, 1 - h_current, mode='full')
