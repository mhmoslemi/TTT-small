import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Thresholding-based optimization
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Threshold parameters
    t = 0.1
    alpha = 0.95
    min_t = 0.01
    max_iterations = 500
    
    # Optimization loop
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for max overlap
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Sort indices by gradient magnitude and apply thresholding to top 50%
        sorted_indices = np.argsort(np.abs(gradient))[::-1]
        top_indices = sorted_indices[:n_points//2]
        
        # Apply thresholding
        h_new = h.copy()
        for i in range(n_points):
            if i in top_indices:
                if h[i] >= 1 - t:
                    h_new[i] = 1
                elif h[i] <= t:
                    h_new[i] = 0
                else:
                    # Keep intermediate values but adjust based on gradient
                    if gradient[i] > 0:
                        h_new[i] = min(1, h[i] + 0.1 * np.abs(gradient[i]))
                    else:
                        h_new[i] = max(0, h[i] - 0.1 * np.abs(gradient[i]))
        
        # Normalize to maintain sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Update h
        h = h_new
        
        # Track best solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Cool threshold
        t = max(min_t, t * alpha)
    
    # Coordinate-wise refinement
    refinement_iters = 100
    for _ in range(refinement_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_refined = best_h.copy()
        for i in range(n_points):
            # Fix all other coordinates, optimize h[i]
            def obj(x):
                h_test = h_refined.copy()
                h_test[i] = x
                # Re-normalize to maintain sum constraint
                h_test = h_test * (n_points/2) / np.sum(h_test)
                return evaluate_erdos_solution(h_test, dx)
            
            # Golden section search between 0 and 1
            a, b = 0, 1
            golden_ratio = 0.618
            x1 = a + (1 - golden_ratio) * (b - a)
            x2 = a + golden_ratio * (b - a)
            
            for _ in range(20):  # Fixed iterations for golden section search
                if abs(b - a) < 1e-6:
                    break
                if obj(x1) < obj(x2):
                    b = x2
                    x2 = x1
                    x1 = a + (1 - golden_ratio) * (b - a)
                else:
                    a = x1
                    x1 = x2
                    x2 = a + golden_ratio * (b - a)
            
            h_refined[i] = (a + b) / 2
            # Re-normalize
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            h_refined = np.clip(h_refined, 0, 1)
        
        current_c5 = evaluate_erdos_solution(h_refined, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_refined.copy()
    
    # Final L-BFGS-B refinement
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    # Final validation
    final_h = best_h.copy()
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_h = np.clip(final_h, 0, 1)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
