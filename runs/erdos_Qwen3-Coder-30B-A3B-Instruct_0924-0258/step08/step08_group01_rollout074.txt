import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        n_points = len(h_current)
    else:
        n_points = 35
        h_current = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Multi-scale optimization with increasing resolution
    scale = 1
    max_scales = int(np.log2(n_points)) + 2
    perturbation_magnitude = 0.05
    perturbation_decay = 0.95
    
    while scale < max_scales and time() - start_time < budget_s * 0.9:
        # Double the resolution
        new_n_points = min(2 ** scale, 2000)  # Cap at reasonable size
        if new_n_points <= n_points:
            scale += 1
            continue
            
        # Interpolate to higher resolution
        h_upsampled = np.interp(np.linspace(0, 2, new_n_points), 
                               np.linspace(0, 2, n_points), h_current)
        
        # Add perturbations
        h_new = h_upsampled + perturbation_magnitude * (2 * np.random.random(new_n_points) - 1)
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain sum constraint
        h_new = h_new * (new_n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, 2.0 / new_n_points)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            perturbation_magnitude = 0.05  # Reset
        else:
            perturbation_magnitude *= perturbation_decay
            
        # Local refinement with SQP
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / len(h))
        
        def constraint_sum(h):
            return np.sum(h) - len(h) / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(len(h_new))]
        
        try:
            res = minimize(objective, h_new, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
        
        h_current = best_h.copy()
        n_points = len(h_current)
        scale += 1
        
        if time() - start_time > budget_s * 0.95:
            break
    
    # Final refinement on best
    def objective(h):
        return evaluate_erdos_solution(h, 2.0 / len(h))
    
    def constraint_sum(h):
        return np.sum(h) - len(h) / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(len(best_h))]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, 2.0 / len(best_h))
    
    return (list(best_h), final_c5, len(best_h))
