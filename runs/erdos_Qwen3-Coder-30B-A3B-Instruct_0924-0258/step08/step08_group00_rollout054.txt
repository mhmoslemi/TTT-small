import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution simulated annealing
    resolutions = [35, 70, 140, 280]
    current_n = n_points
    h_current = h_start.copy()
    h_current = h_current * (current_n/2) / np.sum(h_current)
    
    # Parameters
    initial_temp = 1.0
    cooling_rate = 0.95
    min_temp = 1e-8
    max_iter_per_temp = 100
    patience = 200
    
    for res_idx, target_n in enumerate(resolutions):
        if target_n > current_n:
            # Upscale current h
            h_up = np.interp(np.linspace(0, 1, target_n), 
                           np.linspace(0, 1, current_n), 
                           h_current)
            h_current = h_up
            current_n = target_n
            dx = 2.0 / current_n
        else:
            # Use current h
            pass
            
        # Maintain sum constraint
        h_current = h_current * (current_n/2) / np.sum(h_current)
        
        # Simulated Annealing Parameters
        temp = initial_temp
        no_improvement = 0
        
        for iteration in range(max_iter_per_temp * 50):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb h
            h_new = h_current.copy()
            idx = np.random.randint(0, current_n)
            delta = np.random.normal(0, temp * 0.1)
            h_new[idx] += delta
            h_new[idx] = np.clip(h_new[idx], 0, 1)
            
            # Maintain sum constraint
            h_new = h_new * (current_n/2) / np.sum(h_new)
            
            old_value = evaluate_erdos_solution(h_current, dx)
            new_value = evaluate_erdos_solution(h_new, dx)
            
            if new_value < best_c5:
                best_c5 = new_value
                best_h = h_new.copy()
                no_improvement = 0
            else:
                no_improvement += 1
                
            if no_improvement >= patience:
                break
            
            # Accept or reject
            if new_value < old_value or np.random.rand() < np.exp(-(new_value - old_value) / temp):
                h_current = h_new
            
            # Cool down
            temp *= cooling_rate
            if temp < min_temp:
                temp = min_temp
                
    # Final local refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - current_n / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(current_n)]
        
        h_final = h_current.copy()
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, current_n)
