import numpy as np
from time import time
from scipy.spatial.distance import cdist

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    # Normalize initial solution to satisfy sum constraint
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    population_size = 50
    population = []
    
    # Initialize population
    population.append(h_start)
    for _ in range(population_size - 1):
        # For even n_points: create arrays with n_points/2 ones and the rest zeros
        # For odd n_points: create arrays with (n_points-1)/2 ones, one 0.5, and the rest zeros
        h = np.zeros(n_points)
        if n_points % 2 == 0:
            h[:n_points//2] = 1.0
        else:
            h[:(n_points-1)//2] = 1.0
            h[(n_points-1)//2] = 0.5
        # Randomly shuffle elements
        np.random.shuffle(h)
        # Normalize to sum constraint
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    def get_fitness(h):
        return evaluate_erdos_solution(h, dx)
        
    def tournament_selection(pop, k=5):
        selected = np.random.choice(len(pop), k)
        best_idx = selected[np.argmin([get_fitness(pop[i]) for i in selected])]
        return pop[best_idx]
        
    def crossover(parent1, parent2):
        # Uniform crossover
        mask = np.random.rand(n_points) > 0.5
        child = np.where(mask, parent1, parent2)
        return child
        
    def mutate(h, mut_rate=0.1):
        # Flip mutation
        flip_mask = np.random.rand(n_points) < mut_rate
        if np.any(flip_mask):
            indices = np.where(flip_mask)[0]
            for i in indices:
                # Randomly subtract or add 0.1, clamped to [0,1]
                change = np.random.choice([-0.1, 0.1])
                h[i] = np.clip(h[i] + change, 0, 1)
                
        # Perturbation mutation
        perturb_mask = np.random.rand(n_points) < 0.1
        if np.any(perturb_mask):
            indices = np.where(perturb_mask)[0]
            for i in indices:
                # Multiplicative perturbation
                factor = np.random.uniform(0.9, 1.1)
                h[i] = np.clip(h[i] * factor, 0, 1)
                
        return h

    # Evaluate initial population
    fitness_scores = [get_fitness(h) for h in population]
    best_idx = np.argmin(fitness_scores)
    best_solution = population[best_idx].copy()
    best_fitness = fitness_scores[best_idx]
    
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time() - start_time < budget_s * 0.95:
        generation += 1
        
        # Selection - tournament
        new_population = [best_solution]  # Elitism
        
        while len(new_population) < population_size:
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child = crossover(parent1, parent2)
            child = mutate(child)
            
            # Ensure sum constraint
            child_sum = np.sum(child)
            child = child * (n_points/2) / child_sum
            
            # Clamp to [0,1]
            child = np.clip(child, 0, 1)
            
            new_population.append(child)
            
        # Evaluate new population
        new_fitness_scores = [get_fitness(h) for h in new_population]
        
        # Update best solution
        current_best_idx = np.argmin(new_fitness_scores)
        current_best_fitness = new_fitness_scores[current_best_idx]
        
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_solution = new_population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            # If stagnated, introduce more diversity
            for _ in range(5):
                new_h = np.random.rand(n_points)
                new_h = new_h * (n_points/2) / np.sum(new_h)
                new_h = np.clip(new_h, 0, 1)
                new_population[np.random.randint(1, population_size)] = new_h
            no_improvement_count = 0
            
        population = new_population
        
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final refinement
    try:
        # Use SLSQP to refine the best solution
        from scipy.optimize import minimize
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_solution, method='SLSQP',
                      bounds=bounds, constraints=cons_sum,
                      options={'maxiter': 500, 'ftol': 1e-10})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_fitness:
                best_solution = h_final
                best_fitness = c5_final
    except:
        pass
    
    return (list(best_solution), best_fitness, n_points)
