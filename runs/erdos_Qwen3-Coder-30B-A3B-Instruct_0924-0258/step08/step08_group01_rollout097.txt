import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        n_points = len(h_base)
    else:
        n_points = 35
        h_base = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_base.copy()
    best_c5 = evaluate_erdos_solution(h_base, dx)
    
    # Multi-scale optimization
    scale = 1
    max_scales = int(np.log2(256)) + 1  # Up to 256 points
    max_iter_per_scale = 100
    
    # Scale-based parameters
    perturb_magnitude = 0.05
    perturb_decay = 0.95
    n_scales = min(max_scales, int(np.log2(256)) + 1)
    
    while scale <= n_scales and (time() - start_time) < budget_s * 0.95:
        current_n = min(2**scale, 256)
        if current_n <= n_points:
            # Reuse previous solution if smaller scale
            h_current = best_h.copy()
        else:
            # Up-sample previous solution
            h_current = np.interp(np.linspace(0, 1, current_n), 
                                np.linspace(0, 1, n_points), 
                                best_h)
        
        # Normalize to correct sum
        h_current = h_current * (current_n / 2) / np.sum(h_current)
        
        # Perturb and optimize
        for i in range(max_iter_per_scale):
            if (time() - start_time) > budget_s * 0.95:
                break
                
            # Perturb
            h_perturbed = h_current + perturb_magnitude * (np.random.random(current_n) - 0.5)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (current_n / 2) / np.sum(h_perturbed)
            
            # Evaluate
            c5_perturbed = evaluate_erdos_solution(h_perturbed, 2.0 / current_n)
            
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
                # Reset perturbation magnitude
                perturb_magnitude = 0.05
            else:
                perturb_magnitude *= perturb_decay
        
        # Local refinement
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / current_n)
        
        def constraint_sum(h):
            return np.sum(h) - current_n / 2.0
        
        # SQP refinement
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(current_n)]
            
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-12})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
        
        scale += 1
    
    # Final local refinement with more fine-grained approach
    if len(best_h) > 10:
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / len(h))
        
        def constraint_sum(h):
            return np.sum(h) - len(h) / 2.0
        
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(len(best_h))]
            
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-14})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, len(best_h))
