import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter_sa = 100000
    epsilon = 0.1
    
    # Initialize
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    current_c5 = evaluate_erdos_solution(h, dx)
    
    # Annealing loop
    T = T_init
    iter_count = 0
    
    while T > T_min and iter_count < max_iter_sa:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb two random indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-epsilon, epsilon)
        
        # Store old values and compute change
        old_i, old_j = h[i], h[j]
        new_i, new_j = old_i + delta, old_j - delta
        
        # Ensure bounds
        if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
            continue
            
        # Apply change
        h[i], h[j] = new_i, new_j
        
        # Maintain sum constraint through projection
        h_sum = np.sum(h)
        if abs(h_sum - n_points/2) > 1e-10:
            h = h * (n_points/2) / h_sum
        
        # Compute new overlap
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Accept or reject
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            h[i], h[j] = old_i, old_j
            
        iter_count += 1
        T *= alpha
    
    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return np.max(np.correlate(h, 1 - h, mode='full') * dx)
        
        # Normalize before optimization
        h_refine = best_h * (n_points/2) / np.sum(best_h)
        
        res = minimize(
            objective,
            h_refine,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            h_refine = res.x
            h_refine = h_refine * (n_points/2) / np.sum(h_refine)
            refined_c5 = evaluate_erdos_solution(h_refine, dx)
            if refined_c5 < best_c5:
                best_h = h_refine
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
