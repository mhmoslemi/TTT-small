import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a piecewise constant construction for better locality
        h_start = np.ones(n_points) * 0.5
        # Try to make a piecewise binary-like solution
        h_start[::2] = 1.0
        h_start[1::2] = 0.0
        # Ensure it's feasible
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Threshold-based optimization with adaptive cooling
    threshold = 0.1
    cooling_factor = 0.95
    
    iterations = 0
    max_iterations = 500
    while iterations < max_iterations and (time() - start_time) < budget_s * 0.95:
        # Compute overlaps
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        k_max = np.argmax(overlaps)
        
        # Compute gradient contribution for current max overlap
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient[i] = -2 * dx * (1 - best_h[j] - best_h[i])
        
        # Create a ranking based on gradient magnitude
        grad_magnitude = np.abs(gradient)
        sorted_indices = np.argsort(grad_magnitude)[::-1]  # descending order
        
        # Apply thresholding to top 50% of the gradient magnitude
        top_k = max(1, n_points // 2)
        indices_to_threshold = sorted_indices[:top_k]
        
        # Create candidate with thresholding
        h_candidate = best_h.copy()
        for i in indices_to_threshold:
            if h_candidate[i] >= 1 - threshold:
                h_candidate[i] = 1.0
            elif h_candidate[i] <= threshold:
                h_candidate[i] = 0.0
        
        # Maintain sum constraint by redistributing surplus/deficit
        current_sum = np.sum(h_candidate)
        if abs(current_sum - n_points/2) > 1e-6:
            # Find indices farthest from their targets
            diff = h_candidate - 0.5
            sorted_diff = np.argsort(np.abs(diff))
            redistribute_indices = sorted_diff[:min(10, len(sorted_diff))]
            
            # Adjust those close to 0 or 1 to accomodate sum
            total_diff = n_points/2 - current_sum
            adjustment = total_diff / max(1, len(redistribute_indices))
            for i in redistribute_indices:
                h_candidate[i] = np.clip(h_candidate[i] + adjustment, 0, 1)
        
        # Final check
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        
        # Evaluate the new candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            threshold = max(0.01, threshold * 1.05)  # Slightly increase as we improve
        else:
            threshold = max(0.01, threshold * cooling_factor)
        
        iterations += 1
    
    # Local refinement using a sweeping procedure on top coordinates
    if (time() - start_time) < budget_s * 0.95:
        # Identify top candidates by gradient analysis
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        k_max = np.argmax(overlaps)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient[i] = -2 * dx * (1 - best_h[j] - best_h[i])
        
        # Only re-optimize coordinates with largest gradient influence
        grad_magnitudes = np.abs(gradient)
        sorted_indices = np.argsort(grad_magnitudes)[::-1][:n_points//3]
        
        h_refined = best_h.copy()
        for idx in sorted_indices:
            if time() - start_time > budget_s * 0.95:
                break
                
            # Simple golden section search for this single variable
            def obj(x_new):
                h_test = h_refined.copy()
                h_test[idx] = x_new
                h_test = h_test * (n_points/2) / np.sum(h_test)
                return evaluate_erdos_solution(h_test, dx)
            
            # Golden ratio technique
            gr = (1 + np.sqrt(5)) / 2
            a, b = 0, 1
            resphi = 2 - gr
            
            x1 = a + resphi * (b - a)
            x2 = b - resphi * (b - a)
            f1 = obj(x1)
            f2 = obj(x2)
            
            while abs(b - a) > 1e-6:
                if f1 < f2:
                    b = x2
                    x2 = x1
                    f2 = f1
                    x1 = a + resphi * (b - a)
                    f1 = obj(x1)
                else:
                    a = x1
                    x1 = x2
                    f1 = f2
                    x2 = b - resphi * (b - a)
                    f2 = obj(x2)
            
            h_refined[idx] = x1
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        refined_c5 = evaluate_erdos_solution(h_refined, dx)
        if refined_c5 < best_c5:
            best_h = h_refined
            best_c5 = refined_c5
    
    # Final optimization with scipy for polishing
    if (time() - start_time) < budget_s * 0.95:
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            
            def objective(h):
                return np.max(np.correlate(h, 1 - h, mode='full') * dx)
            
            res = minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 50, 'ftol': 1e-9}
            )
            
            if res.success:
                polished_h = res.x
                polished_h = polished_h * (n_points/2) / np.sum(polished_h)
                polished_c5 = evaluate_erdos_solution(polished_h, dx)
                if polished_c5 < best_c5:
                    best_h = polished_h
                    best_c5 = polished_c5
        except Exception as e:
            pass
    
    # Final validity check
    final_sum = np.sum(best_h)
    if abs(final_sum - n_points/2) > 1e-6:
        best_h = best_h * (n_points/2) / np.sum(best_h)
    
    # Ensure all values are bounded
    best_h = np.clip(best_h, 0, 1)
    
    return (list(best_h), best_c5, n_points)
