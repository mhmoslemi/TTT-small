import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Binary search with continuous refinement
    # We'll start with some heuristic binary-like solutions
    max_iter = 500
    iter_count = 0
    
    # Try different initialization strategies
    initial_hs = []
    
    # Strategy 1: Original
    h1 = h_start.copy()
    h1 = h1 * (n_points/2) / np.sum(h1)
    initial_hs.append(h1)
    
    # Strategy 2: Thresholded version
    h2 = np.zeros(n_points)
    # Place ones in roughly half positions, spaced
    ones_positions = np.arange(0, n_points, 2)
    h2[ones_positions[:n_points//2]] = 1
    h2 = h2 * (n_points/2) / np.sum(h2)
    initial_hs.append(h2)
    
    # Strategy 3: Sorted binary
    h3 = np.zeros(n_points)
    h3[:n_points//2] = 1
    h3 = h3 * (n_points/2) / np.sum(h3)
    initial_hs.append(h3)
    
    # Strategy 4: Random binary with constraint
    h4 = np.random.rand(n_points)
    h4 = (h4 > 0.5).astype(float)
    h4 = h4 * (n_points/2) / np.sum(h4)
    initial_hs.append(h4)
    
    # Start with the best of these initializations
    best_h = min(initial_hs, key=lambda h: evaluate_erdos_solution(h, dx))
    
    # Main optimization loop with early stopping and changes
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        current_c5 = evaluate_erdos_solution(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
            
        # Adaptive gradient-based update
        # Calculate lags that maximize overlap
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        l_max = np.argmax(overlaps)
        
        # Estimate gradient more accurately
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - best_h[j] - best_h[i])
        
        # Update with adaptive step size
        step_size = 0.01 * (1.0 - (iter / max_iter))
        new_h = best_h - step_size * gradient
        
        # Project to valid range
        new_h = np.clip(new_h, 0, 1)
        
        # Enforce sum = n_points / 2 constraint
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Simple refinement if this improves things
        candidate_c5 = evaluate_erdos_solution(new_h, dx)
        if candidate_c5 < current_c5:
            best_h = new_h

    # Final local refinement with scipy
    bounds = [(0, 1) for _ in range(n_points)]
    
    def obj(h):
        return evaluate_erdos_solution(h, dx)
        
    def obj_and_grad(h):
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        l_max = np.argmax(overlaps)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + l_max) % n_points
            grad[i] = -2 * dx * (1 - h[j] - h[i])
        return np.max(overlaps), grad

    try:
        # Start from best solution
        res = minimize(
            obj,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass

    return (list(best_h), best_c5, n_points)
