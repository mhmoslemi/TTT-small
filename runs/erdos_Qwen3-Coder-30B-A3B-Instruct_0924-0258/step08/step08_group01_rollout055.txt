import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 500
    crossover_rate = 0.8
    mutation_rate = 0.1
    
    # Initialize population
    population = []
    population.append(h_start.copy())
    
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    # Track best solution
    best_individual = h_start.copy()
    best_fitness = evaluate_erdos_solution(h_start, dx)
    
    # Flag to indicate if we improved beyond 0.3807
    improvement_target = 0.3807
    max_no_improvement = 100
    no_improvement_count = 0
    
    # Genetic algorithm main loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness of entire population
        fitnesses = []
        for h in population:
            fitness = evaluate_erdos_solution(h, dx)
            fitnesses.append(fitness)
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        current_best = population[current_best_idx]
        current_best_fitness = fitnesses[current_best_idx]
        
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = current_best.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if best_fitness < improvement_target:
            break
            
        if no_improvement_count >= max_no_improvement:
            break
        
        # Tournament selection
        def tournament_select():
            k = 3
            tournament_indices = np.random.choice(len(population), k)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            return population[winner_idx].copy()
        
        # Create new population
        new_population = [best_individual.copy()]
        
        while len(new_population) < population_size:
            parent1 = tournament_select()
            parent2 = tournament_select()
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutate
            if np.random.rand() < mutation_rate:
                mutation_strength = 0.05 * (0.95 ** gen)
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child1[i] += np.random.normal(0, mutation_strength)
                # Renormalize to meet constraint
                child1 = child1 * (n_points / 2) / np.sum(child1)
                child1 = np.clip(child1, 0, 1)
            
            if np.random.rand() < mutation_rate:
                mutation_strength = 0.05 * (0.95 ** gen)
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child2[i] += np.random.normal(0, mutation_strength)
                # Renormalize to meet constraint
                child2 = child2 * (n_points / 2) / np.sum(child2)
                child2 = np.clip(child2, 0, 1)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
    
    # Final local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try refinement if we got a good solution
    if best_fitness < 0.3811 and time() - start_time < budget_s * 0.90:
        try:
            res = minimize(objective, best_individual, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_fitness:
                    best_individual = h_refined
                    best_fitness = c5_refined
        except Exception:
            pass
    
    return (list(best_individual), best_fitness, n_points)
