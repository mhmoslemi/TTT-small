import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Coarse grid optimization
    n_coarse = 15
    n_fine = 35
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.5
    max_gen = 1000
    
    # Generate initial population on coarse grid
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_coarse)
        h = h * (n_coarse/2) / np.sum(h)
        population.append(h)
    
    best_h_coarse = None
    best_c5_coarse = float('inf')
    
    # Coarse grid genetic algorithm
    for gen in range(max_gen):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, 2.0/n_coarse)
            fitness.append(-c5)
            if c5 < best_c5_coarse:
                best_c5_coarse = c5
                best_h_coarse = h.copy()
        
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            idx1, idx2 = np.random.randint(0, pop_size, 2)
            if fitness[idx1] > fitness[idx2]:
                selected.append(population[idx1])
            else:
                selected.append(population[idx2])
        
        # Crossover and Mutation
        new_population = []
        for i in range(0, pop_size, 2):
            p1 = selected[i]
            p2 = selected[i+1] if i+1 < pop_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Blend crossover
                alpha = np.random.rand()
                child1 = alpha * p1 + (1 - alpha) * p2
                child2 = (1 - alpha) * p1 + alpha * p2
            else:
                child1, child2 = p1.copy(), p2.copy()
            
            # Mutation
            for j in range(len(child1)):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                    child1[j] = np.clip(child1[j], 0, 1)
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.05)
                    child2[j] = np.clip(child2[j], 0, 1)
            
            # Renormalize
            child1 = child1 * (n_coarse/2) / np.sum(child1)
            child2 = child2 * (n_coarse/2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
    
    # Upscale best coarse solution to fine grid
    if best_h_coarse is not None:
        h_fine = np.interp(np.linspace(0, 2, n_fine), 
                          np.linspace(0, 2, n_coarse), 
                          best_h_coarse)
        h_fine = h_fine * (n_fine/2) / np.sum(h_fine)
    else:
        h_fine = np.ones(n_fine) * 0.5
        h_fine = h_fine * (n_fine/2) / np.sum(h_fine)
    
    dx_fine = 2.0 / n_fine
    h_current = h_fine.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx_fine)
    
    # Fine grid gradient optimization focusing on max overlaps
    max_iter = 10000
    iter_count = 0
    step_size = 0.01
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        
        # Get the maximum lag value
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Compute gradient for all elements that contribute to that specific overlap
        gradient = np.zeros(n_fine)
        
        if max_lag >= 0:
            for i in range(n_fine - max_lag):
                gradient[i] += 1 - h_current[i + max_lag]
                if i + max_lag < n_fine:
                    gradient[i + max_lag] -= h_current[i]
        else:
            for i in range(n_fine + max_lag):
                gradient[i] += 1 - h_current[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_current[i]
        
        # Gradient descent step
        h_new = h_current - step_size * gradient
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_fine/2) / np.sum(h_new)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx_fine)
        
        # Accept step if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        elif c5_new > evaluate_erdos_solution(h_current, dx_fine):
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            h_current = h_new
            step_size *= 1.05
            
        iter_count += 1
    
    return (list(best_h), best_c5, n_fine)
