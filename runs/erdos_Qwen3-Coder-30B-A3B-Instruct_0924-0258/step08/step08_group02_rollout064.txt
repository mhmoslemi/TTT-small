import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary optimization approach
    max_iterations = 500
    threshold_schedule = np.linspace(0.01, 0.5, 20)
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Attempt binary search approach
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for the maximum overlap lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Find positions with significant gradient
        threshold = threshold_schedule[min(iter // 10, len(threshold_schedule)-1)]
        mask = np.abs(gradient) > threshold
        
        # Update h based on gradient
        h_new = h.copy()
        for i in range(n_points):
            if mask[i]:
                if gradient[i] > 0:
                    h_new[i] = min(1.0, h[i] + 0.02)
                else:
                    h_new[i] = max(0.0, h[i] - 0.02)
        
        # Maintain sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Ensure constraints
        h_new = np.clip(h_new, 0, 1)
        
        # Track best solution
        current_c5 = evaluate_erdos_solution(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        
        h = h_new.copy()
    
    # Local refinement using L-BFGS-B on best solution
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def obj_func(h_vec):
            return np.max(np.correlate(h_vec, 1 - h_vec, mode='full') * dx)
        
        # Ensure sum constraint in refinement
        res = minimize(
            obj_func,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
