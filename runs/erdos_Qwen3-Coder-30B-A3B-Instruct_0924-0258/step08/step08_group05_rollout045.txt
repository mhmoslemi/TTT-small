import numpy as np
from scipy.optimize import minimize
from time import time
from scipy.fftpack import fft, ifft

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    max_iter = 10000
    step_size = 0.01
    tolerance = 1e-9
    no_improvement_limit = 100
    no_improvement_count = 0
    
    # Optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Fast computation using FFT for correlation
        H = fft(h_current)
        G = 1 - h_current
        # FFT of G (conjugate)
        G_fft = fft(G)
        # Compute cross-correlation in frequency domain
        correlation_fft = H * np.conj(G_fft)
        correlation = ifft(correlation_fft).real
        # Only the valid correlations
        overlap_values = correlation[:len(h_current)] * dx

        # Find the lag with maximum overlap
        max_idx = np.argmax(overlap_values)
        max_lag = max_idx - (len(h_current) - 1)
        # Compute gradient using a related method 
        gradient = np.zeros_like(h_current)
        for i in range(len(h_current)):
            if max_lag >= 0:
                if i + max_lag < len(h_current):
                    gradient[i] = -1.0
            else:
                if i - max_lag >= 0:
                    gradient[i] = -1.0
                    
        # Apply gradient with learning rate
        h_new = h_current - step_size * gradient
        
        # Project onto feasible set
        sum_h = np.sum(h_new)
        projection = (n_points / 2 - sum_h) / len(h_new)
        h_new = h_new + projection
        
        # Clamp values to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize sum
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check convergence
        change = np.sum(np.abs(h_new - h_current))
        if change < tolerance:
            step_size *= 0.8
            no_improvement_count += 1
        else:
            no_improvement_count = 0

        # Update solution
        h_current = h_new
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            no_improvement_count = 0
            
        # Early stopping
        if no_improvement_count >= no_improvement_limit:
            break
    
    # Final SLSQP optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
