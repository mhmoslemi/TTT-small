import numpy as np
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 35
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Binary search bounds
    low = 0.38
    high = 0.38085
    best_c5 = 1.0
    best_h = h_init.copy()
    
    # Precompute shift indices for efficient constraint handling
    shifts = np.arange(1, n_points)  # Only positive shifts (excluding 0)
    num_shifts = len(shifts)
    
    # Sample subset of shifts for SDP feasibility checks to reduce complexity
    sampled_shifts = np.random.choice(shifts, size=min(100, num_shifts), replace=False)
    
    while time() - start_time < budget_s * 0.95 and high - low > 1e-5:
        mid = (low + high) / 2.0
        
        # Create SDP variables
        h = cp.Variable(n_points, nonneg=True)
        Y = cp.Variable((n_points, n_points), symmetric=True)
        
        # Constraints
        constraints = [
            Y >> 0,                      # Y is positive semidefinite
            cp.diag(Y) == h,              # Diagonal of Y equals h
            cp.sum(h) == n_points / 2,    # Sum constraint
        ]
        
        # Generate overlap constraints for sampled shifts
        for k in sampled_shifts:
            # Constraint sum_{i} h[i] * (1 - h[i+k]) <= mid * dx
            # This equals sum_{i} h[i] - sum_{i} h[i] * h[i+k] <= mid * dx
            # Which is: sum_{i} h[i] - sum_{i,j} Y[i,j] * delta[i,j+k] <= mid * dx
            # Let's take: sum_{i} h[i] - sum_{i} Y[i, i+k] <= mid * dx
            left_sum = cp.sum(h)
            right_sum = cp.sum(cp.multiply(Y, np.eye(n_points)[np.arange(n_points), np.roll(np.arange(n_points), k)]))
            # Since we have j = i + k, and we're working in linear indices:
            constraint = left_sum - cp.sum(Y[np.arange(n_points), np.minimum(np.arange(n_points) + k, n_points - 1)]) <= mid * dx
            constraints.append(constraint)
            
        # Objective
        objective = cp.Minimize(0)  # dummy since we're checking feasibility
            
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=CPLEX, verbose=False, max_seconds=budget_s * 0.05)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                # The problem is feasible
                h_solution = h.value
                h_solution = np.maximum(0, np.minimum(1, h_solution))
                h_solution /= np.sum(h_solution) * 2 / n_points  # normalize
                c5_eval = evaluate_erdos_solution(h_solution, dx)
                if c5_eval < best_c5:
                    best_c5 = c5_eval
                    best_h = h_solution.copy()
                high = mid
            else:
                low = mid
        except:
            low = mid

    # Local refinement with gradient-based method
    h_refined = best_h.copy()
    dx_refined = 2.0 / n_points
    best_c5_refined = evaluate_erdos_solution(h_refined, dx_refined)

    step_size = 0.001
    max_iter = 20000
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.98:
            break
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for j in range(n_points):
            k = j + max_lag
            if 0 <= k < n_points:
                grad[j] = 1 - 2 * h_refined[j] - h_refined[k]
            else:
                grad[j] = 1 - 2 * h_refined[j]
        
        h_new = h_refined - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx_refined)
        if c5_new < best_c5_refined:
            best_c5_refined = c5_new
            best_h = h_new.copy()
            h_refined = h_new.copy()
        else:
            step_size *= 0.5
            if step_size < 1e-9:
                break

    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
