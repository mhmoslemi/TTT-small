import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Trust region method parameters
    trust_radius = 0.2
    max_iter = 5000
    improvement_threshold = 1e-6
    trust_radius_factor = 1.2
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Solve using CVXPY for trust region subproblem
        h_vars = Variable(n_points)
        C = Variable()
        constraints = [h_vars >= 0, h_vars <= 1]
        constraints.append(sum(h_vars) == n_points / 2)
        constraints.append(C >= 0)
        
        # Add overlap constraints that are likely problematic
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        lag_indices = np.arange(len(corr))
        lags = lag_indices - (len(h_current) - 1)
        overlap_values = corr * dx
        max_idx = np.argmax(overlap_values)
        
        # Add a subset of most violated constraints
        top_k = min(20, len(lags))  # restrict to most violated
        sorted_indices = np.argsort(overlap_values)[::-1][:top_k]
        
        for i in sorted_indices:
            lag = lags[i]
            if lag >= 0:
                constraint = sum(h_vars[j] * (1 - h_vars[j + lag]) for j in range(len(h_vars) - lag)) <= C
            else:
                constraint = sum(h_vars[j] * (1 - h_vars[j - lag]) for j in range(-lag, len(h_vars))) <= C
            constraints.append(constraint)
        
        prob = Problem(Minimize(C), constraints)
        
        try:
            prob.solve(solver=CPLEX, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_new_vals = np.array([float(h_vars[i].value) for i in range(n_points)])
                h_new = h_new_vals
                # Check constraint violations
                sum_violation = abs(np.sum(h_new) - n_points / 2)
                bound_violation = max(0, np.max(overlap_values) - best_c5)
                if sum_violation < 1e-4 and bound_violation < 1e-4:
                    c5_current = evaluate_erdos_solution(h_new, dx)
                    if c5_current < best_c5:
                        best_c5 = c5_current
                        best_h = h_new.copy()
                        trust_radius *= trust_radius_factor  # Expand trust region
                    else:
                        trust_radius *= 0.5  # Contract trust region
                else:
                    h_new = np.clip(h_new, 0, 1) 
                    h_new = h_new * (n_points/2) / np.sum(h_new)
                    c5_current = evaluate_erdos_solution(h_new, dx)
                    if c5_current < best_c5:
                        best_c5 = c5_current
                        best_h = h_new.copy()
        except:
            pass

    # Perform final refinement using SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
