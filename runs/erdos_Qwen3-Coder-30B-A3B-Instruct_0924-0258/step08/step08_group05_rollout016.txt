import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    population = []
    fitness = []
    
    # Add initial solution
    h_normalized = h_start * (n_points/2) / np.sum(h_start)
    population.append(h_normalized)
    fitness.append(evaluate_erdos_solution(h_normalized, dx))
    
    # Add random solutions
    for _ in range(pop_size - 1):
        # For even n_points: create arrays with n_points/2 ones and rest zeros
        # For odd n_points: create arrays with (n_points-1)/2 ones, one 0.5, rest zeros
        h = np.zeros(n_points)
        if n_points % 2 == 0:
            h[:n_points//2] = 1.0
        else:
            h[:(n_points-1)//2] = 1.0
            h[(n_points-1)//2] = 0.5
        # Shuffle and add noise
        np.random.shuffle(h)
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # GA parameters
    max_generations = 10000
    tournament_size = 5
    mutation_rate = 0.1
    
    generation = 0
    no_improvement_count = 0
    min_improvement = 1e-6
    
    while generation < max_generations and (time() - start_time) < budget_s * 0.95:
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = [best_h.copy()]
        for i in range(0, pop_size - 1, 2):
            parent1 = selected[i]
            parent2 = selected[i+1]
            
            # Uniform crossover
            child1 = np.zeros(n_points)
            child2 = np.zeros(n_points)
            for j in range(n_points):
                if np.random.random() < 0.5:
                    child1[j] = parent1[j]
                    child2[j] = parent2[j]
                else:
                    child1[j] = parent2[j]
                    child2[j] = parent1[j]
            
            # Mutation
            if np.random.random() < mutation_rate:
                idx1 = np.random.randint(0, n_points)
                idx2 = np.random.randint(0, n_points)
                child1[idx1] = np.clip(child1[idx1] + np.random.normal(0, 0.1), 0, 1)
                child1[idx2] = np.clip(child1[idx2] - np.random.normal(0, 0.1), 0, 1)
                # Normalize
                child1 = child1 * (n_points/2) / np.sum(child1)
            
            if np.random.random() < mutation_rate:
                idx1 = np.random.randint(0, n_points)
                idx2 = np.random.randint(0, n_points)
                child2[idx1] = np.clip(child2[idx1] + np.random.normal(0, 0.1), 0, 1)
                child2[idx2] = np.clip(child2[idx2] - np.random.normal(0, 0.1), 0, 1)
                # Normalize
                child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        # Evaluate new population
        new_fitness = []
        for h in new_population:
            new_fitness.append(evaluate_erdos_solution(h, dx))
        
        # Replace worst with new individuals
        all_fitness = fitness + new_fitness
        all_pop = population + new_population
        
        # Keep best individuals
        sorted_indices = np.argsort(all_fitness)
        population = [all_pop[i] for i in sorted_indices[:pop_size]]
        fitness = [all_fitness[i] for i in sorted_indices[:pop_size]]
        
        # Update global best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count > 100:
            # Increase mutation rate if stuck
            mutation_rate = min(0.5, mutation_rate * 1.1)
            no_improvement_count = 0
            
        generation += 1
        
        # Early stopping for improvement
        if abs(fitness[current_best_idx] - best_c5) < min_improvement:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
    
    return (list(best_h), best_c5, n_points)
