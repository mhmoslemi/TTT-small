import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search on C5
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    # Sample shifts for constraint checking
    shift_samples = 100
    sampled_shifts = np.random.randint(1, n_points, shift_samples)
    
    def is_feasible(c5):
        # Create optimization variables
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Overlap constraints
        for k in sampled_shifts:
            # Compute overlap for shift k
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h_var[i] * (1 - h_var[j])
            constraints.append(overlap <= c5 * dx)
            
        # Create and solve problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INACCURATE
        except:
            return False
    
    # Binary search
    while high - low > tolerance and time() - start_time < budget_s * 0.9:
        c5_test = (low + high) / 2
        if is_feasible(c5_test):
            high = c5_test
        else:
            low = c5_test
    
    # Final refinement with gradient descent if needed
    final_c5 = high
    best_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Local refinement
    max_iterations = 500
    step_size = 0.1
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient for the max lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * best_h[i] - best_h[j]
            else:
                grad[i] = 1 - 2 * best_h[i]
        
        # Gradient descent step
        h_new = best_h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to maintain constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        else:
            step_size *= 0.95
            if step_size < 1e-6:
                break
    
    final_c5 = best_c5
    return (list(best_h), final_c5, n_points)
