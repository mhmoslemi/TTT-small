import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Thresholding-based optimization
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    threshold = 0.1
    cooling_factor = 0.95
    max_iterations = 500
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient contribution for each coordinate
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Thresholding: Apply to top 50% by gradient magnitude
        indices = np.argsort(-np.abs(gradient))[:n_points//2]
        new_h = h.copy()
        
        # Apply thresholding
        for i in indices:
            if h[i] >= 1 - threshold:
                new_h[i] = 1
            elif h[i] <= threshold:
                new_h[i] = 0
        
        # Maintain sum constraint by redistribution
        current_sum = np.sum(new_h)
        diff = (n_points/2) - current_sum
        
        if abs(diff) > 1e-10:
            # Redistribute if necessary
            if diff > 0:  # Need more value
                # Find indices with small values
                small_indices = np.where(new_h < 1)[0]
                sorted_indices = small_indices[np.argsort(new_h[small_indices])]
                for i in sorted_indices:
                    if diff <= 0:
                        break
                    add_amount = min(1 - new_h[i], diff)
                    new_h[i] += add_amount
                    diff -= add_amount
            else:  # Need less value
                # Find indices with large values
                large_indices = np.where(new_h > 0)[0]
                sorted_indices = large_indices[np.argsort(-new_h[large_indices])]
                for i in sorted_indices:
                    if diff >= 0:
                        break
                    subtract_amount = min(new_h[i], -diff)
                    new_h[i] -= subtract_amount
                    diff += subtract_amount
        
        h = np.clip(new_h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Track best solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Cooling
        threshold = max(0.01, threshold * cooling_factor)
        
        if threshold < 0.01:
            break
    
    # Coordinate-wise refinement for top 30%
    h = best_h.copy()
    for refine_iter in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Refine top 30% by gradient magnitude
        indices = np.argsort(-np.abs(gradient))[:n_points//3]
        
        for i in indices:
            # Golden-section search for optimal h[i]
            def objective(h_val):
                test_h = h.copy()
                test_h[i] = h_val
                test_h = test_h * (n_points/2) / np.sum(test_h)
                overlaps = np.correlate(test_h, 1 - test_h, mode='full') * dx
                return np.max(overlaps)

            a, b = 0.0, 1.0
            phi = (1 + np.sqrt(5)) / 2
            resphi = 2 - phi
            
            # Initial points
            x1 = a + resphi * (b - a)
            x2 = b - resphi * (b - a)
            f1 = objective(x1)
            f2 = objective(x2)
            
            # Golden section iterations
            for _ in range(20):
                if abs(b - a) < 1e-8:
                    break
                if f1 < f2:
                    b = x2
                    x2 = x1
                    f2 = f1
                    x1 = a + resphi * (b - a)
                    f1 = objective(x1)
                else:
                    a = x1
                    x1 = x2
                    f1 = f2
                    x2 = b - resphi * (b - a)
                    f2 = objective(x2)
            
            # Optimal value found
            new_h_val = (a + b) / 2
            new_h_val = max(0, min(1, new_h_val))
            h[i] = new_h_val
            
        h = h * (n_points/2) / np.sum(h)
        
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final L-BFGS refinement
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
