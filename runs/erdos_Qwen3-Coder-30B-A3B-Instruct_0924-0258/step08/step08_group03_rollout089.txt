import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_fine = np.array(initial_h_values)
        n_points = len(h_fine)
    else:
        n_points = 35
        h_fine = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_fine.copy()
    best_c5 = evaluate_erdos_solution(h_fine, dx)
    
    # Coarse grid optimization (n_coarse = 15)
    n_coarse = 15
    h_coarse = np.ones(n_coarse) * 0.5
    coarse_dx = 2.0 / n_coarse
    
    # Initialize coarse h with uniform distribution
    h_coarse = np.ones(n_coarse) * 0.5
    
    # Coarse grid optimization loop
    max_coarse_iter = 500
    step_size = 0.01

    for coarse_iter in range(max_coarse_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate coarse to fine grid
        x_coarse = np.linspace(0, 2, n_coarse)
        x_fine = np.linspace(0, 2, n_points)
        f = interp1d(x_coarse, h_coarse, kind='linear', fill_value='extrapolate')
        h_interp = f(x_fine)
        h_interp = project_feasible(h_interp, n_points)
        
        # Compute overlaps
        corr = np.correlate(h_interp, 1 - h_interp, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros(n_coarse)
        for i in range(n_coarse):
            x_i = i * coarse_dx
            x_j = x_i + max_lag * dx
            j = int(x_j / dx)
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h_interp[i] - h_interp[j]
            else:
                grad[i] = 1 - 2 * h_interp[i]
        
        # Update coarse grid
        h_new_coarse = h_coarse - step_size * grad
        h_new_coarse = project_feasible(h_new_coarse, n_coarse)
        h_coarse = h_new_coarse
        
        # Check if the interpolated version improves
        x_coarse = np.linspace(0, 2, n_coarse)
        x_fine = np.linspace(0, 2, n_points)
        f = interp1d(x_coarse, h_coarse, kind='linear', fill_value='extrapolate')
        h_interp = f(x_fine)
        h_interp = project_feasible(h_interp, n_points)
        c5_new = evaluate_erdos_solution(h_interp, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            # Re-interpolate to update best_h
            x_coarse = np.linspace(0, 2, n_coarse)
            x_fine = np.linspace(0, 2, n_points)
            f = interp1d(x_coarse, h_coarse, kind='linear', fill_value='extrapolate')
            best_h = f(x_fine)
            best_h = project_feasible(best_h, n_points)
    
    # Fine grid optimization with alternating strategy
    fine_iter = 0
    max_fine_iter = 2000
    step_size_fine = 0.01
    
    h_fine = best_h.copy()
    while fine_iter < max_fine_iter and time() - start_time < budget_s * 0.95:
        # Gradient descent phase
        # Compute overlaps
        corr = np.correlate(h_fine, 1 - h_fine, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h_fine[i] - h_fine[j]
            else:
                grad[i] = 1 - 2 * h_fine[i]
        
        # Update
        h_new = h_fine - step_size_fine * grad
        h_new = project_feasible(h_new, n_points)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_fine = h_new.copy()
        else:
            # Mutation phase
            h_mutated = h_fine.copy()
            mutation_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            for idx in mutation_indices:
                h_mutated[idx] = np.random.uniform(0, 1)
            h_mutated = project_feasible(h_mutated, n_points)
            c5_mutated = evaluate_erdos_solution(h_mutated, dx)
            if c5_mutated < best_c5:
                best_c5 = c5_mutated
                best_h = h_mutated.copy()
                h_fine = h_mutated.copy()
        
        fine_iter += 1
    
    # Final check and return
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
