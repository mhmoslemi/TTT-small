import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points/2
    dx = 2.0 / n_points
    h_current = h_start * (n_points/2) / np.sum(h_start)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    max_no_improvement = 2000
    no_improvement_count = 0
    max_iter = int(5e6)  # Roughly 5 million iterations
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor by swapping small delta between two points
        i, j = np.random.randint(0, n_points, 2)
        if i == j:
            j = (j + 1) % n_points
            
        # Compute feasible delta
        max_delta = min(h_current[i], 1 - h_current[j])
        if max_delta <= 0:
            continue
            
        delta = np.random.uniform(0, max_delta)
        
        # Create candidate solution
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on SA criteria
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count >= max_no_improvement:
                # Reset with new random solution
                h_new = np.random.uniform(0, 1, n_points)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                no_improvement_count = 0
                
        # Update temperature with adaptive cooling
        if temp > min_temp:
            temp *= cooling_rate
            
        # Occasionally re-normalize to ensure constraint satisfaction
        if iteration % 100 == 0:
            h_new_sum = np.sum(h_new)
            h_new = h_new * (n_points/2) / h_new_sum
            
        # Early termination if temperature is too low
        if temp < min_temp:
            break
            
    return (list(best_h), best_c5, n_points)
