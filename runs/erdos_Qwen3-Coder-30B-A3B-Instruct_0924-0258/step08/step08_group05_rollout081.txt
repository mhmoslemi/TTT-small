import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    initial_temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    max_iter = 100000
    max_no_improve = 5000
    
    temp = initial_temp
    no_improve_count = 0
    iteration = 0
    
    while iteration < max_iter and time() - start_time < budget_s * 0.95:
        # Generate neighbor by swapping a small delta between two indices
        i, j = random.sample(range(n_points), 2)
        delta = random.uniform(0.001, 0.1) * temp
        
        # Make sure delta is feasible
        max_delta = min(h_current[i], 1 - h_current[j])
        delta = min(delta, max_delta)
        
        if delta > 0:
            h_new = h_current.copy()
            h_new[i] += delta
            h_new[j] -= delta
            
            # Re-normalize to maintain the constraint sum(h) = n_points/2
            sum_h = np.sum(h_new)
            h_new = h_new * (n_points/2) / sum_h
            
            # Evaluate new solution
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject based on SA criteria
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                no_improve_count = 0
            else:
                # Accept worse solution with probability
                if temp > min_temp:
                    if random.random() < np.exp(-(c5_new - best_c5) / temp):
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        no_improve_count = 0
                no_improve_count += 1
                
        # Cool down temperature if no improvement for too long
        if no_improve_count > max_no_improve:
            temp *= cooling_rate
            no_improve_count = 0
            
        temp = max(temp * cooling_rate, min_temp)
        iteration += 1
    
    # Final refinement with local search
    def local_search(h, step=0.01):
        h = h.copy()
        prev_c5 = evaluate_erdos_solution(h, dx)
        improved = True
        
        while improved and time() - start_time < budget_s * 0.98:
            improved = False
            best_h_trial = h.copy()
            best_c5_trial = prev_c5
            
            for i in range(n_points):
                for j in range(i+1, n_points):
                    h_trial = h.copy()
                    delta = step
                    delta = min(delta, h_trial[i], 1 - h_trial[j])
                    
                    if delta > 0:
                        h_trial[i] += delta
                        h_trial[j] -= delta
                        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
                        c5_trial = evaluate_erdos_solution(h_trial, dx)
                        
                        if c5_trial < best_c5_trial:
                            best_c5_trial = c5_trial
                            best_h_trial = h_trial.copy()
                            improved = True
            
            if improved:
                h = best_h_trial
                prev_c5 = best_c5_trial
                
        return h, prev_c5
    
    # If we have a better solution, try local improvement
    if best_c5 < evaluate_erdos_solution(h_current, dx):
        h_final, c5_final = local_search(best_h)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
