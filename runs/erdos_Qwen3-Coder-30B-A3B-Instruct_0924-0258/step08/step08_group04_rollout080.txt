import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        # Use proportional scaling
        scale_factor = (n_points/2) / current_sum
        h = h * scale_factor
        # Re-clip to ensure bounds
        h = np.clip(h, 0, 1)
        # Final adjustment to meet exact constraint
        current_sum = np.sum(h)
        if abs(current_sum - n_points/2) > 1e-10:
            diff = (n_points/2) - current_sum
            # Distribute the difference
            idx = np.argsort(h)
            # Distribute the difference by increasing values from low to high
            i = 0
            while abs(diff) > 1e-10 and i < len(h):
                available = 1.0 - h[idx[i]]
                if available > 1e-10:
                    add = min(diff, available)
                    h[idx[i]] += add
                    diff -= add
                i += 1
            # Ensure no negative values
            h = np.clip(h, 0, 1)
    return h

    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Settings
    K = 5  # Top K lags to consider
    max_iter = 50000
    step_size = 0.05
    line_search_factor = 0.5
    perturb_freq = 500
    perturb_magnitude = 0.05
    max_no_improvement = 2000
    no_improvement_count = 0
    
    # Initial setup
    h_current = project_to_feasible(h_current, n_points, dx)
    c5_prev = evaluate_erdos_solution(h_current, dx)
    
    iter_count = 0
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all correlations for all lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        
        # Get the indices of top K lags by value
        top_k_indices = np.argpartition(corr, -K)[-K:]
        top_k_lags = []
        for idx in top_k_indices:
            lag = idx - (n_points - 1)
            top_k_lags.append(lag)
        
        # Compute combined gradient for top K lags
        gradient = np.zeros(n_points)
        for lag in top_k_lags:
            if lag >= 0:
                for i in range(n_points - lag):
                    gradient[i] += 1 - h_current[i + lag]
                    if i + lag < n_points:
                        gradient[i + lag] -= h_current[i]
            else:
                for i in range(n_points + lag):
                    gradient[i] += 1 - h_current[i - lag]
                    if i - lag >= 0:
                        gradient[i - lag] -= h_current[i]
        
        # Line search
        step_size_ls = step_size
        h_new = h_current - step_size_ls * gradient
        h_new = project_to_feasible(h_new, n_points, dx)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Try decreasing step sizes if no improvement
        while c5_new >= c5_prev and step_size_ls > 1e-12:
            step_size_ls *= line_search_factor
            h_new = h_current - step_size_ls * gradient
            h_new = project_to_feasible(h_new, n_points, dx)
            c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept the step if it improves
        if c5_new < c5_prev:
            h_current = h_new
            c5_prev = c5_new
            no_improvement_count = 0
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_current.copy()
            step_size = min(1.1 * step_size, 0.5)  # Increase step size slowly
        else:
            step_size = max(0.5 * step_size, 1e-12)  # Reduce step size
            no_improvement_count += 1
        
        # Periodic perturbation to escape local minima

            h_current += perturb
            h_current = project_to_feasible(h_current, n_points, dx)
            c5_prev = evaluate_erdos_solution(h_current, dx)
            no_improvement_count = 0

        iter_count += 1
    
    return (list(best_h), best_c5, n_points)
