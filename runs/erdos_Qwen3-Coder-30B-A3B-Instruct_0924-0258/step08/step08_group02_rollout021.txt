import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Threshold-based optimization
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Hyperparameters
    threshold = 0.1
    cooling_rate = 0.95
    max_iterations = 500
    refinement_iterations = 100
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        k_max = np.argmax(overlaps)
        
        # Compute gradient contribution for each coordinate
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Rank coordinates by gradient magnitude
        idxs = np.argsort(np.abs(gradient))[::-1]
        top_indices = idxs[:max(1, n_points // 2)]
        
        # Thresholding operation
        h_new = h.copy()
        for i in top_indices:
            if h[i] >= 1 - threshold:
                h_new[i] = 1.0
            elif h[i] <= threshold:
                h_new[i] = 0.0
                
        # Redistribution to maintain sum constraint
        current_sum = np.sum(h_new)
        deficit = n_points / 2 - current_sum
        
        # Find indices that need adjustment
        if abs(deficit) > 1e-10:
            if deficit > 0:  # Need to increase some values
                # Sort indices by how much they're currently below 1
                below_ones = np.where(h_new < 1)[0]
                diffs = [1.0 - h_new[i] for i in below_ones]
                sorted_diffs = sorted(zip(diffs, below_ones), key=lambda x: x[0])
                for diff, i in sorted_diffs:
                    if deficit <= 0:
                        break
                    add = min(diff, deficit)
                    h_new[i] += add
                    deficit -= add
            else:  # Need to decrease some values
                # Sort indices by how much they're currently above 0
                above_zeros = np.where(h_new > 0)[0]
                diffs = [h_new[i] for i in above_zeros]
                sorted_diffs = sorted(zip(diffs, above_zeros), key=lambda x: x[0])
                for diff, i in sorted_diffs:
                    if deficit >= 0:
                        break
                    subtract = min(diff, -deficit)
                    h_new[i] -= subtract
                    deficit += subtract
        
        # Apply the changes and ensure feasible values
        h = np.clip(h_new, 0, 1)
        # Re-normalize
        h = h * (n_points/2) / np.sum(h)
        
        # Evaluate and track best
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Cool threshold
        threshold = max(0.01, threshold * cooling_rate)
    
    # Coordinate-wise refinement
    for iter in range(refinement_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Optimizing each coordinate individually using golden section search
        h_refined = best_h.copy()
        for i in range(n_points):
            def objective(h_i):
                h_test = h_refined.copy()
                h_test[i] = h_i
                # Normalize to maintain sum constraint
                h_test = h_test * (n_points/2) / np.sum(h_test)
                return evaluate_erdos_solution(h_test, dx)
            
            # Golden section search bounds: [0, 1]
            def golden_section_search():
                phi = (1 + np.sqrt(5)) / 2
                resphi = 2 - phi
                
                a, b = 0.0, 1.0
                # Initial points
                x1 = a + resphi * (b - a)
                x2 = b - resphi * (b - a)
                
                f1 = objective(x1)
                f2 = objective(x2)
                
                while abs(b - a) > 1e-6:
                    if f1 < f2:
                        b = x2
                        x2 = x1
                        f2 = f1
                        x1 = a + resphi * (b - a)
                        f1 = objective(x1)
                    else:
                        a = x1
                        x1 = x2
                        f1 = f2
                        x2 = b - resphi * (b - a)
                        f2 = objective(x2)
                return (a + b) / 2
            
            # Update the coordinate
            new_val = golden_section_search()
            h_refined[i] = new_val
            
        # Re-normalize and re-evaluate
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        new_c5 = evaluate_erdos_solution(h_refined, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_refined.copy()
    
    # Final refinement using L-BFGS-B if time allows
    try:
        if time() - start_time < budget_s * 0.90:
            def objective_and_gradient(h):
                overlaps = np.correlate(h, 1 - h, mode='full') * dx
                max_overlap = np.max(overlaps)
                k_max = np.argmax(overlaps)
                gradient = np.zeros(n_points)
                for i in range(n_points):
                    j = (i + k_max) % n_points
                    gradient[i] = -2 * dx * (1 - h[j] - h[i])
                return max_overlap, gradient
            
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 50, 'ftol': 1e-9}
            )
            
            if res.success:
                refined_h = res.x
                refined_h = refined_h * (n_points/2) / np.sum(refined_h)
                refined_c5 = evaluate_erdos_solution(refined_h, dx)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
