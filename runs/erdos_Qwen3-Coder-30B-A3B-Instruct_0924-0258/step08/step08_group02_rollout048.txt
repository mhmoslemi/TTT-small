import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Threshold-based optimization with adaptive parameters
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    threshold = 0.1
    cooling_rate = 0.95
    max_iterations = 500
    threshold_min = 0.01
    
    # Thresholding loop with coordinate refinement
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for current maximum overlap lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Rank by gradient magnitude and threshold top 50%
        sorted_indices = np.argsort(np.abs(gradient))[::-1]
        top_indices = sorted_indices[:max(1, n_points // 2)]
        
        # Thresholding operation
        h_new = h.copy()
        for i in range(n_points):
            if i in top_indices:
                if h[i] >= 1 - threshold:
                    h_new[i] = 1
                elif h[i] <= threshold:
                    h_new[i] = 0
        
        # Ensure sum constraint after thresholding
        current_sum = np.sum(h_new)
        if abs(current_sum - n_points/2) > 1e-6:
            # Distribute adjustment to most impacted coordinates
            adjustment = (n_points/2 - current_sum) / n_points
            for i in range(n_points):
                if i in top_indices:
                    h_new[i] = np.clip(h_new[i] + adjustment, 0, 1)
        
        h = h_new
        
        # Track best solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Cooling
        threshold = max(threshold_min, threshold * cooling_rate)
        
        # Coordinate-wise refinement (small part of iteration)
        if iter % 3 == 0 and iter > 10:
            # Single coordinate optimization
            for i in range(n_points):
                def obj(x):
                    h_test = h.copy()
                    h_test[i] = x
                    h_test = h_test * (n_points/2) / np.sum(h_test)
                    return evaluate_erdos_solution(h_test, dx)
                
                # Golden section search to optimize h[i] in [0,1]
                a, b = 0.0, 1.0
                res = minimize(lambda x: obj(np.clip(x, 0, 1)), 
                               (a+b)/2, 
                               method='Nelder-Mead', 
                               options={'maxiter': 10})
                if res.success:
                    h[i] = np.clip(res.x[0], 0, 1)
            
            # Ensure sum constraint
            h = h * (n_points/2) / np.sum(h)
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Final L-BFGS refinement
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def obj(h):
            return np.max(np.correlate(h, 1 - h, mode='full') * dx)
        
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 50, 'ftol': 1e-9})
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
