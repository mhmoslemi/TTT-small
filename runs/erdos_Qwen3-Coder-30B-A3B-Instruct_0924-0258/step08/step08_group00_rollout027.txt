import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach with annealing
    resolutions = [35, 70, 140, 280]
    current_n_points = n_points
    h_current = h_start.copy()
    
    patience = 100
    no_improvement = 0
    
    # Cooling schedule parameters
    initial_temp = 0.1
    cooling_rate = 0.95
    min_temp = 1e-6
    
    for res_idx, target_n in enumerate(resolutions):
        if res_idx > 0:
            # Interpolate to higher resolution
            h_current = np.interp(np.linspace(0, 2, target_n), 
                                np.linspace(0, 2, current_n_points), h_current)
            current_n_points = target_n
            dx = 2.0 / current_n_points
            
        # Local optimization
        h_current = h_current * (current_n_points/2) / np.sum(h_current)
        temp = initial_temp
        iterations = 0
        max_iter = 1000 if target_n < 100 else 500
        
        while temp > min_temp and iterations < max_iter:
            if time() - start_time > budget_s * 0.95:
                break
                
            # Try a random perturbation
            h_new = h_current.copy()
            perturb_idx = np.random.randint(current_n_points)
            perturb_amount = np.random.normal(0, temp)
            h_new[perturb_idx] += perturb_amount
            
            # Project back to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Fix sum constraint
            h_new = h_new * (current_n_points/2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            c5_current = evaluate_erdos_solution(h_current, dx)
            
            # Accept or reject based on Metropolis criterion
            delta = c5_new - c5_current
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h_current = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_current.copy()
                    no_improvement = 0
                else:
                    no_improvement += 1
            else:
                no_improvement += 1
                
            temp *= cooling_rate
            iterations += 1
            
            if no_improvement >= patience:
                break
                
        # Local refinement
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - current_n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(current_n_points)]
            
            res = minimize(objective, h_current, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, current_n_points)
