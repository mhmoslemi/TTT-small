import numpy as np
from scipy.optimize import minimize
from time import time
from scipy.fft import fft, ifft, fftfreq

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # FFT-based optimization
    max_iter = 5000
    step_size = 0.01
    tolerance = 1e-9
    no_improvement_count = 0
    no_improvement_limit = 100
    
    # Precompute FFT of 1-h
    G = 1 - h_current
    G_fft = fft(G)
    
    # First pass: gradient-based with FFT
    for iteration in range(max_iter):
        current_time = time()
        if current_time - start_time > budget_s * 0.85:
            break
            
        # Compute overlap using FFT, then get max overlap lag
        H = fft(h_current)
        corr_fft = H * np.conj(G_fft)
        overlap_values = ifft(corr_fft).real * dx
        
        # Since we're looking for max, we're interested in the full correlation
        max_idx = np.argmax(overlap_values)
        # The actual lag is max_idx - (n_points - 1)
        max_lag = max_idx - (n_points - 1)
        
        # Compute gradient of the overlap w.r.t. h for this specific lag
        # Since overlap at lag k corresponds to convolution h * (1-h) at k,
        # and the derivative of h[k] in this context amounts to finding influence
        # from the (1-h) term at lag k
        gradient = np.zeros_like(h_current)
        for i in range(n_points):
            if 0 <= i + max_lag < n_points:
                # Gradient is -1 at position where the lagged 1-h contributes to overlap
                gradient[i] = -1.0
            elif 0 <= i - max_lag < n_points:
                # derivative of 1-h at this point is -1
                gradient[i] = -1.0
                
        # Project onto the constraint (sum = n_points / 2)
        sum_h = np.sum(h_current)
        projection = (n_points / 2 - sum_h) / n_points
        
        # Update h
        h_new = h_current - step_size * gradient + step_size * projection
        
        # Ensure bounds [0, 1] and re-normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Track performance
        c5_current = evaluate_erdos_solution(h_new, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Reduce learning rate if no improvement for a while  
        if no_improvement_count > 50:
            step_size *= 0.8
            
        h_current = h_new
        
    # Final refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
