import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex perturbation optimization
    max_iter = 5000
    epsilon_0 = 0.001
    beta = 0.99
    top_k = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        # Get indices of top k largest overlaps
        top_k_indices = np.argsort(corr)[-top_k:]
        
        # Formulate convex problem
        h_var = cp.Variable(n_points)
        objective = 0
        
        # Sum over top_k dominant shifts
        for k_idx in top_k_indices:
            shift = k_idx - (n_points - 1)
            # Manual computation of shifted terms
            shifted_h = np.roll(h, shift)
            obj_term = cp.sum(cp.multiply(h_var, 1 - shifted_h))
            objective += obj_term
            
        objective = cp.Minimize(objective + epsilon_0 * (beta ** iteration) * cp.sum_squares(h_var))
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
            else:
                h_new = h.copy()
        except:
            h_new = h.copy()
            
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        
        # Early stopping if stagnation
        if iteration > 0 and iteration % 500 == 0:
            prev_c5 = evaluate_erdos_solution(h, dx)
            if abs(prev_c5 - c5_new) < 1e-8:
                break
    
    # Final local refinement
    best_c5 = evaluate_erdos_solution(best_h, dx)
    for _ in range(100):
        if time() - start_time > budget_s * 0.98:
            break
        h_pert = best_h + np.random.normal(0, 0.001, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        c5_test = evaluate_erdos_solution(h_pert, dx)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_pert.copy()
    
    return (list(best_h), best_c5, n_points)
