```python
import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a better starting construction
        h_start = np.zeros(n_points)
        mid = n_points // 2
        h_start[:mid] = 1.0
        h_start[mid:] = 0.0
        h_start = h_start * (n_points/2) / np.sum(h_start)
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Trust region optimization approach
    max_iter = 5000
    trust_radius = 0.2
    improvement_threshold = 1e-8
    no_improvement_count = 0
    max_no_improvement = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Refine with convex relaxation approach
        try:
            # Create a CVXPY optimization problem
            h_vars = cp.Variable(n_points)
            C = cp.Variable()
            
            # Set bounds
            constraints = [h_vars >= 0, h_vars <= 1]
            
            # Sum constraint
            constraints.append(cp.sum(h_vars) == n_points / 2)
            
            # Overlap constraints: for each lag k, enforce overlap bound
            corr_values = np.correlate(h_current, 1 - h_current, mode='full')
            lags = np.arange(len(corr_values)) - (len(h_current) - 1)
            
            for i, (lag, corr_val) in enumerate(zip(lags, corr_values)):
                if abs(lag) < len(h_current):
                    # Define overlap constraint for this lag
                    # The overlap integral for lag k is sum_i h[i] * (1 - h[i+k])
                    # We linearize it over current solution for constraint
                    if lag >= 0:
                        overlap_expr = cp.sum(cp.multiply(h_vars[:len(h_current)-lag], 
                                                        1 - h_vars[lag:]))
                    else:
                        overlap_expr = cp.sum(cp.multiply(h_vars[-lag:], 
                                                        1 - h_vars[:len(h_current)+lag]))
                    constraints.append(overlap_expr <= C)
            
            # Objective: minimize C
            objective = cp.Minimize(C)
            prob = cp.Problem(objective, constraints)
            
            # Solve with ECOS solver
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_vars.value
                h_new = np.clip(h_new, 0, 1)
                # Re-normalize
                h_new = h_new * (n_points/2) / np.sum(h_new)
                
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    trust_radius = min(0.5, trust_radius * 1.2)
                    no_improvement_count = 0
                else:
                    trust_radius = max(1e-6, trust_radius * 0.8)
                    no_improvement_count += 1

            else:
                # If optimization fails, fall back
                h_new = h_current.copy()
                trust_radius = max(1e-6, trust_radius * 0.8)
            
            h_current = h_new
            
        except Exception as e:
            # Backup to simple gradient descent
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            lag_indices = np.arange(len(corr))
            lags = lag_indices - (len(h_current) - 1)
            overlap_values = corr * dx
            max_idx = np.argmax(overlap_values)
            max_lag = lags[max_idx]

            gradient = np.zeros_like(h_current)
            for i in range(len(h_current)):
                if max_lag >= 0:
                    if i + max_lag < len(h_current):
                        gradient[i] = -1.0
                else:
                    if i - max_lag >= 0:
                        gradient[i] = -1.0

            h_new = h_current - 0.005 * gradient
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)

            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            else:
                no_improvement_count += 1
            
            h_current = h_new
        
        if no_improvement_count > max_no_improvement:
            break
    
    # Final SLSQP refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h.copy(), method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-14})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
```
