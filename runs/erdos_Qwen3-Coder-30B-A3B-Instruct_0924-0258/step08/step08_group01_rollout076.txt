import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm Parameters
    population_size = 50
    generations = 2000
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 5
    stagnation_limit = 50
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    if 'initial_h_values' in globals():
        population = [h_start.copy()]
        for _ in range(population_size - 1):
            population.append(create_individual())
    else:
        population = [create_individual() for _ in range(population_size)]
    
    # Evaluate initial population
    fitnesses = [evaluate_erdos_solution(ind, dx) for ind in population]
    best_idx = np.argmin(fitnesses)
    if fitnesses[best_idx] < best_c5:
        best_c5 = fitnesses[best_idx]
        best_h = population[best_idx].copy()
    
    no_improvement = 0
    generation = 0
    
    while time() - start_time < budget_s * 0.95 and no_improvement < stagnation_limit:
        # Tournament selection
        def tournament_select():
            k = 3
            idxs = np.random.choice(population_size, k)
            best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
            return population[best_idx].copy()
        
        # Crossover (blend)
        def blend_crossover(parent1, parent2, alpha=0.5):
            child1 = alpha * parent1 + (1 - alpha) * parent2
            child2 = (1 - alpha) * parent1 + alpha * parent2
            return np.clip(child1, 0, 1), np.clip(child2, 0, 1)
        
        # Mutation
        def mutate(individual, mutation_rate, generation, max_gen):
            mutation_strength = 0.05 * (1 - generation / max_gen)
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        individual[i] += np.random.normal(0, mutation_strength)
                # Renormalize
                individual = individual * (n_points / 2) / np.sum(individual)
                individual = np.clip(individual, 0, 1)
            return individual
    
        # Create offspring
        offspring = []
        while len(offspring) < population_size - elite_size:
            parent1 = tournament_select()
            parent2 = tournament_select()
            
            if np.random.rand() < crossover_rate:
                child1, child2 = blend_crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            child1 = mutate(child1, mutation_rate, generation, generations)
            child2 = mutate(child2, mutation_rate, generation, generations)
            
            offspring.extend([child1, child2])
        
        offspring = offspring[:population_size - elite_size]
        
        # Keep elite
        elite_indices = np.argsort(fitnesses)[:elite_size]
        new_population = [population[i].copy() for i in elite_indices]
        new_population.extend(offspring)
        
        # Evaluate new population
        new_fitnesses = [evaluate_erdos_solution(ind, dx) for ind in new_population]
        
        # Update best solution
        min_idx = np.argmin(new_fitnesses)
        if new_fitnesses[min_idx] < best_c5:
            best_c5 = new_fitnesses[min_idx]
            best_h = new_population[min_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        population = new_population
        fitnesses = new_fitnesses
        generation += 1

    # Final local refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
