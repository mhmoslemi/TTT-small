import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Start with uniform distribution as fallback
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    temp_init = 1.0
    temp_min = 1e-5
    alpha = 0.99
    max_anneal_iter = 100000
    
    # Initialize temperature and best solution
    T = temp_init
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)  # Ensure sum constraint
    current_c5 = evaluate_erdos_solution(h_current, dx)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_current.copy()
    
    # SA iterations
    iter_count = 0
    for i in range(max_anneal_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        iter_count += 1
        # Generate perturbation (change two random points)
        i1 = np.random.randint(0, n_points)
        i2 = np.random.randint(0, n_points)
        
        # Create candidate solution
        h_candidate = h_current.copy()
        delta = np.random.uniform(-0.1, 0.1)
        h_candidate[i1] += delta
        h_candidate[i2] -= delta
        
        # Ensure bounds and sum constraint
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5:
            h_current = h_candidate
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
        else:
            # Accept with probability based on temperature
            delta_c5 = candidate_c5 - current_c5
            if np.random.uniform() < np.exp(-delta_c5 / T):
                h_current = h_candidate
                current_c5 = candidate_c5
        
        # Update temperature
        T *= alpha
        if T < temp_min:
            break
            
    # Local refinement using L-BFGS-B
    try:
        # Funnel optimization function
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return -np.max(corr) * dx  # Negative because we minimize
            
        # Gradient function
        def objective_grad(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            
            grad = np.zeros_like(h)
            for i in range(len(h)):
                j = (i + max_idx) % len(h)
                grad[i] = -2 * dx * (1 - h[j] - h[i])
            
            return grad
            
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimize using L-BFGS-B
        res = minimize(
            lambda x: -np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            jac=lambda x: -objective_grad(x),
            options={'maxiter': 100}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
