import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary initialization
    h_binary = np.zeros(n_points)
    h_binary[:n_points//2] = 1.0
    np.random.shuffle(h_binary)
    
    # Threshold-based optimization
    threshold_range = np.linspace(1.0, 0.01, 50)
    
    h = h_binary.copy()
    h = h * (n_points/2) / np.sum(h)
    
    for threshold in threshold_range:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute pseudo-gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Update step using threshold
        update_mask = np.abs(gradient) > threshold
        h_update = h.copy()
        
        # Update based on gradient sign
        h_update[update_mask] = np.clip(h_update[update_mask] - 0.01 * gradient[update_mask], 0, 1)
        
        # Project to sum constraint
        h_update = h_update * (n_points/2) / np.sum(h_update)
        
        # Check if this improves the solution
        current_c5 = evaluate_erdos_solution(h_update, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_update.copy()
        
        # Update h for next iteration
        h = h_update
    
    # Final refinement with scipy
    try:
        # Define the objective
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)
        
        # Define bounds for each h_i
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Refine using L-BFGS-B
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
