import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convex relaxation approach
    max_iter = 200
    no_improve_iter = 0
    max_no_improve = 50
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Regularization parameters
    lambda_reg = 0.1
    alpha_reg = 0.01
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convex relaxation step
        try:
            # Create a relaxed version of the objective using CVXPY
            h_var = cp.Variable(n_points)
            constraints = [
                cp.sum(h_var) == n_points/2,
                h_var >= 0,
                h_var <= 1
            ]
            
            # Quadratic surrogate for overlaps
            objective = cp.Minimize(
                lambda_reg * cp.sum_squares(cp.convolution(h_var, 1 - h_var, method='direct')) +
                alpha_reg * cp.sum_squares(h_var)
            )
            
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_relaxed = h_var.value
                h = h_relaxed / np.sum(h_relaxed) * n_points/2
        except:
            pass
        
        # Coordinate descent refinement
        # Fix 80% of variables, optimize 20%
        indices = np.random.permutation(n_points)
        num_optimize = max(1, n_points // 5)  # 20% of variables
        optimize_indices = indices[:num_optimize]
        fixed_indices = indices[num_optimize:]
        
        # Optimizing the selected variables using gradient descent
        for _ in range(50):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute current overlaps
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            max_overlap_idx = np.argmax(overlaps)
            
            # Compute gradients for selected indices
            grad = np.zeros(n_points)
            for i in optimize_indices:
                j = (i + max_overlap_idx) % n_points
                grad[i] = -2 * dx * (1 - h[j] - h[i])
            
            # Update selected variables while ensuring constraints
            step_size = 0.01
            h_new = h - step_size * grad
            
            h_new = np.clip(h_new, 0, 1)
            
            # Project back to sum constraint
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Update with sufficient decrease
            current_c5 = evaluate_erdos_solution(h_new, dx)
            if current_c5 < best_c5:
                h = h_new.copy()
                best_c5 = current_c5
                no_improve_iter = 0
            else:
                no_improve_iter += 1
                
            if no_improve_iter >= max_no_improve:
                break
    
    # Final local refinement with scipy
    try:
        from scipy.optimize import minimize
        
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if res.success:
            h_final = res.x
            h_final = h_final * (n_points/2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
