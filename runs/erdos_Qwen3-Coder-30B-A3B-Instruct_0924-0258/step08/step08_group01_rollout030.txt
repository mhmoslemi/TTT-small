import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 2
    max_no_improvement = 100
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    if 'initial_h_values' in globals():
        population = [h_start.copy()]
        for _ in range(population_size - 1):
            population.append(create_individual())
    else:
        population = [create_individual() for _ in range(population_size)]
    
    # Fitness evaluation
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    no_improvement_count = 0
    
    # Main GA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.7:
            break
            
        # Evaluate fitness
        fitnesses = [fitness(ind) for ind in population]
        
        # Track best individual
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = population[best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
        
        # Tournament selection
        def tournament_select():
            k = 3
            tournament_indices = np.random.choice(len(population), k)
            tournament_fitness = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            return population[winner_idx].copy()
        
        # Create new population
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_select()
            parent2 = tournament_select()
            
            if random.random() < crossover_rate:
                # Blend crossover
                alpha = random.random()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
                
                # Normalize to satisfy constraint
                child1 = child1 * (n_points / 2) / np.sum(child1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
                
                # Clip to [0,1]
                child1 = np.clip(child1, 0, 1)
                child2 = np.clip(child2, 0, 1)
                
                new_population.extend([child1, child2])
            else:
                new_population.extend([parent1.copy(), parent2.copy()])
        
        # Mutation
        mutation_strength = 0.1 * (0.95 ** gen)
        for i in range(elite_size, len(new_population)):
            if random.random() < mutation_rate:
                # Gaussian mutation
                noise = np.random.normal(0, mutation_strength, n_points)
                new_population[i] += noise
                # Ensure constraint ∑h = n_points/2
                new_population[i] = new_population[i] * (n_points / 2) / np.sum(new_population[i])
                # Clip to [0,1]
                new_population[i] = np.clip(new_population[i], 0, 1)
        
        population = new_population[:population_size]
    
    # Local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
