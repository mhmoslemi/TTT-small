import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach with simulated annealing
    resolutions = [35, 70, 140, 280]
    current_resolution = n_points
    h = h_start.copy()
    h = h * (current_resolution/2) / np.sum(h)  # Ensure sum constraint
    
    # Simulated Annealing parameters
    initial_temp = 0.1
    cooling_rate = 0.95
    min_temp = 1e-6
    max_iter_per_temp = 1000
    
    # Iteration constants
    max_iter = 5000
    patience = 100
    
    no_improvement = 0
    temp = initial_temp
    
    def interpolate_h(h_old, new_n):
        x_old = np.linspace(0, 2, len(h_old))
        x_new = np.linspace(0, 2, new_n)
        h_new = np.interp(x_new, x_old, h_old)
        return np.clip(h_new, 0, 1)
    
    def project_h(h, n):
        h = np.clip(h, 0, 1)
        h = h * (n/2) / np.sum(h)
        return h
    
    def get_current_resolution():
        return len(h)
    
    for res_idx, target_n in enumerate(resolutions):
        if target_n <= get_current_resolution():
            continue
            
        # Interpolate to higher resolution
        h = interpolate_h(h, target_n)
        current_resolution = target_n
        dx = 2.0 / current_resolution
        
        # Reset cooling
        temp = initial_temp
        no_improvement = 0
        
        for iteration in range(max_iter_per_temp):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb randomly
            h_new = h.copy()
            idx = np.random.randint(0, len(h_new))
            delta = np.random.normal(0, temp * 0.1)
            h_new[idx] += delta
            
            # Project feasible
            h_new = project_h(h_new, current_resolution)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                no_improvement = 0
            else:
                if np.random.rand() < np.exp(-(c5_new - best_c5) / max(temp, 1e-10)):
                    h = h_new
                no_improvement += 1
            
            if no_improvement >= patience:
                break
            
            # Cooling
            temp *= cooling_rate
            if temp < min_temp:
                break
    
    # Final local optimization with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - current_resolution / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(current_resolution)]
        
        res = minimize(objective, h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, len(best_h))
