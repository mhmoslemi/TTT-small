import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    population_size = 50
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 2
    max_generations = 1000
    stagnation_limit = 50
    refine_every = 100
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    if fitnesses[best_idx] < best_c5:
        best_c5 = fitnesses[best_idx]
        best_h = population[best_idx].copy()
    
    no_improvement = 0
    generation = 0
    
    while time() - start_time < budget_s * 0.7 and generation < max_generations:
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Check for improvement
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement >= stagnation_limit:
            break
            
        # Select parents
        def tournament_selection():
            k = 3
            candidates = random.sample(range(population_size), k)
            winner = min(candidates, key=lambda i: fitnesses[i])
            return population[winner]
        
        # Create new population
        new_population = [population[0].copy(), population[1].copy()]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if random.random() < crossover_rate:
                parent1 = tournament_selection()
                parent2 = tournament_selection()
                
                # Blend crossover
                alpha = random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                parent = tournament_selection()
                child = parent.copy()
            
            # Mutation
            if random.random() < mutation_rate:
                # Apply mutation uniformly
                mutate_idx = random.randint(0, n_points - 1)
                delta = (random.random() - 0.5) * 0.1
                child[mutate_idx] += delta
                child[mutate_idx] = np.clip(child[mutate_idx], 0, 1)
                
                # Rescale to maintain constraint
                child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        # Evaluate new population
        fitnesses = [evaluate_erdos_solution(h, dx) for h in new_population]
        population = new_population
        
        # Periodic refinement
        if (generation + 1) % refine_every == 0:
            # Refine best solution
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-10})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        
        generation += 1
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
