import numpy as np
from scipy.optimize import minimize
from time import time
from scipy.fft import fft, ifft

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Use FFT for faster correlation computation
    def compute_overlap_fft(h):
        # Compute FFT of h and (1-h)
        H = fft(h)
        G = fft(1 - h)
        # Cross-correlation in frequency domain
        corr = ifft(H * np.conj(G)).real
        # Return maximum overlap value scaled by dx
        return np.max(corr) * dx
    
    # Gradient-based optimization with FFT
    max_iter = 10000
    step_size = 0.01
    tolerance = 1e-9
    no_improvement_limit = 100
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap using FFT
        c5_current = compute_overlap_fft(h_current)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            
        # Identify maximum overlap lag
        H = fft(h_current)
        G = fft(1 - h_current)
        corr = ifft(H * np.conj(G)).real
        max_idx = np.argmax(corr)
        max_lag = max_idx - (len(h_current) - 1)
        
        # Compute gradient for max lag
        gradient = np.zeros_like(h_current)
        for i in range(len(h_current)):
            if max_lag >= 0:
                if i + max_lag < len(h_current):
                    gradient[i] = -1.0
            else:
                if i - max_lag >= 0:
                    gradient[i] = -1.0
        
        # Update step
        h_new = h_current - step_size * gradient
        
        # Projection onto sum constraint
        sum_h = np.sum(h_new)
        projection = (n_points / 2 - sum_h) / len(h_new)
        h_new = h_new + step_size * projection
        
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to correct sum
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Convergence check
        change = np.sum(np.abs(h_new - h_current))
        if change < tolerance:
            step_size *= 0.5
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            
        if no_improvement_count >= no_improvement_limit:
            step_size *= 0.5
            no_improvement_count = 0
            
        h_current = h_new
    
    # Final refinement using SLSQP
    def objective(h):
        return compute_overlap_fft(h)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = compute_overlap_fft(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
