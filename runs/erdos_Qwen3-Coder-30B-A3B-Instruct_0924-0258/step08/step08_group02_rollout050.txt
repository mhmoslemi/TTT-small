import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter_sa = int(1e5)
    
    # Initialize
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    T = T_init
    current_c5 = evaluate_erdos_solution(h, dx)
    
    iter_count = 0
    stagnation_count = 0
    stagnation_threshold = 1000
    
    while T > T_min and (time() - start_time) < budget_s * 0.9:
        if iter_count >= max_iter_sa:
            break
            
        # Make a perturbation
        i, j = np.random.choice(n_points, 2, replace=False)
        epsilon = np.random.uniform(-0.1, 0.1)
        
        # Adjust h[i] and h[j]
        old_h_i = h[i]
        old_h_j = h[j]
        
        h[i] = np.clip(h[i] + epsilon, 0, 1)
        h[j] = np.clip(h[j] - epsilon, 0, 1)
        
        # Maintain sum constraint
        sum_diff = np.sum(h) - n_points / 2
        if abs(sum_diff) > 1e-10:
            # Distribute sum error to some random points
            indices = np.random.choice(n_points, size=min(3, n_points), replace=False)
            for idx in indices:
                h[idx] = np.clip(h[idx] - sum_diff / len(indices), 0, 1)
        
        # Ensure sum is still correct
        h = h * (n_points/2) / np.sum(h)
        
        # Compute new overlap
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            # Accept with probability
            delta = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                current_c5 = new_c5
            else:
                # Revert changes
                h[i] = old_h_i
                h[j] = old_h_j
        
        # Cool down
        T *= alpha
        iter_count += 1
        
        if stagnation_count > stagnation_threshold:
            T = T_min  # Force early termination if stuck
            
        # Adaptive temperature
        if iter_count % 100 == 0:
            T = max(T_min, T * alpha)  # Prevent too fast cooling
    
    # Local refinement with L-BFGS-B
    try:
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)

        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
