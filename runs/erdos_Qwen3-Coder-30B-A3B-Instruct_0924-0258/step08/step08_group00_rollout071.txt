import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def repair(h, n_points):
    h = np.clip(h, 0, 1)
    if np.sum(h) == 0:
        return h
    h = h * (n_points/2) / np.sum(h)
    return h

    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-start approach with Latin hypercube sampling
    n_starts = 50
    all_h = []
    
    # Add the initial point if it exists
    if 'initial_h_values' in globals():
        h_init = repair(h_start.copy(), n_points)
        all_h.append(h_init)
    
    # Generate Latin hypercube samples
    for _ in range(n_starts - len(all_h)):
        h = np.random.rand(n_points)
        h = repair(h, n_points)
        all_h.append(h)
    
    # Optimize each starting point
    for i, h0 in enumerate(all_h):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Define objective
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        # Define constraint
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h0, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_opt = res.x
                h_opt = repair(h_opt, n_points)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
