import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Parameters
    max_iterations = 500
    k_max = 20  # Number of top overlaps to consider in convex optimization
    filter_bandwidth = 5  # Moving average filter bandwidth
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convex relaxation with top K overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        top_k_indices = np.argsort(corr)[-k_max:][::-1]
        top_k_lags = [i - (n_points - 1) for i in top_k_indices]
        
        # Define variables
        h_var = cp.Variable(n_points)
        
        # Objective: weighted sum of selected overlaps
        objective = cp.Minimize(cp.sum([cp.multiply(h_var[i], 1 - h_var[i + lag]) 
                                       for i in range(n_points) for lag in top_k_lags if 0 <= i+lag < n_points]))
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Solve convex relaxation
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=1000, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_new = h_var.value
            else:
                h_new = h.copy()
        except:
            h_new = h.copy()
        
        # Apply low-pass filter
        h_filtered = np.zeros(n_points)
        for i in range(n_points):
            start_idx = max(0, i - filter_bandwidth)
            end_idx = min(n_points, i + filter_bandwidth + 1)
            h_filtered[i] = np.mean(h_new[start_idx:end_idx])
        
        # Renormalize
        h_new = h_filtered * (n_points / 2) / np.sum(h_filtered)
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Early stopping logic
            if iteration > 10 and iteration % 10 == 0:
                # Gradually reduce filter bandwidth
                filter_bandwidth = max(1, filter_bandwidth - 1)
                
                # Increase K for more overlaps in convex optimization
                k_max = min(50, k_max + 1)
        
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
