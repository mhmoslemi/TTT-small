import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach with simulated annealing
    resolutions = [35, 70, 140, 280]
    cooling_rate = 0.95
    min_temp = 1e-6
    max_iter_per_temp = 500
    
    for res_idx, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to new resolution
        if res_idx == 0:
            h = h_start.copy()
        else:
            old_points = len(h)
            old_x = np.linspace(0, 2, old_points)
            new_x = np.linspace(0, 2, n_points)
            h = np.interp(new_x, old_x, h)
        
        # Ensure sum constraint
        h = h * (n_points/2) / np.sum(h)
        
        # Annealing parameters
        temp = max(1.0, best_c5)
        patience = 0
        patience_limit = 20
        
        # Local optimization with SLSQP before SA
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200})
            
            if res.success:
                h = res.x
                current_c5 = objective(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        except Exception:
            pass
        
        for iter_count in range(max_iter_per_temp * 5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Random perturbation
            h_new = h.copy()
            idx = random.randint(0, n_points - 1)
            perturb_mag = temp * 0.1
            h_new[idx] += np.random.normal(0, perturb_mag)
            
            # Project back to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Adjust to satisfy sum constraint
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate new solution
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject based on SA criteria
            delta = new_c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h = h_new.copy()
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
            else:
                patience += 1
                
            # Reduce temperature
            temp = max(min_temp, temp * cooling_rate)
            
            if patience >= patience_limit:
                break
        
        dx = 2.0 / n_points

    # Final polishing with local optimizer
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 300})
        
        if res.success:
            final_h = res.x
            final_c5 = objective(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
