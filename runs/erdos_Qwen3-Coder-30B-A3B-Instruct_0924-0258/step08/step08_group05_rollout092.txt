import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    population_size = 50
    max_generations = 10000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_ref = np.array(initial_h_values)
        h_ref = h_ref * (n_points/2) / np.sum(h_ref)
        population.append(h_ref.copy())
    else:
        h_ref = np.ones(n_points) * 0.5
        h_ref = h_ref * (n_points/2) / np.sum(h_ref)
        population.append(h_ref.copy())
        
    # Fill rest of population
    for i in range(population_size - 1):
        if n_points % 2 == 0:
            h = np.zeros(n_points)
            h[:n_points//2] = 1.0
        else:
            h = np.zeros(n_points)
            h[:(n_points-1)//2] = 1.0
            h[(n_points-1)//2] = 0.5
        # Randomly shuffle
        np.random.shuffle(h)
        # Normalize
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # GA Main Loop
    no_improvement_count = 0
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, 5, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_index].copy())
        
        # Crossover and mutation
        new_population = [best_h.copy()]  # Elitism
        
        for i in range(0, population_size - 1, 2):
            parent1 = selected[i]
            parent2 = selected[i+1]
            
            # Uniform crossover
            mask = np.random.rand(n_points) > 0.5
            child1 = np.where(mask, parent1, parent2)
            child2 = np.where(mask, parent2, parent1)
            
            # Mutation
            # Flip mutation
            if np.random.rand() < 0.3:
                idx1, idx2 = np.random.choice(n_points, 2, replace=False)
                child1[idx1] = np.clip(child1[idx1] + np.random.choice([-0.1, 0.1]), 0, 1)
                child1[idx2] = np.clip(child1[idx2] + np.random.choice([-0.1, 0.1]), 0, 1)
                
            # Perturbation mutation
            if np.random.rand() < 0.3:
                indices = np.random.choice(n_points, int(0.1 * n_points), replace=False)
                for idx in indices:
                    factor = np.random.uniform(0.9, 1.1)
                    child1[idx] = np.clip(child1[idx] * factor, 0, 1)
            
            # Normalize to maintain sum
            sum1 = np.sum(child1)
            child1 = child1 * (n_points/2) / sum1
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        # Evaluate new population
        new_fitness = []
        for h in new_population:
            f = evaluate_erdos_solution(h, dx)
            new_fitness.append(f)
            
        # Replace worst with new individuals (keep best in population)
        combined_population = list(zip(new_population, new_fitness))
        combined_population.sort(key=lambda x: x[1])
        population = [x[0] for x in combined_population[:population_size]]
        fitness = [x[1] for x in combined_population[:population_size]]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 100:
            break
            
    # Final SLSQP refinement if possible
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons_sum, 
                      options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
