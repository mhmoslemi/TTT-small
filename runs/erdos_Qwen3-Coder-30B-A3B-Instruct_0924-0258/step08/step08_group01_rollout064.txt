import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    # Genetic Algorithm parameters
    population_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    mutation_step = 0.05
    elite_size = population_size // 2
    
    # Initialize population
    def create_individual():
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def create_feasible_individual():
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Create population
    population = [h_start.copy()] if 'initial_h_values' in globals() else []
    
    while len(population) < population_size:
        population.append(create_feasible_individual())
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    def fitness(individual):
        return evaluate_erdos_solution(individual, dx)
    
    # Genetic Algorithm loop
    no_improvement = 0
    for generation in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness of the entire population
        fitness_scores = [fitness(ind) for ind in population]
        
        # Track best individual
        best_idx = np.argmin(fitness_scores)
        if fitness_scores[best_idx] < best_c5:
            best_c5 = fitness_scores[best_idx]
            best_h = population[best_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > 200:
            break
            
        # Create next generation
        sorted_indices = np.argsort(fitness_scores)
        new_population = [population[i] for i in sorted_indices[:elite_size]]
        
        # Tournament selection with replacement
        while len(new_population) < population_size:
            # Tournament selection
            tournament_size = 3
            selected_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in selected_indices]
            winner_idx = selected_indices[np.argmin(tournament_fitness)]
            
            # Create offspring
            if np.random.rand() < crossover_rate:
                # Select second parent
                selected_indices2 = np.random.choice(population_size, tournament_size)
                tournament_fitness2 = [fitness_scores[i] for i in selected_indices2]
                parent2_idx = selected_indices2[np.argmin(tournament_fitness2)]
                
                # Blend crossover
                alpha = np.random.uniform(0.3, 0.7)
                offspring = alpha * population[winner_idx] + (1 - alpha) * population[parent2_idx]
            else:
                offspring = population[winner_idx].copy()

                    offspring[i] += np.random.normal(0, mutation_step)

                offspring = offspring * (n_points / 2) / np.sum(offspring)

        if generation % 100 == 0:

                return evaluate_erdos_solution(h, dx)

                return np.sum(h) - n_points / 2.0

        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP',

                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
