import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter = 100000
    
    # Initialize temperature
    T = T_init
    
    # Ensure sum constraint
    h = h_start * (n_points / 2) / np.sum(h_start)
    
    # Simulated Annealing loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Make a random perturbation
        h_new = h.copy()
        i, j = random.sample(range(n_points), 2)
        delta = random.uniform(-0.1, 0.1)
        
        # Ensure bounds
        h_new[i] = max(0, min(1, h_new[i] + delta))
        h_new[j] = max(0, min(1, h_new[j] - delta))
        
        # Maintain sum constraint (adjust other elements)
        sum_diff = np.sum(h_new) - n_points / 2
        if abs(sum_diff) > 1e-10:
            # Distribute adjustment equally among all elements except i,j
            adjustment = sum_diff / (n_points - 2)
            for k in range(n_points):
                if k != i and k != j:
                    h_new[k] -= adjustment
        
        # Ensure all elements are within bounds again
        h_new = np.clip(h_new, 0, 1)
        
        # Re-adjust to maintain sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        delta_c5 = c5_new - best_c5
        if delta_c5 < 0 or random.random() < np.exp(-delta_c5 / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        
        # Cool down
        T = max(T_min, T * alpha)
    
    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    # Final check and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
