import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def get_initial_h():
    if 'initial_h_values' in globals():
        return np.array(initial_h_values)
    else:
        # Initialize with 35 points
        n = 35
        h = np.ones(n) * 0.5
        # Force some structure to potentially reduce overlap
        h[:10] = 0.0
        h[10:25] = 1.0
        h[25:] = 0.0
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    # Get initial h
    h_start = get_initial_h()
    n_points = len(h_start)
    dx = 2.0 / n_points

    # Normalize to correct sum
    h_current = h_start * (n_points / 2) / np.sum(h_start)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)

    # Convex relaxation approach
    def solve_convex_relaxation(h_init):
        # Define variables
        h = cp.Variable(n_points)
        C = cp.Variable()
        
        # For each lag, add constraint
        constraints = []
        constraints.append(cp.sum(h) == n_points / 2)
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        constraints.append(C >= 0)
        
        # Find all unique lags that could contribute
        max_lag = n_points - 1
        lags = list(range(-max_lag, max_lag + 1))
        
        # Use binary search approach to identify critical lags
        # but use all lags for safety
        for k in lags:
            if k >= 0:
                # h[i] * (1 - h[i+k])
                # For each i from 0 to n_points - k - 1
                expr = 0
                for i in range(n_points - k):
                    expr += h[i] - h[i] * h[i + k]  # Simplified using identity
                constraints.append(expr <= C)
            else:
                # h[i+(-k)] * (1 - h[i])
                expr = 0
                for i in range(n_points + k):
                    expr += h[i - k] - h[i - k] * h[i]
                constraints.append(expr <= C)
        
        # Solve relaxation
        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h.value
        except:
            pass
        return None

    # Try convex relaxation as initialization
    if time() - start_time < budget_s * 0.7:
        try:
            h_relaxed = solve_convex_relaxation(h_current)
            if h_relaxed is not None:
                h_current = h_relaxed
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points/2) / np.sum(h_current)
                best_c5 = evaluate_erdos_solution(h_current, dx)
                best_h = h_current.copy()
        except:
            pass

    # Local refinement using SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    # Run SLSQP optimization
    try:
        res = minimize(objective, h_current, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)

    return (list(best_h), final_c5, n_points)
