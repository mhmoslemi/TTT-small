import numpy as np
from time import time
from cvxpy import *
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search parameters
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    # Sample shifts for constraint checking
    shift_samples = np.random.choice(range(1, n_points), size=min(100, n_points-1), replace=False)
    
    # Binary search over C5
    while time() - start_time < budget_s * 0.9 and high - low > tolerance:
        mid = (low + high) / 2.0
        
        # Formulate feasibility problem
        h_var = Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            sum(h_var) == n_points / 2
        ]
        
        # Add overlap constraints for sampled shifts
        for k in shift_samples:
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h_var[i] * (1 - h_var[j])
            constraints.append(overlap <= mid * dx)
        
        # Create and solve problem
        objective = Minimize(0)  # feasibility problem
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_candidate = h_var.value
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
                high = mid
            else:
                low = mid
        except:
            low = mid
    
    # Local refinement with gradient descent
    h_refined = best_h.copy()
    step_size = 0.1
    max_iter = 500
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap gradients
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute finite difference gradient
        grad = np.zeros(n_points)
        eps = 1e-6
        for j in range(n_points):
            h_test = h_refined.copy()
            h_test[j] += eps
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            corr_test = np.correlate(h_test, 1 - h_test, mode='full')
            grad[j] = (np.max(corr_test) * dx - np.max(corr) * dx) / eps
        
        # Gradient descent
        h_new = h_refined - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check improvement
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_refined = h_new.copy()
        else:
            step_size *= 0.95
    
    return (list(best_h), best_c5, n_points)
