import numpy as np
from time import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h onto feasible set: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to meet sum constraint
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def line_search(h_current, grad, n_points, dx, max_alpha=1.0):
    """Find optimal step size using line search"""
    def objective(alpha):
        h_new = h_current - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_erdos_solution(h_new, dx)
    
    # Try different alpha values
    alphas = [max_alpha * (0.5 ** i) for i in range(10)]
    best_alpha = 0.0
    best_val = evaluate_erdos_solution(h_current, dx)
    
    for alpha in alphas:
        try:
            val = objective(alpha)
            if val < best_val:
                best_val = val
                best_alpha = alpha
        except:
            continue
    return best_alpha

    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Parameters
    max_iter = 5000
    iter_count = 0

    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all correlations and find top lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        corr_indices = np.argsort(corr)[::-1][:5]  # Top 5 lags
        
        # Build gradient using top lags
        gradient = np.zeros(n_points)

                    gradient[i] += 1 - h_current[i + lag]
                    if i + lag < n_points:

            else:
                for i in range(n_points + lag):
                    gradient[i] += 1 - h_current[i - lag]
                    if i - lag >= 0:
                        gradient[i - lag] -= h_current[i]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 1e-10:

        # Line search for step size
        alpha = line_search(h_current, gradient, n_points, dx, step_size)
        
        # Update

        h_new = project_h(h_new, n_points)
