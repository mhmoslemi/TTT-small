import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Cool down approach: mixed strategies
    for iteration in range(1000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try different approaches
        # 1. Gradient descent with adaptive step
        h = best_h.copy()
        h = h * (n_points/2) / np.sum(h)
        
        # Evaluate lag with max overlap
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient approx.
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 2 * (1 - h[j] - h[i]) / n_points
        
        step_size = 0.01
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        # 2. Random perturbation
        if time() - start_time > budget_s * 0.8:
            h_pert = h_start.copy()
            for i in range(n_points):
                if np.random.random() < 0.05:
                    h_pert[i] = np.random.random()
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            c5_pert = evaluate_erdos_solution(h_pert, dx)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
        
        # 3. Simple local search
        h_local = best_h.copy()
        for i in range(n_points):
            orig = h_local[i]
            h_local[i] = min(1, max(0, orig + (np.random.random() - 0.5) * 0.1))
            h_local = h_local * (n_points/2) / np.sum(h_local)
            c5_local = evaluate_erdos_solution(h_local, dx)
            if c5_local < best_c5:
                best_c5 = c5_local
                best_h = h_local.copy()
            else:
                h_local[i] = orig
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 300, 'ftol': 1e-10})
        
        if res.success:
            h_final = res.x
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
