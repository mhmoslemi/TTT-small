import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_onto_feasible(h, n_points):
    """Project h onto feasible set with constraints 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to bounds
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def line_search(h_current, grad, dx, max_iter=20):
    """Perform line search to find optimal step size"""
    alpha = 1.0
    step_size = 0.5
    current_c5 = evaluate_erdos_solution(h_current, dx)
    
    for _ in range(max_iter):
        h_new = h_current - alpha * grad
        h_new = project_onto_feasible(h_new, len(h_current))
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < current_c5:
            return alpha, h_new
        alpha *= step_size
    return alpha, h_current

    no_improve_count = 0
    perturb_freq = 500

        corr = np.correlate(h_current, 1 - h_current, mode='full')

        top_k = 5
        lag_indices = np.argpartition(-corr, top_k)[:top_k]
        lag_indices = lag_indices[np.argsort(-corr[lag_indices])]

        # Combined gradient estimation
        gradient = np.zeros(n_points)

            for lag in lag_indices:
                if lag < n_points:
                    j = i + lag
                    if j < n_points:
                        gradient[i] += (1 - h_current[j])
                        gradient[j] -= h_current[i]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm
            
        # Line search for optimal step
        alpha, h_new = line_search(h_current, gradient, dx)
        
        # Update h
        h_current = h_new

        c5_new = evaluate_erdos_solution(h_current, dx)

            best_c5 = c5_new
            best_h = h_current.copy()
            no_improve_count = 0

        if no_improve_count > perturb_freq:
            noise = np.random.normal(0, 0.01, n_points)
            h_current = h_current + noise
            h_current = project_onto_feasible(h_current, n_points)
            no_improve_count = 0
