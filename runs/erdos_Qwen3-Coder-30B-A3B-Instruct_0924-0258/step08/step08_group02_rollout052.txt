import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter_sa = 100000
    epsilon = 0.1
    
    # Initialize
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    current_c5 = evaluate_erdos_solution(h, dx)
    
    T = T_init
    iter_count = 0
    
    # Simulated Annealing loop
    while T > T_min and iter_count < max_iter_sa and (time() - start_time) < budget_s * 0.95:
        # Perturb two random indices
        i, j = np.random.randint(0, n_points, 2)
        delta = np.random.uniform(-epsilon, epsilon)
        
        # Calculate change in sum and bound violations
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds
        h_new[i] = np.clip(h_new[i], 0, 1)
        h_new[j] = np.clip(h_new[j], 0, 1)
        
        # Adjust to maintain sum constraint exactly
        diff = np.sum(h_new) - n_points / 2
        if abs(diff) > 1e-10:
            h_new[0] -= diff
        
        h_new = np.clip(h_new, 0, 1)
        
        # New overlap
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                h = h_new.copy()
                current_c5 = new_c5
        
        # Cooling
        T *= alpha
        iter_count += 1
    
    # Local refinement with L-BFGS-B
    try:
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Renormalize to meet constraint
        h_refine = best_h * (n_points/2) / np.sum(best_h)
        
        res = minimize(
            objective,
            h_refine,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            h_refine = res.x
            h_refine = h_refine * (n_points/2) / np.sum(h_refine)
            refined_c5 = evaluate_erdos_solution(h_refine, dx)
            if refined_c5 < best_c5:
                best_h = h_refine
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    # Ensure final constraint satisfaction
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
