import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale optimization
    max_scales = int(np.log2(n_points)) + 2
    max_iter_per_scale = 100
    perturbation_decay = 0.95
    initial_perturbation = 0.05
    
    h_current = h_start.copy()
    current_n = n_points
    perturbation_magnitude = initial_perturbation
    
    scale = 0
    while scale < max_scales and (time() - start_time) < budget_s * 0.95:
        # Double the resolution
        new_n = min(current_n * 2, 1000)  # cap at 1000 points
        if new_n <= current_n:
            break
            
        # Upsample the current solution
        h_upsampled = np.interp(np.linspace(0, 2, new_n), 
                               np.linspace(0, 2, current_n), 
                               h_current)
        
        # Perturb 
        h_perturbed = h_upsampled + perturbation_magnitude * (np.random.random(new_n) - 0.5)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (new_n / 2) / np.sum(h_perturbed)
        
        # Evaluate
        c5_perturbed = evaluate_erdos_solution(h_perturbed, 2.0 / new_n)
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
            perturbation_magnitude = initial_perturbation
        else:
            perturbation_magnitude *= perturbation_decay
            
        # Local SQP refinement
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / new_n)
            
        def constraint_sum(h):
            return np.sum(h) - new_n / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(new_n)]
        
        try:
            res = minimize(objective, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-12})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except Exception:
            pass
            
        # Prepare for next scale
        h_current = best_h.copy()
        current_n = new_n
        scale += 1
    
    # Final pass with clean optimization
    if (time() - start_time) < budget_s * 0.95:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, 2.0 / len(best_h))
                
            def constraint_sum(h):
                return np.sum(h) - len(best_h) / 2.0
                
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(len(best_h))]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
            
    return (list(best_h), best_c5, len(best_h))
