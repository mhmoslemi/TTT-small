import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def binary_to_h(binary_string, n_points):
    """Convert binary string to step function h."""
    h = np.zeros(n_points)
    interval_length = n_points // len(binary_string)
    
    for i, bit in enumerate(binary_string):
        start = i * interval_length
        end = min((i + 1) * interval_length, n_points)
        if bit == 1:
            h[start:end] = 1.0
            
    return h

def h_to_binary(h, n_points, threshold=0.5):
    """Convert h to a binary representation."""
    binary = []
    interval_length = max(1, n_points // 10)
    for i in range(0, n_points, interval_length):
        end = min(i + interval_length, n_points)
        avg = np.mean(h[i:end])
        binary.append(1 if avg > threshold else 0)
    return binary

def mutate_binary(binary_string, mutation_rate=0.05):
    """Mutate binary string."""
    new_string = binary_string.copy()
    for i in range(len(new_string)):
        if random.random() < mutation_rate:
            new_string[i] = 1 - new_string[i]
    return new_string

def crossover_binary(parent1, parent2):
    """Uniform crossover."""
    child = []
    for i in range(len(parent1)):
        if random.random() < 0.5:
            child.append(parent1[i])
        else:
            child.append(parent2[i])
    return child

        # Evaluate all individuals
        fitness = []

            c5 = evaluate_erdos_solution(h, dx)

        sorted_indices = np.argsort(fitness)
