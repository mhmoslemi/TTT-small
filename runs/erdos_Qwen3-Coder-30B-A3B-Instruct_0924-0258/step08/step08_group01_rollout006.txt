import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 0
    no_improvement = 0
    max_no_improvement = 100
    
    # Initialize population
    population = []
    population.append(h_start.copy())
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # Fix sum
        h = np.clip(h, 0, 1)  # Clip to [0,1]
        population.append(h)
    
    while time() - start_time < budget_s * 0.7:
        # Evaluate population
        fitness = []
        for h in population:
            f = evaluate_erdos_solution(h, dx)
            fitness.append(f)
        
        # Track best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= max_no_improvement:
            break
            
        # Selection (tournament)
        new_population = []
        for _ in range(population_size):
            # Tournament selection
            tourn_size = 3
            tourn_indices = np.random.choice(population_size, tourn_size)
            tourn_fitness = [fitness[i] for i in tourn_indices]
            winner_idx = tourn_indices[np.argmin(tourn_fitness)]
            new_population.append(population[winner_idx].copy())
        
        # Crossover and mutation
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = new_population[i]
            parent2 = new_population[(i+1) % population_size]
            
            if np.random.rand() < crossover_rate:
                # Blend crossover (BLX-0)
                alpha = 0.5
                for j in range(n_points):
                    a = min(parent1[j], parent2[j])
                    b = max(parent1[j], parent2[j])
                    range_ = b - a
                    lower = a - alpha * range_
                    upper = b + alpha * range_
                    offspring1 = np.random.uniform(lower, upper)
                    offspring2 = np.random.uniform(lower, upper)
                    parent1[j] = np.clip(offspring1, 0, 1)
                    parent2[j] = np.clip(offspring2, 0, 1)
            
            offspring.append(parent1)
            offspring.append(parent2)
            
        # Mutation
        for h in offspring:
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    step = 0.05 * (1 - 2 * np.random.rand())  # Small random step
                    h[j] += step
            # Fix sum constraint
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
        
        # Replace population
        population = offspring
        generations += 1
        
        # Periodic SLSQP refinement of best
        if generations % 20 == 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 300, 'ftol': 1e-10})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final result check
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
