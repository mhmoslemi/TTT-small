import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Coarse optimization parameters
    coarse_n = 15
    coarse_dx = 2.0 / coarse_n
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.5
    max_coarse_iter = 500
    coarse_budget = 60
    
    # Genetic algorithm on coarse grid
    if time() - start_time < coarse_budget:
        coarse_h = np.linspace(0, 1, coarse_n)
        population = [coarse_h.copy() for _ in range(pop_size)]
        for i in range(pop_size):
            population[i] = np.random.rand(coarse_n)
            population[i] = population[i] * (coarse_n/2) / np.sum(population[i])
        
        for iter_count in range(max_coarse_iter):
            if time() - start_time > coarse_budget:
                break
                
            # Evaluate fitness
            fitness = []
            for h in population:
                c5 = evaluate_erdos_solution(h, coarse_dx)
                fitness.append(-c5)
            
            # Selection (tournament)
            selected = []
            for _ in range(pop_size):
                idx1, idx2 = np.random.randint(0, pop_size, 2)
                if fitness[idx1] > fitness[idx2]:
                    selected.append(population[idx1].copy())
                else:
                    selected.append(population[idx2].copy())
            
            # Crossover and mutation
            new_population = []
            for i in range(0, pop_size, 2):
                p1 = selected[i]
                p2 = selected[i+1] if i+1 < pop_size else selected[0]
                
                if np.random.rand() < crossover_rate:
                    # Blend crossover
                    alpha = np.random.rand()
                    child1 = alpha * p1 + (1 - alpha) * p2
                    child2 = (1 - alpha) * p1 + alpha * p2
                else:
                    child1, child2 = p1.copy(), p2.copy()
                
                # Mutation
                for j in range(coarse_n):
                    if np.random.rand() < mutation_rate:
                        child1[j] += np.random.normal(0, 0.1)
                        child2[j] += np.random.normal(0, 0.1)
                
                # Keep in bounds
                child1 = np.clip(child1, 0, 1)
                child2 = np.clip(child2, 0, 1)
                
                # Rescale
                child1 = child1 * (coarse_n/2) / np.sum(child1)
                child2 = child2 * (coarse_n/2) / np.sum(child2)
                
                new_population.extend([child1, child2])
            
            # Keep only pop_size
            population = new_population[:pop_size]
        
        # Find best in population for upsampling
        best_coarse = population[0]
        best_fitness = evaluate_erdos_solution(best_coarse, coarse_dx)
        for h in population:
            c5 = evaluate_erdos_solution(h, coarse_dx)
            if c5 < best_fitness:
                best_fitness = c5
                best_coarse = h.copy()
        
        # Upscale to fine grid (35 points)
        h_current = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, coarse_n), best_coarse)
        h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Gradient optimization for fine grid
    step_size = 0.005
    max_iter = 10000
    iter_count = 0
    patience = 0
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Get correlations
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_current[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= h_current[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_current[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_current[i]
        
        h_new = h_current - step_size * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
            last_improvement = iter_count
        else:
            patience += 1
            if patience > 200:
                step_size *= 0.5
                patience = 0
                if step_size < 1e-10:
                    break
        
        h_current = h_new
        iter_count += 1
    
    return (list(best_h), best_c5, n_points)
