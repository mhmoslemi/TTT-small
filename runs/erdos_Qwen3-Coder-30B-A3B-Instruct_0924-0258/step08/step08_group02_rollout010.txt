import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a binary-like structure but with small perturbations
        h_start = np.zeros(n_points)
        half = n_points // 2
        h_start[:half] = 1.0
        # Random shuffle to get a diverse start
        np.random.shuffle(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-phase optimization
    # Phase 1: Gradient-based with adaptive step
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    
    step_size = 0.05
    max_iterations = 300
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for the maximum overlap lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Update using gradient
        h = h - step_size * gradient
        
        # Enforce constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Track best
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            step_size = min(0.05, step_size * 1.2)
        else:
            step_size = max(1e-6, step_size * 0.8)
    
    # Phase 2: Local refinement with scipy
    try:
        from scipy.optimize import minimize
        
        def objective(h):
            return np.max(np.correlate(h, 1 - h, mode='full') * dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        # Use scipy minimize with L-BFGS-B
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100}
        )
        
        if res.success and res.fun < best_c5:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    # Phase 3: Random restarts around best solution to avoid local minima
    for _ in range(5):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb best solution slightly
        perturbed_h = best_h + np.random.normal(0, 0.02, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points/2) / np.sum(perturbed_h)
        
        # Local optimization
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
                perturbed_h,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 50}
            )
            
            if res.success:
                refined_h = res.x
                refined_h = refined_h * (n_points/2) / np.sum(refined_h)
                refined_c5 = evaluate_erdos_solution(refined_h, dx)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
                    
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
