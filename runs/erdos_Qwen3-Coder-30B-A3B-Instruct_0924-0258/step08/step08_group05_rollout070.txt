import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # FFT-based optimization
    max_iter = 10000
    step_size = 0.01
    tolerance = 1e-9
    no_improvement_limit = 100
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # FFT-based computation
        H = np.fft.fft(h_current)
        G = 1 - h_current
        G_fft = np.fft.fft(G)
        corr_fft = H * np.conj(G_fft)
        overlap_values = np.fft.ifft(corr_fft).real
        overlap_values *= dx
        
        # Find the lag with maximum overlap
        max_lag_idx = np.argmax(overlap_values)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Compute gradient of overlap w.r.t. h for specific lag
        gradient = np.zeros_like(h_current)
        for i in range(len(h_current)):
            if max_lag >= 0:
                if i + max_lag < len(h_current):
                    gradient[i] = -1.0
            else:
                if i - max_lag >= 0:
                    gradient[i] = -1.0
                    
        # Apply step with gradient
        h_new = h_current - step_size * gradient
        
        # Clamp values to [0,1] and re-normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check for improvement
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Reduce step size if no improvement for a while
        if no_improvement_count > 50:
            step_size *= 0.9
            
        h_current = h_new
        
    # Perform final refinement using SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
