import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Configuration
    n_coarse = 15
    n_fine = 35
    coarse_iterations = 500
    fine_iterations = 2000
    mutation_rate = 0.1
    step_size = 0.01
    tolerance = 1e-9
    
    # Coarse grid initialization
    h_coarse = np.random.uniform(0, 1, n_coarse)
    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    # Fine grid initialization
    if 'initial_h_values' in globals():
        h_fine = np.array(initial_h_values)
    else:
        h_fine = np.ones(n_fine) * 0.5
    h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)
    
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # Initialize best solutions
    best_coarse_h = h_coarse.copy()
    best_fine_h = h_fine.copy()
    best_c5_coarse = evaluate_erdos_solution(h_coarse, dx_coarse)
    best_c5_fine = evaluate_erdos_solution(h_fine, dx_fine)
    
    # Coarse grid optimization
    for iteration in range(coarse_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_coarse - 1)
        
        # Compute gradient for max lag
        grad = np.zeros(n_coarse)
        for i in range(n_coarse):
            j = i + max_lag
            if 0 <= j < n_coarse:
                grad[i] = 1 - 2 * h_coarse[i] - h_coarse[j]
            else:
                grad[i] = 1 - 2 * h_coarse[i]
        
        # Gradient descent
        h_coarse_new = h_coarse - step_size * grad
        h_coarse_new = np.clip(h_coarse_new, 0, 1)
        h_coarse_new = h_coarse_new * (n_coarse / 2) / np.sum(h_coarse_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_coarse_new, dx_coarse)
        
        if c5_new < best_c5_coarse:
            best_c5_coarse = c5_new
            best_coarse_h = h_coarse_new.copy()
            h_coarse = h_coarse_new.copy()
        else:
            step_size *= 0.995
            if step_size < tolerance:
                break
    
    # Upsample coarse to fine grid
    coarse_positions = np.linspace(0, 2, n_coarse)
    fine_positions = np.linspace(0, 2, n_fine)
    f = interp1d(coarse_positions, best_coarse_h, kind='linear', fill_value='extrapolate')
    h_fine = f(fine_positions)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)
    
    # Fine grid optimization with alternating descent and mutation
    for iteration in range(fine_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps
        corr = np.correlate(h_fine, 1 - h_fine, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_fine - 1)
        
        # Compute gradient
        grad = np.zeros(n_fine)
        for i in range(n_fine):
            j = i + max_lag
            if 0 <= j < n_fine:
                grad[i] = 1 - 2 * h_fine[i] - h_fine[j]
            else:
                grad[i] = 1 - 2 * h_fine[i]
        
        # Gradient descent step
        h_fine_new = h_fine - step_size * grad
        h_fine_new = np.clip(h_fine_new, 0, 1)
        h_fine_new = h_fine_new * (n_fine / 2) / np.sum(h_fine_new)
        
        # Mutation step (10% of values)
        mutation_indices = np.random.choice(n_fine, size=int(n_fine * mutation_rate), replace=False)
        h_fine_new[mutation_indices] = np.random.uniform(0, 1, len(mutation_indices))
        h_fine_new = h_fine_new * (n_fine / 2) / np.sum(h_fine_new)
        
        # Evaluate both
        c5_new_descent = evaluate_erdos_solution(h_fine_new, dx_fine)
        c5_new_mutation = evaluate_erdos_solution(h_fine_new.copy(), dx_fine)
        
        # Accept best of both
        c5_new = min(c5_new_descent, c5_new_mutation)
        if c5_new < best_c5_fine:
            best_c5_fine = c5_new
            best_fine_h = h_fine_new.copy()
            h_fine = h_fine_new.copy()
        else:
            step_size *= 0.995
            if step_size < tolerance:
                break

    # Final validation with original evaluation
    final_c5 = evaluate_erdos_solution(best_fine_h, dx_fine)
    if final_c5 < best_c5_fine:
        best_c5_fine = final_c5
    
    # Return best solution found
    return (list(best_fine_h), best_c5_fine, n_fine)
