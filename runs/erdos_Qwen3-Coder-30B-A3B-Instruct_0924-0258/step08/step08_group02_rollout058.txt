import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter = 100000
    epsilon = 0.1
    
    # Initialize temperature
    T = T_init
    
    # Initialize current solution
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    c5_current = evaluate_erdos_solution(h_current, dx)
    
    if c5_current < best_c5:
        best_c5 = c5_current
        best_h = h_current.copy()
    
    # Annealing loop
    iter_count = 0
    while T > T_min and time() - start_time < budget_s * 0.9:
        iter_count += 1
        
        # Create new candidate by perturbing two random indices
        i, j = random.sample(range(n_points), 2)
        delta = np.random.uniform(-epsilon, epsilon)
        
        # Apply perturbation
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Enforce bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Ensure sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Compute new overlap
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / T):
            h_current = h_new
            c5_current = c5_new
            
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
        
        # Cool down
        T *= alpha
        
        # Adaptive epsilon
        if iter_count % 1000 == 0:
            epsilon = max(0.001, epsilon * 0.99)
    
    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return np.max(np.correlate(h, 1 - h, mode='full') * dx)
        
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    # Ensure final constraint satisfaction
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
