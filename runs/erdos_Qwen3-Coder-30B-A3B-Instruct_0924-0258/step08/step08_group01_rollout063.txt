import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def crossover(parent1, parent2, alpha=0.5):
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1, step_size=0.05):
    h_mut = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h_mut[i] += np.random.normal(0, step_size)
    return h_mut

def enforce_feasibility(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    # GA parameters
    population_size = 50
    generations = 5000
    mutation_rate = 0.1
    step_size = 0.05
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points) * 0.1 + 0.45
        h = enforce_feasibility(h, target_sum)
        population.append(h)
        
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    no_improvement_count = 0
    
    # GA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitnesses.append(fit)
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= 200:
            break
            
        # Selection and reproduction
        new_population = [population[0]]  # Elitism
        while len(new_population) < population_size:
            # Tournament selection
            idx1 = random.randint(0, population_size // 2)
            idx2 = random.randint(0, population_size // 2)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Crossover
            child = crossover(parent1, parent2)
            
            # Mutation
            child = mutate(child, mutation_rate, step_size)
            
            # Enforce feasibility
            child = enforce_feasibility(child, target_sum)
            
            new_population.append(child)
            
        population = new_population
        
        # Periodic SLSQP refinement
        if gen % 100 == 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 500, 'ftol': 1e-10})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
            
            # Decrease mutation rate
            mutation_rate *= 0.98
            step_size *= 0.98
            
    # Final optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
