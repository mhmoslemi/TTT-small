import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Trust Region with Cutting Planes approach
    trust_region_radius = 0.2
    max_iter = 2000
    improvement_threshold = 1e-6
    no_improvement_limit = 50
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Compute correlation for all lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        lags = np.arange(len(corr)) - (len(h_current) - 1)
        overlap_values = corr * dx
        
        # Find the maximum overlap lag
        max_overlap_idx = np.argmax(overlap_values)
        max_overlap_lag = lags[max_overlap_idx]
        
        # Use CVXPY for convex approximation with current constraint
        h_vars = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add current max overlap constraint
        overlap_constraint = cp.sum(
            cp.multiply(h_vars[:(n_points - abs(max_overlap_lag))], 
                       1 - h_vars[abs(max_overlap_lag):(n_points - abs(max_overlap_lag)) + abs(max_overlap_lag)])
        ) <= C
        constraints.append(overlap_constraint)
        
        # Add other most violated constraints from previous iterations
        # For this implementation, we focus on the main constraint only
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == 'optimal':
                h_new = h_vars.value
                c5_new = C.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                h_new = h_current.copy()
        except Exception:
            h_new = h_current.copy()
        
        # Trust region update based on improvement
        if (no_improvement_count > no_improvement_limit and 
            trust_region_radius > 0.001):
            trust_region_radius *= 0.8
        elif no_improvement_count == 0:
            trust_region_radius = min(trust_region_radius * 1.2, 0.5)
            
        # Bound update with trust region
        diff = h_new - h_current
        if np.linalg.norm(diff) > trust_region_radius:
            h_new = h_current + trust_region_radius * diff / np.linalg.norm(diff)
            
        h_current = h_new.copy()
        
        # Occasionally perform additional refinement using scipy
        if iteration % 50 == 0 and time() - start_time < budget_s * 0.8:
            # Use existing optimization approach to possibly improve
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-10})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_c5 = c5_refined
                        best_h = h_refined.copy()
            except Exception:
                pass
    
    # Final SLSQP refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
