import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter = int(1e5)
    epsilon = 0.1
    
    # Initial temperature and current state
    T = T_init
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    c5_current = evaluate_erdos_solution(h_current, dx)
    
    # Track best solution
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Main Simulated Annealing loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Perturb the current solution
        h_new = h_current.copy()
        i, j = random.sample(range(n_points), 2)
        delta = np.random.uniform(-epsilon, epsilon)
        
        # Ensure the sum constraint is maintained
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clip to bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain sum constraint
        total = np.sum(h_new)
        if total > 0:
            h_new = h_new * (n_points/2) / total
        
        # Compute new overlap
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        delta_c5 = c5_new - c5_current
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h_current = h_new
            c5_current = c5_new
            
            # Update best if improved
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                epsilon = max(0.01, epsilon * 0.999)
        
        # Cool down
        T *= alpha
        if T < T_min:
            T = T_min
    
    # Local refinement using L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
