import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Convex relaxation
    def solve_convex_relaxation(h_init):
        h_vars = cp.Variable(n_points)
        C = cp.Variable()
        
        # Define McCormick envelopes for z[i,j] = h[i] * h[j]
        z_vars = cp.Variable((n_points, n_points))
        
        # Constraints
        constraints = []
        
        # Sum constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # McCormick envelopes
        for i in range(n_points):
            for j in range(n_points):
                if i <= j and i + j < n_points:  # Only for valid combinations
                    constraints.append(z_vars[i,j] >= h_vars[i] + h_vars[j] - 1)
                    constraints.append(z_vars[i,j] <= h_vars[i])
                    constraints.append(z_vars[i,j] <= h_vars[j])
                    constraints.append(z_vars[i,j] >= 0)
        
        # Overlap constraint for each lag k
        for k in range(-n_points + 1, n_points):
            if k >= 0:
                # For positive lags
                lag_sum = 0
                for i in range(n_points - k):
                    lag_sum += h_vars[i] - z_vars[i, i+k]
            else:
                # For negative lags
                lag_sum = 0
                for i in range(n_points + k):
                    lag_sum += h_vars[i] - z_vars[i - k, i]
            constraints.append(lag_sum <= C)
            
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h_vars.value, C.value
        except:
            pass
            
        return h_init, evaluate_erdos_solution(h_init, dx)
    
    # Local refinement with SLSQP
    def refine_solution(h_init):
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-12})
            if res.success:
                return res.x, objective(res.x)
        except:
            pass
        return h_init, evaluate_erdos_solution(h_init, dx)
    
    # Primary optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try convex relaxation
        h_relaxed, c5_relaxed = solve_convex_relaxation(h_current)
        
        # Refine the solution
        h_final, c5_final = refine_solution(h_relaxed)
        
        # Update if better
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
            
        # Update current solution with best found so far
        h_current = h_final.copy()
        
        # Convergence check
        if iteration > 10 and np.abs(best_c5 - c5_final) < 1e-6:
            break
            
    return (list(best_h), best_c5, n_points)
