import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    population = []
    
    # Add initial solution
    h_start = h_start * (n_points/2) / np.sum(h_start)
    population.append(h_start.copy())
    
    # Fill rest of population
    for _ in range(pop_size - 1):
        if n_points % 2 == 0:
            # Even case: n_points/2 ones, rest zeros
            h = np.zeros(n_points)
            h[:n_points//2] = 1.0
        else:
            # Odd case: (n_points-1)/2 ones, one 0.5, rest zeros
            h = np.zeros(n_points)
            h[:n_points//2] = 1.0
            h[n_points//2] = 0.5
        # Randomly shuffle and normalize
        np.random.shuffle(h)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
    
    # Track best solution
    def eval_individual(h):
        return evaluate_erdos_solution(h, dx)
    
    best_h = None
    best_c5 = float('inf')
    
    for h in population:
        c5 = eval_individual(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # GA parameters
    generations = 0
    no_improvement = 0
    max_no_improvement = 100
    
    while time() - start_time < budget_s * 0.95:
        generations += 1
        if no_improvement >= max_no_improvement:
            break
            
        # Evaluate population
        fitness = [eval_individual(h) for h in population]
        sorted_indices = np.argsort(fitness)
        
        # Keep best solution (elitism)
        elite = population[sorted_indices[0]].copy()
        elite_c5 = fitness[sorted_indices[0]]
        
        if elite_c5 < best_c5:
            best_c5 = elite_c5
            best_h = elite.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Tournament selection
        def tournament_selection(k=5):
            tournament = random.sample(list(zip(population, fitness)), k)
            winner = min(tournament, key=lambda x: x[1])
            return winner[0].copy()
        
        # Create new population
        new_population = [elite]
        
        while len(new_population) < pop_size:
            parent1 = tournament_selection()
            parent2 = tournament_selection()
            
            # Uniform crossover
            child = np.zeros(n_points)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            
            # Mutation
            if random.random() < 0.3:  # 30% mutation rate
                # Flip mutation
                i1, i2 = random.sample(range(n_points), 2)
                child[i1] = max(0, min(1, child[i1] + 0.1))
                child[i2] = max(0, min(1, child[i2] - 0.1))
            
            if random.random() < 0.3:  # 30% mutation rate
                # Perturb mutation
                num_perturb = max(1, int(0.1 * n_points))
                indices = random.sample(range(n_points), num_perturb)
                for i in indices:
                    factor = random.uniform(0.9, 1.1)
                    child[i] *= factor
            
            # Adjust to maintain sum constraint
            child_sum = np.sum(child)
            if abs(child_sum - n_points/2) > 1e-6:
                scale = (n_points/2) / child_sum
                child *= scale
            
            new_population.append(child)
        
        population = new_population
    
    # Final refinement with local search for best solution
    try:
        from scipy.optimize import minimize
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0

        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        res = minimize(objective, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 1000, 'ftol': 1e-12})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
