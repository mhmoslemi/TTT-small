import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize with binary structure from the current solution
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    h_binary = (h > 0.5).astype(float)
    h_binary = h_binary * (n_points/2) / np.sum(h_binary)
    
    # Threshold-based optimization
    threshold_vals = np.linspace(0.05, 1.0, 20)
    
    for t_idx, threshold in enumerate(threshold_vals):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Compute overlaps and identify dominating lag
        overlaps = np.correlate(h_binary, 1 - h_binary, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Estimate gradient magnitude
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h_binary[j] - h_binary[i])
        
        # Identify candidates based on gradient threshold
        candidate_indices = np.where(np.abs(gradient) > threshold)[0]
        
        # Update candidates
        h_new = h_binary.copy()
        ones_count = int(n_points/2)
        zeros_count = n_points - ones_count
        
        # Adjust values for candidates
        for idx in candidate_indices:
            if gradient[idx] > 0 and h_new[idx] < 1:
                h_new[idx] = min(1.0, h_new[idx] + 0.05)
            elif gradient[idx] < 0 and h_new[idx] > 0:
                h_new[idx] = max(0.0, h_new[idx] - 0.05)
        
        # Enforce sum constraint
        current_sum = np.sum(h_new)
        if current_sum > n_points/2:
            # Reduce ones
            diff = current_sum - n_points/2
            ones_indices = np.where(h_new > 0)[0]
            reduced = 0
            for idx in sorted(ones_indices, key=lambda x: -h_new[x]):
                if reduced >= diff:
                    break
                if h_new[idx] > 0:
                    new_val = max(0, h_new[idx] - 0.05)
                    reduced += h_new[idx] - new_val
                    h_new[idx] = new_val
        elif current_sum < n_points/2:
            # Increase zeros
            diff = n_points/2 - current_sum
            zeros_indices = np.where(h_new < 1)[0]
            increased = 0
            for idx in sorted(zeros_indices, key=lambda x: h_new[x]):
                if increased >= diff:
                    break
                if h_new[idx] < 1:
                    new_val = min(1, h_new[idx] + 0.05)
                    increased += new_val - h_new[idx]
                    h_new[idx] = new_val
        
        # Normalize to maintain sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Local gradient-based refinement
        def objective(h_vec):
            overlaps = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
            return np.max(overlaps)
        
        # Gradient-based update: gradient descent
        max_iter = 50
        lr = 0.01
        for iter in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            overlaps = np.correlate(h_new, 1 - h_new, mode='full') * dx
            max_overlap = np.max(overlaps)
            l_max = np.argmax(overlaps)
            
            # Approximate gradient for max overlap lag
            gradient = np.zeros(n_points)
            for i in range(n_points):
                j = (i + l_max) % n_points
                gradient[i] = -2 * dx * (1 - h_new[j] - h_new[i])
            
            h_new = h_new - lr * gradient
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Check for convergence
            new_overlap = objective(h_new)
            if abs(new_overlap - max_overlap) < 1e-6:
                break
        
        # Track best solution
        current_c5 = evaluate_erdos_solution(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
    
    # Final local refinement using L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
