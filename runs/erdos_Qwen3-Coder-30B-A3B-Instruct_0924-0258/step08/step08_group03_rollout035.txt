import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible range [0,1] and renormalize to sum to n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_fine = np.array(initial_h_values)
        n_points = len(h_fine)
    else:
        n_points = 35
        h_fine = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Set up coarse grid
    n_coarse = 15
    h_coarse = np.ones(n_coarse) * 0.5
    coarse_dx = 2.0 / n_coarse
    
    best_c5 = float('inf')
    best_h_fine = h_fine.copy()
    
    # Coarse grid optimization
    for iteration in range(500):
        if time() - start_time > budget_s * 0.8:
            break
            
        c5 = evaluate_erdos_solution(h_coarse, coarse_dx)
        if c5 < best_c5:
            best_c5 = c5
            # Interpolate coarse solution to fine grid
            coarse_x = np.linspace(0, 2, n_coarse)
            fine_x = np.linspace(0, 2, n_points)
            f = interp1d(coarse_x, h_coarse, kind='linear', fill_value='extrapolate')
            h_fine = f(fine_x)
            h_fine = project_feasible(h_fine, n_points)
            
        # Compute overlap gradient for coarse
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_coarse - 1)
        
        grad = np.zeros(n_coarse)
        for i in range(n_coarse):
            j = i + max_lag
            if 0 <= j < n_coarse:
                grad[i] = 1 - 2 * h_coarse[i] - h_coarse[j]
            else:
                grad[i] = 1 - 2 * h_coarse[i]

        # Update coarse grid
        h_coarse_new = h_coarse - 0.01 * grad
        h_coarse_new = np.clip(h_coarse_new, 0, 1)
        h_coarse_new = h_coarse_new * (n_coarse / 2) / np.sum(h_coarse_new)
        h_coarse = h_coarse_new
    
    # Convert best coarse solution to fine grid and optimize
    coarse_x = np.linspace(0, 2, n_coarse)
    fine_x = np.linspace(0, 2, n_points)
    f = interp1d(coarse_x, h_coarse, kind='linear', fill_value='extrapolate')
    h_fine = f(fine_x)
    h_fine = project_feasible(h_fine, n_points)
    
    # Fine grid optimization alternating between gradient descent and mutation
    step_size = 0.01
    mutation_rate = 0.1
    
    for iteration in range(2000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current C5
        c5_current = evaluate_erdos_solution(h_fine, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h_fine = h_fine.copy()
        
        # Gradient step
        corr = np.correlate(h_fine, 1 - h_fine, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h_fine[i] - h_fine[j]
            else:
                grad[i] = 1 - 2 * h_fine[i]
                
        h_fine_new = h_fine - step_size * grad
        h_fine_new = project_feasible(h_fine_new, n_points)
        
        # Mutation step (with probability 0.2)
        if np.random.rand() < 0.2:
            mask = np.random.rand(n_points) < mutation_rate
            h_fine_new[mask] = np.random.uniform(0, 1, np.sum(mask))
            h_fine_new = project_feasible(h_fine_new, n_points)
            
        h_fine = h_fine_new
        
        # Reduce step size periodically
        if iteration % 100 == 0 and step_size > 1e-6:
            step_size *= 0.9
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h_fine, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (list(best_h_fine), best_c5, n_points)
