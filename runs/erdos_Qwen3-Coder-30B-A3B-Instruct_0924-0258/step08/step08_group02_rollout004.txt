import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter = 100000
    epsilon = 0.1
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    T = T_init
    iter_count = 0
    
    while T > T_min and time() - start_time < budget_s * 0.9:
        # Random perturbation
        i, j = np.random.randint(0, n_points, 2)
        delta = np.random.uniform(-epsilon, epsilon)
        
        # Ensure the change maintains constraint (sum = n/2)
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        
        # Boundary checks
        if new_h[i] < 0 or new_h[i] > 1 or new_h[j] < 0 or new_h[j] > 1:
            continue
            
        # Adjust to maintain sum constraint
        adjustment = - (np.sum(new_h) - n_points/2)
        new_h[0] += adjustment / n_points
        
        # Clip values to [0,1]
        new_h = np.clip(new_h, 0, 1)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        delta_c5 = new_c5 - best_c5
        
        # Accept or reject
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = new_h
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        T *= alpha
        iter_count += 1
        if iter_count % 1000 == 0 and time() - start_time > budget_s * 0.9:
            break
            
    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        def objective(h):
            return np.max(np.correlate(h, 1 - h, mode='full') * dx)
            
        # Normalize to original sum
        h_refine = best_h * (n_points/2) / np.sum(best_h)
        
        result = minimize(
            objective,
            h_refine,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-12}
        )
        
        if result.success:
            refined_h = result.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
