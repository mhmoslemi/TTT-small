import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize with the given solution
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Try SDP relaxation approach
    try:
        # CVXPY variables
        h = cp.Variable(n_points, bounds=(0, 1))
        
        # Constraint: sum(h) = n_points / 2
        constraints = [cp.sum(h) == n_points / 2]
        
        # Find maximum overlap
        max_lag = n_points - 1
        max_overlap = cp.Variable()
        constraints.append(max_overlap >= 0)
        
        # Formulate all lags
        for k in range(-max_lag + 1, max_lag):
            # Determine valid indices
            if k >= 0:
                valid_i = np.arange(0, n_points - k)
                valid_j = np.arange(k, n_points)
            else:
                valid_i = np.arange(-k, n_points)
                valid_j = np.arange(0, n_points + k)
            
            if len(valid_i) > 0:
                # Correlation term for lag k
                corr_terms = cp.sum(cp.multiply(h[valid_i], 1 - h[valid_j]))
                constraints.append(corr_terms <= max_overlap * dx)
        
        # Objective
        objective = cp.Minimize(max_overlap)
        prob = cp.Problem(objective, constraints)
        
        # Try solving with MOSEK
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s - 10)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            if h_opt is not None:
                # Normalize to satisfy sum constraint exactly
                h_opt = h_opt * (n_points/2) / np.sum(h_opt)
                
                # Recalculate maximum overlap
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt

    except Exception as e:
        # If SDP fails, do nothing and proceed to the next optimization
        pass

    # Hyperswarm optimization refinement
    best_c5 = evaluate_erdos_solution(best_h, dx)
    max_iter = 10000
    iter_count = 0
    step_size = 0.01

    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all correlations for all possible lags
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Get the maximum overlap lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(best_h) - 1)
        
        # Compute gradient for max overlap
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - best_h[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= best_h[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - best_h[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= best_h[i]
        
        # Gradient descent step
        h_new = best_h - step_size * gradient
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        else:
            step_size *= 0.9
        
        iter_count += 1

    return (list(best_h), best_c5, n_points)
