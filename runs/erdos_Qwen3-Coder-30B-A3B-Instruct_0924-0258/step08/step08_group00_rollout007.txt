import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Normalize to satisfy sum constraint
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # First try gradient-based approach
    h = h_start.copy()
    
    # Gradient descent with adaptive step size
    for iteration in range(1000]:
        if time() - start_time > budget_s * 0.9:
            break
            
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Compute gradient
        max_lag = max_lag_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 2 * (1 - h[j] - h[i]) / n_points
        
        # Adaptive step size
        step_size = 0.01
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        if evaluate_erdos_solution(h_new, dx) < evaluate_erdos_solution(h, dx):
            h = h_new
        else:
            step_size *= 0.5
            if step_size < 1e-8:
                break
    
    # Refinement with constrained optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    # Additional hill climbing
    h_current = best_h.copy()
    for _ in range(500):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb one element
        i = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.05, 0.05)
        h_test = h_current.copy()
        h_test[i] += delta
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points/2) / np.sum(h_test)
        
        c5_test = evaluate_erdos_solution(h_test, dx)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()
            h_current = h_test.copy()
        else:
            # Accept with probability based on improvement
            if np.random.rand() < 0.01:
                h_current = h_test.copy()
    
    return (list(best_h), best_c5, n_points)
