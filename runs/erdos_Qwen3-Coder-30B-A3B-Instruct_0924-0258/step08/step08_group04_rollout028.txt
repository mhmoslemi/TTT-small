import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available  
    if 'initial_h_values' in globals():

        n_points = len(h_start)

    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Multiresolution approach:
    # 1. Coarse grid optimization (n=15) 
    # 2. Upscaling to fine grid (n=35)
    # 3. Gradient refinement
    
    # Coarse optimization
    n_coarse = 15
    dx_coarse = 2.0 / n_coarse
    population_size = 50
    elite_size = 10
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = np.random.uniform(0, 1, n_coarse)
        h = h * (n_coarse/2) / np.sum(h)
        population.append(h)
    
    # Add initial solution if it works
    h_init_coarse = np.interp(np.linspace(0, 2, n_coarse), 
                            np.linspace(0, 2, len(h_current)), h_current)
    h_init_coarse = h_init_coarse * (n_coarse/2) / np.sum(h_init_coarse)
    population[0] = h_init_coarse
    
    # Genetic optimization on coarse grid
    for gen in range(500):
        if time() - start_time > budget_s * 0.7:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx_coarse)
            fitness.append(-c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)[::-1]
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check if we improved
        best_in_gen = -fitness[0]
        if best_in_gen < best_c5:
            best_c5 = best_in_gen
            best_h_coarse = population[0].copy()
        
        # Selection + reproduction
        new_pop = population[:elite_size].copy()
        
        while len(new_pop) < population_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(elite_size, 2, replace=False)
            parent1, parent2 = population[idx1], population[idx2]
            
            # Crossover
            if np.random.random() < 0.5:
                point = np.random.randint(1, n_coarse-1)
                child = np.concatenate([parent1[:point], parent2[point:]])
            else:
                child = (parent1 + parent2) / 2
            
            # Mutation
            for i in range(n_coarse):
                if np.random.random() < 0.2:
                    child[i] += np.random.normal(0, 0.1)
            
            # Clip and rescale
            child = np.clip(child, 0, 1)
            child = child * (n_coarse/2) / np.sum(child)
            
            new_pop.append(child)
        
        population = new_pop
    
    # Upscale to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, n_coarse), best_h_coarse)
    h_fine = h_fine * (n_points/2) / np.sum(h_fine)
    h_current = h_fine
    
    # Gradient refinement on fine grid
    step_size = 0.01
    iter_count = 0
    max_iter = 10000
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute overlaps
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Find top contributing lags
        top_lags = np.argsort(corr)[-5:][::-1]  # Top 5 lags
        max_lag = top_lags[0] - (len(h_current) - 1)
        
        # Compute gradients for these lags
        gradient = np.zeros(n_points)
        for lag in top_lags:
            lag_val = lag - (len(h_current) - 1)
            if lag_val >= 0:
                for i in range(n_points - lag_val):
                    gradient[i] += 1 - h_current[i + lag_val]
                    if i + lag_val < n_points:
                        gradient[i + lag_val] -= h_current[i]
            else:
                for i in range(n_points + lag_val):
                    gradient[i] += 1 - h_current[i - lag_val]
                    if i - lag_val >= 0:
                        gradient[i - lag_val] -= h_current[i]
        
        # Gradient descent step
        h_new = h_current - step_size * gradient
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            step_size = min(0.05, step_size * 1.1)  # Increase step size
        else:
            step_size *= 0.95  # Reduce step size
            if step_size < 1e-10:
                break
                
        h_current = h_new.copy()
        iter_count += 1
    
    return (list(best_h), best_c5, n_points)
