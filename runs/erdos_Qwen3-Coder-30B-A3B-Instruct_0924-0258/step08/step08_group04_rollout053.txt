import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Multiresolution approach: coarse grid first to escape local optima
    # Coarse resolution
    n_coarse = 15
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.linspace(0, 1, n_coarse)
    h_coarse = h_coarse * (n_coarse/2) / np.sum(h_coarse)
    
    # Population-based search on coarse grid
    population_size = 50
    population = []
    for _ in range(population_size):
        h_tmp = h_coarse + np.random.normal(0, 0.1, n_coarse)
        h_tmp = np.clip(h_tmp, 0, 1)
        h_tmp = h_tmp * (n_coarse/2) / np.sum(h_tmp)
        population.append(h_tmp)
    
    # Evolve coarse population
    mutation_rate = 0.2
    crossover_rate = 0.5
    generations = 100
    for gen in range(generations):
        if time() - start_time > budget_s * 0.6:
            break
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx_coarse)
            fitness.append(-c5)
        
        # Select top 50%
        sorted_indices = np.argsort(fitness)[::-1][:population_size//2]
        population = [population[i] for i in sorted_indices]
        
        # Generate new population
        new_population = population[:]
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < crossover_rate:
                i1, i2 = np.random.choice(len(population), 2, replace=False)
                parent1 = population[i1]
                parent2 = population[i2]
                crossover_point = np.random.randint(1, n_coarse-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child = child * (n_coarse/2) / np.sum(child)
                new_population.append(child)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                i = np.random.randint(len(population))
                child = population[i].copy()
                mutate_idx = np.random.randint(n_coarse)
                child[mutate_idx] += np.random.normal(0, 0.1)
                child = np.clip(child, 0, 1)
                child = child * (n_coarse/2) / np.sum(child)
                new_population.append(child)
        
        population = new_population[:population_size]
    
    # Find best coarse solution and upscale to fine grid
    best_coarse = None
    best_coarse_c5 = float('inf')
    for h in population:
        c5 = evaluate_erdos_solution(h, dx_coarse)
        if c5 < best_coarse_c5:
            best_coarse_c5 = c5
            best_coarse = h
    
    # Interpolate to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), best_coarse)
    h_fine = h_fine * (n_points/2) / np.sum(h_fine)
    
    # Gradient optimization on fine grid
    h_current = h_fine.copy()
    step_size = 0.01
    max_iter = 10000
    iter_count = 0
    while time() - start_time < budget_s * 0.99 and iter_count < max_iter:
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_current[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= h_current[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_current[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_current[i]
        
        h_new = h_current - step_size * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        if c5_new >= evaluate_erdos_solution(h_current, dx):
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            h_current = h_new
            step_size *= 1.02
        
        iter_count += 1
    
    return (list(best_h), best_c5, n_points)
