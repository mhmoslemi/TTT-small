import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Trust region method with cutting planes
    max_iter = 5000
    trust_radius = 0.2
    min_trust_radius = 1e-6
    trust_radius_factor = 1.2
    trust_shrink_factor = 0.5
    
    # Precompute all lags
    all_lags = np.arange(-n_points + 1, n_points)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlap values
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        lags = np.arange(len(corr)) - (len(h_current) - 1)
        overlap_values = corr * dx
        
        # Find max overlap index
        max_idx = np.argmax(overlap_values)
        max_lag = lags[max_idx]
        
        # Add a small random perturbation to break symmetry
        if iteration % 50 == 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            perturbation = np.clip(perturbation, -trust_radius, trust_radius)
            h_current = np.clip(h_current + perturbation, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Use trust region constrained optimization
        # For large problems, use simpler trust region without CVXPY
        # But compute initial good step
        h_new = h_current.copy()
        
        # Perform gradient step to decrease maximum overlap
        # This is a heuristic: simply sample the effect of modifying h[i]
        best_gain = -1
        best_i = -1
        best_delta = 0
        
        for i in range(n_points):
            target_h = h_current.copy()
            delta = min(0.1, trust_radius)
            new_val = min(1.0, h_current[i] + delta)
            target_h[i] = new_val
            new_val = max(0.0, h_current[i] - delta)
            target_h[i] = new_val
            
            corr_new = np.correlate(target_h, 1 - target_h, mode='full')
            overlap_new = corr_new * dx
            max_new = np.max(overlap_new)
            
            if max_new < best_c5 and max_new < best_gain:
                best_gain = max_new
                best_i = i
                best_delta = delta
                
        if best_i >= 0 and best_gain < best_c5:
            h_new[best_i] = max(0.0, min(1.0, h_new[best_i] + best_delta))
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_current = evaluate_erdos_solution(h_new, dx)
            if c5_current < best_c5:
                h_current = h_new.copy()
                best_c5 = c5_current
                trust_radius = max(min_trust_radius, trust_radius * trust_radius_factor)
        else:
            trust_radius *= trust_shrink_factor
        
        # Occasionally use SLSQP to refine
        if iteration % 200 == 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum,

                if res.success:
                    h_test = res.x
                    c5_test = objective(h_test)
                    if c5_test < best_c5:
                        h_current = h_test
                        best_c5 = c5_test
            except Exception:
                pass
            
            # Also re-evaluate current best solution
            c5_current = evaluate_erdos_solution(h_current, dx)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
    
    # Final step - important SLSQP refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, h_current, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
