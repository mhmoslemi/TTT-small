import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale optimization
    current_h = h_start.copy()
    current_c5 = best_c5
    scale = 0
    max_scale = int(np.floor(np.log2(n_points)) + 2)  # up to 2^max_scale
    perturb_mag = 0.05
    
    while scale < max_scale and (time() - start_time) < budget_s * 0.95:
        # Double the resolution
        new_n_points = 2 ** (scale + 1)
        if new_n_points > 1000:  # Cap at reasonable size
            break
            
        # Upsample current solution
        new_h = np.interp(np.linspace(0, 2, new_n_points), 
                         np.linspace(0, 2, len(current_h)), 
                         current_h)
        
        # Add noise and project to feasible region
        noise = np.random.uniform(-perturb_mag, perturb_mag, new_n_points)
        new_h = new_h + noise
        
        # Clip values and rescale to maintain sum constraint
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (new_n_points / 2) / np.sum(new_h)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, 2.0 / new_n_points)
        if new_c5 < current_c5:
            current_h = new_h.copy()
            current_c5 = new_c5
            perturb_mag = 0.05  # reset perturbation magnitude
        else:
            perturb_mag *= 0.95  # reduce perturbation magnitude
        
        # Local SQP refinement
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / len(h))
        
        def constraint_sum(h):
            return np.sum(h) - len(h) / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(len(current_h))]
        
        try:
            res = minimize(objective, current_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-10})
            if res.success:
                refined_h = res.x
                refined_c5 = objective(refined_h)
                if refined_c5 < current_c5:
                    current_h = refined_h
                    current_c5 = refined_c5
        except Exception:
            pass
            
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        scale += 1
    
    # Final SLSQP refinement
    def final_objective(h):
        return evaluate_erdos_solution(h, 2.0 / len(h))
    
    def final_constraint_sum(h):
        return np.sum(h) - len(h) / 2.0
    
    final_cons = {'type': 'eq', 'fun': final_constraint_sum}
    final_bounds = [(0, 1) for _ in range(len(best_h))]
    
    try:
        res = minimize(final_objective, best_h, method='SLSQP',
                       bounds=final_bounds, constraints=final_cons,
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            final_h = res.x
            final_c5 = final_objective(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
    
    n_points = len(best_h)
    return (list(best_h), best_c5, n_points)
