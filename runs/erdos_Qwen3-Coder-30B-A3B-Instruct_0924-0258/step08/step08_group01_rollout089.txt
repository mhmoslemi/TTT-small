import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    M = 50  # Population size
    generations = 5000
    crossover_rate = 0.8
    mutation_rate = 0.1
    mut_step = 0.05
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        M -= 1
    
    # Generate initial population
    for _ in range(M):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # Normalize sum to n_points/2
        population.append(h)
    
    # GA loop
    no_improvement_count = 0
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            f = evaluate_erdos_solution(h, dx)
            fitness.append(f)
        
        # Track best solution
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_h = population[min_fit_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 200:
            break
        
        # Tournament selection
        selected = []
        tourn_size = 3
        for _ in range(M // 2):
            tournament = np.random.choice(M, tourn_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(winner)
        
        # Generate offspring
        offspring = []
        for i in range(0, len(selected), 2):
            if i + 1 < len(selected) and np.random.rand() < crossover_rate:
                parent1 = population[selected[i]]
                parent2 = population[selected[i+1]]
                
                # Blend crossover
                alpha = np.random.rand()
                alpha = np.clip(alpha, 0.3, 0.7)
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
                
                # Mutation
                if np.random.rand() < mutation_rate:
                    child1 += np.random.normal(0, mut_step, n_points)
                if np.random.rand() < mutation_rate:
                    child2 += np.random.normal(0, mut_step, n_points)
                
                # Ensure feasibility
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points / 2) / np.sum(child1)
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
                
                offspring.append(child1)
                offspring.append(child2)
            else:
                # Keep parents if no crossover or odd number
                offspring.append(population[selected[i]])
                if i + 1 < len(selected):
                    offspring.append(population[selected[i+1]])
        
        # Keep top individuals and offspring
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices[:M//2]]
        population.extend(offspring[:M//2])
        
        # Adaptive mutation step
        mut_step *= 0.98
    
    # Final local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
