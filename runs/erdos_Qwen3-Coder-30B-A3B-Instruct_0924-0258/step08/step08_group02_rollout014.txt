import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Binary initialization and sparse optimization
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Binary search approach
    threshold_levels = np.linspace(1.0, 0.0, 50)
    for t in threshold_levels:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute sparse gradient (only active positions)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Update step with threshold
        update = np.zeros(n_points)
        active_positions = np.abs(gradient) > t
        update[active_positions] = -gradient[active_positions]
        
        # Apply update with projection
        h_new = h + update
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        current_sum = np.sum(h_new)
        target_sum = n_points / 2
        if current_sum > 0:
            h_new = h_new * target_sum / current_sum
        
        # Check improvement
        current_c5 = evaluate_erdos_solution(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        
        # Update for next iteration
        h = h_new.copy()
    
    # Phase 2: Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    # Final validation
    final_sum = np.sum(best_h)
    if abs(final_sum - n_points/2) > 1e-6:
        best_h = best_h * (n_points/2) / final_sum
    
    return (list(best_h), best_c5, n_points)
