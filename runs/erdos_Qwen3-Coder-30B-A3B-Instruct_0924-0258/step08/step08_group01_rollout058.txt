import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    stagnation_limit = 50
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    if 'initial_h_values' in globals():
        population = [h_start.copy()]
        population.extend([create_individual() for _ in range(population_size - 1)])
    else:
        population = [create_individual() for _ in range(population_size)]
    
    def evaluate_population(pop):
        return [evaluate_erdos_solution(ind, dx) for ind in pop]
    
    # Tournament selection
    def tournament_selection(pop, fitness, tournament_size=3):
        selected_idx = np.random.choice(len(pop), tournament_size)
        best_idx = selected_idx[np.argmin([fitness[i] for i in selected_idx])]
        return pop[best_idx].copy()
    
    # Blend crossover
    def blend_crossover(parent1, parent2, alpha=0.5):
        child1 = np.zeros_like(parent1)
        child2 = np.zeros_like(parent2)
        for i in range(len(parent1)):
            if np.random.random() < 0.5:
                child1[i] = parent1[i]
                child2[i] = parent2[i]
            else:
                lower = min(parent1[i], parent2[i])
                upper = max(parent1[i], parent2[i])
                range_val = upper - lower
                child1[i] = lower - alpha * range_val + np.random.random() * (2 * alpha * range_val)
                child2[i] = lower - alpha * range_val + np.random.random() * (2 * alpha * range_val)
        # Ensure constraints
        child1 = np.clip(child1, 0, 1)
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = np.clip(child2, 0, 1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        return child1, child2
    
    # Mutation with adaptive step
    def mutate(individual, step_size=0.05):
        mutated = individual.copy()
        for i in range(len(mutated)):
            if np.random.random() < mutation_rate:
                mutated[i] += np.random.normal(0, step_size)
        # Ensure constraints
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    # Genetic algorithm loop
    no_improvement_count = 0
    prev_best = best_c5

            break

        fitness = evaluate_population(population)
        
        # Track best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Check for stagnation
        if no_improvement_count >= stagnation_limit:
            # Reset population with some variation
            if 'initial_h_values' in globals():
                population = [h_start.copy()]
                population.extend([create_individual() for _ in range(population_size - 1)])
            else:
                population = [create_individual() for _ in range(population_size)]
            no_improvement_count = 0
        
        # Selection, crossover and mutation
        new_population = []
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            if np.random.random() < crossover_rate:
                child1, child2 = blend_crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        new_population = new_population[:population_size]
        population = new_population
    
    # Final local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
