import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Try convex relaxation approach first
    try:
        # Create CVXPY problem
        h_vars = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints for h_vars in [0,1]
        constraints = [h_vars >= 0, h_vars <= 1]
        
        # Sum constraint
        constraints.append(cp.sum(h_vars) == n_points/2)
        
        # Overlay constraints using McCormick envelopes
        # We model h[i] * h[j] with auxiliary variable z[i,j]
        # z[i,j] >= h[i] + h[j] - 1
        # z[i,j] <= h[i]
        # z[i,j] <= h[j]
        # z[i,j] >= 0
        
        # Maximum overlap constraint by lag
        max_lag = n_points - 1
        for k in range(-max_lag, max_lag + 1):
            if k == 0:
                continue
            # For lag k, corr[x] = sum_{i=0}^{n_points-1-|k|} h[i] h[i+k]
            # Our overlap is sum_{i=0}^{n_points-1-|k|} h[i](1-h[i+k])
            # = sum_{i=0}^{n_points-1-|k|} (h[i] - h[i] h[i+k])
            # Let z[i, i+k] = h[i] h[i+k], an auxiliary variable
            # This is a bit more fussy to encode using the "full" convolution
            # But we can construct the constraint for a fixed k directly
            if k > 0:
                # h[i] h[i+k] for i = 0 to n_points-1-k
                sum_term = 0
                for i in range(n_points - k):
                    # The overlaps constrain the max of a convolution
                    # max lag is n_points - 1, so for lag k we have n_points - k terms
                    # overlap = h[i] - h[i] h[i+k] = h[i] - z[i, i+k], for i = 0,...,n_points-1-k
                    sum_term += h_vars[i] - h_vars[i] * h_vars[i + k]
                constraints.append(sum_term <= C)
            else:
                # k < 0: h[i] h[i+k] for i = -k to n_points-1
                sum_term = 0
                for i in range(-k, n_points):
                    sum_term += h_vars[i] - h_vars[i] * h_vars[i + k]
                constraints.append(sum_term <= C)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        # Solve the relaxed problem
        prob.solve(solver=cp.CVXOPT, verbose=False)
        
        # Get the solution from CVXPY
        if prob.status == cp.OPTIMAL:
            h_relaxed = h_vars.value
            if h_relaxed is not None:
                h_current = np.clip(h_relaxed, 0, 1)
                h_current = h_current * (n_points/2) / np.sum(h_current)
                c5_relaxed = evaluate_erdos_solution(h_current, dx)
                if c5_relaxed < best_c5:
                    best_c5 = c5_relaxed
                    best_h = h_current.copy()
    except Exception as e:
        pass
    
    # Local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final check and hardening
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
