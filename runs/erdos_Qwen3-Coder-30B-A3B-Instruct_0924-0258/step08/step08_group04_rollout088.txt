import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    # Make sure sum(h) = n_points / 2
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # PSO Parameters
    num_particles = 75
    inertia = 0.8
    inertia_decay = 0.999
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    perturbation_rate = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    particle_best = []
    particle_best_values = []
    
    # Initialize particles with random feasible solutions
    for _ in range(num_particles):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        particle_best.append(h.copy())
        particle_best_values.append(evaluate_erdos_solution(h, dx))
    
    # Initialize global best
    global_best_idx = np.argmin(particle_best_values)
    global_best = particle_best[global_best_idx].copy()
    global_best_value = particle_best_values[global_best_idx]
    
    # Set max iterations based on remaining time
    max_iter = int(100000 * (budget_s - (time() - start_time)) / budget_s)
    if max_iter <= 0:
        max_iter = 5000
    
    iter_count = 0
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Update inertia
        inertia = max(0.4, inertia * inertia_decay)
        
        for i in range(num_particles):
            h = particles[i]
            v = velocities[i]
            p_best = particle_best[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (inertia * v + 
                 cognitive_weight * r1 * (p_best - h) + 
                 social_weight * r2 * (global_best - h))
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            
            # Update position
            h_new = h + v
            
            # Project to feasible set
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate new solution
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Update particle best
            if c5_new < particle_best_values[i]:
                particle_best[i] = h_new.copy()
                particle_best_values[i] = c5_new
            
            # Update global best
            if c5_new < global_best_value:
                global_best = h_new.copy()
                global_best_value = c5_new
                last_improvement = iter_count
            
            # Store updated values
            particles[i] = h_new.copy()
            velocities[i] = v.copy()
        
        iter_count += 1
        
        # Perturbation to escape local minima
        if iter_count % 1000 == 0 and iter_count > last_improvement + 5000:
            for i in range(num_particles):
                if np.random.rand() < perturbation_rate:
                    h = particles[i]
                    # Add small random perturbation
                    perturb = np.random.normal(0, 0.01, n_points)
                    h = h + perturb
                    h = np.clip(h, 0, 1)
                    h = h * (n_points/2) / np.sum(h)
                    particles[i] = h.copy()
    
    # Final refinement
    best_h = global_best.copy()
    final_c5 = global_best_value
    
    return (list(best_h), final_c5, n_points)
