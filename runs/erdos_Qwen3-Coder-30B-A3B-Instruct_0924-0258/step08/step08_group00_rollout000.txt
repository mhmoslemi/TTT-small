import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time()
    best_h = None
    best_c5 = float('inf')
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # Multi-resolution approach: start coarse and refine
    resolutions = [35, 70, 140, 280]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to current resolution if needed
        if res > n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            h_interp = np.interp(x_new, x_orig, h_start)
            h_start = h_interp
            n_points = res
        elif res < n_points:
            # Downsample
            h_start = h_start[::n_points//res] if n_points//res > 0 else h_start
            n_points = res
            
        dx = 2.0 / n_points
        
        # Simulated Annealing
        T0 = 0.1
        Tmin = 1e-6
        alpha = 0.95
        max_iter = 10000
        patience = 1000
        
        h_current = h_start.copy()
        h_current = h_current * (n_points/2) / np.sum(h_current)
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            
        T = T0
        no_improvement = 0
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                    
            # Perturb solution
            h_new = h_current.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, T)
            h_new[idx] += delta
            h_new[idx] = np.clip(h_new[idx], 0, 1)
            
            # Renormalize
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                h_current = h_new
                current_c5 = new_c5
                no_improvement = 0
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_current.copy()
            else:
                no_improvement += 1
                
            # Cooling schedule
            if T > Tmin:
                T *= alpha
                
            if no_improvement > patience:
                break
                
        # Final refinement using SLSQP
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_current, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined
        except Exception:
            pass
        
        if time() - start_time > budget_s * 0.95:
            break
    
    if best_h is None:
        best_h = h_start.copy()
    
    return (list(best_h), best_c5, n_points)
