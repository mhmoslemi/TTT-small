import numpy as np
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search on C5 using SDP relaxation
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    max_iter = 100
    
    shift_indices = np.random.choice(n_points, size=min(100, n_points), replace=False)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2
        
        # Formulate SDP
        h_var = Variable(n_points)
        Y = Variable((n_points, n_points), symmetric=True)
        
        constraints = [
            Y >> 0,  # Positive semidefinite
            cp.diag(Y) == h_var,  # Diagonal constraint
            cp.sum(h_var) == n_points / 2,  # Normalization
        ]
        
        # Add constraints for overlapping integrals
        for k in shift_indices:
            if k == 0:
                continue
            # Sum_{i=0}^{n-1} h_i(1 - h_{i+k}) <= mid*dx
            # = Sum h_i - Sum h_i h_{i+k} <= mid*dx
            # = Sum h_i - Sum Y[i,i+k] <= mid*dx
            sum_h = cp.sum(h_var)
            sum_Y = cp.sum(cp.diag(Y, k))
            constraints.append(sum_h - sum_Y <= mid * dx)
        
        objective = Minimize(mid)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=MOSEK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_sdp = h_var.value
                if h_sdp is not None:
                    # Project back to original space
                    h_projected = np.clip(h_sdp, 0, 1)
                    h_projected = h_projected * (n_points / 2) / np.sum(h_projected)
                    
                    # Local refinement with gradient descent
                    step_size = 0.01
                    for _ in range(50):
                        corr = np.correlate(h_projected, 1 - h_projected, mode='full')
                        max_lag_idx = np.argmax(corr)
                        max_lag = max_lag_idx - (n_points - 1)
                        
                        grad = np.zeros(n_points)
                        for i in range(n_points):
                            j = i + max_lag
                            if 0 <= j < n_points:
                                grad[i] = 1 - 2 * h_projected[i] - h_projected[j]
                            else:
                                grad[i] = 1 - 2 * h_projected[i]
                        
                        h_new = h_projected - step_size * grad
                        h_new = np.clip(h_new, 0, 1)
                        h_new = h_new * (n_points / 2) / np.sum(h_new)
                        
                        c5_new = evaluate_erdos_solution(h_new, dx)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                        
                        h_projected = h_new.copy()
                        
                    if evaluate_erdos_solution(h_projected, dx) < best_c5:
                        best_c5 = evaluate_erdos_solution(h_projected, dx)
                        best_h = h_projected.copy()
                    high = mid
                else:
                    low = mid
            else:
                low = mid
        except Exception:
            low = mid
            
    return (list(best_h), best_c5, n_points)
