import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # More aggressive gradient-based optimization with FFT
    max_iter = 20000
    step_size = 0.01
    no_improvement_limit = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Use FFT-based computation for overlap
        h_fft = np.fft.fft(h_current)
        g_fft = np.fft.fft(1 - h_current)
        corr_fft = h_fft * np.conj(g_fft)
        overlap_values = np.fft.ifft(corr_fft).real
        # Only consider valid overlaps (not circular)
        overlap_values = overlap_values[len(h_current)-1:]
        
        # Find maximum overlap index
        max_idx = np.argmax(overlap_values)
        max_overlap = overlap_values[max_idx]
        
        # Compute gradient for this specific lag using finite differences
        grad = np.zeros_like(h_current)
        eps = 1e-6
        
        for i in range(len(h_current)):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            max_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (max_plus - max_overlap) / eps
        
        # Apply gradient step
        h_new = h_current - step_size * grad
        
        # Clamp values and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Update
        h_current = h_new
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            
        # Adjust learning rate
        if iteration % 100 == 0 and iteration > 0:
            step_size *= 0.5
    
    # Final refinement with optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
