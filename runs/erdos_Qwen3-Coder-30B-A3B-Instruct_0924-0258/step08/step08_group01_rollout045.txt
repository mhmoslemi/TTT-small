import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale optimization
    max_scales = int(np.log2(n_points)) + 2
    scale = 0
    current_h = h_start.copy()
    current_n = n_points
    perturbation_magnitude = 0.05
    scale_iterations = 10
    scale_decay = 0.95
    
    while scale < max_scales and (time() - start_time) < budget_s * 0.95:
        # Double the resolution for next scale
        next_n = current_n * 2
        if next_n > 1000:  # Prevent oversized arrays
            break
        
        # Upsample current solution
        h_upsampled = np.interp(
            np.linspace(0, 2, next_n),
            np.linspace(0, 2, current_n),
            current_h
        )
        
        # Apply perturbations
        for _ in range(scale_iterations):
            if (time() - start_time) >= budget_s * 0.95:
                break

            h_perturbed = h_upsampled.copy()
            for i in range(next_n):
                h_perturbed[i] += perturbation_magnitude * (2 * np.random.random() - 1)
            
            # Ensure feasibility
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (next_n / 2) / np.sum(h_perturbed)
            
            # Evaluate and accept if improved
            c5_perturbed = evaluate_erdos_solution(h_perturbed, 2.0 / next_n)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
                current_h = h_perturbed.copy()
                current_n = next_n
                perturbation_magnitude = 0.05  # Reset
            else:
                perturbation_magnitude *= scale_decay
        
        # Local refinement at current scale using SLSQP
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / next_n)
        
        def constraint_sum(h):
            return np.sum(h) - next_n / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(next_n)]
        
        h_refined = current_h.copy()
        try:
            res = minimize(objective, h_refined, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined.copy()
        except Exception:
            pass

        current_h = best_h[:current_n].copy() if len(best_h) > current_n else best_h.copy()
        current_n = next_n
        scale += 1
    
    # Final refinement with full optimization
    if (time() - start_time) < budget_s * 0.95:
        # Reinitialize with best result
        h_final = best_h.copy()
        dx_final = 2.0 / len(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx_final)
        
        def constraint_sum(h):
            return np.sum(h) - len(h) / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(len(h_final))]
        
        try:
            res = minimize(objective, h_final, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 500, 'ftol': 1e-12})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, len(best_h))
