import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iterations = 100000
    epsilon = 0.1
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    current_c5 = evaluate_erdos_solution(h, dx)
    
    T = T_init
    iteration = 0
    
    while T > T_min and iteration < max_iterations and (time() - start_time) < budget_s * 0.9:
        iteration += 1
        
        # Make a random perturbation
        i, j = np.random.randint(0, n_points, 2)
        delta = np.random.uniform(-epsilon, epsilon)
        
        # Ensure the change keeps h[j] within bounds
        if h[j] + delta < 0:
            delta = -h[j]
        elif h[j] + delta > 1:
            delta = 1 - h[j]
        
        # Make the change
        h[i] += delta
        h[j] -= delta
    
        # Ensure bounds
        h = np.clip(h, 0, 1)
        
        # Ensure sum constraint
        h = h * (n_points/2) / np.sum(h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Accept with probability based on temperature
            delta_c5 = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta_c5 / T):
                current_c5 = new_c5
            else:
                # Revert the change
                h[i] -= delta
                h[j] += delta
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
        
        # Cool down
        T *= alpha

    # Local refinement using L-BFGS-B
    try:
        # Ensure sum constraint in refinement
        bounds = [(0, 1) for _ in range(n_points)]
        
        def obj(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)
        
        # Normalize the initial solution for L-BFGS
        h_refine = best_h * (n_points/2) / np.sum(best_h)
        
        res = minimize(
            obj,
            h_refine,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
