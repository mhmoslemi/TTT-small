import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale optimization
    max_scales = int(np.log2(n_points)) + 1
    scale = 1
    
    while time() - start_time < budget_s * 0.95 and scale <= max_scales:
        current_n = 2**scale
        if current_n > n_points:
            # Upsample solution
            h_current = np.interp(np.linspace(0, 1, current_n), 
                                np.linspace(0, 1, len(best_h)), 
                                best_h)
        else:
            h_current = best_h.copy()
            
        # Random perturbations with decay
        mu = 0.05 * (0.95 ** (scale - 1))
        perturbation_count = 0
        max_perturbations = max(10, current_n // 10)
        
        while perturbation_count < max_perturbations and time() - start_time < budget_s * 0.95:
            # Generate perturbed solution
            h_new = h_current.copy()
            idx = np.random.randint(0, current_n)
            h_new[idx] += mu * (2 * np.random.random() - 1)
            h_new = np.clip(h_new, 0, 1)
            
            # Adjust to maintain sum constraint
            h_new = h_new * (current_n / 2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, 2.0 / current_n)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                mu = 0.05 * (0.95 ** (scale - 1))  # Reset
            else:
                mu *= 0.95  # Decay
                
            perturbation_count += 1
            
        # Local SQP refinement
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / len(h))
        
        def constraint_sum(h):
            return np.sum(h) - len(h) / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(len(h_current))]
        
        try:
            res = minimize(objective, h_current, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-12})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined
        except Exception:
            pass
            
        scale += 1
        
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / len(h))
        
        def constraint_sum(h):
            return np.sum(h) - len(h) / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(len(best_h))]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            best_h = res.x
            best_c5 = objective(best_h)
    except Exception:
        pass
    
    return (list(best_h), best_c5, len(best_h))
