import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def repair_h(h, target_sum):
    h = np.clip(h, 0, 1)
    if np.sum(h) == 0:
        return h
    h = h * target_sum / np.sum(h)
    return h

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Latin hypercube sampling for diverse initializations
    n_initial_guesses = 50
    
    # Generate Latin hypercube samples
    samples = np.random.random((n_initial_guesses, n_points))
    for i in range(n_points):
        samples[:, i] = np.sort(samples[:, i])
        samples[:, i] = (samples[:, i] * (i + 1) + (n_points - i - 1) * samples[:, i]) / n_points
    
    # Normalize each sample to satisfy sum constraint
    h_candidates = []
    for s in samples:
        h_candidate = s * target_sum / np.sum(s)
        h_candidates.append(h_candidate)
    
    # Add the initial construction as first candidate if available
    if 'initial_h_values' in globals():

    for i, h0 in enumerate(h_candidates):

    try:

        def objective(h):
            return evaluate_erdos_solution(h, dx)

            return np.sum(h) - target_sum
