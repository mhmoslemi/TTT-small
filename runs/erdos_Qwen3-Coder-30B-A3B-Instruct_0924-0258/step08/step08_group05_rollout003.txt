import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Start with a better initial guess: alternating 0 and 1
        n_points = 35
        h_start = np.ones(n_points)
        h_start[::2] = 0.0
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Refinement using local optimization
    max_iter = 5000
    no_improvement_limit = 100
    
    # Alternative: use CVXPY for a convex relaxation approach
    remaining_time = budget_s * 0.7
    if remaining_time > 1.0:
        try:
            # Convex relaxation approach
            h_vars = cp.Variable(n_points)
            C = cp.Variable()
            
            # Set up McCormick envelopes
            z_vars = cp.Variable((n_points, n_points))
            
            constraints = [
                cp.sum(h_vars) == n_points/2,
                h_vars >= 0,
                h_vars <= 1,
                C >= 0
            ]
            
            # Add McCormick envelope constraints
            for i in range(n_points):
                for j in range(n_points):
                    if i <= j:
                        constraints.append(z_vars[i,j] >= h_vars[i] + h_vars[j] - 1)
                        constraints.append(z_vars[i,j] <= h_vars[i])
                        constraints.append(z_vars[i,j] <= h_vars[j])
                        constraints.append(z_vars[i,j] >= 0)
            
            # Set up overlap constraints for all lags
            for k in range(-n_points+1, n_points):
                if k >= 0:
                    # Positive lag
                    lag_indices = range(n_points - k)
                    expr = sum(h_vars[i] - z_vars[i, i+k] for i in lag_indices)
                else:
                    # Negative lag
                    lag_indices = range(-k, n_points)
                    expr = sum(h_vars[i] - z_vars[i+k, i] for i in lag_indices)
                constraints.append(expr <= C)
            
            # Minimize the maximum overlap
            objective = cp.Minimize(C)
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_refined = h_vars.value
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_check = evaluate_erdos_solution(h_refined, dx)
                if c5_check < best_c5:
                    best_c5 = c5_check
                    best_h = h_refined.copy()
        except Exception:
            pass

    # Local refinement using SLSQP
    remaining_time = budget_s * 0.3
    if remaining_time > 1.0:
        # Reinitialize with the best solution
        h_current = best_h.copy()
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_current, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
