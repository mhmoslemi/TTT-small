import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    population_size = 50
    num_generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 3
    no_improvement_limit = 100
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def create_population():
        pop = [h_start.copy()]
        for _ in range(population_size - 1):
            pop.append(create_individual())
        return pop
    
    def fitness(individual):
        return evaluate_erdos_solution(individual, dx)
    
    def tournament_selection(population, fitnesses):
        selected = []
        for _ in range(population_size):
            tournament = random.sample(list(zip(population, fitnesses)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0].copy())
        return selected
    
    def blend_crossover(parent1, parent2, alpha=0.5):
        child1 = np.zeros(n_points)
        child2 = np.zeros(n_points)
        for i in range(n_points):
            gamma = (1 + 2 * alpha) * np.random.rand() - alpha
            child1[i] = (1 - gamma) * parent1[i] + gamma * parent2[i]
            child2[i] = (1 - gamma) * parent2[i] + gamma * parent1[i]
        return child1, child2
    
    def mutate(individual, mutation_rate, generation, max_generations):
        step_size = 0.05 * (1 - generation / max_generations)
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.normal(0, step_size)
        # Normalize to maintain constraint
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        return individual
    
    population = create_population()
    generation = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.9:
        # Evaluate fitness
        fitnesses = [fitness(ind) for ind in population]
        
        # Track best solution
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitnesses.index(min_fitness)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= no_improvement_limit:
            break
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Crossover and mutation
        new_population = [best_h.copy()]
        for i in range(0, population_size - 1, 2):
            if np.random.rand() < crossover_rate and i + 1 < population_size:
                child1, child2 = blend_crossover(selected[i], selected[i+1])
                child1 = mutate(child1, mutation_rate, generation, num_generations)
                child2 = mutate(child2, mutation_rate, generation, num_generations)
                new_population.extend([child1, child2])
            else:
                new_population.append(mutate(selected[i], mutation_rate, generation, num_generations))
                if len(new_population) < population_size:
                    new_population.append(mutate(selected[i+1], mutation_rate, generation, num_generations))
        
        # Trim to regular population size if too many
        population = new_population[:population_size]
        
        generation += 1
        
        # Occasionally refine best solution with SLSQP
        if generation % 10 == 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-12})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    return (list(best_h), best_c5, n_points)
