import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    # Adjust sum to be exactly n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points/2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = project_to_feasible(h_current, n_points)
    
    # PSO parameters
    num_particles = 75
    inertia_start = 0.8
    inertia_end = 0.4
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    max_iter = int(budget_s * 0.9 / 0.01)  # Estimate iterations
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    particle_best_positions = []
    particle_best_fitness = []
    
    for _ in range(num_particles):
        h_init = h_current + np.random.normal(0, 0.05, n_points)
        h_init = project_to_feasible(h_init, n_points)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
        particle_best_positions.append(h_init.copy())
        particle_best_fitness.append(evaluate_erdos_solution(h_init, dx))
    
    # Global best
    global_best_idx = np.argmin(particle_best_fitness)
    global_best_position = particle_best_positions[global_best_idx].copy()
    global_best_fitness = particle_best_fitness[global_best_idx]
    
    # Optimization loop
    iter_count = 0
    last_improvement = 0
    for iteration in range(max_iter):
        elapsed = time() - start_time
        if elapsed > budget_s * 0.95:
            break
            
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * (iteration / max_iter)
        
        # Update each particle
        for i in range(num_particles):
            # Get random values for velocity update
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            # Update velocity
            velocities[i] = (inertia * velocities[i] + 
                           cognitive_weight * r1 * (particle_best_positions[i] - particles[i]) +
                           social_weight * r2 * (global_best_position - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible set
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate fitness
            fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update particle best
            if fitness < particle_best_fitness[i]:
                particle_best_fitness[i] = fitness
                particle_best_positions[i] = particles[i].copy()
                
                # Update global best if needed
                if fitness < global_best_fitness:
                    global_best_fitness = fitness
                    global_best_position = particles[i].copy()
                    last_improvement = iteration
            
        # Periodic perturbation to escape local minima
        if iteration % 1000 == 0 and iteration > 0:
            for i in range(num_particles):
                if np.random.rand() < 0.01:
                    particles[i] += np.random.normal(0, 0.02, n_points)
                    particles[i] = project_to_feasible(particles[i], n_points)
        
        iter_count += 1
        
        # Early termination if no improvement for a while
        if iteration - last_improvement > 5000:
            break
    
    # Final refinement with local search around best solution
    best_h = global_best_position.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # One final gradient-based fine-tune
    step_size = 0.005
    for _ in range(100):
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(best_h) - 1)
        
        # Compute gradient at max lag
        gradient = np.zeros(len(best_h))
        if max_lag >= 0:
            for i in range(len(best_h) - max_lag):
                gradient[i] += 1 - best_h[i + max_lag]
                if i + max_lag < len(best_h):
                    gradient[i + max_lag] -= best_h[i]
        else:
            for i in range(len(best_h) + max_lag):
                gradient[i] += 1 - best_h[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= best_h[i]
        
        # Gradient descent step
        h_new = best_h - step_size * gradient
        h_new = project_to_feasible(h_new, n_points)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        else:
            step_size *= 0.95
            if step_size < 1e-6:
                break
    
    return (list(best_h), best_c5, n_points)
