import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def binary_to_h(binary_vec, n_points):
    """Convert binary representation to step function h"""
    h = np.zeros(n_points)
    # Simple approach: create piecewise constant function
    # Each 1 in binary vector represents a 1 interval
    step_size = n_points // len(binary_vec)
    if step_size == 0:
        step_size = 1
    for i, bit in enumerate(binary_vec):
        start = i * step_size
        end = min(start + step_size, n_points)
        h[start:end] = bit
    return h

def h_to_binary(h, n_points, n_intervals=10):
    """Convert h to a binary representation (approximate)"""
    # Split interval into n_intervals segments
    intervals = []
    for i in range(n_intervals):
        start = i * n_points // n_intervals
        end = (i + 1) * n_points // n_intervals
        segment_mean = np.mean(h[start:end])
        # Convert to binary: if > 0.5, then 1, else 0
        intervals.append(1 if segment_mean > 0.5 else 0)
    return np.array(intervals)

def mutate_binary(binary_vec, mutation_rate=0.05):
    """Mutate binary vector"""
    new_binary = binary_vec.copy()
    for i in range(len(new_binary)):
        if random.random() < mutation_rate:
            new_binary[i] = 1 - new_binary[i]  # Flip bit
    return new_binary

def crossover_binary(b1, b2):
    """Single point crossover"""
    if len(b1) != len(b2):
        return b1, b2
    point = random.randint(1, len(b1)-1)
    child1 = np.concatenate([b1[:point], b2[point:]])
    child2 = np.concatenate([b2[:point], b1[point:]])
    return child1, child2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        binary_start = h_to_binary(h_start, n_points)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        binary_start = np.random.randint(0, 2, size=10)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_binary = binary_start.copy()
    
    # Genetic algorithm parameters
    pop_size = 100
    max_generations = 200
    elite_size = pop_size // 5
    mutation_rate = 0.05
    crossover_prob = 0.7
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if random.random() < 0.5:
            binary = binary_start.copy()
            # Add some randomness
            for i in range(len(binary)):
                if random.random() < 0.1:
                    binary[i] = 1 - binary[i]
        else:
            binary = np.random.randint(0, 2, size=10)
        population.append(binary)
    
    no_improvement = 0
    patience = 50
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate population
        fitness_scores = []
        for binary in population:
            h = binary_to_h(binary, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points/2) > 1e-6:
                penalty = abs(h_sum - n_points/2) * 100
            else:
                penalty = 0
            c5 = evaluate_erdos_solution(h, dx) + penalty
            fitness_scores.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Check improvement
        if fitness_scores[0] < best_c5:
            best_c5 = fitness_scores[0]
            best_binary = population[0].copy()
            best_h = binary_to_h(best_binary, n_points)
            best_h = np.clip(best_h, 0, 1)
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement >= patience:
            break
        
        # Selection
        elites = population[:elite_size]
        
        # Crossover and mutation
        new_population = elites.copy()
        while len(new_population) < pop_size:
            if random.random() < crossover_prob:
                parent1, parent2 = random.sample(elites, 2)
                child1, child2 = crossover_binary(parent1, parent2)
            else:
                parent = random.choice(elites)
                child1 = parent.copy()
                child2 = parent.copy()
            
            # Mutate
            child1 = mutate_binary(child1, mutation_rate)
            child2 = mutate_binary(child2, mutation_rate)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population
    
    # Refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Start with best binary solution
        refined_h = binary_to_h(best_binary, n_points)
        refined_h = np.clip(refined_h, 0, 1)
        refined_h = refined_h * (n_points/2) / np.sum(refined_h)
        
        res = minimize(objective, refined_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
