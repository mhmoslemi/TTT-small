import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 35
    population_size = 50
    crossover_rate = 0.8
    mutation_rate = 0.1
    max_generations_no_improvement = 100
    slsqp_refine_freq = 100
    
    # Initialize population
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
    else:
        population = []
    while len(population) < population_size:
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    best_generation = 0
    
    # Track generation
    generation = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.7:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, 2.0 / n_points)
            fitness.append(fit)
        
        # Find best in population
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            best_generation = generation
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_generations_no_improvement:
            break
        
        # Tournament selection
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx])
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(best_h.copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if np.random.random() < crossover_rate:
                parent1 = selected[np.random.randint(0, population_size)]
                parent2 = selected[np.random.randint(0, population_size)]
                
                # Blend crossover
                alpha = np.random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
                
                # Ensure sum constraint
                child_sum = np.sum(child)
                child = child * (n_points / 2) / child_sum
                
                # Mutation
                if np.random.random() < mutation_rate:
                    mut_strength = 0.05 * (0.95 ** generation)
                    for i in range(n_points):
                        if np.random.random() < 0.1:
                            child[i] += np.random.normal(0, mut_strength)
                    # Clamp to [0,1] and adjust sum
                    child = np.clip(child, 0, 1)
                    child_sum = np.sum(child)
                    child = child * (n_points / 2) / child_sum
                    
            else:
                # Just mutate one of the selected parents
                parent = selected[np.random.randint(0, population_size)]
                child = parent.copy()
                mut_strength = 0.05 * (0.95 ** generation)
                for i in range(n_points):
                    if np.random.random() < 0.1:
                        child[i] += np.random.normal(0, mut_strength)
                # Clamp to [0,1] and adjust sum
                child = np.clip(child, 0, 1)
                child_sum = np.sum(child)
                child = child * (n_points / 2) / child_sum
            
            new_population.append(child)
        
        population = new_population
    
    # Final local refinement with SLSQP if needed
    if time() - start_time < budget_s * 0.95:
        dx = 2.0 / n_points

        def objective(h):
            return evaluate_erdos_solution(h, dx)

        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0

        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]

        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass

    return (list(best_h), best_c5, n_points)
