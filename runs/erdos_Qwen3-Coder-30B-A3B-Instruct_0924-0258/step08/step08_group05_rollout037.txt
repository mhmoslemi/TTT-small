import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Trust region with cutting planes approach
    trust_radius = 0.2
    max_iter = 2000
    improvement_threshold = 1e-8
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        current_time = time()
        
        # Build problem for trust region subproblem
        h_vars = cp.Variable(n_points)
        C = cp.Variable()
        
        # Objective
        objective = cp.Minimize(C)
        
        # Constraints
        constraints = [cp.sum(h_vars) == n_points/2.0]
        constraints += [h_vars >= 0, h_vars <= 1]
        
        # Generate all overlap constraints
        corr_max = 0
        # Evaluate constraint violations in current solution
        # Get lag indices
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        lags = np.arange(len(corr)) - (len(h_current) - 1)
        overlap_values = corr * dx
        
        # Add most violated constraints
        sorted_indices = np.argsort(overlap_values)[::-1]
        num_constraints = min(50, len(sorted_indices))
        for i in range(num_constraints):
            idx = sorted_indices[i]
            lag = lags[idx]
            if lag >= 0:
                # Valid forward correlation
                valid_len = len(h_current) - lag
                if valid_len > 0:
                    # h[i] * (1 - h[i+lag])
                    expr = 0
                    for j in range(valid_len):
                        expr += h_vars[j] * (1 - h_vars[j + lag])
                    constraints.append(expr <= C)
                    corr_max = max(corr_max, overlap_values[idx])
            else:
                # Valid backwards correlation
                valid_len = len(h_current) + lag
                if valid_len > 0:
                    expr = 0
                    for j in range(valid_len):
                        expr += h_vars[j] * (1 - h_vars[j + lag])
                    constraints.append(expr <= C)
                    corr_max = max(corr_max, overlap_values[idx])
        
        # Trust region constraint
        h_current_var = cp.Parameter(n_points, value=h_current)
        constraints.append(cp.sum_squares(h_vars - h_current_var) <= trust_radius**2)
        
        # Solve subproblem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=CPLEX, verbose=False, time_limit=30)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                # Ensure constraint satisfaction
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                
                # Evaluate current solution
                c5_current = evaluate_erdos_solution(h_new, dx)
                if c5_current < best_c5:
                    best_c5 = c5_current
                    best_h = h_new.copy()
                    
                # Adjust trust region based on improvement
                improvement = best_c5 - c5_current
                if improvement > improvement_threshold:
                    trust_radius = min(0.8 * trust_radius, 0.1)
                else:
                    trust_radius = min(1.1 * trust_radius, 0.5)
                    
                h_current = h_new.copy()
            else:
                trust_radius *= 0.8
        except Exception:
            trust_radius *= 0.8
            continue
    
    # Final refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
