import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search on C5 with SDP relaxation approach
    low = 0.38
    high = 0.38085
    tol = 1e-5
    
    while time() - start_time < budget_s * 0.95 and high - low > tol:
        mid = (low + high) / 2
        c5_target = mid
        
        # SDP formulation
        h_var = cp.Variable(n_points)
        Y = cp.Variable((n_points, n_points), symmetric=True)
        
        # Constraints
        constraints = [
            Y >> 0,  # Positive semidefinite
            cp.diag(Y) == h_var,  # Diagonal constraint
            cp.sum(h_var) == n_points / 2,  # Measure constraint
        ]
        
        # Overlap constraints for a sampling of shifts
        shifts = np.random.randint(1, n_points, size=min(100, n_points))
        for k in shifts:
            # Sum over all valid i where i+k < n_points
            expr = 0
            for i in range(n_points):
                if i + k < n_points:
                    expr += h_var[i] - Y[i, i+k]
            constraints.append(expr <= c5_target * dx)
        
        # Solve SDP
        objective = cp.Minimize(0)  # Dummy objective
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h_new = np.clip(h_new, 0, 1)
                    # Renormalize
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    
                    # Local refinement
                    def objective_func(h_vec):
                        return evaluate_erdos_solution(h_vec, dx)
                    
                    # Gradient-based refinement
                    for _ in range(500):
                        if time() - start_time > budget_s * 0.95:
                            break
                        corr = np.correlate(h_new, 1 - h_new, mode='full')
                        max_lag_idx = np.argmax(corr)
                        max_lag = max_lag_idx - (n_points - 1)
                        grad = np.zeros(n_points)
                        for i in range(n_points):
                            j = i + max_lag
                            if 0 <= j < n_points:
                                grad[i] = 1 - 2 * h_new[i] - h_new[j]
                            else:
                                grad[i] = 1 - 2 * h_new[i]
                        h_new = h_new - 0.001 * grad
                        h_new = np.clip(h_new, 0, 1)
                        h_new = h_new * (n_points / 2) / np.sum(h_new)
                    
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        if c5_new < 0.3807:
                            return (list(best_h), best_c5, n_points)
                    
                    low = mid
                else:
                    high = mid
            else:
                high = mid
                
        except Exception:
            # Likely infeasible, try lower bound
            high = mid
    
    # Final refinement
    def objective_func(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    # Local optimization around the best solution
    for _ in range(1000):
        if time() - start_time > budget_s * 0.95:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * best_h[i] - best_h[j]
            else:
                grad[i] = 1 - 2 * best_h[i]
        best_h = best_h - 0.001 * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
