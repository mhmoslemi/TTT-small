import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA Parameters
    population_size = 50
    crossover_rate = 0.8
    mutation_rate = 0.1
    max_generations = 2000
    no_improvement_limit = 100
    
    # Initialize population
    population = []
    population.append(h_start.copy())
    
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    def mutate(h, generation):
        h_new = h.copy()
        step_size = 0.05 * (0.95 ** generation)
        for i in range(n_points):
            if random.random() < mutation_rate:
                h_new[i] += random.gauss(0, step_size)
        # Normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        return h_new
    
    def crossover(h1, h2):
        if random.random() > crossover_rate:
            return h1.copy()
        alpha = random.random()
        h_new = alpha * h1 + (1 - alpha) * h2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        return h_new
    
    # Selection
    def tournament_selection(pop, k=3):
        selected = random.sample(pop, k)
        fitnesses = [evaluate_erdos_solution(h, dx) for h in selected]
        return selected[np.argmin(fitnesses)]
    
    # GA Loop
    no_improvement = 0
    generation = 0
    
    while time() - start_time < budget_s * 0.7 and generation < max_generations:
        # Evaluate current population
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[np.argmin(fitnesses)].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        if no_improvement > no_improvement_limit:
            break
        
        # Create new population
        new_population = []
        # Keep best
        new_population.append(population[np.argmin(fitnesses)].copy())
        
        while len(new_population) < population_size:
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child = crossover(parent1, parent2)
            child = mutate(child, generation)
            new_population.append(child)
        
        population = new_population
        
        generation += 1
    
    # Final SLSQP refinement on best
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
