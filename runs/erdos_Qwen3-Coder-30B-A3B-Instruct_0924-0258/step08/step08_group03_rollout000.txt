import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimizer parameters
    max_iter = 500
    filter_width = 5
    max_k = 20
    k_inc = 2
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Select top K overlaps to focus on in convex optimization
        corr = np.correlate(h, 1 - h, mode='full')
        top_k_indices = np.argpartition(-corr, max_k)[:max_k]
        k = min(max_k, len(top_k_indices))
        selected_indices = np.sort(top_k_indices[-k:])
        
        # Convex optimization setup
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Minimize sum of selected overlaps, using linearization
        objective = 0
        for idx in selected_indices:
            # Approximate with convolution constraints
            # This is a simplified approach - we compute convolution terms directly
            offset = idx - (n_points - 1)
            obj_term = 0
            for i in range(n_points):
                j = i + offset
                if 0 <= j < n_points:
                    obj_term += h_var[i] * (1 - h_var[j])
                else:
                    obj_term += h_var[i] * (1 - 0)  # treat outside as 0
            objective += obj_term
        
        # Try to minimize the sum of selected correlations
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=500)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                # Apply low-pass filtering to smooth
                h_filtered = np.zeros(n_points)
                for i in range(n_points):
                    total = 0
                    count = 0
                    for j in range(max(0, i - filter_width), min(n_points, i + filter_width + 1)):
                        total += h_new[j]
                        count += 1
                    h_filtered[i] = total / count if count > 0 else 0
                h_new = h_filtered
                
                # Ensure constraints
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    # Adjust k for next iteration
                    if k < max_k:
                        max_k = min(max_k, k + k_inc)
                else:
                    # Reducing k and filter width if stagnating
                    max_k = max(5, k - 1)
                    filter_width = max(1, filter_width - 1)
            else:
                # If optimization fails, just use a small step refinement
                # Simple gradient-like update
                h_new = h.copy()
                # For simplicity, let's try small random perturbations
                perturb = np.random.uniform(-0.01, 0.01, n_points)
                h_new += perturb
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
        except Exception:
            # Fallback if problem doesn't solve numerically
            h_new = h.copy()
            perturb = np.random.uniform(-0.01, 0.01, n_points)
            h_new += perturb
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
