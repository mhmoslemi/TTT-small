import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter = 100000
    epsilon = 0.1
    iter_count = 0
    
    # Initialize with normalized sum
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    current_c5 = evaluate_erdos_solution(h, dx)
    
    T = T_init
    while T > T_min and (time() - start_time) < budget_s * 0.9:
        if iter_count >= max_iter:
            break
            
        # Make a perturbation
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-epsilon, epsilon)
        
        # Apply perturbation with sum constraint
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure boundary constraints
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        sum_diff = np.sum(h_new) - n_points/2
        if abs(sum_diff) > 1e-10:
            # Adjust first element to correct sum
            h_new[0] -= sum_diff
        
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                h = h_new.copy()
                current_c5 = new_c5
        
        # Cooling schedule
        T *= alpha
        iter_count += 1
        
        # Occasionally adjust epsilon
        if iter_count % 1000 == 0 and epsilon > 0.001:
            epsilon *= 0.95

    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return np.max(np.correlate(h, 1 - h, mode='full') * dx)
        
        h_refined = best_h.copy()
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        res = minimize(
            objective,
            h_refined,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            h_final = res.x
            h_final = h_final * (n_points/2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
