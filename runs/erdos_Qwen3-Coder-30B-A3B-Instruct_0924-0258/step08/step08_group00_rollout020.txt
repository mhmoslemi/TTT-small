import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def repair(h, n_points):
    h = np.clip(h, 0, 1)
    if np.sum(h) == 0:
        return h
    h = h * (n_points/2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-start approach with Latin hypercube sampling
    n_starts = 50
    n_starts = min(n_starts, 10000 // n_points)  # Avoid too many starts for small n
    
    # Generate diverse initial guesses using Latin hypercube sampling
    initial_guesses = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = repair(h0, n_points)
        initial_guesses.append(h0)
    
    # Generate Latin hypercube samples
    for _ in range(n_starts - len(initial_guesses)):
        # Generate samples from uniform distribution
        samples = np.random.uniform(0, 1, n_points)
        # Normalize to satisfy sum constraint
        h0 = samples * (n_points/2) / np.sum(samples)
        initial_guesses.append(h0)
    
    # Optimize each initial guess using SLSQP
    for i, h0 in enumerate(initial_guesses):
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Ensure feasibility before optimization
            h0 = repair(h0, n_points)
            
            res = minimize(objective, h0, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                h_final = repair(h_final, n_points)
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
