import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    # Normalize the starting point to satisfy sum(h) = n_points / 2
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # Fix the value for SDP based on a suitable approach
    # Utilize CVXPY with a modern SDP solver (e.g., MOSEK or ECOS)
    # Optimization using convex relaxation (SDP formulation)

    # Initialize the continuous variable
    h = cp.Variable(n_points, name='h')
    # Objective: minimize the maximum overlap (C5)
    C5 = cp.Variable(name='C5')

    # Constraints
    constraints = []
    # Add constraints for h being in [0,1]
    constraints.extend([h >= 0, h <= 1])
    
    # Add constraint: sum(h) = n_points / 2
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Add constraint: C5 >= \int h(x)(1 - h(x+k)) dx for all k
    # Compute all valid lags
    shift_range = range(1 - n_points, n_points)
    
    for k in shift_range:
        # Compute valid indices: i, i+k such that both are in [0, n_points)
        valid_indices = []
        valid_indices_k = []
        for i in range(n_points):
            j = i + k
            if 0 <= i < n_points and 0 <= j < n_points:
                valid_indices.append(i)
                valid_indices_k.append(j)
        
        if valid_indices:
            # Sum over the valid pairs 
            expr = cp.sum(cp.multiply(h[valid_indices], 1 - h[valid_indices_k])) * dx
            constraints.append(C5 >= expr)
    
    # Formulate objective
    objective = cp.Minimize(C5)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with ECOS as a fallback
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s - 5)
        result = h.value
        if result is not None and np.isfinite(result).all():
            h_opt = np.array(result)
        else:
            h_opt = h_start
    except:
        try:
            prob.solve(solver=cp.ECOS, verbose=False, time_limit=budget_s - 5)
            result = h.value
            if result is not None and np.isfinite(result).all():
                h_opt = np.array(result)
            else:
                h_opt = h_start
        except:
            h_opt = h_start
    
    # Final validation of h_opt
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points/2) / np.sum(h_opt)
    
    # Evaluate the final C5
    final_c5 = evaluate_erdos_solution(h_opt, dx)
    
    return (list(h_opt), final_c5, n_points)
