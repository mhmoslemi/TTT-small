import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex perturbation optimization
    max_iter = 5000
    epsilon0 = 0.001
    beta = 0.99
    top_k = 100
    tolerance = 1e-6
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlations
        corr = np.correlate(h, 1 - h, mode='full')
        
        # Select top shifts
        top_shifts = np.argsort(corr)[-top_k:]
        
        # Formulate convex optimization problem
        h_var = cp.Variable(n_points)
        
        # Objective: sum of shifted correlations + convex perturbation
        obj = 0
        for shift in top_shifts:
            if shift < n_points:
                obj += cp.sum(cp.multiply(h_var, 1 - h_var[shift]))
            else:
                obj += cp.sum(cp.multiply(h_var, 1))
        
        # Add convex regularization term
        epsilon_t = epsilon0 * (beta ** iteration)
        obj += epsilon_t * cp.norm(h_var, 2)**2
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Solve problem
        prob = cp.Problem(cp.Minimize(obj), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=100)
            if prob.status == cp.OPTIMAL:
                h_new = np.array(h_var.value).reshape(-1)
            else:
                # Fall back to gradient method if optimization fails
                h_new = h.copy()
        except:
            # Fall back to gradient method if optimization fails
            h_new = h.copy()
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update best
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improvement = iteration
        
        h = h_new
        
        # Early termination if no improvement in a while
        if iteration - last_improvement > 500:
            break

    # Final local refinement with random sampling
    if time() - start_time < budget_s * 0.9:
        current_best = best_h.copy()
        for _ in range(100):
            # Generate random perturbation
            pert = np.random.normal(0, 0.01, n_points)
            h_try = current_best + pert
            h_try = np.clip(h_try, 0, 1)
            h_try = h_try * (n_points / 2) / np.sum(h_try)
            
            c5_try = evaluate_erdos_solution(h_try, dx)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    return (list(best_h), best_c5, n_points)
