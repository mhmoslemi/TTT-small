import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Trust-region optimizer
    max_iter = 5000
    trust_radius = 0.2
    min_trust_radius = 1e-6
    trust_radius_factor = 1.2
    
    def project_onto_sum_constraint(h, target_sum):
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        return h + diff / len(h)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Solve convex relaxation
        h_current = np.clip(h_current, 0, 1)
        h_current = project_onto_sum_constraint(h_current, n_points / 2)
        
        # Evaluate and find worst lag
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        overlap_values = corr * dx
        max_idx = np.argmax(overlap_values)
        max_val = overlap_values[max_idx]
        
        # Break if improvement is too small
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h_current.copy()
        
        # Approximate with a trust region subproblem
        # We limit to a subproblem focusing on the main constraint
        # Use simple gradient descent at this moment with projection
        gradient = np.zeros_like(h_current)
        lag = max_idx - (len(h_current) - 1)
        
        # Construct a simplified gradient estimate
        for i in range(len(h_current)):
            if i + lag < len(h_current) and i + lag >= 0:
                gradient[i] -= (1 - h_current[i + lag])
            elif i - lag >= 0 and i - lag < len(h_current):
                gradient[i] -= (1 - h_current[i - lag])
        
        # Take a step in the direction of negative gradient
        h_new = h_current - 0.01 * gradient
        
        # Project onto sum and boundaries
        h_new = np.clip(h_new, 0, 1)
        h_new = project_onto_sum_constraint(h_new, n_points / 2)
        
        # Line search for better step size (simplified)
        step_size = 0.01
        for _ in range(5):
            trial_h = h_current - step_size * gradient
            trial_h = np.clip(trial_h, 0, 1)
            trial_h = project_onto_sum_constraint(trial_h, n_points / 2)
            
            trial_c5 = evaluate_erdos_solution(trial_h, dx)
            if trial_c5 < best_c5:
                h_current = trial_h
                break
            step_size *= 0.5
            
        # Reduce trust region if no improvement
        if iteration % 20 == 0:
            if evaluate_erdos_solution(h_current, dx) >= 0.99 * best_c5:
                trust_radius = max(min_trust_radius, trust_radius * 0.9)
            else:
                trust_radius = min(0.5, trust_radius * trust_radius_factor)
    
    # SLSQP refinement at the end
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
