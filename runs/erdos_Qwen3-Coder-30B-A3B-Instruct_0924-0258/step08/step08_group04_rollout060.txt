import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # Initialize PSO parameters
    num_particles = 75
    inertia = 0.8
    inertia_decay = 0.999
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    particle_best = []
    particle_best_fitness = []
    
    for _ in range(num_particles):
        h = h_start.copy()
        # Add small random perturbation
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        fitness = evaluate_erdos_solution(h, dx)
        particle_best.append(h.copy())
        particle_best_fitness.append(fitness)
    
    global_best_idx = np.argmin(particle_best_fitness)
    global_best = particle_best[global_best_idx].copy()
    global_best_fitness = particle_best_fitness[global_best_idx]
    
    max_iter = int(budget_s * 0.9 * 1000)
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 5000
    
    # PSO loop
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for p in range(num_particles):
            h = particles[p]
            v = velocities[p]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_weight * r1 * (particle_best[p] - h)
            social = social_weight * r2 * (global_best - h)
            v = inertia * v + cognitive + social
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            velocities[p] = v
            
            # Update position
            h_new = h + v
            
            # Apply boundary constraints
            h_new = np.clip(h_new, 0, 1)
            # Adjust to maintain sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate
            fitness_new = evaluate_erdos_solution(h_new, dx)
            
            # Update particle best
            if fitness_new < particle_best_fitness[p]:
                particle_best[p] = h_new.copy()
                particle_best_fitness[p] = fitness_new
                
                # Update global best
                if fitness_new < global_best_fitness:
                    global_best = h_new.copy()
                    global_best_fitness = fitness_new
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
                
            # Update particle
            particles[p] = h_new.copy()
            
        # Update inertia
        inertia = max(0.4, inertia * inertia_decay)
        
        # Perturb some particles occasionally to avoid local optima
        if iter_count % 1000 == 0:
            for p in range(num_particles):
                if np.random.rand() < 0.01:
                    particles[p] += np.random.normal(0, 0.02, n_points)
                    particles[p] = np.clip(particles[p], 0, 1)
                    particles[p] = particles[p] * (n_points / 2) / np.sum(particles[p])
    
    # Final local refinement of best solution
    h_final = global_best.copy()
    dx_final = dx
    
    # Return best solution found
    return (list(h_final), global_best_fitness, n_points)
