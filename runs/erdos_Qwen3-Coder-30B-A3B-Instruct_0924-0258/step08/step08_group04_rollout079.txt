import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Mutate to more diverse solutions:
    # Simple local search enhancement
    
    # Start with a genetic approach on a coarse grid to avoid local optima
    coarse_points = 15
    dx_coarse = 2.0 / coarse_points
    population_size = 50
    mutation_rate = 0.2
    
    # Initial population on coarse scale
    population = []
    for _ in range(population_size):
        h = np.random.rand(coarse_points)
        h = h * (coarse_points/2) / np.sum(h)
        population.append(h)
    
    # Evaluate and select
    fitness = []
    for i, h in enumerate(population):
        c5 = evaluate_erdos_solution(h, dx_coarse)
        fitness.append(-c5)
    
    # Selection, crossover, mutation
    max_iter = 1000
    
    # Coarse optimization
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.2:
            break
            
        # Sort by fitness (negative C5)
        idx_sorted = np.argsort(fitness)
        top_indices = idx_sorted[:population_size//2]
        top_pop = [population[i] for i in top_indices]
        
        # Create new population
        new_pop = top_pop.copy()
        while len(new_pop) < population_size:
            parent1 = np.random.choice(top_pop)
            parent2 = np.random.choice(top_pop)
            # Crossover
            if np.random.rand() < 0.5:
                split = np.random.randint(1, coarse_points)
                child = np.concatenate([parent1[:split], parent2[split:]])
            else:
                child = (parent1 + parent2) / 2
            # Mutation
            if np.random.rand() < mutation_rate:
                mut_idx = np.random.randint(0, coarse_points)
                child[mut_idx] = np.random.rand()
            # Normalize
            child = child * (coarse_points/2) / np.sum(child)
            new_pop.append(child)
        
        population = new_pop
        
        # Evaluate new population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx_coarse)
            fitness.append(-c5)
    
    # Take best of coarse
    best_pop_idx = np.argmin(fitness)
    h_coarse = population[best_pop_idx]
    
    # Upscale to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, coarse_points), 
                       h_coarse)
    h_fine = h_fine * (n_points/2) / np.sum(h_fine)
    h_current = h_fine.copy()
    
    # Compute corresponding C5
    c5_current = evaluate_erdos_solution(h_current, dx)
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Refinement with gradient steps on fine grid
    max_iter_fine = 10000
    step_size = 0.01
    
    for iter_count in range(max_iter_fine):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Get maximum overlap lag
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Compute gradient at max lag
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_current[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= h_current[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_current[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_current[i]
        
        # Gradient descent step
        h_new = h_current - step_size * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new C5
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        if c5_new >= evaluate_erdos_solution(h_current, dx):
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            h_current = h_new
            step_size *= 1.05
            
    return (list(best_h), best_c5, n_points)
