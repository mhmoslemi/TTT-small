import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based local search with smarter initialization
    max_iter = 1000
    patience = 50
    no_improvement = 0
    step_size = 0.01
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate overlap for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            break
            
        # Identify lag with maximum overlap
        max_lag = max_lag_idx - (n_points - 1)  # Convert index to actual lag
        
        # Compute gradient w.r.t. to h for this specific lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 2 * (1 - h[j] - h[i]) / n_points
        
        # Update with gradient descent
        h_new = h - step_size * grad
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check if the update improved the objective
        if evaluate_erdos_solution(h_new, dx) < evaluate_erdos_solution(h, dx):
            h = h_new
        else:
            # Reduce step size if no improvement
            step_size *= 0.5
            if step_size < 1e-8:
                break
    
    # Add more sophisticated local search using a hybrid approach
    try:
        # Start with a few good candidates from the elite population
        candidate_h = best_h.copy()
        # Try different local optimization methods
        method_list = ['L-BFGS-B', 'TNC', 'SLSQP']
        
        for method in method_list:
            if time() - start_time > budget_s * 0.9:
                break
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(objective, candidate_h, method=method, 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-9})
                
                if res.success:
                    h_final = res.x
                    c5_final = objective(h_final)
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
                        
            except Exception:
                continue
                
        # Try a random restart with better initialization
        if time() - start_time < budget_s * 0.8:
            for _ in range(5):
                if time() - start_time > budget_s * 0.9:
                    break
                h_test = np.random.rand(n_points)
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points/2) / np.sum(h_test)
                c5_test = evaluate_erdos_solution(h_test, dx)
                if c5_test < best_c5:
                    best_h = h_test
                    best_c5 = c5_test
                    
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
