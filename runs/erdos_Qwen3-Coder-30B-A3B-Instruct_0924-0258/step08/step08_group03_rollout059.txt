import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search bounds
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    # Sample shifts for feasibility checking
    shift_samples = min(100, n_points)
    
    # Helper function to check feasibility
    def is_feasible(c5_val):
        # Create variables
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Add overlap constraints
        for k in range(shift_samples):
            # Compute the sum: sum h[i] * (1 - h[(i+k) % n])
            overlap_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_sum += h_vars[i] * (1 - h_vars[j])
            
            constraints.append(overlap_sum <= c5_val * dx)
        
        # Create and solve problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            return prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INACCURATE
        except:
            return False
    
    # Binary search
    while time() - start_time < budget_s * 0.9 and high - low > tolerance:
        mid = (low + high) / 2
        if is_feasible(mid):
            high = mid
            # Update best if better
            if mid < best_c5:
                # Reoptimize to get better h
                h_vars = cp.Variable(n_points)
                constraints = [
                    h_vars >= 0,
                    h_vars <= 1,
                    cp.sum(h_vars) == n_points / 2
                ]
                
                # Optimization for improved h
                for k in range(shift_samples):
                    overlap_sum = 0
                    for i in range(n_points):
                        j = (i + k) % n_points
                        overlap_sum += h_vars[i] * (1 - h_vars[j])
                    constraints.append(overlap_sum <= mid * dx)
                
                # Add objective to minimize max overlap over a few shifts
                obj = cp.Minimize(0)
                prob = cp.Problem(obj, constraints)
                try:
                    prob.solve(solver=cp.CVXOPT, verbose=False)
                    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INACCURATE:
                        new_h = h_vars.value
                        if new_h is not None:
                            new_h = np.maximum(0, np.minimum(1, new_h))
                            new_h = new_h * (n_points / 2) / np.sum(new_h)
                            new_c5 = evaluate_erdos_solution(new_h, dx)
                            if new_c5 < best_c5:
                                best_c5 = new_c5
                                best_h = new_h.copy()
                except:
                    pass
        else:
            low = mid

    # Local refinement with gradient descent
    h = best_h.copy()
    dx = 2.0 / n_points
    max_it = 500
    
    for it in range(max_it):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap values
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Gradient approximation
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h[i] - h[j]
            else:
                grad[i] = 1 - 2 * h[i]
        
        # Gradient step
        h_new = h - 0.1 * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
