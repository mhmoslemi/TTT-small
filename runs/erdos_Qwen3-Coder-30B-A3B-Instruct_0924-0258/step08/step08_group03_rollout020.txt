import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex perturbation parameters
    epsilon_0 = 0.001
    beta = 0.99
    max_iter = 5000
    top_k = 100
    
    # Adaptive perturbation
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        
        # Get top_k dominant lags to regularize
        top_lags = np.argsort(corr)[-top_k:]
        
        # Formulate convex problem with quadratic perturbation
        h_var = cp.Variable(n_points)
        objective = 0
        
        # Build convex objective: minimize sum of top_k overlaps + epsilon * ||h||^2
        for i, k in enumerate(top_lags):
            # Compute convolution for specific lag k
            if k < n_points:
                # Only compute when valid
                diff_lag = np.zeros(n_points)
                diff_lag[k:] = h[:n_points-k]
                conv_term = cp.sum(cp.multiply(h_var, diff_lag))
            else:
                conv_term = 0
            
            # For simplicity just use first-order approx of correlations
            # Note: This simplification is for avoid complex convolution functions
            # We'll use a hybrid approach for better convergence
            objective += conv_term
            
        # Add convex perturbation
        epsilon = epsilon_0 * (beta ** iteration)
        objective += epsilon * cp.sum_squares(h_var)
        
        # Define constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Solve convex optimization problem using CVXPY
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            h_new = h_var.value
            if h_new is not None:
                h_new = project_to_feasible(h_new, n_points, dx)
                
                # Evaluate new solution 
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                # Update best if improved
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
        except Exception:
            pass  # Fallback to gradient descent if convex solve fails
            
        # Gradient-based refinement for robustness
        if iteration % 100 == 0:
            corr = np.correlate(h, 1 - h, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (n_points - 1)
            
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + max_lag
                if 0 <= j < n_points:
                    grad[i] = 1 - 2 * h[i] - h[j]
                else:
                    grad[i] = 1 - 2 * h[i]
                    
            step_size = 0.01
            h_new = h - step_size * grad
            h_new = project_to_feasible(h_new, n_points, dx)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
    
    # Final post-processing: Generate nearby points for local refinement
    for _ in range(100):
        if time() - start_time > budget_s * 0.95:
            break
        perturb = np.random.normal(0, 0.001, n_points)
        h_candidate = h + perturb
        h_candidate = project_to_feasible(h_candidate, n_points, dx)
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
    
    return (list(best_h), best_c5, n_points)
