import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Normalize to satisfy sum constraint
    h_start = h_start * (n_points/2) / np.sum(h_start)
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter_sa = 100000
    epsilon = 0.1
    
    T = T_init
    h_current = h_start.copy()
    c5_current = best_c5
    
    accepted = 0
    total = 0
    
    for i in range(max_iter_sa):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Perturb two random indices
        idx1, idx2 = np.random.randint(0, n_points, 2)
        delta = np.random.uniform(-epsilon, epsilon)
        
        # Adjust h[idx1] and h[idx2] to maintain sum constraint
        h_new = h_current.copy()
        h_new[idx1] += delta
        h_new[idx2] -= delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain equality constraint (sum = n_points/2)
        sum_diff = np.sum(h_new) - n_points/2
        h_new[0] -= sum_diff
        
        # Ensure bounds again
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        delta_c5 = c5_new - c5_current
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h_current = h_new
            c5_current = c5_new
            accepted += 1
            
            # Update best
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
        
        total += 1
        
        # Cooling
        T = max(T_min, T * alpha)
        
        if T <= T_min:
            break
    
    # Local refinement using L-BFGS-B
    def obj_and_grad(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        l_max = np.argmax(corr)
        
        # Compute gradient (simplified approximation)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            grad[i] = -2 * dx * (1 - h[j] - h[i])
        
        return max_overlap, grad
    
    try:
        # Normalize h to satisfy sum constraint
        h_norm = best_h * (n_points/2) / np.sum(best_h)
        
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            h_norm,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if res.success:
            h_refined = res.x
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
