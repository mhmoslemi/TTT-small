import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    best_c5 = evaluate_erdos_solution(h_fine, dx)

    n_coarse = 15

    if n_points > n_coarse:
        f_interp = interp1d(np.linspace(0, 2, n_points), h_fine, kind='linear', fill_value='extrapolate')
        x_coarse = np.linspace(0, 2, n_coarse)
        h_coarse = f_interp(x_coarse)
        h_coarse = project_to_feasible(h_coarse, n_coarse)
        
    # Coarse grid optimization
    for iteration in range(500):

        # Compute overlaps for coarse
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_coarse - 1)
        
        # Compute gradient for max lag
        grad = np.zeros(n_coarse)

        step_size = 0.01

    h_fine = project_to_feasible(h_fine, n_points)
