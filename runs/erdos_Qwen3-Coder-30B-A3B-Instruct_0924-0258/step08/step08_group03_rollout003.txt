import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex optimization parameters
    max_iterations = 500
    filter_bandwidth = 5
    k_top = 10
    optimizer_steps = 5
    filter_steps = 3
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Solve convex relaxation of top k overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        top_k_indices = np.argpartition(-corr, k_top)[:k_top]
        
        # Create convex optimization problem
        h_vars = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(h_vars, 1 - h_vars)))
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add constraints for top k overlaps
        for idx in top_k_indices:
            shift = idx - (n_points - 1)
            if shift >= 0:
                constraints.append(
                    cp.sum(cp.multiply(h_vars[:n_points-shift], 1 - h_vars[shift:])) <= 0.5
                )
            else:
                constraints.append(
                    cp.sum(cp.multiply(h_vars[-shift:], 1 - h_vars[:n_points+shift])) <= 0.5
                )
        
        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = np.array(h_vars.value).reshape(-1)
            else:
                h_new = h.copy()
        except:
            h_new = h.copy()
        
        # Apply moving average filter
        low_pass = np.ones(filter_bandwidth) / filter_bandwidth
        h_filtered = np.convolve(h_new, low_pass, mode='same')
        
        # Renormalize
        h_new = h_filtered * (n_points / 2) / np.sum(h_filtered)
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept better solution
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            
        # Adjust n for exploration
        if iteration % 20 == 0:
            k_top = min(k_top + 1, 20)
            filter_bandwidth = max(filter_bandwidth - 1, 2)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
