import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search parameters
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    # Local refinement parameters
    max_refine_iter = 500
    step_size = 0.1
    
    while time() - start_time < budget_s * 0.95 and high - low > tolerance:
        mid = (low + high) / 2
        
        # Create feasibility problem
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Sample shifts for constraints (100 random shifts)
        shifts = np.random.choice(n_points, size=100, replace=False)
        
        # Add overlap constraints
        for k in shifts:
            # Compute overlap sum: sum h[i] * (1 - h[(i+k) % n])
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_var[i] * (1 - h_var[j])
            constraints.append(expr <= mid * dx)
        
        # Solve feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_candidate = h_var.value
                if h_candidate is not None:
                    # Evaluate this candidate
                    c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                    if c5_candidate < best_c5:
                        best_c5 = c5_candidate
                        best_h = h_candidate.copy()
                    high = mid
                else:
                    low = mid
            else:
                low = mid
        except:
            low = mid
    
    # Local gradient refinement around best solution
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
        
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    for iter_refine in range(max_refine_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Optimize using scipy
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=cons, 
                         options={'disp': False, 'maxiter': 1})
        
        if result.success:
            h_new = result.x
            # Renormalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    return (list(best_h), best_c5, n_points)
