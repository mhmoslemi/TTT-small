import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Convex relaxation with McCormick envelopes
    def solve_convex_relaxation(h_init, n_points):
        h = cp.Variable(n_points)
        C = cp.Variable()
        
        # McCormick envelopes for z[i,j] = h[i] * h[j]
        z = cp.Variable((n_points, n_points))
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            C >= 0
        ]
        
        # Add McCormick envelopes
        for i in range(n_points):
            for j in range(n_points):
                if i >= j:
                    continue
                constraints.extend([
                    z[i, j] >= h[i] + h[j] - 1,
                    z[i, j] <= h[i],
                    z[i, j] <= h[j],
                    z[i, j] >= 0
                ])
        
        # Add overlap constraints
        # For lag k from -n_points+1 to n_points-1
        # Need to compute subset: sum_{i=0}^{n_points-1-|k|} h[i]*(1-h[i+k]) <= C
        
        # We'll check k=0 to n_points-1 and -n_points+1 to -1
        for k in range(1, n_points):
            # Lag k: i from 0 to n_points-1-k, sum h[i]*(1-h[i+k])
            # Each sum is h[i] - z[i, i+k]
            if k >= n_points:
                continue
            expr = 0
            for i in range(n_points - k):
                expr += h[i] - z[i, i+k]
            constraints.append(expr <= C)
            
        for k in range(-n_points+1, 0):
            # Lag k: i from 0 to n_points-1+|k|, sum h[i]*(1-h[i+k])
            # Each sum is h[i] - z[i, i+k]
            if k <= -n_points:
                continue
            expr = 0
            for i in range(n_points + k):
                expr += h[i] - z[i, i+k]
            constraints.append(expr <= C)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h.value, prob.value
        except:
            pass
        
        return h_init, evaluate_erdos_solution(h_init, dx)
    
    # Try convex relaxation first
    if time() - start_time < budget_s * 0.7:
        h_relaxed, c5_relaxed = solve_convex_relaxation(h_current, n_points)
        if c5_relaxed < best_c5:
            best_c5 = c5_relaxed
            best_h = h_relaxed.copy()
    
    # Local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Refine using SLSQP
    max_iter = 5000
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            # Start from current best
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined.copy()
        except Exception as e:
            pass
        
        # Random perturbation
        if np.random.random() < 0.1:
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            c5_perturbed = objective(h_perturbed)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
        
        # Early stopping if no improvement
        if iteration > 100 and iteration % 100 == 0:
            # Check last few iterations
            break
    
    return (list(best_h), best_c5, n_points)
