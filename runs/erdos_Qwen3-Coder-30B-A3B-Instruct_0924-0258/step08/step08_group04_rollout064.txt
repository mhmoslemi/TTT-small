import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        h_start[10:25] = 0.8  # Slightly more structured
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # PSO parameters
    num_particles = 75
    inertia = 0.8
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Create initial particles with some diversity
    for i in range(num_particles):
        particle = h_current.copy()
        # Add small random noise to diversify
        noise = np.random.normal(0, 0.05, n_points)
        particle += noise
        # Ensure feasibility for each particle
        particle = np.clip(particle, 0, 1)
        # Renormalize to match constraint
        particle = particle * (n_points/2) / np.sum(particle)
        particles.append(particle.copy())
        velocities.append(np.zeros(n_points))
    
    # Best solutions tracking
    particle_best = [p.copy() for p in particles]
    particle_best_fit = [evaluate_erdos_solution(p, dx) for p in particles]
    global_best = particles[np.argmin(particle_best_fit)].copy()
    global_best_fit = min(particle_best_fit)
    
    max_iter = 50000  # Prevent infinite loop
    iter_count = 0
    last_improvement = 0
    
    # PSO main loop
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        for i in range(num_particles):
            # Evaluate current particle
            current_fit = evaluate_erdos_solution(particles[i], dx)
            
            # Update particle best if improved
            if current_fit < particle_best_fit[i]:
                particle_best[i] = particles[i].copy()
                particle_best_fit[i] = current_fit
            
            # Update global best if needed
            if current_fit < global_best_fit:
                global_best = particles[i].copy()
                global_best_fit = current_fit
                last_improvement = iter_count
                
        # Update velocities and positions
        for i in range(num_particles):
            # Update inertia from 0.8 to 0.4 over time
            current_inertia = inertia * (1.0 - (iter_count / max_iter) * 0.4)
            
            # Random weightings
            r1, r2 = np.random.rand(2)
            
            # Compute new velocity
            vel = (current_inertia * velocities[i] + 
                   cognitive_weight * r1 * (particle_best[i] - particles[i]) + 
                   social_weight * r2 * (global_best - particles[i]))
            
            # Clip velocity
            vel = np.clip(vel, -max_velocity, max_velocity)
            velocities[i] = vel.copy()
            
            # Update position
            new_pos = particles[i] + vel
            
            # Project to feasible region
            new_pos = np.clip(new_pos, 0, 1)
            # Renormalize to match constraint
            new_pos = new_pos * (n_points/2) / np.sum(new_pos)
            particles[i] = new_pos.copy()
        
        iter_count += 1
        
        # Check for early termination (no improvement in long time)
        if iter_count - last_improvement > 5000:
            # Apply perturbation to some particles
            for i in range(num_particles // 10):
                idx = np.random.randint(num_particles)
                particles[idx] = global_best.copy()
                # Add random small noise
                noise = np.random.normal(0, 0.01, n_points)
                particles[idx] += noise
                particles[idx] = np.clip(particles[idx], 0, 1)
                particles[idx] = particles[idx] * (n_points/2) / np.sum(particles[idx])
    
    # Final refinement: local search on best solution
    best_h = global_best.copy()
    best_c5 = global_best_fit
    
    # Perform a few gradient steps to refine
    step_size = 0.01
    for _ in range(100):
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(best_h) - 1)
        
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - best_h[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= best_h[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - best_h[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= best_h[i]
        
        new_h = best_h - step_size * gradient
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
        else:
            step_size *= 0.5
            if step_size < 1e-10:
                break
    
    return (list(best_h), best_c5, n_points)
