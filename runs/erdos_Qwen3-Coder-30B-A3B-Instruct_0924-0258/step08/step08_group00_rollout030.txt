import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Local search with multiple strategies
    max_iter = 2000
    patience = 30
    no_improvement = 0
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            break
            
        # Try different update strategies
        strategy = iteration % 5
        h_new = h.copy()
        
        if strategy == 0:  # Gradient descent on max overlap
            corr = np.correlate(h, 1 - h, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (n_points - 1)
            
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + max_lag
                if 0 <= j < n_points:
                    grad[i] = 2 * (1 - h[j] - h[i]) / n_points
            
            step_size = 0.01
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
        elif strategy == 1:  # Random perturbation
            perturbation = np.random.normal(0, 0.02, n_points)
            h_new = h + perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
        elif strategy == 2:  # Neighborhood search with binary search
            # Create new solution by varying one element
            idx = np.random.randint(0, n_points)
            old_val = h[idx]
            new_val = np.random.uniform(0, 1)
            h_new = h.copy()
            h_new[idx] = new_val
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
        elif strategy == 3:  # Simple pattern matching approach
            # Modify in piecewise fashion
            segment_size = max(2, n_points // 5)
            start_idx = np.random.randint(0, n_points - segment_size)
            factor = np.random.uniform(0.8, 1.2)
            for i in range(start_idx, min(start_idx + segment_size, n_points)):
                h_new[i] = np.clip(h[i] * factor, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
        elif strategy == 4:  # Centered shrinkage
            center = n_points // 2
            h_new = h.copy()
            for i in range(n_points):
                factor = 1.0 - 0.1 * abs(i - center) / (n_points/2)
                h_new[i] = np.clip(h[i] * factor, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
        # Accept if improves
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < current_c5:
            h = h_new
        else:
            # Occasionally accept worse moves to escape local minima
            if np.random.random() < 0.1:
                h = h_new
    
    # Final optimization using scipy
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 300, 'ftol': 1e-10})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Ensure final constraint satisfaction
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
