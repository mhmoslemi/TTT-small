import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Binary initialization and threshold search
    # Initialize binary solution with equal 0s and 1s
    h_binary = np.zeros(n_points)
    h_binary[:n_points//2] = 1
    np.random.shuffle(h_binary)
    
    # Threshold-based update
    threshold = 1.0
    max_iter = 1000
    iter_count = 0
    
    while threshold > 0.01 and iter_count < max_iter and time() - start_time < budget_s * 0.8:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h_binary, 1 - h_binary, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute approximate gradient
        h_grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            h_grad[i] = -2 * dx * (1 - h_binary[j] - h_binary[i])
        
        # Apply threshold
        update_mask = np.abs(h_grad) > threshold
        h_new = h_binary.copy()
        
        # Update based on gradient
        for i in range(n_points):
            if update_mask[i]:
                if h_grad[i] > 0:
                    h_new[i] = 1
                else:
                    h_new[i] = 0
        
        # Maintain sum constraint
        current_sum = np.sum(h_new)
        if current_sum > n_points/2:
            # Convert some 1s to 0s
            inds = np.argsort(h_new)[::-1]
            for i in inds:
                if current_sum <= n_points/2:
                    break
                if h_new[i] == 1:
                    h_new[i] = 0
                    current_sum -= 1
        elif current_sum < n_points/2:
            # Convert some 0s to 1s
            inds = np.argsort(h_new)
            for i in inds:
                if current_sum >= n_points/2:
                    break
                if h_new[i] == 0:
                    h_new[i] = 1
                    current_sum += 1
        
        # Update
        h_binary = h_new.copy()
        
        # Evaluate
        current_c5 = evaluate_erdos_solution(h_binary, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_binary.copy()
            
        threshold *= 0.95
        iter_count += 1
        
        # Early termination
        if threshold < 0.01 and iter_count > 100:
            break
            
    # Phase 2: Local L-BFGS refinement with binary projection
    # Only if we have a good binary solution
    if np.all(np.isin(best_h, [0, 1])):
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(
                lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 100, 'ftol': 1e-9}
            )
            
            if res.success:
                refined_h = res.x
                refined_h = refined_h * (n_points/2) / np.sum(refined_h)
                refined_c5 = evaluate_erdos_solution(refined_h, dx)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
        except Exception:
            pass
    
    # Final projection and validation
    best_h = np.clip(best_h, 0, 1)
    
    # Final check
    c5_check = evaluate_erdos_solution(best_h, dx)
    if c5_check < best_c5:
        best_c5 = c5_check
        
    return (list(best_h), best_c5, n_points)
