import numpy as np
from time import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search over C5 using SDP relaxation
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    def solve_sdp(c5_val):
        # SDP relaxation approach to find if C5 is feasible
        h_vars = cp.Variable(n_points)
        Y = cp.Variable((n_points, n_points), symmetric=True)
        
        # Constraints
        constraints = [
            Y >> 0,  # Y is positive semidefinite
            cp.diag(Y) == h_vars,  # Y[i,i] = h[i]
            cp.sum(h_vars) == n_points / 2,  # constraint on sum
        ]
        
        # Overlap constraints for several sampled shifts
        shifts = np.random.randint(1, n_points, size=100)
        for k in shifts:
            if k < n_points:
                # Sum of h[i] * (1 - h[i + k]) <= c5 * dx
                overlap = cp.sum(cp.multiply(h_vars[:n_points-k], 
                                           1 - h_vars[k:]))
                constraints.append(overlap <= c5_val * dx)
        
        # Minimize zero (just check feasibility)
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            return prob.status == cp.OPTIMAL or prob.status == cp.OPT_INFEASIBLE
        except:
            return False
    
    def project_sdp_solution(Y_sol):
        # Project Y to get a valid h
        u, s, vt = np.linalg.svd(Y_sol)
        # Get rank-1 approximation
        h_proj = np.sqrt(s[0]) * u[:, 0]
        # Clip and renormalize
        h_proj = np.clip(h_proj, 0, 1)
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        return h_proj
    
    # Local refinement helper
    def refine_h(h_input):
        h_refined = h_input.copy()
        dx_ref = 2.0 / len(h_refined)
        
        def objective(h_vals):
            corr = np.correlate(h_vals, 1 - h_vals, mode='full')
            return np.max(corr) * dx_ref
        
        def constraint_sum(h_vals):
            return np.sum(h_vals) - n_points / 2
            
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(objective, h_refined, method='SLSQP', bounds=bounds, 
                         constraints=constraints, options={'maxiter': 100})
        
        if result.success:
            h_refined = result.x
        else:
            # fallback to basic local update
            for _ in range(100):
                corr = np.correlate(h_refined, 1 - h_refined, mode='full')
                max_lag_idx = np.argmax(corr)
                max_lag = max_lag_idx - (n_points - 1)
                grad = np.zeros(n_points)
                for i in range(n_points):
                    j = i + max_lag
                    if 0 <= j < n_points:
                        grad[i] = 1 - 2 * h_refined[i] - h_refined[j]
                    else:
                        grad[i] = 1 - 2 * h_refined[i]
                h_refined = h_refined - 0.001 * grad
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                
        return h_refined
    
    # Binary search with SDP feasibility checks
    while time() - start_time < budget_s * 0.95 and high - low > tolerance:
        mid = (low + high) / 2.0
        if solve_sdp(mid):
            high = mid
            # Try to improve the solution with local search
            try:
                # Simple local search refinement based on gradient
                h_new = refine_h(best_h)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            except:
                pass
        else:
            low = mid
            
        if time() - start_time > budget_s * 0.9:
            break
    
    # Final local refinement
    try:
        h_final = refine_h(best_h)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    except:
        pass
    
    # Ensure final constraint
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
