import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    max_generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    stagnation_limit = 100
    refinement_interval = 100
    
    # Initialize population
    population = []
    fitness_history = []
    
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        fitness_history.append(evaluate_erdos_solution(h_start, dx))
    else:
        population.append(h_start.copy())
        fitness_history.append(evaluate_erdos_solution(h_start, dx))
    
    # Generate random individuals
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
        fitness_history.append(evaluate_erdos_solution(h, dx))
    
    # Main GA loop
    generation = 0
    stagnation_count = 0
    last_best = best_c5
    
    while time() - start_time < budget_s * 0.7 and generation < max_generations:
        # Evaluate all individuals
        fitness_values = []
        for h in population:
            fitness = evaluate_erdos_solution(h, dx)
            fitness_values.append(fitness)
        
        # Tournament selection
        def tournament_selection(k=3):
            indices = np.random.choice(len(population), k, replace=False)
            best_idx = indices[np.argmin([fitness_values[i] for i in indices])]
            return population[best_idx].copy()
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individual
        best_idx = np.argmin(fitness_values)
        new_population.append(population[best_idx].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Selection and crossover
            parent1 = tournament_selection()
            parent2 = tournament_selection()
            
            if np.random.rand() < crossover_rate:
                # Blend crossover (BLX-0)
                alpha = 0.5
                min_vals = np.minimum(parent1, parent2)
                max_vals = np.maximum(parent1, parent2)
                range_vals = max_vals - min_vals
                lower_bound = min_vals - alpha * range_vals
                upper_bound = max_vals + alpha * range_vals
                
                offspring = np.random.rand(n_points) * (upper_bound - lower_bound) + lower_bound
            else:
                # Copy one parent
                offspring = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.1:  # 10% chance to mutate
                        # Add small random change
                        change = np.random.normal(0, 0.05)
                        offspring[i] += change
                        # Re-normalize
                        offspring = offspring * (n_points / 2) / np.sum(offspring)
            
            # Ensure constraints (0 <= h <= 1 and sum = n_points/2)
            offspring = np.clip(offspring, 0, 1)
            if np.sum(offspring) > 0:
                offspring = offspring * (n_points / 2) / np.sum(offspring)
            
            new_population.append(offspring)
        
        population = new_population
        
        # Track best solution
        current_best = min(fitness_values)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(fitness_values)].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        if stagnation_count >= stagnation_limit:
            break
            
        generation += 1
        
        # Periodic refinement with SLSQP
        if generation % refinement_interval == 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP',

                               options={'maxiter': 100, 'ftol': 1e-10})

                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final SLSQP refinement on best solution
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
