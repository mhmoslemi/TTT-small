import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search parameters
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    # Use a subset of shifts to make the feasibility problem tractable
    # Sample 100 shifts from the full range [0, n_points-1]
    shift_set = np.random.choice(n_points, size=100, replace=False)
    
    def solve_feasibility(h_var, target_c5, shifts):
        """Solve the convex feasibility problem: 
           max overlap <= target_c5"""
        n = len(h_var)
        # Constraint: \sum_i h[i] * (1 - h[(i+k)%n]) <= target_c5 * dx
        constraints = []
        constraints.append(cp.sum(h_var) == n_points / 2)
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # Check overlaps for chosen shifts
        for k in shifts:
            overlap = sum(h_var[i] * (1 - h_var[(i+k) % n]) for i in range(n))
            constraints.append(overlap <= target_c5 * dx)
        
        # The problem is feasible if we can find h satisfying all constraints
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return prob.status == 'optimal'
        except:
            return False
    
    # Binary search loop
    while time() - start_time < budget_s * 0.95 and high - low > tolerance:
        mid = (low + high) / 2
        # Set up variable for the feasibility problem
        h_var = cp.Variable(n_points)
        
        if solve_feasibility(h_var, mid, shift_set):
            high = mid
            # Store the best feasible solution
            h_sol = h_var.value
            if h_sol is not None:
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                c5_test = evaluate_erdos_solution(h_sol, dx)
                if c5_test < best_c5:
                    best_h = h_sol.copy()
                    best_c5 = c5_test
        else:
            low = mid
            
    # Local refinement using gradient descent if no much improvement
    # Try several restarts with slightly perturbed solutions
    for restart in range(5):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor solution by slight random perturbation
        h_pert = best_h + np.random.normal(0, 0.01, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        
        # Simple gradient descent on max overlap
        step_size = 0.01
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            # The gradient is about: -2 * h + 1 - (sum of shifted terms)
            # Manually compute for max overlap at a given lag
            corr = np.correlate(h_pert, 1 - h_pert, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (n_points - 1)
            
            grad = np.zeros(n_points)
            for i in range(n_points):
                # Derivative of h[i][1 - h[(i + max_lag) % n]] 
                j = (i + max_lag) % n_points
                grad[i] = -2 * h_pert[i] + 1 - (1 - h_pert[j])
            
            h_new = h_pert - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_pert = h_new.copy()
            else:
                step_size *= 0.95
                
    return (list(best_h), best_c5, n_points)
