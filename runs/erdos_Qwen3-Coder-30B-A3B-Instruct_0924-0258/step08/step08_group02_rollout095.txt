import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a binary-like starting point
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1.0
        np.random.shuffle(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Thresholding optimization loop
    threshold = 0.1
    cooling_factor = 0.95
    max_iterations = 500
    threshold_min = 0.01
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient contributions
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - best_h[j] - best_h[i])
        
        # Sort by gradient magnitude and threshold top half
        indices = np.argsort(np.abs(gradient))[::-1]
        top_indices = indices[:n_points//2]
        
        # Apply thresholding
        new_h = best_h.copy()
        for i in range(n_points):
            if i in top_indices:
                if best_h[i] >= 1 - threshold:
                    new_h[i] = 1.0
                elif best_h[i] <= threshold:
                    new_h[i] = 0.0
                else:
                    # Use a soft threshold
                    new_h[i] = np.clip(best_h[i], 0.0, 1.0)
            else:
                # Keep as is for non-top indices
                new_h[i] = best_h[i]
        
        # Maintain sum constraint
        sum_h = np.sum(new_h)
        if abs(sum_h - n_points/2) > 1e-6:
            # Simple redistribution fix
            diff = (n_points/2 - sum_h) / n_points
            new_h = np.clip(new_h + diff, 0, 1)
            # Normalize more carefully
            new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate updated solution
        current_c5 = evaluate_erdos_solution(new_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_h.copy()
        
        # Cool threshold
        threshold = max(threshold_min, threshold * cooling_factor)
    
    # Additional coordinate-wise refinement
    for iter_refine in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        refine_indices = np.random.permutation(n_points)
        improved = False
        for i in refine_indices[:n_points//2]:
            if time() - start_time > budget_s * 0.95:
                break
                
            # Optimize single coordinate with golden section search
            old_h = best_h[i]
            
            def obj_val(x):
                best_h[i] = x
                # Normalize
                sum_h = np.sum(best_h)
                if abs(sum_h - n_points/2) > 1e-6:
                    best_h = best_h * (n_points/2) / sum_h
                return evaluate_erdos_solution(best_h, dx)
            
            # Golden section search in [0, 1]
            ph = (np.sqrt(5) - 1) / 2
            a, b = 0.0, 1.0
            x1 = a + (1 - ph) * (b - a)
            x2 = a + ph * (b - a)
            f1, f2 = obj_val(x1), obj_val(x2)
            
            while abs(b - a) > 1e-6:
                if f1 < f2:
                    b, x2, f2 = x2, x1, f1
                    x1 = a + (1 - ph) * (b - a)
                    f1 = obj_val(x1)
                else:
                    a, x1, f1 = x1, x2, f2
                    x2 = a + ph * (b - a)
                    f2 = obj_val(x2)
            
            best_h[i] = (a + b) / 2.0
            improved = True
        
        # Normalize
        sum_h = np.sum(best_h)
        if abs(sum_h - n_points/2) > 1e-6:
            best_h = best_h * (n_points/2) / sum_h
            
        if not improved:
            break
    
    # Final polishing with L-BFGS
    try:
        def objective(h):
            return np.max(np.correlate(h, 1 - h, mode='full') * dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    # Final feasibility check
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points/2) > 1e-6:
        best_h = best_h * (n_points/2) / sum_h
    
    return (list(best_h), best_c5, n_points)
