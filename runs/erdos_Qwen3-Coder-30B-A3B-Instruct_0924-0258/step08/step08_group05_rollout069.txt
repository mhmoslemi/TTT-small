import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    
    # Normalize initial solution
    h_start = h_start * (n_points/2) / np.sum(h_start)
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    generations = 10000
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    no_improvement_limit = 100
    
    # Initialize population
    population = []
    population.append(h_start.copy())
    for i in range(population_size - 1):
        if n_points % 2 == 0:
            h = np.zeros(n_points)
            h[:n_points//2] = 1
        else:
            h = np.zeros(n_points)
            h[:(n_points-1)//2] = 1
            h[(n_points-1)//2] = 0.5
        # Normalize
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evolution loop
    no_improvement_count = 0
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= no_improvement_limit:
            break
            
        # Selection (tournament)
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner_idx = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner_idx].copy())
            
        # Crossover and Mutation
        new_population = [best_h.copy()]  # Elitism
        
        while len(new_population) < population_size:
            if np.random.random() < crossover_rate and len(selected) >= 2:
                parent1, parent2 = np.random.choice(len(selected), 2, replace=False)
                parent1, parent2 = selected[parent1], selected[parent2]
                
                # Uniform crossover
                child = np.zeros(n_points)
                mask = np.random.random(n_points) < 0.5
                child[mask] = parent1[mask]
                child[~mask] = parent2[~mask]
                
                # Normalize child
                child = child * (n_points/2) / np.sum(child)
                
                # Mutation
                if np.random.random() < mutation_rate:
                    idx1, idx2 = np.random.choice(n_points, 2, replace=False)
                    child[idx1] = np.clip(child[idx1] + np.random.normal(0, 0.1), 0, 1)
                    child[idx2] = np.clip(child[idx2] + np.random.normal(0, 0.1), 0, 1)
                    child = child * (n_points/2) / np.sum(child)
                    
                new_population.append(child)
            else:
                # Just mutate a random individual
                idx = np.random.randint(0, len(selected))
                child = selected[idx].copy()
                if np.random.random() < mutation_rate:
                    idx1, idx2 = np.random.choice(n_points, 2, replace=False)
                    child[idx1] = np.clip(child[idx1] + np.random.normal(0, 0.1), 0, 1)
                    child[idx2] = np.clip(child[idx2] + np.random.normal(0, 0.1), 0, 1)
                    child = child * (n_points/2) / np.sum(child)
                new_population.append(child)
                
        population = new_population[:population_size]
        
    # Final refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
