import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Try SDP approach
    try:
        # SDP approach using CVXPY
        h_vars = cp.Variable(n_points, lower_bound=0, upper_bound=1)
        # Constraint: sum of h = n_points/2
        constraints = [cp.sum(h_vars) == n_points/2]
        
        # For a faster solver, we'll use a coarse approximation
        # by linearizing the problem for only a few critical lags
        max_lags_to_consider = min(10, n_points-1)
        for k in range(-max_lags_to_consider, max_lags_to_consider+1):
            if k == 0:
                continue
            # Create the correlation constraint: sum(h[i] * (1 - h[i+k]))
            # which is sum(h[i] - h[i]*h[i+k])
            # This relaxes to sum(h[i]) - sum(h[i]*h[i+k])
            # We'll create this using auxiliary variables for bilinear terms
            
            # For simplicity, we avoid full SDP here but focus on heuristic
            # setting of bounds only for most critical lags.
            
        # Instead of full SDP, optimize by gradient descent + constraints
        # But try a hybrid approach with restarts
        
        # Parameters for descent
        step_size = 0.01
        max_iter = 5000
        iter_count = 0
        restart_count = 0
        max_restarts = 10
        
        while time() - start_time < budget_s * 0.9 and iter_count < max_iter and restart_count < max_restarts:
            # Compute all correlations for all possible lags
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            # Get the maximum overlap lag
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (len(h_current) - 1)

            # Gradient estimation using finite differences (simpler)
            gradient = np.zeros(n_points)
            eps = 1e-6
            
            if max_lag >= 0:
                for i in range(n_points - max_lag):
                    gradient[i] += 1 - h_current[i + max_lag]
                    if i + max_lag < n_points:
                        gradient[i + max_lag] -= h_current[i]
            else:
                for i in range(n_points + max_lag):
                    gradient[i] += 1 - h_current[i - max_lag]
                    if i - max_lag >= 0:
                        gradient[i - max_lag] -= h_current[i]
            
            # Apply small gradient step but with adaptive step size
            h_new = h_current - step_size * gradient
            
            # Ensure feasibility
            h_new = np.clip(h_new, 0, 1)
            
            # Renormalize to maintain constraint (sum = n_points/2)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate new C5 value
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept step if improvement
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                step_size = min(0.1, step_size * 1.05)  # increase step if improving
            else:
                step_size = max(1e-6, step_size * 0.5)  # reduce step if not improving
                if step_size < 1e-6:
                    # Restart with better vector
                    restart_count += 1
                    h_new = h_current.copy()
                    # Add some random perturbation to escape local minimum
                    h_new += np.random.normal(0, 0.01, n_points)
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points/2) / np.sum(h_new)
            
            h_current = h_new.copy()
            iter_count += 1
            
    except Exception:
        # Fallback to previous algorithm if SDP fails
        pass

    return (list(best_h), best_c5, n_points)
