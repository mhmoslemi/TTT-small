import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 0
    max_no_improvement = 100
    no_improvement_count = 0
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        population[0] = population[0] * (n_points / 2) / np.sum(population[0])
    
    while len(population) < population_size:
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        population.append(h_rand)
    
    def selection_tournament(pop, k=3):
        selected = []
        for _ in range(len(pop)):
            indices = np.random.choice(len(pop), k)
            fitnesses = [evaluate_erdos_solution(pop[i], dx) for i in indices]
            winner_idx = indices[np.argmin(fitnesses)]
            selected.append(pop[winner_idx].copy())
        return selected
    
    def blend_crossover(parent1, parent2, alpha=0.5):
        child1 = alpha * parent1 + (1 - alpha) * parent2
        child2 = (1 - alpha) * parent1 + alpha * parent2
        return child1, child2
    
    def mutate(h, rate, step_size):
        h_mut = h.copy()
        for i in range(len(h)):
            if np.random.rand() < rate:
                delta = np.random.normal(0, step_size)
                h_mut[i] += delta
        # Renormalize
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
        # Clip to bounds
        h_mut = np.clip(h_mut, 0, 1)
        return h_mut
    
    def evaluate_population(pop):
        fitnesses = [evaluate_erdos_solution(h, dx) for h in pop]
        return fitnesses
    
    # Main GA loop
    while time() - start_time < budget_s * 0.7:
        fitnesses = evaluate_population(population)
        
        # Track best solution
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            break
            
        # Selection
        selected = selection_tournament(population)
        
        # Create new population
        new_population = [population[min_idx].copy()]
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < crossover_rate and len(new_population) + 1 < population_size:
                i, j = np.random.choice(len(selected), 2, replace=False)
                child1, child2 = blend_crossover(selected[i], selected[j])
                new_population.append(child1)
                new_population.append(child2)
            else:
                # Mutation
                parent = selected[np.random.randint(0, len(selected))]
                child = mutate(parent, mutation_rate, 0.05)
                new_population.append(child)
        
        population = new_population[:population_size]
        generations += 1
    
    # Final refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
