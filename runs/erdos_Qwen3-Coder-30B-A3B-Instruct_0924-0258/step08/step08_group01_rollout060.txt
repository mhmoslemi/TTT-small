import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    generations = 0
    max_no_improvement = 200
    no_improvement_count = 0
    mutation_rate = 0.1
    crossover_rate = 0.8
    mutation_step = 0.05
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h = np.random.random(n_points) * 0.2 + 0.4
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    def ensure_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def crossover(h1, h2):
        if random.random() > crossover_rate:
            return h1.copy(), h2.copy()
        alpha = random.uniform(0.3, 0.7)
        child1 = alpha * h1 + (1 - alpha) * h2
        child2 = (1 - alpha) * h1 + alpha * h2
        return ensure_feasible(child1), ensure_feasible(child2)
    
    def mutate(h):
        if random.random() > mutation_rate:
            return h.copy()
        h_mut = h.copy()
        idx = random.randint(0, n_points - 1)
        h_mut[idx] += random.gauss(0, mutation_step)
        return ensure_feasible(h_mut)
    
    while time() - start_time < budget_s * 0.9:
        # Evaluate population fitness
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            break
        
        # Tournament selection
        selected = []
        tournament_size = 3
        for _ in range(population_size // 2):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx])
        
        # Generate offspring
        new_population = []
        for i in range(0, len(selected), 2):
            h1 = selected[i]
            h2 = selected[(i + 1) % len(selected)]
            child1, child2 = crossover(h1, h2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        # Keep best solutions and new offspring
        sorted_indices = np.argsort(fitness)[:population_size // 2]
        population = [population[i] for i in sorted_indices]
        population.extend(new_population[:population_size // 2])
        
        generations += 1
        
        # Periodic local refinement with SLSQP
        if generations % 100 == 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-12})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
