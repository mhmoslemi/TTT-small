import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    """Project h to satisfy constraints: 0<=h<=1 and sum(h)=n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex perturbation optimization
    max_iter = 5000
    epsilon_0 = 0.001
    beta = 0.99
    top_k = 100
    
    for t in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        # Get top k largest correlations (shifts that maximize overlap)
        top_shifts = np.argsort(corr)[-top_k:]
        
        # Construct a convexified objective
        # Target: minimize sum over top_k of (h * (1 - h_shifted)) + epsilon * ||h||_2^2
        # We'll approach this with CVXPY
        
        # Build problem variables
        h_var = cp.Variable(n_points)
        
        # Define constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Define terms for convex objective
        total_obj = 0
        epsilon_t = epsilon_0 * (beta ** t)
        
        # Add regularization term
        total_obj += epsilon_t * cp.norm2(h_var) ** 2
        
        # Add contributions from top_k shifts
        for k in top_shifts:
            # Compute shifted version
            shift = k - (n_points - 1)
            if shift >= 0:
                h_shift = np.concatenate([np.zeros(shift), h[:-shift]])
            else:
                shift = -shift
                h_shift = np.concatenate([h[shift:], np.zeros(shift)])
            
            # Compute the inner product (convolution approx with linearized terms)
            # Since we've linearized this, we'll compute average approximation
            total_obj += cp.sum(cp.multiply(h_var, 1 - h_shift)) / top_k
        
        # Formulate and solve
        obj = cp.Minimize(total_obj)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=100)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
            else:
                h_new = h.copy()
        except:
            h_new = h.copy()
        
        # Project to feasible region
        h_new = project_to_feasible(h_new, n_points, dx)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Adjust perturbation if no improvement
            if t > 0 and t % 100 == 0:
                epsilon_0 *= 0.95
    
    # Final local refinement with random search
    if time() - start_time < budget_s * 0.9:
        best_refined = best_h.copy()
        best_c5_refined = best_c5
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            # Small random perturbation
            pert = np.random.normal(0, 0.001, n_points)
            h_trial = best_h + pert
            h_trial = project_to_feasible(h_trial, n_points, dx)
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            if c5_trial < best_c5_refined:
                best_c5_refined = c5_trial
                best_refined = h_trial.copy()
        
        if best_c5_refined < best_c5:
            best_h = best_refined
            best_c5 = best_c5_refined
    
    return (list(best_h), best_c5, n_points)
