import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search parameters
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    # Sample shifts for feasibility checks
    shift_samples = 100
    shifts = np.random.choice(n_points, shift_samples, replace=False)
    
    while time() - start_time < budget_s * 0.95 and high - low > tolerance:
        mid = (low + high) / 2
        
        # Formulate convex feasibility problem
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Add overlap constraints for sampled shifts
        for k in shifts:
            # Compute convolution term
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_var[i] * (1 - h_var[j])
            constraints.append(expr <= mid * dx)
        
        # Solve feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_candidate = h_var.value
                if h_candidate is not None:
                    # Check if this solution is feasible
                    candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
                    if candidate_c5 < best_c5:
                        best_c5 = candidate_c5
                        best_h = h_candidate.copy()
                    if candidate_c5 <= mid:
                        high = mid
                    else:
                        low = mid
                else:
                    low = mid
            else:
                low = mid
        except:
            low = mid
    
    # Local refinement using gradient-based method
    h = best_h.copy()
    step_size = 0.1
    max_iterations = 500
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.99:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient for the max lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h[i] - h[j]
            else:
                grad[i] = 1 - 2 * h[i]
        
        # Gradient descent step
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to maintain constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Reduce step size
            step_size *= 0.5
            if step_size < 1e-9:
                break
    
    return (list(best_h), best_c5, n_points)
