import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 50
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 2
    generations_no_improvement = 0
    max_generations_no_improvement = 100
    generation = 0
    
    # Initialize population
    population = []
    population_fitness = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        population_fitness.append(evaluate_erdos_solution(h_start, dx))
    
    # Fill population with random solutions
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
        population_fitness.append(evaluate_erdos_solution(h, dx))
    
    # Genetic algorithm main loop
    while time() - start_time < budget_s * 0.7 and generations_no_improvement < max_generations_no_improvement:
        # Sort population by fitness
        sorted_indices = np.argsort(population_fitness)
        population = [population[i] for i in sorted_indices]
        population_fitness = [population_fitness[i] for i in sorted_indices]
        
        if population_fitness[0] < best_c5:
            best_c5 = population_fitness[0]
            best_h = population[0].copy()
            generations_no_improvement = 0
        else:
            generations_no_improvement += 1
            
        # Create new population
        new_population = []
        new_population_fitness = []
        
        # Elitism: keep the best individuals
        for i in range(elite_size):
            new_population.append(population[i].copy())
            new_population_fitness.append(population_fitness[i])
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, population_fitness, 3)
            parent2 = tournament_selection(population, population_fitness, 3)
            
            # Crossover
            if random.random() < crossover_rate:
                child1, child2 = blend_crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            mutation_strength = 0.05 * (0.95 ** generation)
            mutate(child1, mutation_rate, mutation_strength, dx, n_points)
            mutate(child2, mutation_rate, mutation_strength, dx, n_points)
            
            # Evaluate offspring
            new_population.append(child1)
            new_population_fitness.append(evaluate_erdos_solution(child1, dx))
            
            if len(new_population) < population_size:
                new_population.append(child2)
                new_population_fitness.append(evaluate_erdos_solution(child2, dx))
        
        population = new_population
        population_fitness = new_population_fitness
        generation += 1
    
    # Final local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)

def tournament_selection(population, fitness, k):
    """Select one individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), k, replace=False)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_index].copy()

def blend_crossover(parent1, parent2):
    """Blend crossover operation"""
    alpha = 0.5
    child1 = np.zeros_like(parent1)
    child2 = np.zeros_like(parent2)
    for i in range(len(parent1)):
        if parent1[i] > parent2[i]:
            child1[i] = parent1[i] + alpha * (parent2[i] - parent1[i])
            child2[i] = parent2[i] + alpha * (parent1[i] - parent2[i])
        else:
            child1[i] = parent1[i] + alpha * (parent2[i] - parent1[i])
            child2[i] = parent2[i] + alpha * (parent1[i] - parent2[i])
    return child1, child2

def mutate(h, mutation_rate, mutation_strength, dx, n_points):
    """Mutate an individual preserving constraints"""
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h[i] += np.random.normal(0, mutation_strength)
    
    # Ensure h stays in [0, 1]
    h = np.clip(h, 0, 1)
    
    # Adjust to maintain sum = n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        scale_factor = (n_points / 2) / current_sum
        h *= scale_factor
