import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s > 0:
        h = h * (n_points / 2) / s
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = project_feasible(h_start, n_points)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    T0 = 0.05
    alpha = 0.999
    T_min = 1e-6
    sigma0 = 0.05
    beta = 0.999
    
    T = T0
    sigma = sigma0
    
    max_iter = int(1e6)
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and T > T_min and iter_count < max_iter:
        # Generate neighbor by perturbing h
        delta_h = np.random.normal(0, sigma, n_points)
        h_new = h_current + delta_h
        
        # Project to feasible set
        h_new = project_feasible(h_new, n_points)
        
        # Evaluate new C5
        c5_new = evaluate_erdos_solution(h_new, dx)
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        # Accept or reject
        delta = c5_new - c5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_current = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_current.copy()
        
        # Cooling
        T *= alpha
        sigma *= beta
        
        iter_count += 1
    
    # Final validation
    best_h = project_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
