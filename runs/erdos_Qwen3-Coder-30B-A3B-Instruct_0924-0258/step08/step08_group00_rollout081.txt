import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different initialization strategies
    initial_strategies = []
    
    # Strategy 1: Initial construction
    initial_strategies.append(h_start.copy())
    
    # Strategy 2: Simple alternating pattern
    h_alt = np.zeros(n_points)
    for i in range(n_points):
        h_alt[i] = 1 if i % 2 == 0 else 0
    h_alt = h_alt * (n_points/2) / np.sum(h_alt)
    initial_strategies.append(h_alt)
    
    # Strategy 3: Piecewise constant regions
    h_piece = np.ones(n_points) * 0.5
    mid = n_points // 2
    h_piece[:mid] = 0.3
    h_piece[mid:] = 0.7
    h_piece = h_piece * (n_points/2) / np.sum(h_piece)
    initial_strategies.append(h_piece)
    
    # Strategy 4: More localized approach
    h_local = np.ones(n_points) * 0.1
    h_local[n_points//4:3*n_points//4] = 0.9
    h_local = h_local * (n_points/2) / np.sum(h_local)
    initial_strategies.append(h_local)
    
    # Try all strategies
    for h_init in initial_strategies:
        h = h_init.copy()
        h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
        
        # Local optimization with limited iterations
        max_iter = 500
        patience = 20
        no_improvement = 0
        step_size = 0.02
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.9:
                break
                
            corr = np.correlate(h, 1 - h, mode='full')
            max_lag_idx = np.argmax(corr)
            max_overlap = corr[max_lag_idx] * dx
            
            if max_overlap < best_c5:
                best_c5 = max_overlap
                best_h = h.copy()
                no_improvement = 0
            else:
                no_improvement += 1
                
            if no_improvement >= patience:
                break
                
            # Compute gradient for maximum overlap lag
            max_lag = max_lag_idx - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + max_lag
                if 0 <= j < n_points:
                    grad[i] = 2 * (1 - h[j] - h[i]) / n_points
            
            # Update
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            if evaluate_erdos_solution(h_new, dx) < evaluate_erdos_solution(h, dx):
                h = h_new
            else:
                step_size *= 0.5
                if step_size < 1e-8:
                    break
    
    # Final SLSQP optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 300, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
