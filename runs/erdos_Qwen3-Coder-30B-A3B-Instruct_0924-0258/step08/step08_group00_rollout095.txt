import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def repair(h, n_points):
    h = np.clip(h, 0, 1)
    if np.sum(h) == 0:
        return h
    h = h * (n_points/2) / np.sum(h)
    return h

    random.seed(seed)

    best_h = h_start.copy()

    # Multi-start approach with Latin hypercube initialization
    n_starts = 50
    best_h_list = []
    best_c5_list = []
    
    # Generate Latin hypercube samples
    lhs_samples = np.random.uniform(0, 1, (n_starts, n_points))
    for i in range(n_starts):
        # Normalize each sample to satisfy sum constraint
        lhs_samples[i] = lhs_samples[i] * (n_points/2) / np.sum(lhs_samples[i])
    
    # Add initial_h_values if available
    if 'initial_h_values' in globals():
        lhs_samples[0] = np.array(initial_h_values)
    
    # Process all starting points
    for i in range(n_starts):
        if time() - start_time > budget_s * 0.95:
            break
            
        h0 = lhs_samples[i].copy()
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)

        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0

        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            # Use SLSQP for local optimization
            res = minimize(objective, h0, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                h_final = repair(h_final, n_points)
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
        except Exception:
            continue
    
    return (list(best_h), best_c5, n_points)
