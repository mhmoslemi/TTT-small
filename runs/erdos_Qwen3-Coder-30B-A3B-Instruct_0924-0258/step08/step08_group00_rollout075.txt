import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time()
    best_h = None
    best_c5 = float('inf')
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Multi-resolution approach
    resolutions = [35, 70, 140, 280]
    current_h = h_start.copy()
    current_n = n_points
    
    for resolution in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to new resolution
        if resolution > current_n:
            # Upsample
            x_old = np.linspace(0, 2, current_n)
            x_new = np.linspace(0, 2, resolution)
            current_h = np.interp(x_new, x_old, current_h)
            current_n = resolution
        elif resolution < current_n:
            # Downsample
            x_old = np.linspace(0, 2, current_n)
            x_new = np.linspace(0, 2, resolution)
            current_h = np.interp(x_new, x_old, current_h)
            current_n = resolution
            
        dx = 2.0 / current_n
        
        # Normalize sum to correct value
        current_h = current_h * (current_n/2) / np.sum(current_h)
        
        # Simulated annealing at this resolution
        temp = 0.1
        cooling_rate = 0.995
        min_temp = 1e-8
        patience = 100
        no_improvement = 0
        
        h_current = current_h.copy()
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            
        # SA iterations
        max_sa_iter = 5000
        for sa_iter in range(max_sa_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb a random index
            idx = random.randint(0, current_n - 1)
            delta = random.uniform(-0.05, 0.05)
            h_proposed = h_current.copy()
            h_proposed[idx] += delta
            h_proposed[idx] = max(0, min(1, h_proposed[idx]))
            
            # Rescale to maintain sum constraint
            h_proposed = h_proposed * (current_n/2) / np.sum(h_proposed)
            
            proposed_c5 = evaluate_erdos_solution(h_proposed, dx)
            
            # Accept or reject
            if proposed_c5 < current_c5:
                h_current = h_proposed
                current_c5 = proposed_c5
                no_improvement = 0
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_current.copy()
            else:
                # Accept with probability
                delta_e = proposed_c5 - current_c5
                if random.random() < np.exp(-delta_e / temp):
                    h_current = h_proposed
                    current_c5 = proposed_c5
                no_improvement += 1
                
            # Anneal temperature
            temp *= cooling_rate
            
            if no_improvement > patience or temp < min_temp:
                break
                
        # Local refinement with SLSQP
        if time() - start_time < budget_s * 0.9:
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - current_n / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(current_n)]
                
                res = minimize(objective, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-9})
                
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_c5 = c5_refined
                        best_h = h_refined
            except Exception:
                pass
                
        current_h = h_current.copy()
    
    # Final refinement
    if best_h is not None:
        dx = 2.0 / len(best_h)
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - len(best_h) / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(len(best_h))]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-10})
            
            if res.success:
                final_h = res.x
                final_c5 = objective(final_h)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except Exception:
            pass
            
    return (list(best_h), best_c5, len(best_h))
