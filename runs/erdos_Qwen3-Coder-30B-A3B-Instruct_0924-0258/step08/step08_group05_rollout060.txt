import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point than uniform
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1.0
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Trust Region method with cutting planes
    max_iter = 5000
    delta = 0.2  # Trust region radius
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate current solution
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            
        # Build constraints for current best solution
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        lag_indices = np.arange(len(corr))
        lags = lag_indices - (len(h_current) - 1)
        overlap_values = corr * dx
        
        # Find most violated constraints (top 10%)
        max_indices = np.argsort(overlap_values)[-int(0.1 * len(overlap_values)):]
        max_lags = lags[max_indices]
        
        # Solve convex relaxation within trust region
        h_vars = cp.Variable(n_points)
        C = cp.Variable()
        
        constraints = [
            cp.sum(h_vars) == n_points / 2.0,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add cutting planes for max-lag overlaps
        for k in max_lags:
            if k >= 0:
                expr = cp.sum(cp.multiply(
                    h_vars[:n_points - k],
                    1 - h_vars[k:]
                )) * dx
            else:
                expr = cp.sum(cp.multiply(
                    h_vars[-k:],
                    1 - h_vars[:-k]
                )) * dx
            constraints.append(expr <= C)
        
        # Minimize C
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_vars.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                
                # Update trust region
                d = h_new - h_current
                if np.linalg.norm(d) <= delta:
                    h_current = h_new
                else:
                    h_current = h_current + delta * d / np.linalg.norm(d)
                    
                # Adapt trust region
                new_c5 = evaluate_erdos_solution(h_current, dx)
                improvement = c5_current - new_c5
                if improvement > 1e-8:
                    delta = min(2.0 * delta, 1.0)
                else:
                    delta *= 0.5
                    
        except Exception:
            # Fallback to simple update
            delta *= 0.5
            h_current = np.clip(h_current + np.random.normal(0, delta, n_points), 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Final refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
