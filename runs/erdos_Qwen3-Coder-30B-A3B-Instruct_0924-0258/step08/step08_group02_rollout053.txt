import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter = 100000
    epsilon = 0.1
    
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    c5_current = evaluate_erdos_solution(h_current, dx)
    
    T = T_init
    accepted = 0
    total_iter = 0
    
    while T > T_min and time() - start_time < budget_s * 0.9:
        if total_iter >= max_iter:
            break
            
        # Perturb two random indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-epsilon, epsilon)
        
        # Ensure feasibility
        new_h = h_current.copy()
        new_h[i] += delta
        new_h[j] -= delta
        
        # Project back to feasible region
        if new_h[i] < 0:
            new_h[j] += new_h[i]
            new_h[i] = 0
        elif new_h[i] > 1:
            new_h[j] -= (new_h[i] - 1)
            new_h[i] = 1
            
        if new_h[j] < 0:
            new_h[i] += new_h[j]
            new_h[j] = 0
        elif new_h[j] > 1:
            new_h[i] -= (new_h[j] - 1)
            new_h[j] = 1
            
        # Adjust to maintain sum constraint
        sum_diff = np.sum(new_h) - n_points/2
        if abs(sum_diff) > 1e-10:
            if sum_diff > 0:
                # Reduce some values
                idx = np.argsort(new_h)[::-1]
                for k in idx:
                    if sum_diff <= 0:
                        break
                    reduction = min(new_h[k], sum_diff)
                    new_h[k] -= reduction
                    sum_diff -= reduction
            else:
                # Increase some values
                idx = np.argsort(new_h)
                for k in idx:
                    if sum_diff >= 0:
                        break
                    increment = min(1 - new_h[k], -sum_diff)
                    new_h[k] += increment
                    sum_diff += increment
                    
        c5_new = evaluate_erdos_solution(new_h, dx)
        
        # Acceptance probability
        if c5_new < c5_current or np.random.random() < np.exp(-(c5_new - c5_current) / T):
            h_current = new_h
            c5_current = c5_new
            accepted += 1
            
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
        
        # Cooling schedule
        T *= alpha
        total_iter += 1
        
        if total_iter % 5000 == 0:
            # Occasionally adjust epsilon based on progress
            if accepted == 0:
                epsilon *= 0.9
            else:
                accepted = 0
    
    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        # Refine using scipy local optimizer
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-12}
        )
        
        # Ensure it's feasible
        refined_h = np.clip(res.x, 0, 1)
        refined_h = refined_h * (n_points/2) / np.sum(refined_h)
        refined_c5 = evaluate_erdos_solution(refined_h, dx)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
            
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
