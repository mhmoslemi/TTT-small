import numpy as np
from time import time
import cvxpy as cp
from scipy.linalg import svd

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # SDP-based optimization
    # Binary search on C5
    low, high = 0.38, 0.38085
    eps = 1e-5
    max_iters = 100
    
    for iteration in range(max_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        
        # Formulate SDP
        h_var = cp.Variable(n_points)
        Y = cp.Variable((n_points, n_points), symmetric=True)
        
        constraints = [
            Y >> 0,  # Positive semidefinite
            cp.diag(Y) == h_var,  # Diagonal constraint
            cp.sum(h_var) == n_points / 2  # Normalization constraint
        ]
        
        # Overlap constraints for a subset of shifts
        shifts = np.random.choice(n_points, size=min(100, n_points), replace=False)
        for k in shifts:
            # Sum i h_i (1 - h_{i+k}) <= mid * dx
            # This is equivalent to sum h_i - sum Y_{i, i+k} <= mid * dx
            if k == 0:
                # No shift
                expr = cp.sum(h_var) - cp.sum(Y.diagonal())
            else:
                # Shifted sum
                expr = cp.sum(h_var) - cp.sum(Y[np.arange(n_points), np.roll(np.arange(n_points), k)])
            constraints.append(expr <= mid * dx)
        
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    # Renormalize
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    c5_sol = evaluate_erdos_solution(h_sol, dx)
                    if c5_sol < best_c5:
                        best_c5 = c5_sol
                        best_h = h_sol.copy()
                    high = mid
                else:
                    low = mid
            else:
                low = mid
        except Exception:
            low = mid
        
        if high - low < eps:
            break
    
    # Local refinement
    h_refined = best_h.copy()
    dx_ref = 2.0 / n_points
    for i in range(1000):
        if time() - start_time > budget_s * 0.95:
            break
            
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i_idx in range(n_points):
            j_idx = i_idx + max_lag
            if 0 <= j_idx < n_points:
                grad[i_idx] = 1 - 2 * h_refined[i_idx] - h_refined[j_idx]
            else:
                grad[i_idx] = 1 - 2 * h_refined[i_idx]
        
        h_new = h_refined - 0.005 * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx_ref)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_refined = h_new.copy()
        else:
            break
    
    return (list(best_h), best_c5, n_points)
