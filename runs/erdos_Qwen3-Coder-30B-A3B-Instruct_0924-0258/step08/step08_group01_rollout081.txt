import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for genetic algorithm
    pop_size = 50
    num_generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    stagnation_limit = 100

    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)
    else:
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)
    
    # Helper function for creating new individuals
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        return h
    
    # Tournament selection
    def tournament_selection(pop, k=3):
        tournament = np.random.choice(len(pop), k)
        fitnesses = [evaluate_erdos_solution(pop[i], dx) for i in tournament]
        winner_idx = tournament[np.argmin(fitnesses)]
        return pop[winner_idx].copy()
    
    # Blend crossover
    def blend_crossover(parent1, parent2, alpha=0.3):
        child1 = np.zeros_like(parent1)
        child2 = np.zeros_like(parent2)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child1[i] = parent1[i]
                child2[i] = parent2[i]
            else:
                a = min(parent1[i], parent2[i])
                b = max(parent1[i], parent2[i])
                range_ = b - a
                child1[i] = np.clip(a - alpha * range_ + np.random.rand() * (1 + 2 * alpha) * range_, 0, 1)
                child2[i] = np.clip(a - alpha * range_ + np.random.rand() * (1 + 2 * alpha) * range_, 0, 1)
        # Adjust to meet sum constraint
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        return child1, child2
    
    # Mutation
    def mutate(h, mutation_rate, generation):
        step_size = 0.05 * (0.95 ** generation)
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.normal(0, step_size)
        # Enforce constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

            child2 = mutate(child2, mutation_rate, generation)
            
            new_population.extend([child1, child2])

                return np.sum(h) - n_points / 2.0

                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:

    def objective(h):
        return evaluate_erdos_solution(h, dx)

        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}

                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Ensure final solution meets constraints exactly
