import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) * dx = 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (1.0 / n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex perturbation parameters
    epsilon_0 = 0.001
    beta = 0.99
    max_iter = 5000
    top_k = 100
    tolerance = 1e-6
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        # Get top k shifts that cause maximum overlaps
        top_shifts = np.argsort(corr)[-top_k:]
        
        # Solve convex optimization problem
        h_var = cp.Variable(n_points)
        objective = 0
        
        # Add convex regularization term
        epsilon = epsilon_0 * (beta ** iteration)
        objective += epsilon * cp.sum_squares(h_var)
        
        # Add main objective terms
        for k in top_shifts:
            # Extract the shifted copy
            h_shifted = np.roll(1 - h, k)
            h_shifted = h_shifted[:n_points]
            objective += cp.sum(cp.multiply(h_var, h_shifted))
            
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Solve optimization problem
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h_new = h_var.value
        except:
            h_new = h.copy()
            
        # Project to feasible region
        h_new = project_to_feasible(h_new, n_points, dx)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            last_improvement = iteration
        else:
            # Check for stagnation
            if iteration - last_improvement > 1000:
                break
                
        # Reduce perturbation strength over time  
        if iteration % 500 == 0 and iteration > 0:
            epsilon_0 *= 0.9
    
    # Final local refinement
    if time() - start_time < budget_s * 0.9:
        num_refinements = 100
        for _ in range(num_refinements):
            if time() - start_time > budget_s * 0.95:
                break
            # Generate random perturbation
            noise = np.random.uniform(-0.01, 0.01, n_points)
            h_test = h + noise
            h_test = project_to_feasible(h_test, n_points, dx)
            
            c5_test = evaluate_erdos_solution(h_test, dx)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
    
    return (list(best_h), best_c5, n_points)
