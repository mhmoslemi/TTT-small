import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convex relaxation and coordinate descent hybrid
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Parameters for convex relaxation
    lambda_reg = 0.1
    alpha_reg = 0.01
    
    # Coordinate descent parameters
    coord_ratio = 0.2
    coord_iters = 50
    
    no_improvement_count = 0
    max_no_improvement = 50
    
    for iter in range(1000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convex relaxation step to initialize
        try:
            h_relaxed = np.zeros(n_points)
            
            # Formulate the convex problem
            h_var = cp.Variable(n_points)
            constraints = [
                cp.sum(h_var) == n_points/2,
                h_var >= 0,
                h_var <= 1
            ]
            
            # Objective function (sum of squared overlaps)
            overlaps = []
            for k in range(n_points):
                overlap = 0
                for i in range(n_points):
                    j = (i + k) % n_points
                    overlap += h_var[i] * (1 - h_var[j])
                overlaps.append(overlap)
            
            objective = cp.Minimize(
                lambda_reg * cp.sum_squares(overlaps) + 
                alpha_reg * cp.sum_squares(h_var)
            )
            
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_relaxed = h_var.value
                h_relaxed = np.clip(h_relaxed, 0, 1)
                h_relaxed = h_relaxed * (n_points/2) / np.sum(h_relaxed)
                h = h_relaxed.copy()
        except:
            pass
            
        # Coordinate descent step
        indices = np.random.permutation(n_points)
        coord_indices = indices[:int(n_points * coord_ratio)]
        
        # Store original values
        h_orig = h.copy()
        
        # Optimize selected coordinates
        for _ in range(coord_iters):
            grad = np.zeros(n_points)
            for k in range(n_points):
                for i in coord_indices:
                    j = (i + k) % n_points
                    grad[i] += -2 * dx * (1 - h[j] - h[i])
            
            h_new = h - 0.1 * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Check if improving
            h = h_new.copy()
            
        # Normalize and check improvement
        h = h * (n_points/2) / np.sum(h)
        current_c5 = evaluate_erdos_solution(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                break
                
        # Early stopping if time limit nears
        if time() - start_time > budget_s * 0.95:
            break
    
    # Final refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        # Renormalize best_h to enforce sum constraint
        best_h = best_h * (n_points/2) / np.sum(best_h)
        
        def objective(h):
            return np.max(np.correlate(h, 1 - h, mode='full') * dx)
            
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
