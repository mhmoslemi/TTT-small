import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Try SDP approach
    try:
        # Create CVXPY variables
        h_vars = cp.Variable(n_points, bounds=(0, 1))
        
        # Constraint: sum(h) = n_points/2
        sum_constraint = cp.Equality(cp.sum(h_vars), n_points / 2)
        
        # Objective to minimize
        obj = cp.Minimize(cp.Variable(1))
        
        # Build constraints for the overlap
        constraints = [sum_constraint]
        max_k = n_points - 1
        
        # For each lag k, add constraint for the overlap
        for k in range(-max_k, max_k + 1):
            if k >= 0:
                # h[i] * (1 - h[i + k]) for valid i
                valid_indices = range(n_points - k)
                expr = cp.sum(cp.multiply(h_vars[valid_indices], 1 - h_vars[valid_indices + k])) * dx
            else:
                # h[i] * (1 - h[i + k]) for valid i
                valid_indices = range(-k, n_points)
                expr = cp.sum(cp.multiply(h_vars[valid_indices], 1 - h_vars[valid_indices + k])) * dx
            
            # Constraint: expr <= obj
            constraints.append(expr <= obj)
        
        # Create and solve problem
        prob = cp.Problem(obj, constraints)
        
        # Solve with time limit
        prob.solve(solver=cp.MOSEK, time_limit=budget_s * 0.9)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_vars.value
            if h_opt is not None:
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points/2) / np.sum(h_opt)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    
    except Exception:
        # If SDP fails, fall back to gradient descent
        pass
    
    return (list(best_h), best_c5, n_points)
