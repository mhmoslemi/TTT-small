import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter = int(1e5)
    
    # Perturbation size
    epsilon = 0.1
    
    # Current state
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)  # Ensure sum constraint
    c5_current = evaluate_erdos_solution(h_current, dx)
    
    if c5_current < best_c5:
        best_c5 = c5_current
        best_h = h_current.copy()
    
    T = T_init
    iter_count = 0
    
    while T > T_min and time() - start_time < budget_s * 0.9:
        iter_count += 1
        
        # Create new candidate by perturbing two random indices
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Create perturbation
        delta = np.random.uniform(-epsilon, epsilon)
        
        # Apply perturbation
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / T):
            h_current = h_new
            c5_current = c5_new
            
            # Update best solution
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
        
        # Cool down
        T *= alpha
        
        # Adaptive epsilon
        if iter_count % 1000 == 0:
            epsilon = max(1e-4, epsilon * 0.99)
    
    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        h_refined = best_h.copy()
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        res = minimize(
            objective,
            h_refined,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            h_refined = res.x
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final check
    final_h = best_h.copy()
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
