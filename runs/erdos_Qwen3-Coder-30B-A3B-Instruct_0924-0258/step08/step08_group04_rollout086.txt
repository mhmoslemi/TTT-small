import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # PSO parameters
    num_particles = 75
    inertia = 0.8
    inertia_decay = 0.999
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    perturbation_rate = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    particle_best = []
    particle_best_values = []
    
    for i in range(num_particles):
        if i == 0:
            h = h_current.copy()
        else:
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        particle_best.append(h.copy())
        particle_best_values.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    global_best_idx = np.argmin(particle_best_values)
    global_best = particle_best[global_best_idx].copy()
    global_best_value = particle_best_values[global_best_idx]
    
    # PSO iterations
    max_iter = int(1e6)
    iter_count = 0
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = cognitive_weight * r1 * (particle_best[i] - particles[i])
            social = social_weight * r2 * (global_best - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible set
            particles[i] = np.clip(particles[i], 0, 1)
            # Renormalize to satisfy sum constraint
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            current_value = evaluate_erdos_solution(particles[i], dx)
            
            # Update particle best
            if current_value < particle_best_values[i]:
                particle_best[i] = particles[i].copy()
                particle_best_values[i] = current_value
                
                # Update global best
                if current_value < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = current_value
                    last_improvement = iter_count
        
        # Decay inertia
        inertia *= inertia_decay
        
        # Occasionally perturb particles to avoid local minima
        if iter_count % 1000 == 0 and iter_count > 0:
            for i in range(num_particles):
                if np.random.random() < perturbation_rate:
                    particles[i] = np.random.uniform(0, 1, n_points)
                    particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
        
        # Check for early stopping
        if iter_count - last_improvement > 5000:
            break
            
        iter_count += 1
    
    # Final refinement with local search
    h_refined = global_best.copy()
    dx = 2.0 / n_points
    
    # Local search around global best
    best_h = h_refined.copy()
    best_c5 = global_best_value
    
    # Try small adjustments
    step_size = 0.001
    for _ in range(1000):
        # Sample a random direction
        direction = np.random.normal(0, 0.01, n_points)
        # Adjust the direction to preserve sum constraint
        direction -= np.mean(direction)
        
        h_new = h_refined + step_size * direction
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Adjust step size based on progress
        if c5_new >= evaluate_erdos_solution(h_refined, dx):
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            h_refined = h_new.copy()
            step_size = min(step_size * 1.05, 0.01)
    
    return (list(best_h), best_c5, n_points)
