import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convex relaxation initialization
    try:
        # Use CVXPY to solve a relaxed version
        h_relax = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum_squares(h_relax) + 0.01 * cp.sum(cp.square(1 - h_relax)))
        constraints = [cp.sum(h_relax) == n_points/2,
                       h_relax >= 0,
                       h_relax <= 1]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CLARABEL, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_start = np.array(h_relax.value).reshape(-1)
            h_start = np.clip(h_start, 0, 1)
            h_start = h_start * (n_points/2) / np.sum(h_start)
    except:
        pass
    
    h = h_start.copy()
    
    # Local coordinate descent with random cycles
    max_iter = 500
    no_improve_count = 0
    patience = 50
    
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Select a random subset for optimization (20%)
        indices = np.random.permutation(n_points)[:max(1, n_points // 5)]
        indices.sort()
        
        # Prepare the fixed values
        h_fixed = h.copy()
        for i in indices:
            h_fixed[i] = 0  # Will optimize this
        
        # Create the optimization function for indices
        def obj_and_grad(h_var):
            h_current = h_fixed.copy()
            for i, val in zip(indices, h_var):
                h_current[i] = val
            
            h_current = h_current * (n_points/2) / np.sum(h_current)
            
            # Evaluate overlaps
            overlaps = np.correlate(h_current, 1 - h_current, mode='full') * dx
            max_overlap = np.max(overlaps)
            l_max = np.argmax(overlaps)
            
            # Compute gradient
            gradient = np.zeros(len(indices))
            for idx, i in enumerate(indices):
                j = (i + l_max) % n_points
                gradient[idx] = -2 * dx * (1 - h_current[j] - h_current[i])
            
            return max_overlap, gradient
        
        # Optimize the selected indices
        init_vals = [h[i] for i in indices]
        bounds = [(0, 1) for _ in indices]
        
        try:
            # Use L-BFGS-B to optimize selected coordinates
            res = minimize(
                lambda x: obj_and_grad(x)[0],
                init_vals,
                method='L-BFGS-B',
                bounds=bounds,
                jac=lambda x: obj_and_grad(x)[1],
                options={'ftol': 1e-9, 'maxiter': 20}
            )
            
            if res.success:
                for i, val in zip(indices, res.x):
                    h[i] = max(0, min(1, val))
                
                # Normalize
                h = h * (n_points/2) / np.sum(h)
                
                # Update best solution if improved
                current_c5 = evaluate_erdos_solution(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1
                    
        except Exception:
            pass
        
        # Early stopping condition
        if no_improve_count >= patience:
            break
    
    # Final refinement
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
        
    return (list(best_h), best_c5, n_points)
