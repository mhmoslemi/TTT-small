import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a more structured initial solution
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1.0
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Thresholding-based approach
    threshold = 0.1
    cooling_rate = 0.95
    max_iterations = 500
    n_refine = max(1, n_points // 2)
    
    h = h_start.copy()
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        k_max = np.argmax(overlaps)
        
        # Compute per-coordinate gradients
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Sort by gradient magnitude to identify candidates for thresholding
        idx_sort = np.argsort(np.abs(gradient))[::-1][:n_refine]
        
        # Thresholding operation
        h_new = h.copy()
        for i in idx_sort:
            if h[i] >= 1 - threshold:
                h_new[i] = 1.0
            elif h[i] <= threshold:
                h_new[i] = 0.0
        
        # Rebalance to satisfy sum constraint
        current_sum = np.sum(h_new)
        if abs(current_sum - n_points/2) > 1e-6:
            # Distribute deviation to indices with largest |gradient|
            delta = (n_points/2) - current_sum
            if abs(delta) > 1e-8:
                idx_grad = np.argsort(np.abs(gradient))[::-1]
                for i in idx_grad:
                    if abs(delta) < 1e-8:
                        break
                    # Compute how much we can move
                    move = min(abs(delta), 1.0 - h_new[i]) if delta > 0 else min(abs(delta), h_new[i])
                    h_new[i] += move * (1 if delta > 0 else -1)
                    delta -= move * (1 if delta > 0 else -1)
        
        h_new = np.clip(h_new, 0, 1)
        
        # Local coordinate-wise refinement
        for i in range(n_points):
            def obj_val(h_i):
                h_test = h_new.copy()
                h_test[i] = h_i
                return evaluate_erdos_solution(h_test, dx)
            
            # Use a golden-section search between 0 and 1
            a, b = 0.0, 1.0
            tol = 1e-5
            res = 0.381966 * (b - a)  # Golden ratio proportion
            x1 = a + res
            x2 = b - res
            
            while abs(b - a) > tol:
                if obj_val(x1) < obj_val(x2):
                    b = x2
                    x2 = x1
                    x1 = a + 0.381966 * (b - a)
                else:
                    a = x1
                    x1 = x2
                    x2 = b - 0.381966 * (b - a)
                
                if time() - start_time > budget_s * 0.95:
                    break
            
            h_new[i] = (a + b) / 2.0
        
        # Ensure sum constraint after refinement
        h_new = h_new * (n_points/2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        
        # Check if improvement
        current_c5 = evaluate_erdos_solution(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        
        # Update h for next iteration
        h = h_new.copy()
        
        # Reduce threshold
        threshold = max(0.01, threshold * cooling_rate)
        
        if threshold < 0.01:
            # Transition to coordinate-wise optimization
            n_refine = max(1, n_points // 4)
            
    # Final refinement with L-BFGS
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
