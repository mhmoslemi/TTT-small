import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter = int(1e5)
    
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    T = T_init
    
    # Annealing loop
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Perturb two random indices
        idx1, idx2 = random.randint(0, n_points-1), random.randint(0, n_points-1)
        delta = random.uniform(-0.1, 0.1)
        
        # Adjust values with boundary handling
        new_h = h_current.copy()
        new_h[idx1] += delta
        new_h[idx2] -= delta
        
        # Clip to [0, 1]
        new_h = np.clip(new_h, 0, 1)
        
        # Maintain sum constraint
        sum_diff = np.sum(new_h) - n_points/2
        if abs(sum_diff) > 1e-10:
            # Distribute difference to first element
            new_h[0] -= sum_diff
            new_h = np.clip(new_h, 0, 1)
        
        # Compute overlaps
        current_c5 = evaluate_erdos_solution(h_current, dx)
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or random.random() < np.exp(-delta_c5 / T):
            h_current = new_h
            if new_c5 < current_c5:
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = h_current.copy()
                    best_c5 = current_c5
        
        # Cooling
        T = max(T_min, T * alpha)
    
    # Local refinement
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def obj(h):
            return np.max(np.correlate(h, 1 - h, mode='full') * dx)
        
        res = minimize(
            obj,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
