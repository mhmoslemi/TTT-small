import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        n_points = len(h_start)

    # Population-based approach: genetic algorithm
    population_size = 50
    population = []
    
    # Initialize population
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        population[0] = population[0] * (n_points/2) / np.sum(population[0])
        
    for i in range(population_size - len(population)):
        # Create mixed feasible solutions
        h = np.zeros(n_points)
        if n_points % 2 == 0:
            ones_count = n_points // 2
            h[:ones_count] = 1.0
        else:
            ones_count = (n_points - 1) // 2
            h[:ones_count] = 1.0
            h[ones_count] = 0.5
        np.random.shuffle(h)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    def select_tournament(pop, k=5):
        selected = random.sample(pop, min(k, len(pop)))
        return min(selected, key=fitness)
        
    def crossover(h1, h2):
        # Uniform crossover
        mask = np.random.random(n_points) > 0.5
        child = np.where(mask, h1, h2)
        # Re-normalize to sum constraint
        child = child * (n_points/2) / np.sum(child)
        return child
    
    def mutate(h, rate=0.1):
        child = h.copy()
        # Flip mutation
        indices = np.random.choice(n_points, int(n_points * rate), replace=False)
        for i in indices:
            if random.random() < 0.5:
                child[i] = max(0, child[i] - 0.1)
            else:
                child[i] = min(1, child[i] + 0.1)
        # Re-normalize
        child = child * (n_points/2) / np.sum(child)
        return child
    
    # Main GA loop
    best_individual = min(population, key=fitness)
    best_fitness = fitness(best_individual)
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95:
        # Evaluate all individuals
        population_fitness = [fitness(h) for h in population]
        
        # Create new population through selection, crossover, mutation
        new_population = [best_individual]  # Elitism
        
        while len(new_population) < population_size:
            parent1 = select_tournament(population)
            parent2 = select_tournament(population)
            
            child = crossover(parent1, parent2)
            child = mutate(child)
            
            new_population.append(child)
            
        population = new_population
        current_best = min(population, key=fitness)
        current_fitness = fitness(current_best)
        
        if current_fitness < best_fitness:
            best_fitness = current_fitness
            best_individual = current_best.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 100:
            break
    
    # Final refinement using local search
    try:
        from scipy.optimize import minimize
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_individual, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            final_h = res.x
            final_c5 = objective(final_h)
            if final_c5 < best_fitness:
                best_fitness = final_c5
                best_individual = final_h
    except Exception:
        pass
    
    return (list(best_individual), best_fitness, n_points)
