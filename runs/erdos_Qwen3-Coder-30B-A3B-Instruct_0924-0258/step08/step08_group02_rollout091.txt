import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from existing construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # alternate between convex relaxation and coordinate descent
    max_iterations = 1000
    no_improve_count = 0
    max_no_improve = 50
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # convex relaxation step
        try:
            # set up CVXPY problem for relaxation
            h_cvx = cp.Variable(n_points)
            # proxy objective to minimize max overlap
            # we'll use a quadratic penalty instead
            objective = cp.Minimize(
                cp.sum_squares(
                    cp.reshape(cp.Variable(n_points + n_points - 1), (n_points, n_points)) @ 
                    cp.kron(cp.ones((1, n_points)), h_cvx) * dx  # placeholder for more precise formulation
                )
            )
            # hard constraints
            constraints = [
                cp.sum(h_cvx) == n_points / 2,
                h_cvx >= 0,
                h_cvx <= 1
            ]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_relaxed = h_cvx.value
                h_relaxed = np.clip(h_relaxed, 0, 1)
                h_relaxed = h_relaxed * (n_points/2) / np.sum(h_relaxed)
                h = h_relaxed.copy()
        except:
            pass
            
        # coordinate descent with random subset
        coords_to_optimize = np.random.choice(n_points, size=int(n_points*0.3), replace=False)
        h_old = h.copy()
        
        # optimize selected subset using gradient descent
        for coord in coords_to_optimize:
            # Store current value
            old_val = h[coord]
            best_val = old_val
            
            # Try several values in a small neighborhood
            best_overlap = float('inf')
            for val in np.linspace(0, 1, 20):
                h[coord] = val
                # normalize to sum constraint
                h = h * (n_points/2) / np.sum(h)
                overlap = evaluate_erdos_solution(h, dx)
                if overlap < best_overlap:
                    best_overlap = overlap
                    best_val = val
                    
            h[coord] = best_val
            h = h * (n_points/2) / np.sum(h)
            
        # normalize and check progress
        h = h * (n_points/2) / np.sum(h)
        current_c5 = evaluate_erdos_solution(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count >= max_no_improve:
            break
    
    # final refinement with scipy
    try:
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return -np.max(overlaps)
        
        bounds = [(0, 1) for _ in range(n_points)]
        h_normalized = best_h * (n_points/2) / np.sum(best_h)
        
        res = minimize(objective, h_normalized, method='L-BFGS-B', bounds=bounds)
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
        
    return (list(best_h), best_c5, n_points)
