import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def normalize_h(h, n_points):
    """Normalize h to satisfy sum constraint"""
    h_sum = np.sum(h)
    if abs(h_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / h_sum
    return np.clip(h, 0, 1)

def crossover(parent1, parent2):
    """Uniform crossover"""
    mask = np.random.rand(len(parent1)) > 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(h, mutation_rate=0.1):
    """Mutate h with random changes"""
    h = h.copy()
    n_points = len(h)
    
    # Flip mutation
    if np.random.rand() < 0.5:
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        h[idx1] = np.clip(h[idx1] + np.random.choice([-0.1, 0.1]), 0, 1)
        h[idx2] = np.clip(h[idx2] + np.random.choice([-0.1, 0.1]), 0, 1)
    
    # Perturb mutation
    n_mutations = int(mutation_rate * n_points)
    if n_mutations > 0:
        indices = np.random.choice(n_points, n_mutations, replace=False)
        for i in indices:
            factor = np.random.uniform(0.9, 1.1)
            h[i] = np.clip(h[i] * factor, 0, 1)
    
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Tournament selection"""
    selected_indices = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        selected_indices.append(winner_idx)
    return selected_indices

    best_c5 = np.min(fitnesses)
    
    no_improvement_count = 0
    max_no_improvement = 100

    while time() - start_time < budget_s * 0.95:
        # Selection
        selected_indices = tournament_selection(population, fitnesses)
        selected_population = [population[i] for i in selected_indices]
        
        # Crossover and mutation to create new population
        new_population = [best_h.copy()]
