import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def repair(h, n_points):
    h = np.clip(h, 0, 1)
    if np.sum(h) == 0:
        return h
    h = h * (n_points/2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-start optimization with Latin hypercube sampling
    n_starts = 50
    best_c5 = float('inf')
    best_h = None
    
    # Generate diverse starting points using Latin hypercube
    def latin_hypercube_sample(n, samples):
        result = np.zeros((samples, n))
        for i in range(n):
            # Generate samples for dimension i
            samples_i = np.random.random(samples)
            # Sort and space out
            samples_i = np.sort(samples_i)
            # Assign to intervals
            for j in range(samples):
                result[j, i] = (j + samples_i[j]) / samples
        return result
    
    # Sample from LHS and normalize to satisfy sum constraint
    initial_guesses = latin_hypercube_sample(n_points, n_starts)
    for i in range(n_starts):
        h0 = initial_guesses[i]
        # Normalize to sum to n_points/2
        h0 = h0 * (n_points/2) / np.sum(h0)
        # Ensure bounds
        h0 = np.clip(h0, 0, 1)
        initial_guesses[i] = h0
    
    # Add the initial_h_values if available
    if 'initial_h_values' in globals():
        initial_guesses[0] = np.array(initial_h_values)
    
    for i, h0 in enumerate(initial_guesses):
        if time() - start_time > budget_s * 0.95:
            break
        h = h0.copy()
        
        # Run SLSQP optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                h_final = repair(h_final, n_points)
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
        except Exception:
            # If there's an error, keep trying others
            continue
    
    if best_h is None:
        # fallback to original
        best_h = h_start.copy()
        best_c5 = evaluate_erdos_solution(h_start, dx)
    
    return (list(best_h), best_c5, n_points)
