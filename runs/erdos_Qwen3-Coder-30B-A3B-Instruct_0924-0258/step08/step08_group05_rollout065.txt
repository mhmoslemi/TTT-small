import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    # Normalize to satisfy sum constraint
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # GA parameters
    population_size = 50
    max_generations = 10000
    tournament_size = 5
    mutation_rate = 0.1
    crossover_rate = 0.8
    no_improvement_limit = 100
    
    # Initialize population
    population = []
    population.append(h_start.copy())
    for i in range(1, population_size):
        if n_points % 2 == 0:
            h = np.zeros(n_points)
            h[:n_points//2] = 1.0
        else:
            h = np.zeros(n_points)
            h[:(n_points-1)//2] = 1.0
            h[(n_points-1)//2] = 0.5
        # Normalize
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # GA main loop
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    no_improvement_count = 0
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= no_improvement_limit:
            break
            
        # Selection
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Crossover and mutation
        new_population = [best_h.copy()]  # Elitism
        while len(new_population) < population_size:
            if np.random.random() < crossover_rate and len(new_population) + 1 < population_size:
                parent1_idx, parent2_idx = np.random.choice(population_size, 2, replace=False)
                parent1 = selected[parent1_idx]
                parent2 = selected[parent2_idx]
                
                child1 = np.zeros(n_points)
                child2 = np.zeros(n_points)
                
                # Uniform crossover
                for i in range(n_points):
                    if np.random.random() < 0.5:
                        child1[i] = parent1[i]
                        child2[i] = parent2[i]
                    else:
                        child1[i] = parent2[i]
                        child2[i] = parent1[i]
                
                # Adjust children to satisfy sum constraint
                child1 = child1 * (n_points/2) / np.sum(child1)
                child2 = child2 * (n_points/2) / np.sum(child2)
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Mutation only
                parent_idx = np.random.choice(population_size)
                child = selected[parent_idx].copy()
                
                # Flip mutation: randomly select two indices
                if np.random.random() < mutation_rate:
                    i, j = np.random.choice(n_points, 2, replace=False)
                    child[i] = np.clip(child[i] + 0.1, 0, 1)
                    child[j] = np.clip(child[j] - 0.1, 0, 1)
                    # Adjust to meet constraint
                    child = child * (n_points/2) / np.sum(child)
                
                new_population.append(child)
        
        # Update population
        population = new_population
    
    # Final refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
