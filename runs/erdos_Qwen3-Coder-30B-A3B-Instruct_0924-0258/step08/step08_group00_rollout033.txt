import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different starting points
    for attempt in range(3):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Initialize with different configurations
        if attempt == 0:
            h = h_start.copy()
        elif attempt == 1:
            # Try a more sparse construction
            h = np.zeros(n_points)
            h[0:n_points//2] = 1.0
            h = h * (n_points/2) / np.sum(h)
        else:
            # Try a more structured approach 
            h = np.ones(n_points) * 0.25
            h[0:n_points//4] = 0.75
            h[-n_points//4:] = 0.75
            h = h * (n_points/2) / np.sum(h)
        
        # Local optimization with different methods
        # First, try gradient descent with adaptive steps
        for iter_num in range(100):
            if time() - start_time > budget_s * 0.9:
                break
                
            corr = np.correlate(h, 1 - h, mode='full')
            max_lag_idx = np.argmax(corr)
            max_overlap = corr[max_lag_idx] * dx
            
            if max_overlap < best_c5:
                best_c5 = max_overlap
                best_h = h.copy()
                
            # Compute direction of improvement
            max_lag = max_lag_idx - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + max_lag
                if 0 <= j < n_points:
                    # Simple gradient: approximating derivative of overlap
                    grad[i] = (1 - 2 * h[i] - h[j]) / n_points
                
            step_size = 0.01
                
            # Update with gradient descent
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            if evaluate_erdos_solution(h_new, dx) < max_overlap:
                h = h_new
            else:
                step_size *= 0.5
                if step_size < 1e-8:
                    break
    
    # Final refinement using scipy.optimize
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
