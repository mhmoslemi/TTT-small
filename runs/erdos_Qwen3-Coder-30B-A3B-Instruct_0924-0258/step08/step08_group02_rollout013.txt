import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Threshold-based binary search
    max_iterations = 500
    step_size = 0.01
    
    # Initial binary approximation
    h_binary = np.round(h_start).astype(float)
    h_binary = h_binary * (n_points/2) / np.sum(h_binary)
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.8:
            break
        
        # Compute overlaps
        overlaps = np.correlate(h_binary, 1 - h_binary, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute effective gradient for max overlap
        # gradient[i] = -2 * dx * sum_{j} (1 - h[j] - h[i]) where j = (i + l_max) % n
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h_binary[j] - h_binary[i])
        
        # Update using threshold-based approach
        threshold = 0.01 + (0.05 * (1 - iter / max_iterations))  # Adaptive threshold
        update_positions = np.where(np.abs(gradient) > threshold)[0]
        
        for pos in update_positions:
            if gradient[pos] > 0:
                h_binary[pos] = min(1.0, h_binary[pos] + step_size)
            else:
                h_binary[pos] = max(0.0, h_binary[pos] - step_size)
        
        # Rebalance to satisfy constraint: sum(h) = n/2
        current_sum = np.sum(h_binary)
        if abs(current_sum - n_points/2) > 1e-6:
            delta = (n_points/2 - current_sum) / n_points
            h_binary = np.clip(h_binary + delta, 0, 1)
        
        # Verify constraints
        h_binary = h_binary * (n_points/2) / np.sum(h_binary)
        
        # Track best solution
        current_c5 = evaluate_erdos_solution(h_binary, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_binary.copy()
            
        # Reduce step size gradually
        step_size = max(1e-6, step_size * 0.999)
    
    # Phase 2: Local refinement using projected L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)
        
        # Project to constraint
        def project(h):
            h = h * (n_points/2) / np.sum(h)
            return np.clip(h, 0, 1)
        
        # Refine the best binary solution
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = project(res.x)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception as e:
        pass

    # Final validation
    final_h = best_h
    final_c5 = best_c5
    final_sum = np.sum(final_h)
    
    # Ensure correct sum
    final_h = final_h * (n_points/2) / final_sum
    
    return (list(final_h), final_c5, n_points)
