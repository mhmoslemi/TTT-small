import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Try to construct a good initial solution
        h_start = np.ones(n_points) * 0.5

    # Normalize sum to n_points/2
    dx = 2.0 / n_points
    h_current = h_start * (n_points / 2) / np.sum(h_start)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing
    temp = 1.0  # Initial temperature
    cooling_rate = 0.999
    min_temp = 1e-6
    max_iter = 100000
    no_improvement_limit = 5000
    no_improvement_count = 0
    
    # Ensure we have a valid starting solution
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try to make a small change
        idx1 = random.randint(0, n_points - 1)
        idx2 = random.randint(0, n_points - 1)
        
        # Determine feasible delta
        max_delta = min(h_current[idx1], 1 - h_current[idx2])
        if max_delta == 0:
            continue
            
        delta = random.uniform(0, max_delta)
        
        # Make the move
        h_new = h_current.copy()
        h_new[idx1] -= delta
        h_new[idx2] += delta
        
        # Ensure bounds
        if h_new[idx1] < 0 or h_new[idx2] > 1:
            continue
            
        # Renormalize to maintain sum constraint
        sum_h = np.sum(h_new)
        h_new = h_new * (n_points/2) / sum_h
        
        # Ensure bounds again after normalization
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        # Accept or reject
        if c5_new < c5_current:
            h_current = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                no_improvement_count = 0
        else:
            # Probabilistic acceptance
            try:
                p = np.exp(-(c5_new - c5_current) / temp)
                if random.random() < p:
                    h_current = h_new
            except:
                pass
                
        # Cool down
        temp *= cooling_rate
        no_improvement_count += 1
        if no_improvement_count > no_improvement_limit:
            temp = max(temp * 0.9, min_temp)  # Reduce temp if no improvement
            no_improvement_count = 0
            
        if temp < min_temp:
            break
    
    # Final SoP-based optimization
    try:
        from scipy.optimize import minimize
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
