import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Normalize initial h to satisfy sum constraint
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Helper to project h to feasible region
    def project_h(h, target_sum):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    # Coordinate descent optimization with convex relaxation
    max_iterations = 200

    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convex relaxation step using CVXPY
        if n_points <= 100:  # Only use convex relaxation for smaller problems
            try:
                h_relaxed = np.zeros(n_points)
                # Create variables
                h_vars = cp.Variable(n_points)
                
                # Objective: minimize sum of squared overlaps plus regularization
                objective = cp.Minimize(
                    cp.sum_squares(cp.vstack([cp.sum(h_vars[i] * (1 - h_vars[(i + k) % n_points]) 
                                              for i in range(n_points)]) for k in range(n_points)]) / n_points
                    + 0.01 * cp.sum_squares(h_vars)
                )
                
                # Constraints
                constraints = [
                    cp.sum(h_vars) == n_points/2,
                    h_vars >= 0,
                    h_vars <= 1
                ]
                
                # Solve
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                
                if prob.status == cp.OPTIMAL:
                    h_relaxed = h_vars.value
                    h_relaxed = project_h(h_relaxed, n_points/2)
                    h = h_relaxed.copy()
                else:
                    h = best_h.copy()
            except:
                h = best_h.copy()
        else:
            h = best_h.copy()
            
        # Coordinate descent step: optimize a subset of variables
        indices = np.random.permutation(n_points)[:max(1, n_points // 5)]
        orig_h = h.copy()
        
        for _ in range(20):  # Inner iterations
            for idx in indices:
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Evaluate gradient for current variable
                grad = 0.0
                for k in range(n_points):
                    j = (idx + k) % n_points
                    grad += (1 - h[j] - h[idx]) * dx
                
                # Update using gradient step
                h[idx] = h[idx] - 0.01 * grad
                h[idx] = max(0, min(1, h[idx]))
                
            # Project to satisfy sum constraint
            h = project_h(h, n_points/2)
            
        # Update best solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Use L-BFGS for final refinement
        if iter % 10 == 0 and time() - start_time < budget_s * 0.9:
            try:
                bounds = [(0, 1) for _ in range(n_points)]
                def obj_func(h_trial):
                    overlaps = np.correlate(h_trial, 1 - h_trial, mode='full') * dx
                    return np.max(overlaps)
                    
                res = minimize(obj_func, h, method='L-BFGS-B', bounds=bounds, 
                              options={'maxiter': 50, 'ftol': 1e-9})
                if res.success:
                    h_refined = res.x
                    h_refined = project_h(h_refined, n_points/2)
                    refined_c5 = evaluate_erdos_solution(h_refined, dx)
                    if refined_c5 < best_c5:
                        best_h = h_refined
                        best_c5 = refined_c5
            except Exception:
                pass
    
    return (list(best_h), best_c5, n_points)
