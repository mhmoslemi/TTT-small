import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    generations = 0
    max_no_improvement = 200
    no_improvement_count = 0
    mutation_rate = 0.1
    crossover_rate = 0.8
    mutation_step = 0.05
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)
    
    def feasible_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def crossover(h1, h2):
        if random.random() > crossover_rate:
            return h1.copy(), h2.copy()
        alpha = random.uniform(0.3, 0.7)
        h_child1 = alpha * h1 + (1 - alpha) * h2
        h_child2 = (1 - alpha) * h1 + alpha * h2
        return feasible_h(h_child1), feasible_h(h_child2)
    
    def mutate(h):
        if random.random() > mutation_rate:
            return h.copy()
        h_mut = h.copy()
        for i in range(n_points):
            if random.random() < 0.1:
                delta = random.gauss(0, mutation_step)
                h_mut[i] += delta
        return feasible_h(h_mut)
    
    while time() - start_time < budget_s * 0.95:
        # Evaluate population
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Track best
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitness.index(min_fitness)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            break
        
        # Tournament selection
        selected = []
        for _ in range(population_size // 2):
            tournament_size = 3
            indices = random.sample(range(population_size), tournament_size)
            tournament_fitness = [fitness[i] for i in indices]
            winner_idx = indices[tournament_fitness.index(min(tournament_fitness))]
            selected.append(population[winner_idx].copy())
        
        # Generate offspring
        offspring = []
        for i in range(0, len(selected), 2):
            if i + 1 < len(selected):
                child1, child2 = crossover(selected[i], selected[i+1])
                offspring.extend([mutate(child1), mutate(child2)])
            else:
                offspring.append(mutate(selected[i]))
        
        # Replace population
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Combine current population and offspring
        all_candidates = population + offspring
        all_fitness = fitness + [evaluate_erdos_solution(h, dx) for h in offspring]
        
        # Keep top half
        combined = list(zip(all_candidates, all_fitness))
        combined.sort(key=lambda x: x[1])
        population = [x[0] for x in combined[:population_size]]
        
        # SLSQP refinement every 100 generations
        if generations % 100 == 0 and generations > 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 1000, 'ftol': 1e-12})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        
        generations += 1
    
    # Final local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
