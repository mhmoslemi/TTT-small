import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h based on available construction or default
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Trust region method parameters
    trust_radius = 0.2
    trust_radius_min = 1e-6
    trust_radius_max = 0.5
    improvement_threshold = 1e-8
    max_trust_iter = 500
    
    for trust_iter in range(max_trust_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Solve convex relaxation with trust region constraint
        try:
            # Create variables
            h_vars = cp.Variable(n_points)
            C = cp.Variable()
            
            # Constraints
            constraints = [
                cp.sum(h_vars) == n_points / 2.0,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Add overlap constraints for most violated lags
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            lag_indices = np.arange(len(corr))
            lags = lag_indices - (len(h_current) - 1)
            overlap_values = corr * dx
            max_indices = np.argsort(overlap_values)[-10:]  # Top 10 violated
            for i in max_indices:
                k = lags[i]
                if k > 0:
                    expr = cp.sum(cp.multiply(h_vars[:-k], 1 - h_vars[k:]))
                elif k < 0:
                    expr = cp.sum(cp.multiply(h_vars[-k:], 1 - h_vars[:-k]))
                else:
                    expr = cp.sum(cp.multiply(h_vars, 1 - h_vars))
                constraints.append(expr <= C * dx)
            
            # Add trust region constraint
            constraints.append(cp.sum_squares(h_vars - h_current) <= trust_radius**2)
            
            # Objective
            objective = cp.Minimize(C)
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == 'optimal':
                h_new = h_vars.value
                c5_new = C.value * dx
                
                # Update trust region based on progress
                if c5_new < best_c5 - improvement_threshold:
                    trust_radius *= 1.2
                    trust_radius = min(trust_radius, trust_radius_max)
                    best_h = h_new.copy()
                    best_c5 = c5_new
                else:
                    trust_radius *= 0.8
                    trust_radius = max(trust_radius, trust_radius_min)
                    
                h_current = h_new.copy()
                
        except Exception:
            # Fallback to simple modification if CVXOPT fails
            h_new = h_current + np.random.normal(0, trust_radius/10, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            h_current = h_new.copy()
            c5_new = evaluate_erdos_solution(h_current, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_current.copy()
    
    # Final refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
