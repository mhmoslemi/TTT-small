import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def low_pass_filter(h, bandwidth=5):
    """Apply a simple moving average filter to smooth the function"""
    n = len(h)
    filtered = np.zeros(n)
    for i in range(n):
        start = max(0, i - bandwidth)
        end = min(n, i + bandwidth + 1)
        filtered[i] = np.mean(h[start:end])
    return filtered

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex optimization parameters
    max_iterations = 1000
    k_top = 10  # Number of top overlaps to consider in convex relaxation
    filter_bandwidth = 5
    patience = 50
    
    # Convex relaxation with top K overlaps
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find top k overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        top_k_indices = np.argsort(corr)[-k_top:][::-1]
        
        # Create a convex problem to minimize these top overlaps
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Objective: minimize the sum of top overlaps (approximate)
        objective_value = 0
        for idx in top_k_indices:
            # Align the correlation properly
            lag = idx - (n_points - 1)
            # Compute the overlap contribution of the lag
            if lag >= 0:
                obj = cp.sum(cp.multiply(h_var[:n_points-lag], 1 - h_var[lag:]))
            else:
                obj = cp.sum(cp.multiply(h_var[-lag:], 1 - h_var[:n_points+lag]))
            objective_value += obj
            
        # Solve the convex program
        objective = cp.Minimize(objective_value)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_new = np.array(h_var.value).flatten()
                h_new = np.clip(h_new, 0, 1)
                # Renormalize
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Apply low-pass filter
                h_filtered = low_pass_filter(h_new, filter_bandwidth)
                h_filtered = np.clip(h_filtered, 0, 1)
                h_filtered = h_filtered * (n_points / 2) / np.sum(h_filtered)
                
                # Evaluate new solution
                c5_new = evaluate_erdos_solution(h_filtered, dx)
                
                # Accept if better
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_filtered.copy()
                    h = h_filtered.copy()
                    patience = 50  # reset patience
                    k_top = min(20, k_top + 1)  # Increase K if improving
                else:
                    patience -= 1
                    if patience <= 0:
                        break
            else:
                break
        except Exception:
            break
    
    # Final refinement
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
