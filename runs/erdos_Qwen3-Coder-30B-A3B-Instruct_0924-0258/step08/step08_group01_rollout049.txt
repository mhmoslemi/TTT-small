import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale optimization with scale doubling
    scale = 0
    max_scales = int(np.log2(n_points)) + 2
    perturbation_magnitude = 0.05
    decay_factor = 0.95
    
    while scale < max_scales and (time() - start_time) < budget_s * 0.95:
        current_n_points = min(2**scale, 512)  # Cap at reasonable size
        if current_n_points <= n_points:
            h_current = best_h.copy()
        else:
            # Upsample from current best
            h_current = np.interp(np.linspace(0, 1, current_n_points), 
                                 np.linspace(0, 1, n_points), 
                                 best_h)
        
        # Perturb and evaluate
        best_local_h = h_current.copy()
        best_local_c5 = evaluate_erdos_solution(h_current, 2.0/current_n_points)
        
        # Perturbation loop
        perturb_iter = 0
        max_perturb = 1000
        while perturb_iter < max_perturb and (time() - start_time) < budget_s * 0.95:
            # Create new perturbed solution
            h_perturbed = h_current.copy()
            for i in range(current_n_points):
                # Random perturbation within bounds
                delta = perturbation_magnitude * (np.random.rand() - 0.5) * 2
                h_perturbed[i] += delta
            
            # Ensure feasibility
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (current_n_points / 2) / np.sum(h_perturbed)
            
            # Evaluate
            c5_perturbed = evaluate_erdos_solution(h_perturbed, 2.0/current_n_points)
            if c5_perturbed < best_local_c5:
                best_local_h = h_perturbed.copy()
                best_local_c5 = c5_perturbed
                perturbation_magnitude = min(perturbation_magnitude * 1.1, 0.5)  # Adaptive increase
            else:
                perturbation_magnitude *= decay_factor  # Adaptive decrease
            
            perturb_iter += 1
        
        # Local refinement with SLSQP
        def objective(h):
            return evaluate_erdos_solution(h, 2.0/current_n_points)
        
        def constraint_sum(h):
            return np.sum(h) - current_n_points / 2.0
        
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(current_n_points)]
            res = minimize(objective, best_local_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 1000, 'ftol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_local_c5:
                    best_local_h = h_refined
                    best_local_c5 = c5_refined
        except Exception:
            pass
        
        # Update global best if better
        if best_local_c5 < best_c5:
            best_c5 = best_local_c5
            best_h = best_local_h.copy()
        
        scale += 1
        
        # Early exit if no improvement
        if abs(best_c5 - evaluate_erdos_solution(best_h, dx)) < 1e-12:
            break
    
    # Final refinement with full optimization on best
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
