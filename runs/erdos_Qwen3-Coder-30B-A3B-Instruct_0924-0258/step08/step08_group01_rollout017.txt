import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    learning_rate = 0.01
    
    # Gradient descent with Adam
    h_current = h_start.copy()
    
    def compute_gradient(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_lag = max_idx - (len(h) - 1)
        
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + max_lag) % len(h)
            grad[i] = 2 * (1 - h[j] - h[i])
        return grad * dx
    
    no_improvement_count = 0
    max_no_improvement = 200
    
    while time() - start_time < budget_s * 0.90:
        t += 1
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            # Reset if no improvement for too long
            h_current = best_h.copy()
            m.fill(0)
            v.fill(0)
            t = 0
            learning_rate *= 0.5
            no_improvement_count = 0
            if learning_rate < 1e-6:
                break
        
        # Compute gradient
        grad = compute_gradient(h_current, dx)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h_new = h_current - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Apply constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_current = h_new
        
        # Occasionally recompute better gradient and update
        if t % 50 == 0:
            c5_new = evaluate_erdos_solution(h_current, dx)
            if c5_new < c5_current:
                learning_rate *= 1.05
            else:
                learning_rate *= 0.5
        
        learning_rate = np.clip(learning_rate, 1e-6, 0.1)
    
    # Final refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
