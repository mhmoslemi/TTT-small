import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h onto feasible set: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to target sum
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-10:
        # Avoid division by zero
        h = np.ones_like(h) * (n_points / 2) / n_points
    else:
        h = h * (n_points/2) / current_sum
    return h

def line_search(h_current, grad, dx, max_iter=20):
    """Perform backtracking line search for step size"""
    alpha = 1.0
    step_size = 1e-2
    c1 = 1e-4
    
    for _ in range(max_iter):
        h_new = h_current - alpha * grad
        h_new = project_h(h_new, len(h_current))
        c5_new = evaluate_erdos_solution(h_new, dx)
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        if c5_new < c5_current:
            return alpha
        alpha *= 0.5
    
    return step_size
