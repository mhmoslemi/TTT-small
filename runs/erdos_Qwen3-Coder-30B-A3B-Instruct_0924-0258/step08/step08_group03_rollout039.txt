import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_fine = np.array(initial_h_values)
        n_points = len(h_fine)
    else:
        n_points = 35
        h_fine = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_fine.copy()
    best_c5 = evaluate_erdos_solution(h_fine, dx)
    
    # Coarse grid optimization (15 points)
    n_coarse = 15
    h_coarse = np.ones(n_coarse) * 0.5
    dx_coarse = 2.0 / n_coarse
    
    # Optimize coarse grid for 500 iterations
    for iteration in range(500):
        if time() - start_time > budget_s * 0.8:
            break
            
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_coarse - 1)
        
        grad = np.zeros(n_coarse)
        for i in range(n_coarse):
            j = i + max_lag
            if 0 <= j < n_coarse:
                grad[i] = 1 - 2 * h_coarse[i] - h_coarse[j]
            else:
                grad[i] = 1 - 2 * h_coarse[i]
        
        h_coarse_new = h_coarse - 0.01 * grad
        h_coarse_new = np.clip(h_coarse_new, 0, 1)
        
        h_coarse_new = h_coarse_new * (n_coarse / 2) / np.sum(h_coarse_new)
        
        c5_new = evaluate_erdos_solution(h_coarse_new, dx_coarse)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_coarse_new.copy()
            h_coarse = h_coarse_new.copy()
        else:
            if iteration % 100 == 0:
                # Add some noise to avoid local minimum
                h_coarse += np.random.normal(0, 0.001, n_coarse)
                h_coarse = np.clip(h_coarse, 0, 1)
    
    # Upsample to fine grid
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_points)
    
    f = interp1d(x_coarse, h_coarse, kind='linear', fill_value='extrapolate')
    h_fine = f(x_fine)
    
    # Renormalize to exactly match the constraint
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)
    
    # Fine grid optimization: gradient descent + random mutations
    for iteration in range(2000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Gradient step
        corr = np.correlate(h_fine, 1 - h_fine, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h_fine[i] - h_fine[j]
            else:
                grad[i] = 1 - 2 * h_fine[i]
        
        h_fine = h_fine - 0.01 * grad
        h_fine = np.clip(h_fine, 0, 1)
        
        # Renormalize
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        # Random mutation
        if iteration % 50 == 0:
            idx = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            h_fine[idx] = np.random.uniform(0, 1, len(idx))
            h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_fine, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_fine.copy()
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (list(best_h), best_c5, n_points)
