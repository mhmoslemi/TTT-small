import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # PSO parameters
    num_particles = 75
    inertia = 0.8
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    perturbation_rate = 0.01
    max_iter = int(budget_s * 0.9 * 1000)  # rough estimate
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    personal_best_positions = []
    personal_best_scores = []
    
    # Initialize particles
    if 'initial_h_values' in globals():
        particles.append(h_current.copy())
        particles.append(h_current.copy())
        particles.append(h_current.copy())
        for _ in range(num_particles - 3):
            h = np.random.rand(n_points) * 0.5 + 0.25
            h = h * (n_points/2) / np.sum(h)
            particles.append(h)
    else:
        for _ in range(num_particles):
            h = np.random.rand(n_points) * 0.5 + 0.25
            h = h * (n_points/2) / np.sum(h)
            particles.append(h)
    
    # Initialize velocities
    for _ in range(num_particles):
        velocities.append(np.zeros(n_points))
    
    # Initialize personal bests
    for i in range(num_particles):
        score = evaluate_erdos_solution(particles[i], dx)
        personal_best_positions.append(particles[i].copy())
        personal_best_scores.append(score)
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO loop
    for iteration in range(max_iter):
        current_time = time()
        if current_time - start_time > budget_s * 0.95:
            break
            
        # Update inertia
        if max_iter > 10000:
            inertia = 0.8 - (0.4 * iteration / max_iter)
        
        # Update each particle
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive_component = cognitive_weight * r1 * (personal_best_positions[i] - particles[i])
            social_component = social_weight * r2 * (global_best_position - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive_component + social_component
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible set
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate fitness
            score = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = particles[i].copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best_position = particles[i].copy()
        
        # Perturbation to escape local minima
        if iteration % 1000 == 0 and iteration > 0:
            for i in range(num_particles):
                if np.random.rand() < perturbation_rate:
                    particles[i] = np.random.rand(n_points) * 0.5 + 0.25
                    particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
    
    # Final refinement using local search on best solution
    refined_h = global_best_position.copy()
    dx = 2.0 / n_points
    best_c5 = global_best_score
    
    # Simple local search with small perturbations
    for _ in range(1000):
        perturbation_magnitude = 0.01
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, perturbation_magnitude)
        
        # Apply perturbation
        h_test = refined_h.copy()
        h_test[idx] += delta
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points/2) / np.sum(h_test)
        
        # Evaluate
        c5_test = evaluate_erdos_solution(h_test, dx)
        
        if c5_test < best_c5:
            best_c5 = c5_test
            refined_h = h_test.copy()
    
    return (list(refined_h), best_c5, n_points)
