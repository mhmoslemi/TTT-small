import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) < 1e-10:
        return h
    
    # Simple projection: rescale and clip
    scale = (n_points/2) / current_sum
    h = h * scale
    h = np.clip(h, 0, 1)
    
    # Final adjustment to ensure sum constraint
    diff = n_points/2 - np.sum(h)
    if abs(diff) > 1e-10:
        # Add small amounts to some values
        idx = np.argsort(h)[:int(abs(diff) / dx)]
        for i in idx:
            if h[i] < 1:
                h[i] += min(dx, 1 - h[i])
                diff -= dx
                if abs(diff) < 1e-10:
                    break
    
    return h

    # Renormalize
    h_current = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Parameters
    max_iter = 10000
    step_size = 0.01
    backtracking_factor = 0.5
    perturbation_freq = 500
    perturbation_magnitude = 0.05
    
    iter_count = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all correlations for all possible lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Get the maximum overlap lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Compute gradient for top k lags to improve diversification
        K = min(5, len(h_current))  # Consider top K lags
        top_lags = np.argsort(corr)[-K:]
        
        gradient = np.zeros(n_points)
        for lag in top_lags:
            lag_val = lag - (len(h_current) - 1)
            if lag_val >= 0:
                for i in range(n_points - lag_val):
                    gradient[i] += 1 - h_current[i + lag_val]
                    if i + lag_val < n_points:
                        gradient[i + lag_val] -= h_current[i]
            else:
                for i in range(n_points + lag_val):
                    gradient[i] += 1 - h_current[i - lag_val]
                    if i - lag_val >= 0:
                        gradient[i - lag_val] -= h_current[i]
        
        # Line search for step size
        step_size_new = step_size
        h_new = h_current - step_size_new * gradient
        h_new = project_to_feasible(h_new, n_points, dx)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Backtracking line search
        while c5_new >= evaluate_erdos_solution(h_current, dx) and step_size_new > 1e-10:
            step_size_new *= backtracking_factor
            h_new = h_current - step_size_new * gradient
            h_new = project_to_feasible(h_new, n_points, dx)
            c5_new = evaluate_erdos_solution(h_new, dx)
            
        if c5_new < best_c5:
