import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.random.rand(n_points)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Threshold-based optimization
    threshold = 0.1
    cooling_rate = 0.95
    max_iterations = 500
    refinement_iterations = 100
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps and find maximum
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        max_overlap = np.max(overlaps)
        k_max = np.argmax(overlaps)
        
        # Compute gradient contribution for each point
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient[i] = -2 * dx * (1 - best_h[j] - best_h[i])
        
        # Sort indices by gradient magnitude
        sorted_indices = np.argsort(np.abs(gradient))[::-1]
        top_indices = sorted_indices[:max(1, n_points // 2)]
        
        # Apply thresholding to top indices
        new_h = best_h.copy()
        for i in top_indices:
            if new_h[i] >= 1 - threshold:
                new_h[i] = 1.0
            elif new_h[i] <= threshold:
                new_h[i] = 0.0
        
        # Adjust to satisfy sum constraint
        current_sum = np.sum(new_h)
        diff = (n_points/2) - current_sum
        if abs(diff) > 1e-10:
            # Distribute difference to most impacted coordinates
            sort_idx = np.argsort(np.abs(gradient))[::-1][:n_points//4]
            for i in sort_idx:
                if abs(diff) < 1e-10:
                    break
                delta = min(abs(diff), 1.0 - new_h[i]) if diff > 0 else max(-new_h[i], diff)
                new_h[i] += delta
                diff -= delta
                
        new_h = np.clip(new_h, 0, 1)
        # Re-adjust to ensure sum is correct
        new_h = new_h * (n_points/2) / np.sum(new_h)
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            
        # Cool threshold
        threshold = max(0.01, threshold * cooling_rate)
        
        if threshold < 0.01:
            break
    
    # Coordinate-wise refinement
    for iter in range(refinement_iterations):
        if time() - start_time > budget_s * 0.99:
            break
            
        old_c5 = evaluate_erdos_solution(best_h, dx)
        for i in range(n_points):
            # Fix all other points and optimize h[i] only
            def obj(h_i):
                temp_h = best_h.copy()
                temp_h[i] = h_i
                return evaluate_erdos_solution(temp_h, dx)
            
            # Golden section search in [0,1]
            a, b = 0.0, 1.0
            resphi = 2.0 - 0.5 * (1.0 + np.sqrt(5.0))
            c = a + resphi * (b - a)
            d = b - resphi * (b - a)
            
            while abs(b - a) > 1e-8:
                if obj(c) < obj(d):
                    b = d
                else:
                    a = c
                    
                # Recompute
                c = a + resphi * (b - a)
                d = b - resphi * (b - a)
                
            # Update h[i]
            best_h[i] = (a + b) / 2.0
                        
        # Re-normalize to sum constraint
        best_h = best_h * (n_points/2) / np.sum(best_h)
        
        new_c5 = evaluate_erdos_solution(best_h, dx)
        if new_c5 >= old_c5:
            break
    
    # Final optimization with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        if res.success:
            final_h = res.x
            final_h = final_h * (n_points/2) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
