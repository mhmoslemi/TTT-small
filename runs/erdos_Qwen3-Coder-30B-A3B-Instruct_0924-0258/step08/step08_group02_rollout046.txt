import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Threshold-based optimization
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    threshold = 0.1
    cooling_factor = 0.95
    max_iterations = 500
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Sort indices by gradient magnitude
        sorted_indices = np.argsort(np.abs(gradient))[::-1]
        top_indices = sorted_indices[:n_points//2]
        
        # Threshold operation
        new_h = h.copy()
        for i in range(n_points):
            if i in top_indices:
                if h[i] >= 1 - threshold:
                    new_h[i] = 1.0
                elif h[i] <= threshold:
                    new_h[i] = 0.0
                else:
                    new_h[i] = h[i]
            else:
                new_h[i] = h[i]
        
        # Re-distribute to maintain sum constraint
        current_sum = np.sum(new_h)
        deficit = (n_points/2) - current_sum
        
        if abs(deficit) > 1e-10:
            # Distribute evenly to most impacted coordinates
            sort_indices = np.argsort(np.abs(gradient))
            count = min(len(sort_indices), 10)
            for i in range(count):
                idx = sort_indices[i]
                if new_h[idx] < 1:
                    add = min(deficit, 1 - new_h[idx])
                    new_h[idx] += add
                    deficit -= add
                    if abs(deficit) < 1e-10:
                        break
            if abs(deficit) > 1e-10:
                for i in range(count):
                    idx = sort_indices[-(i+1)]
                    if new_h[idx] > 0:
                        subtract = min(-deficit, new_h[idx])
                        new_h[idx] -= subtract
                        deficit += subtract
                        if abs(deficit) < 1e-10:
                            break
        
        # Update h
        h = np.clip(new_h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Track best solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Cool threshold
        threshold = max(0.01, threshold * cooling_factor)
        
        # Occasionally do small local optimizations
        if iter % 20 == 0 and threshold < 0.05:
            # Local refinement for a few coordinates
            for i in range(n_points):
                if np.abs(gradient[i]) > 0.001:
                    original = h[i]
                    # Try a few small perturbations
                    best_val = original
                    best_loss = current_c5
                    for pert in [0.01, -0.01, 0.005, -0.005]:
                        h[i] = original + pert
                        h = np.clip(h, 0, 1)
                        h = h * (n_points/2) / np.sum(h)
                        new_c5 = evaluate_erdos_solution(h, dx)
                        if new_c5 < best_loss:
                            best_loss = new_c5
                            best_val = h[i]
                    h[i] = best_val
                    h = h * (n_points/2) / np.sum(h)
                    if best_loss < current_c5:
                        current_c5 = best_loss
                        if current_c5 < best_c5:
                            best_c5 = current_c5
                            best_h = h.copy()
    
    # Final L-BFGS-B refinement
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return np.max(np.correlate(h, 1 - h, mode='full') * dx)
        
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
