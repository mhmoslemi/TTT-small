import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    if np.sum(h_clipped) == 0:
        return h_clipped
    h_scaled = h_clipped * (n_points / 2.0) / np.sum(h_clipped)
    return h_scaled

def compute_gradient(h, n_points, max_lag):
    shifted = np.roll(1 - h, max_lag)
    gradient = 2 * (1 - 2 * h) * shifted
    return gradient

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_iter = 1000
    patience = 50
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = project(h_start, n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    step_size = 0.1
    no_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        max_overlap = corr[max_lag_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            step_size *= 0.5
            no_improvement = 0
            if step_size < 1e-8:
                break
        
        # Compute gradient for the max lag
        grad = compute_gradient(h, n_points, max_lag)
        
        # Update h using gradient descent
        h_new = h - step_size * grad
        h_new = project(h_new, n_points)
        
        # Check if the update improved the objective
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < max_overlap:
            h = h_new
        else:
            step_size *= 0.5
            if step_size < 1e-8:
                break
            
    # Fallback refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        h_init = best_h.copy()
        res = minimize(objective, h_init, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    if 'initial_h_values' not in globals():
        # Try a different initialization pattern
        try:
            h_alternative = np.zeros(n_points)
            for i in range(n_points):
                h_alternative[i] = 0.5 if i % 2 == 0 else 0.0
            h_alternative = project(h_alternative, n_points)
            c5_alt = evaluate_erdos_solution(h_alternative, dx)
            if c5_alt < best_c5:
                best_h = h_alternative
                best_c5 = c5_alt
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
