import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Coarse grid genetic algorithm (n=15) for global structure
    coarse_n = 15
    coarse_dx = 2.0 / coarse_n
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.5
    
    # Initialize coarse population
    coarse_pop = []
    for _ in range(pop_size):
        h_coarse = np.random.rand(coarse_n)
        h_coarse = h_coarse * (coarse_n/2) / np.sum(h_coarse)
        coarse_pop.append(h_coarse)
    
    # Coarse optimization for 60 seconds
    coarse_time = 60
    coarse_start = time()
    while time() - coarse_start < coarse_time and time() - start_time < budget_s * 0.8:
        # Evaluate all solutions
        fitness = []
        for h_coarse in coarse_pop:
            val = evaluate_erdos_solution(h_coarse, coarse_dx)
            fitness.append(-val)
        
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            candidates = np.random.choice(pop_size, 3, replace=False)
            best_idx = candidates[np.argmax([fitness[i] for i in candidates])]
            selected.append(coarse_pop[best_idx].copy())
        
        # Crossover and mutation
        new_pop = []
        for i in range(0, pop_size, 2):
            p1 = selected[i]
            p2 = selected[min(i+1, pop_size-1)]
            
            if np.random.rand() < crossover_rate:
                # Blend crossover
                alpha = np.random.rand()
                child1 = alpha * p1 + (1 - alpha) * p2
                child2 = (1 - alpha) * p1 + alpha * p2
            else:
                child1, child2 = p1.copy(), p2.copy()
            
            # Mutation
            for j in range(coarse_n):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.05)
            
            # Clamp and rescale
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (coarse_n/2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (coarse_n/2) / np.sum(child2)
            
            new_pop.extend([child1, child2])
        
        coarse_pop = new_pop[:pop_size]
    
    # Find best coarse solution
    coarse_fitness = []
    for h_coarse in coarse_pop:
        val = evaluate_erdos_solution(h_coarse, coarse_dx)
        coarse_fitness.append(-val)
    
    best_coarse_idx = np.argmax(coarse_fitness)
    best_coarse = coarse_pop[best_coarse_idx]
    
    # Upscale to fine grid (n=35)
    h_fine = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, coarse_n), 
                       best_coarse)
    h_fine = h_fine * (n_points/2) / np.sum(h_fine)
    
    h_current = h_fine
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Gradient optimization on fine grid
    step_size = 0.01
    max_iter = 10000
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all correlations for all possible lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Get the maximum overlap lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Compute gradient for max overlap at max_lag
        # For h[i] with contribution to overlap at lag max_lag, 
        # the contribution is h[i] * (1 - h[i + max_lag])  
        # So d/dh[i] = 1 - h[i + max_lag] 
        # and d/dh[i + max_lag] = -h[i]
        
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_current[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= h_current[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_current[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_current[i]
        
        # Let's also consider second-largest overlaps
        # Make multiple gradients with top couple of lags
        corr_sorted = np.argsort(corr)[::-1][:3]  # Top 3
        for lag_idx in corr_sorted:
            lag = lag_idx - (len(h_current) - 1)
            if lag == max_lag:
                continue  # Already addressed
            grad_temp = np.zeros(n_points)
            if lag >= 0:
                for i in range(n_points - lag):
                    grad_temp[i] += 1 - h_current[i + lag]
                    if i + lag < n_points:
                        grad_temp[i + lag] -= h_current[i]
            else:
                for i in range(n_points + lag):
                    grad_temp[i] += 1 - h_current[i - lag]
                    if i - lag >= 0:
                        grad_temp[i - lag] -= h_current[i]
            # Weight the gradient contribution from the lag
            # Higher weight if the lag is significantly large
            weight = corr[lag_idx] / corr[max_lag_idx]  # relative size of this correlation
            gradient += 0.2 * weight * grad_temp  # Reduce effect of secondary lags
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm
        
        # Gradient descent step
        h_new = h_current - step_size * gradient
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # If not improving, reduce step size
        if c5_new >= evaluate_erdos_solution(h_current, dx):
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            h_current = h_new
            step_size *= 1.05  # Increase step size slightly
        
        iter_count += 1
    
    return (list(best_h), best_c5, n_points)
