import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    pop_size = 100
    max_generations = 500
    mutation_rate = 0.05
    elite_ratio = 0.2
    crossover_rate = 0.7
    
    # Initialize population with binary representations
    def to_binary(h):
        # Map h values to binary intervals (0 or 1)
        return (h > 0.5).astype(int)
    
    def from_binary(b):
        # Convert binary to continuous h with sum constraint
        h = b.astype(float)
        h = h * (n_points/2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    # Create initial population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals() and len(population) == 0:
            binary = to_binary(h_start)
        else:
            binary = np.random.randint(0, 2, n_points)
        h_candidate = from_binary(binary)
        population.append(h_candidate)
    
    # GA main loop
    best_fitness_history = []
    patience = 50
    no_improvement = 0
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness (C5)
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            break
            
        # Selection (keep elites)
        elite_count = int(pop_size * elite_ratio)
        elites = population[:elite_count]
        
        # Crossover and mutation
        new_population = elites[:]
        
        while len(new_population) < pop_size:
            if random.random() < crossover_rate:
                parent1, parent2 = random.sample(elites, 2)
                crossover_point = random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                parent = random.choice(elites)
                child1 = parent.copy()
                child2 = parent.copy()
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child1[i] = 1 - child1[i]
                if random.random() < mutation_rate:
                    child2[i] = 1 - child2[i]
            
            # Ensure constraints
            child1 = from_binary(child1)
            child2 = from_binary(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:pop_size]
    
    # Post-processing: local refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Refine top candidates
        for i in range(10):  # Try refining top 10
            if time() - start_time > budget_s * 0.95:
                break
            h_candidate = population[i]
            res = minimize(objective, h_candidate, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
