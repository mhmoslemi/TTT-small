import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a structured binary initial solution
        n_points = 35
        h_start = np.zeros(n_points)
        # Place 1's in a binary-like pattern to avoid large overlaps
        for i in range(n_points):
            if i % 3 == 0:
                h_start[i] = 1.0
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Define the structure of the problem
    # Use a convex relaxation approach with McCormick envelopes
    
    # We'll try an alternating optimization approach:
    # 1. High-level relaxation for global bounds
    # 2. Local refinement using SLSQP for tightness
    
    # Try CVXOPT relaxation first
    try:
        # Set up CVX problem
        # Variables
        h = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraint: sum(h) = n_points / 2
        sum_constraint = cp.Equality(cp.sum(h), n_points / 2)
        
        # Overlap constraints
        constraints = [sum_constraint]
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        
        # For each lag, add the overlap constraint using a convex approximation
        # We approximate: h[i](1 - h[i + k]) with z[i][i + k] = h[i]*h[i+k]
        for k in range(-n_points + 1, n_points):
            if k >= 0:
                # offset = k
                # Exclude the last |k| points
                indices = range(n_points - k)
                lag_sum = cp.sum([h[i] - h[i] * h[i + k] for i in indices])
            else:
                # offset = |k|, should be negative
                indices = range(n_points + k)
                lag_sum = cp.sum([h[i] - h[i] * h[i - k] for i in indices])
            
            # Use McCormick envelopes:
            # Let z[i][i+k] = h[i] * h[i+k]
            # h[i] + h[i+k] - 1 <= z[i][i+k] <= h[i]
            # z[i][i+k] <= h[i+k] 
            # z[i][i+k] >= 0
            
            # For now, add simpler linear constraints
            constraints.append(lag_sum <= C)
        
        # Objective
        objective = cp.Minimize(C)
        
        if time() - start_time > budget_s * 0.7:  # Early stopping
            return (list(best_h), best_c5, n_points)
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=1000)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_relaxed = h.value
            h_projected = np.clip(h_relaxed, 0, 1)
            # Re-normalize
            h_projected = h_projected * (n_points/2) / np.sum(h_projected)
            
            c5_relaxed = evaluate_erdos_solution(h_projected, dx)
            if c5_relaxed < best_c5:
                best_h = h_projected.copy()
                best_c5 = c5_relaxed
                
    except Exception as e:
        # Fallback to SLSQP
        pass
        
    # Now perform local refinement using SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Refine with SLSQP from current best
    try:
        # If the number of times we've refined exceeds a limit, just optimize with small steps
        # We can reuse the best_h from the relaxation if we had a good one
        if time() - start_time > budget_s * 0.95:
            return (list(best_h), best_c5, n_points)
            
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12})
        
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            # Normalize
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        # Ignore errors in refinement
        pass
    
    # Final evaluation with fine-tuning
    try:
        # Perform one more round of optimization using different optimization approaches
        # Try to start with a more refined version if we're not in the budget exhaustion zone
        if time() - start_time > budget_s * 0.9:
            return (list(best_h), best_c5, n_points)
            
        # Try to make a few more small local improvements 
        # Use a tighter "lsq" approach with less tolerance
        res = minimize(objective, best_h, method='L-BFGS-B', 
                       bounds=bounds, 
                       options={'maxiter': 500, 'ftol': 1e-15,
                               'gtol': 1e-15, 'eps': 1e-8})
        
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            # Re-normalize
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
                
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
