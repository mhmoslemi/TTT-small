import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Define coarse grid for global search (15 points)
    n_coarse = 15
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.ones(n_coarse) * 0.5
    
    # Population for genetic search
    population_size = 50
    population = []
    population_fitness = []
    
    # Initialize coarse population with random feasible solutions
    for _ in range(population_size):
        h_test = np.random.rand(n_coarse)
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_coarse/2) / np.sum(h_test)
        population.append(h_test)
        c5_val = evaluate_erdos_solution(h_test, dx_coarse)
        population_fitness.append(-c5_val)  # Negative for minimization
    
    # Coarse optimization loop
    coarse_budget = 60
    coarse_start_time = time()
    coarse_iter = 0
    while time() - coarse_start_time < coarse_budget and coarse_iter < 1000:
        # Sort by fitness
        sorted_indices = np.argsort(population_fitness)
        top_indices = sorted_indices[:population_size//2]
        
        # Keep top performers
        new_population = [population[i] for i in top_indices]
        
        # Generate offspring through crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(top_indices, 2, replace=False)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Crossover
            if np.random.rand() < 0.5:
                crossover_point = np.random.randint(1, n_coarse)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            # Apply Gaussian perturbation
            for i in range(len(child1)):
                if np.random.rand() < 0.2:
                    child1[i] += np.random.normal(0, 0.02)
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_coarse/2) / np.sum(child1)
            
            for i in range(len(child2)):
                if np.random.rand() < 0.2:
                    child2[i] += np.random.normal(0, 0.02)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_coarse/2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
        
        # Evaluate all
        population_fitness = []
        for h in population:
            c5_val = evaluate_erdos_solution(h, dx_coarse)
            population_fitness.append(-c5_val)
        
        coarse_iter += 1
    
    # Select best from coarse
    best_coarse_idx = np.argmax(population_fitness)
    best_coarse = population[best_coarse_idx]
    
    # Upscale to fine grid (35 points)
    h_fine = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, n_coarse), best_coarse)
    
    # Normalize to correct sum
    h_fine = h_fine * (n_points/2) / np.sum(h_fine)
    
    h_current = h_fine.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Fine-grained gradient-based optimization
    fine_budget = budget_s - coarse_budget - 10
    fine_start_time = time()
    step_size = 0.01
    iter_count = 0
    max_iter = 10000
    no_improvement_count = 0
    
    while time() - fine_start_time < fine_budget and iter_count < max_iter:
        # Compute overlaps and get lags with highest values
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # We'll target the top 3 lags to reduce
        top_lag_indices = np.argpartition(corr, -3)[-3:]
        
        # Compute gradient for all these lags
        gradient = np.zeros(n_points)
        for lag_idx in top_lag_indices:
            lag = lag_idx - (len(h_current) - 1)
            if lag >= 0:
                for i in range(n_points - lag):
                    gradient[i] += 1 - h_current[i + lag]
                    if i + lag < n_points:
                        gradient[i + lag] -= h_current[i]
            else:
                for i in range(n_points + lag):
                    gradient[i] += 1 - h_current[i - lag]
                    if i - lag >= 0:
                        gradient[i - lag] -= h_current[i]
        
        # Gradient descent step
        h_new = h_current - step_size * gradient
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # If not improving, reduce step size
        if c5_new >= evaluate_erdos_solution(h_current, dx):
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            h_current = h_new
            step_size = min(step_size * 1.05, 0.1)  # Increase step size rather than too aggressive
            
        iter_count += 1
        
        # Early termination if no improvement for a while
        if no_improvement_count > 1000:
            break
    
    return (list(best_h), best_c5, n_points)
