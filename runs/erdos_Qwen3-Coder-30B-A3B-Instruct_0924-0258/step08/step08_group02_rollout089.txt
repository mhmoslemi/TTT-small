import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Threshold-based optimization
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Phase 2: Sparse optimization with gradient updates
    max_iterations = 500
    threshold_min = 0.01
    threshold_max = 1.0
    threshold_step = 0.1
    
    thresholds = np.arange(threshold_max, threshold_min, -threshold_step)
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for the maximum overlap lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Apply threshold-based updates
        gradient_abs = np.abs(gradient)
        threshold = np.percentile(gradient_abs, 90) if np.any(gradient_abs > 0) else 1e-6
        
        # Update h elements based on gradient
        for i in range(n_points):
            if gradient_abs[i] >= threshold:
                if gradient[i] < 0:
                    h[i] = min(1.0, h[i] - 0.02 * np.sign(gradient[i]))
                else:
                    h[i] = max(0.0, h[i] - 0.02 * np.sign(gradient[i]))
        
        # Enforce constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Track best solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Phase 3: Refinement with L-BFGS-B
    try:
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)
        
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
