import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points/2) / current_sum
    return h

def line_search(h, grad, step_size, n_points, dx):
    """Simple backtracking line search"""
    c1 = 1e-4
    alpha = step_size
    while alpha > 1e-10:
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, n_points, dx)
        new_val = evaluate_erdos_solution(h_new, dx)
        if new_val < evaluate_erdos_solution(h, dx) - c1 * alpha * np.dot(grad, grad):
            return h_new, alpha
        alpha *= 0.5
    return h, step_size

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        n_points = len(h_start)

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Parameters
    step_size = 0.01
    max_iter = 10000
    iter_count = 0
    no_improvement_count = 0
    perturbation_rate = 0.1
    perturbation_size = 0.05
    k_lags = 5  # Number of top lags to consider
    
    # Main optimization loop
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all correlations
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Get indices and values of top k lags by overlap
        top_k_indices = np.argsort(corr)[-k_lags:][::-1]
        # Create gradient from these top k lags
        gradient = np.zeros(n_points)
        for lag in top_k_indices:
            lag_idx = lag - (n_points - 1)
            # Handle lags
            if lag_idx >= 0:
                # Positive lag
                for i in range(n_points - lag_idx):
                    gradient[i] += 1 - h_current[i + lag_idx]
                    if i + lag_idx < n_points:
                        gradient[i + lag_idx] -= h_current[i]
            else:
                # Negative lag
                for i in range(n_points + lag_idx):
                    gradient[i] += 1 - h_current[i - lag_idx]
                    if i - lag_idx >= 0:
                        gradient[i - lag_idx] -= h_current[i]
        
        # Do line search
        h_new, actual_step = line_search(h_current, gradient, step_size, n_points, dx)
        
        # Project after update
        h_new = project_to_feasible(h_new, n_points, dx)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement_count = 0
            step_size *= 1.1  # Increase step size slightly
        else:
            no_improvement_count += 1
            step_size *= 0.5  # Reduce step size
            
            # Perturb in case stuck
            if no_improvement_count % 500 == 0:
                perturb_mask = np.random.rand(n_points) < perturbation_rate
                perturbation = (np.random.rand(n_points) - 0.5) * perturbation_size
                h_current[perturb_mask] += perturbation[perturb_mask]
                h_current = project_to_feasible(h_current, n_points, dx)

        iter_count += 1
        
        # Early exit if near target
        if best_c5 <= 0.3807:

    return (list(best_h), best_c5, n_points)
