import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Heuristic: try different starting points
    candidates = []
    for _ in range(5):
        h = h_start.copy()
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        candidates.append(h)
    
    # Local optimization with various methods
    methods = ['Nelder-Mead', 'Powell', 'CG', 'BFGS', 'L-BFGS-B']
    
    for h in candidates:
        for method in methods:
            if time() - start_time > budget_s * 0.95:
                break
                
            def objective(h_vec):
                h_rescaled = h_vec * (n_points/2) / np.sum(h_vec)
                return evaluate_erdos_solution(h_rescaled, dx)
            
            def constraint_sum(h_vec):
                return np.sum(h_vec) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h, method=method,
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 100, 'ftol': 1e-9})
                if res.success:
                    h_final = res.x
                    h_final = h_final * (n_points/2) / np.sum(h_final)
                    c5_final = objective(h_final)
                    if c5_final < best_c5:
                        best_c5 = c5_final
                        best_h = h_final.copy()
            except Exception:
                continue
    
    # Try gradient-free optimization: Nelder-Mead with adaptive steps
    for _ in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_new = best_h + np.random.normal(0, 0.02, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Final refinement
    try:
        def objective(h_vec):
            h_rescaled = h_vec * (n_points/2) / np.sum(h_vec)
            return evaluate_erdos_solution(h_rescaled, dx)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='L-BFGS-B',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 300, 'ftol': 1e-10, 'gtol': 1e-10})
        
        if res.success:
            h_final = res.x
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
