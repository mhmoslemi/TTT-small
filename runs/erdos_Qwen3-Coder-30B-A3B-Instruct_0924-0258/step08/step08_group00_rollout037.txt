import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm with local refinement
    pop_size = 100
    generations = budget_s // 3
    patience = 50
    no_improvement = 0
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # If initial construction available, add it to population
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = h_start * (n_points/2) / np.sum(h_start)
        population.append(h_start)
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]
        fitness = [fitness[i] for i in sorted_idx]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            break
        
        # Selection (keep top 30%)
        elites = population[:pop_size//3]
        
        # Create new population
        new_population = elites.copy()
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitness, 3)
            parent2 = tournament_selection(population, fitness, 3)
            
            # Crossover
            if random.random() < 0.7:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            child = mutate(child, 0.05, dx)
            
            # Ensure sum constraint
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:pop_size]
        
        # Local refinement of best individual
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, population[0], method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)

def tournament_selection(population, fitness, k):
    idxs = np.random.choice(len(population), size=k)
    best = np.argmin([fitness[i] for i in idxs])
    return population[idxs[best]].copy()

def crossover(parent1, parent2):
    point = np.random.randint(1, len(parent1)-1)
    child = np.concatenate([parent1[:point], parent2[point:]])
    return child

def mutate(h, mutation_rate, dx):
    for i in range(len(h)):
        if random.random() < mutation_rate:
            # Add small random perturbation
            h[i] += np.random.normal(0, 0.05)
    # Ensure constraints
    h = np.clip(h, 0, 1)
    return h
