import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible set: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Simple scaling approach
    if current_sum > 0:
        h = h * target_sum / current_sum
        # Re-clip to ensure bounds
        h = np.clip(h, 0, 1)
        
        # Final adjustment to exactly meet sum (if needed due to clipping)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            # Find largest values and scale down proportionally
            # Or if they are all at bound, do nothing
            # Simply clamp to avoid violating bounds further
            pass
            
    return h

def compute_gradients(h, k_lags, n_points):
    """Compute gradient for top k_lags"""
    gradient = np.zeros(n_points)
    
    # For each lag k in top k_lags
    for k in k_lags:
        if k >= 0:
            # h(i) * (1 - h(i + k))
            # grad wrt h(i) = 1 - h(i + k)
            # grad wrt h(i + k) = -h(i)
            for i in range(n_points - k):
                gradient[i] += 1 - h[i + k]
                gradient[i + k] -= h[i]
        else:
            # k is negative
            # h(i) * (1 - h(i + k)) where i+k >= 0
            for i in range(-k, n_points):
                gradient[i] += 1 - h[i + k]
                gradient[i + k] -= h[i]
                
    return gradient

def line_search(h, grad, n_points, dx, max_step_size=0.1):
    """Simple backtracking line search"""
    # Try different step sizes: 1.0, 0.5, 0.25, 0.125, etc.
    step_size = max_step_size
    for _ in range(10):  # maximum 10 steps
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < evaluate_erdos_solution(h, dx) - 1e-10:
            return h_new, step_size
        step_size *= 0.5
        
    return h, max_step_size * 0.5

    # Parameters
    K = 5  # Top K lags to consider
    step_size = 0.05
    max_iter = 100000
    iter_count = 0
    no_improvement_count = 0
    perturbation_frequency = 500

        # Compute all correlations for all possible lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')

        lags = np.arange(-len(h_current) + 1, len(h_current))
        # rank lags by overlap value
        lag_values = [(lags[i], corr[i]) for i in range(len(corr)) if i != len(corr) // 2]
