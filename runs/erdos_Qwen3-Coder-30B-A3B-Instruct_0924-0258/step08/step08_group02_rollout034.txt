import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter = 100000
    epsilon = 0.1
    
    # Normalize to meet sum constraint
    h = h_start * (n_points / 2) / np.sum(h_start)
    current_c5 = evaluate_erdos_solution(h, dx)
    
    T = T_init
    iter_count = 0
    
    while T > T_min and (time() - start_time) < budget_s * 0.9:
        if iter_count >= max_iter:
            break
            
        # Random perturbation
        i, j = random.randint(0, n_points-1), random.randint(0, n_points-1)
        delta = np.random.uniform(-epsilon, epsilon)
        delta_i = delta * 0.5
        delta_j = -delta * 0.5
        
        # Ensure we don't violate bounds
        if h[i] + delta_i < 0 or h[i] + delta_i > 1:
            delta_i = -delta_i
        if h[j] + delta_j < 0 or h[j] + delta_j > 1:
            delta_j = -delta_j
            
        h_new = h.copy()
        h_new[i] += delta_i
        h_new[j] += delta_j
        
        # Maintain the sum constraint
        sum_diff = np.sum(h_new) - n_points / 2
        h_new[0] -= sum_diff / 2
        h_new[1] -= sum_diff / 2
        
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject the new solution
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            h = h_new
            current_c5 = new_c5
            
            # Update best if better
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        T *= alpha
        iter_count += 1
    
    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def obj_and_grad(h):
            corr = np.correlate(h, 1 - h, mode='full') * dx
            max_overlap = np.max(corr)
            l_max = np.argmax(corr)
            
            # Approximate gradient
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = (i + l_max) % n_points
                grad[i] = -2 * dx * (1 - h[j] - h[i])
            return max_overlap, grad
        
        # Refine with L-BFGS-B
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        refined_h = res.x
        refined_h = refined_h * (n_points/2) / np.sum(refined_h)
        refined_c5 = evaluate_erdos_solution(refined_h, dx)
        
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

    except Exception:
        pass
        
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
