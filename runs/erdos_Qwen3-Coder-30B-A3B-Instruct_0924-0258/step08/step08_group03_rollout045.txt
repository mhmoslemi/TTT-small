import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    """Project h to satisfy integral constraint and bounds"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex perturbation parameters
    epsilon_0 = 0.001
    beta = 0.99
    max_iter = 5000
    top_k = 100
    
    def convex_solve(h_current, epsilon):
        # Create variables
        h_vars = cp.Variable(n_points)
        
        # Objective: minimize the sum over top-k correlations + perturbation
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        top_indices = np.argsort(corr)[-top_k:]
        
        objective = 0
        for k in top_indices:
            # Compute the correlation coefficient
            h_shifted = np.roll(1 - h_current, k - (n_points - 1))
            if k <= (n_points - 1):
                # Positive shift
                h_left = h_current[:n_points - k]
                h_right = h_shifted[k:]
            else:
                # Negative shift
                h_left = h_current[k - (n_points - 1):]
                h_right = h_shifted[:n_points - k]
            objective += cp.sum(cp.multiply(h_vars[:len(h_left)], h_right))
        
        # Add regularization term
        objective += epsilon * cp.sum_squares(h_vars)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Solve
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            return np.array(h_vars.value).reshape(-1)
        except:
            # If CVXPY fails, fall back to basic projection
            return project_to_feasible(h_current, n_points, dx)
    
    epsilon = epsilon_0
    t = 0
    
    while t < max_iter and (time() - start_time) < budget_s * 0.95:
        # Get the current best overlap
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        
        # Generate new solution via convex optimization with perturbation
        h_new = convex_solve(h, epsilon)
        
        # Project to feasible region
        h_new = project_to_feasible(h_new, n_points, dx)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update best
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Update current solution
        h = h_new
        
        # Update parameters
        epsilon *= beta
        t += 1
        
        # Early stopping if not improving for a while
        if t % 500 == 0 and c5_new >= best_c5:
            # Random restart
            h = np.random.rand(n_points) * 0.5 + 0.25
            h = project_to_feasible(h, n_points, dx)
    
    # Final refinement - try local perturbations
    final_h = best_h.copy()
    final_c5 = best_c5
    for _ in range(100):
        if (time() - start_time) > budget_s * 0.95:
            break
        # Generate random small perturbations
        pert = np.random.normal(0, 0.001, n_points)
        h_trial = final_h + pert
        h_trial = project_to_feasible(h_trial, n_points, dx)
        c5_trial = evaluate_erdos_solution(h_trial, dx)
        if c5_trial < final_c5:
            final_c5 = c5_trial
            final_h = h_trial
    
    return (list(final_h), final_c5, n_points)
