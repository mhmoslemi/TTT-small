import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    pop_size = 50
    generations = 1000
    cross_rate = 0.8
    mut_rate = 0.1
    mut_step = 0.05
    elite_size = 10
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # Normalize sum
        h = np.clip(h, 0, 1)
        return h
    
    if 'initial_h_values' in globals():
        population = [h_start.copy()]
        for _ in range(pop_size - 1):
            population.append(create_individual())
    else:
        population = [create_individual() for _ in range(pop_size)]
        
    def fitness(individual):
        return evaluate_erdos_solution(individual, dx)
    
    # GA loop
    gen_no_improvement = 0
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness_scores = [fitness(ind) for ind in population]
        sorted_indices = np.argsort(fitness_scores)
        best_in_gen = fitness_scores[sorted_indices[0]]
        
        if best_in_gen < best_c5:
            best_c5 = best_in_gen
            best_h = population[sorted_indices[0]].copy()
            gen_no_improvement = 0
        else:
            gen_no_improvement += 1
            
        if gen_no_improvement > 200:
            break
            
        # Tournament selection
        new_population = []
        
        # Elitism
        elite_indices = sorted_indices[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
            
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = population[random.choice(sorted_indices[:pop_size//2])]
            parent2 = population[random.choice(sorted_indices[:pop_size//2])]
            
            # Crossover
            if random.random() < cross_rate:
                alpha = random.uniform(0.3, 0.7)
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
                
            # Mutation
            if random.random() < mut_rate:
                perturb = np.random.normal(0, mut_step, n_points)
                child += perturb
                # Restore feasibility
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                
            new_population.append(child)
            
        population = new_population
        mut_step *= 0.98  # Decrease mutation rate
        
        # Periodic SLSQP refinement
        if gen % 100 == 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
                
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 200, 'ftol': 1e-12})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    return (list(best_h), best_c5, n_points)
