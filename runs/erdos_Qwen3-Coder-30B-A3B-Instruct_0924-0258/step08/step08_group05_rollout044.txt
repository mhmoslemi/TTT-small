import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    population_size = 50
    n_points = 35
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Create initial population with diversity
    population = []
    population.append(h_start.copy())
    
    # Fill with diverse initial solutions
    for i in range(population_size - 1):
        if i % 3 == 0:
            # Create a step-like function
            h = np.zeros(n_points)
            h[:n_points//2] = 1.0
            h = h * (n_points/2) / np.sum(h)
        elif i % 3 == 1:
            # Create a random feasible solution
            h = np.random.random(n_points)
            h = h * (n_points/2) / np.sum(h)
        else:
            # Create another diverse solution
            h = np.ones(n_points) * 0.5
            h[:n_points//3] = 0.0
            h[n_points//3:2*n_points//3] = 1.0
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fitness.append(evaluate_erdos_solution(h, dx))
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # GA parameters
    max_generations = 10000
    tournament_size = 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    no_improvement_limit = 100
    no_improvement_count = 0
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create new population through crossover and mutation
        new_population = [best_h.copy()]  # Elitism: keep best
        
        while len(new_population) < population_size:
            # Crossover
            if random.random() < crossover_rate:
                parent1_idx, parent2_idx = np.random.choice(population_size, 2, replace=False)
                parent1 = selected[parent1_idx]
                parent2 = selected[parent2_idx]
                
                # Uniform crossover
                child = np.where(np.random.random(n_points) > 0.5, parent1, parent2)
                
                # Adjust child to satisfy constraint
                child_sum = np.sum(child)
                if abs(child_sum - n_points/2) > 1e-10:
                    child = child * (n_points/2) / child_sum
            else:
                # Selection for mutation (pick one randomly)
                parent_idx = np.random.choice(population_size)
                child = selected[parent_idx].copy()
            
            # Mutation
            if random.random() < mutation_rate:
                # Flip mutation on two random indices
                idx1, idx2 = np.random.choice(n_points, 2, replace=False)
                child[idx1] = max(0, min(1, child[idx1] + np.random.choice([-0.1, 0.1])))
                child[idx2] = max(0, min(1, child[idx2] + np.random.choice([-0.1, 0.1])))
                
                # Ensure sum constraint
                child_sum = np.sum(child)
                child = child * (n_points/2) / child_sum
            
            # Ensure bounds
            child = np.clip(child, 0, 1)
            
            new_population.append(child)
        
        # Evaluate new population
        population = new_population
        fitness = []
        for h in population:
            fitness.append(evaluate_erdos_solution(h, dx))
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= no_improvement_limit:
            break
    
    # Final refinement using a local search
    try:
        from scipy.optimize import minimize
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            if evaluate_erdos_solution(h_final, dx) < best_c5:
                best_h = h_final
                best_c5 = evaluate_erdos_solution(h_final, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
