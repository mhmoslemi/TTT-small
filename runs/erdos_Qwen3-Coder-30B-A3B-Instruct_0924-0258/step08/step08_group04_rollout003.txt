import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Genetic algorithm parameters
    population_size = 50
    num_generations = 200
    mutation_rate = 0.2
    crossover_rate = 0.5
    
    # Track best solution found
    best_solution = best_h.copy()
    best_value = best_c5
    
    # Coarse grid for initial search (n=15)
    n_coarse = 15
    dx_coarse = 2.0 / n_coarse
    population = []
    
    # Initialize population with feasible solutions
    for _ in range(population_size):
        h = np.random.rand(n_coarse)
        h = h * (n_coarse/2) / np.sum(h)
        population.append(h.copy())
        
    # Genetic algorithm loop
    for gen in range(num_generations):
        if time() - start_time > budget_s * 0.7:
            break
            
        # Evaluate all solutions
        fitness_scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx_coarse)
            fitness_scores.append(-c5)  # Negative for maximization
            
        # Selection using tournament
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament = random.sample(list(zip(population, fitness_scores)), 
                                     tournament_size)
            winner = max(tournament, key=lambda x: x[1])
            selected.append(winner[0].copy())
            
        # Create new population through crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            if random.random() < crossover_rate and i+1 < population_size:
                parent1 = selected[i]
                parent2 = selected[i+1]
                
                # Blend crossover
                alpha = random.random()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
                
                # Rescale to satisfy constraint
                child1 = child1 * (n_coarse/2) / np.sum(child1)
                child2 = child2 * (n_coarse/2) / np.sum(child2)
            else:
                child1 = selected[i].copy()
                child2 = selected[i+1].copy()
                
            # Mutation
            if random.random() < mutation_rate:
                mut_idx = random.randint(0, n_coarse-1)
                child1[mut_idx] += np.random.normal(0, 0.1)
                child1[mut_idx] = max(0, min(1, child1[mut_idx]))
                child1 = child1 * (n_coarse/2) / np.sum(child1)
                
            if random.random() < mutation_rate:
                mut_idx = random.randint(0, n_coarse-1)
                child2[mut_idx] += np.random.normal(0, 0.1)
                child2[mut_idx] = max(0, min(1, child2[mut_idx]))
                child2 = child2 * (n_coarse/2) / np.sum(child2)
                
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        
        # Check for improvement in coarse grid
        for h in population:
            c5 = evaluate_erdos_solution(h, dx_coarse)
            if c5 < best_value:
                best_value = c5
                best_solution = h.copy()
                
    # Upscale and refine best solution
    if time() - start_time > budget_s * 0.9:
        return (list(best_solution), best_value, n_coarse)
        
    # Upscale coarse solution to fine grid (n=35)
    h_fine = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, n_coarse), 
                       best_solution)
    h_fine = h_fine * (n_points/2) / np.sum(h_fine)
    
    # Refine with gradient-like optimization
    step_size = 0.01
    max_iter = 10000
    iter_count = 0
    
    while time() - start_time < budget_s * 0.99 and iter_count < max_iter:
        corr = np.correlate(h_fine, 1 - h_fine, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_fine) - 1)
        
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_fine[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= h_fine[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_fine[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_fine[i]
        
        h_new = h_fine - step_size * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        if c5_new >= evaluate_erdos_solution(h_fine, dx):
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            h_fine = h_new
            step_size *= 1.05
            
        iter_count += 1
    
    # Always return the best solution found
    return (list(best_h), best_c5, n_points)
