import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy constraints"""
    h = np.clip(h, 0, 1)
    if np.sum(h) == 0:
        h = np.ones(n_points) * 0.5
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-start approach with Latin hypercube sampling
    n_starts = 50
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Generate diverse starting points
    starts = []
    # Starting point 1: initial_h_values or uniform
    starts.append(h_start.copy())
    
    # Latin hypercube sampling for remaining points
    for _ in range(n_starts - 1):
        # Generate Latin hypercube samples
        h_sample = np.random.rand(n_points)
        starts.append(h_sample)
    
    # Optimize each starting point
    for i, h0 in enumerate(starts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Ensure sum constraint
        h0 = repair_h(h0, n_points)
        
        # Define objective
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        # Define constraints
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h0, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                h_final = repair_h(h_final, n_points)  # Ensure constraints
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
