import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach with simulated annealing
    resolutions = [35, 70, 140, 280]
    current_n = n_points
    current_h = h_start.copy()
    
    # Simulated Annealing parameters
    initial_temp = 0.1
    cooling_rate = 0.995
    min_temp = 1e-6
    max_iter_per_temp = 100
    patience = 100
    
    no_improvement = 0
    
    for res_idx, target_n in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to target resolution if needed
        if target_n > current_n:
            x_orig = np.linspace(0, 2, current_n)
            x_new = np.linspace(0, 2, target_n)
            current_h = np.interp(x_new, x_orig, current_h)
            current_n = target_n
            
        # SA for this resolution
        temp = initial_temp
        h_current = current_h.copy()
        h_current = h_current * (target_n/2) / np.sum(h_current)
        
        current_c5 = evaluate_erdos_solution(h_current, 2.0/target_n)
        
        for iter_count in range(max_iter_per_temp * 5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate neighbor solution
            h_new = h_current.copy()
            perturb_idx = np.random.randint(0, target_n)
            delta = np.random.normal(0, temp * 0.1)
            h_new[perturb_idx] += delta
            
            # Boundary checks
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (target_n/2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, 2.0/target_n)
            
            # Accept or reject
            if c5_new < current_c5:
                h_current = h_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_current.copy()
                    no_improvement = 0
            else:
                # Accept with probability
                p_accept = np.exp(-(c5_new - current_c5) / temp)
                if np.random.rand() < p_accept:
                    h_current = h_new
                    current_c5 = c5_new
                no_improvement += 1
                
            # Cooling schedule
            if iter_count % 50 == 0:
                temp *= cooling_rate
                if temp < min_temp:
                    temp = min_temp
                    
            if no_improvement >= patience:
                break
        
        current_h = h_current
    
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, 2.0/current_n)
        
        def constraint_sum(h):
            return np.sum(h) - current_n / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(current_n)]
        
        res = minimize(objective, current_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Ensure constraint and validate final solution
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (current_n/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, 2.0/current_n)
    
    return (list(best_h), best_c5, current_n)
