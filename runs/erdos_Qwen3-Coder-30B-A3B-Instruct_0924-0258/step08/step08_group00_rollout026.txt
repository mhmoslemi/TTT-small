import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization
    resolutions = [35, 70, 140, 280]
    current_n = n_points
    current_h = h_start.copy()
    
    for res_idx, target_n in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to target resolution
        if target_n > current_n:
            x_orig = np.linspace(0, 2, current_n)
            x_new = np.linspace(0, 2, target_n)
            current_h = np.interp(x_new, x_orig, current_h)
            current_n = target_n
        elif target_n < current_n:
            # Downsample by averaging
            new_h = []
            step = current_n // target_n
            for i in range(target_n):
                start_idx = i * step
                end_idx = min((i + 1) * step, current_n)
                new_h.append(np.mean(current_h[start_idx:end_idx]))
            current_h = np.array(new_h)
            current_n = target_n
            
        # Simulated Annealing at this resolution
        current_temp = 0.1
        cooling_rate = 0.995
        max_iter = 10000
        patience = 100
        no_improvement = 0
        
        h_current = current_h.copy()
        h_current = h_current * (current_n/2) / np.sum(h_current)
        current_obj = evaluate_erdos_solution(h_current, 2.0 / current_n)
        
        # Initialize with best so far if this is first resolution
        if res_idx == 0:
            h_current = best_h.copy()
            h_current = h_current * (current_n/2) / np.sum(h_current)
            current_obj = evaluate_erdos_solution(h_current, 2.0 / current_n)
            
        for iter_num in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Propose new solution
            h_new = h_current.copy()
            
            # Randomly perturb
            idx = np.random.randint(0, current_n)
            delta = np.random.normal(0, current_temp)
            h_new[idx] += delta
            h_new[idx] = max(0, min(1, h_new[idx]))
            
            # Maintain sum constraint
            h_new = h_new * (current_n/2) / np.sum(h_new)
            
            # Evaluate
            new_obj = evaluate_erdos_solution(h_new, 2.0 / current_n)
            
            # Accept or reject
            if new_obj < current_obj or np.random.rand() < np.exp(-(new_obj - current_obj) / current_temp):
                h_current = h_new
                current_obj = new_obj
                
                if current_obj < best_c5:
                    best_c5 = current_obj
                    best_h = h_current.copy()
                    no_improvement = 0
                else:
                    no_improvement += 1
            else:
                no_improvement += 1
                
            if no_improvement >= patience:
                break
                
            current_temp *= cooling_rate
            if current_temp < 1e-8:
                break
                
        current_h = h_current.copy()

    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / current_n)
        
        def constraint_sum(h):
            return np.sum(h) - current_n / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(current_n)]
        
        res = minimize(objective, current_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    # Ensure last check
    final_c5 = evaluate_erdos_solution(best_h, 2.0 / len(best_h))
    
    return (list(best_h), final_c5, len(best_h))
