import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # SDP approach
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Variable for h
    h = cp.Variable(n_points, bounds=(0, 1))
    
    # Constraint: sum(h) = n_points / 2
    sum_constraint = cp.Equality(cp.sum(h), n_points / 2)
    
    # Constraint: C5 >= integral for all lags
    # For each k in [-n+1, n-1], compute correlation
    # We'll add constraints directly in a loop
    constraints = [sum_constraint]
    
    # Objective
    C5 = cp.Variable()
    
    # Compute all possible lags
    for k in range(1 - n_points, n_points):
        if k >= 0:
            # lag positive: h[i] (1-h[i+k]) for i in [0, n_points-k)
            indices_i = list(range(n_points - k))
            indices_j = list(range(k, n_points))
            if len(indices_i) > 0:
                # sum_{i=0}^{n_points-k-1} h[i] (1 - h[i+k]) <= C5 * dx
                # = sum_{i=0}^{n_points-k-1} h[i] - sum_{i=0}^{n_points-k-1} h[i]*h[i+k] <= C5 * dx
                # We model the product using big-M relaxation or truncate to linear constraint:
                # Let's simplify and just enforce the explicit form:
                # sum(h[i] * (1 - h[i+k])) <= C5 * dx
                corr_term = cp.sum(cp.multiply(h[indices_i], 1 - h[indices_j]))
                constraints.append(corr_term <= C5 * dx)
        else:
            # lag negative: h[i+k] (1-h[i]) for i in [0, n_points+k)
            indices_i = list(range(-k, n_points))
            indices_j = list(range(n_points + k))
            if len(indices_i) > 0:
                corr_term = cp.sum(cp.multiply(h[indices_i], 1 - h[indices_j]))
                constraints.append(corr_term <= C5 * dx)
    
    # Optimization problem
    objective = cp.Minimize(C5)
    prob = cp.Problem(objective, constraints)
    
    solver_max_time = 580.0  # Leave some time for postprocessing
    
    # Try solving SDP
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=solver_max_time)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_optimal = h.value
            # Rescale if needed to match mean constraint
            h_optimal = h_optimal * (n_points/2) / np.sum(h_optimal)
            # Clip to [0, 1] if needed
            h_optimal = np.clip(h_optimal, 0, 1)
            # Evaluate
            c5_new = evaluate_erdos_solution(h_optimal, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_optimal.copy()
    except Exception:
        pass
    
    # Fallback to gradient descent
    if best_c5 >= 0.381:
        step_size = 0.01
        max_iter = 10000
        iter_count = 0
        h_current = best_h.copy()
        
        while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
            # Compute all correlations for all possible lags
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            # Get the maximum overlap lag
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (len(h_current) - 1)
            
            gradient = np.zeros(n_points)
            if max_lag >= 0:
                for i in range(n_points - max_lag):
                    gradient[i] += 1 - h_current[i + max_lag]
                    if i + max_lag < n_points:
                        gradient[i + max_lag] -= h_current[i]
            else:
                for i in range(n_points + max_lag):
                    gradient[i] += 1 - h_current[i - max_lag]
                    if i - max_lag >= 0:
                        gradient[i - max_lag] -= h_current[i]
            
            # Gradient descent step
            h_new = h_current - step_size * gradient
            
            # Ensure feasibility
            h_new = np.clip(h_new, 0, 1)
            # Renormalize
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate new C5 value
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept step if improvement
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
            # If not improving, reduce step size
            if c5_new >= evaluate_erdos_solution(h_current, dx):
                step_size *= 0.5
                if step_size < 1e-10:
                    break
            else:
                h_current = h_new
                step_size *= 1.05  # Increase step size slightly
            
            iter_count += 1
    
    return (list(best_h), best_c5, n_points)
