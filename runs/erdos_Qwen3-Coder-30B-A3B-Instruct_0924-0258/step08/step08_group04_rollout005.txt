import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) < 1e-10:
        return h
    
    # Use a more robust projection method
    scale_factor = (n_points/2) / current_sum
    h = h * scale_factor
    h = np.clip(h, 0, 1)
    
    # Final adjustment
    diff = n_points/2 - np.sum(h)
    if abs(diff) > 1e-10:
        # Find largest values to reduce
        sorted_indices = np.argsort(h)[::-1]
        for i in sorted_indices:
            if abs(diff) < 1e-10:
                break
            decrease = min(diff, h[i])
            h[i] -= decrease
            diff += decrease
            
    return h

def compute_gradient(h, max_lags, n_points, dx):
    """Compute gradient for given top lags"""
    gradient = np.zeros(n_points)
    
    for k in max_lags:
        if k >= 0:
            # h[i] contributes to overlap with h[i+k]
            for i in range(n_points - k):
                gradient[i] += 1 - h[i + k]
                if i + k < n_points:
                    gradient[i + k] -= h[i]
        else:
            # h[i] contributes to overlap with h[i+k] where k is negative
            for i in range(n_points + k):
                gradient[i] += 1 - h[i - k]
                if i - k < n_points:
                    gradient[i - k] -= h[i]
    
    return gradient

def line_search(h, gradient, n_points, dx, max_iter=10):
    """Backtracking line search for optimal step size"""
    step_size = 1.0
    alpha = 0.5
    beta = 0.7
    
    current_c5 = evaluate_erdos_solution(h, dx)
    
    for _ in range(max_iter):
        h_new = h - step_size * gradient
        h_new = project_to_feasible(h_new, n_points, dx)
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < current_c5:
            return step_size, h_new
        step_size *= beta
    
    return step_size, h_new
