import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter_sa = int(1e5)
    
    # Initialize
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    T = T_init
    current_c5 = evaluate_erdos_solution(h, dx)
    
    # Track the best solution
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing loop
    iter_count = 0
    for i in range(max_iter_sa):
        if time() - start_time > budget_s * 0.9:
            break
            
        iter_count += 1
        
        # Perturb two random indices
        i1 = random.randint(0, n_points - 1)
        i2 = random.randint(0, n_points - 1)
        delta = random.uniform(-0.1, 0.1)
        
        # Store original values
        h_old = h.copy()
        c5_old = current_c5
        
        # Update h[i1] and h[i2]
        h[i1] += delta
        h[i2] -= delta
        
        # Keep within bounds
        h[i1] = max(0, min(1, h[i1]))
        h[i2] = max(0, min(1, h[i2]))
        
        # Adjust to maintain sum constraint
        diff = np.sum(h) - n_points / 2
        if abs(diff) > 1e-10:
            # Distribute difference
            h[0] += diff
        
        # Ensure new values are within bounds
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Compute new overlap
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Accept or reject
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or random.random() < np.exp(-delta_c5 / T):
            current_c5 = new_c5
        else:
            # Restore original state
            h = h_old
            current_c5 = c5_old
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Cooling schedule
        T = max(T_min, T * alpha)
    
    # Local refinement using L-BFGS-B on the best solution
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
