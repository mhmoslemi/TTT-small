import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex optimization approach to escape local minima
    max_iter = 5000
    epsilon_0 = 0.001
    beta = 0.99
    num_shifts = 100
    
    # Convert to CVXPY variables
    h_var = cp.Variable(n_points)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Identify dominant shifts
        corr = np.correlate(h, 1 - h, mode='full')
        top_shifts = np.argsort(corr)[-num_shifts:]
        
        # Form convexified objective
        objective = 0
        for k in top_shifts:
            # Compute correlation for specific shift k
            if k <= n_points - 1:
                # Right shift case
                corr_k = np.convolve(h, 1 - h, mode='full')
                objective += corr_k[k + n_points - 1]  # Adjust index
            else:
                # Left shift case
                corr_k = np.convolve(h, 1 - h, mode='full')
                objective += corr_k[k - (n_points - 1)]
        
        # Add regularization term
        epsilon = epsilon_0 * (beta ** iteration)
        objective += epsilon * cp.sum_squares(h_var)
        
        # Define constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Create and solve problem
        problem = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                # Project back
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Evaluate new solution
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                # Accept if better
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
            else:
                # Fall back to gradient step with small random perturbation
                h_new = h + (np.random.rand(n_points) - 0.5) * 0.01
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    
        except:
            # Fallback to basic method
            h_new = h + (np.random.rand(n_points) - 0.5) * 0.01
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                
    # Final refinement: random local search
    for _ in range(100):
        if time() - start_time > budget_s:
            break
        perturb = np.random.rand(n_points) * 0.001
        h_test = h + perturb
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        c5_test = evaluate_erdos_solution(h_test, dx)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()
    
    return (list(best_h), best_c5, n_points)
