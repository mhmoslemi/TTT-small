import numpy as np
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_fine = np.array(initial_h_values)
        n_points = len(h_fine)
    else:
        n_points = 35
        h_fine = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_fine.copy()
    best_c5 = evaluate_erdos_solution(h_fine, dx)
    
    # Coarse grid optimization (15 points)
    n_coarse = 15
    h_coarse = h_fine[::(n_points // n_coarse)] if n_points >= n_coarse else h_fine.copy()
    h_coarse = np.clip(h_coarse, 0, 1)
    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    # Coarse grid optimization for 500 iterations
    step_size = 0.01
    for iteration in range(500):
        if time() - start_time > budget_s * 0.8:
            break
        # Compute overlaps
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_coarse - 1)
        
        # Gradient computation
        grad = np.zeros(n_coarse)
        for i in range(n_coarse):
            j = i + max_lag
            if 0 <= j < n_coarse:
                grad[i] = 1 - 2 * h_coarse[i] - h_coarse[j]
            else:
                grad[i] = 1 - 2 * h_coarse[i]
        
        # Gradient descent step
        h_coarse_new = h_coarse - step_size * grad
        h_coarse_new = np.clip(h_coarse_new, 0, 1)
        
        # Renormalize to maintain integral constraint  
        h_coarse_new = h_coarse_new * (n_coarse / 2) / np.sum(h_coarse_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_coarse_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_coarse_new.copy()
            h_coarse = h_coarse_new.copy()
        else:
            step_size *= 0.99
            
    # Upsample to fine grid
    coarse_indices = np.linspace(0, n_points - 1, n_coarse, dtype=int)
    fine_indices = np.arange(n_points)
    f = interp1d(coarse_indices, h_coarse, kind='linear', fill_value='extrapolate')
    h_fine = f(fine_indices)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    
    # Fine grid optimization: alternating gradient descent and mutation
    step_size = 0.01
    mutation_rate = 0.1
    for phase in range(2):
        for iteration in range(1000):
            if time() - start_time > budget_s * 0.95:
                break
            # Gradient descent
            corr = np.correlate(h_fine, 1 - h_fine, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (n_points - 1)
            
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + max_lag
                if 0 <= j < n_points:
                    grad[i] = 1 - 2 * h_fine[i] - h_fine[j]
                else:
                    grad[i] = 1 - 2 * h_fine[i]
            
            h_fine_new = h_fine - step_size * grad
            h_fine_new = np.clip(h_fine_new, 0, 1)
            h_fine_new = h_fine_new * (n_points / 2) / np.sum(h_fine_new)
            
            if evaluate_erdos_solution(h_fine_new, dx) < best_c5:
                best_c5 = evaluate_erdos_solution(h_fine_new, dx)
                best_h = h_fine_new.copy()
                h_fine = h_fine_new.copy()
            else:
                # Mutation
                mutation_mask = np.random.random(n_points) < mutation_rate
                h_fine_new = h_fine.copy()
                h_fine_new[mutation_mask] = np.random.uniform(0, 1, np.sum(mutation_mask))
                h_fine_new = h_fine_new * (n_points / 2) / np.sum(h_fine_new)
                
                if evaluate_erdos_solution(h_fine_new, dx) < best_c5:
                    best_c5 = evaluate_erdos_solution(h_fine_new, dx)
                    best_h = h_fine_new.copy()
                    h_fine = h_fine_new.copy()
                    
    return (list(best_h), best_c5, n_points)
