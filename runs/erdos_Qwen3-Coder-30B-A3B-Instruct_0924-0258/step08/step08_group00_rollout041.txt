import numpy as np
from scipy.optimize import minimize
from time import time
import random

def project(h, n_points):
    """Project h onto feasible set: [0,1]^n with sum(h) = n/2"""
    h = np.clip(h, 0, 1)
    if np.sum(h) == 0:
        return h
    h = h * (n_points / 2.0) / np.sum(h)
    return h

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def compute_gradient(h, max_lag, n_points):
    """Compute gradient of correlation at max_lag"""
    shifted = np.roll(1 - h, max_lag)
    grad = 2 * (1 - 2 * h) * shifted
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = project(h_start, n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    step_size = 0.1
    patience = 50
    no_improvement = 0
    max_iter = 1000
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient
        grad = compute_gradient(h, max_lag, n_points)
        
        # Update step
        h_new = h - step_size * grad
        h_new = project(h_new, n_points)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        h = h_new
        
        if no_improvement >= patience:
            step_size *= 0.5
            no_improvement = 0
            if step_size < 1e-8:
                break
    
    # Try a refinement with SLSQP if final result is better than baseline
    if time() - start_time < budget_s * 0.9:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            
            if res.success:
                h_final = project(res.x, n_points)
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
            
    return (list(best_h), best_c5, n_points)
