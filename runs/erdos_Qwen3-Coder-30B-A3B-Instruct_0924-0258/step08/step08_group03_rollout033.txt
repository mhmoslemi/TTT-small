import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search parameters
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    def is_feasible(c5):
        # Formulate convex feasibility problem
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Sample shifts for feasibility check
        shifts = np.random.choice(n_points, size=min(100, n_points), replace=False)
        
        for k in shifts:
            # Compute overlap constraint
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h_var[i] * (1 - h_var[j])
            
            constraints.append(overlap_expr <= c5 * dx)
            
        # Create and solve problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return prob.status == cp.OPTIMAL
        except:
            return False
    
    # Binary search
    while time() - start_time < budget_s * 0.9 and high - low > tolerance:
        mid = (low + high) / 2
        if is_feasible(mid):
            high = mid
            # Refine solution
            h_refined = h.copy()
            # Perform local optimization
            bounds = [(0, 1) for _ in range(n_points)]
            for _ in range(500):
                if time() - start_time > budget_s * 0.95:
                    break
                x = h_refined.copy()
                h_refined = x.copy()
                
                # Evaluate current value
                c5_current = evaluate_erdos_solution(h_refined, dx)
                if c5_current < best_c5:
                    best_c5 = c5_current
                    best_h = h_refined.copy()
            
            # Significant improvement - restart binary search
            if best_c5 < 0.3807:
                break
        else:
            low = mid
    
    # Final refinement with gradient descent
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    nfev = 0
    max_iter = 1000
    step_size = 0.1
    
    for _ in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h[i] - h[j]
            else:
                grad[i] = 1 - 2 * h[i]
        
        # Gradient descent step
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to maintain constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        nfev += 1
        
        # Accept improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        
        # Stop if no improvement
        if nfev > 500 and abs(c5_new - objective(h)) < 1e-8:
            break
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
