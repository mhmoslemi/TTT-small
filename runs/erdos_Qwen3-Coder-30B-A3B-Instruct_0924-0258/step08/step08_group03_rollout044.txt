import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from cvxpy import Semidefinite, Variable, Minimize, sum as cp_sum, trace, diag, multiply

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    # Add a bit of randomness to avoid local minima
    h += np.random.normal(0, 1e-4, n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search approach with SDP relaxation
    def check_feasible(c5_val):
        # SDP relaxation approach
        try:
            Y = Semidefinite(n_points)
            h_var = Variable(n_points)
            
            # Constraints
            constraints = [
                Y >> 0,                        # Positive semidefinite
                diag(Y) == h_var,              # Diagonal elements
                cp_sum(h_var) == n_points/2,   # Constraint on sum
            ]
            
            # Add overlap constraints for sampled shifts
            shifts = np.random.choice(n_points, size=min(100, n_points), replace=False)
            for k in shifts:
                # This is an approximate constraint
                constraint = cp_sum(multiply(h_var, 1 - h_var)) <= c5_val * dx
                constraints.append(constraint)
            
            # Check if solution exists
            prob = cp.Problem(Minimize(0), constraints)
            prob.solve(solver=cp.MOSEK, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_sol = h_var.value
                if h_sol is not None:
                    return True, h_sol
            return False, None
        except:
            return False, None
    
    # Binary search over C5
    low, high = 0.38, 0.38085
    best_c5 = 0.38085
    best_h_local = h.copy()
    
    while time() - start_time < budget_s * 0.95 and high - low > 1e-6:
        mid = (low + high) / 2
        feasible, h_sol = check_feasible(mid)
        
        if feasible:
            high = mid
            if h_sol is not None:
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                c5_val = evaluate_erdos_solution(h_sol, dx)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h_local = h_sol.copy()
        else:
            low = mid
    
    # Local refinement using gradient ascent
    def obj_func(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    h_init = best_h_local.copy()
    h_opt = h_init.copy()
    
    # Use scipy.optimize for better local refinement
    loss_start = obj_func(h_init)
    result = minimize(obj_func, h_init, method="L-BFGS-B", bounds=[(0,1) for _ in range(n_points)])
    
    h_refined = project_h(result.x)
    loss_end = obj_func(h_refined)
    
    if loss_end < best_c5:
        best_h_local = h_refined
        best_c5 = loss_end
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h_local, dx)
    return (list(best_h_local), final_c5, n_points)
