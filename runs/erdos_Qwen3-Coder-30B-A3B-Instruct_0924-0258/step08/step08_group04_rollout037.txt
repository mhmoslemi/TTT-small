import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum(h) = target_sum while keeping in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize PSO parameters
    num_particles = 75
    max_iter = 5000
    inertia = 0.8
    inertia_decay = 0.995
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    perturbation_rate = 0.01
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    particle_best = []
    particle_best_fitness = []
    
    for _ in range(num_particles):
        h = h_start.copy()
        h = project_to_feasible(h + np.random.normal(0, 0.05, n_points), target_sum)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        particle_best.append(h.copy())
        particle_best_fitness.append(evaluate_erdos_solution(h, dx))
    
    global_best_idx = np.argmin(particle_best_fitness)
    global_best = particle_best[global_best_idx].copy()
    global_best_fitness = particle_best_fitness[global_best_idx]
    
    iter_count = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_weight * r1 * (particle_best[i] - particles[i])
            social = social_weight * r2 * (global_best - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update particle best
            if fitness < particle_best_fitness[i]:
                particle_best[i] = particles[i].copy()
                particle_best_fitness[i] = fitness
            
            # Update global best
            if fitness < global_best_fitness:
                global_best = particles[i].copy()
                global_best_fitness = fitness
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Update inertia
        inertia = max(0.4, inertia * inertia_decay)
        
        # Apply perturbation occasionally
        if iter_count % 1000 == 0 and iter_count > 0:
            for i in range(num_particles):
                if np.random.random() < perturbation_rate:
                    particles[i] += np.random.normal(0, 0.05, n_points)
                    particles[i] = project_to_feasible(particles[i], target_sum)
        
        iter_count += 1
        
        # Early termination if no improvement
        if no_improvement_count > 5000:
            break
    
    # Final refinement with local search
    final_h = global_best.copy()
    final_c5 = global_best_fitness
    
    # Try small local perturbations
    for _ in range(1000):
        if time() - start_time > budget_s * 0.98:
            break
        pert_idx = np.random.randint(0, n_points)
        step_size = 0.005
        pert_h = final_h.copy()
        pert_h[pert_idx] += np.random.normal(0, step_size)
        pert_h = project_to_feasible(pert_h, target_sum)
        new_c5 = evaluate_erdos_solution(pert_h, dx)
        if new_c5 < final_c5:
            final_h = pert_h
            final_c5 = new_c5
    
    return (list(final_h), final_c5, n_points)
