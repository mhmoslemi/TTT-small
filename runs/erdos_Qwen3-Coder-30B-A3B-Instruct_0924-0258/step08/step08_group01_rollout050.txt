import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    generations = 500
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 5
    max_no_improvement = 100
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        population.append(create_individual())
    
    # Track best solution
    best_fitness_history = []
    no_improvement_count = 0
    
    # Main GA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness for all individuals
        fitnesses = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitnesses.append(fit)
        
        # Track best fitness
        min_fitness = min(fitnesses)
        best_fitness_history.append(min_fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitnesses.index(min_fitness)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
        # Tournament selection
        def tournament_selection():
            k = 3
            candidates = random.sample(list(enumerate(population)), k)
            winner = min(candidates, key=lambda x: fitnesses[x[0]])
            return winner[1]
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection()
            parent2 = tournament_selection()
            
            # Crossover
            if random.random() < crossover_rate:
                alpha = random.random()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if random.random() < mutation_rate:
                # Random mutation with clip and rescale
                mutate_idx = random.randint(0, n_points - 1)
                child1[mutate_idx] += np.random.normal(0, 0.05)
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points / 2) / np.sum(child1)
            
            if random.random() < mutation_rate:
                mutate_idx = random.randint(0, n_points - 1)
                child2[mutate_idx] += np.random.normal(0, 0.05)
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        # Keep population size
        population = new_population[:population_size]
    
    # Final refinement with SLSQP for best individual
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
