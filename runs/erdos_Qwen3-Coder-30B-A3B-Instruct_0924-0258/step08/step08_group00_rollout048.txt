import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-start approach with Latin hypercube sampling
    num_starts = 50
    best_h_global = best_h.copy()
    best_c5_global = best_c5
    
    def repair_h(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return h
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Generate Latin hypercube samples
    latin_samples = np.random.random((num_starts, n_points))
    for i in range(n_points):
        latin_samples[:, i] = np.sort(latin_samples[:, i])
        latin_samples[:, i] = (latin_samples[:, i] * (1 - 0) + 0)
    
    # Ensure sum constraint
    latin_samples = np.array([repair_h(s) for s in latin_samples])
    
    # Add the initial solution if available
    if 'initial_h_values' in globals():
        latin_samples[0] = h_start.copy()
    
    for i in range(num_starts):
        h0 = latin_samples[i].copy()
        if time() - start_time > budget_s * 0.95:
            break
            
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h0, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                h_final = repair_h(h_final)
                c5_final = objective(h_final)
                if c5_final < best_c5_global:
                    best_c5_global = c5_final
                    best_h_global = h_final.copy()
        except Exception:
            pass
    
    # Final refinement with a few additional local optimizations
    h_refine = best_h_global.copy()
    for _ in range(10):
        if time() - start_time > budget_s * 0.95:
            break
            
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_refine, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            
            if res.success:
                h_refine = res.x
                h_refine = repair_h(h_refine)
                c5_refine = objective(h_refine)
                if c5_refine < best_c5_global:
                    best_c5_global = c5_refine
                    best_h_global = h_refine.copy()
        except Exception:
            break
    
    return (list(best_h_global), best_c5_global, n_points)
