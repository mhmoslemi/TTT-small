import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_onto_feasible(h, n_points):
    """Project h onto feasible set: 0 <= h <= 1 and sum(h) = n_points/2"""
    # First clip to [0,1]
    h = np.clip(h, 0, 1)
    # Then rescale to meet sum constraint
    h_sum = np.sum(h)
    if abs(h_sum) < 1e-10:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Optimization parameters
    max_iter = 5000
    step_size = 0.01
    line_search_factor = 0.5
    perturb_freq = 500
    perturb_magnitude = 0.05
    
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlations for all lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Get indices of top k lags by overlap
        k = 5
        top_lag_indices = np.argsort(corr)[-k:][::-1]
        top_lags = [i - (n_points - 1) for i in top_lag_indices]
        
        # Compute combined gradient from top k lags
        gradient = np.zeros(n_points)
        for lag in top_lags:
            if lag >= 0:
                # h[i] contributes to overlap with h[i+lag]
                for i in range(n_points - lag):
                    gradient[i] += 1 - h_current[i + lag]
                    if i + lag < n_points:
                        gradient[i + lag] -= h_current[i]
            else:
                # negative lag: h[i] overlaps with h[i-lag] with smaller indices
                for i in range(-lag, n_points):
                    gradient[i] += 1 - h_current[i + lag]
                    if i + lag >= 0:
                        gradient[i + lag] -= h_current[i]
        
        # Line search for optimal step size
        step_size_ls = step_size
        improved = False
        while step_size_ls > 1e-10 and not improved:
            h_new = h_current - step_size_ls * gradient
            h_new = project_onto_feasible(h_new, n_points)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                h_current = h_new
                best_c5 = c5_new
                improved = True
                step_size *= 1.1  # Boost step size on improvement
            else:
                step_size_ls *= line_search_factor  # Reduce step size
                
        if not improved:
            step_size *= 0.8  # Reduce step size generally
            
        # Periodic perturbation to escape stagnation
        if iter_count % perturb_freq == 0 and iter_count > 0:
            perturb = np.random.normal(0, perturb_magnitude, n_points)
            h_current = h_current + perturb
            h_current = project_onto_feasible(h_current, n_points)
            new_c5 = evaluate_erdos_solution(h_current, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                
    return (list(best_h), best_c5, n_points)
