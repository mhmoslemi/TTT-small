import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    crossover_rate = 0.8
    mutation_rate = 0.1
    max_no_improvement = 100
    no_improvement_count = 0
    generation = 0
    checkpoint_interval = 100
    max_generations = 10000
    
    # Initialize population
    def create_individual():
        h = np.random.random(n_points) * 0.8 + 0.1  # Keep values away from boundaries
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    if 'initial_h_values' in globals():
        population = [h_start.copy()]
        for _ in range(population_size - 1):
            population.append(create_individual())
    else:
        population = [create_individual() for _ in range(population_size)]
    
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    def tournament_selection(pop, k=3):
        tournament = random.sample(pop, min(k, len(pop)))
        return min(tournament, key=fitness)
    
    def blend_crossover(h1, h2, alpha=0.3):
        # Blend crossover
        if random.random() > crossover_rate:
            return h1.copy(), h2.copy()
        child1 = np.zeros_like(h1)
        child2 = np.zeros_like(h2)
        for i in range(len(h1)):
            if random.random() > 0.5:
                child1[i] = h1[i]
                child2[i] = h2[i]
            else:
                d = abs(h1[i] - h2[i])
                lower = min(h1[i], h2[i]) - alpha * d
                upper = max(h1[i], h2[i]) + alpha * d
                child1[i] = np.clip(random.uniform(lower, upper), 0, 1)
                child2[i] = np.clip(random.uniform(lower, upper), 0, 1)
        # Fix sum constraint
        sum1 = np.sum(child1)
        sum2 = np.sum(child2)
        child1 = child1 * (n_points / 2) / sum1
        child2 = child2 * (n_points / 2) / sum2
        return child1, child2
    
    def mutate(h, mutation_strength=0.05):
        # Gaussian mutation and normalize
        h_new = h.copy()
        for i in range(len(h_new)):
            if random.random() < mutation_rate:
                # Add noise
                noise = np.random.normal(0, mutation_strength)
                h_new[i] += noise
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Main GA loop
    try:
        while time() - start_time < budget_s * 0.7 and generation < max_generations:
            # Evaluate current population
            fitness_values = [fitness(ind) for ind in population]
            current_best = min(fitness_values)
            
            if current_best < best_c5:
                best_c5 = current_best
                best_individual = population[np.argmin(fitness_values)].copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            if no_improvement_count > max_no_improvement:
                break
            
            # Select parents
            parents = [tournament_selection(population) for _ in range(population_size)]
            
            # Create offspring
            offspring = []
            for i in range(0, population_size, 2):
                if i + 1 < population_size:
                    child1, child2 = blend_crossover(parents[i], parents[i+1])
                    offspring.extend([child1, child2])
                else:
                    offspring.append(mutate(parents[i]))
            
            # Apply mutation
            mutated_offspring = [mutate(ind, 0.05 * (0.95 ** generation)) for ind in offspring]
            
            # Replace worst individuals
            all_individuals = population + mutated_offspring
            all_fitness = [fitness(ind) for ind in all_individuals]
            sorted_indices = np.argsort(all_fitness)
            population = [all_individuals[i] for i in sorted_indices[:population_size]]
            
            generation += 1
            
            # Periodic refinement
            if generation % checkpoint_interval == 0:
                # Use best from population as basis for SLSQP refinement
                try:
                    best_candidate = population[np.argmin([fitness(ind) for ind in population])]
                    def objective(h):
                        return evaluate_erdos_solution(h, dx)
                    def constraint_sum(h):
                        return np.sum(h) - n_points / 2.0
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    bounds = [(0, 1) for _ in range(n_points)]
                    res = minimize(objective, best_candidate, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-10})
                    if res.success:
                        refined_h = res.x
                        refined_c5 = objective(refined_h)
                        if refined_c5 < best_c5:
                            best_c5 = refined_c5
                            best_h = refined_h
                except Exception:
                    pass
                
    except Exception as e:
        pass
    
    # Final refinement step with SLSQP on the best solution found
    try:
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(lambda h: evaluate_erdos_solution(h, dx), best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum)
        if res.success:
            final_h = res.x
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
    
    # Ensure bounds and constraints one final time
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
