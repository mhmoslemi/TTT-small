import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        n_points = len(h_start)

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    population_size = 50
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 2
    max_generations = 10000
    no_improvement_limit = 500
    refinement_interval = 100
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        population[0] = population[0] * (n_points / 2) / np.sum(population[0])
    
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    generation = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.7:
        # Evaluate fitness
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= no_improvement_limit:
            break
        
        # Create new population
        new_population = []
        
        # Elite
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Tournament selection and crossover
        while len(new_population) < population_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(population_size, 2, replace=False)
            parent1 = population[idx1] if fitness[idx1] < fitness[idx2] else population[idx2]
            
            idx3, idx4 = np.random.choice(population_size, 2, replace=False)
            parent2 = population[idx3] if fitness[idx3] < fitness[idx4] else population[idx4]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
                # Make sure it sums to n_points / 2
                child = child * (n_points / 2) / np.sum(child)
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        delta = np.random.normal(0, 0.05)
                        child[i] += delta
                # Clip to bounds
                child = np.clip(child, 0, 1)
                # Adjust to maintain constraint
                child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        generation += 1
    
    # Final refinement with SLSQP for best solution
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
