import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search on C5
    low, high = 0.38, 0.38085
    tolerance = 1e-5
    
    # Sample shifts to reduce constraint count
    shift_indices = np.random.choice(n_points, size=min(100, n_points), replace=False)
    
    while high - low > tolerance and time() - start_time < budget_s * 0.9:
        mid = (low + high) / 2
        
        # Convex feasibility problem formulation
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Overlap constraints for sampled shifts
        for k_idx in shift_indices:
            # Compute overlap constraint for shift k_idx
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k_idx) % n_points
                overlap_expr += h_var[i] * (1 - h_var[j])
            constraints.append(overlap_expr <= mid * dx)
        
        # Formulate and solve
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_candidate = h_var.value
                # Check actual constraint satisfaction
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
                high = mid
            else:
                low = mid
        except:
            low = mid
    
    # Local refinement with gradient descent
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    def constraint_bounds(h_vec):
        return np.concatenate([h_vec, 1 - h_vec])
    
    # Use scipy minimize for local refinement
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    ref_start = best_h.copy()
    result = minimize(objective, ref_start, method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': 500, 'ftol': 1e-9}, callback=lambda x: None)
    
    if result.success:
        ref_h = np.clip(result.x, 0, 1)
        ref_h = ref_h * (n_points / 2) / np.sum(ref_h)
        ref_c5 = evaluate_erdos_solution(ref_h, dx)
        if ref_c5 < best_c5:
            best_c5 = ref_c5
            best_h = ref_h.copy()
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
