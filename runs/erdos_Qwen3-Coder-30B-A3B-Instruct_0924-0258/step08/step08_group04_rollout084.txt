import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    # Project h onto the feasible set: 0 <= h[i] <= 1 and sum(h) = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) < 1e-10:
        return h
    
    # Use a simple iterative approach for projection
    # First ensure we don't exceed bounds when rescaling
    target_sum = n_points / 2.0
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
        
    # Simple rescaling approach:
    # if sum < target, we increase values
    # if sum > target, we decrease values
    # to minimize the squared difference
    scaling_factor = target_sum / current_sum
    h = h * scaling_factor
    
    # Clamp to bounds again
    h = np.clip(h, 0, 1)
    
    # Final re-normalization to ensure sum == target
    current_sum = np.sum(h)
    h = h * target_sum / current_sum
    
    return h

def compute_gradients(h, max_lags, n_points):
    """Compute combined gradient from top lags"""
    grad = np.zeros(n_points)
    for lag in max_lags:
        if lag >= 0:
            for i in range(n_points - lag):
                grad[i] += 1 - h[i + lag]
                if i + lag < n_points:
                    grad[i + lag] -= h[i]
        else:
            for i in range(n_points + lag):
                grad[i] += 1 - h[i - lag]
                if i - lag >= 0:
                    grad[i - lag] -= h[i]
    return grad

def line_search(h, grad, step_size, n_points, dx):
    """Simple backtracking line search"""
    alpha = step_size
    # try different step sizes
    for _ in range(10):
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, n_points)
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < evaluate_erdos_solution(h, dx):
            return h_new, c5_new, alpha
        alpha *= 0.5
    return h, evaluate_erdos_solution(h, dx), alpha

    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:

        # Get the top K lags
