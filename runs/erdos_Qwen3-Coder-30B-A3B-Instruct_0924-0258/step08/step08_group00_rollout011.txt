import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based local search with restarts
    max_iter = 5000
    patience = 50
    no_improvement = 0
    step_size = 0.01
    
    # Add more sophisticated local search
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate overlap for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            # Restart with a better initial point
            h = np.random.rand(n_points) * 0.5
            h = h * (n_points/2) / np.sum(h)
            no_improvement = 0
            step_size = 0.01  # Reset step size
            
        # Limited gradient computation for a random lag
        rand_lag = np.random.randint(-n_points+1, n_points)
        lag_idx = rand_lag + n_points - 1
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + rand_lag
            if 0 <= j < n_points:
                grad[i] = 2 * (1 - h[j] - h[i]) / n_points
        
        # Update with gradient descent
        h_new = h - step_size * grad
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check improvement
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < max_overlap:
            h = h_new
            step_size *= 1.1  # Increase step size for faster convergence
        else:
            step_size *= 0.9  # Reduce step size if no improvement
            if step_size < 1e-8:
                step_size = 1e-6
    
    # Final refinement with SLSQP 
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try multiple random starting points for robustness
        best_final = best_c5
        best_final_h = best_h.copy()
        
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            h_restart = np.random.rand(n_points) * 0.5
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_final:
                    best_final = c5_final
                    best_final_h = h_final
        
        if best_final < best_c5:
            best_h = best_final_h
            best_c5 = best_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
