import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter = 100000
    
    # Initialize
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    current_c5 = evaluate_erdos_solution(h, dx)
    
    T = T_init
    iter_count = 0
    
    while T > T_min and time() - start_time < budget_s * 0.9:
        iter_count += 1
        if iter_count > max_iter:
            break
            
        # Generate perturbation
        h_new = h.copy()
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.1, 0.1)
        
        # Apply perturbation
        h_new[i1] += delta
        h_new[i2] -= delta
        
        # Ensure boundaries
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Compute new overlap
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
        else:
            prob = np.exp(-(new_c5 - current_c5) / T)
            if np.random.random() < prob:
                h = h_new
                current_c5 = new_c5
                
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Cooling
        T *= alpha
    
    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
