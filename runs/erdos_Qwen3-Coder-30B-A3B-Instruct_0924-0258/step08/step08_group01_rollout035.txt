import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for genetic algorithm
    population_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    mutation_step = 0.05
    slsqp_frequency = 100
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        for _ in range(population_size - 1):
            h = np.random.rand(n_points) * 0.2 + 0.4  # Random distribution
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points) * 0.2 + 0.4  # Random distribution
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    no_improvement_count = 0
    
    # GA loop
    for generation in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population and track best
        fitness = []
        for h in population:
            f = evaluate_erdos_solution(h, dx)
            fitness.append(f)
            if f < best_c5:
                best_c5 = f
                best_h = h.copy()
                no_improvement_count = 0
        
        # Select parents using tournament selection
        selected = []
        for _ in range(population_size // 2):
            # Tournament selection
            indices = np.random.choice(population_size, size=3, replace=False)
            best_idx = indices[np.argmin([fitness[i] for i in indices])]
            selected.append(population[best_idx].copy())
        
        # Generate offspring
        offspring = []
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < len(selected) else selected[0]
            
            # Crossover
            if random.random() < crossover_rate:
                alpha = random.uniform(0.3, 0.7)
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if random.random() < mutation_rate:
                    child1[j] += random.gauss(0, mutation_step)
                    child2[j] += random.gauss(0, mutation_step)
            
            # Enforce constraints
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            offspring.extend([child1, child2])
        
        # Replace population
        # Keep best from current population
        elite_indices = np.argsort(fitness)[:population_size//2]
        elite = [population[i] for i in elite_indices]
        # Add offspring
        population = elite + offspring[:population_size-len(elite)]
        
        no_improvement_count += 1
        if no_improvement_count > 200:
            # Restart with new population if no improvement
            for i in range(population_size):
                h = np.random.rand(n_points) * 0.2 + 0.4
                h = h * (n_points / 2) / np.sum(h)
                population[i] = h
            no_improvement_count = 0
        
        # Occasionally refine with SLSQP
        if generation % slsqp_frequency == 0:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 500, 'ftol': 1e-12})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final local refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
