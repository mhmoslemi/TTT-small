import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter = int(1e5)
    epsilon = 0.1
    
    # Initialize temperature
    T = T_init
    
    # Run Simulated Annealing
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Perturb two random indices
        i, j = np.random.randint(0, n_points, 2)
        delta = np.random.uniform(-epsilon, epsilon)
        
        # Ensure the perturbation maintains sum = n/2
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate overlaps
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject the move
        if c5_new < best_c5:
            h_current = h_new.copy()
            best_c5 = c5_new
            best_h = h_new.copy()
        else:
            # Accept with probability
            delta_c5 = c5_new - best_c5
            if np.random.rand() < np.exp(-delta_c5 / T):
                h_current = h_new.copy()
        
        # Cool down
        T *= alpha
        
        # Adapt epsilon based on improvement
        if iter % 1000 == 0 and iter > 0:
            epsilon = max(1e-6, epsilon * 0.95)
    
    # Local refinement with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        # Normalize starting point for refinement
        h_refine = best_h.copy()
        h_refine = h_refine * (n_points/2) / np.sum(h_refine)
        
        res = minimize(obj, h_refine, method='L-BFGS-B', bounds=bounds, options={'ftol': 1e-10, 'maxiter': 100})
        
        if res.success:
            final_h = res.x
            final_h = final_h * (n_points/2) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
