import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-scale optimization with increasing resolution
    max_scale = int(np.log2(n_points)) + 3
    current_scale = 0
    current_n_points = n_points
    h_current = h_start.copy()
    
    # Initialize step size
    step_size = 0.05
    
    scale_iterations = 10
    
    while current_scale < max_scale and (time() - start_time) < budget_s * 0.95:
        # Increase resolution
        current_n_points = min(2**current_scale, 1000) if current_scale > 0 else n_points
        if current_n_points > n_points:
            # Upsample
            h_new = np.interp(np.linspace(0, 1, current_n_points), np.linspace(0, 1, n_points), h_current)
            h_current = h_new
        elif current_n_points < n_points:
            # Downsample
            h_current = h_current[::n_points//current_n_points] if n_points % current_n_points == 0 else h_current[:current_n_points]
        
        # Perturb and optimize
        for _ in range(scale_iterations):
            # Create perturbed solution
            h_perturbed = h_current + step_size * (np.random.random(current_n_points) - 0.5)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            # Enforce sum constraint
            h_perturbed = h_perturbed * (current_n_points / 2) / np.sum(h_perturbed)
            
            # Evaluate
            c5_perturbed = evaluate_erdos_solution(h_perturbed, 2.0 / current_n_points)
            
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
                h_current = h_perturbed.copy()
                step_size = min(0.1, step_size * 1.05)
            else:
                step_size *= 0.95
            
            if (time() - start_time) > budget_s * 0.95:
                break
        
        # Local SQP refinement
        try:
            def objective(h):
                return evaluate_erdos_solution(h, 2.0 / len(h))
            
            def constraint_sum(h):
                return np.sum(h) - len(h) / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(len(h_current))]
            
            res = minimize(objective, h_current, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 50, 'ftol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except Exception:
            pass
        
        current_scale += 1

    return (list(best_h), best_c5, len(best_h))
