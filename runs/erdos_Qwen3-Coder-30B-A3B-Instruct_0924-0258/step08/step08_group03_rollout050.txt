import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    # Normalize h to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    n_iter = 0
    max_iter = 1000  # Prevent infinite loop
    
    while time() - start_time < budget_s * 0.95 and n_iter < max_iter:
        # Get current overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        sorted_indices = np.argsort(corr)[::-1]
        
        # Select top K most significant overlaps
        K = min(10 + n_iter // 100, 20)  # Gradually increase K
        top_indices = sorted_indices[:K]
        
        # Set up convex optimization problem
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Objective: weighted sum of selected overlaps
        objective = 0
        for i, idx in enumerate(top_indices):
            lag = idx - (n_points - 1)
            if lag >= 0:
                # h[i] * (1 - h[i+lag]) for valid indices
                objective += (h_var[i] * (1 - h_var[i + lag])) * (1.0 / (i + 1))
            else:
                # h[i] * (1 - h[i+lag]) where i+lag < 0 -> wrap around
                objective += (h_var[i] * (1 - h_var[i + lag + n_points])) * (1.0 / (i + 1))
        
        objective = cp.Minimize(objective)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h_new = h_var.value
        except:
            # Fallback to gradient method if convex solve fails
            h_new = h.copy()
            for _ in range(10):
                corr = np.correlate(h_new, 1 - h_new, mode='full')
                max_lag_idx = np.argmax(corr)
                max_lag = max_lag_idx - (n_points - 1)
                grad = np.zeros(n_points)
                for i in range(n_points):
                    j = i + max_lag
                    if 0 <= j < n_points:
                        grad[i] = 1 - 2 * h_new[i] - h_new[j]
                    else:
                        grad[i] = 1 - 2 * h_new[i]
                h_new = h_new - 0.01 * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Apply moving average filter to smooth
        filter_width = max(3, n_points // 10)
        kernel = np.ones(filter_width) / filter_width
        h_filtered = np.convolve(h_new, kernel, mode='same')
        h_filtered = np.clip(h_filtered, 0, 1)
        
        # Renormalize
        h_filtered = h_filtered * (n_points / 2) / np.sum(h_filtered)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_filtered, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_filtered.copy()
            h = h_filtered.copy()
        else:
            # Sometimes accept slightly worse if we're making progress
            if best_c5 - c5_new < 1e-8:
                h = h_filtered.copy()
        
        n_iter += 1
        
        # Prevent stagnation
        if n_iter % 200 == 0:
            # Random reset with some probability
            if np.random.rand() < 0.1:
                h = np.random.rand(n_points) * 0.5 + 0.25
                h = h * (n_points / 2) / np.sum(h)
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
