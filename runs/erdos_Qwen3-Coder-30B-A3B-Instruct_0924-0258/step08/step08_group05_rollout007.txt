import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        n_points = len(h_current)
    else:
        n_points = 35
        # Start with binary-like construction
        h_current = np.ones(n_points)
        h_current[:n_points//2] = 0.0
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
    dx = 2.0 / n_points
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    T_min = 1e-6
    cooling_rate = 0.9995
    max_iter = int(1e7)
    no_improvement_count = 0
    max_no_improvement = 5000
    
    # Iteration budget
    max_iterations = int(max_iter * 0.8)
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighborhood by swapping values between two indices
        i, j = np.random.choice(n_points, 2, replace=False)
        # Compute feasible delta
        delta_max = min(h_current[i], 1.0 - h_current[j])
        if delta_max <= 0:
            continue
        delta = np.random.uniform(0, delta_max)
        
        # Create new candidate solution
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        # Re-normalize to satisfy sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept with Metropolis criterion
        delta_energy = c5_new - best_c5
        if delta_energy < 0 or np.random.rand() < np.exp(-delta_energy / T):
            h_current = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_current.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Cooling schedule
        T *= cooling_rate
        
        # Early stopping if temperature too low or no improvement
        if T < T_min or no_improvement_count > max_no_improvement:
            break
    
    # Final refinement with optimization if time allows
    if time() - start_time < budget_s * 0.9:
        try:
            from scipy.optimize import minimize
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
