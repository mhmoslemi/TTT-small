import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convex relaxation to find starting point
    if n_points <= 100:  # Only use CVXPY for smaller problems
        try:
            h_cvx = cp.Variable(n_points)
            h_constraint = cp.sum(h_cvx) == n_points/2
            h_bounds = [0 <= h_cvx, h_cvx <= 1]
            
            # Create surrogate problem with quadratic penalties
            obj = 0
            for k in range(n_points):
                term = 0
                for i in range(n_points):
                    j = (i + k) % n_points
                    term += h_cvx[i] * (1 - h_cvx[j])
                obj += cp.square(term)
            
            prob = cp.Problem(cp.Minimize(obj), h_bounds + [h_constraint])
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_start = np.array(h_cvx.value).flatten()
                h_start = np.clip(h_start, 0, 1)
                h_start = h_start * (n_points/2) / np.sum(h_start)
        except:
            pass
    
    # Coordinate descent refinement
    h = h_start.copy()
    n_coord = max(1, n_points // 5)  # Optimize 20% of variables
    
    # Local refinement using L-BFGS-B
    try:
        from scipy.optimize import minimize
        
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)
        
        def gradient(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            max_k = np.argmax(overlaps)
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = (i + max_k) % n_points
                grad[i] = -2 * dx * (1 - h[j] - h[i])
            return grad
        
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                      jac=gradient, options={'maxiter': 50, 'ftol': 1e-9})
        
        if res.success:
            h = res.x
            h = h * (n_points/2) / np.sum(h)
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                
    except Exception as e:
        pass
    
    # Adaptive coordinate descent
    iterations = 0
    no_improve_count = 0
    max_no_improve = 50
    
    while time() - start_time < budget_s * 0.95 and no_improve_count < max_no_improve:
        iterations += 1
        if iterations > 2000:
            break
            
        h_current = best_h.copy()
        
        # Randomly select subset for update
        indices = np.random.permutation(n_points)[:n_coord]
        h_old = h_current[indices].copy()
        
        # Simple gradient-based update on selected indices
        overlaps = np.correlate(h_current, 1 - h_current, mode='full') * dx
        max_k = np.argmax(overlaps)
        
        # Compute gradient for selected indices
        for i in range(n_coord):
            idx = indices[i]
            j = (idx + max_k) % n_points
            grad = -2 * dx * (1 - h_current[j] - h_current[idx])
            h_current[idx] -= 0.01 * grad
            h_current[idx] = np.clip(h_current[idx], 0, 1)
        
        # Enforce sum constraint
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Evaluate and update best
        c5 = evaluate_erdos_solution(h_current, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_current.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
    
    # Final refinement
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
