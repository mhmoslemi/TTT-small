import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points / 2 and bounds [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # PSO parameters
    num_particles = 75
    inertia_start = 0.8
    inertia_end = 0.4
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    perturbation_rate = 0.01
    perturbation_interval = 1000
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize particles
    particles = []
    velocities = []
    particle_best = []
    particle_best_c5 = []
    
    for _ in range(num_particles):
        h = h_start.copy()
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        particle_best.append(h.copy())
        particle_best_c5.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    global_best_idx = np.argmin(particle_best_c5)
    global_best = particle_best[global_best_idx].copy()
    global_best_c5 = particle_best_c5[global_best_idx]
    
    max_iter = int(1e6)
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_count / max_iter)
        
        for i in range(num_particles):
            # Evaluate current particle
            c5_current = evaluate_erdos_solution(particles[i], dx)
            
            # Update particle best if improved
            if c5_current < particle_best_c5[i]:
                particle_best[i] = particles[i].copy()
                particle_best_c5[i] = c5_current
                
                # Update global best if improved
                if c5_current < global_best_c5:
                    global_best = particles[i].copy()
                    global_best_c5 = c5_current
        
        # Update particles
        for i in range(num_particles):
            # Update velocity
            r1 = np.random.random(n_points)
            r2 = np.random.random(n_points)
            
            cognitive = cognitive_weight * r1 * (particle_best[i] - particles[i])
            social = social_weight * r2 * (global_best - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible set
            particles[i] = project_to_feasible(particles[i], n_points)
        
        # Apply perturbation occasionally
        if iter_count % perturbation_interval == 0 and iter_count > 0:
            for i in range(num_particles):
                if np.random.random() < perturbation_rate:
                    noise = np.random.normal(0, 0.01, n_points)
                    particles[i] = particles[i] + noise
                    particles[i] = project_to_feasible(particles[i], n_points)
        
        iter_count += 1
    
    # Final refinement of the best solution
    refined_h = project_to_feasible(global_best, n_points)
    final_c5 = evaluate_erdos_solution(refined_h, dx)
    
    return (list(refined_h), final_c5, n_points)
