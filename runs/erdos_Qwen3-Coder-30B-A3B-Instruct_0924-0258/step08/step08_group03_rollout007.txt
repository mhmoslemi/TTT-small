import numpy as np
from time import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search on C5
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    # Function to test feasibility for a given C5
    def is_feasible(c5):
        # Create variables
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Sample shifts for practical feasibility check
        shifts = np.random.choice(range(1, n_points), size=min(100, n_points-1), replace=False)
        
        # Add overlap constraints for sampled shifts
        for k in shifts:
            # For circular correlation
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h_vars[i] * (1 - h_vars[j])
            constraints.append(overlap <= c5 * dx)
        
        # Create and solve problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INACCURACY
        except:
            return False
    
    # Binary search
    while high - low > tolerance and time() - start_time < budget_s * 0.9:
        mid = (low + high) / 2
        if is_feasible(mid):
            high = mid
            best_c5 = mid
        else:
            low = mid
    
    # Local refinement with gradient descent
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def gradient(h_vec):
        # Approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad
    
    h = best_h.copy()
    step_size = 0.01
    max_iter = 500
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = gradient(h)
        
        # Update with gradient descent
        h_new = h - step_size * grad
        
        # Clip to [0, 1] and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = objective(h_new)
        
        # Accept if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            step_size *= 0.95  # Reduce step size
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
