import numpy as np
from scipy.optimize import minimize
from time import time
import math

def project(h, n_points):
    """Project h onto feasible set: [0,1]^n and sum = n/2"""
    h_clipped = np.clip(h, 0, 1)
    if np.sum(h_clipped) == 0:
        return h_clipped
    h_scaled = h_clipped * (n_points / 2.0) / np.sum(h_clipped)
    return h_scaled

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def compute_gradient(h, n_points):
    """Compute gradient for current maximum overlap"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_lag = max_idx - (n_points - 1)
    
    # Create shifted version of (1-h) 
    shifted = np.roll(1 - h, max_lag)
    
    # Gradient according to chain rule
    gradient = 2 * (1 - 2 * h) * shifted
    return gradient

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 64
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = project(h_start, n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    max_iter = 5000
    patience = 50
    no_improvement = 0
    step_size = 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlap and get max lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        # Record best
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement = 0 
        else:
            no_improvement += 1
            
        # Early stopping
        if no_improvement >= patience:
            break
            
        # Compute gradient
        grad = compute_gradient(h, n_points)
        
        # Update step
        h_new = h - step_size * grad
        
        # Project back
        h_new = project(h_new, n_points)
        
        # Accept if improved
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < max_overlap:
            h = h_new
        else:
            # Reduce step size
            step_size *= 0.5
            if step_size < 1e-8:
                break
    
    # Final refinement with scipy
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        # Use L-BFGS-B
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        res = minimize(objective, h, method='L-BFGS-B', 
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 1000, 'ftol': 1e-12})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final check of constraints
    assert abs(np.sum(best_h) - n_points / 2.0) < 1e-6
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    
    return (list(best_h), best_c5, n_points)
