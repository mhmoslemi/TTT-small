import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.random.rand(n_points)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Thresholding-based optimization
    threshold = 0.1
    alpha = 0.95
    max_iterations = 500
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        max_overlap = np.max(overlaps)
        k_max = np.argmax(overlaps)
        
        # Compute gradients for maximum overlap lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient[i] = -2 * dx * (1 - best_h[j] - best_h[i])
        
        # Apply thresholding to top gradient coordinates
        top_indices = np.argsort(np.abs(gradient))[::-1][:n_points//2]
        
        # Store current h
        h_old = best_h.copy()
        
        # Threshold the top indices
        for i in top_indices:
            if best_h[i] >= 1 - threshold:
                best_h[i] = 1
            elif best_h[i] <= threshold:
                best_h[i] = 0
            else:
                # Apply intermediate adjustment
                best_h[i] = np.clip(best_h[i], 0, 1)
        
        # Maintain sum constraint
        current_sum = np.sum(best_h)
        if abs(current_sum - n_points/2) > 1e-6:
            # Distribute adjustment evenly
            adjustment = (n_points/2 - current_sum) / n_points
            best_h += adjustment
            np.clip(best_h, 0, 1, out=best_h)
        
        # Check if best solution improved
        current_c5 = evaluate_erdos_solution(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            # Update the best solution only if better
            if abs(current_c5 - best_c5) > 1e-9:
                # Make a new copy to avoid reference issues
                best_h = best_h.copy()
        
        # Cool the threshold
        threshold = max(0.01, threshold * alpha)
    
    # Coordinate-wise refinement
    for _ in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Fix all except one coordinate and optimize that one
        for i in range(n_points):
            if time() - start_time > budget_s * 0.95:
                break
            # Store current state
            h_test = best_h.copy()
            
            # Function to optimize for given i
            def opt_func(x):
                h_test[i] = x
                # Re-normalize to satisfy sum constraint
                current_sum = np.sum(h_test)
                if abs(current_sum - n_points/2) > 1e-6:
                    # Distribute adjustment among all elements except i
                    diff = (n_points/2 - current_sum) / (n_points - 1)
                    for j in range(n_points):
                        if j != i:
                            h_test[j] += diff
                # Ensure values stay in [0,1]
                h_test = np.clip(h_test, 0, 1)
                return evaluate_erdos_solution(h_test, dx)
            
            # Golden section search on [0,1]
            a, b = 0.0, 1.0
            resphi = 2.0 - 0.5 * (1 + np.sqrt(5))
            resphi2 = 0.5 * (3 - np.sqrt(5))
            x1 = resphi * a + resphi2 * b
            x2 = resphi2 * a + resphi * b
            f1 = opt_func(x1)
            f2 = opt_func(x2)
            
            while abs(b - a) > 1e-6:
                if f1 < f2:
                    b = x2
                    x2 = x1
                    f2 = f1
                    x1 = resphi * a + resphi2 * b
                    f1 = opt_func(x1)
                else:
                    a = x1
                    x1 = x2
                    f1 = f2
                    x2 = resphi2 * a + resphi * b
                    f2 = opt_func(x2)
            
            # Take final point
            best_h[i] = (a + b) / 2
            
            # Re-normalize
            current_sum = np.sum(best_h)
            if abs(current_sum - n_points/2) > 1e-6:
                diff = (n_points/2 - current_sum) / (n_points - 1)
                for j in range(n_points):
                    if j != i:
                        best_h[j] += diff
            best_h = np.clip(best_h, 0, 1)
            
            # Check if improved
            current_c5 = evaluate_erdos_solution(best_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                
    # Final optimization with L-BFGS
    try:
        def objective(h):
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlaps)

        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use scipy minimize with L-BFGS-B
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if res.success:
            h_final = res.x
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
