import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm Setup
    pop_size = 50
    generations = 0
    max_generations = 1000
    no_improvement_count = 0
    max_no_improvement = 200
    
    # Initialize population
    population = []
    fitnesses = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        fitnesses.append(best_c5)
    
    # Fill remaining population with random valid solutions
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        population.append(h)
        fitnesses.append(evaluate_erdos_solution(h, dx))
    
    best_fitness = min(fitnesses)
    if best_fitness < best_c5:
        best_c5 = best_fitness
        best_h = population[np.argmin(fitnesses)].copy()
    
    # GA parameters
    crossover_rate = 0.8
    mutation_rate = 0.1
    mutation_step = 0.05
    tournament_size = 3
    
    while time() - start_time < budget_s * 0.95 and no_improvement_count < max_no_improvement:
        generations += 1
        
        # Evaluate current population
        current_fitnesses = []
        for h in population:
            f = evaluate_erdos_solution(h, dx)
            current_fitnesses.append(f)
        
        # Selection (tournament)
        selected_indices = []
        for _ in range(pop_size // 2):
            tournament = np.random.choice(pop_size, tournament_size, replace=False)
            winner = tournament[np.argmin([current_fitnesses[i] for i in tournament])]
            selected_indices.append(winner)
        
        # Generate offspring
        offspring = []
        for i in range(0, len(selected_indices), 2):
            idx1 = selected_indices[i]
            idx2 = selected_indices[i+1] if i+1 < len(selected_indices) else selected_indices[0]
            
            if np.random.rand() < crossover_rate:
                alpha = np.random.uniform(0.3, 0.7)
                child1 = alpha * population[idx1] + (1 - alpha) * population[idx2]
                child2 = (1 - alpha) * population[idx1] + alpha * population[idx2]
            else:
                child1 = population[idx1].copy()
                child2 = population[idx2].copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, mutation_step)
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, mutation_step)
            
            # Enforce bounds
            child1 = np.clip(child1, 0, 1)
            child2 = np.clip(child2, 0, 1)
            
            # Correct sum to match constraint
            sum1 = np.sum(child1)
            sum2 = np.sum(child2)
            child1 = child1 * (n_points / 2) / sum1
            child2 = child2 * (n_points / 2) / sum2
            
            offspring.extend([child1, child2])
        
        # Combine population
        combined_fitness = current_fitnesses + [evaluate_erdos_solution(h, dx) for h in offspring]
        combined_population = population + offspring
        
        # Select top individuals
        top_indices = np.argsort(combined_fitness)[:pop_size]
        population = [combined_population[i] for i in top_indices]
        fitnesses = [combined_fitness[i] for i in top_indices]
        
        # Update best solution
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:

                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 500, 'ftol': 1e-12})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
