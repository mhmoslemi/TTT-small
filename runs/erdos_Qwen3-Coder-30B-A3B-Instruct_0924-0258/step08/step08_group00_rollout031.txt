import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Annealing-inspired approach with multiple restarts
    max_iter = 2000
    T_start = 0.1
    T_decay = 0.995
    restarts = 30
    patience = 100
    
    for restart in range(restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        h = h_start.copy()
        h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
        
        T = T_start
        no_improvement = 0
        current_c5 = evaluate_erdos_solution(h, dx)
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Generate neighbor
            h_new = h.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.1, 0.1)
            h_new[idx] += delta
            
            # Maintain bounds and sum constraint
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate neighbor
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                h = h_new
                current_c5 = new_c5
                no_improvement = 0
            else:
                no_improvement += 1
                
            if no_improvement >= patience:
                break
                
            T *= T_decay
            
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final refinement with constrained optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
