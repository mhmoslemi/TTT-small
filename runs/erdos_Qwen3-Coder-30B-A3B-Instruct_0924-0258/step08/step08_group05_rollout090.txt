import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Population-based approach
    population_size = 50
    max_generations = 10000
    no_improvement_limit = 100
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a reasonable initial construction
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize population with feasible solutions
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
    
    # Fill population with diverse candidates
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        # Ensure values are in [0,1]
        h = np.clip(h, 0, 1)
        population.append(h.copy())
    
    # Evolution loop
    best_fitness = float('inf')
    best_individual = None
    no_improvement = 0
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitnesses.append(fit)
        
        # Track best
        min_fitness = min(fitnesses)
        if min_fitness < best_fitness:
            best_fitness = min_fitness
            best_individual = population[fitnesses.index(min_fitness)].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement > no_improvement_limit:
            break
            
        # Selection - tournament
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, size=5, replace=False)
            tournament_fitness = [fitnesses[i] for i in tournament]
            winner_idx = tournament[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = [best_individual.copy()]  # Elitism
        
        for i in range(0, population_size - 1, 2):
            parent1 = selected[i]
            parent2 = selected[i + 1] if i + 1 < len(selected) else selected[0]
            
            # Uniform crossover
            child1 = np.where(np.random.rand(n_points) > 0.5, parent1, parent2)
            child2 = np.where(np.random.rand(n_points) > 0.5, parent2, parent1)
            
            # Mutation
            for child in [child1, child2]:
                # Flip mutation (10% chance per element)
                flip_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
                for idx in flip_indices:
                    delta = np.random.choice([-0.1, 0.1])
                    child[idx] += delta
                    child[idx] = np.clip(child[idx], 0, 1)
                
                # Rescale to maintain sum constraint
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
            
            if len(new_population) >= population_size:
                break
        
        # Keep population size
        population = new_population[:population_size]
    
    # Final refinement
    if best_individual is not None:
        # Try to locally refine using scipy
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_individual, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-12})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_fitness:
                    best_fitness = c5_final
                    best_individual = h_final
        except Exception:
            pass
            
    return (list(best_individual), best_fitness, n_points)
