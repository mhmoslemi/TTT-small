import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    max_iter = 100000
    no_improvement = 0
    max_no_improvement = 5000
    
    def get_neighbor(h, temp_val):
        h_new = h.copy()
        i, j = random.sample(range(n_points), 2)
        delta = random.uniform(0.001, min(0.1, temp_val))
        if random.random() < 0.5:
            delta = -delta
        if h_new[i] + delta >= 0 and h_new[i] + delta <= 1:
            h_new[i] += delta
            h_new[j] -= delta
            # Ensure sum constraint
            sum_diff = np.sum(h_new) - n_points/2
            if abs(sum_diff) > 1e-6:
                # Adjust one more element
                idx = random.randint(0, n_points-1)
                h_new[idx] -= sum_diff
                h_new[idx] = np.clip(h_new[idx], 0, 1)
        return h_new
    
    # Main Simulated Annealing Loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break

        # Generate neighbor
        h_new = get_neighbor(h_current, temp)
        
        # Calculate energy
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Calculate energy difference
        delta = c5_new - best_c5
        
        # Accept or reject the new solution
        if delta < 0 or random.random() < np.exp(-delta / temp):
            h_current = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_current.copy()
                no_improvement = 0
            else:
                no_improvement += 1
        else:
            no_improvement += 1

        # Cool down
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp
        if no_improvement >= max_no_improvement:
            temp = max(temp * 1.1, 1e-3)
            no_improvement = 0
    
    # Final refinement using SLSQP
    try:
        from scipy.optimize import minimize
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
