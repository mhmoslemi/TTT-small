import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach
    resolutions = [35, 70, 140, 280]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to new resolution
        if res > n_points:
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            h_interp = np.interp(x_new, x_old, h_start)
            h = h_interp
        else:
            h = h_start[:res]
        
        # Ensure sum constraint
        h = h * (res/2) / np.sum(h)
        n_points = res
        dx = 2.0 / n_points
        
        # Simulated Annealing parameters
        temp = best_c5 * 0.1  # Initial temperature
        cooling_rate = 0.999
        min_temp = 1e-8
        max_iter = 2000
        
        current_h = h.copy()
        current_c5 = evaluate_erdos_solution(current_h, dx)
        
        patience = 100
        no_improvement = 0
        
        for iter in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb solution
            h_new = current_h.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, temp * 0.1)
            h_new[idx] += delta
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Compute new objective
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            if c5_new < current_c5:
                current_h = h_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                no_improvement = 0
            else:
                # Accept with probability
                delta_e = c5_new - current_c5
                if np.random.rand() < np.exp(-delta_e / temp):
                    current_h = h_new
                    current_c5 = c5_new
                no_improvement += 1
            
            # Cooling
            temp = max(min_temp, temp * cooling_rate)
            
            if no_improvement > patience:
                break
        
        h_start = current_h.copy()
    
    # Final optimization with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
