import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Binary search with CVXPY for feasibility
    low = 0.38
    high = 0.38085
    tolerance = 1e-5
    
    # Sample shifts for feasibility checking
    shift_samples = 100
    shifts = np.random.choice(n_points, shift_samples, replace=False)
    
    # Keep track of best solution in binary search
    best_feasible_h = None
    best_feasible_c5 = high
    
    while time() - start_time < budget_s * 0.95 and high - low > tolerance:
        mid = (low + high) / 2
        
        # Formulate CVXPY problem
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Add overlap constraints for sampled shifts
        for k in shifts:
            # Compute overlap constraint for shift k
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h_var[i] * (1 - h_var[j])
            constraints.append(overlap_expr <= mid * dx)
        
        # Solve feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                feasible_h = h_var.value
                if feasible_h is not None:
                    c5_val = evaluate_erdos_solution(feasible_h, dx)
                    if c5_val < best_feasible_c5:
                        best_feasible_c5 = c5_val
                        best_feasible_h = feasible_h.copy()
                    high = mid
                else:
                    low = mid
            else:
                low = mid
        except:
            low = mid
            
        if best_feasible_h is not None and evaluate_erdos_solution(best_feasible_h, dx) < best_c5:
            best_c5 = evaluate_erdos_solution(best_feasible_h, dx)
            best_h = best_feasible_h.copy()
    
    # Local refinement with gradient descent
    h_refined = best_h.copy()
    step_size = 0.1
    tolerance = 1e-8
    max_iter = 500
    
    for _ in range(max_iter):
        if time() - start_time > budget_s * 0.99:
            break
            
        # Gradient of overlap function for max contributing lag
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h_refined[i] - h_refined[j]
            else:
                grad[i] = 1 - 2 * h_refined[i]
        
        # Projected gradient descent step
        h_new = h_refined - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check for convergence
        if np.linalg.norm(h_new - h_refined) < tolerance:
            break
            
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_refined = h_new.copy()
        else:
            step_size *= 0.5
            if step_size < tolerance:
                break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
