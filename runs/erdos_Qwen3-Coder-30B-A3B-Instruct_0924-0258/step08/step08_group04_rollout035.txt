import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # PSO Parameters
    num_particles = 75
    inertia = 0.8
    inertia_decay = 0.999
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    perturbation_rate = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    particle_best = []
    particle_best_values = []
    
    for _ in range(num_particles):
        # Initialize each particle with a feasible solution
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        particle_best.append(h.copy())
        particle_best_values.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    global_best_idx = np.argmin(particle_best_values)
    global_best = particle_best[global_best_idx].copy()
    global_best_value = particle_best_values[global_best_idx]
    
    max_iter = int(1e6)  # Very large number, will be limited by time
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 5000
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        for i in range(num_particles):
            h = particles[i]
            v = velocities[i]
            
            # Compute fitness
            fitness = evaluate_erdos_solution(h, dx)
            
            # Update particle best
            if fitness < particle_best_values[i]:
                particle_best[i] = h.copy()
                particle_best_values[i] = fitness
            
            # Update global best
            if fitness < global_best_value:
                global_best = h.copy()
                global_best_value = fitness
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_weight * r1 * (particle_best[i] - h)
            social = social_weight * r2 * (global_best - h)
            
            v = inertia * v + cognitive + social
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            
            # Update position
            h_new = h + v
            
            # Project to feasible region
            h_new = np.clip(h_new, 0, 1)
            # Renormalize
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Apply perturbation occasionally
            if iter_count % 1000 == 0 and np.random.rand() < perturbation_rate:
                h_new += np.random.normal(0, 0.01, n_points)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Update particle
            particles[i] = h_new
            velocities[i] = v
            
            iter_count += 1
            
            if time() - start_time >= budget_s * 0.95:
                break
                
        # Update inertia
        inertia = max(0.4, inertia * inertia_decay)
        
        if time() - start_time >= budget_s * 0.95:
            break
            
        if no_improvement_count > max_no_improvement:
            break
    
    # Final local search on best solution
    best_h = global_best.copy()
    best_c5 = global_best_value
    dx = 2.0 / n_points
    
    # Simple local perturbation
    for _ in range(100):
        h_trial = best_h + np.random.normal(0, 0.001, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        c5_trial = evaluate_erdos_solution(h_trial, dx)
        if c5_trial < best_c5:
            best_h = h_trial
            best_c5 = c5_trial
    
    return (list(best_h), best_c5, n_points)
