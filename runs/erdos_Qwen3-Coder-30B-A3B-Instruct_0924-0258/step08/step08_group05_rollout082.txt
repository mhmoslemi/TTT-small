import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    # Ensure sum constraint
    dx = 2.0 / n_points
    h_current = h_start * (n_points/2) / np.sum(h_start)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    T = 0.1
    cooling_rate = 0.999
    min_T = 1e-6
    max_iter = int(1e6)
    no_improvement_limit = 1000
    no_improvement_count = 0
    
    # Initial energy calculation
    current_energy = evaluate_erdos_solution(h_current, dx)
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by perturbing two random indices
        idx1 = random.randint(0, n_points - 1)
        idx2 = random.randint(0, n_points - 1)
        
        # Ensure different indices
        while idx2 == idx1:
            idx2 = random.randint(0, n_points - 1)
            
        # Determine max delta
        max_delta = min(h_current[idx1], 1 - h_current[idx2])
        if max_delta <= 0:
            continue
            
        delta = random.uniform(0, max_delta)
        
        # Create candidate solution
        h_candidate = h_current.copy()
        h_candidate[idx1] -= delta
        h_candidate[idx2] += delta
        
        # Apply sum constraint
        sum_diff = np.sum(h_candidate) - n_points / 2
        if abs(sum_diff) > 1e-10:
            # Adjust to satisfy constraint
            if sum_diff > 0:
                # Need to decrease sum
                h_candidate -= sum_diff / n_points
            else:
                # Need to increase sum
                h_candidate += (-sum_diff) / n_points
                
        # Clamp to [0,1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Check if we made it feasible
        if abs(np.sum(h_candidate) - n_points/2) > 1e-6:
            continue
        
        # Compute energy of candidate
        candidate_energy = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        if candidate_energy < current_energy:
            h_current = h_candidate
            current_energy = candidate_energy
            if candidate_energy < best_c5:
                best_c5 = candidate_energy
                best_h = h_candidate.copy()
            no_improvement_count = 0
        else:
            # Accept with probability
            try:
                if random.random() < np.exp(-(candidate_energy - current_energy) / T):
                    h_current = h_candidate
                    current_energy = candidate_energy
                    no_improvement_count = 0
            except:
                # Prevent overflow issues
                pass
            
        # Cooling
        T *= cooling_rate
        if T < min_T:
            T = min_T
            
        no_improvement_count += 1
        if no_improvement_count > no_improvement_limit:
            T = min(1.0, T * 1.1)
            no_improvement_count = 0
            
    return (list(best_h), best_c5, n_points)
