import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Thresholding-based optimization
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    # Parameters
    threshold = 0.1
    alpha = 0.95
    max_iterations = 500
    refinements = 100
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps and find maximum
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for each index
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Sort by gradient magnitude and threshold top 50%
        sorted_indices = np.argsort(np.abs(gradient))[::-1]
        top_indices = sorted_indices[:max(1, n_points // 2)]
        
        # Thresholding step
        new_h = h.copy()
        for i in top_indices:
            if h[i] >= 1 - threshold:
                new_h[i] = 1
            elif h[i] <= threshold:
                new_h[i] = 0
            else:
                new_h[i] = h[i]
        
        # Maintain sum constraint
        current_sum = np.sum(new_h)
        adjustment = (n_points/2 - current_sum) / max(1, n_points - np.sum((new_h >= 1 - threshold) | (new_h <= threshold)))
        new_h = np.clip(new_h + adjustment, 0, 1)
        
        # Ensure sum is exactly right
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        h = new_h
        current_c5 = evaluate_erdos_solution(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Adaptive threshold cooling
        threshold = max(0.01, threshold * alpha)
    
    # Coordinate-wise refinement
    for _ in range(refinements):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Try to optimize each point individually
        h_refined = best_h.copy()
        for i in range(n_points):
            # Golden section search in the neighborhood of current value
            def objective(val):
                temp_h = h_refined.copy()
                temp_h[i] = val
                temp_h = temp_h * (n_points/2) / np.sum(temp_h)
                return evaluate_erdos_solution(temp_h, dx)
                
            # Bound search to [0,1]
            a, b = 0.0, 1.0
            res = minimize(objective, h_refined[i], bounds=[(a,b)], method='bounded')
            if res.success:
                h_refined[i] = res.x[0]
                
        # Re-normalize
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        # Check improvement
        refined_c5 = evaluate_erdos_solution(h_refined, dx)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = h_refined.copy()
            
    # Final refinement using scipy
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if res.success:
            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
