import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_init = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter_sa = int(1e5)
    
    # Initialize cooling schedule
    T = T_init
    num_sa_iter = 0
    
    # Initial normalization
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    
    # Annealing loop
    while T > T_min and time() - start_time < budget_s * 0.9:
        # Perturbation: modify two random indices
        i, j = random.sample(range(n_points), 2)
        delta = np.random.uniform(-0.1, 0.1)
        
        # Ensure constraint preservation by adjusting both values
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clip to bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize to enforce sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)

        new_c5 = evaluate_erdos_solution(h_new, dx)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new.copy()

            delta_c5 = new_c5 - best_c5
            if random.random() < np.exp(-delta_c5 / T):
                h = h_new.copy()

        T = T * alpha

    # Local refinement with L-BFGS-B

        bounds = [(0, 1) for _ in range(n_points)]

            return evaluate_erdos_solution(h, dx)
        
        # Use scipy minimize with L-BFGS-B
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,

        )

            refined_h = res.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
