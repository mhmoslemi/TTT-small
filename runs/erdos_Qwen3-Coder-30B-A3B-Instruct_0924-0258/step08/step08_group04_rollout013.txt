import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # PSO parameters
    num_particles = 75
    inertia = 0.8
    cognitive_weight = 0.5
    social_weight = 1.2
    max_velocity = 0.1
    perturbation_rate = 0.01
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    particle_best = []
    particle_best_values = []
    
    # Create initial population
    for i in range(num_particles):
        # Use some randomness but respect constraints
        h = h_current.copy()
        if i > 0:
            # Add some perturbation to the base solution
            perturb_mag = 0.1
            noise = np.random.normal(0, perturb_mag, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            # Ensure sum constraint
            h = h * (n_points/2) / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        particle_best.append(h.copy())
        particle_best_values.append(evaluate_erdos_solution(h, dx))
    
    # Global best tracking
    global_best_idx = np.argmin(particle_best_values)
    global_best = particle_best[global_best_idx].copy()
    global_best_value = particle_best_values[global_best_idx]
    
    # PSO iteration loop
    iter_count = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Check for time limit
        if time() - start_time > budget_s * 0.94:
            break
            
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            inertia_term = inertia * velocities[i]
            cognitive_term = cognitive_weight * r1 * (particle_best[i] - particles[i])
            social_term = social_weight * r2 * (global_best - particles[i])
            velocities[i] = inertia_term + cognitive_term + social_term
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Projection to feasible set
            particles[i] = np.clip(particles[i], 0, 1)
            # Adjust to make sum = n_points/2 exactly
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate and update personal best
            current_value = evaluate_erdos_solution(particles[i], dx)
            if current_value < particle_best_values[i]:
                particle_best[i] = particles[i].copy()
                particle_best_values[i] = current_value
                
                # Update global best if needed
                if current_value < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = current_value
                    no_improvement_count = 0
        
        # Adaptive inertia decay
        inertia = 0.8 - (0.4 * (iter_count / max_iter))
        if inertia < 0.4:
            inertia = 0.4
        
        # Perturb some particles to escape local optima
        if iter_count % 1000 == 0 and iter_count > 0:
            for i in range(num_particles):
                if np.random.rand() < perturbation_rate:
                    noise = np.random.normal(0, 0.05, n_points)
                    particles[i] = particles[i] + noise
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
        
        iter_count += 1
        no_improvement_count += 1
    
    # Final refinement on the best particle
    final_h = global_best.copy()
    final_c5 = global_best_value
    
    return (list(final_h), final_c5, n_points)
