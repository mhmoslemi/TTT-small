import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 500
    inertia = 0.7
    phi1 = phi2 = 1.5
    v_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Add initial solution as first particle
    particles.append(h_start.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Generate other particles
    for _ in range(n_particles - 1):
        h_new = h_start + np.random.normal(0, 0.1, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        particles.append(h_new)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize personal bests
    p_best = [p.copy() for p in particles]
    p_best_vals = [evaluate_erdos_solution(p, dx) for p in particles]
    
    # Initialize global best
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    # PSO iteration
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            v_new = (inertia * velocities[i] + 
                    phi1 * r1 * (p_best[i] - particles[i]) + 
                    phi2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            v_new = np.clip(v_new, -v_max, v_max)
            
            # Update position
            particles[i] += v_new
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to maintain integral constraint
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if fitness < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = fitness
                
                # Update global best
                if fitness < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = fitness
                    stagnation_count = 0
                else:
                    stagnation_count += 1
        
        # Check for stagnation
        if stagnation_count > 100:
            break
    
    # Final SLSQP refinement of best solution
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, g_best, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        
        if res.success:
            refined_h = res.x
            refined_c5 = objective(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
