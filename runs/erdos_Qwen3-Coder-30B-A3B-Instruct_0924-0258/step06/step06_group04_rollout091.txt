import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured starting point
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter_pso = 200
    w = 0.7  # inertia weight
    c1 = 1.4  # cognitive parameter
    c2 = 1.4  # social parameter
    v_max = 0.2  # maximum velocity
    
    # Initialize particles
    particles = [h_start.copy() for _ in range(n_particles)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
    personal_best = [h.copy() for h in particles]
    personal_best_c5 = [evaluate_erdos_solution(h, dx) for h in particles]
    global_best = best_h.copy()
    global_best_c5 = best_c5
    
    # PSO iterations
    for iter in range(max_iter_pso):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if c5 < personal_best_c5[i]:
                personal_best[i] = particles[i].copy()
                personal_best_c5[i] = c5
                
                # Update global best
                if c5 < global_best_c5:
                    global_best = particles[i].copy()
                    global_best_c5 = c5
                    if global_best_c5 < best_c5:
                        best_h = global_best.copy()
                        best_c5 = global_best_c5
    
    # SQP refinement using most critical constraints
    try:
        # Select critical lags based on the current best solution
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Get critical lags from correlation analysis of best solution
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Focus on the largest correlation values (positive lags only for efficiency)
        critical_lags = np.argsort(corr)[-10:][::-1]  # Top 10 lags
        # Later, check both positive and negative lags for better accuracy
        # For simplicity in few iterations, just deploy the set
        
        for k in critical_lags:
            if k >= n_points:
                k = k - n_points
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    
    except Exception as e:
        pass
    
    # Final normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
