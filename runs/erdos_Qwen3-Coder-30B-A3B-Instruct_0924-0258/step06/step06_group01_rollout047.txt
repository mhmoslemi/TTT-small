import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.random.rand(n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    T_min = 1e-5
    alpha = 0.999
    max_iter_per_T = 100
    
    T = T0
    h_current = h_start.copy()
    current_c5 = best_c5
    
    # Annealing loop
    while T > T_min and time() - start_time < budget_s * 0.95:
        for _ in range(max_iter_per_T):
            # Generate neighbor solution
            h_new = h_current.copy()
            
            # Select two random indices
            i, j = random.sample(range(n_points), 2)
            
            # Perturb h[i] and adjust h[j] to maintain sum constraint
            delta = random.uniform(-0.1, 0.1)
            old_i = h_new[i]
            h_new[i] = h_new[i] + delta
            h_new[j] = h_new[j] - delta * (h_new[i] - old_i) / (h_new[i] - old_i) if (h_new[i] - old_i) != 0 else 0
            
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Adjust to maintain sum constraint
            total = np.sum(h_new)
            scale_factor = (n_points / 2) / total
            h_new = h_new * scale_factor
            
            # Evaluate new solution
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject the new solution
            delta_c5 = new_c5 - current_c5
            if delta_c5 < 0 or np.exp(-delta_c5 / T) > random.random():
                h_current = h_new.copy()
                current_c5 = new_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
        
        # Decrease temperature
        T *= alpha
    
    # Final refinement with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
