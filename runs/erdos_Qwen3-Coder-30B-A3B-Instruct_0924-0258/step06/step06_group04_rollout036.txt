import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Grid refinement
    n_levels = 5
    for level in range(n_levels):
        if time() - start_time > budget_s * 0.95:
            break
            
        current_n = min(10 * (2 ** level), 35)
        if current_n < 10:
            continue
            
        # Initialize from previous solution or sinusoidal pattern
        if level == 0:
            x = np.linspace(0, 2, current_n)
            h_init = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / current_n))
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (current_n / 2) / np.sum(h_init)
        else:
            # Interpolate from previous resolution and refine
            old_h = best_h
            old_n = len(old_h)
            indices = np.linspace(0, old_n-1, current_n)
            h_init = np.interp(indices, np.arange(old_n), old_h)
            # Add some noise
            noise_factor = 0.05
            noise = np.random.normal(0, noise_factor, current_n)
            h_init = h_init + noise
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (current_n / 2) / np.sum(h_init)
            
        h_current = h_init.copy()
        current_c5 = evaluate_erdos_solution(h_current, 2.0/current_n)
        
        # Simulated annealing
        for iteration in range(200):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Create neighbor
            noise = np.random.normal(0, 0.05, current_n)
            h_new = h_current + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (current_n / 2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, 2.0/current_n)
            
            if c5_new < current_c5:
                h_current = h_new
                current_c5 = c5_new
            elif np.random.random() < np.exp(-(c5_new - current_c5) / (0.1 * max(1e-6, current_c5))):
                h_current = h_new
                current_c5 = c5_new
                
        if current_c5 < best_c5:
            best_h = h_current
            best_c5 = current_c5
            
    # Binary search to tight bound
    if best_c5 < 0.3807:
        low_bound = 0.0
        high_bound = best_c5
        for _ in range(15):
            if time() - start_time > budget_s * 0.95:
                break
            candidate_bound = (low_bound + high_bound) / 2.0
            try:
                h_opt = cp.Variable(len(best_h))
                constraints = [
                    cp.sum(h_opt) == len(best_h) / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                corr = np.correlate(best_h, 1 - best_h, mode='full')
                max_corr = np.max(corr)
                
                # Add constraint on max correlation
                corr_sum = cp.sum(cp.multiply(h_opt[:len(best_h)-1], 1 - h_opt[1:]))
                constraints.append(corr_sum <= candidate_bound * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
                
                if prob.status == cp.OPTIMAL:
                    new_c5 = evaluate_erdos_solution(h_opt.value, dx)
                    if new_c5 < candidate_bound:
                        high_bound = candidate_bound
                        best_c5 = new_c5
                        best_h = h_opt.value
                else:
                    low_bound = candidate_bound
            except:
                low_bound = candidate_bound
                
    # Final check to ensure constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (len(best_h) / 2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, len(best_h))
