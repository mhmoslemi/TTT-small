import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n):
    """Create a triangular window centered at the middle"""
    x = np.arange(n)
    mid = (n - 1) / 2
    triangle = 1 - np.abs(x - mid) / (n / 2)
    return np.clip(triangle, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with either provided construction or triangular window
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = triangular_window(n_points)
    
    dx = 2.0 / n_points
    
    # Normalize the initial function
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # DE parameters
    popsize = 50
    maxiter = 300
    mutation_factor = 0.7
    crossover_rate = 0.9
    
    # Define bounds for DE
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE objective function that also ensures sum constraint
    def de_objective(h):
        # Normalize to ensure sum is n_points/2
        h_normalized = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h_normalized, dx)
    
    # Perform Differential Evolution
    de_result = differential_evolution(
        de_objective,
        bounds,
        maxiter=maxiter,
        popsize=popsize,
        mutation=(mutation_factor, 0.9),
        recombination=crossover_rate,
        seed=seed,
        callback=lambda x, convergence: None,
        disp=False
    )
    
    if de_result.success:
        h_de = de_result.x
        h_de = h_de * (n_points / 2) / np.sum(h_de)
        c5_de = evaluate_erdos_solution(h_de, dx)
        if c5_de < best_c5:
            best_h = h_de
            best_c5 = c5_de
    
    end_time = time()
    remaining_time = budget_s - (end_time - start_time)
    
    # Refinement with local optimization if time allows
    if remaining_time > 10 and best_c5 > 0.35:
        # Use L-BFGS-B for local refinement
        def objective(h):
            h_normalized = h * (n_points / 2) / np.sum(h)
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds_ref = [(0, 1) for _ in range(n_points)]
        
        # Try multiple starting points for refinement
        for _ in range(10):
            if time() - start_time > budget_s * 0.9:
                break
            # Generate a random perturbation of the best solution
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            try:
                res = minimize(objective, h_perturbed, method='L-BFGS-B', 
                              bounds=bounds_ref, constraints=cons_sum, 
                              options={'maxiter': 50, 'ftol': 1e-10})
                
                if res.success:
                    h_refined = res.x
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
