import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    n_points = 35
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
    else:
        population = [np.ones(n_points) * 0.5]
    
    # Generate diverse initial population
    while len(population) < 100:
        h = population[0].copy()
        noise = np.random.normal(0, 0.1, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # GA parameters
    pop_size = len(population)
    generations = 200
    elite_size = pop_size // 10
    tournament_size = pop_size // 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = [evaluate_erdos_solution(h, 2.0 / n_points) for h in population]
        
        # Track best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
        
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            candidates = random.sample(list(zip(population, fitness)), tournament_size)
            winner = min(candidates, key=lambda x: x[1])
            selected.append(winner[0])
        
        # Crossover and mutation
        new_population = []
        
        # Elite
        elites = [x[0] for x in sorted(zip(population, fitness), key=lambda x: x[1])[:elite_size]]
        new_population.extend(elites)
        
        # Generate offspring
        while len(new_population) < pop_size:
            h1, h2 = random.sample(selected, 2)
            
            if random.random() < crossover_rate:
                # Binomial crossover
                mask = np.random.random(n_points) < 0.5
                offspring = np.where(mask, h1, h2)
            else:
                offspring = h1.copy()
            
            # Mutation
            if random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                offspring = offspring + noise
                offspring = np.clip(offspring, 0, 1)
            
            # Normalize
            offspring = offspring * (n_points/2) / np.sum(offspring)
            new_population.append(offspring)
        
        population = new_population[:pop_size]
    
    # Local refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / n_points)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Refine top candidates
        fitness = [evaluate_erdos_solution(h, 2.0 / n_points) for h in population]
        top_indices = np.argsort(fitness)[:10]
        
        for i in top_indices:
            h_refined = population[i].copy()
            try:
                res = minimize(objective, h_refined, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 500, 'ftol': 1e-9})
                if res.success:
                    h_candidate = res.x
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
            except:
                pass
    except:
        pass
        
    return (list(best_h), best_c5, n_points)
