import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate diverse initial population
    def create_initial_population(n_points, n_pop=100):
        pop = []
        # Include the initial solution
        pop.append(h_start.copy())
        
        # Add random noisy versions
        for _ in range(n_pop - 1):
            pert = np.random.normal(0, 0.1, n_points)
            h = h_start + pert
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            pop.append(h)
        return pop
    
    # Define bounds for DE
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Differential Evolution optimization
    def de_objective(h):
        # Normalize the sum
        h = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    # Use differential evolution for global search
    de_result = differential_evolution(
        de_objective,
        bounds,
        maxiter=500,
        popsize=20,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        callback=lambda x, convergence: None
    )
    
    if de_result.success:
        h_de = de_result.x
        h_de = np.clip(h_de, 0, 1)
        h_de = h_de * (n_points/2) / np.sum(h_de)
        c5_de = de_objective(h_de)
        if c5_de < best_c5:
            best_h = h_de
            best_c5 = c5_de
    
    # Local refinement with SLSQP
    def refine_with_slsqp(h_initial):
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_initial, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            return res.x, objective(res.x)
        else:
            return h_initial, objective(h_initial)
    
    # Refine best solution from DE
    try:
        h_refined, c5_refined = refine_with_slsqp(best_h)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    except Exception:
        pass
    
    # Try multiple random restarts for SLSQP with different starting points
    num_restarts = 50
    for i in range(num_restarts):
        if time() - start_time > budget_s * 0.9:
            break
        h_restart = np.random.rand(n_points) * 0.5 + 0.25
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        try:
            h_local, c5_local = refine_with_slsqp(h_restart)
            if c5_local < best_c5:
                best_h = h_local
                best_c5 = c5_local
        except Exception:
            continue
    
    return (list(best_h), best_c5, n_points)
