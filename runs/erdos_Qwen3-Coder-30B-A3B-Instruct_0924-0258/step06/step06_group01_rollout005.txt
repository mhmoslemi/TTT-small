import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.random.rand(n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    rho = 1e-4
    max_iter = 1000
    tol = 1e-6
    iter_count = 0
    
    # Initialize variables
    h = h_start.copy()
    z = h.copy()
    u = np.zeros(n_points)
    
    # ADMM iterations
    while time() - start_time < budget_s * 0.8 and iter_count < max_iter:
        # Primal update: solve convex subproblem
        # minimize (rho/2)||z - h + u||_2^2 + (1/rho) * C5(h)
        # subject to sum(h) = n_points/2, 0 <= h <= 1
        
        # Create the problem using cvxpy
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(
            cp.sum_squares(h_var - z + u) / 2 + 
            (1.0 / rho) * evaluate_erdos_solution(h_var.value, dx) if iter_count == 0 else 0
        )
        constraints = [
            cp.sum(h_var) == n_points/2,
            h_var >= 0,
            h_var <= 1
        ]
        
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                
                # Update z and u
                z = h_new
                u = u + h - z
                
                # Update h
                h = z - (1.0 / rho) * (z - h + u)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                
                c5 = evaluate_erdos_solution(h, dx)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                
                # Check for convergence
                if np.linalg.norm(h - z) < tol:
                    break
        except:
            break
            
        iter_count += 1
    
    # Final refinement with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        
        if res.success:
            h_ref = res.x
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (n_points/2) / np.sum(h_ref)
            c5_ref = evaluate_erdos_solution(h_ref, dx)
            if c5_ref < best_c5:
                best_h = h_ref
                best_c5 = c5_ref
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
