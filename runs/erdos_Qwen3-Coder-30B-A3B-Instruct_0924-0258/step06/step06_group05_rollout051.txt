import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary algorithm with adaptive mutation
    population_size = 50
    generations = 1000
    mutation_rate = 0.05
    mutation_decay = 0.95
    min_mutation_rate = 0.01
    stagnation_limit = 100
    stagnation_count = 0
    best_c5_history = []
    
    # Initialize population
    population = [h_start.copy()]
    population_c5 = [best_c5]
    
    for _ in range(population_size - 1):
        h_noise = h_start + np.random.normal(0, 0.05, n_points)
        h_noise = np.clip(h_noise, 0, 1)
        h_noise = h_noise * (n_points/2) / np.sum(h_noise)
        population.append(h_noise)
        population_c5.append(evaluate_erdos_solution(h_noise, dx))
    
    # Evolution loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by C5
        sorted_indices = np.argsort(population_c5)
        population = [population[i] for i in sorted_indices]
        population_c5 = [population_c5[i] for i in sorted_indices]
        
        # Track best
        if population_c5[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = population_c5[0]
            stagnation_count = 0
            best_c5_history.append(best_c5)
        else:
            stagnation_count += 1
            
        # Check for stagnation
        if stagnation_count > stagnation_limit:
            # Increase mutation rate
            mutation_rate = max(min_mutation_rate, mutation_rate * mutation_decay)
            stagnation_count = 0
        
        # Select top 20% for breeding
        elite_count = max(1, population_size // 5)
        elite = population[:elite_count]
        
        # Generate offspring
        new_population = elite.copy()
        
        # Mutation with constraint-aware noise
        for _ in range(population_size - elite_count):
            parent = elite[np.random.randint(0, len(elite))]
            noise = np.random.normal(0, mutation_rate, n_points)
            h_mut = parent + noise
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * (n_points/2) / np.sum(h_mut)
            new_population.append(h_mut)
        
        # Update population
        population = new_population
        
        # Evaluate new population
        population_c5 = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            population_c5.append(c5)
        
        # Periodic convex relaxation refinement
        if gen % 100 == 0 and gen > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Use fewer k values to avoid numerical issues
                k_vals = np.linspace(0.05, 1.95, min(15, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if 0 < k < n_points:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    h_feasible = np.clip(h_feasible, 0, 1)
                    h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible.copy()
                        best_c5 = c5_feasible
            except Exception:
                pass
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
