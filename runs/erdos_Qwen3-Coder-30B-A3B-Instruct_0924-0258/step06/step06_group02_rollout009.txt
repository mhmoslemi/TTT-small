import numpy as np
from scipy import signal
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        return h
    scale = n / (2 * sum_h)
    h *= scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n = len(h)
    else:
        n = 70
        h = np.ones(n) * 0.5
    
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Annealing with focused perturbations
    T = 0.5
    a = 0.95
    max_steps = 2000
    step_count = 0
    
    while step_count < max_steps and time() - start_time < budget_s * 0.8:
        step_count += 1
        
        # Perturb top 15% indices with highest overlap
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Get positions where correlation is large
        high_indices = np.argsort(corr)[-int(0.15*n):]
        
        h_new = best_h.copy()
        # Perturb each of these indices
        for i in high_indices:
            if i < n:
                # Add noise to the perturbed indices
                noise = np.random.normal(0, 0.1 * T)
                h_new[i] += noise
        h_new = project_feasible(h_new, n)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Temperature decrease
        T = max(1e-4, T * a)
    
    # QCP Refinement with All Shifts
    try:
        h_opt = cp.Variable(n)
        constraints = [
            cp.sum(h_opt) == n / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraints for all shifts
        for k in range(1, n):
            corr_sum = cp.sum(cp.multiply(h_opt[:n-k], 1 - h_opt[k:]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    h_final = project_feasible(best_h, n)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), c5_final, n)
