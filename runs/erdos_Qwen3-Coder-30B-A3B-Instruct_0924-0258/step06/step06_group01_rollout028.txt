def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points/2 and h[i] in [0,1]"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)

    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.ones(n_points) * 0.5
    return h
