import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search approach
    h_binary = (h_start > 0.5).astype(int)
    sum_h = np.sum(h_binary)
    if abs(sum_h - n_points/2) > 1e-6:
        # Adjust to correct sum
        diff = int(n_points/2 - sum_h)
        if diff > 0:  # need more 1s
            zero_indices = np.where(h_binary == 0)[0]
            if len(zero_indices) >= diff:
                for i in range(diff):
                    h_binary[zero_indices[i]] = 1
        else:
            one_indices = np.where(h_binary == 1)[0]
            if len(one_indices) >= -diff:
                for i in range(-diff):
                    h_binary[one_indices[i]] = 0
        sum_h = int(n_points/2)
    
    def get_c5_binary(h_binary):
        h = h_binary.astype(float)
        h = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    def random_flip(h_binary):
        h_new = h_binary.copy()
        ones = np.where(h_new == 1)[0]
        zeros = np.where(h_new == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h_new[i] = 0
            h_new[j] = 1
        return h_new
    
    # Binary hill climbing
    temp = 1.0
    temp_decay = 0.995
    max_iter = 1000
    
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_new = random_flip(h_binary)
        c5_new = get_c5_binary(h_new)
        
        if c5_new < best_c5:
            h_binary = h_new
            best_c5 = c5_new
            best_h = h_binary.astype(float) * (n_points/2) / np.sum(h_binary)
        else:
            # Simulated annealing acceptance
            if np.random.random() < np.exp(-(c5_new - best_c5) / temp):
                h_binary = h_new
        temp *= temp_decay
        
        # Periodic QCP refinement
        if iter_count % 50 == 0 and iter_count > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample k values for constraints
                k_vals = np.linspace(1, n_points-1, min(20, n_points//3)).astype(int)
                for k in k_vals:
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass

    # Final refinement with local optimization
    try:
        h_final = best_h.copy()
        dx_final = 2.0 / n_points
        
        def objective(h):
            return evaluate_erdos_solution(h, dx_final)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
