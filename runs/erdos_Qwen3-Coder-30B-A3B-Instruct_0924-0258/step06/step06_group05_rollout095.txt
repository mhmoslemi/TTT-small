import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Use binary representation approach
    dx = 2.0 / n_points
    h_current = h_start.copy()
    current_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Binary representation approach
    h_binary = (h_current > 0.5).astype(int)
    ones_count = np.sum(h_binary)
    target_ones = n_points // 2
    if ones_count != target_ones:
        # Fix the count by flipping some bits
        diff = ones_count - target_ones
        if diff > 0:  # Too many 1s - flip some to 0
            zero_indices = np.where(h_binary == 0)[0]
            flip_indices = np.random.choice(zero_indices, diff, replace=False)
            h_binary[flip_indices] = 1
        else:  # Too few 1s - flip some to 1
            one_indices = np.where(h_binary == 1)[0]
            flip_indices = np.random.choice(one_indices, -diff, replace=False)
            h_binary[flip_indices] = 0
    
    # Set up temperature for annealing
    temp = 1.0
    temp_decay = 0.995
    min_temp = 0.0001
    
    # Search loop
    for iteration in range(1000):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Generate neighbor by swapping two bits
        zero_indices = np.where(h_binary == 0)[0]
        one_indices = np.where(h_binary == 1)[0]
        
        if len(zero_indices) > 0 and len(one_indices) > 0:
            i = np.random.choice(zero_indices)
            j = np.random.choice(one_indices)
            
            h_new = h_binary.copy()
            h_new[i] = 1
            h_new[j] = 0
            
            # Evaluate
            h_new_float = h_new.astype(float)
            c5_new = evaluate_erdos_solution(h_new_float, dx)
            
            # Accept or reject based on simulated annealing
            if c5_new < current_c5 or np.random.rand() < np.exp(-(c5_new - current_c5) / temp):
                h_binary = h_new
                current_c5 = c5_new
        
        # Periodic convex refinement (every 50 iterations)
        if iteration % 50 == 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample lags (reduce density)
                k_vals = np.linspace(0.1, 1.9, min(10, n_points//4))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= current_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=200)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < current_c5:
                        h_binary = (h_feasible > 0.5).astype(int)
                        current_c5 = c5_feasible
            except Exception:
                pass
                
        # Update temperature
        temp = max(temp * temp_decay, min_temp)
    
    # Final cleanup and return best
    h_final = h_binary.astype(float)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), c5_final, n_points)
