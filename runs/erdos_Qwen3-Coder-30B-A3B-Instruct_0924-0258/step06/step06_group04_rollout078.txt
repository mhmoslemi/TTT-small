```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with a sinusoidal pattern
    n_points = 35
    x = np.linspace(0, 2, n_points, endpoint=False)
    h_init = 0.5 + 0.5 * np.sin(2 * np.pi * x / 2)
    h_init = np.clip(h_init, 0, 1)
    h_init = h_init * (n_points/2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(h_init, 2.0/n_points)
    
    # Gradient-based refinement using scipy
    def objective(h):
        dx = 2.0 / n_points
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.concatenate([h, 1-h])
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # LBFGS optimization with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    for i in range(5000):
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, 
                              constraints=cons, options={'maxiter': 10})
            if result.success:
                h_new = result.x
                h_new = h_new * (n_points/2) / np.sum(h_new)
                h_new = np.clip(h_new, 0, 1)
                c5_new = evaluate_erdos_solution(h_new, 2.0/n_points)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
        except:
            pass
        
        # Perturbation step
        noise = np.random.normal(0, 0.01, n_points)
        h_new = best_h + noise
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        c5_new = evaluate_erdos_solution(h_new, 2.0/n_points)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Final refinement with a smaller step using CVXPY
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        dx = 2.0 / n_points
        # Use a subset of lags for smaller size problem
        for k in range(1, n_points, 3):
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-8)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-8, max_iters=2000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Ensure validity of the final solution
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, 2.0/n_points)
    
    return (list(best_h), final_c5, n_points)
```
