import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def check_feasibility(C_candidate, h_start, n, dx):
    # Randomly sample shifts
    k_samples = np.random.choice(range(1, n), size=max(1, int(n * 0.2)), replace=False)
    
    # Define decision variables
    h_opt = cp.Variable(n)
    
    # Integral constraint
    constraints = [cp.sum(h_opt) == n / 2]
    
    # Bounds
    constraints += [0 <= h_opt, h_opt <= 1]
    
    # Overlap constraints for sampled shifts
    for k in k_samples:
        if k < n:
            corr_sum = cp.sum(cp.multiply(h_opt[:n-k], 1 - h_opt[k:]))
        else:
            corr_sum = cp.sum(cp.multiply(h_opt[k-n+1:], 1 - h_opt[:n-(k-n+1)]))
        constraints.append(corr_sum <= C_candidate)
    
    # Formulate and solve problem
    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=cp.ECOS, eps=1e-6)
        return prob.status == cp.OPTIMAL
    except:
        return False

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search approach
    low = 0.38
    high = 0.38085
    iterations = 20
    sample_rate = 0.2
    
    # Look for minimal C5 using binary search
    for _ in range(iterations):
        if time() - start_time > budget_s * 0.9:
            break
            
        mid = (low + high) / 2
        if check_feasibility(mid, h_start, n_points, dx):
            best_c5 = mid
            high = mid
        else:
            low = mid
    
    # Use all constraints for final refinement
    try:
        h_final = cp.Variable(n_points)
        constraints = [
            cp.sum(h_final) == n_points / 2.0,
            h_final >= 0,
            h_final <= 1
        ]
        
        # Add overlap constraints with full shift set
        for k in range(1, n_points):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_final[:n_points-k], 1 - h_final[k:]))
            else:

            constraints.append(corr_sum <= best_c5)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_optimal = h_final.value
            c5_optimal = evaluate_erdos_solution(h_optimal, dx)
            if c5_optimal < best_c5:

    if sum_h > 0:
        h_final = h_final * (n_points / (2 * sum_h))
    final_c5 = evaluate_erdos_solution(h_final, dx)
