import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n):
    """Create a triangular window centered at the middle"""
    x = np.arange(n)
    mid = (n - 1) / 2
    triangle = 1 - np.abs(x - mid) / (n / 2)
    return np.clip(triangle, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = triangular_window(n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Normalize initial solution
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Objective function for differential evolution
    def de_objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    # Bounds for differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Differential Evolution
    try:
        de_result = differential_evolution(
            de_objective,
            bounds,
            maxiter=300,
            popsize=50,
            mutation=(0.5, 1.0),
            recombination=0.9,
            seed=seed,
            callback=lambda x, convergence: None,
            strategy='best1bin'
        )
        
        if de_result.success:
            h_de = de_result.x
            h_de = np.clip(h_de, 0, 1)
            h_de = h_de * (n_points / 2) / np.sum(h_de)
            c5_de = evaluate_erdos_solution(h_de, dx)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    except Exception:
        pass
    
    # Local refinement with L-BFGS-B
    def lbfgsb_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Run several L-BFGS-B refinements from top individuals
    for _ in range(5):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Random perturbation of best solution
        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            result = minimize(
                lbfgsb_objective,
                h_perturbed,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 200}
            )
            
            if result.success:
                h_refined = result.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            continue
    
    return (list(best_h), best_c5, n_points)
