from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    max_iters = 1000
    step_size = 0.01
    
    # Projected gradient descent
    for i in range(max_iters):

            break
            
        # Compute correlation and find max lag
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(best_h) - 1)
        
        # Compute gradient at the current best lag
        # This is simplified - we compute finite difference approximations
        gradient = np.zeros_like(best_h)
        eps = 1e-6
        for j in range(len(best_h)):
            h_plus = best_h.copy()
            h_minus = best_h.copy()
            h_plus[j] += eps
            h_minus[j] -= eps

            gradient[j] = (corr_plus - corr_minus) / (2 * eps)
        
        # Update with gradient descent
        new_h = best_h - step_size * gradient
        
        # Project to feasible region
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5

            return evaluate_erdos_solution(h, dx)

    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
