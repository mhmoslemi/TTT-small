import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a simple initialization
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization
    max_iterations = 2000
    learning_rate = 0.01
    decay = 0.999
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute gradient
        gradient = np.zeros(n_points)
        for k in range(1, n_points):
            # Compute contribution to gradient from shift k
            if k < n_points:
                # h[i] contributes to (1-h[i+k]) for i < n-k
                gradient[:n_points-k] += (1 - best_h[k:])
                # h[i] contributes to -h[i-k] for i >= k
                if k > 0:
                    gradient[k:] -= best_h[:n_points-k]
        
        # Update h using gradient descent
        h_new = best_h - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct integral
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Decay learning rate
        learning_rate *= decay
    
    # Final refinement with tiny perturbations to escape local optima
    for _ in range(50):
        if time() - start_time > budget_s * 0.95:
            break
        noise = np.random.normal(0, 1e-5, n_points)
        h_candidate = best_h + noise
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
    
    return (list(best_h), best_c5, n_points)
