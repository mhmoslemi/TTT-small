import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a more structured initialization for better starting point
        # Create a piecewise constant function with some randomness
        h_start = np.ones(n_points) * 0.5
        
        # perturb in a way that helps avoid plateaus
        for i in range(n_points):
            if i % 4 == 0:
                h_start[i] = 0.3
            elif i % 4 == 2:
                h_start[i] = 0.7
            else:
                h_start[i] = 0.5
        
        # Add small random noise for diversification
        noise = np.random.normal(0, 0.02, n_points)
        h_start += noise
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization using a simple gradient descent approach
    learning_rate = 0.01
    decay_factor = 0.9995
    max_iterations = 1000
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute gradient numerically
        gradient = np.zeros(n_points)
        # For each shift k, compute contribution to gradient
        for k in range(1, n_points):
            # valid indices for left part
            left_indices = slice(0, n_points - k)  
            # valid indices for right part
            right_indices = slice(k, n_points)  
            
            # contributions from h[i] * (1 - h[i+k])
            gradient[left_indices] += (1 - best_h[right_indices])
            
            # contributions from (1 - h[i]) * h[i+k]
            # which becomes -h[i+k] in d/dh[i] (since we're differentiating h[i])
            if k < n_points:  # avoid out of bounds
                gradient[right_indices] -= best_h[left_indices]
        
        # Update with gradient descent
        new_h = best_h - learning_rate * gradient
        
        # Ensure feasible values
        new_h = np.clip(new_h, 0, 1)
        
        # Normalize to sum constraint:
        # sum(h) = n_points / 2
        sum_h = np.sum(new_h)
        if sum_h > 0:
            new_h = new_h * (n_points / 2) / sum_h
            
        # Update learning rate
        learning_rate *= decay_factor
        
        # Evaluate new solution and update if improving
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
    
    # Final validation of the solution
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    sum_h = np.sum(final_h)
    if sum_h > 0:
        final_h = final_h * (n_points / 2) / sum_h
    
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
