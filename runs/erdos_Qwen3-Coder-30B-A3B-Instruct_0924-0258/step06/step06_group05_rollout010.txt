import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    T = 1.0  # initial temperature
    cooling_rate = 0.995
    
    # Initialize particles and velocities
    particles = np.random.uniform(0, 1, (n_particles, n_points))
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    
    # Normalize particles to satisfy constraint
    for i in range(n_particles):
        particles[i] *= (n_points / 2) / np.sum(particles[i])
    
    # Track best particles
    pbest_positions = particles.copy()
    pbest_values = np.array([evaluate_erdos_solution(p, dx) for p in particles])
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) +
                           c2 * r2 * (gbest_position - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair and normalize
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest_positions[i] = particles[i].copy()
                pbest_values[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_value:
                    gbest_position = particles[i].copy()
                    gbest_value = current_c5
        
        # Apply simulated annealing
        if np.random.rand() < np.exp(-gbest_value / T):
            # Perturb gbest slightly
            perturbation = np.random.normal(0, 0.02, n_points)
            new_gbest = gbest_position + perturbation
            new_gbest = np.clip(new_gbest, 0, 1)
            new_gbest *= (n_points / 2) / np.sum(new_gbest)
            new_c5 = evaluate_erdos_solution(new_gbest, dx)
            if new_c5 < gbest_value:
                gbest_position = new_gbest
                gbest_value = new_c5
                
        # Cool down temperature
        T *= cooling_rate
        
        # Convex relaxation every 50 iterations
        if iteration % 50 == 0 and iteration > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample k values for constraints
                k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)  # to index space
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= gbest_value * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < gbest_value:
                        gbest_position = h_feasible
                        gbest_value = c5_feasible
                        
            except Exception:
                pass
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, gbest_position, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < gbest_value:
                best_h = h_refined
                best_c5 = c5_refined
        else:
            best_h = gbest_position
            best_c5 = gbest_value
    except Exception:
        best_h = gbest_position
        best_c5 = gbest_value
    
    # Final verification
    if np.abs(np.sum(best_h) - n_points / 2.0) > 1e-6:
        best_h *= (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
