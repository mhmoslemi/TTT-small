import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)

    # SDP-based initialization
    try:
        h_sdp = cp.Variable(n_points)
        overlap_terms = [cp.sum(cp.multiply(h_sdp[:n_points-k], 1 - h_sdp[k:])) for k in range(1, n_points)]
        max_overlap = cp.max(cp.vstack([ot * dx for ot in overlap_terms]))

        constraints = [
            cp.sum(h_sdp) == n_points / 2.0,
            h_sdp >= 0,
            h_sdp <= 1
        ]

        prob = cp.Problem(cp.Minimize(max_overlap), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_init = h_sdp.value
            h_init = np.clip(h_init, 0, 1)
            sum_h = np.sum(h_init)
            if sum_h > 0:
                h_init = h_init * (n_points / (2 * sum_h))
            c5_init = evaluate_erdos_solution(h_init, dx)
            if c5_init < best_c5:
                best_h = h_init.copy()
                best_c5 = c5_init
    except Exception:
        pass

    # Trust Region Optimization
    current_tr_radius = 0.3
    lambda_merit = 0.1
    max_iterations = 3000

    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Perturb within trust region
        perturbation = np.random.uniform(-current_tr_radius, current_tr_radius, n_points)
        h_new = best_h + perturbation
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to integral constraint
        sum_h = np.sum(h_new)
        if sum_h > 0:
            h_new = h_new * (n_points / (2 * sum_h))
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Constraint violation
        constraint_violation = max(0, np.max(h_new - 1), np.min(h_new), abs(np.sum(h_new) - n_points/2))
        merit_new = c5_new + lambda_merit * constraint_violation
        
        merit_best = best_c5 + lambda_merit * max(0, np.max(best_h - 1), np.min(best_h), abs(np.sum(best_h) - n_points/2))
        
        if merit_new < merit_best:
            best_h = h_new.copy()
            best_c5 = c5_new
            current_tr_radius = min(0.5, current_tr_radius * 1.1)
        else:
            current_tr_radius = max(1e-6, current_tr_radius * 0.9)
        
        # Adjust merit weight
        if constraint_violation > 1e-4:
            lambda_merit *= 1.5
        else:
            lambda_merit *= 0.9

    # Final check with exact evaluation
    try:
        best_h = np.clip(best_h, 0, 1)
        sum_h = np.sum(best_h)
        if sum_h > 0:
            best_h = best_h * (n_points / (2 * sum_h))
        final_c5 = evaluate_erdos_solution(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
