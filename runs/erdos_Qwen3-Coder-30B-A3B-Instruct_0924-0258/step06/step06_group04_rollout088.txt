import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Grid refinement approach
    n_levels = 0
    max_levels = 5
    
    while n_levels < max_levels and (time() - start_time) < budget_s * 0.95:
        current_n = min(10 * (2 ** n_levels), 35)
        
        # Skip if already at this resolution
        if current_n == n_points:
            n_levels += 1
            continue
            
        # Initialize or interpolate from previous
        h_new = np.ones(current_n) * 0.5
        if n_points > 1:
            # Interpolate from existing
            old_x = np.linspace(0, 2, n_points)
            new_x = np.linspace(0, 2, current_n)
            h_new = np.interp(new_x, old_x, best_h)
            h_new = np.clip(h_new, 0, 1)
        
        # Sinusoidal initialization if not interpolateable
        if current_n == 10:
            x = np.linspace(0, 2, current_n)
            h_new = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / current_n))
            h_new = np.clip(h_new, 0, 1)
            
        h_new = h_new * (current_n / 2) / np.sum(h_new)
        h_opt = h_new.copy()
        
        # Local optimization with simulated annealing
        T = 0.1
        T_decay = 0.99
        max_iter = 100
        sa_freq = 0.1
        
        for i in range(max_iter):
            if (time() - start_time) > budget_s * 0.95:
                break
            
            # Gradient step
            perturbation = np.random.normal(0, 0.05, current_n)
            h_test = h_opt + perturbation
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (current_n / 2) / np.sum(h_test)
            
            c5_test = evaluate_erdos_solution(h_test, 2.0 / current_n)
            
            # Acceptance probability
            if c5_test < best_c5:
                h_opt = h_test
                best_c5 = c5_test
            else:
                accept_prob = np.exp(-(c5_test - best_c5) / T)
                if np.random.rand() < accept_prob:
                    h_opt = h_test
            
            # Simulated Annealing Step
            if np.random.rand() < sa_freq:
                noise = np.random.normal(0, T, current_n)
                h_sa = h_opt + noise
                h_sa = np.clip(h_sa, 0, 1)
                h_sa = h_sa * (current_n / 2) / np.sum(h_sa)
                c5_sa = evaluate_erdos_solution(h_sa, 2.0 / current_n)
                if c5_sa < best_c5:
                    h_opt = h_sa
                    best_c5 = c5_sa
            
            T = max(0.001, T * T_decay)
            
        # Final check and constraint normalization
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (current_n / 2) / np.sum(h_opt)
        c5_global = evaluate_erdos_solution(h_opt, 2.0 / current_n)
        
        if c5_global < best_c5:
            best_h = h_opt
            best_c5 = c5_global
            n_points = current_n
            
        n_levels += 1
    
    # Binary search refinement for final tight bound
    try:
        tol = 1e-6
        target_c5 = 0.3807
        low_c5 = 0.0
        high_c5 = best_c5
        
        while high_c5 - low_c5 > tol and (time() - start_time) < budget_s * 0.98:
            mid_c5 = (low_c5 + high_c5) / 2.0
            # Compute C5 with this constraint
            # Rather than solving a hard constraint, we just evaluate
            # A good SAA approximation should suffice for refinement
            
            # Fast trial update
            h_trial = best_h.copy()
            corr = np.correlate(h_trial, 1 - h_trial, mode='full')
            if np.max(corr) * (2.0 / n_points) <= mid_c5:
                high_c5 = mid_c5
            else:
                low_c5 = mid_c5
        
        final_c5 = high_c5
        
        # Do tighter evaluation
        final_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    except:
        final_c5 = best_c5
    
    return (list(best_h), final_c5, n_points)
