import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n_points):
    """Create a triangular window that peaks in the center"""
    half = n_points // 2
    window = np.zeros(n_points)
    for i in range(half + 1):
        window[i] = i / half
        window[n_points - 1 - i] = i / half
    return window

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with either parent construction or a triangular window
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = triangular_window(n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Normalize to sum to n_points/2
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Set up optimization bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        # Normalize to sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Differential Evolution parameters
    DE_bounds = [(0, 1) for _ in range(n_points)]
    
    # Run DE for exploration
    def de_objective(x):
        return objective(x)
    
    # Run DE for 60% of the budget
    de_budget = budget_s * 0.6
    de_start = time()
    
    # Initialize population with variations of the triangular window
    pop_init = []
    for i in range(20):
        h = h_start.copy()
        # Add some random noise
        noise = np.random.normal(0, 0.01, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        pop_init.append(h)
    
    # Run DE
    try:
        result = differential_evolution(
            func=de_objective,
            bounds=DE_bounds,
            maxiter=100,
            popsize=20,
            seed=seed,
            disp=False,
            polish=True
        )
        
        if result.success:
            h_de = result.x
            h_de = h_de * (n_points / 2) / np.sum(h_de)
            c5_de = evaluate_erdos_solution(h_de, dx)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    except:
        pass
    
    # Refinement with L-BFGS-B for the top individuals
    refine_budget = budget_s * 0.3
    refine_start = time()
    
    # Sample the best individuals found so far and refine them
    num_refinements = min(10, n_points)
    for _ in range(num_refinements):
        if time() - refine_start > refine_budget:
            break
            
        # Start with a slightly perturbed solution
        h_refine = best_h.copy()
        noise = np.random.normal(0, 0.005, n_points)
        h_refine += noise
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_points/2) / np.sum(h_refine)
        
        try:
            # Optimize by minimizing the objective with constraints
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            res = minimize(
                objective,
                h_refine,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if res.success:
                h_refined = res.x
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Final check at the best solution
    final_h = best_h.copy()
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
