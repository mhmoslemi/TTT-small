import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a construction inspired by symmetric and staggered patterns
        h_start = np.ones(n_points) * 0.5
        # Add some structured variation
        for i in range(n_points):
            h_start[i] = 0.5 + 0.2 * np.sin(2 * np.pi * i / n_points)
        # Normalize to sum to n/2
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive steps
    iterations = 1000
    for i in range(iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute gradient
        gradient = np.zeros(n_points)
        for k in range(1, n_points):
            # Forward correlations
            if k < n_points:
                gradient[:n_points-k] += (1 - best_h[k:])
            # Backward correlations
            if k < n_points:
                gradient[k:] -= best_h[:-k]
        
        # Update with adaptive learning rate
        lr = 0.01 * (0.99 ** i)
        new_h = best_h - lr * gradient
        
        # Project to feasible region
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(new_h, dx)
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
            
    # Final refinement
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
