import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different initializations and optimization strategies
    initial_strategies = []
    
    # Strategy 1: Periodic wave-like construction
    h_periodic = np.zeros(n_points)
    for i in range(n_points):
        h_periodic[i] = 0.5 + 0.3 * np.sin(2 * np.pi * i / n_points)
    h_periodic = np.clip(h_periodic, 0, 1)
    h_periodic = h_periodic * (n_points/2) / np.sum(h_periodic)
    initial_strategies.append(h_periodic)
    
    # Strategy 2: Piecewise constant with random perturbations
    h_piecewise = np.ones(n_points) * 0.5
    # Create some variation
    for i in range(0, n_points, 3):
        if i + 2 < n_points:
            # Randomly adjust small segments
            h_piecewise[i:i+3] = np.clip(h_piecewise[i:i+3] + np.random.normal(0, 0.05, 3), 0, 1)
    
    # Normalize
    h_piecewise = h_piecewise * (n_points/2) / np.sum(h_piecewise)
    initial_strategies.append(h_piecewise)
    
    # Strategy 3: Use the existing one
    initial_strategies.append(h_start.copy())
    
    # Try all strategies
    for h_init in initial_strategies:
        # Start with a normalized version
        h_current = h_init.copy()
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Gradient-based local optimization  
        max_restarts = 20
        restart_count = 0
        
        while time() - start_time < budget_s * 0.90 and restart_count < max_restarts:
            # Random perturbation
            h_restart = h_current + np.random.normal(0, 0.03, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
                
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 300, 'ftol': 1e-9})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        restart_count = 0
                    else:
                        restart_count += 1
                else:
                    restart_count += 1
            except Exception:
                restart_count += 1
    
    # Final refinement with maybe improvement
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
