import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a more structured initial guess
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 200
    inertia = 0.7
    cognitive = 1.4
    social = 1.4
    velocity_max = 0.2
    
    # Initialize particles
    particles = [h_start.copy() for _ in range(n_particles)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
    personal_best = [h.copy() for h in particles]
    personal_best_vals = [evaluate_erdos_solution(h, dx) for h in particles]
    global_best = best_h.copy()
    global_best_val = best_c5
    
    # PSO iterations
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample key lags for constraint adjustment
        if iter_num % 20 == 0:
            # Explore current best to find most critical lags
            corr_full = np.correlate(global_best, 1 - global_best, mode='full')
            lags = np.argmax(np.abs(corr_full))
            critical_lags = [i for i in range(1, n_points//2) if abs(corr_full[n_points-1+i]) > 0.8 * np.max(np.abs(corr_full))]
            critical_lags = critical_lags[:10]  # Keep top 10
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (personal_best[i] - particles[i]) + 
                            social * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to sum constraint
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            c5_val = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if c5_val < personal_best_vals[i]:
                personal_best[i] = particles[i].copy()
                personal_best_vals[i] = c5_val
                
                # Update global best
                if c5_val < global_best_val:
                    global_best = particles[i].copy()
                    global_best_val = c5_val
                    if global_best_val < best_c5:
                        best_h = global_best.copy()
                        best_c5 = global_best_val
    
    # Refine with SQP if time allows
    if time() - start_time < budget_s * 0.9:
        try:
            # Use a reduced set of constraints for efficiency
            n_constraints = min(20, n_points)
            indices = np.linspace(0, n_points-1, n_constraints, dtype=int)
            indices = list(indices) + [i + n_points for i in indices[:-1]]
            
            # Create and solve SQP problem
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Sample constraints based on the best solution so far
            corr_vals = np.correlate(global_best, 1 - global_best, mode='full')
            top_lags = np.argsort(np.abs(corr_vals))[-n_constraints:][::-1]
            
            for k in top_lags:
                if k <= n_points:
                    # Extract correlation for the specific lag
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points:], 1 - h_opt[:n_points-(k-n_points)]))
                constraints.append(corr_sum <= global_best_val * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception as e:
            pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
