import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        # Initialize with symmetric pattern
        h_start = np.zeros(n_points)
        centers = [0.6, 1.0, 1.4]
        spread = 0.4
        for c in centers:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - c))
            h_start[max(0, idx-3):min(n_points, idx+4)] = 1.0
        h_start = h_start * (n_points / (2 * np.sum(h_start)))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Trust Region Optimization with Adaptive Steps
    max_iterations = 1000
    tr_radius = 0.3
    lambda_merit = 0.1
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Generate perturbations within trust region
        perturbation = np.random.uniform(-tr_radius, tr_radius, n_points)
        h_new = best_h + perturbation
        
        # Project to feasible space
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / (2 * np.sum(h_new)))
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        constraint_violation = max(0, max(0, np.max(h_new - 1), np.min(h_new)), 
                                   abs(np.sum(h_new) - n_points / 2))
        
        # Merit function
        merit_new = c5_new + lambda_merit * constraint_violation
        merit_best = best_c5 + lambda_merit * max(0, max(0, np.max(best_h - 1), np.min(best_h)), 
                                                   abs(np.sum(best_h) - n_points / 2))
        
        # Accept or reject
        if merit_new < merit_best:
            best_h = h_new.copy()
            best_c5 = c5_new
            tr_radius = min(1.1 * tr_radius, 0.8)
        else:
            tr_radius = max(0.9 * tr_radius, 0.02)
            
        # Adjust lambda based on constraint satisfaction
        if constraint_violation > 1e-4:
            lambda_merit *= 1.2
        else:
            lambda_merit *= 0.9
    
    # Final correction and validation
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * (n_points / (2 * sum_h))
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
