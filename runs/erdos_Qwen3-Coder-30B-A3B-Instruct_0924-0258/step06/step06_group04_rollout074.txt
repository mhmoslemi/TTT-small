import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use adaptive grid sampling to build better initial solutions
    n_iter = 200
    max_iter = 1000
    
    # Create a better initial solution using simple heuristic
    h_heuristic = np.zeros(n_points)
    # Place a bump in the center with varying height
    center = n_points // 2
    width = n_points // 4
    for i in range(max(0, center - width), min(n_points, center + width)):
        h_heuristic[i] = 1.0
    h_heuristic = h_heuristic * (n_points/2) / np.sum(h_heuristic)
    
    h_current = h_heuristic
    for i in range(n_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Slightly perturb
        noise = np.random.normal(0, 0.02, n_points)
        h_new = h_current + noise
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h_current = h_new
    
    # Try a variation of a different algorithm at the end
    try:
        # Use DCP with constraint relaxation for final refinement
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Use a few key correlations in constraining
        for k in [1, 2, 3, 5, 7, 10]:
            if k < n_points:
                corr = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr <= best_c5 * dx * 1.01)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        # Try multiple solvers
        solvers = [cp.ECOS, cp.CBC, cp.CVXOPT]
        for solver in solvers:
            try:
                prob.solve(solver=solver, eps=1e-6, max_iters=300)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible.copy()
                        best_c5 = c5_feasible
                break
            except:
                continue
                
    except:
        pass
    
    # Final polishing with gradient based method
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def proj_h(h):
        return np.clip(h, 0, 1) * (n_points/2) / np.sum(h)
    
    # Gradient decent
    step_size = 0.1
    for i in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_new = best_h + np.random.normal(0, step_size, n_points)
        h_new = proj_h(h_new)
        c5_new = obj(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        step_size *= 0.99
    
    return (list(best_h), best_c5, n_points)
