import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        return h
    scale = n / (2 * sum_h)
    h *= scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with provided construction or create new
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Annealing with focused perturbations
    T = 0.5
    cooling_rate = 0.95
    max_iter = 2000
    min_iter_no_improvement = 100
    iter_no_improvement = 0
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Calculate overlap indices
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        max_indices = np.argsort(corr_full)[-int(0.15 * len(corr_full)):]
        
        # Generate perturbation
        h_new = best_h.copy()
        for idx in max_indices:
            # Convert full correlation index back to shift k
            if idx < n_points:
                k = idx
            else:
                k = idx - n_points + 1
                
            # Perturb only on relevant region
            if k < n_points:
                # Random perturbation
                delta = np.random.uniform(-0.1, 0.1)
                if 0 <= idx < len(h_new):
                    h_new[idx] += delta
        
        # Project to feasible region
        h_new = project_feasible(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # acceptance probability
        if c5_new < best_c5 or np.random.rand() < np.exp(-((c5_new - best_c5) / T)):
            best_h = h_new.copy()
            best_c5 = c5_new
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
            
        # Temperature update
        T *= cooling_rate
        
        # Check for early termination
        if iter_no_improvement >= min_iter_no_improvement:
            break
    
    # QCP Refinement with all shifts
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add all overlap constraint
        for k in range(1, n_points):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
                print('QCP refined!')
    except Exception as e:
        pass
    
    # Final projection and validation
    try:
        final_h = project_feasible(best_h, n_points)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    except:
        pass
    
    # Ensure final solution is valid
    final_h = project_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
