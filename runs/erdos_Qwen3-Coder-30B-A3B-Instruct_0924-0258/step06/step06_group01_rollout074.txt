import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    T_min = 1e-5
    alpha = 0.999
    iterations_per_T = 100
    
    T = T0
    current_h = h.copy()
    current_c5 = best_c5
    
    # SA loop
    while T > T_min and time() - start_time < budget_s * 0.95:
        for _ in range(iterations_per_T):
            # Generate neighbor
            h_new = current_h.copy()
            i, j = random.sample(range(n_points), 2)
            delta = random.uniform(-0.1, 0.1)
            h_new[i] += delta
            h_new[j] -= delta
            
            # Maintain sum constraint
            sum_diff = np.sum(h_new) - n_points/2
            if abs(sum_diff) > 1e-10:  # Only adjust if needed
                # Adjust the last element to satisfy sum constraint
                h_new[-1] -= sum_diff
                
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate new solution
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            delta_c5 = new_c5 - current_c5
            if delta_c5 < 0 or np.exp(-delta_c5 / T) > random.random():
                current_h = h_new.copy()
                current_c5 = new_c5
                
                # Update best
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
        
        # Cool down
        T *= alpha
    
    # Final refinement with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
