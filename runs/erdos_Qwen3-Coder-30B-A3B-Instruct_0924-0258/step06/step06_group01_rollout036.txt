import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
        h = h * (n_points/2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient-based optimization
    max_iter = 10000
    iter_count = 0
    last_improvement = 0
    tol = 1e-6
    
    # Use limited memory BFGS with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points/2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Prepare for projected gradient descent
    h_prev = h.copy()
    v = np.zeros_like(h)
    momentum = 0.9
    learning_rate = 0.05
    
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient numerically 
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            val_plus = evaluate_erdos_solution(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            val_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (val_plus - val_minus) / (2 * eps)
            
        # Update with momentum
        v = momentum * v - learning_rate * grad
        h_new = h + v
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check for improvement
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = iter_count
            
        # Update h
        h = h_new.copy()
        
        # Check convergence
        if iter_count - last_improvement > 100:
            break

    # Final refinement
    try:
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
