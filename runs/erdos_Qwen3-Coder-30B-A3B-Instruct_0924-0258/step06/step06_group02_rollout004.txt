from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    T = 0.5
    T_min = 1e-4
    cooling_rate = 0.99
    max_steps = 2000

    current_h = h_start.copy()
    current_c5 = best_c5

            break

        corr_full = np.correlate(current_h, 1 - current_h, mode='full')

        overlap_indices = np.argsort(corr_full)[-max(1, n_points // 10):]

        h_new = current_h.copy()

            idx = np.random.randint(0, n_points)
            if np.random.rand() < 0.7:  # 70% chance to perturb

                h_new[idx] += delta

        if perturbed > 0:
            h_new = h_new * (n_points / 2) / np.sum(h_new)

                # Compute gradient for overlap objective
                corr = np.correlate(h_refined, 1 - h_refined, mode='full')

                grad = np.zeros_like(h_refined)

        except Exception:

    return (list(best_h), best_c5, n_points)
