import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a pattern that has good spread and low overlap
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with momentum
    iterations = 1000
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    for i in range(iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute gradient using finite differences for better numerical stability
        gradient = np.zeros(n_points)
        for k in range(1, n_points):
            # For correlation involving shift k
            if k < n_points:
                # Gradient contribution from h[i] due to term h[i](1 - h[i+k])
                gradient[:n_points-k] += (1 - best_h[k:])
                # Gradient contribution from h[i+k] due to term h[i](1 - h[i+k]) 
                if k < n_points:
                    gradient[k:] -= best_h[:-k]
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * gradient
        best_h = best_h + velocity
        
        # Project to feasible region
        best_h = np.clip(best_h, 0, 1)
        sum_h = np.sum(best_h)
        if sum_h > 0:
            best_h = best_h * (n_points / 2) / sum_h
            
        # Evaluate
        c5_new = evaluate_erdos_solution(best_h, dx)
        if c5_new < best_c5:
            best_h = best_h.copy()
            best_c5 = c5_new
            
        # Adaptively decrease learning rate
        if i % 100 == 0:
            learning_rate *= 0.95
    
    # Final refinement with local search in neighborhood
    for _ in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_new = best_h + np.random.normal(0, 0.005, n_points)
        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)
        if sum_h > 0:
            h_new = h_new * (n_points / 2) / sum_h
            
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    return (list(best_h), best_c5, n_points)
