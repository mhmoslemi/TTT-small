import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Grid refinement
    resolutions = []
    n_current = 10
    while n_current <= 35 and time() - start_time < budget_s * 0.9:
        resolutions.append(n_current)
        n_current *= 2
    resolutions = resolutions[-1::-1]  # reverse for descending order
    
    for n in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize
        h_init = np.ones(n) * 0.5
        if n_points <= n:
            # Interpolate from previous solution
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n)
            h_init = np.interp(x_new, x_old, best_h)
        else:
            # Create sinusoidal
            x = np.linspace(0, 2, n)
            h_init = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2.0 / n))
        
        # Scale to sum = n/2
        h_init = h_init * (n/2) / np.sum(h_init)
        
        # Hybrid optimization
        h_current = h_init.copy()
        c5_current = evaluate_erdos_solution(h_current, 2.0/n)
        
        # Simulated Annealing with gradient steps
        step_size = 0.1
        temperature = 0.1
        decay_rate = 0.99
        max_iter = 100
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Gradient descent step
            # We compute gradient numerically
            epsilon = 1e-4
            grad = np.zeros(n)
            
            for j in range(n):
                h_plus = h_current.copy()
                h_plus[j] += epsilon
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n/2) / np.sum(h_plus)
                c5_plus = evaluate_erdos_solution(h_plus, 2.0/n)
                
                h_minus = h_current.copy()
                h_minus[j] -= epsilon
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n/2) / np.sum(h_minus)
                c5_minus = evaluate_erdos_solution(h_minus, 2.0/n)
                
                grad[j] = (c5_plus - c5_minus) / (2*epsilon)
            
            # Update with gradient
            h_new = h_current - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, 2.0/n)
            
            # SA acceptance
            if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / temperature):
                h_current = h_new
                c5_current = c5_new
            
            # Update best if improved
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
            
            # Cooling
            temperature *= decay_rate
            step_size *= decay_rate
        
        n_points = n
    
    # Final refinement with optimization
    try:
        # Define optimization variables with constraints
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraint for current bound
        corr = cp.conv(h_opt, 1 - h_opt)
        max_corr = cp.max(corr)
        constraints.append(max_corr <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception: 
        pass
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
