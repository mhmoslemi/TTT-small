import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based evolutionary algorithm
    population_size = 50
    mutation_rate = 0.05
    generations = 1000
    elite_size = population_size // 5
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Keep track of best individual across generations
    last_improvement = 0
    patience = 100
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        indices = np.argsort(fitness)
        population = [population[i] for i in indices]
        fitness = [fitness[i] for i in indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            last_improvement = gen
        
        # Check for stagnation
        if gen - last_improvement > patience:
            mutation_rate *= 0.95
            last_improvement = gen
            
        # Selection: keep elite
        new_population = population[:elite_size]
        
        # Generation of offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = population[np.random.randint(0, elite_size)]
            parent2 = population[np.random.randint(0, elite_size)]
            
            # Crossover (uniform)
            mask = np.random.rand(n_points) > 0.5
            child = np.where(mask, parent1, parent2)
            
            # Mutation
            if np.random.rand() < 0.8:  # 80% mutation rate
                noise = np.random.normal(0, mutation_rate, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                # Renormalize
                child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        
        # Occasionally apply convex relaxation
        if gen % 20 == 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                # Sample k values for constraints - use a subset to reduce density
                k_vals = np.linspace(0.05, 1.95, min(20, n_points//2))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
    
    # Final local refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
