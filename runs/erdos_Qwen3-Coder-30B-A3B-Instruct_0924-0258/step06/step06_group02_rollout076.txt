import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with focused perturbations
    T = 0.5
    T_min = 1e-4
    cooling_rate = 0.99
    steps = 2000
    
    for i in range(steps):
        if time() - start_time > budget_s * 0.80:
            break
            
        # Select indices with highest overlap contribution
        h_current = best_h.copy()
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Find peaks in correlation (used as guidance for perturbations)
        peak_indices = np.argsort(corr)[-int(n_points/10):]
        indices_to_change = set()
        for k in peak_indices:
            if k < n_points:
                indices_to_change.add(k)
            else:
                indices_to_change.add(k - n_points)
        
        # Perturb selected indices
        h_new = h_current.copy()
        for idx in indices_to_change:
            perturbation = np.random.uniform(-0.1, 0.1)
            h_new[idx] += perturbation
        
        # Keep within bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to maintain integral constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on simulated annealing
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Decay temperature
        T = max(T_min, T * cooling_rate)
    
    # Gradient-based refinement
    h_refined = best_h.copy()
    step_size = 1e-3
    
    # Projected gradient descent
    for _ in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient (approximate)
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            h_minus = h_refined.copy()
            h_minus[i] -= 1e-5
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2e-5)
        
        # Apply gradient step
        h_refined = h_refined - step_size * grad
        
        # Project to feasible region
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        # Evaluate new value
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
            
        step_size *= 0.99
    
    return (list(best_h), best_c5, n_points)
