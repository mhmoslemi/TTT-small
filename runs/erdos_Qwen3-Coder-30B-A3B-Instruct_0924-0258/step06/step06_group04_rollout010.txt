import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Grid refinement parameters
    n_start = 10
    n_max = 35
    n_points = n_start
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        dx = 2.0 / n_points
    else:
        h_start = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n_points))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive refinement with grid doubling
    n_levels = 0
    max_levels = 5
    
    while n_points <= n_max and n_levels < max_levels:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate previous solution or create new one
        if n_levels > 0:
            # Interpolate to new grid
            old_x = np.linspace(0, 2, len(best_h))
            new_x = np.linspace(0, 2, n_points)
            h_start = np.interp(new_x, old_x, best_h)
            h_start = np.clip(h_start, 0, 1)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
        else:
            # Sinusoidal initialization
            x = np.linspace(0, 2, n_points)
            h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n_points))
            h_start = np.clip(h_start, 0, 1)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        # Optimization with hybrid approach
        h = h_start.copy()
        dx = 2.0 / n_points
        current_c5 = evaluate_erdos_solution(h, dx)
        
        # Simulated Annealing with gradient step
        temperature = 0.1
        max_iter = 100
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Gradient-based step
            # Compute gradient direction 
            # We use finite differences for simplicity
            eps = 1e-5
            grad = np.zeros(n_points)
            base_c5 = current_c5
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                if h_plus[j] > 1: h_plus[j] = 1
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[j] -= eps
                if h_minus[j] < 0: h_minus[j] = 0
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                c5_minus = evaluate_erdos_solution(h_minus, dx)
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update based on gradient
            alpha = 0.1 * temperature
            h_new = h - alpha * grad
            
            # Project to feasible region
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # If it's better, accept it
            new_c5 = evaluate_erdos_solution(h_new, dx)
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
            else:
                # Accept with probability
                p_accept = np.exp(-(new_c5 - current_c5) / temperature)
                if np.random.random() < p_accept:
                    h = h_new
                    current_c5 = new_c5
                    
            # Decrease temperature
            temperature *= 0.99
            
        # Apply second order refinement: try to minimize using scipy
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        # Optimization with constraints
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        def constraint_bounds(h_vec):
            return np.concatenate([
                h_vec,
                1 - h_vec
            ])
        
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add a few key constraints from the correlation to improve solution
            corr_pairs = []
            step = max(1, n_points // 10)
            for k in range(1, n_points, step):
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                corr_pairs.append(corr_sum)
            
            # Add those as constraints
            for corr_sum in corr_pairs[: min(10, len(corr_pairs))]:
                constraints.append(corr_sum <= current_c5 * dx + 1e-6)
                
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < current_c5:
                    current_c5 = c5_feasible
                    h = h_feasible
        except Exception:
            pass
            
        # Check if this is the best solution so far
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        n_points = min(n_points * 2, n_max)
        n_levels += 1
    
    # Binary search refinement of the final solution
    dx_final = 2.0 / len(best_h)
    candidates = [best_c5]
    for i in range(20):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find a new solution that is possibly better
        h_perturbed = best_h + np.random.normal(0, 0.01, len(best_h))
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (len(best_h) / 2) / np.sum(h_perturbed)
        c5_perturbed = evaluate_erdos_solution(h_perturbed, dx_final)
        if c5_perturbed < best_c5:
            best_h = h_perturbed
            best_c5 = c5_perturbed
            candidates.append(best_c5)
    
    # Final evaluation to ensure we have right value
    final_c5 = evaluate_erdos_solution(best_h, dx_final)
    
    return (list(best_h), final_c5, len(best_h))
