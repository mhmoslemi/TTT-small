from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    # Improved Algorithm: Evolutionary Strategy with Local Refinement
    # Start with the known solution or random
    n_generations = 100
    population_size = 20
    mutation_rate = 0.1
    elite_size = 2
    
    # Generate initial population
    population = []
    for _ in range(population_size):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Add the best known starting point if available
    if 'initial_h_values' in globals():
        population[0] = h_start.copy()
    
    for gen in range(n_generations):
        if time() - start_time > budget_s * 0.95:

        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create next generation
        new_population = population[:elite_size]  # Elitism
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = np.random.choice(elite_size)
            parent2_idx = np.random.choice(elite_size)
            
            # Blend crossover
            alpha = np.random.random()
            child = alpha * population[parent1_idx] + (1 - alpha) * population[parent2_idx]
            
            # Mutation
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    child[i] += np.random.normal(0, 0.05)
            
            # Ensure bounds
            child = np.clip(child, 0, 1)
            # Renormalize
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Local refinement with trust-constr
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        # Refine solution with scipy's trust-constr
        result = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,

            bounds=bounds,

        )
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun
    except:

    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
