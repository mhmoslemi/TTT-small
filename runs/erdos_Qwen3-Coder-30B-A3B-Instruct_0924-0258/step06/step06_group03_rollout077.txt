import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Triangular initialization
        h_start = np.array([min(i, 2*n_points-i-2) for i in range(n_points)])
        h_start = h_start / np.sum(h_start) * (n_points/2)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Constraint for integral = 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Objective with penalty for constraint violation
    def constrained_objective(h):
        penalty = 1e6 * abs(constraint_sum(h))
        return objective(h) + penalty
    
    # Differential Evolution for global search
    de_res = differential_evolution(
        constrained_objective,
        bounds,
        maxiter=300,
        popsize=50,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        callback=lambda x, convergence: None,
        disp=False
    )
    
    if de_res.success:
        h_de = de_res.x
        c5_de = objective(h_de)
        if c5_de < best_c5:
            best_h = h_de
            best_c5 = c5_de
    
    # Local refinement with L-BFGS-B
    def lbfgs_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def lbfgs_constraint(h):
        return np.sum(h) - n_points / 2.0
    
    # Keep trying different valid refinements
    for _ in range(10):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Try to improve with different starting points
        h_trial = np.random.random(n_points) * 0.5
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        
        # Ensure it's feasible
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        
        try:
            res = minimize(
                lbfgs_objective,
                h_trial,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': lbfgs_constraint},
                options={'maxiter': 200}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = lbfgs_objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    
        except Exception:
            continue
    
    return (list(best_h), best_c5, n_points)
