import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def feasibility_fix(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points/2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 45
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization approach
    resolutions = [35, 40, 45, 50, 55, 60, 70]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = feasibility_fix(h_current, n_points)
        
        # SLSQP local optimization with multiple restarts
        max_restarts = 40
        restart_count = 0
        improvement_count = 0
        
        while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
            # Random perturbation with structured elements
            h_restart = h_current.copy()
            
            # Inject some structure
            if np.random.random() > 0.5 and len(h_restart) > 10:
                # Add some block-like structure
                block_size = max(2, len(h_restart) // 10)
                block_start = np.random.randint(0, len(h_restart) - block_size)
                block_type = np.random.choice([0, 1])
                h_restart[block_start:block_start+block_size] = 1.0 if block_type == 1 else 0.0
            
            # Add small random noise
            h_restart += np.random.normal(0, 0.02, n_points)
            h_restart = feasibility_fix(h_restart, n_points)
            
            # SLSQP local optimization
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 1000, 'ftol': 1e-9})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        restart_count = 0
                        improvement_count += 1
                    else:
                        restart_count += 1
                else:
                    restart_count += 1
            except Exception:
                restart_count += 1

            if improvement_count > 0 and improvement_count % 5 == 0 and np.random.random() > 0.7:

                h_alt = np.ones(n_points)
                if n_points >= 20:
                    h_alt[:n_points//3] = 0
                    h_alt[n_points//3:2*n_points//3] = 1
                    h_alt[2*n_points//3:] = 0
                    h_alt = feasibility_fix(h_alt, n_points)
                    c5_alt = evaluate_erdos_solution(h_alt, dx)
                    if c5_alt < best_c5:
                        best_h = h_alt
                        best_c5 = c5_alt
                        improvement_count = 0
            
            # If stuck for too long, add more randomness
            if restart_count > 15 and np.random.random() > 0.5:
                h_current = np.random.rand(n_points) * 0.5 + 0.25
                h_current = feasibility_fix(h_current, n_points)
                restart_count = 0

    try:
        # Final SLSQP refinement
        h_final = feasibility_fix(best_h.copy(), n_points)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
