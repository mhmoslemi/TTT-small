import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing approach
    max_iter = 5000
    temp = 1.0
    min_temp = 1e-6
    decay_rate = 0.995
    
    # Define frequency basis for sparse representation
    freqs = [1, 2, 3, 4, 5]
    n_freqs = len(freqs)
    
    # Initialize sparse representation
    coeffs = np.random.uniform(-0.5, 0.5, n_freqs * 2 + 1)  # a_f, b_f, c
    max_iter = 5000
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate new candidate via perturbation
        coeffs_new = coeffs.copy()
        idx = np.random.randint(0, len(coeffs_new))
        coeffs_new[idx] += np.random.normal(0, 0.05)
        
        # Reconstruct function
        h_new = np.zeros(n_points)
        x_vals = np.linspace(0, 2, n_points)
        for j, f in enumerate(freqs):
            h_new += coeffs_new[j] * np.sin(2 * np.pi * f * x_vals)
            h_new += coeffs_new[n_freqs + j] * np.cos(2 * np.pi * f * x_vals)
        h_new += coeffs_new[-1]  # constant term
        
        # Ensure non-negativity and normalization
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5:
            coeffs = coeffs_new
            best_h = h_new
            best_c5 = c5_new
        else:
            # Metropolis criterion
            if np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
                coeffs = coeffs_new
                best_h = h_new
                best_c5 = c5_new

        temp = max(min_temp, temp * decay_rate)
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
