import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier-based refinement approach
    max_iter = 500
    iter_count = 0
    
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Periodic initialization + Fourier filtering approach
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Apply FFT-based smoothing to reduce high-frequency components
        h_fft = np.fft.fft(h_current)
        
        # Create frequency domain window (Hamming window to dampen high frequencies)
        window = np.hamming(len(h_fft))
        h_filtered = h_fft * window
        
        # Inverse FFT to get refined function
        h_refined = np.real(np.fft.ifft(h_filtered))
        
        # Project back to [0,1] and normalize
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        # Local SLSQP optimization
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Perturb slightly before optimization
            h_perturbed = h_refined + np.random.normal(0, 0.005, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            
            res = minimize(objective, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            
            if res.success:
                h_opt = res.x
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
                    h_current = h_opt.copy()
            else:
                # Fall back to the filtering approach if optimization fails
                h_current = h_refined.copy()
        except Exception:
            # If optimization fails, fall back to filtered version
            h_current = h_refined.copy()
        
        iter_count += 1
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
