import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 500
    omega = 0.7
    phi1 = phi2 = 1.5
    v_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    # Seed particles around initial solution
    if 'initial_h_values' in globals():
        h_seed = h_start
    else:
        h_seed = np.ones(n_points) * 0.5
        
    for i in range(n_particles):
        if i == 0:
            h = h_seed.copy()
        else:
            # Add noise to initial solution
            h = h_seed + np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
        
        # Normalize
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        p_best.append(h.copy())
        p_best_vals.append(evaluate_erdos_solution(h, dx))
        
    # Global best
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    # PSO iterations
    stagnation_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = phi1 * r1 * (p_best[i] - particles[i])
            social = phi2 * r2 * (g_best - particles[i])
            velocities[i] = omega * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Ensure bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            current_val = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_val
                
                # Update global best
                if current_val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = current_val
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                    
        # Early termination if no improvement
        if stagnation_count > 100:
            break
    
    # Final refinement using SLSQP
    h_final = g_best.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
