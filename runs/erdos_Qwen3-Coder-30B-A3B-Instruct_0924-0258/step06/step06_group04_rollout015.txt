import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 200
    w = 0.7
    c1 = 1.4
    c2 = 1.4
    v_max = 0.2
    
    # Initialize particles
    particles = [h_start.copy() for _ in range(n_particles)]
    velocities = [np.random.uniform(-v_max, v_max, n_points) for _ in range(n_particles)]
    pbest = [h.copy() for h in particles]
    pbest_c5 = [evaluate_erdos_solution(h, dx) for h in particles]
    
    # Global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # Adaptive constraint sampling
    def get_critical_lags(h, num=10):
        corr = np.correlate(h, 1 - h, mode='full')
        # Only consider positive lags (first half)
        corr_positive = corr[len(corr)//2:]
        indices = np.argsort(corr_positive)[-num:][::-1]
        return indices
    
    # PSO iteration
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Get critical lags for current global best
        crit_lags = get_critical_lags(gbest, num=max(5, n_points//10))
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2), np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Clip and normalize
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            c5 = evaluate_erdos_solution(particles[i], dx)
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
            
            if c5 < gbest_c5:
                gbest = particles[i].copy()
                gbest_c5 = c5
        
        # Occasionally update list of critical lags
        if iter_count % 10 == 0:
            crit_lags = get_critical_lags(gbest, num=max(5, n_points//10))
    
    # Final SQP refinement with adaptive constraints
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add adaptive constraints based on critical lags
        crit_lags = get_critical_lags(gbest, num=max(5, n_points//5))
        for k in crit_lags:
            if k > 0:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt, 1 - h_opt))
            constraints.append(corr_sum <= gbest_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < gbest_c5:
                gbest = h_feasible
                gbest_c5 = c5_feasible
    except Exception:
        pass
    
    # Final check and normalization
    gbest = np.clip(gbest, 0, 1)
    gbest = gbest * (n_points/2) / np.sum(gbest)
    final_c5 = evaluate_erdos_solution(gbest, dx)
    
    return (list(gbest), final_c5, n_points)
