import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n_points))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search approach
    low, high = 0.3807, 0.38085
    tolerance = 1e-6
    max_iter = 50
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        c_candidate = (low + high) / 2.0
        
        # LP feasibility check for this candidate
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Sample lags adaptively
            sampled_lags = []
            for k in range(1, n_points // 2):
                sampled_lags.append(k)
            
            # Add lag constraints
            for k in sampled_lags:
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= c_candidate * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000, verbose=False)
            if prob.status == cp.OPTIMAL:
                # Refine solution with gradient descent
                h_feasible = h_opt.value
                h_feasible = np.clip(h_feasible, 0, 1)
                h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
                high = c_candidate
            else:
                low = c_candidate
        except Exception:
            low = c_candidate
    
    # Final local refinement using scipy optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    # Keep the constrained sum as a constraint
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0})
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'maxiter': 500, 'ftol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
