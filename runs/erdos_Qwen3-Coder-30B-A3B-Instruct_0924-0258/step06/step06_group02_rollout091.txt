import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def check_feasibility(C_candidate, n, dx, sample_rate=0.2):
    # Randomly sample shifts
    k_samples = np.random.choice(range(1, n), size=int(n * sample_rate), replace=False)
    
    # Define decision variables
    h_opt = cp.Variable(n)
    
    # Integral constraint
    constraints = [cp.sum(h_opt) == n / 2]
    
    # Bounds
    constraints += [0 <= h_opt, h_opt <= 1]
    
    # Overlap constraints for sampled shifts
    for k in k_samples:
        corr_sum = cp.sum(cp.multiply(h_opt[:n-k], 1 - h_opt[k:]))
        constraints.append(corr_sum <= C_candidate / dx)
    
    # Formulate and solve problem
    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=cp.ECOS, eps=1e-6)
        return prob.status == cp.OPTIMAL
    except:
        return False

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search approach
    low = 0.38
    high = 0.38085
    iterations = 20
    sample_rate = 0.2
    
    for _ in range(iterations):
        if time() - start_time > budget_s * 0.90:
            break
        mid = (low + high) / 2
        feasible = check_feasibility(mid, n_points, dx, sample_rate)
        if feasible:
            best_c5 = mid
            high = mid
        else:
            low = mid
    
    # Final optimization with all constraints
    try:
        h_final = cp.Variable(n_points)
        constraints = [
            cp.sum(h_final) == n_points / 2.0,
            h_final >= 0,
            h_final <= 1
        ]
        
        # Add all overlap constraints
        for k in range(1, n_points):
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_final[:n_points-k], 1 - h_final[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_final[k-n_points+1:], 1 - h_final[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_final_values = h_final.value
            final_c5 = evaluate_erdos_solution(h_final_values, dx)
            if final_c5 < best_c5:
                best_h = h_final_values
                best_c5 = final_c5
    except Exception:
        pass
    
    # Final refinement
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        final_c5 = evaluate_erdos_solution(h_refined, dx)
        if final_c5 < best_c5:
            best_h = h_refined
            best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
