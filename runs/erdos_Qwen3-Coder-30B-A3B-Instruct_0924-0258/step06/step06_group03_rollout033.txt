import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Structured initialization with triangular window
        h_start = np.array([min(i, n_points - i - 1) for i in range(n_points)])
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Differential Evolution settings
    maxiter = 300
    popsize = 50
    mutation = 0.7
    crossover = 0.9
    
    # DE optimization
    def de_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Run DE
    de_result = differential_evolution(
        de_objective,
        bounds,
        maxiter=maxiter,
        popsize=popsize,
        mutation=mutation,
        crossover=crossover,
        seed=seed,
        disp=False,
        atol=1e-8,
        rtol=1e-8
    )
    
    if de_result.success:
        h_de = de_result.x
        c5_de = de_result.fun
        if c5_de < best_c5:
            best_h = h_de
            best_c5 = c5_de
    
    if time() - start_time > budget_s * 0.8:
        return (list(best_h), best_c5, n_points)
    
    # Local refinement with L-BFGS-B
    def lbfgsb_refinement():
        refine_count = 0
        for _ in range(10):
            if time() - start_time > budget_s * 0.95:
                break
            try:
                # Random small perturbation to avoid local minima
                h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
                
                res = minimize(
                    de_objective,
                    h_perturbed,
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 100, 'ftol': 1e-10}
                )
                
                if res.success:
                    h_refined = res.x
                    c5_refined = de_objective(h_refined)
                    if c5_refined < best_c5:
                        best_h[:] = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
            refine_count += 1
            if refine_count >= 5:
                break
    
    lbfgsb_refinement()
    
    # Final validation check
    try:
        h_final = best_h.copy()
        h_final = h_final * (n_points/2) / np.sum(h_final)
        final_c5 = evaluate_erdos_solution(h_final, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
