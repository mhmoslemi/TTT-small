import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():

    else:

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based approach with projected updates
    max_iter = 10000
    lr = 0.1
    momentum = 0.9
    conv_radius = 0.01
    patience = 100
    patience_counter = 0
    
    h = h_start.copy()
    v = np.zeros_like(h)
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient approximations (simplified)
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update with momentum
        v = momentum * v - lr * grad
        h_new = h + v
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        h = h_new
        
        # Early stopping based on improvement
        if iteration > 50:
            if abs(current_c5 - best_c5) < 1e-6:
                patience_counter += 1
            else:
                patience_counter = 0
                
            if patience_counter >= patience:
                break
    
    # Ensure we return the best solution
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    # Final polishing with scipy
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 100}

            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            final_c5 = evaluate_erdos_solution(best_h, dx)
            best_c5 = final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
