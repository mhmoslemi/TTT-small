import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    inertia_decay = 0.995
    temperature = 1.0
    temp_decay = 0.995
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            h_init = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h_init = np.random.uniform(0, 1, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points/2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h_init.copy())
        pbest_values.append(evaluate_erdos_solution(h_init, dx))
    
    # Global best initialization
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Annealing schedule
        temperature *= temp_decay
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) + 
                             c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Clip to feasible region
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Repair to sum constraint
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            current_c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_c5
            
            # Simulated annealing acceptance
            if current_c5 > gbest_value:
                prob_accept = np.exp(-(current_c5 - gbest_value) / temperature)
                if np.random.random() < prob_accept:
                    pbest[i] = particles[i].copy()
                    pbest_values[i] = current_c5
            
            # Update inertia weight
            w *= inertia_decay
            
        # Periodic convex relaxation every 50 iterations
        if iteration % 50 == 0 and iteration > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample k values for constraints
                k_vals = np.random.choice(n_points, min(20, n_points//3), replace=False)
                for k in k_vals:
                    if k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= gbest_value * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=200)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    h_feasible = np.clip(h_feasible, 0, 1)
                    h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < gbest_value:
                        gbest = h_feasible.copy()
                        gbest_value = c5_feasible
            except Exception:
                pass
    
    # Refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, gbest, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < gbest_value:
                gbest = h_refined
                gbest_value = c5_refined
    except Exception:
        pass
    
    return (list(gbest), gbest_value, n_points)
