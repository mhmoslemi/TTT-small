import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def compute_gradients(h, k_max, n_points):
    grad = np.zeros(n_points)
    for i in range(n_points):
        if i + k_max < n_points:
            grad[i] += 1 - h[i + k_max]
        if i - k_max >= 0:
            grad[i] -= h[i - k_max]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with adaptive steps and random restarts
    max_iter = 2000
    step_size = 0.1
    noise_scale = 0.05
    patience = 100
    
    for idx in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all correlations
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient
        grad = compute_gradients(best_h, k_max, n_points)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
        
        # Line search
        alpha = step_size
        h_trial = best_h + alpha * grad
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        c5_trial = evaluate_erdos_solution(h_trial, dx)
        
        # Backtracking line search
        while c5_trial >= best_c5 and alpha > 1e-6:
            alpha *= 0.5
            h_trial = best_h + alpha * grad
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points/2) / np.sum(h_trial)
            c5_trial = evaluate_erdos_solution(h_trial, dx)
        
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            patience = 100
            
        # Random perturbations
        if idx % 20 == 0:
            noise = np.random.normal(0, noise_scale, n_points)
            h_new = best_h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                
        # Reduce noise scale
        noise_scale *= 0.995
        
        # Check for early termination
        patience -= 1
        if patience <= 0:
            break
    
    # Final convex relaxation with fewer constraints
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample subset of possible lags for constraints
        k_vals = np.arange(1, min(30, n_points))
        for k in k_vals:
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
