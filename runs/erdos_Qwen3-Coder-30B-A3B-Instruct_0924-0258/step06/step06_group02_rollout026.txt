import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Foveated Perturbations
    T = 0.5
    T_min = 1e-4
    cooling_rate = 0.99
    max_iter = 2000
    current_h = best_h.copy()
    current_c5 = best_c5
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Select indices for perturbation based on current overlap
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        # Find top 10% indices where overlap is highest
        indices = np.where(corr > np.percentile(corr, 90))[0]
        if len(indices) == 0:
            indices = np.arange(len(corr) // 2, len(corr))
        
        # Sample indices to perturb
        k = min(5, len(indices))
        idxs = np.random.choice(indices, k, replace=False)
        
        # Create perturbed version
        h_new = current_h.copy()
        for idx in idxs:
            perturbation = np.random.choice([-0.1, 0.1])
            h_new[idx] += perturbation
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct integral
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Compute new overlap
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept with probability based on Metropolis criterion
        if c5_new < current_c5 or np.random.rand() < np.exp(-(c5_new - current_c5) / T):
            current_h = h_new.copy()
            current_c5 = c5_new
            
        # Update best if necessary
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Cool down temperature
        if T > T_min:
            T *= cooling_rate
    
    # Gradient Descent Refinement
    try:
        # Convert to continuous variables for gradient-based optimization
        h_cont = best_h.copy()
        # Project back to valid solution
        h_cont = np.clip(h_cont, 0, 1)
        h_cont = h_cont * (n_points/2) / np.sum(h_cont)
        
        # Iterative refinement using scipy minimize with trust-constr
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Local refinement
        result = minimize(objective, h_cont, method='trust-constr', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'disp': False})
        
        if result.success:
            h_refined = result.x.copy()
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final validation
    try:
        final_h = best_h.copy()
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points/2) / np.sum(final_h)
        c5_final = evaluate_erdos_solution(final_h, dx)
        if c5_final < best_c5:
            best_h = final_h
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
