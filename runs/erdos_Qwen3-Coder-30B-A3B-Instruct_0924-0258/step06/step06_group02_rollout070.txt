import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use Halton sequence for low-discrepancy initialization
        def halton_sequence(n, base=2):
            sequence = []
            for i in range(n):
                x = 0
                f = 1.0 / base
                while i > 0:
                    x += (i % base) * f
                    i //= base
                    f /= base
                sequence.append(x)
            return np.array(sequence)
        
        h_start = halton_sequence(n_points)
        h_start = (h_start - np.min(h_start)) / (np.max(h_start) - np.min(h_start))
        h_start = h_start * 0.5  # Scale to average 0.5
        h_start = h_start * (n_points / 2) / np.sum(h_start)  # Ensure sum h = n/2
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with adaptive learning rate
    iterations = 2000
    learning_rate = 0.01
    
    for i in range(iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute gradient
        gradient = np.zeros(n_points)
        for k in range(1, n_points):
            # Forward difference for k-shifted correlations
            if k < n_points:
                gradient[:n_points-k] += (1 - best_h[k:])
            # Backward difference for k-shifted correlations
            if k < n_points:
                gradient[k:] -= best_h[:n_points-k]
        
        # Update with gradient
        best_h -= learning_rate * gradient
        best_h = np.clip(best_h, 0, 1)
        
        # Project to correct integral
        sum_h = np.sum(best_h)
        if sum_h > 0:
            best_h = best_h * (n_points / (2 * sum_h))
        
        # Evaluate and update best
        c5_new = evaluate_erdos_solution(best_h, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = best_h.copy()
            
        # Decaying learning rate
        learning_rate *= 0.999
    
    # Final check and refinement
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * (n_points / (2 * sum_h))
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
