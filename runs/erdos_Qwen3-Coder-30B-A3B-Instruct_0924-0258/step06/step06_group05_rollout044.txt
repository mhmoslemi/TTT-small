```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based descent with periodic random restarts
    max_iter = 3000
    step_size = 0.02
    noise_scale = 0.05
    noise_decay = 0.995
    noise_interval = 50
    conv_interval = 100
    
    h_current = best_h.copy()
    current_c5 = best_c5
    no_improvement_count = 0
    global_iteration = 0
    
    for idx in range(max_iter):
        global_iteration += 1
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all correlation values
        corr_full = np.correlate(h_current, 1 - h_current, mode='full')
        k_vals = np.arange(len(corr_full))
        k_half = len(k_vals) // 2
        
        # Find index of max overlap (k_max)
        max_idx = np.argmax(corr_full)
        k_max = k_vals[max_idx] - k_half
        
        # Compute gradient w.r.t. h for current k_max
        grad = np.zeros(n_points)
        if k_max > 0:
            # contribution from h[i] * (1 - h[i+k_max])
            grad[k_max:] += -h_current[:-k_max]
            grad[:-k_max] += 1 - h_current[k_max:]
        else:
            k_abs = -k_max
            grad[:k_abs] += 1 - h_current[k_abs:]
            grad[k_abs:] += -h_current[:-k_abs]
            
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
        
        # Line search for step size
        alpha = step_size
        h_new = h_current + alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # If not improving, try smaller step
        if c5_new >= current_c5:
            for _ in range(10):
                alpha *= 0.5
                h_test = h_current + alpha * grad
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                c5_test = evaluate_erdos_solution(h_test, dx)
                if c5_test < c5_new:
                    h_new = h_test
                    c5_new = c5_test
                    break
        
        # Accept new solution
        h_current = h_new
        current_c5 = c5_new
        
        if c5_new < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Random perturbation
        if idx % noise_interval == 0 and idx > 0:
            noise = np.random.normal(0, noise_scale, n_points)
            h_current += noise
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            noise_scale *= noise_decay
            
        # Convex relaxation every conv_interval iterations
        if idx % conv_interval == 0 and idx > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample some k values (avoid duplicating)
                k_vals_small = np.linspace(1, n_points//2, min(20, n_points//3))
                k_vals_small = k_vals_small.astype(int)
                k_vals_small = k_vals_small[k_vals_small > 0]
                
                for k_val in k_vals_small:
                    if k_val < n_points:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k_val], 1 - h_opt[k_val:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass

    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
```
