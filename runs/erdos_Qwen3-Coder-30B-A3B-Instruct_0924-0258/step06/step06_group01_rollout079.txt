import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        h_start[:n_points//2] = 0.8
        h_start[n_points//2:] = 0.2
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Iterative convex relaxation approach
    max_iterations = 500
    iteration = 0
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Compute overlap correlation for current solution
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_overlap_idx = np.argmax(corr)
        max_overlap_val = corr[max_overlap_idx] * dx
        
        # If we're already good enough, break early
        if max_overlap_val < best_c5:
            best_c5 = max_overlap_val
            best_h = best_h.copy()
        
        # Generate a quadratic approximation to relax the problem
        # Approximate the maximum as a quadratic in the variables
        # Using second-order Taylor approximation around current point
        
        # Construct QP: minimize 0.5 * x^T P x + q^T x
        # subject to Ax = b, lb <= x <= ub
        
        # Square of the constraint sum
        # We want sum(h) = n_points / 2
        
        # Linearize the maximum overlap contribution around current solution
        # This section is a simplified heuristic due to non-trivial nature
        # of creating the precise convex approximation
        
        # Direct optimization approach: project gradient ascent with constraint
        try:
            # Reformulate as constrained optimization over the feasible set
            def objective(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            # Use trust-constr or SLSQP for better handling of constraints
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
            
            res = minimize(
                objective,
                best_h.copy(),
                method='SLSQP',

                constraints=cons_sum,

            )
            
            if res.success and res.fun < best_c5:
                best_h = res.x.copy()
                best_c5 = res.fun
                
        except Exception as e:
            pass #
        
        # Alternative improvement: local search with perturbation
        try:
            # Small random perturbations on best solution
            h_pert = best_h + np.random.normal(0, 0.01, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            
            c5_trial = evaluate_erdos_solution(h_pert, dx)
            if c5_trial < best_c5:
                best_h = h_pert
                best_c5 = c5_trial
                
        except Exception as e:
            pass
        
        iteration += 1
        
    # Final projection to meet constraints exactly
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
