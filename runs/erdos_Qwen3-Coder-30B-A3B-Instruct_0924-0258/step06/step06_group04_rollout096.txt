import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Local Search
    max_iter = 10000
    temperature = 1.0
    min_temp = 1e-6
    decay_rate = 0.9995
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb
        h_new = best_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.1 * temperature)
        h_new[idx] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        elif np.random.rand() < np.exp(-(c5_new - best_c5) / temperature):
            best_h = h_new
            best_c5 = c5_new
            
        # Cool down
        temperature = max(min_temp, temperature * decay_rate)
    
    # Local refinement with gradient descent
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points/2
    
    def constraint_bounds(h):
        return np.concatenate([h, 1-h])
    
    # Use scipy.optimize with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
    ]
    
    try:
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-10})
        if result.success:
            h_refined = result.x
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
