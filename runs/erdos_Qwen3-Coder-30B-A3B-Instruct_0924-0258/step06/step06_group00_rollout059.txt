import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 500
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    
    # First particle is the initial solution
    h_init = h_start.copy()
    h_init = np.clip(h_init, 0, 1)
    h_init = h_init * (n_points/2) / np.sum(h_init)
    particles.append(h_init.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Generate rest of particles around the initial solution
    for i in range(1, n_particles):
        h = h_init + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        
    # Keep track of best positions
    personal_best_positions = [p.copy() for p in particles]
    personal_best_scores = [evaluate_erdos_solution(p, dx) for p in particles]
    global_best_idx = np.argmin(personal_best_scores)
    global_best_score = personal_best_scores[global_best_idx]
    global_best_position = personal_best_positions[global_best_idx].copy()
    
    # PSO iterations
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            v_new = (w * velocities[i] + 
                     c1 * r1 * (personal_best_positions[i] - particles[i]) +
                     c2 * r2 * (global_best_position - particles[i]))
            
            # Clip velocity
            v_new = np.clip(v_new, -v_max, v_max)
            
            # Update position
            particles[i] += v_new
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to sum constraint
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate fitness
            fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if fitness < personal_best_scores[i]:
                personal_best_scores[i] = fitness
                personal_best_positions[i] = particles[i].copy()
                
                # Update global best if needed
                if fitness < global_best_score:
                    global_best_score = fitness
                    global_best_position = particles[i].copy()
                    global_best_idx = i
                    
        # Check for convergence
        if iter_count > 100 and abs(global_best_score - best_c5) < 1e-8:
            break
        
        # Update the overall best
        if global_best_score < best_c5:
            best_c5 = global_best_score
            best_h = global_best_position.copy()
    
    # Final optimization with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
