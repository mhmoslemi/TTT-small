import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle swarm optimization with simulated annealing
    n_particles = 50
    inertia = 0.7
    c1, c2 = 2.0, 2.0
    max_iter = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_scores = []
    gbest = best_h.copy()
    gbest_score = best_c5
    
    # Initial particles with small Gaussian noise
    for _ in range(n_particles):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_scores.append(evaluate_erdos_solution(h, dx))

    # Main PSO loop
    temp = 1.0
    temp_decay = 0.995
    anneal_prob = 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Clip and repair
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            score = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                
                # Update global best
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    if gbest_score < best_c5:
                        best_h = gbest.copy()
                        best_c5 = gbest_score
            
            # Simulated annealing: accept worse solutions with probability
            if score > gbest_score and np.random.random() < np.exp(-score / temp):
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    if gbest_score < best_c5:
                        best_h = gbest.copy()
                        best_c5 = gbest_score

        # Annealing
        temp *= temp_decay
        
        # Periodic convex relaxation to prevent stagnation
        if iteration % 50 == 0 and iteration > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                k_vals = np.random.choice(n_points-1, min(20, n_points//3), replace=False)
                for k_val in k_vals:
                    if k_val > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k_val], 1 - h_opt[k_val:]))
                        constraints.append(corr_sum <= gbest_score * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
                    h_feasible = np.clip(h_feasible, 0, 1)
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
                        gbest = h_feasible
                        gbest_score = c5_feasible
                        
            except Exception:
                pass

    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
