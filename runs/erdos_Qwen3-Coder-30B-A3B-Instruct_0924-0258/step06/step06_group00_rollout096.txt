import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different starting points
    start_configs = []
    
    # Uniform distribution
    start_configs.append(np.ones(n_points) * 0.5)
    
    # Slightly perturbed uniform
    start_configs.append(np.clip(np.random.normal(0.5, 0.1, n_points), 0, 1))
    
    # Step function similar to previous
    h_step = np.zeros(n_points)
    mid = n_points // 2
    h_step[:mid] = 1.0
    h_step = h_step * (n_points/2) / np.sum(h_step)
    start_configs.append(h_step)
    
    # Periodic pattern
    h_periodic = np.ones(n_points) * 0.5
    period = max(2, n_points // 5)
    for i in range(0, n_points, period):
        if i + period//2 < n_points:
            h_periodic[i:i+period//2] = 1.0
    h_periodic = h_periodic * (n_points/2) / np.sum(h_periodic)
    start_configs.append(h_periodic)
    
    # Initialize with best result from multiple starting points
    for h_start in start_configs:
        h_current = h_start.copy()
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, h_current, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 300, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    # Additional optimization pass with higher resolution
    if time() - start_time < budget_s * 0.7:
        # Try to improve with a different resolution if needed
        n_high_res = min(100, n_points * 2)
        dx_hr = 2.0 / n_high_res
        
        # Interpolate
        x_full = np.linspace(0, 2, n_points)
        x_hr = np.linspace(0, 2, n_high_res)
        h_interp = np.interp(x_hr, x_full, best_h)
        h_interp = h_interp * (n_high_res/2) / np.sum(h_interp)
        
        try:
            def objective_hr(h):
                return evaluate_erdos_solution(h, dx_hr)
            
            def constraint_sum_hr(h):
                return np.sum(h) - n_high_res / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum_hr}
            bounds = [(0, 1) for _ in range(n_high_res)]
            res = minimize(objective_hr, h_interp, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective_hr(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    n_points = n_high_res
                    dx = dx_hr
        except Exception:
            pass
    
    # Final refinement
    if time() - start_time < budget_s * 0.9:
        try:
            # Final refinement with SLSQP
            h_final = np.clip(best_h, 0, 1)  # Ensure bounds are respected
            h_final = h_final * (n_points/2) / np.sum(h_final) 
            
            def objective_final(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum_final(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum_final}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective_final, h_final, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-10})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective_final(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
