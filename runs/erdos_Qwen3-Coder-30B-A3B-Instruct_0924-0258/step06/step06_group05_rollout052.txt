import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary algorithm with adaptive mutation
    population_size = 50
    num_generations = 1000
    
    # Initialize population
    population = []
    for i in range(population_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    mutation_rate = 0.05
    no_improvement_count = 0
    best_individual = best_h.copy()
    best_fitness = best_c5
    
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            f = evaluate_erdos_solution(h, dx)
            fitness.append(f)
        
        # Select top 20%
        idxs = np.argsort(fitness)
        top_idx = idxs[:population_size//5]
        
        # Update best
        if fitness[idxs[0]] < best_fitness:
            best_individual = population[idxs[0]].copy()
            best_fitness = fitness[idxs[0]]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Reduce mutation rate if stagnating
        if no_improvement_count > 100:
            mutation_rate = max(0.01, mutation_rate * 0.95)
            no_improvement_count = 0
        
        # Create offspring through mutation
        new_population = []
        for i in range(population_size):
            if i < len(top_idx):
                parent = population[top_idx[i % len(top_idx)]]
            else:
                parent = population[np.random.randint(0, population_size)]
            
            # Mutate
            noise = np.random.normal(0, mutation_rate, n_points)
            child = parent + noise
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            new_population.append(child)
        
        # Convex relaxation periodically
        if generation % 100 == 0 and generation > 0:
            try:
                h_var = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_var) == n_points / 2.0,
                    h_var >= 0,
                    h_var <= 1
                ]
                
                # Add constraints from a subset of k values to avoid too many
                k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
                        constraints.append(corr_sum <= best_fitness * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_var.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_fitness:
                        best_individual = h_feasible
                        best_fitness = c5_feasible
            except Exception:
                pass
         
        
        # Keep new population
        population = new_population
        
    # Final refinement step using local optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        # Run a few iterations of optimization on best individual
        for _ in range(5):
            res = minimize(objective, best_individual, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                std_h = res.x
                std_c5 = objective(std_h)
                if std_c5 < best_fitness:
                    best_individual = std_h
                    best_fitness = std_c5
            else:
                break
    except Exception:
        pass
    
    return (list(best_individual), best_fitness, n_points)
