import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n_points):
    half = n_points // 2
    if n_points % 2 == 0:
        h = np.concatenate([np.arange(1, half + 1), np.arange(half, 0, -1)])
    else:
        h = np.concatenate([np.arange(1, half + 2), np.arange(half, 0, -1)])
    return h / np.sum(h) * (n_points / 2)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = triangular_window(n_points)
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Constraint: sum(h) == n_points/2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE parameters
    popsize = 50
    maxiter = 300
    
    # Allow for DE to potentially use more time
    de_budget = min(0.7 * budget_s, maxiter * 3)
    de_start = time()
    
    result = differential_evolution(
        objective, 
        bounds, 
        args=(),
        maxiter=maxiter,
        popsize=popsize,
        strategy='best1bin',
        tol=1e-6,
        recombination=0.9,
        mutation=(0.5, 1.0),
        seed=seed,
        callback=lambda x, convergence: None,
        disp=False
    )
    
    if result.success:
        h_de = result.x
        c5_de = objective(h_de)
        if c5_de < best_c5:
            best_h = h_de
            best_c5 = c5_de

    # Local refinement with L-BFGS-B
    if time() - de_start < de_budget:
        try:
            # More targeted refinement with L-BFGS-B
            def objective_lbfgs(h):
                return evaluate_erdos_solution(h, dx)
            
            # Project back to sum constraint
            def project(h):
                h_proj = h * (n_points / 2) / np.sum(h)
                return np.clip(h_proj, 0, 1)
            
            # Refine from best individuals
            top_individuals = [best_h]
            for _ in range(5):
                pert = np.random.normal(0, 0.01, n_points)
                h_pert = best_h + pert
                h_pert = project(h_pert)
                top_individuals.append(h_pert)
                
            for h_ref in top_individuals:
                if time() - start_time > budget_s * 0.95:
                    break
                res = minimize(objective_lbfgs, h_ref, method='L-BFGS-B', 
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective_lbfgs(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        
        except Exception:
            pass

    # Final check
    return (list(best_h), best_c5, n_points)
