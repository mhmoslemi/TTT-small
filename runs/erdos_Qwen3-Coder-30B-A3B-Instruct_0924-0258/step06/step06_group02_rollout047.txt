import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        # Use a more structured initialization based on good known patterns
        h_start = np.zeros(n_points)
        # Create a symmetric, smooth function that spans the interval well
        x = np.linspace(0, 2, n_points)
        # Create a bump-like function with some randomness
        h_start = 0.5 + 0.3 * np.sin(np.pi * x) + 0.1 * np.random.randn(n_points)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with decay
    iterations = 1000
    learning_rate = 0.01
    decay = 0.999
    
    for i in range(iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute gradient for total overlap
        gradient = np.zeros(n_points)
        for k in range(1, n_points):
            # For each shift k, compute impact on gradient
            if k < n_points:
                # h[i] interacts with (1-h[i+k]) and h[i-k] when valid
                for i in range(n_points - k):
                    gradient[i] += (1 - best_h[i + k])
                for i in range(k, n_points):
                    gradient[i] -= best_h[i - k]
        
        # Apply gradient descent
        best_h -= learning_rate * gradient
        best_h = np.clip(best_h, 0, 1)
        
        # Project to correct integral
        sum_h = np.sum(best_h)
        if sum_h > 0:
            best_h = best_h * (n_points / 2) / sum_h
        
        # Evaluate current solution
        c5_new = evaluate_erdos_solution(best_h, dx)
        if c5_new < best_c5:
            best_h = best_h.copy()
            best_c5 = c5_new
            
        # Decay learning rate
        learning_rate *= decay
    
    # Final refinement to ensure feasibility and correctness
    try:
        best_h = np.clip(best_h, 0, 1)
        sum_h = np.sum(best_h)
        if sum_h > 0:
            best_h = best_h * (n_points / 2) / sum_h
        final_c5 = evaluate_erdos_solution(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
