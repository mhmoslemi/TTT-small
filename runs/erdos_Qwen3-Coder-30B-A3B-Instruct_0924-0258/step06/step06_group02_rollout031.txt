import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        return h
    scale = n / (2 * sum_h)
    h *= scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Annealing with focused perturbations
    T = 0.5
    cooling = 0.95
    min_T = 1e-4
    steps = 2000
    perturb_frac = 0.15
    
    for i in range(steps):
        if time() - start_time > budget_s * 0.80:
            break
            
        # Identify indices with high overlap
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Don't consider zero shifts
        corr_full = corr[len(best_h):] 
        # Normalize to get relative overlap
        max_corr_idx = np.argsort(corr_full)[-int(n_points*perturb_frac):]
        # Update picks
        h_new = best_h.copy()
        for k in max_corr_idx:
            if k < len(h_new):
                # Small change to elements with high correlations
                delta = np.random.normal(0, 0.1)
                h_new[k] += delta
        # Apply projections
        h_new = project_feasible(h_new, n_points)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Acceptance probability
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Cool down
        T = max(min_T, T * cooling)
    
    # QCP refinement with all possible shifts
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add all overlap constraints
        for k in range(1, n_points):
            # Calculate overlap for shift k
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final projection and validation
    h_final = project_feasible(best_h, n_points)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), c5_final, n_points)
