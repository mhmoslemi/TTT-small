import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create sinusoidal pattern to start with
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n_points))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Grid search with refinement
    resolutions = [10, 20, 35]
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate or create initial guess
        if 'initial_h_values' in globals() and n_points == len(initial_h_values):
            h_init = h_start.copy()
        else:
            x = np.linspace(0, 2, n_points)
            h_init = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n_points))
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            
        dx = 2.0 / n_points
        
        # Simulated Annealing
        h_current = h_init.copy()
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        temp = 0.1
        decay = 0.995
        max_iter = 100
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb
            perturb = np.random.normal(0, temp, n_points)
            h_new = h_current + perturb
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
            elif np.random.random() < np.exp(-(c5_new - c5_current) / temp):
                h_current = h_new
                c5_current = c5_new
                
            temp *= decay
            
        if c5_current < best_c5:
            best_h = h_current
            best_c5 = c5_current
    
    # Final refinement with quadratic programming
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final optimization with constrained optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use scipy for final refinement
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8})
        if result.success:
            h_final = result.x
            # Enforce sum constraint exactly
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
