import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n_points))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search in the range of interest
    low, high = 0.3807, 0.38085
    tolerance = 1e-6
    max_iter = 50
    
    for _ in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        candidate_c5 = (low + high) / 2.0
        
        # Solve LP feasibility
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Sample lags for constraints
            lags_to_check = []
            for k in range(1, n_points):
                if k <= n_points // 2:
                    lags_to_check.append(k)
                else:
                    lags_to_check.append(k)
            
            # Sample a subset of critical lags
            sampled_lags = lags_to_check[::max(1, len(lags_to_check)//10)]
            
            for k in sampled_lags:
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= candidate_c5 * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = np.array(h_opt.value).flatten()
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
                low = candidate_c5
            else:
                high = candidate_c5
        except Exception:
            high = candidate_c5
    
    # Final refinement
    try:
        # Gradient descent on best found solution
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                         options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8})
        if result.success:
            best_h = result.x
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    # Final normalization and validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
