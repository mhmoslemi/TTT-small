import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        # Symmetric initialization inspired by root systems
        h_start = np.zeros(n_points)
        centers = np.array([0.5, 1.0, 1.5])
        for c in centers:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - c))
            h_start[max(0, idx-3):min(n_points, idx+3)] = 0.8
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Trust-region based optimization with adaptive perturbations
    max_iterations = 1000
    tr_radius = 0.3
    lambda_merit = 0.1
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.80:
            break
            
        # Perturb with trust region
        perturbation = np.random.uniform(-tr_radius, tr_radius, n_points)
        h_new = best_h + perturbation
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct integral
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Constraint violation for merit function
        violation = max(0, np.max(h_new - 1), np.max(-h_new), abs(np.sum(h_new) - n_points/2))
        merit_new = c5_new + lambda_merit * violation
        
        violation_best = max(0, np.max(best_h - 1), np.max(-best_h), abs(np.sum(best_h) - n_points/2))
        merit_best = best_c5 + lambda_merit * violation_best
        
        if merit_new < merit_best:
            best_h = h_new.copy()
            best_c5 = c5_new
            tr_radius = min(0.9 * tr_radius, 0.5)
            # Increase lambda for stricter constraint enforcement
            if violation > 1e-4:
                lambda_merit = min(2.0 * lambda_merit, 1.0)
        else:
            tr_radius = max(0.9 * tr_radius, 1e-4)
            # Decrease lambda if constraint violations are rare
            if violation < 1e-5:
                lambda_merit *= 0.95 if lambda_merit > 0.001 else 0.001
        
        # Periodic SDP reinitialization
        if i % 100 == 0 and i > 0:
            try:
                h_sdp = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_sdp) == n_points / 2.0,
                    h_sdp >= 0,
                    h_sdp <= 1
                ]
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_sdp.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except:
                pass
    
    # Final verification and refinement
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
