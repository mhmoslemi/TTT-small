import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population
    population_size = 100
    population = []
    
    # Seed with initial solution and variations
    population.append(h_start.copy())
    for _ in range(population_size - 1):
        h_var = h_start + np.random.normal(0, 0.1, n_points)
        h_var = np.clip(h_var, 0, 1)
        h_var = h_var * (n_points/2) / np.sum(h_var)
        population.append(h_var)
    
    # GA parameters
    generations = 200
    elite_size = population_size // 10
    tournament_size = population_size // 5
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitnesses = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitnesses.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        if fitnesses[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitnesses[0]
        
        # Selection: tournament selection
        new_population = population[:elite_size]
        
        # Create new individuals
        while len(new_population) < population_size:
            # Tournament selection for parents
            parent1_idx = sorted(random.sample(range(population_size), tournament_size), 
                                key=lambda i: fitnesses[i])[0]
            parent2_idx = sorted(random.sample(range(population_size), tournament_size), 
                                key=lambda i: fitnesses[i])[0]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if random.random() < crossover_rate:
                crossover_point = random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            if random.random() < mutation_rate:
                mutation_indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
                for i in mutation_indices:
                    child[i] += np.random.normal(0, 0.05)
            
            # Ensure feasibility
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Final SLSQP refinement on top solutions
    top_candidates = population[:10]
    for h in top_candidates:
        h_local = h.copy()
        h_local = np.clip(h_local, 0, 1)
        h_local = h_local * (n_points/2) / np.sum(h_local)
        
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(objective, h_local, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            continue
    
    return (list(best_h), best_c5, n_points)
