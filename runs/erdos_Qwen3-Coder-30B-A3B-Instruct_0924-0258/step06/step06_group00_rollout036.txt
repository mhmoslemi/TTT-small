import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    population_size = 100
    max_iterations = 500
    F = 0.5
    CR = 0.7
    
    # Initialize population
    population = []
    fitness = []
    
    # Add initial solution
    h_normalized = h_start * (n_points/2) / np.sum(h_start)
    population.append(h_normalized)
    fitness.append(evaluate_erdos_solution(h_normalized, dx))
    
    # Generate additional diverse initial solutions
    for _ in range(population_size - 1):
        if time() - start_time > budget_s * 0.95:
            break
        # Random perturbation
        h_pert = h_start + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
        fitness.append(evaluate_erdos_solution(h_pert, dx))
    
    # Differential Evolution main loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        new_population = []
        new_fitness = []
        
        for i in range(population_size):
            # Select three different individuals
            indices = list(range(population_size))
            indices.remove(i)
            a, b, c = random.sample(indices, 3)
            
            # Mutation
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (n_points/2) / np.sum(mutant)
            
            # Crossover
            trial = mutant.copy()
            jrand = random.randint(0, n_points - 1)
            for j in range(n_points):
                if random.random() < CR or j == jrand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            
            # Selection
            trial_fitness = evaluate_erdos_solution(trial, dx)
            if trial_fitness < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_fitness)
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
        
        population = new_population
        fitness = new_fitness
        
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
    
    # Local refinement with SLSQP on best solutions
    top_candidates = 10
    sorted_indices = np.argsort(fitness)[:top_candidates]
    for idx in sorted_indices:
        if time() - start_time > budget_s * 0.95:
            break
        
        h_refine = population[idx].copy()
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_points/2) / np.sum(h_refine)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_refine, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 1000, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
