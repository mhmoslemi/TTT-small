import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    population_size = 50
    mutation_rate = 0.05
    max_generations = 1000
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Track best solution globally
    global_best_h = best_h.copy()
    global_best_c5 = best_c5
    
    generations_without_improvement = 0
    
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break

        # Evaluate current population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if fitness[0] < global_best_c5:
            global_best_c5 = fitness[0]
            global_best_h = population[0].copy()
            generations_without_improvement = 0
        else:
            generations_without_improvement += 1
            
        # Check for stagnation
        if generations_without_improvement > 100:
            mutation_rate = max(mutation_rate * 0.95, 0.01)
            generations_without_improvement = 0
            
        # Selection: keep top 20%
        elite_size = max(1, population_size // 5)
        elite = population[:elite_size]
        
        # Create offspring
        new_population = elite.copy()
        
        # Mutation and crossover
        for i in range(population_size - elite_size):
            # Select parent
            parent = elite[np.random.randint(0, elite_size)]
            
            # Add Gaussian noise
            noise = np.random.normal(0, mutation_rate, n_points)
            h_new = parent + noise
            
            # Clip and normalize
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_population.append(h_new)
        
        population = new_population
        
        # Periodic convex relaxation
        if gen % 100 == 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Constraints using subset of k values
                k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
                for k_val in k_vals:
                    k = max(1, min(n_points - 1, int(k_val * n_points / 2)))
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= global_best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < global_best_c5:
                        global_best_h = h_feasible.copy()
                        global_best_c5 = c5_feasible
                        
            except Exception:
                pass
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, global_best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < global_best_c5:
                global_best_h = h_refined
                global_best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(global_best_h), global_best_c5, n_points)
