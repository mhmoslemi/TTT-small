import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier-based refinement
    max_iter = 1000
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Create a frequency-domain optimization step
        h_freq = np.fft.fft(h_start)
        
        # Apply a window to suppress high frequencies
        window = np.hamming(len(h_freq))
        h_freq_windowed = h_freq * window
        
        # Inverse FFT to get refined h
        h_refined = np.real(np.fft.ifft(h_freq_windowed))
        
        # Project to valid bounds [0,1]
        h_refined = np.clip(h_refined, 0, 1)
        
        # Normalize to correct integral
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        # Evaluate and check improvement
        c5_new = evaluate_erdos_solution(h_refined, dx)
        if c5_new < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_new
        
        # Perturb for next iteration
        perturbation = np.random.normal(0, 0.01, n_points)
        h_start = h_refined + perturbation
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
        
        iter_count += 1
    
    # Final local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
                
    return (list(best_h), best_c5, n_points)
