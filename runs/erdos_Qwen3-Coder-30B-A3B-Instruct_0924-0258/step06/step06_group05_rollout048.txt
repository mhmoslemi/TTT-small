import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create initial binary configuration (periodic)
        h_start = np.zeros(n_points)
        ones_count = n_points // 2
        for i in range(ones_count):
            h_start[i * (n_points // ones_count)] = 1
        h_start[0] = 1
        h_start[-1] = 1
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def binary_to_continuous(h_binary):
        h_cont = h_binary.astype(float)
        h_cont = h_cont * (n_points/2) / np.sum(h_cont)
        return h_cont
        
    def update_best(h_new, c5_new):
        nonlocal best_h, best_c5
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

    # Simulated Annealing with binary variables
    T = 0.1  # Initial temperature
    decay = 0.995
    min_T = 1e-6
    iterations = 0
    max_iter = 1000
    
    h_current = best_h.copy()
    c5_current = best_c5
    
    while time() - start_time < budget_s * 0.95 and iterations < max_iter:
        iterations += 1
        h_new = h_current.copy()
        
        # Make a random bit swap
        indices_0 = np.where(h_new == 0)[0]
        indices_1 = np.where(h_new == 1)[0]
        
        if len(indices_0) == 0 or len(indices_1) == 0:
            break
            
        i0 = np.random.choice(indices_0)
        i1 = np.random.choice(indices_1)
        
        h_new[i0] = 1
        h_new[i1] = 0
        
        # Rescale to maintain sum constraint
        h_new = binary_to_continuous(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / T):
            h_current = h_new.copy()
            c5_current = c5_new
        
        update_best(h_current, c5_current)
        T = max(min_T, T * decay)
        
        # Periodic convex refinement
        if iterations % 100 == 0 and time() - start_time < budget_s * 0.9:
            try:
                h_refine = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_refine) == n_points / 2.0,
                    h_refine >= 0,
                    h_refine <= 1
                ]
                
                # Add a subset of constraints based on sampled lags
                sample_k = np.linspace(0.1, 1.9, min(20, n_points//3))
                for k in sample_k:
                    k_int = int(k * n_points / 2)
                    if k_int < n_points-1 and k_int > 0:
                        corr_sum = cp.sum(cp.multiply(h_refine[:n_points-k_int], 1 - h_refine[k_int:]))
                        # Conservative bound
                        constraints.append(corr_sum <= c5_current * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=300)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_refine.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < c5_current:
                        h_current = h_feasible
                        c5_current = c5_feasible
                        update_best(h_current, c5_current)
            except Exception:
                pass
    
    # Final local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final check and return 
    if np.sum(best_h) < n_points/2-1e-6 or np.sum(best_h) > n_points/2+1e-6:
        best_h = binary_to_continuous(best_h)
    
    return (list(best_h), best_c5, n_points)
