import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    bounds = [(0, 1) for _ in range(n_points)]
    maxiter = 500
    popsize = 100
    mutation = (0.5, 1)
    recombination = 0.7
    
    # Generate diverse initial population
    def create_initial_population():
        pop = []
        pop.append(h_start.copy())
        
        # Add random perturbations
        for i in range(30):
            h_rand = np.random.uniform(0, 1, n_points)
            h_rand = h_rand * (n_points/2) / np.sum(h_rand)
            pop.append(h_rand)
        
        # Add structured patterns
        for i in range(30):
            h_pattern = np.ones(n_points) * 0.5
            # Alternating blocks
            for j in range(0, n_points, 4):
                if j + 2 < n_points:
                    h_pattern[j:j+2] = 1.0
                if j + 4 < n_points:
                    h_pattern[j+2:j+4] = 0.0
            h_pattern = h_pattern * (n_points/2) / np.sum(h_pattern)
            pop.append(h_pattern)
        
        # Add clipped normal random
        for i in range(30):
            h_norm = np.random.normal(0.5, 0.2, n_points)
            h_norm = np.clip(h_norm, 0, 1)
            h_norm = h_norm * (n_points/2) / np.sum(h_norm)
            pop.append(h_norm)
            
        return np.array(pop)
    
    # Normalize to required integral
    def normalize_h(h):
        return h * (n_points/2) / np.sum(h)
    
    # Wrap objective for scipy differential_evolution
    def objective_wrap(h):
        h_normalized = normalize_h(h)
        return evaluate_erdos_solution(h_normalized, dx)
    
    # DE with diverse initialization
    try:
        pop = create_initial_population()
        res_de = differential_evolution(
            objective_wrap,
            bounds,
            maxiter=maxiter,
            popsize=popsize,
            mutation=mutation,
            recombination=recombination,
            seed=seed,
            disp=False,
            init=pop
        )
        if res_de.success:
            h_de = normalize_h(res_de.x)
            c5_de = objective_wrap(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    except Exception:
        pass

    # Local refinement with SLSQP on top candidates
    try:
        # Collect top candidates from the population
        if 'pop' in locals():
            top_candidates = []
            for p in pop:
                c = evaluate_erdos_solution(normalize_h(p), dx)
                top_candidates.append((c, p))
            top_candidates.sort(key=lambda x: x[0])
            top_candidates = top_candidates[:10]  # Top 10%
        else:
            top_candidates = [(best_c5, best_h)]
            
        for _, h_candidate in top_candidates:
            h_candidate_normalized = normalize_h(h_candidate)
            
            # SLSQP refinement
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_candidate_normalized, 
                           method='SLSQP', bounds=bounds, 
                           constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-10})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
