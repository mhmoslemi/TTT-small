import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Sparse Fourier Optimization with Dithering
    max_iter = 1000
    learning_rate = 0.1
    dither_std = 0.01
    patience = 50
    patience_counter = 0
    best_c5_history = [best_c5]
    
    # Initialize frequency representation (sinusoids)
    num_freqs = 5
    frequencies = np.arange(1, num_freqs + 1)
    coeffs = np.random.normal(0, 0.1, 2 * num_freqs)
    
    def compute_h_from_coeffs():
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points, endpoint=False)
        for i, f in enumerate(frequencies):
            h += coeffs[2*i] * np.sin(2 * np.pi * f * x)
            h += coeffs[2*i+1] * np.cos(2 * np.pi * f * x)
        # Add constant term and clip
        h = h + np.mean(h)  # Adjust to have same mean as starting h
        h = np.clip(h, 0, 1)
        # Normalize to desired integral
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def compute_gradient():
        h = compute_h_from_coeffs()
        # Compute gradients with respect to frequencies
        grad = np.zeros_like(coeffs)
        eps = 1e-5
        base_c5 = evaluate_erdos_solution(h, dx)
        for i in range(len(coeffs)):
            coeffs_pert = coeffs.copy()
            coeffs_pert[i] += eps
            h_pert = np.zeros(n_points)
            x = np.linspace(0, 2, n_points, endpoint=False)
            for j, f in enumerate(frequencies):
                h_pert += coeffs_pert[2*j] * np.sin(2 * np.pi * f * x)
                h_pert += coeffs_pert[2*j+1] * np.cos(2 * np.pi * f * x)
            h_pert = h_pert + np.mean(h_pert)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            pert_c5 = evaluate_erdos_solution(h_pert, dx)
            grad[i] = (pert_c5 - base_c5) / eps
        return grad
    
    # Main optimization loop
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        try:
            grad = compute_gradient()
            # Update coefficients
            coeffs -= learning_rate * grad
            # Add dithering
            coeffs += np.random.normal(0, dither_std, len(coeffs))
            # Check feasibility
            h_new = compute_h_from_coeffs()
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                patience_counter = 0
                best_c5_history.append(best_c5)
            else:
                patience_counter += 1
                
            # Decay learning rate and dithering
            learning_rate *= 0.995
            dither_std *= 0.999
        except Exception:
            pass
        
        # Early stopping if no improvement for a while
        if patience_counter >= patience:
            break
    
    # Final refinement with quadratic programming
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
