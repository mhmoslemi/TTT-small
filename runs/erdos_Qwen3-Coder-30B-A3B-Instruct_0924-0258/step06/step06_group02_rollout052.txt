import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Annealing parameters
    T = 0.5
    T_min = 1e-4
    cooling_rate = 0.99
    max_steps = 2000
    step = 0
    
    while step < max_steps and time() - start_time < budget_s * 0.8:
        # Create candidate solution
        h_new = best_h.copy()
        
        # Focused perturbation on high overlap regions
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Find top 10% of overlap indices
        top_k = max(1, n_points // 10)
        overlap_indices = np.argsort(corr)[-top_k:]
        for i in overlap_indices:
            # Add noise to indices around high overlap
            if i < n_points:
                idx = i % n_points
                perturbation = np.random.choice([-0.1, 0.1])
                h_new[idx] += perturbation
                h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Normalize integral
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Cool down
        T = max(T_min, T * cooling_rate)
        step += 1
        
        if time() - start_time > budget_s * 0.90:
            break
    
    # Gradient refinement
    try:
        # Use scipy minimize for refinement
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint}
        bounds = [(0, 1) for _ in range(n_points)]
        
        h_refined = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds,
            constraints=cons,
            options={'eps': 1e-6, 'maxiter': 100}
        )
        
        if h_refined.success:
            h_final = h_refined.x
            h_final = np.clip(h_final, 0, 1)
            # Ensure integral constraint
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
