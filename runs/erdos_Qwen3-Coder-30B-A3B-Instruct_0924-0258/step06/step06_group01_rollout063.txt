import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initialize parameters
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize h using the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points/2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient-based optimization with constraints
    max_iter = 10000
    iter_count = 0
    last_improvement = 0
    tolerance = 1e-6
    
    # Optimization parameters
    learning_rate = 0.05
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute current correlation to identify maximum overlap
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_overlap = corr[max_k]
        
        # Formulate gradient based on correlation
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Compute partial derivative of overlap w.r.t. h[i]
            # This is approximated: d/dh[i] (sum_j h[j] * (1 - h[j+k]))
            # For the overlapping term h[i] * (1 - h[i+k]), 
            # we get (1 - h[i+k]) and not h[i+k] due to the (1-h) in correlation
            
            # foward shift to align with convolution indices
            forward_idx = (i + max_k - (n_points - 1)) % n_points
            backward_idx = (i - max_k + (n_points - 1)) % n_points
            
            # Gradient: -1 + 2*h[i] 
            # Even though this is an approximation, it remains reasonable for gradient descent
            grad[i] = -1 + 2 * h[i]
        
        # Apply momentum
        velocity = momentum * velocity - learning_rate * grad
        h_new = h + velocity
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = iter_count
        
        # Update current solution
        h = h_new
        
        iter_count += 1
        
        # Early stopping if no improvement
        if iter_count - last_improvement > 1000:
            break
    
    # Final polishing with scipy optimize
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use L-BFGS-B for better performance
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        
        if res.success:
            h_final = res.x.copy()
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception as e:
        pass
    
    return (list(best_h), best_c5, n_points)
