import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Get initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create triangular window initial solution
        n_points = 35
        h_start = 1 - np.abs(np.arange(n_points) / (n_points - 1) - 0.5) * 2
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    temp = 0.5
    cooling_rate = 0.99
    max_iter = 5000
    min_temp = 1e-6
    
    # SA main loop
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Generate candidate by perturbing current solution
        candidate = best_h + np.random.normal(0, temp, n_points)
        
        # Apply convex normalization to maintain constraints
        candidate_sum = np.sum(candidate)
        if candidate_sum > 0:
            candidate = candidate * (n_points/2) / candidate_sum
        candidate = np.clip(candidate, 0, 1)
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(candidate, dx)
        
        # Accept or reject
        if candidate_c5 < best_c5:
            best_h = candidate.copy()
            best_c5 = candidate_c5
        else:
            # Metropolis criterion
            delta = candidate_c5 - best_c5
            if random.random() < np.exp(-delta / temp):
                best_h = candidate.copy()
                best_c5 = candidate_c5
                
        # Cool down
        temp = max(temp * cooling_rate, min_temp)
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use best solution from SA as starting point
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
