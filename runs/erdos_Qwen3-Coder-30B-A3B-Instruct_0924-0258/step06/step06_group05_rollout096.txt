import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convert to binary representation
    def to_binary(h):
        # Convert continuous h to binary with fixed sum
        h_sorted = np.argsort(h)[::-1]  # indices in decreasing order
        h_bin = np.zeros(n_points)
        ones_count = int(n_points / 2)
        h_bin[h_sorted[:ones_count]] = 1
        return h_bin
    
    # Refine binary representation to satisfy sum constraint
    def binary_refine(h_bin):
        current_sum = np.sum(h_bin)
        target_sum = n_points / 2
        
        if np.abs(current_sum - target_sum) < 1e-6:
            return h_bin
            
        # Ensure sum constraint by swapping
        if current_sum > target_sum:
            # Find 1s to convert to 0s (find indices with 1)
            ones_idx = np.where(h_bin == 1)[0]
            to_zero = int(current_sum - target_sum)
            for i in range(to_zero):
                if i < len(ones_idx):
                    h_bin[ones_idx[i]] = 0
        else:
            # Need to convert 0s to 1s
            zeros_idx = np.where(h_bin == 0)[0]
            to_one = int(target_sum - current_sum)
            for i in range(to_one):
                if i < len(zeros_idx):
                    h_bin[zeros_idx[i]] = 1
                    
        return h_bin
        
    # Initialize binary representation
    h_bin = to_binary(best_h)
    h_bin = binary_refine(h_bin)
    best_c5 = evaluate_erdos_solution(h_bin, dx)
    
    # Binary local search with simulated annealing
    temp = 0.5
    cooling_rate = 0.995
    max_iter = 10000
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor by swapping two random positions
        pos1, pos2 = np.random.choice(n_points, 2, replace=False)
        h_new_bin = h_bin.copy()
        h_new_bin[pos1], h_new_bin[pos2] = h_new_bin[pos2], h_new_bin[pos1]
        
        # Rebalance if needed
        h_new_bin = binary_refine(h_new_bin)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new_bin, dx)
        
        # Accept or reject
        if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / temp):
            h_bin = h_new_bin
            if c5_new < best_c5:
                best_h = h_bin.copy()
                best_c5 = c5_new
                
        temp *= cooling_rate
        
        # Every 100 iterations, do convex optimization refinement
        if i % 100 == 0 and time() - start_time < budget_s * 0.95:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample k values for constraints
                k_vals = np.linspace(0.05, 1.95, min(20, n_points // 3))
                k_vals = [int(k_val * n_points / 2) for k_val in k_vals]
                k_vals = [k for k in k_vals if 0 < k < n_points]
                
                for k in k_vals:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass

    # Final refinement using continuous optimization on best binary solution
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Ensure it is the final binary solution
    if isinstance(best_h, list):
        best_h = np.array(best_h)
    # Convert back to continuous representation if needed for final computation
    h_final = best_h
    
    # Evaluate final result
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), c5_final, n_points)
