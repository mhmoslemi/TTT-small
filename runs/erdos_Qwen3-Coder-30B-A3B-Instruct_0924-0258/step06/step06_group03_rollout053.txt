import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create structured initialization using triangular window
        h_start = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h_start[i] = max(0, 1.0 - abs(i - mid) / (n_points/2))
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds for differential evolution (0 to 1 for each point)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint function: sum should equal n_points/2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    # Define the objective function for DE
    def objective_de(x):
        return evaluate_erdos_solution(x, dx)
    
    # Differential Evolution optimization
    de_res = differential_evolution(
        objective_de,
        bounds,
        maxiter=300,
        popsize=50,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        disp=False,
        tol=1e-6,
        callback=lambda x, convergence: None
    )
    
    # Check if we got a better solution from DE
    if de_res.success:
        de_h = de_res.x
        de_c5 = objective_de(de_h)
        if de_c5 < best_c5:
            best_h = de_h.copy()
            best_c5 = de_c5
    
    # Local refinement with L-BFGS-B
    # First, project to feasible region (sum constraint)
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Refine with L-BFGS-B
    def objective_lbfgs(h):
        return evaluate_erdos_solution(h, dx)
    
    # Use the best solution found so far for L-BFGS-B
    h_refined = project_to_feasible(best_h.copy())
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds_lbfgs = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(
            objective_lbfgs,
            h_refined,
            method='L-BFGS-B',
            bounds=bounds_lbfgs,
            constraints=cons,
            options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 100}
        )
        if res.success:
            h_final = res.x
            c5_final = objective_lbfgs(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final check with projection
    best_h = project_to_feasible(best_h)
    
    return (list(best_h), best_c5, n_points)
