import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.random.rand(n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_h = h_start.copy()
    
    # ADMM parameters
    rho = 1e-4
    max_iter = 1000
    tol = 1e-6
    iter_count = 0
    
    # Initial values
    z = h_current.copy()
    u = np.zeros_like(h_current)
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Primal update: minimize quadratic approximation
        # minimize (rho/2)||z - h + u||_2^2 + (1/rho) * C5_approx(z)
        # subject to sum(z) = n_points/2, 0 <= z[i] <= 1
        
        try:
            # Create a simple convex approximation of the objective
            # Since the full objective can be complex, we compute a surrogate
            h_local = z.copy()
            h_local = np.clip(h_local, 0, 1)
            h_local = h_local * (n_points/2) / np.sum(h_local)
            
            # Use simple gradient descent as a proxy
            def objective(h_test):
                return evaluate_erdos_solution(h_test, dx)
            
            def grad_objective(h_test):
                # Approximate gradient via finite differences
                eps = 1e-6
                grad = np.zeros_like(h_test)
                for i in range(len(h_test)):
                    h_plus = h_test.copy()
                    h_plus[i] += eps
                    h_minus = h_test.copy()
                    h_minus[i] -= eps
                    grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
                return grad
            
            # Simple gradient descent step
            grad = grad_objective(h_local)
            z_new = h_local - 0.01 * grad
            z_new = np.clip(z_new, 0, 1)
            z_new = z_new * (n_points/2) / np.sum(z_new)
            
            # Projection to sum constraint
            sum_diff = np.sum(z_new) - n_points/2
            z_new = z_new - sum_diff / n_points
            
            z_new = np.clip(z_new, 0, 1)
            z_new = z_new * (n_points/2) / np.sum(z_new)
            
            z = z_new.copy()
            
            # Dual update
            u = u + h_local - z
            
            # Update h with proximal step
            h_new = z - (1/rho) * (z - h_local + u)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            h_current = h_new.copy()
            
            # Check convergence
            if np.linalg.norm(h_current - z) < tol:
                break
                
            iter_count += 1
            
            # Evaluate and update best solution
            current_c5 = evaluate_erdos_solution(h_current, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                
        except Exception as e:
            # Fallback to simple local optimization if ADMM fails
            break
    
    # Final SLSQP refinement
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        if res.success:
            refined_h = res.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
