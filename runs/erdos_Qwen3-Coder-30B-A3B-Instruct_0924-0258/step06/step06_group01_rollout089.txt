import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a gradient-based approach with adaptive steps
    max_iter = 10000
    patience = 100
    patience_counter = 0
    prev_best = best_c5
    learning_rate = 0.1
    momentum = 0.9
    conv_radius = 0.01
    
    h_current = h_start.copy()
    velocity = np.zeros_like(h_current)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlap and find the maximum shift
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        if max_overlap < best_c5:
            best_h = h_current.copy()
            best_c5 = max_overlap
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            break
            
        # Compute gradient with finite differences
        eps = 1e-6
        grad = np.zeros_like(h_current)

            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)

            overlap_plus = evaluate_erdos_solution(h_plus, dx)
            
            grad[i] = (overlap_plus - max_overlap) / eps
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h_new = h_current + velocity
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Update current solution
        h_current = h_new
        
        # Decrease learning rate as optimization progresses
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.95
    
    # Final polishing with scipy
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),

            method='SLSQP',

            constraints=cons_sum,

            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points/2) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
