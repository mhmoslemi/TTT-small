import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Population-based GA approach
    population_size = 100
    num_generations = 200
    elite_size = population_size // 10
    mutation_rate = 0.1
    
    # Create initial population
    population = []
    population.append(h_start.copy())
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / n_points)
    
    # GA loop
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, 2.0 / n_points)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection - top 50%
        selected = population[:population_size // 2]
        
        # Create new population
        new_population = selected[:elite_size]  # Elitism
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = selected[random.randint(0, len(selected) - 1)]
            parent2 = selected[random.randint(0, len(selected) - 1)]
            
            # Crossover
            child = np.copy(parent1)
            mask = np.random.rand(n_points) < 0.5
            child[mask] = parent2[mask]
            
            # Mutation
            if random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            
            # Normalize sum
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Perform local refinement on best solutions
    top_candidates = [best_h] + population[:10]
    for h_candidate in top_candidates:
        if time() - start_time > budget_s * 0.95:
            break
            
        # SLSQP refinement
        try:
            def objective(h):
                return evaluate_erdos_solution(h, 2.0 / n_points)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_candidate, method='SLSQP',
                          bounds=bounds, constraints=cons_sum,
                          options={'maxiter': 200, 'ftol': 1e-9})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
