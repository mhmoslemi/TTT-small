from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    max_iter = 500
    iteration = 0
    
    # Convert to CVXPY problem setup
    h_var = Variable(n_points)
    
    def fix_sum_constraint(h_vec):
        return h_vec * (n_points/2) / np.sum(h_vec)
    
    def project_onto_constraints(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = fix_sum_constraint(h_vec)
        return h_vec
    
    # Initialize current solution
    current_h = h_start.copy()
    
    # Iterative refinement using projected gradient or convex relaxation approach
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Calculate current correlation for guidance
        corr = np.correlate(current_h, 1-current_h, mode='full')
        max_corr = np.max(corr) * dx
        
        if max_corr < best_c5:
            best_h = current_h.copy()
            best_c5 = max_corr
        
        # Formulate and solve convex relaxation subproblem
        # Objective: minimize max correlation
        # Since we can't directly minimize max, we approach it differently
        
        # Project current solution to feasible set
        current_h = project_onto_constraints(current_h)
        
        # Define a quadratic approximation around the current point
        # Solve QP with constraints: sum(h) = n_points/2, 0 <= h <= 1
        
        # Try solving using scipy with better formulation
        try:
            h_current = current_h.copy()
            
            # Create a new problem from scratch
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            
            # Define objective to minimize max overlap
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
            
            # Local refinement with different solver
            res = minimize(
                objective,
                h_current,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            
            if res.success:
                new_h = res.x
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points/2) / np.sum(new_h)
                current_h = new_h
            else:
                # If minimization fails, just project

            current_h = project_onto_constraints(current_h)
    
    # Final boundary check

    if final_c5 < best_c5:
        best_h = current_h
        best_c5 = final_c5
    
    # Ensure final solution satisfies all constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
