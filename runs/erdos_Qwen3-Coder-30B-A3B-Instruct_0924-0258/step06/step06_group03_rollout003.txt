import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create triangular initialization
        h_start = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h_start[i] = 1.0 - abs(i - mid) / (mid + 1)
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # DE optimization
    de_start = time()
    def de_objective(h):
        return objective(h)
    
    # Use differential evolution for global search
    de_result = differential_evolution(
        de_objective,
        bounds,
        maxiter=300,
        popsize=50,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        disp=False
    )
    
    if time() - start_time > 0.9 * budget_s:
        h_best = h_start
        c5_best = evaluate_erdos_solution(h_start, dx)
    else:
        h_best = de_result.x.copy()
        c5_best = de_result.fun
    
    # Local refinement using L-BFGS-B
    if time() - start_time < 0.95 * budget_s:
        try:
            # Refinement loop: apply several local optimizations on best individuals
            top_candidates = [h_best]
            for _ in range(10):
                # Generate perturbed version of current best
                h_pert = h_best + np.random.normal(0, 0.01, n_points)
                h_pert = np.clip(h_pert, 0, 1)
                # Re-normalize
                h_pert = h_pert * (n_points/2) / np.sum(h_pert)
                top_candidates.append(h_pert)
            
            # Evaluate candidates
            candidate_scores = [objective(h) for h in top_candidates]
            best_idx = np.argmin(candidate_scores)
            h_candidate = top_candidates[best_idx]
            c5_candidate = candidate_scores[best_idx]
            
            if c5_candidate < c5_best:
                h_best = h_candidate
                c5_best = c5_candidate

            # Final local optimization
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, h_best, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < c5_best:
                    h_best = h_refined
                    c5_best = c5_refined
        except Exception:
            pass
    
    return (list(h_best), c5_best, n_points)
