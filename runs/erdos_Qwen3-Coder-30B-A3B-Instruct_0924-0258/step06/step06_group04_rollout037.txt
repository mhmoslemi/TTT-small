import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Generate sinusoidal initial guess
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.5 * np.sin(np.pi * x)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Grid refinement
    grid_sizes = [10, 20, 35]  # Up to 35 points
    for n in grid_sizes:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate if needed
        if n_points < n:
            x_new = np.linspace(0, 2, n)
            x_old = np.linspace(0, 2, n_points)
            h_new = np.interp(x_new, x_old, best_h)
            h_new = h_new * (n / 2) / np.sum(h_new)
        else:
            h_new = best_h[:n] if n_points > n else best_h.copy()
            
        n_points = n
        dx = 2.0 / n_points
        
        # Annealing + gradient descent
        temp = 0.1
        alpha = 0.1
        max_iter = 200
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Stochastic gradient 
            # Approximate gradient through finite differences
            h_trial = h_new.copy()
            noise = np.random.normal(0, temp, n_points)
            h_trial += noise
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
            
            # Simulated Annealing step
            if np.random.rand() < 0.1:  # 10% chance
                h_perturb = h_new + np.random.normal(0, temp, n_points)
                h_perturb = np.clip(h_perturb, 0, 1)
                h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
                c5_perturb = evaluate_erdos_solution(h_perturb, dx)
                if c5_perturb < best_c5:
                    best_h = h_perturb
                    best_c5 = c5_perturb
            
            # Decrease temperature
            temp *= 0.995
            # Decrease step size
            alpha = max(0.001, alpha * 0.995)
            
        # Final refinement with local optimizer
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            result = minimize(
                lambda x: evaluate_erdos_solution(x, dx),
                best_h,
                bounds=bounds,
                method='L-BFGS-B',
                options={'maxiter': 50}
            )
            if result.success:
                refined = result.x
                refined = np.clip(refined, 0, 1)
                refined = refined * (n_points / 2) / np.sum(refined)
                c5_refined = evaluate_erdos_solution(refined, dx)
                if c5_refined < best_c5:
                    best_h = refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Binary search refinement on the last found
    try:
        # Binary search on bound
        low = best_c5
        high = 0.3807
        for _ in range(20):
            if time() - start_time > budget_s * 0.95:
                break
            mid = (low + high) / 2.
            # Check if we can construct solution with this bound
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            # Sample few key lags for tighter constraints
            for k in range(1, min(n_points//2+1, 20)):
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= mid * dx + 1e-6)

            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                low = mid
            else:
                high = mid
                
        # Final result with refined C5
        best_c5 = low
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
