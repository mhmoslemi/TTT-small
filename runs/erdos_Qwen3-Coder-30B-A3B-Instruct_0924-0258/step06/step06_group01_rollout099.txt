import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    T_min = 1e-5
    alpha = 0.999
    max_iter_per_T = 100
    
    # SA loop
    while T > T_min and (time() - start_time) < budget_s * 0.95:
        for _ in range(max_iter_per_T):
            if (time() - start_time) >= budget_s * 0.95:
                break
                
            # Generate candidate solution
            h_candidate = h.copy()
            i, j = np.random.randint(0, n_points, 2)
            
            # Perturb h[i]
            delta = np.random.normal(0, 0.05)
            h_candidate[i] += delta
            
            # Adjust h[j] to maintain sum constraint
            sum_diff = np.sum(h_candidate) - n_points / 2
            h_candidate[j] -= sum_diff / (n_points - 1) if n_points > 1 else 0
            
            # Clip to [0, 1] and rescale
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Evaluate
            candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept or reject
            delta_c5 = candidate_c5 - best_c5
            if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
                h = h_candidate.copy()
                if candidate_c5 < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = candidate_c5
        
        T *= alpha
    
    # Final refinement with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
