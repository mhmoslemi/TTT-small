import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # New strategy: Homotopy method
    homotopy_steps = 100
    t_vals = np.linspace(0, 1, homotopy_steps)
    
    h_current = h_start.copy()
    current_c5 = best_c5
    
    for t in t_vals:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Define homotopy problem
        try:
            h_opt = cp.Variable(n_points)
            
            # Regularization term (to simplify problem)
            reg_term = cp.sum_squares(h_opt - np.ones(n_points) * 0.5)
            
            # Objective: weighted combination
            objective = t * cp.Variable() + (1 - t) * reg_term
            
            # Constraint for sum
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # For large t, add constraints from sampled k values
            if t > 0.7:
                # Sample a few k values to add constraints
                k_vals = np.linspace(0.1, 1.9, min(15, n_points//4))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if 0 < k < n_points:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= 1.0)  # placeholder
                
            prob = cp.Problem(cp.Minimize(objective), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_opt.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                if c5_new < current_c5:
                    h_current = h_new.copy()
                    current_c5 = c5_new
                    
                    if c5_new < best_c5:
                        best_h = h_current.copy()
                        best_c5 = c5_new
                        
        except Exception:
            pass
    
    # Final local refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_current, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    # Final repair to ensure all constraints 
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
