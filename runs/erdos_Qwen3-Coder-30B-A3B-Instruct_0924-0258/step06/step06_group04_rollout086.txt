import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization
    n_levels = 4
    resolutions = [10, 20, 35]  # Reduced from 10,20,40,80 to fit budget
    current_h = h_start.copy()
    current_n = n_points
    
    for level, n_res in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        if n_res != current_n:
            # Interpolate to new resolution
            x_orig = np.linspace(0, 2, current_n)
            x_new = np.linspace(0, 2, n_res)
            current_h = np.interp(x_new, x_orig, current_h)
            current_n = n_res
            
        dx_current = 2.0 / current_n
        
        # Sinusoidal initialization
        x = np.linspace(0, 2, current_n)
        h_init = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / current_n))
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (current_n / 2) / np.sum(h_init)
        
        # Simulated Annealing parameters
        temp = 0.1
        cooling_rate = 0.995
        max_steps = 100
        
        h_current = h_init.copy()
        c5_current = evaluate_erdos_solution(h_current, dx_current)
        
        for step in range(max_steps):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturbation
            noise = np.random.normal(0, temp, current_n)
            h_new = h_current + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (current_n / 2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx_current)
            
            # Accept or reject based on SA
            delta = c5_new - c5_current
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h_current = h_new
                c5_current = c5_new
                
            # Cool temperature
            temp *= cooling_rate
            
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                
        current_h = h_current.copy()
        
    # Binary search refinement for better precision
    target = 0.3807
    left = best_c5
    right = target
    tolerance = 1e-6
    
    for _ in range(20):  # Limit iterations
        if time() - start_time > budget_s * 0.95:
            break
            
        if right - left < tolerance:
            break
            
        c5_test = (left + right) / 2
        
        # Try to construct feasible solution with this bound
        try:
            h_opt = cp.Variable(current_n)
            constraints = [
                cp.sum(h_opt) == current_n / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add constraints for correlation ≤ c5_test * dx
            for k in range(1, current_n):
                if k <= current_n // 2:
                    corr_k = cp.sum(cp.multiply(h_opt[:current_n-k], 1 - h_opt[k:]))
                else:
                    corr_k = cp.sum(cp.multiply(h_opt[k-current_n+1:], 1 - h_opt[:current_n-(k-current_n+1)]))
                constraints.append(corr_k <= c5_test * dx_current + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx_current)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
                right = c5_test
            else:
                left = c5_test
                
        except Exception:
            left = c5_test
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (current_n/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx_current)
    
    return (list(best_h), best_c5, current_n)
