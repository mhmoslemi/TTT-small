import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n_points))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters  
    low, high = 0.3807, 0.38085
    tolerance = 1e-6
    max_iter = 50
    
    def check_feasibility(h, C, dx):
        # Check if h is feasible for C_5 <= C
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx <= C
    
    # Modified binary search
    for _ in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        C = (low + high) / 2.0
        
        # Try to find feasible solution using LP with constraint sampling
        try:
            # Initialize constraint set with a few key constraints
            h_vars = cp.Variable(n_points)
            constraints = [
                cp.sum(h_vars) == n_points / 2.0,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Sample lags adaptively
            lags_to_check = list(range(1, n_points // 2 + 1))
            np.random.shuffle(lags_to_check)
            sampled_lags = lags_to_check[:max(5, n_points // 10)]
            
            # Add initial constraints
            for k in sampled_lags:
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_vars[:n_points-k], 1 - h_vars[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_vars[k-n_points+1:], 1 - h_vars[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= C * dx + 1e-8)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_vars.value
                if check_feasibility(h_feasible, C, dx):
                    high = C
                    best_h = h_feasible
                    best_c5 = evaluate_erdos_solution(best_h, dx)
                else:
                    low = C
            else:
                low = C
        except Exception:
            low = C
    
    # Final gradient-based refinement if we're close to the target
    if abs(low - high) < tolerance * 10:
        # Try to refine with local optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        # Gradient descent refinement
        h_refined = best_h.copy()
        for _ in range(500):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Compute gradient numerically
            eps = 1e-4
            grad = np.zeros(n_points)
            orig_val = objective(h_refined)
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                grad[i] = (objective(h_plus) - orig_val) / eps
            
            # Step towards lower overlap
            step_size = 0.01
            h_new = h_refined - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            new_val = objective(h_new)
            if new_val < objective(h_refined):
                h_refined = h_new
                best_c5 = new_val
                best_h = h_refined
    
    # Final adjustment to ensure constraints are satisfied
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
