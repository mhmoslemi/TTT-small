import numpy as np
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def generate_initial_candidates(n_points, num_candidates=100):
    candidates = []
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
        candidates.append(h_base.copy())
    
    # Add random variations of base solution
    for _ in range(30):
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values) + np.random.normal(0, 0.02, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        candidates.append(h)
    
    # Add structured patterns
    # Alternating high-low blocks
    pattern1 = np.ones(n_points) * 0.5
    for i in range(n_points):
        if i % 4 < 2:
            pattern1[i] = 0.8
        else:
            pattern1[i] = 0.2
    pattern1 = pattern1 * (n_points / 2) / np.sum(pattern1)
    candidates.append(pattern1)
    
    # Smooth step pattern
    x = np.linspace(0, 2, n_points)
    pattern2 = 1.0 / (1.0 + np.exp(5 * (x - 1.0)))
    pattern2 = pattern2 * (n_points / 2) / np.sum(pattern2)
    candidates.append(pattern2)
    
    # Random candidates to increase diversity
    for _ in range(30):
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        candidates.append(h)
        
    return candidates

def differential_evolution_step(candidates, n_points, dx, F=0.5, CR=0.7):
    new_candidates = []
    num_candidates = len(candidates)
    
    for idx in range(num_candidates):
        # Select three distinct candidates
        candidates_idx = list(range(num_candidates))
        candidates_idx.remove(idx)
        idx1, idx2, idx3 = np.random.choice(candidates_idx, 3, replace=False)
        
        # Mutation
        mutant = candidates[idx1] + F * (candidates[idx2] - candidates[idx3])
        mutant = np.clip(mutant, 0, 1)
        mutant = mutant * (n_points / 2) / np.sum(mutant)
        
        # Crossover
        trial = candidates[idx].copy()
        for i in range(n_points):
            if np.random.rand() < CR:
                trial[i] = mutant[i]
        
        # Selection
        c5_old = evaluate_erdos_solution(candidates[idx], dx)
        c5_new = evaluate_erdos_solution(trial, dx)
        
        if c5_new < c5_old:
            new_candidates.append(trial)
        else:
            new_candidates.append(candidates[idx])
    
    return new_candidates

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 35
    dx = 2.0 / n_points
    
    # Generate initial candidates
    candidates = generate_initial_candidates(n_points, 100)
    
    best_c5 = float('inf')
    best_h = None
    
    # Evaluate initial candidates
    for h in candidates:
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Differential evolution main loop
    max_iter = 500
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        candidates = differential_evolution_step(candidates, n_points, dx)
        
        # Update best
        for h in candidates:
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Local refinement with SLSQP
    if best_h is not None:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try a few restarts
        for _ in range(10):
            if time() - start_time > budget_s * 0.95:
                break
            try:
                # Perturb best solution
                h_restart = best_h + np.random.normal(0, 0.005, n_points)
                h_restart = np.clip(h_restart, 0, 1)
                h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
                
                res = minimize(objective, h_restart, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 500, 'ftol': 1e-9})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_c5 = c5_refined
                        best_h = h_refined
            except:
                continue
    
    # Ensure valid result
    if best_h is None:
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
        else:
            best_h = np.ones(n_points) * 0.5
            best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
