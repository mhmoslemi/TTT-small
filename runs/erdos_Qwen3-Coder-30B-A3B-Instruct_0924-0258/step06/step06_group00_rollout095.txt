import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try multiple different strategies
    strategies = [
        "random_local",
        "structured_local", 
        "fourier_refine"
    ]
    
    for strategy in strategies:
        if time() - start_time > budget_s * 0.95:
            break
            
        h_current = h_start.copy()
        max_restarts = 25 if strategy == "random_local" else 15
        
        # Special handling for fourier strategy
        if strategy == "fourier_refine":
            # Apply Fourier-based refinement
            h_refined = h_current.copy()
            try:
                # FFT-based frequency filtering
                H = np.fft.fft(h_refined)
                # Apply Hamming window to suppress high frequencies
                window = np.hamming(len(H))
                H_filtered = H * window
                h_filtered = np.real(np.fft.ifft(H_filtered))
                h_refined = np.clip(h_filtered, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
            except Exception:
                pass
                
        # Local optimization for other strategies
        if strategy != "fourier_refine":
            for restart in range(max_restarts):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # For structured approach, create more balanced distribution
                if strategy == "structured_local":
                    h_restart = np.ones(n_points) * 0.5
                    # Create alternating pattern
                    for i in range(0, n_points, 2):
                        if i < n_points:
                            h_restart[i] = 0.7
                    # Project to integral constraint  
                    h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                else:  # random_local
                    h_restart = h_current + np.random.normal(0, 0.02, n_points)
                    h_restart = np.clip(h_restart, 0, 1)
                    h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                
                # Local optimization 
                try:
                    def objective(h):
                        return evaluate_erdos_solution(h, dx)
                    
                    def constraint_sum(h):
                        return np.sum(h) - n_points / 2.0
                    
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    bounds = [(0, 1) for _ in range(n_points)]
                    res = minimize(objective, h_restart, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 300, 'ftol': 1e-9})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass

    # Final optimization to clean up solution
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
