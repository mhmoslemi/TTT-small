import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Homotopy method with convex relaxation
    max_homotopy_steps = 100
    t_values = np.linspace(0, 1, max_homotopy_steps)
    
    h_current = h_start.copy()
    current_c5 = best_c5
    
    for t_idx, t in enumerate(t_values):
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            # Convex relaxation: t*C5 + (1-t)*||h - uniform||^2
            h_opt = cp.Variable(n_points)
            
            # Objective: convex relaxation
            corr = cp.sum(cp.multiply(h_opt[:n_points-1], 1 - h_opt[1:]))
            for k in range(2, n_points):
                corr += cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            
            obj = t * corr + (1 - t) * cp.sum_squares(h_opt - 0.5 * np.ones(n_points))
            
            # Constraints
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Solve relaxed problem
            prob = cp.Problem(cp.Minimize(obj), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                # Renormalize to exact constraint
                h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                
                if c5_feasible < current_c5:
                    h_current = h_feasible.copy()
                    current_c5 = c5_feasible
                    if c5_feasible < best_c5:
                        best_h = h_feasible.copy()
                        best_c5 = c5_feasible
        except Exception:
            pass
    
    # Local refinement with trust-region
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use previous solution as starting point
        res = minimize(objective, h_current, method='trust-constr', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'gtol': 1e-8, 'xtol': 1e-8})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
