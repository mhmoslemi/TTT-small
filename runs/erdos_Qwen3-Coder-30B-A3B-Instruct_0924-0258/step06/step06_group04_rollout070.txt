import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Precompute correlation indices for faster computation
    def compute_overlap(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Hybrid approach: alternating between gradient and random search
    # Initial pass with gradient-based optimization
    max_iter = 1000
    lr = 0.1
    decay = 0.995
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation with adaptive step size
        perturbation = np.random.normal(0, lr, n_points)
        h_new = best_h + perturbation
        h_new = np.clip(h_new, 0, 1)
        # Normalize to satisfy integral constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = compute_overlap(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            
        # Adapt learning rate
        lr = max(1e-4, lr * decay)
    
    # Coarse search with multiple random restarts
    for restart in range(20):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random initialization with constraint
        h_rand = np.random.random(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        h_rand = np.clip(h_rand, 0, 1)  
        
        c5_rand = compute_overlap(h_rand)
        if c5_rand < best_c5:
            best_h = h_rand
            best_c5 = c5_rand
    
    # Final quadratic optimization refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Simple linear constraints on key lags
        for k in range(1, n_points//2 + 1, max(1, n_points//20)):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = compute_overlap(h_feasible)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final normalization and evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = compute_overlap(best_h)
    
    return (list(best_h), final_c5, n_points)
