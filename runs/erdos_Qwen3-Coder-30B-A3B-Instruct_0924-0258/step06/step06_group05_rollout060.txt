import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    T_init = 1.0
    T_decay = 0.995
    repair_freq = 50
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # Use initial solution as first particle
    particles.append(h_start.copy())
    velocities.append(np.zeros(n_points))
    pbest.append(h_start.copy())
    pbest_values.append(best_c5)
    
    # Add other random particles
    for i in range(1, n_particles):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_values.append(evaluate_erdos_solution(h, dx))
        
    # Global best tracking
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO main loop
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        T = T_init * (T_decay ** iter)
        
        for p_idx in range(n_particles):
            h = particles[p_idx]
            v = velocities[p_idx]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = w * v + c1 * r1 * (pbest[p_idx] - h) + c2 * r2 * (gbest - h)
            
            # Update position
            h_new = h + v
            
            # Repair and normalize
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new solution
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Update personal best
            if c5_new < pbest_values[p_idx]:
                pbest[p_idx] = h_new.copy()
                pbest_values[p_idx] = c5_new
                
                if c5_new < gbest_value:
                    gbest = h_new.copy()
                    gbest_value = c5_new
                    if gbest_value < best_c5:
                        best_h = gbest.copy()
                        best_c5 = gbest_value
                        
            # Simulated annealing
            if c5_new > pbest_values[p_idx] and np.random.rand() < np.exp(-(c5_new - pbest_values[p_idx]) / max(T, 1e-10)):
                pbest[p_idx] = h_new.copy()
                pbest_values[p_idx] = c5_new
                
                if c5_new < gbest_value:
                    gbest = h_new.copy()
                    gbest_value = c5_new
                    if gbest_value < best_c5:
                        best_h = gbest.copy()
                        best_c5 = gbest_value
            
            particles[p_idx] = h_new
            velocities[p_idx] = v
            
        # Convex relaxation periodically
        if iter % repair_freq == 0 and iter > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample k values for constraints
                k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=200)
                
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
                        # Update particles with feasible solution
                        particles[0] = h_feasible.copy()
                        velocities[0] = np.zeros(n_points)
                        pbest[0] = h_feasible.copy()
                        pbest_values[0] = c5_feasible
                        
            except Exception:
                pass

    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
