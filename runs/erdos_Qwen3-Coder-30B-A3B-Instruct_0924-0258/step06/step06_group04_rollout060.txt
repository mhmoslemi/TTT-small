import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        n_points = len(h_start)

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Grid refinement
    resolutions = [10, 20, 35]  # Start from low res, go up to our target
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        # Initialization: interpolate or create sinusoidal
        if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
            h = np.array(initial_h_values)
        else:
            # Try to interpolate from best solution found so far
            if len(best_h) > 1 and n_points > len(best_h):
                x = np.linspace(0, 2, len(best_h))
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, x, best_h)
            elif len(best_h) > 1 and n_points < len(best_h):
                x = np.linspace(0, 2, len(best_h))
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, x, best_h)
            else:
                # Create sinusoidal pattern
                x = np.linspace(0, 2, n_points)
                h = 0.5 + 0.5 * np.sin(np.pi * x)  # Low overlap pattern
                h = np.clip(h, 0, 1)
    
        # Adjust to match sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Hybrid optimization: gradient method + simulated annealing
        max_iter = 200
        step_size = 0.1
        temp = 0.1
        decay_rate = 0.995
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Gradient-based update
            # For simplicity, we use finite differences for gradients
            h_new = h.copy()
            for j in range(n_points):
                # Small perturbations to estimate gradient
                h_plus = h.copy()
                h_minus = h.copy()
                eps = 1e-4
                h_plus[j] += eps
                h_minus[j] -= eps
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                # Renormalize
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                c5_minus = evaluate_erdos_solution(h_minus, dx)
                grad = (c5_plus - c5_minus) / (2 * eps)
                h_new[j] = h[j] - step_size * grad
                h_new[j] = np.clip(h_new[j], 0, 1)

            # Normalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
                        
            # Simulated Annealing Step
            if np.random.rand() < 0.05:  # 5% chance to perturb with SA
                noise = np.random.normal(0, temp, n_points)
                h_sa = h + noise
                h_sa = np.clip(h_sa, 0, 1)
                h_sa = h_sa * (n_points / 2) / np.sum(h_sa)
                c5_sa = evaluate_erdos_solution(h_sa, dx)
                if c5_sa < c5_new:
                    h_new = h_sa
                    c5_new = c5_sa
                    
            # Accept or reject new solution with SA acceptance probability
            if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
                h = h_new
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new

            temp *= decay_rate
            step_size = max(1e-6, step_size * decay_rate)
            
        # Final Check
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

            best_h = h
            best_c5 = c5

    # Binary search refinement
    target = 0.3807
    exact_refine = 0.38085  # Our current target

        low = target
        high = best_c5
        max_iter_refine = 20

        for _ in range(max_iter_refine):
            if time() - start_time > budget_s * 0.95:
                break
            mid = (low + high) / 2.0
            try:
                h_refine = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_refine) == n_points / 2.0,
                    h_refine >= 0,
                    h_refine <= 1
                ]

                # Constraint for max overlap should be less than or equal to mid
                corr = cp.sum(cp.multiply(h_refine[:n_points], 1 - h_refine[n_points:]))
                constraints.append(corr <= mid * dx + 1e-6)

                # Solve with ECOS
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_refine.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < mid:
                        high = mid
                    else:
                        low = mid
                else:
                    low = mid
            except Exception:
                low = mid

        best_c5 = low

    # Final normalization and validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)

    return (list(best_h), best_c5, n_points)
