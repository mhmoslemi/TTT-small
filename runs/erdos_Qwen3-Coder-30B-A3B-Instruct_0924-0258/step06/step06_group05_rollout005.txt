import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Homotopy-based optimization
    max_homotopy_steps = 100
    t_values = np.linspace(0, 1, max_homotopy_steps)
    
    h_current = h_start.copy()
    current_c5 = best_c5
    
    # For each homotopy step
    for t_idx, t in enumerate(t_values):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Define the relaxed problem
        h_opt = cp.Variable(n_points)
        uniform_h = np.ones(n_points) * (n_points/2) / n_points
        
        # Objective: t * C5 + (1-t) * penalty
        objective_expr = t * cp.MaxValue(cp.matmul(cp.diag(h_opt), 1 - h_opt[1:])) + (1-t) * cp.sum_squares(h_opt - uniform_h)
        
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add a few key correlation constraints to make it more manageable
        for k in range(1, min(10, n_points)):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        try:
            prob = cp.Problem(cp.Minimize(objective_expr), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
            if prob.status == cp.OPTIMAL:
                h_new = h_opt.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                if c5_new < current_c5:
                    h_current = h_new.copy()
                    current_c5 = c5_new
                    if c5_new < best_c5:
                        best_h = h_current.copy()
                        best_c5 = c5_new
                        
        except Exception:
            pass

    # Local refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use current best as starting point
        h_refined = h_current.copy()
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
