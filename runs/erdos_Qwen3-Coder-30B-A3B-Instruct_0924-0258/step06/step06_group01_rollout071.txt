import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    # Simulated Annealing parameters
    T0 = 1.0
    T_min = 1e-5
    alpha = 0.999
    max_iter_per_T = 100
    
    # Initial evaluation
    current_overlap = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_overlap = current_overlap
    T = T0
    iteration = 0
    
    while T > T_min and (time() - start_time) < budget_s * 0.9:
        for _ in range(max_iter_per_T):
            # Generate candidate solution
            h_candidate = h.copy()
            
            # Select two random indices
            i, j = random.sample(range(n_points), 2)
            
            # Perturb h[i]
            delta = random.gauss(0, 0.05)
            h_candidate[i] += delta
            
            # Adjust h[j] to maintain sum constraint
            # Sum constraint: sum(h) = n_points/2
            # If we change h[i] by delta, we should change h[j] by -delta * (h[i] / h[j]) 
            # But this isn't right. Simpler:
            # Let's just change both to keep sum constraint
            sum_diff = np.sum(h_candidate) - n_points/2
            if abs(sum_diff) > 1e-10 and n_points > 1:
                # Adjust h[j] to satisfy sum constraint
                h_candidate[j] = h_candidate[j] - sum_diff
                # Clip to [0,1]
                h_candidate = np.clip(h_candidate, 0, 1)
                # Rescale to maintain exactly sum = n_points/2
                h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
            
            # Evaluate candidate
            candidate_overlap = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept or reject according to simulated annealing
            delta = candidate_overlap - current_overlap
            if delta < 0 or np.exp(-delta / T) > random.random():
                h = h_candidate
                current_overlap = candidate_overlap
            
            # Update best solution
            if current_overlap < best_overlap:
                best_h = h.copy()
                best_overlap = current_overlap
            
            iteration += 1
        
        # Cool down
        T *= alpha
    
    # Final refinement with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_overlap = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    # Ensure feasibility
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "h values must be in [0,1]"
    assert np.isclose(np.sum(best_h), n_points/2), "Sum constraint violated"
    
    return (list(best_h), best_overlap, n_points)
