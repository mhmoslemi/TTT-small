```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    T0 = 1.0  # initial temperature
    alpha = 0.995  # cooling rate
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            h_init = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h_init = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points/2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h_init.copy())
        pbest_vals.append(evaluate_erdos_solution(h_init, dx))
    
    # Global best
    global_best_idx = np.argmin(pbest_vals)
    gbest = pbest[global_best_idx].copy()
    gbest_val = pbest_vals[global_best_idx]
    
    # PSO iterations
    T = T0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Repair to bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Repair to constraint
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            current_val = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                
                # Update global best
                if current_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_val
                    if gbest_val < best_c5:
                        best_c5 = gbest_val
                        best_h = gbest.copy()
            
            # Simulated Annealing acceptance
            if current_val > gbest_val:
                p_accept = np.exp(-(current_val - gbest_val) / T)
                if np.random.rand() < p_accept:
                    pbest[i] = particles[i].copy()
                    pbest_vals[i] = current_val
                    if current_val < gbest_val:
                        gbest = particles[i].copy()
                        gbest_val = current_val
                        if gbest_val < best_c5:
                            best_c5 = gbest_val
                            best_h = gbest.copy()
                            
        # Update temperature
        T *= alpha
        
        # Occasionally solve convex relaxation
        if iteration % 50 == 0 and iteration > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample a few lags for constraints
                lag_indices = np.random.choice(n_points//2, min(20, n_points//3), replace=False)
                for k in lag_indices:
                    if k > 0:
                        k_idx = int(k * n_points / 2)
                        if k_idx < n_points:
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k_idx], 1 - h_opt[k_idx:]))
                            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=200)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible.copy()
                        best_c5 = c5_feasible
            except Exception:
                pass
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
```
