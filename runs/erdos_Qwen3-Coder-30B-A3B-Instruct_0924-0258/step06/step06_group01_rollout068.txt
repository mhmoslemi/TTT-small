import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on the bound C5
    C_low = 0.0
    C_high = 0.38085
    tolerance = 1e-5
    
    # Use CVXOPT for convex optimization
    while time() - start_time < budget_s * 0.95 and C_high - C_low > tolerance:
        C_mid = (C_low + C_high) / 2.0
        
        # Construct the feasibility problem
        h_vars = cp.Variable(n_points)
        
        # Constraints for the correlation
        constraints = [cp.sum(h_vars) == n_points / 2]
        for k in range(n_points):
            # Compute the correlation term using CVXPY
            corr_term = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_term += h_vars[i] * (1 - h_vars[j])
            constraints.append(corr_term <= C_mid * n_points / 2)
        
        # Box constraints
        constraints.extend([h_vars >= 0, h_vars <= 1])
        
        # Dummy objective
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=500)
            if prob.status == cp.OPTIMAL:
                h_sol = h_vars.value
                # Project back to feasible set
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                current_c5 = evaluate_erdos_solution(h_sol, dx)
                if current_c5 < best_c5:
                    best_h = h_sol.copy()
                    best_c5 = current_c5
                C_high = C_mid
            else:
                C_low = C_mid
        except:
            C_low = C_mid

    # Final refinement using local optimization
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
