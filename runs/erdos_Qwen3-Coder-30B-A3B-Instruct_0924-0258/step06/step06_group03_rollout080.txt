import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n_points):
    """Create a triangular window function for initialization."""
    half = n_points // 2
    if n_points % 2 == 0:
        left = np.arange(1, half + 1)
        right = np.arange(half, 0, -1)
    else:
        left = np.arange(1, half + 2)
        right = np.arange(half, 0, -1)
    window = np.concatenate([left, right])
    return window * (n_points/2) / np.sum(window)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = triangular_window(n_points)
    
    dx = 2.0 / n_points
    
    # Ensure sum constraint
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Set up bounds for differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Differential Evolution parameters
    de_maxiter = 300
    de_popsize = 50
    
    # Optimization callback to track progress
    def callback(x, convergence):
        nonlocal best_h, best_c5
        h = np.array(x)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Perform differential evolution
    de_result = differential_evolution(
        lambda x: evaluate_erdos_solution(x * (n_points/2) / np.sum(x), dx),
        bounds,
        maxiter=de_maxiter,
        popsize=de_popsize,
        callback=callback,
        seed=seed,
        disp=False
    )
    
    # Refine with L-BFGS-B if time allows
    if time() - start_time < budget_s * 0.8:
        # Prepare constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        # Refine top individuals
        top_individuals = [best_h]
        for _ in range(9):  # Get 10 top individuals
            # Add small perturbations to get diverse points
            pert = np.random.normal(0, 0.01, n_points)
            h_pert = best_h + pert
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            top_individuals.append(h_pert)
            
        for h_ref in top_individuals:
            try:
                res = minimize(
                    lambda x: evaluate_erdos_solution(x, dx),
                    h_ref,
                    method='L-BFGS-B',
                    bounds=[(0, 1) for _ in range(n_points)],
                    options={'maxiter': 100, 'ftol': 1e-8}
                )
                if res.success:
                    h_final = res.x
                    h_final = h_final * (n_points / 2) / np.sum(h_final)
                    c5_final = evaluate_erdos_solution(h_final, dx)
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
