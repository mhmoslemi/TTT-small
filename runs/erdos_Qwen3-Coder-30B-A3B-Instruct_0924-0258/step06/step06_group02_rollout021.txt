import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a pattern that spreads mass
        h_start = np.ones(n_points) * 0.5
        # Add small random perturbations to avoid symmetry
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with structured updates
    max_iterations = 1500
    learning_rate = 0.01
    decay_factor = 0.995
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute gradient for the total overlap (smooth approximation)
        gradient = np.zeros(n_points)
        for k in range(1, n_points):
            # For each shift k, compute contribution to gradient
            if k < n_points:
                # h[i] interacts with (1 - h[i+k]) 
                for j in range(n_points - k):
                    gradient[j] += (1 - best_h[j + k])
                # h[i] also interacts with h[i-k] when i >= k
                for j in range(k, n_points):
                    gradient[j] -= best_h[j - k]
        
        # Update h
        new_h = best_h - learning_rate * gradient
        new_h = np.clip(new_h, 0, 1)
        
        # Project to correct integral
        sum_h = np.sum(new_h)
        if sum_h > 0:
            new_h = new_h * (n_points/2) / sum_h
            
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(new_h, dx)
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
            
        # Update learning rate
        learning_rate *= decay_factor
    
    # Final refinement with local search 
    try:
        # Try a few random perturbations to escape local minima
        for _ in range(10):
            if time() - start_time > budget_s * 0.95:
                break
            noise = np.random.normal(0, 0.01, n_points)
            trial_h = best_h + noise
            trial_h = np.clip(trial_h, 0, 1)
            sum_h = np.sum(trial_h)
            if sum_h > 0:
                trial_h = trial_h * (n_points/2) / sum_h
            c5_trial = evaluate_erdos_solution(trial_h, dx)
            if c5_trial < best_c5:
                best_h = trial_h.copy()
                best_c5 = c5_trial
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
