import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        return h
    scale = n / (2 * sum_h)
    h *= scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initial projection
    best_h = project_feasible(best_h, n_points)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Simulated Annealing with focused perturbations
    T = 0.5
    cooling = 0.95
    min_T = 1e-4
    max_iter = 2000
    iterations = 0
    
    while T > min_T and iterations < max_iter:
        if time() - start_time > budget_s * 0.80:
            break
            
        # Find indices with highest current overlaps
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        overlap_indices = np.argsort(corr)[-int(0.15 * n_points):]
        
        # Perturb selected indices
        h_new = best_h.copy()
        for _ in range(3):
            idx = np.random.choice(overlap_indices)
            step = np.random.choice([-0.1, 0.1])
            h_new[idx] += step
        
        h_new = project_feasible(h_new, n_points)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            best_h = h_new.copy()
            best_c5 = c5_new
        
        T *= cooling
        iterations += 1
    
    # QCP Refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add all overlap constraints
        for k in range(1, n_points):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final refinement
    try:
        h_final = project_feasible(best_h, n_points)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    # Validate
    h_valid = project_feasible(best_h, n_points)
    c5_final = evaluate_erdos_solution(h_valid, dx)
    
    return (list(h_valid), c5_final, n_points)
