import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # SDP-based initialization for better starting point
    try:
        h_sdp = cp.Variable(n_points)
        constraints = [
            cp.sum(h_sdp) == n_points / 2.0,
            h_sdp >= 0,
            h_sdp <= 1
        ]
        
        # Add overlap constraints for a few key k values to relax the problem
        step = max(1, n_points // 20)
        for k in range(step, min(n_points // 2, 15), step):
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_sdp[:n_points-k], 1 - h_sdp[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_sdp[k-n_points+1:], 1 - h_sdp[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= 0.5)  # Dummy constraint to make it solvable
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_start = np.clip(h_sdp.value, 0, 1)
            h_start = h_start * (n_points/2) / np.sum(h_start)
    except Exception:
        pass
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Trust Region Optimization approach with adaptive steps
    n_iterations = 0
    current_tr_radius = 0.1
    while time() - start_time < budget_s * 0.95 and n_iterations < 5000:
        n_iterations += 1
        
        # Perturb within trust region
        perturbation = np.random.uniform(-current_tr_radius, current_tr_radius, n_points)
        
        h_new = best_h + perturbation
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct integral
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (n_points/2) / h_new_sum

        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on merit
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            current_tr_radius = min(current_tr_radius * 1.1, 0.5)
        else:
            current_tr_radius *= 0.95
            
        if n_iterations % 20 == 0:
            current_tr_radius = max(current_tr_radius, 1e-6)
    
    # Final refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final_sum = np.sum(h_final)
        if h_final_sum > 0:
            h_final = h_final * (n_points/2) / h_final_sum
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
