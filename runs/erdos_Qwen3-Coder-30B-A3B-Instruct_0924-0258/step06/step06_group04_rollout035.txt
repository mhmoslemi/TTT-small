import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Grid refinement parameters
    n_start = 10
    n_max = 35
    n_points = n_start
    best_h = None
    best_c5 = float('inf')
    
    # Binary search parameters
    target_c5 = 0.3807
    binary_search_tolerance = 1e-6
    
    # Try different grid resolutions
    while n_points <= n_max:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize h
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            if len(h_init) < n_points:
                h = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, len(h_init)), h_init)
            else:
                h = h_init[:n_points]
        else:
            x = np.linspace(0, 2, n_points)
            # Sine-based initialization
            h = 0.5 + 0.3 * np.sin(3 * np.pi * x)
            h = np.clip(h, 0, 1)
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Optimization loop
        for iter in range(100):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Gradient approximation using finite differences
            c5_original = evaluate_erdos_solution(h, 2.0/n_points)
            grad = np.zeros(n_points)
            epsilon = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += epsilon
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                c5_plus = evaluate_erdos_solution(h_plus, 2.0/n_points)
                
                grad[i] = (c5_plus - c5_original) / epsilon
            
            # Gradient descent step (with projection)
            alpha = 0.05
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Keep new if it's better
            c5_new = evaluate_erdos_solution(h_new, 2.0/n_points)
            if c5_new < c5_original:
                h = h_new
                c5_original = c5_new
            
            # Occasionally add noise like simulated annealing
            if np.random.rand() < 0.05:
                noise = np.random.normal(0, 0.02, n_points)
                h_new = h + noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, 2.0/n_points)
                if c5_new < c5_original:
                    h = h_new
                    c5_original = c5_new
        
        # Final evaluation
        final_c5 = evaluate_erdos_solution(h, 2.0/n_points)
        
        if final_c5 < best_c5:
            best_h = h.copy()
            best_c5 = final_c5
            
        # Increase resolution
        n_points = min(n_points * 2, n_max)
    
    # Refine using convex optimization if applicable
    if best_h is not None and best_c5 > target_c5:
        try:
            h_opt = cp.Variable(len(best_h))
            constraints = [
                cp.sum(h_opt) == len(best_h) / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add constraints for multiple lags
            dx = 2.0 / len(best_h)
            for k in range(1, len(best_h)):
                lag = min(k, len(best_h) - k)
                if lag == 0:
                    continue
                corr_sum = cp.sum(cp.multiply(h_opt[:len(best_h)-lag], 1 - h_opt[lag:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, 2.0 / len(best_h))
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except:
            pass
    
    # Final verification and binary search refinement
    if best_h is not None:
        dx = 2.0 / len(best_h)
        final_c5 = evaluate_erdos_solution(best_h, dx)
        
        # Binary search around the solution to tighten bound
        low, high = 0.0, final_c5
        for _ in range(20):
            if time() - start_time > budget_s * 0.95:
                break
            mid = (low + high) / 2.0
            # Simple feasibility check
            eval_c5 = evaluate_erdos_solution(best_h, dx)
            if eval_c5 <= mid:
                high = mid
            else:
                low = mid
            if abs(high - low) < binary_search_tolerance:
                break
        
        final_c5 = high
        
        # Normalize again
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (len(best_h) / 2) / np.sum(best_h)
        final_c5 = evaluate_erdos_solution(best_h, dx)
    
    if best_h is None:
        # Fallback with basic construction
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.3 * np.sin(3 * np.pi * x)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_h = h
        best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    return (list(best_h), best_c5, len(best_h))
