import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use Hamming window as a better starting point
        h_start = 0.54 + 0.46 * np.cos(np.pi * (np.arange(n_points) / n_points - 0.5) * 2)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Projected gradient descent
    max_iter = 1000
    eta = 0.01
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Get the current maximum correlation lag
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(best_h) - 1)
        
        # Evaluate gradient approximation
        # For simplicity, we use finite differences to approximate gradient
        eps = 1e-4
        grad = np.zeros_like(best_h)
        for j in range(len(best_h)):
            h_plus = best_h.copy()
            h_minus = best_h.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            h_plus = np.clip(h_plus, 0, 1) * (n_points/2) / np.sum(h_plus)
            h_minus = np.clip(h_minus, 0, 1) * (n_points/2) / np.sum(h_minus)
            val_plus = evaluate_erdos_solution(h_plus, dx)
            val_minus = evaluate_erdos_solution(h_minus, dx)
            grad[j] = (val_plus - val_minus) / (2 * eps)
        
        # Update with gradient descent step
        new_h = best_h - eta * grad
        # Project back to feasible set
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
