import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initialize with provided construction or create a simple one

    # Normalize h to satisfy integral constraint
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # Initial best solution
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # We keep track of all solutions evaluated to avoid re-calculation
    eval_count = 0
    
    # Differential evolution with bounds for [0,1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Custom constraint handling for integral constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Wrapper function for differential evolution that handles integral constraint
    def obj_de(h):
        # Ensure integral constraint
        h = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    # Run DE to global search
    de_iter = 0
    max_de_iter = 500

    while time() - start_time < budget_s * 0.9 and de_iter < max_de_iter:
        try:
            # Multi-start DE with different random seeds
            for _ in range(2):
                if time() - start_time > budget_s * 0.9:
                    break
                # Generate random starting points within bounds
                x0 = np.random.uniform(0, 1, n_points)
                x0 = x0 * (n_points/2) / np.sum(x0)
                
                result = differential_evolution(
                    obj_de, 
                    bounds, 
                    maxiter=200, 
                    popsize=15,
                    seed=seed + de_iter + _
                )
                
                if result.success:
                    h_candidate = result.x
                    h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
                    c5_candidate = obj_de(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
                eval_count += 1

        except Exception:
            break
    
    # Local refinement using SLSQP (Global to local)
    max_slsqp_refinements = 20
    refinement_count = 0
    
    while time() - start_time < budget_s * 0.95 and refinement_count < max_slsqp_refinements:
        try:
            # Generate restarts
            for i in range(10):
                if time() - start_time > budget_s * 0.95:
                    break
                # Slightly perturb best solution
                h_restart = best_h + np.random.normal(0, 0.01, n_points)
                h_restart = np.clip(h_restart, 0, 1)
                h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(objective, h_restart, method='SLSQP',

                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                refinement_count += 1
        except Exception:
            break
    
    # Final validation step
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        final_c5 = evaluate_erdos_solution(h_final, dx)
        best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
