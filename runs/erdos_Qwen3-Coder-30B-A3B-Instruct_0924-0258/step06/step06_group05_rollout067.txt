import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with random perturbations
    max_iter = 5000
    step_size = 0.05
    noise_decay = 0.95
    noise_level = 0.05
    patience = 0
    max_patience = 100
    
    h_current = best_h.copy()
    current_c5 = best_c5
    
    for idx in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlations to find worst case
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        max_corr = corr[k_max]
        
        # Compute gradient for the worst case
        gradient = np.zeros(n_points)
        k = k_max - (n_points - 1)  # convert to shift value k
        
        # For k > 0: gradient[i] += (1 - h[i+k]) - h[i-k]
        if k > 0:
            # forward parts
            gradient[:-k] += 1 - h_current[k:]
            # backward parts
            gradient[k:] -= h_current[:-k]
        elif k < 0:
            # backward parts
            gradient[abs(k):] += 1 - h_current[:-abs(k)]
            # forward parts
            gradient[:abs(k)] -= h_current[abs(k):]
        else:  # k == 0
            # This should not happen for non-zero correlations
            gradient[:] += -h_current[:]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 1e-8:
            gradient = gradient / grad_norm
        
        # Line search
        alpha = step_size
        h_new = h_current - alpha * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < current_c5:
            h_current = h_new.copy()
            current_c5 = c5_new
            patience = 0
            if c5_new < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_new
        else:
            patience += 1
            
        # Add noise periodically to escape local minima
        if idx % 50 == 0 and idx > 0:
            noise = np.random.normal(0, noise_level, n_points)
            h_current += noise
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
            current_c5 = evaluate_erdos_solution(h_current, dx)
            noise_level *= noise_decay
            patience = 0
            
        # Early termination
        if patience > max_patience:
            break
    
    # Final convex optimization refinement
    try:
        h_opt = cp.Variable(n_points, name="h")
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample k values for constraints
        k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
        for k_val in k_vals:
            k = int(k_val * n_points / 2)
            if k < n_points and k > 0:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
