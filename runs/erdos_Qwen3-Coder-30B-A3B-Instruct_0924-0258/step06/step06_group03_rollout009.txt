import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n_points):
    """Create a triangular window function for initialization."""
    center = n_points / 2.0
    window = np.array([max(0, 1 - abs(i - center) / (n_points / 2)) for i in range(n_points)])
    # Normalize
    window = window * (n_points / 2) / np.sum(window)
    return window

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # First, use differential evolution for global search
    de_maxiter = 100
    de_popsize = 30
    
    def de_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Add constraint to sum to n_points/2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Get a better initial solution using DE
    try:
        de_result = differential_evolution(
            de_objective,
            bounds,
            maxiter=de_maxiter,
            popsize=de_popsize,
            seed=seed,
            strategy='best1bin',
            atol=1e-6,
            rtol=1e-6,
            polish=True,
        )
        if de_result.success:
            h_de = np.clip(de_result.x, 0, 1)
            h_de = h_de * (n_points / 2) / np.sum(h_de)
            c5_de = de_objective(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    except Exception as e:
        pass
    
    # Then do local refinement for best found so far
    def refine_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def refine_constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Use L-BFGS-B for local refinement
    cons_refine = {'type': 'eq', 'fun': refine_constraint_sum}
    bounds_refine = [(0, 1) for _ in range(n_points)]
    
    # Try to refine the best solution found
    try:
        ref_res = minimize(
            refine_objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds_refine,
            constraints=cons_refine,
            options={'maxiter': 200, 'ftol': 1e-9}
        )
        if ref_res.success:
            h_refined = ref_res.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            c5_refined = refine_objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final check - normal verification might take time, so just return best solution
    return (list(best_h), best_c5, n_points)
