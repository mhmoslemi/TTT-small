import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)

    # Optimization parameters
    max_iterations = 10000
    patience = 100
    patience_counter = 0
    learning_rate = 0.1
    momentum = 0.9
    conv_radius = 0.01
    
    # Gradient tracking
    velocity = np.zeros_like(h)
    
    # Main optimization loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            break
            
        # Compute overlap correlations
        corr_h = np.correlate(h, 1 - h, mode='full')
        max_k_idx = np.argmax(corr_h)
        max_k = max_k_idx - (len(h) - 1)
        
        # Compute gradient approximation
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_shifted = np.roll(h, max_k)
            h_shifted = np.roll(h_shifted, -i)
            grad[i] = (1 - h[i]) - h_shifted[i] if i < len(h) else -h_shifted[i]
            
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h_new = h + velocity
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h_new)
        h_new = h_new * scale_factor
        
        # Ensure sum equals exactly n_points/2
        h = h_new
        
    # Final polishing with scipy
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
