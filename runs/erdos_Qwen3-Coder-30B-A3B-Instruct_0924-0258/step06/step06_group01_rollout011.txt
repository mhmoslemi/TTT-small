import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with projected updates
    max_iter = 10000
    alpha = 0.05  # learning rate
    beta = 0.9    # momentum
    epsilon = 1e-8
    momentum_v = np.zeros(n_points)
    
    def compute_gradient(h, dx):
        # This is an approximation of the gradient
        # For true gradient we would need to compute derivatives of correlations
        # Since this is complex, we approximate with finite differences
        h = h.copy()
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            val_plus = evaluate_erdos_solution(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            val_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (val_plus - val_minus) / (2 * eps)
        return grad
    
    iter_count = 0
    last_improvement = 0
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute gradient
        grad = compute_gradient(best_h, dx)
        
        # Update momentum
        momentum_v = beta * momentum_v + (1 - beta) * grad
        
        # Update h
        h_new = best_h - alpha * momentum_v
        
        # Project onto [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to sum to n_points/2
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = iter_count
        
        iter_count += 1
        
        if iter_count - last_improvement > 1000:
            break

    # Final polishing with local optimization
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception as e:
        pass

    return (list(best_h), best_c5, n_points)
