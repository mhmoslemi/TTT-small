import numpy as np
from scipy.optimize import minimize
from scipy import optimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def proximal_operator(z, u, rho, n_points):
    # Solve min_z (rho/2)||z - (h - u)||_2^2 + (1/rho)*C5(z)
    # s.t. sum(z) = n_points/2, 0 <= z[i] <= 1
    # For our specific problem, just project onto constraints:
    # 1. Sum constraint: project to sum = n_points/2
    # 2. Bound constraints: clip to [0,1]
    
    # Apply proximal step to get updated h
    h = z - (1.0/rho) * u
    
    # Project onto bounds [0,1]
    h = np.clip(h, 0, 1)
    
    # Project onto sum constraint
    h_sum = np.sum(h)
    if abs(h_sum - n_points/2) > 1e-10:
        scaling = (n_points/2) / h_sum
        h = h * scaling
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    # ADMM parameters
    rho = 1e-4
    max_iter = 1000
    tol = 1e-6
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # ADMM variables
    z = h.copy()
    u = np.zeros(n_points)
    
    # ADMM iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Store previous h for convergence check
        h_prev = h.copy()
        
        # Primal update: solve convex optimization problem
        try:
            # This gradients calculation is simplified for efficiency
            # Direct evaluation via linear program would be more robust
            # but this heuristic approach is faster
            h_new = h.copy()
            
            # Local refinement of current solution
            cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Try a local optimizer
            res = minimize(
                lambda x: evaluate_erdos_solution(x, dx),
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'ftol': 1e-8, 'maxiter': 200}
            )
            
            if res.success:
                h_new = res.x
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Update h using proximal operator
            h = h_new
            
            # Update z and u
            z = h.copy()
            u = u + h - z
            
            # Check for convergence
            if np.linalg.norm(h - h_prev) < tol:
                break
                
        except Exception as e:
            pass
            
        # Update best solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final refinement using SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
