import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T = 0.5
    T_min = 1e-4
    cooling_rate = 0.99
    steps = 2000
    
    for i in range(steps):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Focused perturbation - focus on areas of high overlap
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Identify highest overlap regions
        top_indices = np.argsort(corr)[-max(1, n_points//10):]
        
        h_new = best_h.copy()
        # Perturb top overlap indices
        for idx in top_indices:
            if idx < n_points:
                # Perturb value with jitter
                delta = np.random.uniform(-0.1, 0.1)
                h_new[idx] += delta
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct integral
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Acceptance probability
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Cool down
        T = max(T_min, T * cooling_rate)
    
    # Gradient descent refinement
    def objective(h_vec):
        h = h_vec.reshape(-1)
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h_vec):
        h = h_vec.reshape(-1)
        return np.sum(h) - n_points/2
    
    # Create constraint dictionaries
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run optimization with rescaled h
    h_flat = best_h.flatten()
    res = minimize(objective, h_flat, method='SLSQP', bounds=bounds, constraints=constraints, 
                   options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-6})
    
    if res.success:
        final_h = res.x.reshape(-1)
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points/2) / np.sum(final_h)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
