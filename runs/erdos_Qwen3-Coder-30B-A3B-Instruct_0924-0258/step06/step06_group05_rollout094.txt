import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with random perturbations
    max_iter = 10000
    step_size = 0.05
    noise_level = 0.05
    patience = 0
    max_patience = 100
    
    h_current = best_h.copy()
    current_c5 = best_c5
    
    for idx in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Identify the worst shift k
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        
        # Compute gradient for current worst shift
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                gradient[i] += (1 - h_current[i + k_max])
            if i - k_max >= 0:
                gradient[i] -= h_current[i - k_max]
        
        # Normalize gradient and take step
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 1e-10:
            gradient = gradient / grad_norm
        h_new = h_current + step_size * gradient
        
        # Add noise periodically
        if idx % 20 == 0:
            noise = np.random.normal(0, noise_level, n_points)
            h_new += noise
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < current_c5:
            h_current = h_new.copy()
            current_c5 = c5_new
            patience = 0
            
            if c5_new < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_new
                print(f"New best C5: {best_c5:.6f} at iteration {idx}")
        else:
            patience += 1
            if patience > max_patience:
                # Reset with noise
                h_current = best_h.copy()
                h_current += np.random.normal(0, noise_level*0.5, n_points)
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points/2) / np.sum(h_current)
                current_c5 = evaluate_erdos_solution(h_current, dx)
                patience = 0
                
        # Decrease noise level over time
        noise_level *= 0.9995
    
    # Final QCP refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample k values for constraints
        sample_k = np.linspace(0.1, 1.9, min(20, n_points//4)).astype(int)
        for k in sample_k:
            if k < n_points and k > 0:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=300)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
                print(f"QCP refinement improved C5 to {best_c5:.6f}")
    except Exception:
        pass
    
    # Final SLSQP optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
                print(f"SLSQP refinement improved C5 to {best_c5:.6f}")
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
