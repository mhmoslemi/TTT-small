import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population
    population_size = 100
    population = []
    
    # Add initial solution
    h_normalized = h_start * (n_points/2) / np.sum(h_start)
    population.append(h_normalized)
    
    # Generate diverse initial candidates
    for _ in range(population_size - 1):
        h_candidate = h_normalized + np.random.normal(0, 0.05, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        population.append(h_candidate)
    
    # DE parameters
    F = 0.5
    CR = 0.7
    max_iter = 500
    
    # DE optimization
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        new_population = []
        
        for i, target in enumerate(population):
            # Select three different vectors
            candidates = list(range(population_size))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Mutation
            mutant = target + F * (population[a] - population[b])
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (n_points/2) / np.sum(mutant)
            
            # Crossover
            trial = np.copy(target)
            for j in range(n_points):
                if np.random.random() < CR or j == n_points - 1:
                    trial[j] = mutant[j]
            
            # Selection
            trial_c5 = evaluate_erdos_solution(trial, dx)
            target_c5 = evaluate_erdos_solution(target, dx)
            
            if trial_c5 < target_c5:
                new_population.append(trial)
            else:
                new_population.append(target)
        
        population = new_population
        
        # Keep the best solution so far
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    
    # Local refinement with SLSQP on top solutions
    top_solutions = sorted(population, key=lambda h: evaluate_erdos_solution(h, dx))[:10]
    
    for h_refine in top_solutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_points/2) / np.sum(h_refine)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_refine, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            if res.success:
                h_opt = res.x
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
