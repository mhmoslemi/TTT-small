import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Homotopy method to escape local optima
    max_homotopy_steps = 100
    t_values = np.linspace(0, 1, max_homotopy_steps)
    
    h_current = h_start.copy()
    
    for t in t_values:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Define optimization variables
        h = cp.Variable(n_points)
        
        # Objective: weighted combination
        obj = t * cp.max(cp.multiply(h[:-1], 1-h[1:])) + (1-t) * cp.sum_squares(h - 0.5 * np.ones(n_points))
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Solve the problem
        try:
            prob = cp.Problem(cp.Minimize(obj), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=200)
            
            if prob.status == cp.OPTIMAL:
                h_solution = h.value
                # Ensure feasibility
                h_solution = np.clip(h_solution, 0, 1)
                h_solution = h_solution * (n_points/2) / np.sum(h_solution)
                c5_solution = evaluate_erdos_solution(h_solution, dx)
                
                if c5_solution < best_c5:
                    best_h = h_solution
                    best_c5 = c5_solution
                    
                h_current = h_solution
            else:
                # Fallback to previous solution in case of solver failure
                pass
                
        except Exception:
            pass

    # Final refinement using local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
