import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Homotopy method with convex relaxation
    max_homotopy_steps = 100
    t_values = np.linspace(0, 1, max_homotopy_steps)
    
    h_current = h_start.copy()
    
    # Precompute uniform distribution for regularization
    uniform_h = np.ones(n_points) * (n_points / 2) / n_points
    
    for t in t_values:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Build convex relaxation problem
        h_opt = cp.Variable(n_points)
        
        # Objective: weighted combination of original objective and regularization
        # We're approximating the original problem with a convex relaxation
        # The regularization makes it tractable and helps escape local optima
        obj = cp.Minimize(
            t * cp.max(cp.multiply(h_opt, 1 - h_opt)) + 
            (1 - t) * cp.sum_squares(h_opt - uniform_h)
        )
        
        # Constraints
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Solve this step's convex relaxation
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            if prob.status == cp.OPTIMAL:
                h_new = h_opt.value
                # Basic repair to maintain feasibility
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                h_current = h_new.copy()
        except Exception:
            pass
        # Adaptive step size control based on progress
        if t > 0.7 and time() - start_time > budget_s * 0.8:
            break
    
    # Final refinement with direct optimization
    try:
        # Function to minimize (C5)
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        # Constraint ensuring sum = n_points / 2
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a more robust optimizer for final improvement
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-6})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
