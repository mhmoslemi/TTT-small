import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        return h
    scale = n / (2 * sum_h)
    h *= scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with focused perturbations
    temp = 0.5
    cooling_rate = 0.95
    max_iterations = 2000
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.80:
            break
            
        # Identify indices with high overlap contributions
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Focus on central region for perturbation
        mid_idx = len(corr) // 2
        start_idx = max(0, mid_idx - n_points//4)
        end_idx = min(len(corr), mid_idx + n_points//4)
        idxs = np.argsort(corr[start_idx:end_idx])[::-1][:n_points//10] + start_idx
        
        # Perturb selected indices
        h_new = best_h.copy()
        for idx in idxs:
            # Add small perturbation
            h_new[idx] += np.random.normal(0, 0.05)
        h_new = project_feasible(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        elif np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        temp *= cooling_rate
    
    # QCP refinement with all constraints
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Include all shifting constraints for tight bounds
        for k in range(1, n_points):
            if k <= n_points//2:
                corr_k = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_k = cp.sum(cp.multiply(h_opt[k:], 1 - h_opt[:n_points-k]))
            constraints.append(corr_k <= best_c5 * dx + 1e-6)
            
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final refinement with gradient descent
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        h_final = best_h.copy()
        h_final = project_feasible(h_final, n_points)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
