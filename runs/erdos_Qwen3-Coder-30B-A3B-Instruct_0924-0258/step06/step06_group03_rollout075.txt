import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with structured approach if possible
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create triangular window as structured initialization
        h_start = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h_start[i] = 1.0 - abs(i - mid) / (mid + 1)
        # Normalize to satisfy constraint
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Use Differential Evolution for global search
    de_result = differential_evolution(
        lambda h: evaluate_erdos_solution(h, dx),
        bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        maxiter=300,
        popsize=50,
        seed=seed,
        tol=1e-8,
        recombination=0.9,
        mutation=0.7
    )
    
    if de_result.success:
        h_de = de_result.x
        c5_de = evaluate_erdos_solution(h_de, dx)
        if c5_de < best_c5:
            best_h = h_de
            best_c5 = c5_de
    
    # Select top solutions for refinement
    top_indices = np.argsort([evaluate_erdos_solution(h, dx) for h in [h_start, best_h]])[:10]
    
    # Refinement with L-BFGS-B
    for i in top_indices:
        if time() - start_time > budget_s * 0.9:
            break
            
        h_init = [h_start, best_h][i]
        
        try:
            # Define bounds (clipped)
            bounds_l = [(0, 1) for _ in range(n_points)]
            
            # Objective function for refinement
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            # Constraint for sum
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            # L-BFGS-B local refinement
            res = minimize(obj, h_init, method='L-BFGS-B', bounds=bounds_l, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-10})
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined
                    
        except Exception:
            pass
    
    # Final check and cleanup
    best_h = np.clip(best_h, 0, 1)
    
    return (list(best_h), best_c5, n_points)
