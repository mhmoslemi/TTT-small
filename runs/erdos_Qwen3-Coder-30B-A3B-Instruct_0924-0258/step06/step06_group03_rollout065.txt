import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent or create triangular window
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        # Validate parent construction
        assert np.all(h_start >= 0) and np.all(h_start <= 1)
        assert abs(np.sum(h_start) - n_points/2.0) < 1e-10
    else:
        n_points = 35
        # Create triangular window
        h_start = 1 - np.abs(np.arange(n_points) / (n_points - 1) - 0.5) * 2
        # Normalize to sum = n_points/2
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing settings
    T = 0.5
    cooling_rate = 0.99
    max_iter = 5000
    iteration = 0
    
    h_current = h_start.copy()
    c5_current = best_c5
    
    while iteration < max_iter and (time() - start_time) < budget_s * 0.8:
        # Generate candidate by perturbation
        delta = np.random.normal(0, T, n_points)
        h_candidate = h_current + delta
        
        # Convex normalization to maintain constraints
        h_candidate = np.clip(h_candidate, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h_candidate)
        h_candidate = h_candidate * scale_factor
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        delta_c5 = c5_candidate - c5_current
        if delta_c5 < 0 or random.random() < np.exp(-delta_c5 / T):
            h_current = h_candidate
            c5_current = c5_candidate
            
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
        
        # Cool down
        T *= cooling_rate
        iteration += 1
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use best solution from SA as starting point
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
