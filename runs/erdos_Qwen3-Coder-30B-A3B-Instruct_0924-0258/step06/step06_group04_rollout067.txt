import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a smooth initial pattern
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization (PSO) 
    population_size = 50
    max_iter = 200
    inertia_w = 0.7
    c1 = 1.4
    c2 = 1.4
    velocity_limit = 0.2
    
    # Initialize particles
    particles = np.random.rand(population_size, n_points) * 0.2 + 0.4
    for i in range(population_size):
        particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
    velocities = np.zeros((population_size, n_points))
    pbest = particles.copy()
    pbest_values = np.array([evaluate_erdos_solution(p, dx) for p in particles])
    
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO iterations
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Update particles
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Apply velocity limits
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            
            # Update position
            particles[i] += velocities[i]
            
            # Clip to valid range
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to meet sum constraint
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            current_value = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value
                    
    # Refine using SQP with adaptive constraint sampling
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample key lags based on current gbest
        corr_vals = np.correlate(gbest, 1 - gbest, mode='full')
        lags = np.arange(len(corr_vals))
        # Focus on lag positions with high overlap values
        top_k = min(20, len(corr_vals) // 4)
        top_lags = np.argsort(corr_vals)[-top_k:]
        
        # For each top lag, add constraint
        for k in top_lags:
            # For full correlation, handle circularity properly
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= gbest_value * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
                
    except Exception:
        pass
    
    # Final optimization of the best solution using scipy
    def obj_func(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_eq(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    # Use result from PSO as starting point
    result = minimize(obj_func, best_h, method='SLSQP', bounds=bounds, constraints=constraints, 
                      options={'maxiter': 100, 'ftol': 1e-9})
    
    if result.success:
        final_h = result.x
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
