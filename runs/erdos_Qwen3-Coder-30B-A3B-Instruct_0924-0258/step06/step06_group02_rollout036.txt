import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.zeros(n_points)
        # Use pattern inspired by binary codes and symmetry
        centers = [0.5, 1.0, 1.5]
        for c in centers:
            idx = int(c * n_points / 2)
            h_start[max(0, idx-5):min(n_points, idx+5)] = 0.8
        h_start = h_start * (n_points / (2 * np.sum(h_start)))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Trust Region Optimization with Merit Function
    tr_radius = 0.2
    merit_weight = 0.1
    max_iterations = 5000
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Adapt trust region size
        if i > 0 and i % 100 == 0:
            tr_radius = max(0.01, tr_radius * 0.95)
        
        # Generate perturbation within trust region
        perturbation = np.random.uniform(-tr_radius, tr_radius, n_points)
        h_new = best_h + perturbation
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to maintain integral constraint
        sum_h_new = np.sum(h_new)
        if sum_h_new > 0:
            h_new = h_new * (n_points / (2 * sum_h_new))
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Compute constraint violations
        violation = max(0, np.max(h_new - 1), np.min(h_new), abs(np.sum(h_new) - n_points/2))
        merit_new = c5_new + merit_weight * violation
        
        violation_best = max(0, np.max(best_h - 1), np.min(best_h), abs(np.sum(best_h) - n_points/2))
        merit_best = best_c5 + merit_weight * violation_best
        
        # Accept if improvement
        if merit_new < merit_best:
            best_h = h_new.copy()
            best_c5 = c5_new
            tr_radius = min(0.5, tr_radius * 1.1)
        else:
            tr_radius = max(0.001, tr_radius * 0.9)
        
        # Adjust merit weight dynamically
        if violation > 1e-4:
            merit_weight *= 1.2
        else:
            merit_weight *= 0.9
    
    # Final refinement with convex optimization
    try:
        h_sdp = cp.Variable(n_points)
        constraints = [
            cp.sum(h_sdp) == n_points / 2.0,
            h_sdp >= 0,
            h_sdp <= 1
        ]
        
        # Add overlap constraints for a subset of shifts
        # This is a simplified version to fit the budget
        shifts = list(range(1, min(n_points//2 + 1, 10)))
        for k in shifts:
            if k < n_points - k:
                corr_sum = cp.sum(cp.multiply(h_sdp[:n_points-k], 1 - h_sdp[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_sdp[k:], 1 - h_sdp[:n_points-k]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, verbose=False, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_sdp.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final validation and normalization
    try:
        h_final = np.clip(best_h, 0, 1)
        sum_h = np.sum(h_final)
        if sum_h > 0:
            h_final = h_final * (n_points / (2 * sum_h))
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
