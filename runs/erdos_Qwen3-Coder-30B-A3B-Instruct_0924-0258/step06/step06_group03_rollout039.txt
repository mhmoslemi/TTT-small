import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with existing construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create triangular window as base
        h_start = 1 - np.abs(np.arange(n_points) / (n_points-1) - 0.5) * 2
        # Add small random noise
        h_start += np.random.normal(0, 0.02, n_points)
        # Normalize to sum = n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # DE parameters
    popsize = 50
    maxiter = 300
    F = 0.7
    CR = 0.9
    
    # Reference to the objective function for DE
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Constraint function for normalization
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Ensure initial solution satisfies constraint
    h_start = h_start * (n_points/2) / np.sum(h_start)
    h_start = np.clip(h_start, 0, 1)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE optimization
    for iteration in range(maxiter):
        if time() - start_time > budget_s * 0.8:
            break
        
        # Use differential evolution
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=5,
            popsize=popsize,
            mutation=(0.5, 1.0),
            recombination=CR,
            atol=1e-8,
            seed=seed + iteration,
            disp=False
        )
        
        if de_result.success:
            h_de = de_result.x
            h_de = np.clip(h_de, 0, 1)
            h_de = h_de * (n_points/2) / np.sum(h_de)
            c5_de = evaluate_erdos_solution(h_de, dx)
            if c5_de < best_c5:
                best_h = h_de.copy()
                best_c5 = c5_de
        
        # Local refinement for several top candidates
        # Create several candidates based on top performers
        candidates = []
        # Select candidates from current best
        for _ in range(10):
            # Perturb the current best
            perturbation = np.random.normal(0, 0.01, n_points)
            h_pert = best_h + perturbation
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            candidates.append(h_pert)
        
        for candidate in candidates:
            try:
                # Local optimization
                res = minimize(
                    objective,
                    candidate,
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 50, 'ftol': 1e-9}
                )
                if res.success:
                    h_refined = res.x
                    h_refined = np.clip(h_refined, 0, 1)
                    h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = c5_refined
            except Exception:
                pass
    
    return (list(best_h), best_c5, n_points)
