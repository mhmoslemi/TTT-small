import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 100
    generations = 200
    elite_size = population_size // 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = population_size // 5
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        # Add noisy variants
        for _ in range(population_size - 1):
            h_noisy = h_start + np.random.normal(0, 0.1, n_points)
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = h_noisy * (n_points/2) / np.sum(h_noisy)
            population.append(h_noisy)
    else:
        for _ in range(population_size):
            h_rand = np.random.random(n_points)
            h_rand = h_rand * (n_points/2) / np.sum(h_rand)
            population.append(h_rand)
    
    best_in_generations = []
    stagnation_count = 0
    max_stagnation = 50
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count >= max_stagnation:
            break
            
        # Keep elite
        new_population = population[:elite_size].copy()
        
        # Tournament selection and reproduction
        while len(new_population) < population_size:
            # Tournament selection
            selected = []
            for _ in range(2):
                tournament = random.sample(range(population_size), tournament_size)
                tournament_fitness = [fitness[i] for i in tournament]
                winner_idx = tournament[np.argmin(tournament_fitness)]
                selected.append(population[winner_idx])
            
            # Crossover
            if random.random() < crossover_rate:
                child = np.zeros(n_points)
                for i in range(n_points):
                    if random.random() < 0.5:
                        child[i] = selected[0][i]
                    else:
                        child[i] = selected[1][i]
            else:
                child = selected[0].copy()
            
            # Mutation
            if random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            
            # Normalize sum
            child = child * (n_points/2) / np.sum(child)
            new_population.append(child)
        
        population = new_population
    
    # Local refinement on top candidates
    top_candidates = population[:elite_size]
    for h_ref in top_candidates:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, h_ref, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            continue
    
    return (list(best_h), best_c5, n_points)
