import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial solution - triangular window
        h_start = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h_start[i] = 1.0 - abs(i - mid) / (mid + 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution for global optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Properly formatted bounds for differential_evolution
    de_bounds = [(0, 1) for _ in range(n_points)]
    
    # DE optimization with termination based on time
    try:
        result = differential_evolution(
            objective, 
            de_bounds, 
            strategy='best1bin',
            maxiter=int(5000),  # Will kill early anyway
            popsize=30,
            mutation=(0.5, 1),
            recombination=0.7,
           _tol=1e-6,
            constraints=cons,
            seed=seed,
            callback=None
        )
        
        if result.success:
            h_de = result.x
            c5_de = objective(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    except Exception:
        pass
    
    # L-BFGS-B refinement of best solution found by DE
    if time() - start_time < budget_s * 0.9:
        try:
            # Apply bounds before optimization
            h_proj = np.clip(best_h, 0, 1)
            h_proj = h_proj * (n_points/2) / np.sum(h_proj)
            
            res = minimize(
                objective, 
                h_proj,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-8}
            )
            
            if res.success:
                h_refined = res.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    # Final check if we still have time
    if time() - start_time < budget_s * 0.95:
        try:
            # Try an alternative: direct minimization over a smaller subset of parameters
            # If we have a better initialization, it might help refine further
            # Try SLSQP with different initial guesses
            h_guess = best_h.copy()
            for _ in range(3):  # Try 3 different random initializations
                if time() - start_time > budget_s * 0.95:
                    break
                h_init = h_guess.copy()
                # Add small random perturbations
                h_init += np.random.normal(0, 0.01, n_points)
                h_init = np.clip(h_init, 0, 1)
                h_init = h_init * (n_points/2) / np.sum(h_init)
                
                res = minimize(
                    objective,
                    h_init,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 100, 'ftol': 1e-6}
                )
                if res.success:
                    h_candidate = res.x
                    h_candidate = np.clip(h_candidate, 0, 1)
                    h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
