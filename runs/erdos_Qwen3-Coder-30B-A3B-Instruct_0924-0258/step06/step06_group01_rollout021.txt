import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    T_min = 1e-5
    alpha = 0.999
    iterations_per_T = 100
    
    # Initialize current solution
    h_current = h_start.copy()
    current_c5 = best_c5
    
    # Main SA loop
    while T > T_min and (time() - start_time) < budget_s * 0.95:
        for _ in range(iterations_per_T):
            if (time() - start_time) > budget_s * 0.95:
                break
                
            # Generate candidate solution
            h_candidate = h_current.copy()
            i, j = random.sample(range(n_points), 2)
            
            # Small random perturbation
            delta = random.gauss(0, 0.05)
            h_candidate[i] += delta
            
            # Ensure it's still in valid range
            h_candidate[i] = np.clip(h_candidate[i], 0, 1)
            
            # Adjust other value to maintain sum constraint
            diff = h_candidate[i] - h_current[i]
            h_candidate[j] -= diff
            h_candidate[j] = np.clip(h_candidate[j], 0, 1)
            
            # Re-scale to meet sum constraint
            h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
            
            # Compute overlap
            candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept or reject based on simulated annealing
            delta_c5 = candidate_c5 - current_c5
            if delta_c5 < 0 or np.exp(-delta_c5 / T) > random.random():
                h_current = h_candidate.copy()
                current_c5 = candidate_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
                
        # Decrease temperature
        T *= alpha
    
    # Final refinement with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
