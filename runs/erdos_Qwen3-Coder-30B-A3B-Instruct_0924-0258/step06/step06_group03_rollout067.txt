import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create triangular window
        h_start = 1 - np.abs(np.arange(n_points) / (n_points - 1) - 0.5) * 2
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    max_iter = 5000
    T = 0.5
    alpha = 0.99
    min_T = 1e-8
    
    def accept_prob(delta_E, T):
        if delta_E < 0:
            return 1.0
        return np.exp(-delta_E / T)
    
    # SA loop
    h_current = best_h.copy()
    c5_current = best_c5
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Generate new candidate
        delta = np.random.normal(0, 0.02, n_points)
        h_candidate = h_current + delta
        
        # Ensure 0 <= h <= 1
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Normalize to sum to n_points/2
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        delta_E = c5_candidate - c5_current
        
        # Accept with probability
        if accept_prob(delta_E, T) > random.random():
            h_current = h_candidate.copy()
            c5_current = c5_candidate
            
        # Update best
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        # Cool down
        T = max(T * alpha, min_T)
    
    # Local refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Refine with SLSQP
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
