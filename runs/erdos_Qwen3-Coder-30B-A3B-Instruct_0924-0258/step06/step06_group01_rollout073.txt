import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

        h_start = np.random.rand(n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    T_min = 1e-5
    alpha = 0.999
    max_iter_per_temp = 100
    
    h_current = h_start.copy()
    current_c5 = best_c5
    
    # Simulated Annealing loop
    while T > T_min and time() - start_time < budget_s * 0.95:
        for _ in range(max_iter_per_temp):
            # Create neighbor solution
            h_new = h_current.copy()
            
            # Randomly perturb two indices
            i, j = random.sample(range(n_points), 2)
            
            # Perturb h[i]
            delta = random.uniform(-0.1, 0.1)
            h_new[i] += delta
            
            # Adjust h[j] to maintain sum constraint
            sum_diff = np.sum(h_new) - n_points / 2
            # We adjust j such that the sum constraint is satisfied
            if sum_diff != 0:
                h_new[j] -= sum_diff / (n_points - 1)
            
            # Clip values to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to maintain sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Compute new overlap
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject the new solution
            delta_c5 = new_c5 - current_c5
            if delta_c5 < 0 or np.exp(-delta_c5 / T) > random.random():
                h_current = h_new.copy()
                current_c5 = new_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
                
        # Cool down temperature
        T *= alpha
        
    # Final refinement with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
