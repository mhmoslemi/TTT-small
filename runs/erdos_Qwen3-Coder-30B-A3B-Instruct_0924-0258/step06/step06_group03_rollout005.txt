import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create triangular window initialization
        h_start = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h_start[i] = i / mid
            h_start[n_points-1-i] = i / mid
        h_start[mid] = 1.0
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Objective function for optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Constraint for sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints dictionary
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Differential Evolution for global search
    de_options = {'maxiter': 300, 'popsize': 50, 'tol': 1e-6, 'recombination': 0.9, 'mutation': (0.5, 1)}
    
    # Initial population with some randomization around the triangular window
    def initialize_population(popsize, n_points):
        population = []
        for _ in range(popsize):
            h = h_start.copy()
            # Add small random noise
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            # Normalize
            h = h * (n_points/2) / np.sum(h)
            population.append(h)
        return population
    
    # Try DE with multiple restarts
    for i in range(3):  # 3 restarts
        if time() - start_time > budget_s * 0.8:
            break
            
        de_result = differential_evolution(objective, bounds, constraints=cons, 
                                          maxiter=100, popsize=30, seed=seed+i,
                                          disp=False, polish=False)
        
        if not de_result.success:
            continue
            
        h_de = de_result.x
        c5_de = objective(h_de)
        if c5_de < best_c5:
            best_h = h_de.copy()
            best_c5 = c5_de
            
        # Local refinement with L-BFGS-B
        try:
            refinement = minimize(objective, h_de, method='L-BFGS-B', 
                                bounds=bounds, constraints=cons, 
                                options={'maxiter': 100, 'ftol': 1e-8})
            if refinement.success:
                h_refined = refinement.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
            
        # Try to improve with random restarts
        for _ in range(10):
            if time() - start_time > budget_s * 0.8:
                break
            h_trial = np.random.random(n_points)
            h_trial = h_trial * (n_points/2) / np.sum(h_trial)
            h_trial = np.clip(h_trial, 0, 1)
            c5_trial = objective(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
    
    # Final local optimization
    if time() - start_time < budget_s * 0.9:
        try:
            final_opt = minimize(objective, best_h, method='L-BFGS-B', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 200, 'ftol': 1e-9})
            if final_opt.success:
                h_final = final_opt.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
