import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search loop
    c_low = 0.0
    c_high = 0.38085
    tolerance = 1e-5
    
    def is_feasible(h, c):
        # Check if all overlaps are bounded by c * dx * n_points
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap <= c, max_overlap
    
    # Convex optimization approach for checking feasibility
    def conv_feasible(h_start, c_val):
        # Variables
        h = cp.Variable(n_points)
        # Constraints
        constraints = [cp.sum(h) == n_points / 2, h >= 0, h <= 1]
        
        # Add overlap constraints
        for k in range(n_points):
            # Compute correlation for shift k
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h[i] * (1 - h[j])
            constraints.append(expr <= c_val * n_points / 2)
        
        # Dummy objective
        obj = cp.Minimize(0)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=500)
            return prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INACCURATE
        except:
            return False

        pass

    except:
        pass
    
    return (list(best_h), best_c5, n_points)
