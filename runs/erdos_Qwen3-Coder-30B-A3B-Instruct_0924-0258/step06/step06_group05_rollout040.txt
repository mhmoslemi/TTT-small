import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Main optimization loop
    max_iter = 10000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Perturbation parameters
    noise_scale = 0.1
    noise_decay = 0.995
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        iter_count += 1
        
        # Compute all correlation values and find maximum
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        k_max = max_idx - (n_points - 1)
        
        # Compute gradient for the max correlation
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                gradient[i] += 1 - best_h[i + k_max]
            if i - k_max >= 0:
                gradient[i] -= best_h[i - k_max]
        
        # Normalize gradient to control step size
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm * 0.01
        
        # Perform line search
        step_size = 1.0
        h_trial = best_h - step_size * gradient
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        
        c5_trial = evaluate_erdos_solution(h_trial, dx)
        
        # Accept the step if it improves
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Periodic random perturbations
        if iter_count % 20 == 0:
            noise = np.random.normal(0, noise_scale, n_points)
            h_perturbed = best_h + noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
                no_improvement_count = 0
            
            noise_scale *= noise_decay
        
        # Occasionally solve convex relaxation
        if iter_count % 50 == 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add constraints for some random shifts
                k_vals = np.random.randint(1, n_points//2, min(15, n_points//3))
                for k_val in k_vals:
                    if k_val < n_points and k_val > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k_val], 1 - h_opt[k_val:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=200)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
        
        # Early termination if no improvement
        if no_improvement_count > max_no_improvement:
            break
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
