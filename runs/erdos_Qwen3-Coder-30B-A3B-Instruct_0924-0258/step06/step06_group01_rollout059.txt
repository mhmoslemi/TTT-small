import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.random.rand(n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    rho = 1e-4
    max_iterations = 1000
    tolerance = 1e-6
    iteration = 0
    
    # Initialize variables
    h = h_start.copy()
    z = h.copy()
    u = np.zeros(n_points)
    
    convergence_history = []
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Primal update: solve convex subproblem
        # min (rho/2) * ||z - h + u||_2^2 + (1/rho) * max_k sum h(x) * (1 - h(x+k)) dx
        # s.t. sum(z) = n_points/2, 0 <= z[i] <= 1
        
        # Create the optimization problem
        z_var = cp.Variable(n_points)
        objective = cp.Minimize(
            cp.sum_squares(z_var - h + u) * rho / 2 +
            cp.max(cp.matmul(cp.eye(n_points), z_var)) * (1/rho)  # simplified constraint
        )
        
        constraints = [
            cp.sum(z_var) == n_points/2,
            z_var >= 0,
            z_var <= 1
        ]
        
        # Simplified approach: optimize directly
        try:
            # Use the existing solution as a good starting point for local optimization
            cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Direct optimization using L-BFGS-B to minimize overlap integrals
            def objective_func(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            # Local refinement using SLSQP
            res = minimize(
                objective_func,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            
            if res.success:
                h_refined = res.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                new_c5 = evaluate_erdos_solution(h_refined, dx)
                if new_c5 < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = new_c5
                    
                # Update h with proximal step
                h = h_refined.copy()
                
            # Incremental update for ADMM-like step
            z = h.copy()
            u = u + h - z
            
        except Exception:
            pass
        
        # Check for convergence
        diff = np.linalg.norm(h - z)
        convergence_history.append(diff)
        if len(convergence_history) > 50:
            convergence_history.pop(0)
            
        if diff < tolerance:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            
        if no_improvement_count > 50:
            break
            
        iteration += 1
    
    # Final refinement with global optimization
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        
        if res.success:
            final_h = res.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points/2) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h.copy()
                best_c5 = final_c5
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
