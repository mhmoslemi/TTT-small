import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points/2 and h[i] in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points/2) / current_sum
    else:
        h = np.ones_like(h) * 0.5 * (n_points/2) / n_points
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Iterative refinement with convex relaxation
    max_iterations = 500
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Current correlation matrix
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr_idx = np.argmax(corr)
        
        # Create quadratic approximation: 
        # minimize sum_i sum_j h_i * h_j * corr[i+j] / dx
        # subject to sum(h) = n_points/2 and h[i] in [0,1]
        
        # Attempt to solve convex relaxation with CVXPY
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points/2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # use approximation of psi as quadratic with linear term
        # and approximate the objective differently
        
        # Simplified approach: direct optimization
        try:
            result = minimize(
                lambda x: evaluate_erdos_solution(x, dx),
                best_h,
                method='SLSQP',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2},
                options={'ftol': 1e-8, 'maxiter': 50}
            )
            if result.success:
                new_h = project_to_feasible(result.x, n_points)
                new_c5 = evaluate_erdos_solution(new_h, dx)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
        except Exception:
            pass
            
        # Occasionally try a perturbation
        if iteration % 30 == 0:
            perturbation = np.random.normal(0, 0.02, n_points)
            trial_h = best_h + perturbation
            trial_h = project_to_feasible(trial_h, n_points)
            trial_c5 = evaluate_erdos_solution(trial_h, dx)
            if trial_c5 < best_c5:
                best_h = trial_h.copy()
                best_c5 = trial_c5
                
        # Accelerated refinement
        if iteration % 10 == 0:
            try:
                cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
                res = minimize(
                    lambda x: evaluate_erdos_solution(x, dx),
                    best_h,
                    method='SLSQP',
                    bounds=[(0, 1) for _ in range(n_points)],
                    constraints=cons,
                    options={'ftol': 1e-9, 'maxiter': 100}
                )
                if res.success:
                    h_ref = project_to_feasible(res.x, n_points)
                    c5_ref = evaluate_erdos_solution(h_ref, dx)
                    if c5_ref < best_c5:
                        best_h = h_ref.copy()
                        best_c5 = c5_ref
            except Exception:
                pass

    return (list(best_h), best_c5, n_points)
