import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # ADMM parameters
    rho = 1e-4
    max_iterations = 1000
    convergence_tol = 1e-6
    iteration = 0
    
    # Initializations for ADMM
    z = h.copy()
    u = np.zeros(n_points)
    
    # ADMM iterations
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Primal update - solve convex subproblem using CVXPY
        try:
            # Define optimization variables
            z_var = cp.Variable(n_points)
            
            # Objective: minimize (rho/2) * ||z - h + u||_2^2 + (1/rho) * C5
            # But we need to approximate C5 term, so we linearize or use a proxy
            # For now, we'll solve a simpler version and refine later
            
            # The key is: minimize ||z - (h - u)||_2^2 subject to constraints
            # Plus we can approximate max correlation with a norm or update like this:
            
            # Simplified primal solve
            # minimize (rho/2)||z - (h - u)||_2^2 
            # subject to sum(z) = n_points/2 and 0 <= z <= 1
            objective = cp.Minimize(
                cp.norm2(z_var - (h - u))**2
            )
            
            constraints = [
                cp.sum(z_var) == n_points/2,
                z_var >= 0,
                z_var <= 1
            ]
            
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                z_new = z_var.value
                z = z_new
                
                # Update dual variables
                u = u + h - z
                
                # Update h for next iteration (proximal step)
                h = z - (1/rho) * (z - (h - u))
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                
                # Evaluate and check improvement
                current_c5 = evaluate_erdos_solution(h, dx)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                
                # Check for convergence
                if np.linalg.norm(h - z) < convergence_tol:
                    break
                    
        except Exception:
            # Fall back to simple update if optimization fails
            z = h.copy()
            z = np.clip(z, 0, 1)
            z = z * (n_points/2) / np.sum(z)
            
        iteration += 1
    
    # Final refinement
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
