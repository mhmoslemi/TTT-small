import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        return h
    scale = n / (2 * sum_h)
    h *= scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with focused perturbations
    T = 0.5
    cooling_rate = 0.95
    min_T = 1e-4
    max_iter = 2000
    
    # Modified version: focused perturbations based on high overlaps
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.80:
            break
            
        # Perturb only top 15% indices with highest correlations
        h_copy = best_h.copy()
        corr = np.correlate(h_copy, 1 - h_copy, mode='full')
        # Get high overlap positions
        max_idx = np.argmax(corr)
        # Perturb around the high overlap (shoot for little change)
        perturb_indices = np.argpartition(corr, -int(0.15 * len(corr)))[-int(0.15 * len(corr)):]
        # For each, add or reduce a bit
        step_size = 0.05 * T
        for idx in perturb_indices:
            if np.random.rand() > 0.5:
                h_copy[idx] += step_size
            else:
                h_copy[idx] -= step_size
        
        # Project to feasible set
        h_copy = project_feasible(h_copy, n_points)
        # Evaluate
        c5_new = evaluate_erdos_solution(h_copy, dx)
        
        # Accept with probability
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            best_h = h_copy.copy()
            best_c5 = c5_new
        
        # Cool temperature
        T = max(min_T, T * cooling_rate)
        
        if i % 100 == 0:
            # Periodically check time
            if time() - start_time > budget_s * 0.90:
                break
    
    # QCP Refinement with all constraints
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add overlap constraints for all shifts
        for k in range(1, n_points):
            if k <= n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final feasible projection
    try:
        h_final = project_feasible(best_h, n_points)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    # Verification step
    h_verified = project_feasible(best_h, n_points)
    c5_verified = evaluate_erdos_solution(h_verified, dx)
    
    return (list(h_verified), c5_verified, n_points)
