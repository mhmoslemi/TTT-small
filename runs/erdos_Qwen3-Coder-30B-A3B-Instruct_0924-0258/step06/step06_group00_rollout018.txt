import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 100
    generations = 200
    elite_count = population_size // 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = population_size // 5
    
    # Initialize population
    population = []
    population.append(h_start.copy())
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    last_improvement_gen = 0
    gen = 0
    
    while gen < generations and time() - start_time < budget_s * 0.95:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            last_improvement_gen = gen
            
        # Early termination
        if gen - last_improvement_gen > 50:
            break
            
        # Keep elite
        new_population = population[:elite_count]
        
        # Tournament selection and reproduction
        while len(new_population) < population_size:
            # Tournament selection
            selected_indices = np.random.choice(population_size, tournament_size, replace=False)
            selected_fitness = [fitness[i] for i in selected_indices]
            winner_idx = selected_indices[np.argmin(selected_fitness)]
            
            # Select another parent
            selected_indices2 = np.random.choice(population_size, tournament_size, replace=False)
            selected_fitness2 = [fitness[i] for i in selected_indices2]
            winner_idx2 = selected_indices2[np.argmin(selected_fitness2)]
            
            # Crossover
            if np.random.random() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([population[winner_idx][:crossover_point], 
                                       population[winner_idx2][crossover_point:]])
                child2 = np.concatenate([population[winner_idx2][:crossover_point], 
                                       population[winner_idx][crossover_point:]])
            else:
                child1 = population[winner_idx].copy()
                child2 = population[winner_idx2].copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                    child1[i] = max(0, min(1, child1[i]))
                if np.random.random() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.05)
                    child2[i] = max(0, min(1, child2[i]))
            
            # Normalize
            child1 = child1 * (n_points/2) / np.sum(child1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
        gen += 1
    
    # Local refinement
    for _ in range(10):  # Fine-tune top candidates
        if time() - start_time > budget_s * 0.95:
            break
        # Select top candidates  
        h_refine = best_h.copy()
        h_refine = np.clip(h_refine, 0, 1)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_refine, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
