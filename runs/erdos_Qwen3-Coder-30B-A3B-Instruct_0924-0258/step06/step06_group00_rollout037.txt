import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate diverse initial population
    def generate_initial_population(n_pop, n_points):
        pop = []
        # Start with initial solution
        h0 = h_start.copy()
        h0 = h0 * (n_points/2) / np.sum(h0)
        pop.append(h0)
        
        # Add random perturbations
        for _ in range(n_pop - 1):
            h_pert = h0 + np.random.normal(0, 0.05, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            pop.append(h_pert)
        return pop
    
    # Differential evolution optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Apply DE with multiple restarts
    n_pop = 50
    max_iter = 500
    n_restarts = 5
    
    for restart in range(n_restarts):
        if time() - start_time > budget_s * 0.90:
            break
            
        population = generate_initial_population(n_pop, n_points)
        
        # Custom differential evolution loop
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.90:
                break
                
            # Shuffle population
            random.shuffle(population)
            
            # DE strategy: rand/1/bin
            new_pop = []
            for i in range(n_pop):
                # Select three different individuals
                candidates = [p for j, p in enumerate(population) if j != i]
                if len(candidates) < 3:
                    candidates = population[:]
                a, b, c = random.sample(candidates, 3)
                
                F, CR = 0.7, 0.9
                # Mutation
                mutant = a + F * (b - c)
                mutant = np.clip(mutant, 0, 1)
                
                # Crossover
                trial = []
                j_rand = random.randint(0, n_points-1)
                for j in range(n_points):
                    if random.random() < CR or j == j_rand:
                        trial.append(mutant[j])
                    else:
                        trial.append(population[i][j])
                trial = np.array(trial)
                
                # Normalize
                trial = trial * (n_points/2) / np.sum(trial)
                
                # Selection
                if objective(trial) < objective(population[i]):
                    new_pop.append(trial)
                else:
                    new_pop.append(population[i])
            
            population = new_pop
            
        # Find best in this DE run
        for h in population:
            c5 = objective(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    
    # SLSQP refinement of best solution
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Local refinement
    max_refinements = 100
    for i in range(max_refinements):
        if time() - start_time > budget_s * 0.95:
            break
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-12})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            break

    return (list(best_h), best_c5, n_points)
