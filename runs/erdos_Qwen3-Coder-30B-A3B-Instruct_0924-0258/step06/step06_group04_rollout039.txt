import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Grid refinement
    resolutions = [10, 20, 35]
    for n in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize h for current resolution
        if n == 10:
            h_current = h_start[:10]
        elif n == 20:
            h_current = np.interp(np.linspace(0, 2, 20), np.linspace(0, 2, len(h_start)), h_start)
        else:
            h_current = h_start.copy()
        
        # Sinusoidal initialization for larger grids
        if n > len(h_current):
            x = np.linspace(0, 2, n)
            h_current = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n))
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n / 2) / np.sum(h_current)
        else:
            h_current = h_current[:n]
        
        # Gradient descent with simulated annealing
        max_iter = 100
        step_size = 0.1
        temp = 0.1
        decay_rate = 0.99
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Stochastic gradient descent
            h_new = h_current.copy()
            noise = np.random.normal(0, step_size, n)
            h_new += noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n / 2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, 2.0/n)
            
            # Acceptance probability for simulated annealing
            if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
                h_current = h_new
                if c5_new < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_new
            
            temp *= decay_rate
            step_size *= decay_rate
            
        h_current = h_current[:n]
        c5_current = evaluate_erdos_solution(h_current, 2.0/n)
        if c5_current < best_c5:
            best_h = h_current
            best_c5 = c5_current
    
    return (list(best_h), best_c5, len(best_h))
