import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with foveated perturbations
    T = 0.5
    T_min = 1e-4
    cooling_rate = 0.99
    steps = 2000
    step_count = 0
    
    while step_count < steps and time() - start_time < budget_s * 0.8:
        # Select indices with highest overlap for perturbation
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        # Get top 20% of indices by overlap
        top_indices = np.argsort(corr_full)[-max(1, n_points//5):]
        
        h_new = best_h.copy()
        # Perturb the top indices more significantly
        for i in top_indices:
            if np.random.rand() < 0.8:  # 80% chance to perturb
                delta = np.random.uniform(-0.1, 0.1)
                h_new[i] += delta
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct integral
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        elif np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Cool down
        T = max(T_min, T * cooling_rate)
        step_count += 1
    
    # Gradient refinement with projected gradient descent
    try:
        # Start from the best solution
        h_refine = best_h.copy()
        step_size = 1e-3
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Compute gradient of overlap w.r.t. h
            # This is a simplified approximation using finite differences
            grad = np.zeros(n_points)
            eps = 1e-6
            h_eps = h_refine.copy()
            
            for i in range(n_points):
                h_eps[i] += eps
                c5_plus = evaluate_erdos_solution(h_eps, dx)
                h_eps[i] -= 2*eps
                c5_minus = evaluate_erdos_solution(h_eps, dx)
                grad[i] = (c5_plus - c5_minus) / (2*eps)
                h_eps[i] += eps  # reset
            
            # Update using gradient descent
            h_refine -= step_size * grad
            # Project to feasible set
            h_refine = np.clip(h_refine, 0, 1)
            h_refine = h_refine * (n_points/2) / np.sum(h_refine)
            
            # Evaluate
            c5_refine = evaluate_erdos_solution(h_refine, dx)
            if c5_refine < best_c5:
                best_h = h_refine.copy()
                best_c5 = c5_refine
                
            step_size *= 0.995  # decay step size
            
    except Exception:
        pass
        
    # Final check and return
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
