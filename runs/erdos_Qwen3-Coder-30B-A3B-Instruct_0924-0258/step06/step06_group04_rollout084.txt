import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n_points))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search for C5
    low, high = 0.3807, 0.38085
    tolerance = 1e-6
    max_iter = 50
    
    for _ in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        C = (low + high) / 2.0
        
        # LP feasibility check 
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Sample key lags for constraints
            step = max(1, n_points // 10)
            k_list = list(range(1, n_points, step))
            k_list = [k for k in k_list if k < n_points // 2]
            
            # Include a few more lags
            k_list += [n_points//4, n_points//3, n_points//2]
            k_list = list(set(k_list))
            
            # Add constraints for these lags
            corr_vals = []
            for k in k_list:
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= C * dx + 1e-6)
                corr_vals.append(corr_sum)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
                high = C
            else:
                low = C
        except Exception:
            low = C
    
    # Post-optimization with local refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try multiple random restarts to explore
    for _ in range(10):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Add small noise to current best
        h_trial = best_h + np.random.normal(0, 0.01, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        
        result = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                         options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-6})
        
        if result.success:
            c5_trial = objective(result.x)
            if c5_trial < best_c5:
                best_h = result.x
                best_c5 = c5_trial
    
    # Final check and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
