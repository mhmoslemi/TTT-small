import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm Parameters
    population_size = 100
    generations = 200
    elite_size = population_size // 10
    mutate_prob = 0.1
    crossover_prob = 0.8
    tournament_size = population_size // 5
    
    # Initialize population
    population = []
    population.append(h_start.copy())
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # For early termination
    last_best = float('inf')
    no_improvement_count = 0
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        # Early termination
        if abs(fitness[0] - last_best) < 1e-8:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
        last_best = fitness[0]
        
        if no_improvement_count >= 50:
            break
            
        # Selection (tournament)
        new_population = population[:elite_size]
        while len(new_population) < population_size:
            # Tournament selection
            selected_indices = random.sample(range(population_size), tournament_size)
            selected_fitness = [fitness[i] for i in selected_indices]
            winner_idx = selected_indices[np.argmin(selected_fitness)]
            parent1 = population[winner_idx]
            
            # Another parent
            selected_indices = random.sample(range(population_size), tournament_size)
            selected_fitness = [fitness[i] for i in selected_indices]
            winner_idx = selected_indices[np.argmin(selected_fitness)]
            parent2 = population[winner_idx]
            
            # Crossover
            if random.random() < crossover_prob:
                crossover_point = random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
                
            # Mutation
            if random.random() < mutate_prob:
                for i in range(n_points):
                    if random.random() < 0.1:
                        child[i] += random.gauss(0, 0.05)
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                
            new_population.append(child)
            
        population = new_population
    
    # SLSQP refinement on best solutions
    for i in range(min(10, len(population))):
        h_candidate = population[i].copy()
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_candidate, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    
    return (list(best_h), best_c5, n_points)
