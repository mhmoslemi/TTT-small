import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def feasibility_correct(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points/2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Experiment with multiple strategies
    strategies = [
        lambda h, n: h,  # Identity
        lambda h, n: np.random.rand(n),  # Random
        lambda h, n: np.concatenate([np.ones(n//3), np.zeros(n - n//3)]),  # Block
        lambda h, n: np.concatenate([np.ones(n//2), np.zeros(n - n//2)]),  # Split
    ]
    
    # Multi-resolution grid
    resolutions = [30, 40, 50, 60]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        h_current = h_start.copy()
        if res > n_points:
            # Interpolate to higher resolution
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            h_current = np.interp(x_new, x_orig, h_current)
        elif res < n_points:
            # Downsample
            h_current = h_current[:res]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        h_current = feasibility_correct(h_current, n_points)
        
        # Multiple restarts with different strategies
        base_h = h_current.copy()
        
        for strategy in strategies:
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = strategy(base_h.copy(), n_points)
            h_restart = feasibility_correct(h_restart, n_points)
            
            max_restarts = 15
            for _ in range(max_restarts):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Add noise
                noise = np.random.normal(0, 0.05, n_points)
                h_perturbed = h_restart + noise
                h_perturbed = feasibility_correct(h_perturbed, n_points)
                
                # Local SLSQP optimization
                try:
                    def objective(h):
                        return evaluate_erdos_solution(h, dx)
                    
                    def constraint_sum(h):
                        return np.sum(h) - n_points / 2.0
                    
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    bounds = [(0, 1) for _ in range(n_points)]
                    
                    res = minimize(objective, h_perturbed, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 500, 'ftol': 1e-10})
                    if res.success:
                        h_refined = res.x
                        h_refined = feasibility_correct(h_refined, n_points)
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
                
                if time() - start_time > budget_s * 0.95:
                    break
    
    # Final refinement
    try:
        h_final = feasibility_correct(best_h, n_points)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-11})
        if res.success:
            h_refined = res.x
            h_refined = feasibility_correct(h_refined, n_points)
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
