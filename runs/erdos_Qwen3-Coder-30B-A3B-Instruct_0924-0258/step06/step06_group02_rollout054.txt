import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with foveated perturbations
    T = 0.5
    T_min = 1e-4
    cooling_rate = 0.99
    steps = 2000
    
    # Track recent improvements
    last_improvement = 0
    patience = 100
    
    for i in range(steps):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Calculate current overlap for focus selection
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Focus on highest overlap regions (top 10%)
        max_indices = np.argsort(corr)[-max(1, n_points // 10):]
        
        # Make perturbation to focused regions
        h_new = best_h.copy()
        for idx in max_indices:
            # Ensure we don't go out of bounds when perturbing
            if idx < n_points:
                perturbation = np.random.uniform(-0.1, 0.1)
                h_new[idx] += perturbation
                h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Adjust for sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        delta_c5 = c5_new - best_c5
        
        # Accept based on simulated annealing probability
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Update temperature
        T = max(T_min, T * cooling_rate)
        
        # Check for stagnation
        if delta_c5 < 0:
            last_improvement = 0
        else:
            last_improvement += 1
        if last_improvement > patience:
            # Restart with current best
            T = 0.5
            last_improvement = 0
    
    # Gradient descent refinement
    try:
        h_refined = best_h.copy()
        for step in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Compute gradient approximately using finite differences
            grad_h = np.zeros(n_points)
            epsilon = 1e-4
            base_c5 = evaluate_erdos_solution(h_refined, dx)
            
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += epsilon
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                grad_h[j] = (c5_plus - base_c5) / epsilon

            # Update step
            learning_rate = 1e-3 * (0.95 ** step)
            h_update = h_refined - learning_rate * grad_h
            
            # Apply constraints
            h_update = np.clip(h_update, 0, 1)
            h_update = h_update * (n_points/2) / np.sum(h_update)
            
            # Check improvement
            c5_update = evaluate_erdos_solution(h_update, dx)
            if c5_update < best_c5:
                best_h = h_update.copy()
                best_c5 = c5_update
            else:
                # If no improvement, reduce learning rate and continue
                if learning_rate < 1e-6:
                    break
            
    except Exception as e:
        pass
    
    # Final checkpoint
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
