import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate diverse initial candidates
    initial_candidates = [h_start]
    
    # Add noisy variants
    for i in range(5):
        noise = np.random.normal(0, 0.05, n_points)
        h_noise = h_start + noise
        h_noise = np.clip(h_noise, 0, 1)
        h_noise = h_noise * (n_points/2) / np.sum(h_noise)
        initial_candidates.append(h_noise)
    
    # Block initializations
    for i in range(3):
        h_block = np.zeros(n_points)
        h_block[:n_points//2] = 1
        h_block = np.roll(h_block, i*5)
        h_block = h_block * (n_points/2) / np.sum(h_block)
        initial_candidates.append(h_block)
    
    # Differential Evolution parameters
    bounds = [(0, 1) for _ in range(n_points)]
    
    def de_objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    # Run DE with multiple initial candidates
    for candidate in initial_candidates:
        if time() - start_time > budget_s * 0.9:
            break
            
        # Ensure candidate satisfies constraint
        candidate = candidate * (n_points/2) / np.sum(candidate)
        
        # Use differential evolution
        try:
            result = differential_evolution(
                de_objective,
                bounds,
                maxiter=200,
                popsize=20,
                mutation=(0.5, 1.0),
                recombination=0.7,
                seed=seed,
                tol=1e-6,
                callback=None,
                disp=False,
                polish=True
            )
            
            if result.success:
                h_refined = np.clip(result.x, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = de_objective(h_refined)
                
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            continue
    
    # Final SLSQP refinement of best
    if time() - start_time < budget_s * 0.95:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 1000, 'ftol': 1e-12}
            )
            
            if res.success:
                h_final = np.clip(res.x, 0, 1)
                h_final = h_final * (n_points/2) / np.sum(h_final)
                c5_final = objective(h_final)
                
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
