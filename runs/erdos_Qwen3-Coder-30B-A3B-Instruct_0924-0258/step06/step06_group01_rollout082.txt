import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.random.rand(n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / n_points)

    # ADMM parameters
    rho = 1e-4
    max_iterations = 1000
    convergence_tol = 1e-6
    iteration = 0

    h = h_start.copy()
    z = h.copy()
    u = np.zeros_like(h)
        
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Primal update: solve convex subproblem
        # Minimize (rho/2)||z - h + u||_2^2 + (1/rho) * [max_k (h * (1 - h_k))]
        # subject to sum(z) = n_points/2, 0 <= z[i] <= 1
        
        # Setup CVXPY problem for quadratic approximation
        z_var = cp.Variable(n_points)
        objective = cp.Minimize(
            cp.sum_squares(z_var - (h - u)) * rho / 2 +
            cp.maximum(cp.hstack([
                cp.sum(cp.multiply(z_var, 1 - np.roll(h, k))) for k in range(n_points)
            ])) / rho
        )
        constraints = [
            cp.sum(z_var) == n_points/2,
            z_var >= 0,
            z_var <= 1
        ]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                z_new = z_var.value
                # Ensure feasibility
                z_new = np.clip(z_new, 0, 1)
                z_new = z_new * (n_points/2) / np.sum(z_new)
            else:
                z_new = z
        except:
            z_new = z
            
        # Update h
        h = z_new - (1/rho) * (z_new - h + u)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Update z and u
        z = z_new
        u = u + h - z
        
        # Check convergence
        if np.linalg.norm(h - z) < convergence_tol:
            break
            
        # Update best solution
        current_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        iteration += 1

    # Final SLSQP refinement
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, 2.0 / n_points),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        if res.success:
            refined_h = res.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            current_c5 = evaluate_erdos_solution(refined_h, 2.0 / n_points)
            if current_c5 < best_c5:
                best_h = refined_h
                best_c5 = current_c5
    except:
        pass

    return (list(best_h), best_c5, n_points)
