import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Homotopy method with convex relaxation
    max_steps = 100
    t_vals = np.linspace(0, 1, max_steps)
    
    h_current = best_h.copy()
    current_c5 = best_c5
    
    for t in t_vals:
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            # Convex relaxation: minimize t*C5 + (1-t)*||h - uniform||^2
            h_opt = cp.Variable(n_points)
            uniform = np.ones(n_points) * (n_points / 2) / n_points
            
            # Objective
            obj = t * cp.quad_form(h_opt, np.eye(n_points))  # placeholder for C5
            # We'll compute the actual correlation later
            
            # For the convex reformulation, use a different approach
            # Create the convex relaxation using the dual angle approach
            
            # Instead of full convex formulation, use convex relaxation
            # Set up constraints
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Now we consider a proxy problem to avoid directly computing max overlap
            # Supervise the relaxed convex problem to converge towards better overlap
            
            # Reset the objective to be the standard convex proxy
            # Minimize the squared difference from uniform plus t * approx C5
            objective = cp.Minimize(
                cp.sum_squares(h_opt - uniform) + t * 0  # placeholder
            )
            
            # Use convex relaxation with affine constraints to step toward better solution
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=300)
            
            if prob.status == cp.OPTIMAL:
                h_temp = h_opt.value
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points/2) / np.sum(h_temp)
                
                c5_temp = evaluate_erdos_solution(h_temp, dx)
                if c5_temp < current_c5:
                    h_current = h_temp.copy()
                    current_c5 = c5_temp
                    
                if c5_temp < best_c5:
                    best_h = h_temp.copy()
                    best_c5 = c5_temp
        except Exception:
            pass

    # Refinement step: optimize the best found solution using local methods
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Global optimization with several restarts
        for _ in range(10):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Random restart initialization
            h_restart = np.random.rand(n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined

    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
