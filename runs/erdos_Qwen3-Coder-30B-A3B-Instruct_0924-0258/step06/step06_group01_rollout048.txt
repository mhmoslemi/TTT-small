import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.random.rand(n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # Initialize
    h = h_start.copy()
    current_overlap = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_overlap = current_overlap
    
    # Simulated Annealing parameters
    T = 1.0
    T_min = 1e-5
    alpha = 0.999
    max_iter_per_T = 100
    
    while T > T_min and (time() - start_time) < budget_s * 0.95:
        for _ in range(max_iter_per_T):
            # Generate neighbor: perturb two random indices
            i, j = random.randint(0, n_points-1), random.randint(0, n_points-1)
            h_new = h.copy()
            
            # Perturb h[i]
            delta = random.gauss(0, 0.05)
            h_new[i] += delta
            
            # Adjust h[j] to maintain sum constraint
            h_new[j] -= delta * (h[i] / h[j]) if h[j] != 0 else 0
            
            # Ensure non-negativity and clipping
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate
            new_overlap = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            delta_overlap = new_overlap - current_overlap
            if delta_overlap < 0 or np.exp(-delta_overlap / T) > random.random():
                h = h_new
                current_overlap = new_overlap
                if current_overlap < best_overlap:
                    best_h = h.copy()
                    best_overlap = current_overlap
        
        T *= alpha
    
    # Final polishing with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_overlap = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
        
    return (list(best_h), best_overlap, n_points)
