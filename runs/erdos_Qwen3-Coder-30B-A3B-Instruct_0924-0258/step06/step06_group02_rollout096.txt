import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a low-discrepancy sequence for better initialization
        def halton_sequence(n, base=2):
            sequence = []
            for i in range(n):
                x = 0
                f = 1.0 / base
                while i > 0:
                    x += (i % base) * f
                    i //= base
                    f /= base
                sequence.append(x)
            return np.array(sequence)
        
        h_start = halton_sequence(n_points)
        # Normalize to [0,1] and scale to match integral constraint
        h_start = (h_start - h_start.min()) / (h_start.max() - h_start.min())
        h_start = h_start * 0.5  # Make average around 0.5
        # Adjust to make sum = n/2
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use gradient-based optimization with gradient descent
    max_iterations = 5000
    lr = 0.01
    decay = 0.999
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute gradient with respect to total overlap (smooth approximation)
        gradient = np.zeros(n_points)
        for k in range(1, n_points):
            # Gradient contributions for shift k
            for i in range(n_points - k):
                # h[i] interacts with (1 - h[i+k])
                gradient[i] += (1 - best_h[i + k])
                # h[i+k] interacts with h[i] (but from the other side)
                if i - k >= 0:
                    gradient[i] -= best_h[i - k]
        
        # Update using gradient descent
        updated_h = best_h - lr * gradient
        updated_h = np.clip(updated_h, 0, 1)
        
        # Project to correct integral constraint
        sum_h = np.sum(updated_h)
        if sum_h > 0:
            updated_h = updated_h * (n_points / 2) / sum_h
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(updated_h, dx)
        if c5_new < best_c5:
            best_h = updated_h.copy()
            best_c5 = c5_new
            
        # Learning rate decay
        lr *= decay
    
    # Final refinement using convex optimization for small window (if feasible)
    try:
        # Reconstruct step for solving small QCP problem (if small enough)
        if n_points <= 60:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Only add a subset of constraints to keep it manageable
            num_constraints = min(n_points // 2, 20)
            for k in range(1, num_constraints):
                # Approximate most important differences (prioritize larger k for reduction)
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final feasibility check
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * (n_points / 2) / sum_h
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
