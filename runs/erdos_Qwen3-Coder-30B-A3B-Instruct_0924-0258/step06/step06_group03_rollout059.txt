import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Get initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create triangular window as initial guess
        h_start = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h_start[i] = 1.0 - abs(i - mid) / (mid + 1)
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize best solution
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    max_iter = 300
    popsize = 50
    mutation_factor = 0.7
    crossover_rate = 0.9
    
    # Custom constraint handling
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Objective function with constraint enforcement
    def objective(h):
        # Normalize to maintain constraint
        h_normalized = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h_normalized, dx)
    
    # DE optimization
    de_result = differential_evolution(
        objective, 
        bounds, 
        maxiter=max_iter, 
        popsize=popsize, 
        mutation=mutation_factor,
        recombination=crossover_rate,
        seed=seed,
        disp=False,
        tol=1e-6,
        callback=lambda x, convergence: None
    )
    
    if de_result.success:
        h_de = de_result.x
        h_de_normalized = h_de * (n_points/2) / np.sum(h_de)
        c5_de = evaluate_erdos_solution(h_de_normalized, dx)
        
        if c5_de < best_c5:
            best_h = h_de_normalized
            best_c5 = c5_de
    
    # Local refinement using L-BFGS-B
    def refine_solution(h_start):
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        # Project to maintain constraint
        def project(h):
            return h * (n_points/2) / np.sum(h)
            
        # Wrapper that enforces constraint
        def constrained_obj(h):
            h_proj = project(h)
            return obj(h_proj)
        
        # Sample a few points to see if we can improve
        max_refinements = 10
        for _ in range(max_refinements):
            if time() - start_time > budget_s * 0.95:
                break
                
            try:
                # These are a bit too heavy, simpler approach
                # just do a quick mutation on good candidates
                h_refined = h_start + np.random.normal(0, 0.005, n_points)
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    
            except Exception:

            refined_h, refined_c5 = refine_solution(best_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
