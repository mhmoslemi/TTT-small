import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing approach with adaptive cooling
    max_iter = 1000
    temp = 1.0
    cooling_rate = 0.995
    min_temp = 1e-6
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor solution
        h_new = best_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.1)
        h_new[idx] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5 or np.random.random() < np.exp((best_c5 - c5_new) / temp):
            best_h = h_new
            best_c5 = c5_new
        
        # Cool down
        temp = max(min_temp, temp * cooling_rate)
    
    # Final refinement with gradient-based optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_integral(h):
        return np.sum(h) - n_points / 2
    
    try:
        h_opt = cp.Variable(n_points)
        objective_cp = cp.Minimize(0)  # dummy objective
        constraints = [
            h_opt >= 0,
            h_opt <= 1,
            cp.sum(h_opt) == n_points / 2.0
        ]
        
        prob = cp.Problem(objective_cp, constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
