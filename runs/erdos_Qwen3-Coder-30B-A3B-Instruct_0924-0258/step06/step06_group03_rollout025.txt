import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Structured initialization: triangular window
        h_start = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h_start[i] = max(0, 1.0 - abs(i - mid) / (mid + 1))
        # Scale to meet sum constraint
        h_start = h_start * (n_points/2) / np.sum(h_start)
        
    dx = 2.0 / n_points
    
    # Global optimization with Differential Evolution
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Constraint function: sum(h) = n_points/2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution for global search
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=300,
        popsize=50,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        tol=1e-6
    )
    
    best_h = de_result.x
    best_c5 = de_result.fun
    
    # Refinement with L-BFGS-B for top solutions
    # Select top 10 solutions from population
    top_solutions = []
    if hasattr(de_result, 'population'):
        for sol in de_result.population:
            if abs(np.sum(sol) - n_points/2) < 1e-3:  # Check constraint
                top_solutions.append(sol)
    
    if not top_solutions:
        top_solutions = [best_h]
    
    # Refine these solutions
    for sol in top_solutions[:10]:
        try:
            # Project to feasible set
            sol = np.clip(sol, 0, 1)
            sol = sol * (n_points/2) / np.sum(sol)
            
            # L-BFGS-B refinement
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds_local = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, sol, method='L-BFGS-B', 
                          bounds=bounds_local, constraints=cons_sum,
                          options={'maxiter': 200, 'ftol': 1e-8})
            
            if res.success:
                refined_sol = res.x
                refined_c5 = objective(refined_sol)
                if refined_c5 < best_c5:
                    best_h = refined_sol
                    best_c5 = refined_c5
        except Exception:
            continue
    
    # Ensure final solution meets constraints exactly
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
