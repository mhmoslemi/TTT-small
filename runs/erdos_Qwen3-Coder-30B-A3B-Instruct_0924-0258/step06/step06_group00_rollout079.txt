import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization approach
    resolutions = [35, 40, 50, 60, 80]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Randomized restart based optimizations
        max_restarts = 25
        restart_count = 0
        local_best = h_current.copy()
        local_best_c5 = evaluate_erdos_solution(h_current, dx)
        
        while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
            # Create perturbed version
            h_perturbed = local_best.copy()
            
            # Add Gaussian noise with adaptive std
            std = 0.05 + np.random.rand() * 0.05
            noise = np.random.normal(0, std, n_points)
            h_perturbed += noise
            
            # Clip to feasible region
            h_perturbed = np.clip(h_perturbed, 0, 1)
            
            # Re-normalize
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            
            # Local optimization
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res_mini = minimize(objective, h_perturbed, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 300, 'ftol': 1e-9})
                
                if res_mini.success:
                    h_candidate = res_mini.x
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < local_best_c5:
                        local_best = h_candidate
                        local_best_c5 = c5_candidate
                        restart_count = 0  # Reset counter if improvement
                    else:
                        restart_count += 1
                else:
                    restart_count += 1
            except Exception:
                restart_count += 1
        
        # Update global best if we improved
        if local_best_c5 < best_c5:
            best_h = local_best
            best_c5 = local_best_c5
            
        h_start = local_best.copy()  # Keep the best from this resolution
    
    # Final refinement with higher tolerance constraints
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res_final = minimize(objective, h_final, method='SLSQP', 
                            bounds=bounds, constraints=cons_sum, 
                            options={'maxiter': 1000, 'ftol': 1e-12})
        
        if res_final.success:
            h_opt = res_final.x
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    except Exception as e:
        pass
    
    return (list(best_h), best_c5, n_points)
