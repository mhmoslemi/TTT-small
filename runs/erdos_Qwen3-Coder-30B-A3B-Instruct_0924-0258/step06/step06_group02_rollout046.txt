import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a simple pattern
        h_start = np.ones(n_points) * 0.5
        # Add some variation
        for i in range(n_points):
            h_start[i] += 0.1 * np.sin(i * 2 * np.pi / n_points)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive step size
    max_iterations = 2000
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute gradient approximation
        gradient = np.zeros(n_points)
        # For each shift k compute contribution to gradient
        for k in range(1, n_points):
            # h[i] contributes to gradient of h[i] through:
            # - (1 - h[i+k]) for the term h[i] * (1 - h[i+k]) 
            # - h[i-k] for the term h[i-k] * (1 - h[i]) when i-k >= 0
            if k < n_points:
                # h[i] * (1 - h[i+k]) for i = 0 to n-k-1
                for i in range(n_points - k):
                    gradient[i] += (1 - best_h[i + k])
                # h[i-k] * (1 - h[i]) for i = k to n
                for i in range(k, n_points):
                    gradient[i] -= best_h[i - k]
        
        # Update with gradient descent
        step_size = 0.05 * np.exp(-i * 0.002)
        h_new = best_h - step_size * gradient
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct integral
        sum_h = np.sum(h_new)
        if sum_h > 0:
            h_new = h_new * (n_points / 2) / sum_h
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
    
    # Final refinement using local search
    try:
        # Try slight variations to improve result
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            h_try = best_h.copy()
            # Slightly perturb and project
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.02)
            h_try[idx] += delta
            h_try = np.clip(h_try, 0, 1)
            sum_h = np.sum(h_try)
            if sum_h > 0:
                h_try = h_try * (n_points / 2) / sum_h
            c5_try = evaluate_erdos_solution(h_try, dx)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
