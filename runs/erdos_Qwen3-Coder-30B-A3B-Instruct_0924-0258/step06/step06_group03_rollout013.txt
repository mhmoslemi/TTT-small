import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create triangular window with small random perturbations
        h_start = 1 - np.abs(np.arange(n_points) / (n_points - 1) - 0.5) * 2
        h_start += np.random.normal(0, 0.02, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Differential Evolution with custom callback
    def de_callback(x, convergence):
        nonlocal best_h, best_c5
        h = np.array(x)
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        return False
    
    # Run DE for the majority of the time budget
    de_budget = budget_s * 0.8
    de_start = time()
    
    # Custom DE with constraints
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Try to make a reasonable population size and iterations
    population_size = min(50, n_points * 5)
    maxiter = max(100, int(de_budget / 10))
    
    try:
        # Use scipy's differential_evolution
        result = differential_evolution(
            objective,
            bounds,
            seed=seed,
            maxiter=maxiter,
            popsize=population_size,
            callback=de_callback,
            strategy='best1bin',
            atol=1e-8,
            rtol=1e-8
        )
        
        if result.success:
            h_final = result.x
            c5_final = result.fun
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    # Refinement step with local optimization
    if time() - start_time < de_budget * 0.9:
        # Use L-BFGS-B to refine the best solution
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds_local = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                objective, 
                best_h, 
                method='L-BFGS-B',
                bounds=bounds_local,
                constraints=cons_sum,
                options={'maxiter': 200, 'ftol': 1e-10}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
