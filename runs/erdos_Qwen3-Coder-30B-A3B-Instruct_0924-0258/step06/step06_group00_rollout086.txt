import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate diverse initial candidates
    def generate_candidate(h_ref, noise_level=0.1):
        h_new = h_ref + np.random.normal(0, noise_level, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        return h_new
    
    def generate_block_pattern():
        h = np.zeros(n_points)
        block_size = n_points // 4
        h[:block_size] = 1.0
        h[block_size:2*block_size] = 0.0
        h[2*block_size:3*block_size] = 1.0
        h[3*block_size:] = 0.0
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Initialize population for DE
    population_size = 100
    population = [h_start]
    for _ in range(population_size - 1):
        if np.random.random() < 0.3:
            population.append(generate_block_pattern())
        else:
            population.append(generate_candidate(h_start, 0.1))
    
    # Differential evolution
    def de_objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    de_result = differential_evolution(de_objective, bounds, 
                                       maxiter=500, popsize=population_size,
                                       mutation=(0.5, 1), recombination=0.7,
                                       seed=seed, callback=lambda x, convergence: None)
    
    if de_result.success:
        best_h_de = de_result.x
        best_h_de = np.clip(best_h_de, 0, 1)
        best_h_de = best_h_de * (n_points/2) / np.sum(best_h_de)
        c5_de = evaluate_erdos_solution(best_h_de, dx)
        if c5_de < best_c5:
            best_h = best_h_de
            best_c5 = c5_de
    
    # Local refinement using SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Test top candidates from DE
    top_candidates = [de_result.x]
    for h in population:
        if len(top_candidates) < 10:
            top_candidates.append(h)
    top_candidates = sorted(top_candidates, key=objective)[:10]
    
    for h_refined in top_candidates:
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        try:
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            continue
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
