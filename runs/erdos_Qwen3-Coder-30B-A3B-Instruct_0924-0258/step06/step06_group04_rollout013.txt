import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Grid refinement with increasing resolution
    n_start = 10
    n_max = 35
    n_points = n_start
    best_h = None
    best_c5 = float('inf')
    
    while n_points <= n_max:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize h
        if 'initial_h_values' in globals() and n_points == 10:
            h_start = np.array(initial_h_values)
        else:
            # Sinusoidal initialization
            x = np.linspace(0, 2, n_points)
            h_start = 0.5 + 0.5 * np.sin(np.pi * x)
            h_start = np.clip(h_start, 0, 1)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        dx = 2.0 / n_points
        h_cur = h_start.copy()
        c5_cur = evaluate_erdos_solution(h_start, dx)
        
        # Hybrid optimization: gradient descent + simulated annealing
        max_iter = 100
        step_size = 0.1
        temp = 0.1
        decay_rate = 0.99
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Gradient descent step
            grad = np.zeros(n_points)
            for k in range(n_points):
                if k < n_points // 2:
                    corr = np.sum(h_cur[:n_points-k] * (1 - h_cur[k:]))
                else:
                    corr = np.sum(h_cur[k-n_points+1:] * (1 - h_cur[:n_points-(k-n_points+1)]))
                grad[k] = corr
                
            # Compute gradient at current point and adjust
            h_new = h_cur - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject based on temperature (simulated annealing)
            if c5_new < c5_cur or np.random.rand() < np.exp(-(c5_new - c5_cur) / temp):
                h_cur = h_new
                c5_cur = c5_new
                
            # Simulated annealing: random perturbation
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, temp, n_points)
                h_pert = h_cur + noise
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
                c5_pert = evaluate_erdos_solution(h_pert, dx)
                if c5_pert < c5_cur:
                    h_cur = h_pert
                    c5_cur = c5_pert
                    
            temp *= decay_rate
            step_size *= decay_rate
            
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            
        # Double the resolution
        n_points *= 2
        
    # Binary search refinement
    if best_h is not None:
        target_c5 = 0.3807
        low = 0.0
        high = best_c5
        dx = 2.0 / len(best_h)
        
        for _ in range(20):
            if time() - start_time > budget_s * 0.95:
                break
            mid = (low + high) / 2
            # Approximate check using a feasible constraint
            try:
                h_opt = cp.Variable(len(best_h))
                constraints = [
                    cp.sum(h_opt) == len(best_h) / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                # Add constraints for different lags
                for k in range(1, len(best_h)):
                    if k < len(best_h) // 2:
                        corr_sum = cp.sum(cp.multiply(h_opt[:len(best_h)-k], 1 - h_opt[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h_opt[k-len(best_h)+1:], 1 - h_opt[:len(best_h)-(k-len(best_h)+1)]))
                    constraints.append(corr_sum <= mid * dx + 1e-8)
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-8, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    low = mid
                    best_c5 = mid
                else:
                    high = mid
            except:
                break
                
    # Final refinement
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (len(best_h) / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, 2.0 / len(final_h))
    
    return (list(final_h), final_c5, len(final_h))
