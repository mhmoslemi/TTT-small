from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    resolutions = [35, 40, 50, 60, 80]

        if time() - start_time > budget_s * 0.95:
            break

            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res),

        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
