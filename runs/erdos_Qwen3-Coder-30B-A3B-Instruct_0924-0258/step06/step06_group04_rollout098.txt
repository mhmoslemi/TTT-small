import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm approach
    def mutate(h, mutation_rate=0.1):
        h_new = h.copy()
        for i in range(len(h_new)):
            if np.random.random() < mutation_rate:
                h_new[i] += np.random.normal(0, 0.05)
        # Clamp to [0,1] and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        return h_new
    
    # Population-based approach with selection
    population_size = 20
    population = []
    fitness = []
    
    # Initialize population
    for _ in range(population_size):
        h_temp = mutate(h_start, 0.2)
        population.append(h_temp)
        fitness.append(evaluate_erdos_solution(h_temp, dx))
    
    # Evolution loop
    num_generations = 500
    for gen in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size//2]
        
        # Selection, crossover, mutation
        new_population = [population[i] for i in top_indices]
        
        for _ in range(population_size - len(new_population)):
            # Tournament selection
            parent1_idx = np.random.choice(top_indices)
            parent2_idx = np.random.choice(top_indices)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover (uniform)
            mask = np.random.random(n_points) > 0.5
            child = np.where(mask, parent1, parent2)
            
            # Mutation
            child = mutate(child, 0.1)
            new_population.append(child)
        
        # Evaluate new population
        population = new_population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
    
    # Refinement with local optimization
    try:
        # Use a simple local search with gradients
        h_refine = best_h.copy()
        dx_refine = 2.0 / n_points
        c5_refine = evaluate_erdos_solution(h_refine, dx_refine)
        
        # Simple gradient descent step
        for iter_step in range(200):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Estimate gradient using finite differences
            eps = 1e-4
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h_refine.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                
                c5_plus = evaluate_erdos_solution(h_plus, dx_refine)
                grad[i] = (c5_plus - c5_refine) / eps
            
            # Update step
            step_size = 0.01
            h_refine_new = h_refine - step_size * grad
            h_refine_new = np.clip(h_refine_new, 0, 1)
            h_refine_new = h_refine_new * (n_points/2) / np.sum(h_refine_new)
            
            c5_new = evaluate_erdos_solution(h_refine_new, dx_refine)

                h_refine = h_refine_new
                c5_refine = c5_new
            else:
                break
        
        if c5_refine < best_c5:
            best_h = h_refine
            best_c5 = c5_refine
    except Exception:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
