import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search approach with bit flips
    h_binary = (h_start > 0.5).astype(int)
    ones_count = int(n_points / 2)
    
    # Ensure correct number of ones
    actual_ones = np.sum(h_binary)
    if actual_ones > ones_count:
        # Turn some ones into zeros
        zero_indices = np.where(h_binary == 1)[0]
        for i in range(actual_ones - ones_count):
            idx = np.random.choice(zero_indices)
            h_binary[idx] = 0
            zero_indices = np.where(h_binary == 1)[0]
    elif actual_ones < ones_count:
        # Turn some zeros into ones
        one_indices = np.where(h_binary == 0)[0]
        for i in range(ones_count - actual_ones):
            idx = np.random.choice(one_indices)
            h_binary[idx] = 1
            one_indices = np.where(h_binary == 0)[0]
    
    # Convert to float array for evaluation
    h_float = h_binary.astype(float)
    
    # Simulated annealing with binary flips
    temp = 1.0
    cooling_rate = 0.995
    min_temp = 1e-6
    iter_per_temp = 100
    max_iter = int(budget_s * 1000)
    
    current_h = h_float.copy()
    current_c5 = evaluate_erdos_solution(current_h, dx)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Flip two random bits
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Create new configuration
        new_h = current_h.copy()
        new_h[i] = 1 - new_h[i]
        new_h[j] = 1 - new_h[j]
        
        # Keep sum constraint
        old_sum = np.sum(current_h)
        new_sum = np.sum(new_h)
        if abs(new_sum - old_sum) > 1e-6:
            # Adjust to meet constraint
            diff = old_sum - new_sum
            new_h[0] = max(0, min(1, new_h[0] + diff))
            new_sum = np.sum(new_h)
        
        # Evaluate new configuration
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Acceptance probability
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / temp):
            current_h = new_h.copy()
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    # Final refinement with continuous optimization
    try:
        # Convert best binary to continuous form
        h_refined = cp.Variable(n_points)
        constraints = [
            cp.sum(h_refined) == n_points / 2.0,
            h_refined >= 0,
            h_refined <= 1
        ]
        
        # Use a smaller subset of constraints for efficiency
        k_vals = np.arange(1, min(20, n_points))
        for k in k_vals:
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_refined[:n_points-k], 1 - h_refined[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-8)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7, max_iters=200)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_refined.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
        
    # Final local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        h_final = minimize(objective, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 100, 'ftol': 1e-8})
        if h_final.success:
            final_h = h_final.x
            final_c5 = objective(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
