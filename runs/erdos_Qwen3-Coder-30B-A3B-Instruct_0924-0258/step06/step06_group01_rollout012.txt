import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with existing construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    max_iter = 10000
    iter_count = 0
    last_improvement = 0
    min_improvement = 1e-6
    
    # Gradient-based optimization with alternating projections
    h_current = h_start.copy()
    h_prev = h_start.copy()
    
    # Momentum term
    momentum = 0.9
    learning_rate = 0.1
    
    # Compute initial overlap
    current_c5 = evaluate_erdos_solution(h_current, dx)
    
    if current_c5 < best_c5:
        best_h = h_current.copy()
        best_c5 = current_c5
    
    # Main optimization loop
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1
        
        # Get current correlation
        corr_full = np.correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr_full)
        max_corr_val = corr_full[k_max]
        
        # Create quadratic approximation around current h
        # This is a simplified approach - we approximate the gradient at current point
        # We use finite differences to approximate gradient for the max overlap
        eps = 1e-4
        grad = np.zeros(n_points)
        
        # Approximate gradient using finite differences
        for i in range(n_points):
            h_pert = h_current.copy()
            h_pert[i] += eps
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            
            corr_new = np.correlate(h_pert, 1 - h_pert, mode='full')
            max_new = corr_new[k_max]
            grad[i] = (max_new - max_corr_val) / eps
        
        # Update with momentum
        h_new = h_current - learning_rate * grad
        h_new = h_new * (1 - momentum) + momentum * h_prev
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if new_c5 < current_c5:
            h_current = h_new.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
                last_improvement = iter_count
        else:
            # If no improvement, try small random perturbations
            if np.random.random() < 0.1:
                h_pert = h_current.copy()

                h_pert[pert_idx] += delta
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points/2) / np.sum(h_pert)
                pert_c5 = evaluate_erdos_solution(h_pert, dx)
                if pert_c5 < current_c5:
                    h_current = h_pert.copy()
                    current_c5 = pert_c5
        
        h_prev = h_current.copy()
        
        # Early stopping condition
        if iter_count - last_improvement > 1000:

    # Final polishing with scipy
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        if res.success:
            final_h = res.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points/2) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h.copy()
                best_c5 = final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
