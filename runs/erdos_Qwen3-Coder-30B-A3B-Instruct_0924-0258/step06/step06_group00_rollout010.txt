import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate diverse initial seeds
    initial_seeds = [h_start.copy()]
    # Add random perturbations
    for _ in range(5):
        h_pert = h_start + np.random.normal(0, 0.03, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        initial_seeds.append(h_pert)
    
    # Structural patterns
    # Alternating high/low blocks
    h_blocks = np.zeros(n_points)
    block_size = max(1, n_points // 4)
    for i in range(0, n_points, block_size*2):
        h_blocks[i:i+block_size] = 1.0
        h_blocks[i+block_size:min(i+block_size*2, n_points)] = 0.0
    h_blocks = h_blocks * (n_points/2) / np.sum(h_blocks)
    initial_seeds.append(h_blocks)
    
    # Ensure all seeds satisfy constraint
    for i, h in enumerate(initial_seeds):
        initial_seeds[i] = h * (n_points/2) / np.sum(h)
    
    # Target: C5 <= 0.3807
    # Differential Evolution for global search
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE configuration
    popsize = 100
    maxiter = 500  # ~0.5s per iteration, budget ~600s
    mutation_factor = 0.5
    crossover_prob = 0.7
    
    # Get rough global best from initial seeds
    for h_seed in initial_seeds:
        c5_seed = objective(h_seed)
        if c5_seed < best_c5:
            best_c5 = c5_seed
            best_h = h_seed.copy()
    
    # DE optimization
    de_start = time()
    
    # We'll use a custom loop instead of scipy's diff.evol. 
    # We start with best seed and then use DE
    # Use diverse seed population to avoid local minima
    
    # Run DE on the problem
    max_iter = max(100, int(0.5 * budget_s / 0.5))  # Up to 500 iterations
    
    # Initialize population from seeds
    population = initial_seeds[:min(len(initial_seeds), popsize)]
    while len(population) < popsize:
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
    
    best_population = population.copy()
    best_fvals = [objective(h) for h in population]
    best_idx = np.argmin(best_fvals)
    global_best_h = population[best_idx].copy()
    global_best_c5 = best_fvals[best_idx]
    
    # DE iteration loop
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # DE selection
        # Random 3 distinct candidates (check if valid indices)
        indices = np.random.choice(len(population), size=3, replace=False)
        x1, x2, x3 = population[indices[0]], population[indices[1]], population[indices[2]]
        
        # Mutation
        trial = x1 + mutation_factor * (x2 - x3)
        trial = np.clip(trial, 0, 1)
        trial = trial * (n_points/2) / np.sum(trial)
        
        # Crossover
        cross_points = np.random.rand(n_points) < crossover_prob
        if not np.any(cross_points):
            cross_points[np.random.randint(0, n_points)] = True
        
        trial = np.where(cross_points, trial, population[indices[0]])
        
        # Evaluate
        trial_c5 = objective(trial)
        old_c5 = objective(population[indices[0]])
        
        # Selection - replace if better
        if trial_c5 < old_c5:
            population[indices[0]] = trial
            
        # Update global best
        if trial_c5 < global_best_c5:
            global_best_c5 = trial_c5
            global_best_h = trial.copy()
        
        # Update best
        for i, h in enumerate(population):
            c5_val = objective(h)
            if c5_val < global_best_c5:
                global_best_c5 = c5_val
                global_best_h = h.copy()
    
    # Final refinement with SLSQP on the best DE candidate
    def ref_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def ref_constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    best_h = global_best_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': ref_constraint_sum}
    
    try:
        res = minimize(ref_objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_fine = res.x
            c5_fine = ref_objective(h_fine)
            if c5_fine < global_best_c5:
                global_best_c5 = c5_fine
                global_best_h = h_fine.copy()
    except Exception:
        pass
    
    return (list(global_best_h), global_best_c5, n_points)
