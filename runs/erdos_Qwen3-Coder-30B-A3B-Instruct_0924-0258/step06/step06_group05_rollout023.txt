import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create binary initial construction
        h_start = np.zeros(n_points)
        ones_count = n_points // 2
        h_start[:ones_count] = 1
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search with bit flips
    temperature = 0.5
    min_temp = 1e-6
    decay = 0.995
    
    iterations = 0
    max_iterations = 1000
    
    while time() - start_time < budget_s * 0.95 and iterations < max_iterations:
        # Try bit swaps
        h_new = best_h.copy()
        
        # Select two random indices to swap
        i, j = np.random.choice(n_points, 2, replace=False)
        # Swap values
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept with probability based on temperature
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / temperature):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        temperature *= decay
        
        # Periodically refine with QCP
        if iterations % 100 == 0 and time() - start_time < budget_s * 0.95:
            try:
                # Reformulate QCP to bound overlaps
                h_opt = cp.Variable(n_points, boolean=True)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0
                ]
                
                # Sample lags to form constraints
                k_vals = np.linspace(0.05, 1.95, min(15, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)  # to index space
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=200)
                
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
            
        iterations += 1
        
    # Final local refinement 
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Make sure h is feasible and scaled
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2.0) > 1e-6:
        # Rescale to enforce constraint
        best_h = best_h * (n_points/2) / sum_h
        
    return (list(best_h), best_c5, n_points)
