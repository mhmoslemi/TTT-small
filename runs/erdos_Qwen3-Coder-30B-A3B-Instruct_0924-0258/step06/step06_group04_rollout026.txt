import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with sinusoidal pattern
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / 2)
        h_start = np.clip(h_start, 0, 1)
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with projection
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.concatenate([h, 1-h])
    
    # Define constraints for scipy
    cons = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': lambda h: constraint_bounds(h)[n_points:]}
    ]
    
    # Optimize with projected gradient method
    max_iter = 1000
    step_size = 0.01
    decay_rate = 0.999
    min_step = 1e-6
    
    # Apply gradient descent to improve solution
    h_current = best_h.copy()
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient approximation using finite differences
        grad = np.zeros_like(h_current)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            grad[j] = (objective(h_plus) - objective(h_minus)) / (2*eps)
        
        # Update with gradient step
        h_new = h_current - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            
        h_current = h_new
        step_size = max(min_step, step_size * decay_rate)
    
    # Final quadratic programming refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add overlap constraints
        corr = np.correlate(best_h, 1-best_h, mode='full')
        max_c = np.max(corr) * dx
        for k in range(len(corr)):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= max_c * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final refinement to ensure quality and feasibility
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
