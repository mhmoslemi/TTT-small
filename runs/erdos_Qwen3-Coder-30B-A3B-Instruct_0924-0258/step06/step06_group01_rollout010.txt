import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    max_iter = 10000
    learning_rate = 0.05
    momentum = 0.9
    conv_radius = 0.01
    decay_factor = 0.999
    
    # Initialize velocity for momentum
    velocity = np.zeros_like(h_start)
    
    # Gradient descent with momentum and projection
    for i in range(max_iter):
        if time() - start_time >= budget_s * 0.95:
            break
            
        # Compute gradients (approximate for non-differentiable max)
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        max_k = max_idx - (len(best_h) - 1)
        
        # Compute gradient approximating max operation around the peak
        grad = np.zeros_like(best_h)
        for j in range(len(best_h)):
            # Forward difference approximation
            h_plus = best_h.copy()
            h_plus[j] += conv_radius
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_current = np.correlate(best_h, 1 - best_h, mode='full')
            
            # Approximate gradient contribution for each position
            if j + max_k < len(best_h):
                grad[j] = (corr_plus[j + max_k] - corr_current[j + max_k]) / conv_radius
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        new_h = best_h + velocity
        
        # Project back to feasible region
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            
        # Adaptive learning rate
        learning_rate *= decay_factor
    
    # Final refinement using scipy with bounds and constraint
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-10, 'maxiter': 500}
        )
        if res.success:
            final_h = res.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points/2) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
