import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Set population size and parameters
    pop_size = 50
    n_points = 35
    max_gens = 1000
    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
    dx = 2.0 / n_points
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = initial_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(initial_h, dx)
    
    # Mutation parameters
    sigma = 0.05
    stagnation_count = 0
    stagnation_limit = 100
    
    for gen in range(max_gens):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all candidates
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adjust mutation rate
        if stagnation_count > stagnation_limit:
            sigma *= 0.95
            stagnation_count = 0
            sigma = max(sigma, 0.01)
        
        # Select top 20% for breeding
        elite_size = max(1, pop_size // 5)
        elite = population[:elite_size]
        
        # Generate offspring
        offspring = []
        while len(offspring) < pop_size - elite_size:
            # Select parent
            parent = elite[np.random.randint(0, len(elite))]
            # Add Gaussian noise
            noise = np.random.normal(0, sigma, n_points)
            h_new = parent + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            offspring.append(h_new)
            
        # Keep elite and add offspring
        population = elite + offspring
        
        # Periodic convex relaxation to improve solutions
        if gen % 100 == 0 and gen > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add constraints to limit overlap for various k
                for k in range(1, n_points):
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=200)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    h_feasible = np.clip(h_feasible, 0, 1)
                    h_feasible = h_feasible * (n_points / 2) / np.sum(h_feasible)
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
        
    # Final local refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
