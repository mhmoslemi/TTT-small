import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        # Create a Hamming window as initial guess
        h = 0.54 + 0.46 * np.cos(np.pi * (np.arange(n_points) / n_points - 0.5) * 2)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Projected gradient descent
    max_iter = 1000
    step_size = 0.01
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Find the lag that maximizes correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h) - 1)
        
        # Compute gradient (approximate)
        grad = np.zeros_like(h)
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c_plus = evaluate_erdos_solution(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= 1e-5
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            c_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[j] = (c_plus - c_minus) / (2e-5)
        
        # Update h
        h_new = h - step_size * grad
        
        # Project h_new to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new.copy()
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
