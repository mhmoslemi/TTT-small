import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based evolutionary algorithm
    population_size = 50
    mutation_rate = 0.05
    max_generations = 1000
    generation = 0
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    stagnation_count = 0
    last_improvement = 0
    
    while generation < max_generations and time() - start_time < budget_s * 0.95:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Tournament selection
        selected = []
        for _ in range(population_size):
            idx1, idx2 = np.random.choice(population_size, 2, replace=False)
            if fitness[idx1] < fitness[idx2]:
                selected.append(population[idx1].copy())
            else:
                selected.append(population[idx2].copy())
        
        # Create offspring with constraint-aware mutation
        offspring = []
        for h in selected:
            # Mutation with Gaussian noise
            noise = np.random.normal(0, mutation_rate, n_points)
            h_new = h + noise
            h_new = np.clip(h_new, 0, 1)
            # Repair to maintain sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            offspring.append(h_new)
        
        # Combine and keep best
        population = selected + offspring
        # Evaluate all and select top
        fitness = []
        for h in population:
            fitness.append(evaluate_erdos_solution(h, dx))
        
        # Sort by fitness
        idxs = np.argsort(fitness)
        population = [population[i] for i in idxs[:population_size]]
        fitness = [fitness[i] for i in idxs[:population_size]]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            last_improvement = generation
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adjust mutation rate
        if stagnation_count > 100:
            mutation_rate = max(0.01, mutation_rate * 0.95)
            stagnation_count = 0
            
        if generation % 100 == 0:
            # Periodic convex relaxation refinement
            try:
                h_var = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_var) == n_points / 2.0,
                    h_var >= 0,
                    h_var <= 1
                ]
                
                # Sample more k values for better constraint coverage
                k_vals = np.linspace(0.05, 1.95, min(30, n_points//2))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_var.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
                
        generation += 1
    
    # Final local refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Ensure final bounds hold
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
