import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    h_start = h_start * (n_points/2) / np.sum(h_start)
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define constraint for sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Random initialization strategies
    def random_perturbation(h_base, factor=0.1):
        h = h_base + np.random.normal(0, factor, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def alternating_pattern(n):
        h = np.zeros(n)
        for i in range(n):
            h[i] = 0.5 if i % 2 == 0 else 0.2
        h = h * (n/2) / np.sum(h)
        return h
    
    def block_pattern(n):
        h = np.zeros(n)
        mid = n // 2
        h[:mid] = 0.8
        h[mid:] = 0.2
        h = h * (n/2) / np.sum(h)
        return h
    
    # Population initialization
    population = [h_start]
    for _ in range(9):
        population.append(random_perturbation(h_start, 0.05))
    population.append(alternating_pattern(n_points))
    population.append(block_pattern(n_points))
    
    # Differential Evolution (global search)
    def de_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Set DE options
    maxiter = 500
    
    # DE with multiple restarts
    for i in range(20):
        if time() - start_time > budget_s - 10:
            break
            
        # Select random subset of population as initial population
        de_pop = [population[np.random.randint(len(population))] for _ in range(20)]
        
        try:
            res = differential_evolution(
                de_objective,
                bounds,
                maxiter=maxiter // 20,
                popsize=10,
                mutation=(0.5, 1.0),
                recombination=0.7,
                seed=seed + i,
                disp=False,
                polish=True
            )
            
            if res.success:
                h_de = res.x
                c5_de = de_objective(h_de)
                if c5_de < best_c5:
                    best_h = h_de
                    best_c5 = c5_de
        except Exception:
            continue
    
    # SLSQP Refinement for top candidates
    top_candidates = []
    for h in population:
        try:
            c5 = evaluate_erdos_solution(h, dx)
            top_candidates.append((h, c5))
        except:
            continue
    
    # Sort by C5 value
    top_candidates.sort(key=lambda x: x[1])
    top_candidates = top_candidates[:10]  # Top 10
    
    # Local refinement with SLSQP
    for h_init, _ in top_candidates:
        if time() - start_time > budget_s - 10:
            break
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(de_objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_refined = res.x
                c5_refined = de_objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            continue

    # Final refinement
    try:
        if time() - start_time < budget_s - 10:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(de_objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = de_objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
