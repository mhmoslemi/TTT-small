import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with periodic noise
    max_iter = 1000
    current_h = best_h.copy()
    current_c5 = best_c5
    iter_no_improve = 0
    max_no_improve = 100
    
    # Time-based stopping
    while time() - start_time < budget_s * 0.95:
        # Compute all correlations and find maximum
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_k = np.argmax(corr)
        k_max = max_k - (n_points - 1)
        
        # Gradient computation for the worst correlation
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += (1 - current_h[i + k_max])
            if i - k_max >= 0:
                grad[i] -= current_h[i - k_max]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            grad = grad / grad_norm
            
        # Line search for optimal step size
        alpha = 0.1
        for _ in range(10):  # Max backtracking steps
            h_new = current_h + alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < current_c5:
                break
            alpha *= 0.5
            
        # Update if improvement
        if c5_new < current_c5:
            current_h = h_new
            current_c5 = c5_new
            iter_no_improve = 0
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            iter_no_improve += 1
            
        # Add noise occasionally to escape local minima
        if iter_no_improve % 20 == 0:
            noise = np.random.normal(0, 0.02, n_points)
            current_h = current_h + noise
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points/2) / np.sum(current_h)
            
        if iter_no_improve > max_no_improve:
            break
            
    # Final convex relaxation 
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample k values to include in constraints
        k_vals = np.linspace(0.1, 1.9, min(30, n_points//2))
        for k_val in k_vals:
            k = int(k_val * n_points / 2)  # to index space
            if k < n_points and k > 0:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
        
    # Final local refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
