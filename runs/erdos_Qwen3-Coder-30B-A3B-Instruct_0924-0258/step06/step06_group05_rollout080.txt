import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Homotopy approach to escape local optima
    max_homotopy_steps = 100
    t_values = np.linspace(0, 1, max_homotopy_steps)
    
    # Initialize h with the starting solution
    h_current = best_h.copy()
    current_c5 = best_c5
    
    # Adaptive step size
    for t_idx, t in enumerate(t_values):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Define homotopy problem
        try:
            # Convex relaxation with homotopy parameter t
            h_opt = cp.Variable(n_points)
            h_uniform = np.ones(n_points) * (n_points/2) / n_points
            
            # Objective: t * C5 + (1-t) * regularization
            # For regularization: Euclidean distance to uniform
            reg_term = cp.sum_squares(h_opt - h_uniform)
            corr = cp.Variable()
            
            # Compute correlation constraint
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add the max_k constraint for a few k values to keep problem tractable
            k_vals = np.linspace(0.05, 1.95, min(15, n_points//3))
            for k_val in k_vals:
                k = int(k_val * n_points / 2)  # convert to index
                if k < n_points and k > 0:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    constraints.append(corr_sum <= corr + 1e-8)
            
            # Add objective
            constraints.append(corr <= current_c5 * dx * 1.01)  # Allow small relaxation
            prob = cp.Problem(cp.Minimize(t * corr + (1 - t) * reg_term), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=300)
            
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                
                if c5_feasible < current_c5:
                    h_current = h_feasible
                    current_c5 = c5_feasible
                    
                if c5_feasible < best_c5:
                    best_h = h_feasible.copy()
                    best_c5 = c5_feasible
        except Exception:
            pass
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
