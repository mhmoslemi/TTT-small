import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    n_points = 35
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        # Create sinusoidal initial guess
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n_points))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on C5
    low, high = 0.3807, 0.38085
    tolerance = 1e-6
    max_iter = 50
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        mid = (low + high) / 2
        
        # Try to solve feasibility LP
        try:
            h_vars = cp.Variable(n_points)
            constraints = [
                cp.sum(h_vars) == n_points / 2.0,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Sample some lags instead of all
            sample_lags = list(range(1, n_points//2, max(1, n_points//20)))
            
            # Add constraints for sampled lags
            for k in sample_lags:
                corr_sum = 0
                for i in range(n_points - k):
                    corr_sum += h_vars[i] * (1 - h_vars[i + k])
                for i in range(n_points - k, n_points):
                    corr_sum += h_vars[i] * (1 - h_vars[i + k - n_points])
                constraints.append(corr_sum <= mid * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_vars.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
                high = mid
            else:
                low = mid
        except Exception:
            low = mid
    
    # Final gradient-based refinement if time allows
    if time() - start_time < budget_s * 0.9:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_eq(h):
            return np.sum(h) - n_points / 2
        
        def constraint_ineq(h):
            return np.clip(h, 0, 1) - h  # h >= 0
        cons = [{'type': 'eq', 'fun': constraint_eq}, 
                {'type': 'ineq', 'fun': constraint_ineq}]
        
        # Gradient-based local optimization
        try:
            result = minimize(objective, best_h, bounds=[(0, 1)]*n_points,
                              constraints=cons, method='SLSQP', 
                              options={'maxiter': 1000})
            if result.success:
                h_refined = result.x
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass

    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
