import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

        h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n_points))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    low = 0.3807
    high = 0.38085
    tolerance = 1e-6
    max_iter = 50
    best_feasible_h = None
    best_feasible_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        c_mid = (low + high) / 2.0
        
        # Try to find feasible solution for C = c_mid
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Sample lags for constraint tightening
            lags_to_check = list(range(1, n_points // 2))
            np.random.shuffle(lags_to_check)
            lags_to_check = lags_to_check[:min(20, len(lags_to_check))]
            
            for k in lags_to_check:
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= c_mid * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)

                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_feasible_c5:
                    best_feasible_h = h_feasible.copy()
                    best_feasible_c5 = c5_feasible
                high = c_mid
            else:
                low = c_mid
        except Exception:
            low = c_mid
    
    # Refine best feasible solution with gradient-based optimization
    if best_feasible_h is not None:
        h_final = best_feasible_h
        best_c5 = best_feasible_c5
    else:
        h_final = best_h
        
    # Local optimization with Nelder-Mead
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    # Use scipy.optimize.minimize with Nelder-Mead on a subset of parameters
    bounds = [(0, 1)] * n_points
    # Transform to ensure sum = n_points/2
    scale_factor = (n_points/2) / np.sum(h_final)
    h_initial = np.clip(h_final * scale_factor, 0, 1)
    
    # Apply local refinement
    h_refined = h_initial.copy()
    iter_count = 0
    while time() - start_time < budget_s * 0.9:
        try:
            res = minimize(objective, h_refined, method='Nelder-Mead', options={'maxiter': 50})
            if not res.success:
                break
            h_refined = res.x
            # Normalize
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            new_c5 = evaluate_erdos_solution(h_refined, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                h_final = h_refined.copy()
            iter_count += 1
        except:
            break
            
    # Final validation
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
