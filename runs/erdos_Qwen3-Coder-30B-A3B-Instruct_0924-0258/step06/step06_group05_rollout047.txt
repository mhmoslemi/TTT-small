import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search with bit flips
    h_bin = (h_start > 0.5).astype(int)
    h_bin = h_bin * (n_points/2) / np.sum(h_bin)  # Normalize
    h_bin = (h_bin > 0.5).astype(int)
    
    # Set initial binary configuration to have exactly n_points/2 ones
    ones_count = int(np.round(n_points / 2))
    if np.sum(h_bin) != ones_count:
        diff = ones_count - np.sum(h_bin)
        if diff > 0:
            # Find zeros and convert to ones
            zeros = np.where(h_bin == 0)[0]
            for i in range(min(diff, len(zeros))):
                h_bin[zeros[i]] = 1
        else:
            # Find ones and convert to zeros
            ones = np.where(h_bin == 1)[0]
            for i in range(min(-diff, len(ones))):
                h_bin[ones[i]] = 0
    
    # Convert back to continuous values for optimization
    h_current = h_bin.astype(float)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    current_c5 = evaluate_erdos_solution(h_current, dx)
    
    if current_c5 < best_c5:
        best_h = h_current.copy()
        best_c5 = current_c5
    
    # Tabu search for binary configurations
    def generate_candidate(h_bin, tabu_list, temperature):
        candidate = h_bin.copy()
        idxs = np.where(candidate == 1)[0]
        idx0s = np.where(candidate == 0)[0]
        
        if len(idxs) == 0 or len(idx0s) == 0:
            return candidate
        
        i = np.random.choice(idxs)
        j = np.random.choice(idx0s)
        
        # Check tabu list
        if (i, j) in tabu_list or (j, i) in tabu_list:
            if np.random.random() < temperature:
                # Accept tabu move with low probability
                pass
            else:
                return candidate  # Do not make this move
        
        candidate[i] = 0
        candidate[j] = 1
        
        # Maintain exact sum constraint (n_points/2)
        if np.sum(candidate) != n_points/2:
            # Revert and try again
            return h_bin
        
        return candidate
    
    # Annealing parameters
    temp_start = 0.5
    temp_decay = 0.999
    tabu_list = []
    max_tabu = 20
    num_iters = 0
    
    def optimize_continuous(h_init, dx):
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                return h_refined, c5_refined
        except Exception:
            pass
        return h_init, evaluate_erdos_solution(h_init, dx)
    
    # Main loop
    while time() - start_time < budget_s * 0.95:
        num_iters += 1
        # Generate new binary candidate
        h_candidate = generate_candidate(h_current, tabu_list, temp_start)
        
        # Evaluate
        h_candidate_cont = h_candidate.astype(float)
        h_candidate_cont = h_candidate_cont * (n_points/2) / np.sum(h_candidate_cont)
        c5_candidate = evaluate_erdos_solution(h_candidate_cont, dx)
        
        if c5_candidate < current_c5:
            h_current = h_candidate_cont.copy()
            current_c5 = c5_candidate
            if c5_candidate < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_candidate
                
        # Add move to tabu list
        idxs = np.where(h_current > 0.5)[0]
        idx0s = np.where(h_current < 0.5)[0]
        if len(idxs) > 0 and len(idx0s) > 0:
            i = np.random.choice(idxs)
            j = np.random.choice(idx0s)
            tabu_list.append((i, j))
            if len(tabu_list) > max_tabu:
                tabu_list.pop(0)
        
        # Adaptive annealing
        temp_start *= temp_decay
        
        # Periodic convex optimization refinement
        if num_iters % 50 == 0:
            h_refined, c5_refined = optimize_continuous(h_current, dx)
            if c5_refined < current_c5:
                h_current = h_refined
                current_c5 = c5_refined
                if c5_refined < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_refined
    
    # Final continuous refinement
    final_h, c5_final = optimize_continuous(best_h, dx)
    if c5_final < best_c5:
        best_h = final_h
        best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
