import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO Parameters
    n_particles = 50
    max_iter = 500
    omega = 0.7
    phi1 = phi2 = 1.5
    v_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial solution as reference
    particles.append(h_start.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Generate other particles
    for _ in range(n_particles - 1):
        h = h_start + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO main loop
    p_best = particles.copy()
    p_best_c5 = [evaluate_erdos_solution(h, dx) for h in particles]
    g_best_idx = np.argmin(p_best_c5)
    g_best = p_best[g_best_idx].copy()
    g_best_c5 = p_best_c5[g_best_idx]
    
    iteration = 0
    stagnation = 0
    
    while iteration < max_iter and time() - start_time < budget_s * 0.95:
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (omega * v 
                 + phi1 * r1 * (p_best[i] - h) 
                 + phi2 * r2 * (g_best - h))
            
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            h_new = h + v
            
            # Enforce constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Update personal best
            if c5_new < p_best_c5[i]:
                p_best[i] = h_new.copy()
                p_best_c5[i] = c5_new
                
                # Update global best
                if c5_new < g_best_c5:
                    g_best = h_new.copy()
                    g_best_c5 = c5_new
                    stagnation = 0
                else:
                    stagnation += 1
            
            # Update particles and velocities
            particles[i] = h_new
            velocities[i] = v
        
        iteration += 1
        if stagnation > 100:
            break
    
    # Final refinement using SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, g_best, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < g_best_c5:
                g_best = h_refined
                g_best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(g_best), g_best_c5, n_points)
