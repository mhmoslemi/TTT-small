import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a sinusoidal initialization for better overlap reduction
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)
        h_start = np.clip(h_start, 0, 1)
    
    # Normalize
    h_start = h_start * (n_points/2) / np.sum(h_start)
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])

    # Use L-BFGS-B for constrained optimization
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
    ]
    
    # Limit iterations and use early stopping
    options = {'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6}
    
    for _ in range(20):  # Limit number of optimization runs
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation to escape local minimum
        perturbation = np.random.normal(0, 0.02, n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Optimize
        try:
            res = minimize(
                objective,
                h_perturbed,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options=options,
                tol=1e-8
            )
            if res.success and res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun
        except:
            pass
    
    # Final LP refinement attempt to enforce bounds properly
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add a few key constraints to tighten the feasible region
        step = max(1, n_points // 15)
        for k in range(1, n_points, step):
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception as e:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
