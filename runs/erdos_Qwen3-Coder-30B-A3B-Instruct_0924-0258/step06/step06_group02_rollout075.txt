import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    temperature = 0.5
    cooling_rate = 0.99
    max_steps = 2000
    
    # For foveated perturbations
    top_k = max(1, n_points // 10)
    
    for step in range(max_steps):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Calculate overlaps for foveated perturbations
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        # We only care about the middle range (valid shifts)
        middle = len(corr_full) // 2
        overlaps = corr_full[middle - n_points//2:middle + n_points//2]
        
        # Select top indices for perturbation
        top_indices = np.argsort(overlaps)[-top_k:]
        
        # Mutation: modify top overlap indices
        h_new = best_h.copy()
        # Perturb selected indices
        for i in top_indices:
            if np.random.rand() < 0.5:
                h_new[i] += 0.1 * np.random.choice([-1, 1])
            else:
                h_new[i] += 0.05 * np.random.choice([-1, 1])
        
        # Bound constraint
        h_new = np.clip(h_new, 0, 1)
        
        # Integral constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Annealing acceptance
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Cool down
        temperature *= cooling_rate
    
    # Gradient-based refinement (if time allows)
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        try:
            # Simplified gradient descent approach
            h_refined = best_h.copy()
            step_size = 1e-3
            for _ in range(100):
                if time() - start_time > budget_s * 0.98:
                    break
                # Evaluate gradients approximately using finite differences
                grad = np.zeros_like(h_refined)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[i] += eps
                    h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                    c5_plus = evaluate_erdos_solution(h_plus, dx)
                    
                    h_minus = h_refined.copy()
                    h_minus[i] -= eps
                    h_minus = h_minus * (n_points/2) / np.sum(h_minus)
                    c5_minus = evaluate_erdos_solution(h_minus, dx)
                    
                    grad[i] = (c5_plus - c5_minus) / (2*eps)
                
                # Update with gradient
                h_refined = h_refined - step_size * grad
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                
                # Evaluate
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
                step_size *= 0.995
                
        except Exception:
            pass
    
    # Ensure final constraints
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
