import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    # Binary search bounds for C5
    C_low = 0.0
    C_high = 0.38085
    tolerance = 1e-5
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex feasibility check using CVXPY
    def check_feasibility(C):
        # Create variables
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points/2]
        
        # For all shifts k, add constraint
        for k in range(n_points):
            # Compute the sum for this shift
            shift_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                shift_sum += h_var[i] * (1 - h_var[j])
            constraints.append(shift_sum <= C * n_points / 2)
        
        # Dummy objective (minimize 0)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=500)
            return prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INACCURATE
        except:
            return False
    
    # Binary search loop
    while time() - start_time < budget_s * 0.95 and C_high - C_low > tolerance:
        C_mid = (C_low + C_high) / 2
        feasible = check_feasibility(C_mid)
        
        if feasible:
            C_high = C_mid
            # Get the actual solution
            try:
                # Reoptimize with this value to get a feasible solution
                h_var = cp.Variable(n_points)
                constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points/2]
                for k in range(n_points):
                    shift_sum = 0
                    for i in range(n_points):
                        j = (i + k) % n_points
                        shift_sum += h_var[i] * (1 - h_var[j])
                    constraints.append(shift_sum <= C_mid * n_points / 2)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=500)
                
                if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INACCURATE:
                    h_new = h_var.value
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points/2) / np.sum(h_new)
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
            except:
                pass
        else:
            C_low = C_mid
    
    # Final local optimization
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
