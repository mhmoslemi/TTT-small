import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 200
    inertia = 0.7
    c1 = 1.4
    c2 = 1.4
    vel_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    particle_best = []
    particle_best_val = []
    
    for _ in range(n_particles):
        h = h_start.copy()
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-vel_max, vel_max, n_points))
        particle_best.append(h.copy())
        particle_best_val.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    global_best_idx = np.argmin(particle_best_val)
    global_best = particle_best[global_best_idx].copy()
    global_best_val = particle_best_val[global_best_idx]
    
    # Adaptive constraint sampling
    def get_critical_lags(h, dx, n_lags=10):
        corr = np.correlate(h, 1-h, mode='full')
        # Only consider non-trivial lags that cause overlap
        indices = np.argsort(-np.abs(corr))[:n_lags]
        return indices
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample critical lags for constraint refinement
        critical_lags = get_critical_lags(best_h, dx, n_lags=10)
        
        # PSO update
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (particle_best[i] - particles[i]) + 
                            c2 * r2 * (global_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -vel_max, vel_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            current_c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update particle best
            if current_c5 < particle_best_val[i]:
                particle_best[i] = particles[i].copy()
                particle_best_val[i] = current_c5
                
                # Update global best
                if current_c5 < global_best_val:
                    global_best = particles[i].copy()
                    global_best_val = current_c5
        
        # SQP refinement with adaptive constraints
        try:
            # Refine with maximum likelihood constraint adjustment
            n_lags = len(critical_lags)
            if n_lags > 0:
                # Create a feasible approximation
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add constraints based on critical lags
                for k in critical_lags:
                    if k < n_points // 2:
                        corr_sum = cp.sum(cp.multiply(
                            h_opt[:n_points-k], 1 - h_opt[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(
                            h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                    constraints.append(corr_sum <= global_best_val * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
                
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < global_best_val:
                        global_best = h_feasible
                        global_best_val = c5_feasible
                        
        except Exception:
            pass
        
        # Update best solution found
        if global_best_val < best_c5:
            best_h = global_best.copy()
            best_c5 = global_best_val
    
    # Final check and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
