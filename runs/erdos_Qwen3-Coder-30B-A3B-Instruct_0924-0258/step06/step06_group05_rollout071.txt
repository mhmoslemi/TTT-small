import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary approach with bit flips
    h_binary = (h_start > 0.5).astype(int)
    ones_count = int(n_points / 2)
    
    # Ensure correct sum
    current_ones = np.sum(h_binary)
    if current_ones > ones_count:
        # Flip some 1s to 0s
        zero_indices = np.where(h_binary == 0)[0]
        flip_indices = np.random.choice(zero_indices, int(current_ones - ones_count), replace=False)
        h_binary[flip_indices] = 1
    elif current_ones < ones_count:
        # Flip some 0s to 1s
        one_indices = np.where(h_binary == 1)[0]
        flip_indices = np.random.choice(one_indices, int(ones_count - current_ones), replace=False)
        h_binary[flip_indices] = 0
    
    current_c5 = evaluate_erdos_solution(h_binary.astype(float), dx)
    
    # Annealing parameters
    temp = 1.0
    decay = 0.995
    min_temp = 1e-6
    
    # Binary search with simulated annealing
    max_iter = 1000
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate new candidate by flipping two random bits
        zero_indices = np.where(h_binary == 0)[0]
        one_indices = np.where(h_binary == 1)[0]
        
        if len(zero_indices) > 0 and len(one_indices) > 0:
            i = np.random.choice(one_indices)
            j = np.random.choice(zero_indices)
            
            h_new = h_binary.copy()
            h_new[i] = 0
            h_new[j] = 1
            
            c5_new = evaluate_erdos_solution(h_new.astype(float), dx)
            
            # Accept with simulated annealing probability
            if c5_new < current_c5 or np.random.rand() < np.exp((current_c5 - c5_new) / temp):
                h_binary = h_new
                current_c5 = c5_new
                
                if current_c5 < best_c5:
                    best_h = h_binary.astype(float)
                    best_c5 = current_c5
                    
        temp *= decay
        if temp < min_temp:
            temp = min_temp
            
        # Periodic refinement with QCQP
        if iteration % 100 == 0 and iteration > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample a few lags for constraints
                sample_k = np.linspace(0.1, 1.9, min(15, n_points//3))
                for k_val in sample_k:
                    k = int(k_val * n_points / 2)
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=300)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
    
    # Final optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
