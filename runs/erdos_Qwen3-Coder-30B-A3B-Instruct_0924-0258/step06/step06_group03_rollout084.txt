import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the provided starting construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create a triangular window as starting point
        h_start = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h_start[i] = i / mid
            h_start[n_points-1-i] = i / mid
        h_start[mid] = 1.0
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Normalize to sum constraint
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define mutation and crossover strategies for DE
    def de_mutation_strategy(population, i, F=0.7):
        # DE/rand/1/bin
        idxs = [idx for idx in range(len(population)) if idx != i]
        a, b, c = population[np.random.choice(idxs, 3, replace=False)]
        return a + F * (b - c)
    
    def de_crossover(parent, mutant, CR=0.9):
        # Binomial crossover
        trial = np.copy(parent)
        crossover_point = np.random.randint(0, len(parent))
        for j in range(len(parent)):
            if np.random.rand() < CR or j == crossover_point:
                trial[j] = mutant[j]
        return trial
    
    def de_iteration(population, F=0.7, CR=0.9):
        new_population = []
        for i in range(len(population)):
            mutant = de_mutation_strategy(population, i, F)
            trial = de_crossover(population[i], mutant, CR)
            # Ensure trial is within bounds
            trial = np.clip(trial, 0, 1)
            # Normalize to satisfy sum constraint
            trial = normalize(trial)
            # Evaluate and select
            if objective(trial) < objective(population[i]):
                new_population.append(trial)
            else:
                new_population.append(population[i])
        return new_population
    
    # Start DE
    population_size = 50
    population = [h_start.copy()]
    for i in range(1, population_size):
        # Create perturbed solution
        perturbed = h_start + np.random.normal(0, 0.02, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = normalize(perturbed)
        population.append(perturbed)
    
    max_de_iterations = 300
    for generation in range(max_de_iterations):
        if time() - start_time > budget_s * 0.95:
            break
        population = de_iteration(population, F=0.7, CR=0.9)
        for ind in population:
            c5 = objective(ind)
            if c5 < best_c5:
                best_h = ind.copy()
                best_c5 = c5
    
    # Refinement with local optimization
    # Limit to top individuals for refinement
    top_individuals = sorted(population, key=objective)[:10]
    
    for individual in top_individuals:
        try:
            # Use L-BFGS-B for refinement starting from the top individual
            result = minimize(objective, individual, method='L-BFGS-B', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 100, 'ftol': 1e-9})
            if result.success:
                refined = result.x
                c5_refined = objective(refined)
                if c5_refined < best_c5:
                    best_h = refined
                    best_c5 = c5_refined
        except Exception:
            continue
    
    # Final check with the best solution so far
    return (list(best_h), best_c5, n_points)
