import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)

    # PSO parameters
    n_particles = 50
    inertia = 0.7
    c1 = c2 = 1.4
    vel_max = 0.2
    
    # Initialize particles
    particles = [h_start.copy() for _ in range(n_particles)]
    velocities = [np.random.uniform(-vel_max, vel_max, n_points) for _ in range(n_particles)]
    
    # Personal best
    pbest = [p.copy() for p in particles]
    pbest_c5 = [evaluate_erdos_solution(p, dx) for p in particles]
    
    # Global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # PSO iterations
    max_iter = 200
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            velocities[i] = np.clip(velocities[i], -vel_max, vel_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            c5 = evaluate_erdos_solution(particles[i], dx)
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
                if c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = c5

    # Refinement using Sequential Quadratic Programming
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Adaptive constraint sampling - sample lags based on the current best
        # First, compute overlap integrals for all lags
        corr_all = np.correlate(gbest, 1 - gbest, mode='full')
        # Sample significant lags (top 15% by absolute value of the correlation)
        abs_corr = np.abs(corr_all)
        k_indices = np.argsort(abs_corr)[-int(n_points*0.15):]
        # Also include some random lags to prevent overfitting to specific lags
        random_indices = np.random.choice(len(corr_all), int(n_points*0.1), replace=False)
        k_indices = np.unique(np.concatenate([k_indices, random_indices]))
        
        # Add these constraints to the optimization
        # Since corr_all has length 2*n_points-1, we need to map indices back to start/end
        for k_idx in k_indices:
            if k_idx < n_points:
                # corr sum for positive lags
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k_idx], 1 - h_opt[k_idx:]))
            else:
                # corr sum for negative lags
                k_neg = k_idx - (n_points-1)
                corr_sum = cp.sum(cp.multiply(h_opt[-k_neg:], 1 - h_opt[:n_points+k_neg]))
                
            constraints.append(corr_sum <= gbest_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < gbest_c5:
                gbest = h_feasible
                gbest_c5 = c5_feasible
    except Exception as e:
        pass
    
    # Final validation
    gbest = np.clip(gbest, 0, 1)
    gbest = gbest * (n_points/2) / np.sum(gbest)
    final_c5 = evaluate_erdos_solution(gbest, dx)
    
    return (list(gbest), final_c5, n_points)
