import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective function
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Start with the initial solution to make sure we have at least one candidate
    initial_c5 = objective(h_start)
    if initial_c5 < best_c5:
        best_h = h_start.copy()
        best_c5 = initial_c5
    
    # Differential Evolution for global search
    try:
        # First run DE to explore the space
        de_results = differential_evolution(
            objective,
            bounds,
            constraints=cons_sum,
            seed=seed,
            maxiter=500,
            popsize=30,
            mutation=(0.5, 1),
            recombination=0.7,
            atol=1e-8,
            tol=1e-8,
            callback=lambda x, convergence: None
        )
        
        if de_results.success:
            h_de = de_results.x
            h_de = np.clip(h_de, 0, 1)
            h_de = h_de * (n_points/2) / np.sum(h_de)
            c5_de = objective(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
                
        # Refine the best solution with local optimization
        h_local = best_h.copy()
        h_local = np.clip(h_local, 0, 1)
        h_local = h_local * (n_points/2) / np.sum(h_local)
        
        # Local SLSQP optimization
        res = minimize(objective, h_local, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
                
    except Exception as e:
        pass

    # Multiple random restarts of SLSQP to prevent local minima
    max_restarts = 30
    restart_count = 0
    
    while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        h_restart = np.random.rand(n_points) * 0.5
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        h_restart = np.clip(h_restart, 0, 1)
        
        try:
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0  # Reset if improvement made
            else:
                restart_count += 1
        except Exception:
            restart_count += 1

    # Final refinement with SLSQP using the best available
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined

    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
