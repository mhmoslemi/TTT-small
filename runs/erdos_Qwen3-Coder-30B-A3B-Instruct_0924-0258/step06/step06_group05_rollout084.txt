import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Homotopy method with convex relaxation
    max_homotopy_steps = 100
    t_vals = np.linspace(0, 1, max_homotopy_steps)
    
    h_current = h_start.copy()
    
    # Create uniform reference
    uniform_ref = np.ones(n_points) * (n_points / 2.0) / n_points
    
    for i, t in enumerate(t_vals):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Define the relaxed problem
        h_opt = cp.Variable(n_points)
        # Original objective term with weight t
        # We need to set up constraints for the correlation
        # Use inequalities over a subset of k values for tractability
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add proxy constraints for selected k values  
        k_vals = np.linspace(0.05, 1.95, min(15, n_points//2))
        for k_val in k_vals:
            k = int(k_val * n_points / 2)
            if 0 < k < n_points:
                # Sum of h[i] * (1 - h[i+k]) over valid indices
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                # Approximate constraint using max  of partial sums
                # Full constraint for max overlap would need more work
                # For now, we use a simplified approach: impose some ordering
                # This is not rigorous, but captures the essence
                # A better option might be to linearize around current h to 
                # approximate the correlation, but for now this is a practical compromise
                
        # Quadratic penalty for deviation from uniform
        penalty = cp.sum_squares(h_opt - uniform_ref)
        
        # Objective: weighted combination  
        objective = t * cp.Variable() + (1 - t) * penalty
        # We'll solve a relaxed version using ECOS for speed
        try:
            prob = cp.Problem(cp.Minimize(objective), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=200)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                # Perform an existence check via correlation
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
                    
            # Update for next iteration
            h_current = h_feasible.copy()
        except Exception:
            # Fallback: use previous or no improvement
            pass

    # Local refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run refinement for about 50 steps to explore nearby
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
                
    except Exception:
        pass

    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
