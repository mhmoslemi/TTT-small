import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    # Simulated Annealing parameters
    T0 = 1.0
    Tmin = 1e-5
    alpha = 0.999
    iterations_per_T = 100
    
    # Current and best solution
    current_h = h.copy()
    current_overlap = evaluate_erdos_solution(current_h, dx)
    best_h = current_h.copy()
    best_overlap = current_overlap
    
    T = T0
    while T > Tmin and time() - start_time < budget_s * 0.95:
        for _ in range(iterations_per_T):
            # Make a perturbation
            h_new = current_h.copy()
            
            # Randomly select two indices to perturb
            i, j = np.random.choice(n_points, 2, replace=False)
            
            # Perturb h[i]
            delta = np.random.normal(0, 0.05)
            h_new[i] += delta
            
            # Adjust h[j] to maintain sum constraint
            sum_diff = np.sum(h_new) - n_points / 2
            h_new[j] -= sum_diff / (n_points - 1)
            
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Normalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new solution
            new_overlap = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject based on SA criterion
            delta = new_overlap - current_overlap
            if delta < 0 or np.exp(-delta / T) > np.random.rand():
                current_h = h_new
                current_overlap = new_overlap
            
            # Update best solution if needed
            if current_overlap < best_overlap:
                best_h = current_h.copy()
                best_overlap = current_overlap
        
        T *= alpha
    
    # Final polishing with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            best_overlap = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_overlap, n_points)
