import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Generate structured initial guess
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 200
    w = 0.7
    c1 = 1.4
    c2 = 1.4
    v_max = 0.2
    
    # Initialize PSO
    particles = [h_start.copy() for _ in range(n_particles)]
    velocities = [np.random.uniform(-v_max, v_max, n_points) for _ in range(n_particles)]
    personal_best = [p.copy() for p in particles]
    personal_best_values = [evaluate_erdos_solution(p, dx) for p in particles]
    
    # Global best
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # Adaptive constraint sampling
    def get_critical_lags(h, n_samples=10):
        corr = np.correlate(h, 1 - h, mode='full')
        # Only consider lags from full convolution that don't wrap
        center = len(corr) // 2
        corr_values = np.abs(corr[center - n_points + 1:center + n_points])
        idxs = np.argsort(corr_values)[-n_samples:]
        return [i - n_points + 1 + center for i in idxs if i < len(corr)]
    
    # PSO iterations
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample critical lags
        critical_lags = get_critical_lags(global_best, 15)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2), np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) +
                            c2 * r2 * (global_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to sum = n_points / 2
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            new_value = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if new_value < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = new_value
                
                # Update global best if needed
                if new_value < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = new_value
        
        # Occasionally refine with more precise optimization
        if iter_count % 20 == 0 and iter_count > 0:
            try:
                # Refinement with quadratic programming
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add key constraints from current best
                for k in critical_lags:
                    if k < 0:
                        # Wrap around
                        corr_sum = cp.sum(cp.multiply(h_opt[-k:], 1 - h_opt[:n_points+k]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    constraints.append(corr_sum <= global_best_value * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < global_best_value:
                        global_best = h_feasible
                        global_best_value = c5_feasible
            except Exception:
                pass
    
    # Final check
    final_h = np.clip(global_best, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
