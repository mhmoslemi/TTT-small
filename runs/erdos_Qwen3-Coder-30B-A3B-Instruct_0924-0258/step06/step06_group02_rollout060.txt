import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        # Create symmetric pattern inspired by root systems
        h_start = np.zeros(n_points)
        centers = np.array([0.5, 1.0, 1.5])
        for c in centers:
            idx = int(np.round(c * n_points / 2))
            h_start[max(0, idx-5):min(n_points, idx+5)] = 0.7
        h_start = h_start * (n_points / (2 * np.sum(h_start)))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Project to correct integral
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Trust Region Optimization with Adaptive Radius
    current_tr_radius = 0.2
    lambda_merit = 0.1
    penalty_factor = 1.5
    
    for i in range(2000):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Perturb within trust region
        noise = np.random.uniform(-current_tr_radius, current_tr_radius, n_points)
        h_new = best_h + noise
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct integral
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Constraint violation penalty
        constraint_violation = max(0, np.max(h_new - 1), np.min(h_new), abs(np.sum(h_new) - n_points/2))
        merit_new = c5_new + lambda_merit * constraint_violation
        merit_best = best_c5 + lambda_merit * max(0, np.max(best_h - 1), np.min(best_h), abs(np.sum(best_h) - n_points/2))
        
        if merit_new < merit_best:
            # Accept and expand trust region
            best_h = h_new.copy()
            best_c5 = c5_new
            current_tr_radius = min(0.8, current_tr_radius * 1.1)
            if constraint_violation > 1e-5:
                lambda_merit = min(1.0, lambda_merit * penalty_factor)
        else:
            # Reject and contract trust region
            current_tr_radius = max(0.01, current_tr_radius * 0.8)
            if constraint_violation < 1e-5:
                lambda_merit = max(0.01, lambda_merit / penalty_factor)
    
    # Final refinement with quadratic programming
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Constraint for the maximum overlap to be at most current best
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr = np.max(corr)
        # Apply approximate constraints to avoid too many constraints
        for k in range(1, n_points//2):
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= max_corr * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final projection and validation
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * (n_points / (2 * sum_h))
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
