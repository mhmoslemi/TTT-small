import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    Tf = 1e-5
    alpha = 0.999
    steps_per_temp = 100
    
    T = T0
    current_overlap = best_c5
    
    # Main annealing loop
    while T > Tf and (time() - start_time) < budget_s * 0.9:
        for _ in range(steps_per_temp):
            # Generate neighbor solution
            h_new = h.copy()
            i, j = np.random.choice(n_points, 2, replace=False)
            
            # Small random perturbation
            delta = np.random.normal(0, 0.05)
            h_new[i] += delta
            
            # Adjust j to maintain sum constraint
            diff = np.sum(h_new) - n_points / 2
            if i != j:
                h_new[j] -= diff
            
            # Clip to bounds
            h_new = np.clip(h_new, 0, 1)
            
            # Re-scale to meet sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new solution
            new_overlap = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            delta = new_overlap - current_overlap
            if delta < 0 or np.exp(-delta / T) > np.random.rand():
                h = h_new
                current_overlap = new_overlap
                if new_overlap < best_c5:
                    best_h = h_new.copy()
                    best_c5 = new_overlap
        
        T *= alpha
    
    # Final refinement with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
