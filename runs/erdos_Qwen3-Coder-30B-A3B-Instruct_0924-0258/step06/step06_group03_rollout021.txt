import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a Hamming window-based initialization
        n_points = 35
        i = np.arange(n_points)
        h_start = 0.54 + 0.46 * np.cos(np.pi * (i / n_points - 0.5) * 2)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based refinement using projected gradient descent
    max_iter = 1000
    eta = 0.01
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Compute correlation for all lags
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        max_lag = max_idx - (len(best_h) - 1)
        
        # Compute gradient (numerically approximate for current max lag)
        eps = 1e-5
        grad = np.zeros_like(best_h)
        for i in range(len(best_h)):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (corr_plus[max_idx] - corr[max_idx]) / eps
            
        # Update with projected gradient descent
        new_h = best_h - eta * grad
        # Project to [0,1]
        new_h = np.clip(new_h, 0, 1)
        # Project to correct sum
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
    
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
