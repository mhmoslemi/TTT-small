import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 100
    generations = 200
    elite_size = population_size // 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    population.append(h_start.copy())
    
    for _ in range(population_size - 1):
        h_pert = h_start + np.random.normal(0, 0.1, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    # GA loop
    no_improvement_count = 0
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        indices = np.argsort(fitness)
        population = [population[i] for i in indices]
        fitness = [fitness[i] for i in indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 50:
            break
            
        # Selection (tournament)
        selected = []
        for _ in range(population_size):
            tournament = random.sample(range(population_size // 2), 5)
            winner = min(tournament, key=lambda i: fitness[i])
            selected.append(population[winner].copy())
        
        # Crossover and mutation
        new_population = []
        new_population.extend(population[:elite_size])
        
        for i in range(elite_size, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < len(selected) else selected[0]
            
            if random.random() < crossover_rate:
                # Uniform crossover
                mask = np.random.random(n_points) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if random.random() < mutation_rate:
                child1 += np.random.normal(0, 0.05, n_points)
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points/2) / np.sum(child1)
                
            if random.random() < mutation_rate:
                child2 += np.random.normal(0, 0.05, n_points)
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points/2) / np.sum(child2)
                
            new_population.append(child1)
            new_population.append(child2)
        
        population = new_population[:population_size]
        
    # Local refinement on best solutions
    top_candidates = population[:10]
    for h_candidate in top_candidates:
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_candidate, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            continue
    
    return (list(best_h), best_c5, n_points)
