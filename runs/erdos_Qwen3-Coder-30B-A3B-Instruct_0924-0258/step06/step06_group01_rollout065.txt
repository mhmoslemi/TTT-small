import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    C_low = 0.0
    C_high = 0.38085
    tolerance = 1e-5
    max_iterations = 1000
    
    # Solver configuration
    solver = 'ECOS'  # For compatibility
    
    def is_feasible(C):
        # Create optimization variables
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points/2]
        
        # Overlap constraints for each shift
        for k in range(n_points):
            # Compute overlap for shift k
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            constraints.append(overlap <= C * n_points / 2)
        
        # Dummy objective
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=solver, verbose=False, max_iters=500)
            return prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INACCURATE
        except:
            return False
    
    # Binary search
    iterations = 0
    while time() - start_time < budget_s * 0.9 and iterations < max_iterations:
        C_mid = (C_low + C_high) / 2
        feasible = is_feasible(C_mid)
        
        if feasible:
            C_high = C_mid
        else:
            C_low = C_mid
            
        # Check convergence
        if C_high - C_low < tolerance:
            break
            
        iterations += 1
    
    # Get final solution with refinement
    C_final = C_high  # The upper bound found
    
    # Refine using optimization
    h_refined = cp.Variable(n_points)
    constraints = [h_refined >= 0, h_refined <= 1, cp.sum(h_refined) == n_points/2]
    
    # Add overlap constraints
    for k in range(n_points):
        overlap = 0
        for i in range(n_points):
            j = (i + k) % n_points
            overlap += h_refined[i] * (1 - h_refined[j])
        constraints.append(overlap <= C_final * n_points / 2)
    
    objective = cp.Minimize(0)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver='ECOS', verbose=False, max_iters=500)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INACCURATE:
            h_solution = h_refined.value
        else:
            h_solution = best_h
    except:
        h_solution = best_h
    
    # Final projection
    h_solution = np.clip(h_solution, 0, 1)
    h_solution = h_solution * (n_points/2) / np.sum(h_solution)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h_solution, dx)
    
    # If not better, return best
    if final_c5 < best_c5:
        return (list(h_solution), final_c5, n_points)
    else:
        return (list(best_h), best_c5, n_points)
