import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Enhanced Optimization: Simulated Annealing with Local Search
    max_iter = 3000
    temp = 1.0
    cooling_rate = 0.995
    min_temp = 1e-6
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturbation step
        noise = np.random.normal(0, 0.05, n_points)
        h_new = best_h + noise
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on temperature
        if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / temp):
            best_h = h_new
            best_c5 = c5_new
        
        # Cool down temperature
        temp = max(min_temp, temp * cooling_rate)
        
        # Occasionally apply local refinement using scipy minimize
        if i % 50 == 0:
            try:
                # Local refinement
                h_refined = best_h.copy()
                constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1}]
                bounds = [(0, 1) for _ in range(n_points)]
                
                result = minimize(
                    lambda x: evaluate_erdos_solution(x, dx),
                    h_refined,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 50}
                )
                if result.success:
                    h_refined = result.x
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    # Final convex relaxation refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        # Directly constrain the maximum overlap value
        max_overlap = cp.Parameter()
        max_overlap.value = best_c5 * dx
        
        # Add linear constraints based on some lags for tighter relaxation
        for k in range(1, max(1, n_points // 4)):
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= max_overlap)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final normalization and evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
