import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n_points))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search between 0.3807 and 0.38085
    low = 0.3807
    high = 0.38085
    tolerance = 1e-6
    max_iter = 50
    
    # Adaptive constraint sampling
    def solve_feasibility(h_guess, max_val):
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Determine lag range to sample
            sample_lags = list(range(1, n_points//2 + 1, max(1, n_points//20)))
            for k in sample_lags:
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    constraints.append(corr_sum <= max_val * dx + 1e-8)
                else:
                    break
                    
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            if prob.status == cp.OPTIMAL:
                return h_opt.value
        except:
            pass
        return None
    
    # Refinement with local search
    def refine_solution(h_start, max_val):
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(objective, h_start, method='SLSQP', bounds=bounds, 
                         constraints=constraints, options={'maxiter': 1000})
        
        if result.success:
            h_refined = np.clip(result.x, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            return h_refined
        return h_start
    
    for _ in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        h_feasible = solve_feasibility(best_h, mid)
        
        if h_feasible is not None:
            # Refine to improve the solution
            h_final = refine_solution(h_feasible, mid)
            c5_new = evaluate_erdos_solution(h_final, dx)
            
            if c5_new < best_c5:
                best_h = h_final
                best_c5 = c5_new
                
            high = mid
        else:
            low = mid
            
        if high - low < tolerance:
            break
    
    # Final refinement step
    try:
        h_final = refine_solution(best_h, best_c5)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    # Final result
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
