import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Simulated Annealing
    temp = 0.5
    cooling_rate = 0.99
    max_steps = 2000
    
    for step in range(max_steps):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Focused perturbation: target high overlap regions
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Get indices where overlap is high
        overlap_indices = np.argsort(corr)[-max(1, n_points//10):]  # Top ~10%
        # Select a random index from these
        indices_to_modify = np.random.choice(overlap_indices, 
                                             size=max(1, n_points//20), 
                                             replace=False)
        
        h_new = best_h.copy()
        for i in indices_to_modify:
            # Perturb in a targeted way: smaller changes around peak
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
            h_new[i] = np.clip(h_new[i], 0, 1)
        
        # Project to correct integral
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Acceptance probability
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        else:
            prob = np.exp(-(c5_new - best_c5) / temp)
            if np.random.rand() < prob:
                best_h = h_new.copy()
                best_c5 = c5_new
        
        # Cool down
        temp *= cooling_rate
    
    # Phase 2: Gradient-based refinement using CVXPY
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        # Approximate the maximum overlap constraint
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        # Use a single constraint that approximates the max overlap
        if max_idx < n_points:
            # First part of correlation  
            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-max_idx], 1 - h_opt[max_idx:]))
        else:
            # Second part of correlation
            corr_sum = cp.sum(cp.multiply(h_opt[max_idx-n_points+1:], 1 - h_opt[:n_points-(max_idx-n_points+1)]))
        
        # Use a slightly conservative bound for numerical stability
        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        # Minimize a smoothed version of the objective
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    c5_final = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), c5_final, n_points)
