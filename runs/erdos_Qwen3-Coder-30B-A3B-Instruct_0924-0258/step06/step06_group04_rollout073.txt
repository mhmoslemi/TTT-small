import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Hybrid approach: Simulated Annealing + Local Optimization
    # Generate a refined seed pattern
    def simulate_annealing_step(h, temp):
        h_new = h + np.random.normal(0, temp, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        return h_new
    
    # Simulated Annealing
    temp_start = 0.1
    temp_decay = 0.995
    min_temp = 1e-6
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Adaptive perturbation
    step_size = 0.1
    
    # Track progress
    last_improvement = 0
    patience = 50
    
    for i in range(1000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try simulated annealing step
        h_candidate = simulate_annealing_step(current_h, temp_start)
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject based on Metropolis criterion
        if c5_candidate < current_c5 or np.random.rand() < np.exp((current_c5 - c5_candidate) / temp_start):
            current_h = h_candidate
            current_c5 = c5_candidate
            if c5_candidate < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_candidate
                last_improvement = i
        
        # Update temperature
        temp_start = max(min_temp, temp_start * temp_decay)
        
        # Early stopping if no improvement for a while
        if i - last_improvement > patience:
            break

    # Post-processing with CVXPY if time allows
    try:
        # More aggressive constraint tightening
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        # Constraint for the key moments
        key_lags = [1, 2, 3, 4, 5]
        for k in key_lags:
            if 2*k <= n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k:], 1 - h_opt[:n_points-k]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        # Solve the relaxed problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final refinement using gradient descent on the best found solution
    try:
        # Create a copy for gradient-based refinement
        h_refine = best_h.copy()
        dx = 2.0 / n_points
        
        # Optimizer parameters
        learning_rate = 0.01
        max_iter_refine = 100
        for _ in range(max_iter_refine):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Compute gradient using finite differences
            h_plus = h_refine.copy()
            h_minus = h_refine.copy()
            eps = 1e-4
            
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus[j] += eps
                h_minus[j] -= eps
                # Ensure boundary constraints
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                # Normalize
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                h_minus = h_minus * (n_points/2) / np.sum(h_minus)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                c5_minus = evaluate_erdos_solution(h_minus, dx)
                grad[j] = (c5_plus - c5_minus) / (2*eps)
                h_plus[j] -= eps
                h_minus[j] += eps
            
            # Update with gradient
            h_refine = h_refine - learning_rate * grad
            # Project back to feasible region
            h_refine = np.clip(h_refine, 0, 1)
            h_refine = h_refine * (n_points/2) / np.sum(h_refine)
            
            # Evaluate
            new_c5 = evaluate_erdos_solution(h_refine, dx)
            if new_c5 < best_c5:
                best_h = h_refine
                best_c5 = new_c5
    except Exception:
        pass
    
    # Final checks
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
