import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # ADMM parameters
    rho = 1e-4
    max_iterations = 1000
    tol = 1e-6
    iteration = 0
    
    # Initialize ADMM variables
    z = h.copy()
    u = np.zeros_like(h)
    h_prev = h.copy()
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Primal update - solve convex relaxation
        # Minimize (rho/2) * ||z - (h - u)||_2^2 + (1/rho) * C5(z)
        # Subject to sum(z) = n_points/2, 0 <= z[i] <= 1
        
        # Try solving with CVXPY for the convex subproblem
        try:
            # Create variables
            z_vars = cp.Variable(n_points)
            
            # Objective: (rho/2) * ||z - (h - u)||_2^2 + (1/rho) * C5(z)
            # Approximate C5(z) with max over correlations
            # Use a tractable lower bound or direct computation
            # For now, use direct computing
            
            # Build a convex approximation using current h
            h_current = h - u
            
            # Objective function (proximal term, hard to compute exactly)
            # We'll solve iteratively
        
            # Simple approach: project gradient step with trust regions
            # First compute current gradient for the overlap term
            # We build a full matrix of h(x) * (1-h(x+k)) overlap computations
            # Then compute as max over all lags which we can compute
            
            def compute_c5(z_vec):
                # Assuming z_vec is already scaled and in range [0,1]
                # Compute: max_k sum_i z[i] * (1 - z[i+k]) * dx
                corr = np.correlate(z_vec, 1 - z_vec, mode='full')
                return np.max(corr) * dx
            
            def grad_c5_approx(z_vec):
                # Gradient approximation for projection
                # This is a rough approximation assuming smooth behavior
                # We'll just compute numerically for direction of steepest descent
                eps = 1e-6
                grad = np.zeros_like(z_vec)
                for i in range(len(z_vec)):
                    z_plus = z_vec.copy()
                    z_plus[i] += eps
                    z_minus = z_vec.copy()
                    z_minus[i] -= eps
                    grad[i] = (compute_c5(z_plus) - compute_c5(z_minus)) / (2 * eps)
                return grad
            
            # Perform one step of gradient descent on convexified version
            # Approximate the cost function using current h, then use that for update
            
            def prox_objective(z_vec):
                # Quadratic penalty from ADMM + non-convex term
                a = np.sum((z_vec - (h - u))**2) * 0.5 * rho
                # Simplified: just evaluate cost at current iterate, for update direction
                return a + compute_c5(z_vec)
                
            # Use scipy minimize with bounds
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2.0}
            
            try:
                # Minimize the proximal term with Hessian-free AMG method
                # Actually just use L-BFGS-B for its nature of constraining in bounds
                res = minimize(prox_objective, z, method='L-BFGS-B', bounds=bounds, options={'ftol': 1e-10})
                if res.success:
                    z_new = res.x
                else:
                    # fallback if optimization fails
                    z_new = z
            except:
                # fallback
                z_new = z
            
            # Project z_new to feasible set
            z_new = np.clip(z_new, 0, 1)
            z_new_scale = z_new * (n_points/2) / np.sum(z_new)
            
            # Update h (dual update)
            h_new = z_new_scale - (1.0/rho) * grad_c5_approx(z_new_scale)

            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Update dual variable u
            u = u + h_new - z_new_scale            
            z = z_new_scale
            h = h_new

                cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2.0}

                    h = res.x
                    h = np.clip(h, 0, 1)
                    h = h * (n_points/2) / np.sum(h)
            except:
                pass
                
            z = h.copy()
            u = np.zeros_like(h)
            
        # Evaluate
        c5_val = evaluate_erdos_solution(h, dx)
        if c5_val < best_c5:
            best_h = h.copy()
            best_c5 = c5_val
            
        # Check convergence
        if np.linalg.norm(h - h_prev) < tol:
            break

        iteration += 1
        
    # Final refinement
    try:
        bounds = [(0,1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2.0}
        res = minimize(lambda x: evaluate_erdos_solution(x, dx), best_h, method='SLSQP', bounds=bounds, constraints=cons_sum, options={'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
