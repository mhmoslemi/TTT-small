import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create triangular window with random perturbation
        h_start = 1 - np.abs(np.arange(n_points) / (n_points-1) - 0.5) * 2
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Wrapper function for differential evolution
    def de_objective(h):
        # Ensure constraint
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    # Differential evolution with custom bounds
    try:
        result = differential_evolution(
            de_objective,
            bounds,
            maxiter=100,
            popsize=50,
            mutation=(0.5, 1.0),
            recombination=0.9,
            seed=seed,
            disp=False,
            tol=1e-8
        )
        
        if result.success and evaluate_erdos_solution(result.x, dx) < best_c5:
            best_h = result.x
            best_c5 = evaluate_erdos_solution(result.x, dx)
    except Exception:
        pass
    
    # Local refinement with L-BFGS-B
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Try 10 random initializations for local refinement
    for _ in range(10):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Generate diversified starting point
        h_init = h_start.copy() + np.random.normal(0, 0.02, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points/2) / np.sum(h_init)
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='L-BFGS-B',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-8})
            
            if res.success:
                h_refined = res.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            continue
    
    # Final check
    if best_c5 == np.inf or not np.isfinite(best_c5):
        # Fall back to initial solution
        best_c5 = evaluate_erdos_solution(h_start, dx)
        best_h = h_start
        
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
