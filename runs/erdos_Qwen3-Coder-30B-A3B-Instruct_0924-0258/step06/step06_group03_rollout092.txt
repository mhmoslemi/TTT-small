import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Handle initialization
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create triangular window
        h_start = 1 - np.abs(np.arange(n_points) / (n_points - 1) - 0.5) * 2
        # Normalize to sum = n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 0.5
    alpha = 0.99
    max_iter = 5000
    temp_decay = 0.999
    
    # Run simulated annealing
    current_h = h_start.copy()
    current_c5 = best_c5
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Generate candidate
        delta = np.random.normal(0, T, n_points)
        candidate_h = current_h + delta
        
        # Apply convex combination scaling
        candidate_h = np.clip(candidate_h, 0, 1)
        sum_h = np.sum(candidate_h)
        if sum_h > 0:
            candidate_h = candidate_h * (n_points/2) / sum_h
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5 or random.random() < np.exp(-(candidate_c5 - current_c5) / T):
            current_h = candidate_h
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Cool down temperature
        T *= alpha
        if i % 100 == 0:
            T = max(0.001, T * temp_decay)
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
