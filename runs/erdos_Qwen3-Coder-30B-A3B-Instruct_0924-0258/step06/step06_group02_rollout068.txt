import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def check_feasibility(C_candidate, n, h_start, dx):
    # Randomly sample shifts
    k_samples = np.random.choice(range(1, n), size=max(1, int(n * 0.2)), replace=False)
    
    # Define decision variables
    h_opt = cp.Variable(n)
    
    # Integral constraint
    constraints = [cp.sum(h_opt) == n / 2]
    
    # Bounds
    constraints += [0 <= h_opt, h_opt <= 1]
    
    # Overlap constraints for sampled shifts
    for k in k_samples:
        corr_sum = cp.sum(cp.multiply(h_opt[:n-k], 1 - h_opt[k:]))
        constraints.append(corr_sum <= C_candidate * dx)
    
    # Formulate and solve problem
    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=cp.ECOS, eps=1e-6)
        return prob.status == cp.OPTIMAL
    except:
        return False

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Binary search for minimal C5
    low = 0.38
    high = 0.38085
    iterations = 20
    search_time = budget_s * 0.8
    remaining_time = budget_s - (time() - start_time)
    
    for _ in range(iterations):
        if time() - start_time > search_time:
            break
        mid = (low + high) / 2
        if check_feasibility(mid, n_points, h_start, dx):
            c5_best = mid
            high = mid
        else:
            low = mid
    
    # Final refinement using exact constraints
    try:
        h_refine = cp.Variable(n_points)
        constraints = [cp.sum(h_refine) == n_points / 2.0]
        constraints += [0 <= h_refine, h_refine <= 1]
        
        # Add all constraints to refine
        for k in range(1, n_points):
            corr_sum = cp.sum(cp.multiply(h_refine[:n_points-k], 1 - h_refine[k:]))
            constraints.append(corr_sum <= c5_best * dx)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_opt = h_refine.value
            h_opt = np.clip(h_opt, 0, 1)
            # Project to correct integral
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < c5_best:
                h_best = h_opt
                c5_best = c5_opt
    except Exception:
        pass
    
    # Final refinement using scipy
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_integral(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_integral}
        
        result = minimize(objective, h_best, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-6)

            h_final = result.x
            h_final = np.clip(h_final, 0, 1)
