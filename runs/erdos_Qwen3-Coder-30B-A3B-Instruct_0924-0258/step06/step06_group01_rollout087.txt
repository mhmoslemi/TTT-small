import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
        h = h * (n_points/2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Parameters
    max_iter = 10000
    learning_rate = 0.1
    momentum = 0.9
    conv_radius = 0.01
    patience = 100
    patience_counter = 0
    prev_best = float('inf')
    
    # Gradient tracking
    v = np.zeros_like(h)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap and find max
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Compute gradient w.r.t. h
        gradient = np.zeros_like(h)
        # Efficient computation of gradient without full convolution
        for i in range(n_points):
            # Compute how change in h[i] affects max overlap
            # For simplicity, we'll implement an approximation
            # This approach assumes the effect of changing h[i] on overlap is 
            # mostly related to overlap at shift k where max occurs
            for j in range(n_points):
                if i + j == max_idx - (n_points - 1):  # Account for shift
                    gradient[i] += (1 - 2 * h[i]) * dx
        if np.linalg.norm(gradient) > 1e-10:
            gradient = gradient / np.linalg.norm(gradient)
        
        # Apply momentum
        v = momentum * v - learning_rate * gradient
        h_new = h + v
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new) 
        
        # Evaluate new solution
        c5 = evaluate_erdos_solution(h_new, dx)
        
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        h = h_new.copy()
        
        # Check convergence
        if abs(prev_best - best_c5) < 1e-6:
            patience_counter += 1
        prev_best = best_c5
        
        if patience_counter >= patience:
            break
    
    # Refinement with optimization
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            h_ref = res.x
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (n_points/2) / np.sum(h_ref)
            c5_ref = evaluate_erdos_solution(h_ref, dx)
            if c5_ref < best_c5:
                best_h = h_ref
                best_c5 = c5_ref
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
