import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Smaller initialization for faster optimization
        n_points = 30
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Heuristic: initialize with pattern that reduces overlap
    # Start with a low-high pattern
    if 'initial_h_values' not in globals():
        h_start = np.ones(n_points)
        for i in range(n_points):
            if i % 4 < 2:
                h_start[i] = 0.2
            else:
                h_start[i] = 0.8
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # Refinement with alternating optimization approach
    max_iters = 1000
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iters:
        # Current solution
        current_h = best_h.copy()
        current_c5 = best_c5
        
        # Project to feasible set
        h_projected = np.clip(current_h, 0, 1)
        h_projected = h_projected * (n_points/2) / np.sum(h_projected)
        
        # Try convex relaxation approach
        try:
            h_new = h_projected.copy()
            # Refine using gradient-based method with constraints
            cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Try with different optimizers
            for optimizer in ['SLSQP', 'L-BFGS-B']:
                try:
                    res = minimize(
                        lambda x: evaluate_erdos_solution(x, dx),
                        h_new,
                        method=optimizer,
                        bounds=bounds,
                        constraints=cons_sum,
                        options={'ftol': 1e-9, 'maxiter': 100}
                    )
                    if res.success:
                        h_ref = res.x
                        h_ref = np.clip(h_ref, 0, 1)
                        h_ref = h_ref * (n_points/2) / np.sum(h_ref)
                        new_c5 = evaluate_erdos_solution(h_ref, dx)
                        if new_c5 < best_c5:
                            best_h = h_ref.copy()
                            best_c5 = new_c5
                        break
                except:
                    continue
            
            # Fallback to simple projected gradient step
            if time() - start_time > budget_s * 0.9:
                break
                
            # Simple greedy update with projection
            gradient = np.zeros(n_points)
            h_current = h_new.copy()
            
            # Estimate gradient numerically via small perturbations
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                
                h_minus = h_current.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points/2) / np.sum(h_minus)
                
                grad_i = (evaluate_erdos_solution(h_plus, dx) - evaluate_erdos_solution(h_minus, dx)) / (2*eps)
                gradient[i] = grad_i
            
            # Perform gradient update
            lr = 0.1
            h_updated = h_current - lr * gradient
            h_updated = np.clip(h_updated, 0, 1)
            h_updated = h_updated * (n_points/2) / np.sum(h_updated)
            
            updated_c5 = evaluate_erdos_solution(h_updated, dx)
            if updated_c5 < best_c5:
                best_h = h_updated.copy()
                best_c5 = updated_c5
            
            iter_count += 1
            
        except Exception as e:
            # Fallback to straightforward refinement
            try:
                cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(
                    lambda x: evaluate_erdos_solution(x, dx),
                    current_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons_sum,
                    options={'ftol': 1e-9, 'maxiter': 100}
                )
                if res.success:
                    h_ref = res.x
                    h_ref = np.clip(h_ref, 0, 1)
                    h_ref = h_ref * (n_points/2) / np.sum(h_ref)
                    c5_ref = evaluate_erdos_solution(h_ref, dx)
                    if c5_ref < best_c5:
                        best_h = h_ref.copy()
                        best_c5 = c5_ref
            except Exception:
                pass
            
            iter_count += 1
    
    # Final verification
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (list(best_h), best_c5, n_points)
