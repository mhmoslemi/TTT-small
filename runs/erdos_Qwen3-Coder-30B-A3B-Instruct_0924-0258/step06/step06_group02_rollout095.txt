import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a structured pattern to avoid local minima
        h_start = np.ones(n_points) * 0.5
        # Perturb slightly to break symmetry
        h_start += np.random.normal(0, 0.01, n_points)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive learning rate
    learning_rate = 0.01
    decay_rate = 0.995
    max_iterations = 1000
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute gradients using finite differences for stability
        gradient = np.zeros(n_points)
        eps = 1e-6
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            h_minus = best_h.copy()
            h_minus[j] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            
            gradient[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update step with clipped gradient
        h_new = best_h - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            # Reset learning rate when improvement is made
            learning_rate = 0.01
        else:
            # Reduce learning rate when no improvement
            learning_rate *= decay_rate
    
    # Quadratic constraint refinement using CVXPY
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add overlap constraints for key shifts
        for k in range(1, n_points // 2 + 1):
            k_valid = min(k, n_points - k)
            if k_valid > 0:
                # Compute correlation for this shift
                if k < n_points:
                    corr_term = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_term = cp.sum(cp.multiply(h_opt[n_points-k:], 1 - h_opt[:k]))
                constraints.append(corr_term <= best_c5 * dx + 1e-6)
        
        # Minimize a regularizer (could attempt to minimize sum of squares of h)
        objective = cp.Minimize(cp.sum_squares(h_opt))
        prob = cp.Problem(objective, constraints)
        # Use ECOS solver which is fast and reliable for this size
        prob.solve(solver=cp.ECOS, eps=1e-6)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception as e:
        pass
    
    # Final validation and projection back to feasible set
    try:
        best_h = np.clip(best_h, 0, 1)
        sum_h = np.sum(best_h)
        if sum_h > 0:
            best_h = best_h * (n_points / (2 * sum_h))
        final_c5 = evaluate_erdos_solution(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
