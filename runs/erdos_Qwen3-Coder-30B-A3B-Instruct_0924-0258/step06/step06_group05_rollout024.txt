import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search approach
    h_binary = (h_start > 0.5).astype(int)
    ones_count = int(n_points / 2)
    if np.sum(h_binary) != ones_count:
        # Adjust to have exactly ones_count ones
        diff = ones_count - np.sum(h_binary)
        if diff > 0:
            # Need to flip zeros to ones
            zero_indices = np.where(h_binary == 0)[0]
            flip_indices = zero_indices[:diff]
            h_binary[flip_indices] = 1
        else:
            # Need to flip ones to zeros
            one_indices = np.where(h_binary == 1)[0]
            flip_indices = one_indices[:abs(diff)]
            h_binary[flip_indices] = 0
    
    # Normalize if needed for sum constraint
    h_binary = h_binary.astype(float)
    h_binary *= (n_points/2) / np.sum(h_binary)
    
    current_h = h_binary.copy()
    current_c5 = evaluate_erdos_solution(current_h, dx)
    
    if current_c5 < best_c5:
        best_h = current_h.copy()
        best_c5 = current_c5

    # Binary hill climbing with simulated annealing
    def get_neighbors(h, n=20):
        """Generate neighbors by flipping two random elements."""
        neighbors = []
        for _ in range(n):
            h_new = h.copy()
            # Select two different indices
            i, j = np.random.choice(len(h), size=2, replace=False)
            h_new[i] = 1 - h_new[i]
            h_new[j] = 1 - h_new[j]
            neighbors.append(h_new)
        return neighbors
    
    temp = 1.0
    temp_decay = 0.995
    
    max_iters = 1000
    
    for iteration in range(max_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbors
        neighbors = get_neighbors(current_h, n=20)
        
        # Evaluate all neighbors
        neighbor_c5s = [evaluate_erdos_solution(hn, dx) for hn in neighbors]
        
        # Find best neighbor
        best_neighbor_idx = np.argmin(neighbor_c5s)
        best_neighbor_c5 = neighbor_c5s[best_neighbor_idx]
        best_neighbor = neighbors[best_neighbor_idx]
        
        # Accept or reject based on temperature
        if best_neighbor_c5 < current_c5 or np.random.rand() < np.exp(-(best_neighbor_c5 - current_c5) / temp):
            current_h = best_neighbor
            current_c5 = best_neighbor_c5
            
            # Update best if necessary
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Periodic QCP refinement
        if iteration % 100 == 0 and iteration > 0:
            try:
                # Construct constraints using a sample of k values for QCP
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # For smaller k values, use every 2nd position up to 50% of n_points
                k_indices = [k for k in range(1, min(n_points//2, 50))]
                for k in np.random.choice(k_indices, size=min(10, len(k_indices)), replace=False):
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
                
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
        
        temp *= temp_decay

    # Final refinement with continuous optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h *= (n_points / 2.0) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
