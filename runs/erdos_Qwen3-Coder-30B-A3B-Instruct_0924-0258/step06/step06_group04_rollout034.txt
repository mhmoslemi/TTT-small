import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create sinusoidal initial guess
        x = np.linspace(0, 2, 35)
        h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / 35))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (35 / 2) / np.sum(h_start)
        n_points = 35
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on C5 bound
    low, high = 0.3807, 0.38085
    tolerance = 1e-6
    max_iter = 50
    
    def is_feasible(h, c, dx):
        # Check if all correlations are bounded by c*dx
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx <= c + 1e-8
    
    # Initial refinement with gradient-based optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Adaptive refinement
    for _ in range(50):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Use scipy minimize with bounds
        res = minimize(objective, best_h.copy(), method='L-BFGS-B', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100})
        
        if res.success:
            h_new = res.x
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # Final tuning via integer programming relaxation
    try:
        # Convex relaxation with bounds
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample lags for constraint checking
        lags_to_check = list(range(1, n_points))
        for k in lags_to_check[:20]:  # Sample first 20 lags
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except:
        pass
    
    # Final refinement
    bounds = [(0, 1) for _ in range(n_points)]
    res = minimize(objective, best_h.copy(), method='L-BFGS-B', 
                   bounds=bounds, constraints=cons, 
                   options={'maxiter': 200})
    
    if res.success:
        final_h = res.x
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points/2) / np.sum(final_h)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
            
    return (list(best_h), best_c5, n_points)
