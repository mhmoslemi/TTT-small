import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from cvxpy import Problem, Minimize, Variable, norm, sum as cp_sum

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # ADMM parameters
    rho = 1e-4
    max_iter = 1000
    tol = 1e-6
    iteration = 0
    
    # Initialize variables
    z = h.copy()
    u = np.zeros(n_points)
    
    # ADMM iterations
    prev_c5 = float('inf')
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iter:
        # Primal update step
        # Minimize (rho/2) * ||z - h + u||^2 + (1/rho) * C5_obj(z)
        # subject to sum(z) = n_points/2 and 0 <= z <= 1
        
        # Use CVXOPT for convex subproblem
        try:
            # Create the convex optimization problem
            z_var = Variable(n_points)
            
            # Objective: quadratic term plus C5 approximation
            # The C5 term can't be treated exactly in ADMM as it's non-convex,
            # so we work around it with iterative refinement
            
            # Instead, we solve a simpler projected problem
            z_new = z - u
            z_new = np.clip(z_new, 0, 1)
            # Project onto sum constraint
            z_new = z_new * (n_points/2) / np.sum(z_new)
            
            # For direct objective minimization, we do a local refinement:
            # From the current h, use first-order approximation to compute better z
            
            # Compute current objective
            current_c5 = evaluate_erdos_solution(h, dx)
            
            # A small heuristic local search using simple gradient-based method
            # short-lived local optimization
            cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
            bounds = [(0, 1) for _ in range(n_points)]
            
            def obj_func(x):
                return evaluate_erdos_solution(x, dx)
            
            try:
                res = minimize(
                    obj_func,
                    h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons_sum,
                    options={'ftol': 1e-8, 'maxiter': 50}
                )
                if res.success:
                    z_new = res.x.copy()
                    z_new = np.clip(z_new, 0, 1)
                    z_new = z_new * (n_points/2) / np.sum(z_new)
            except:
                pass  # fall back to projection
            
            # Update h
            h = z_new.copy()
            
            # Dual update
            u = u + (z - h)
            z = h.copy()
            
            # Update best if improved
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
            # Check convergence
            norm_diff = np.linalg.norm(h - z)
            if norm_diff < tol:
                break
            
            iteration += 1
            
            # Track improvement
            if abs(current_c5 - prev_c5) < 1e-10:
                no_improvement_count += 1
            else:
                no_improvement_count = 0
            prev_c5 = current_c5
            
            if no_improvement_count > 50:
                break
                
        except Exception as e:
            break
    
    # Final refinement after ADMM
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
