import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n_points))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on C5 with adaptive constraint sampling
    low, high = 0.3807, 0.38085
    tolerance = 1e-6
    max_iterations = 50
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        C = (low + high) / 2.0
        
        # Try to find feasible solution with this C
        try:
            # Try initial solution first
            h_feasible = best_h.copy()
            c5_feasible = best_c5
            
            # If we are close to being feasible, just do gradient descent refinement
            if c5_feasible <= C * 1.1:
                # Use gradient descent to refine
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                def constraint_bounds(h):
                    return np.concatenate([h, 1 - h])
                
                cons = [{'type': 'eq', 'fun': constraint_sum},
                        {'type': 'ineq', 'fun': lambda h: 1 - h},
                        {'type': 'ineq', 'fun': lambda h: h}]
                
                result = minimize(objective, best_h, method='SLSQP', 
                                constraints=cons, bounds=[(0, 1)]*n_points,
                                options={'maxiter': 100, 'ftol': 1e-8})
                if result.success:
                    h_feasible = result.x
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            
            # Use CVXPY for more robust constraint checking
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]

            # Sample some key lags, not all
            sample_lags = list(range(1, n_points, max(1, n_points // 10)))
            for k in sample_lags:
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= C * dx + 1e-6)
            
            try:
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            except:
                pass
            
            if c5_feasible <= C + tolerance:
                high = C
            else:
                low = C
                
            # Update best solution if we found a better one
            if c5_feasible < best_c5:
                best_h = h_feasible.copy()
                best_c5 = c5_feasible
                
        except Exception:
            low = C
            
    # Final refinement with gradient descent
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons = [{'type': 'eq', 'fun': constraint_sum},
                {'type': 'ineq', 'fun': lambda h: 1 - h},
                {'type': 'ineq', 'fun': lambda h: h}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         constraints=cons, bounds=[(0, 1)]*n_points,
                         options={'maxiter': 100, 'ftol': 1e-8})
        if result.success:
            best_h = result.x
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
