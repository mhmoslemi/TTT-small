import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with sinusoidal pattern
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing approach
    max_iter = 10000
    temperature = 1.0
    cooling_rate = 0.9995
    min_temp = 1e-8
    step_size = 0.1
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor
        noise = np.random.normal(0, step_size, n_points)
        h_new = best_h + noise
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        else:
            # Accept with probability
            delta = c5_new - best_c5
            if np.random.random() < np.exp(-delta / temperature):
                best_h = h_new
                best_c5 = c5_new
                
        temperature = max(min_temp, temperature * cooling_rate)
    
    # Final improvement using gradient-based optimization
    # Create objective function for scipy
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Use scipy for final refinement
    try:
        constraints = [
            {'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': lambda h: 1 - h},  # h <= 1
            {'type': 'ineq', 'fun': lambda h: h}      # h >= 0
        ]
        
        # Simple bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        
        if result.success:
            optimized_h = result.x
            optimized_h = np.clip(optimized_h, 0, 1)
            optimized_h = optimized_h * (n_points/2) / np.sum(optimized_h)
            c5_optimized = evaluate_erdos_solution(optimized_h, dx)
            if c5_optimized < best_c5:
                best_h = optimized_h
                best_c5 = c5_optimized
    except:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
