import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 100
    num_generations = 200
    elite_size = population_size // 10
    tournament_size = population_size // 5
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    population.append(h_start)
    for _ in range(population_size - 1):
        h_perturbed = h_start + np.random.normal(0, 0.1, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        population.append(h_perturbed)
    
    # Genetic algorithm
    no_improvement_count = 0
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= 50:
            break
            
        # Selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, 
                                                 tournament_size, 
                                                 replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_index].copy())
        
        # Elitism
        new_population = population[:elite_size]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if np.random.random() < crossover_rate:
                parent1_idx, parent2_idx = np.random.choice(population_size, 
                                                           2, 
                                                           replace=False)
                parent1, parent2 = selected[parent1_idx], selected[parent2_idx]
                
                # Binomial crossover
                mask = np.random.random(n_points) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1 = selected[np.random.randint(population_size)].copy()
                child2 = selected[np.random.randint(population_size)].copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child1 = child1 + noise
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points/2) / np.sum(child1)
            
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child2 = child2 + noise
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points/2) / np.sum(child2)
                
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
    
    # Local refinement on top candidates
    top_candidates = population[:elite_size]
    for h_refined in top_candidates:
        if time() - start_time > budget_s * 0.95:
            break
        
        # SLSQP local optimization
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
