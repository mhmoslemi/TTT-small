import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 500
    inertia_w = 0.7
    phi1 = phi2 = 1.5
    v_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # Seed with initial solution
    particles.append(h_start.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    pbest.append(h_start.copy())
    pbest_values.append(evaluate_erdos_solution(h_start, dx))
    
    # Initialize rest of particles
    for _ in range(n_particles - 1):
        h_rand = h_start + np.random.normal(0, 0.1, n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        particles.append(h_rand)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h_rand.copy())
        pbest_values.append(evaluate_erdos_solution(h_rand, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO iterations
    stagnation_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = phi1 * r1 * (pbest[i] - particles[i])
            social = phi2 * r2 * (gbest - particles[i])
            velocities[i] = inertia_w * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to correct integral
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            current_value = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                    
        if stagnation_count > 100:
            break
            
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, gbest, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
        else:
            best_h = gbest
            best_c5 = gbest_value
    except Exception:
        best_h = gbest
        best_c5 = gbest_value
    
    return (list(best_h), best_c5, n_points)
