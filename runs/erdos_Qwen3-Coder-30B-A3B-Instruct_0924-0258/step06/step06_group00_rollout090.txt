import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 200
    elite_size = population_size // 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        for _ in range(population_size - 1):
            h = h_start + np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points/2) / np.sum(h)
            population.append(h)

    generation = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time() - start_time < budget_s * 0.95 and generation < generations:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort population by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            break
            
        # Selection: keep top 50%
        selected_indices = sorted_indices[:population_size // 2]
        selected_population = [population[i] for i in selected_indices]
        
        # Create new population
        new_population = []
        
        # Elitism: keep top 10%
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = selected_population[random.randint(0, len(selected_population)-1)]
            parent2 = selected_population[random.randint(0, len(selected_population)-1)]
            
            if random.random() < crossover_rate:
                # Binomial crossover
                child1 = np.zeros(n_points)
                child2 = np.zeros(n_points)
                for i in range(n_points):
                    if random.random() < 0.5:
                        child1[i] = parent1[i]
                        child2[i] = parent2[i]
                    else:
                        child1[i] = parent2[i]
                        child2[i] = parent1[i]
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                if random.random() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.05)
            
            # Ensure feasibility
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
        generation += 1
    
    # Local refinement on top solutions with SLSQP
    top_candidates = population[:10]
    
    for h_init in top_candidates:
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
