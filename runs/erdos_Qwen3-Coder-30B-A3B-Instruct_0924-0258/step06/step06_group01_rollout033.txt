import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    rho = 1e-4
    max_iter = 1000
    tol = 1e-6
    iter_count = 0
    
    # Initialize variables
    h = h_start.copy()
    z = h.copy()
    u = np.zeros_like(h)
    
    def prox_step(h, z, u, rho):
        # minimize (rho/2)||z - h + u||_2^2 + (1/rho)*objective
        # Dual variable update
        u = u + h - z
        # Proximal update: solve constrained optimization problem
        # We solve: minimize (rho/2)||z - (h - u)||_2^2 + (1/rho)*f(z)
        # subject to sum(z) = n_points/2, 0 <= z[i] <= 1
        
        # This is exposed as a convex QP. We use a quadratic solver
        # For simplicity, just project the gradient and use projection onto box + sum constraints
        return z - (1.0 / rho) * (h - u)
    
    # ADMM Iteration
    last_improvement = 0
    while time() - start_time < budget_s * 0.9 and iter_count < max_iter:
        # Update h using a local refinement step
        try:
            # SLSQP optimization with sum constraint
            cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                lambda x: evaluate_erdos_solution(x, dx),
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            if res.success:
                h = res.x
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
        except Exception:
            pass

        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iter_count
            
        # ADMM step: update z and h
        old_h = h.copy()
        
        # Proximal update based on previous solution
        z = prox_step(h, z, u, rho)
        # Project z to box constraints [0,1]
        z = np.clip(z, 0, 1)
        # Project to sum constraint
        z = z * (n_points/2) / np.sum(z)
        
        # Update dual variable
        u = u + h - z
        
        # Update h
        h = z - (1.0 / rho) * (h - z)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        iter_count += 1
        # Early stopping
        if iter_count - last_improvement > 50:
            break
    
    # Final refinement
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
