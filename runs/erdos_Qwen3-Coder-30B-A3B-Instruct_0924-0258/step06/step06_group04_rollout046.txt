import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimize using scipy minimize with constraints
    def objective(h_vec):
        # Normalize
        h_norm = h_vec * (n_points/2) / np.sum(h_vec)
        h_norm = np.clip(h_norm, 0, 1)
        return evaluate_erdos_solution(h_norm, dx)
    
    def constraint_func(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use L-BFGS-B algorithm with bounds and constraints
    try:
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 1000})
        if result.success:
            h_opt = result.x
            h_opt = h_opt * (n_points/2) / np.sum(h_opt)
            h_opt = np.clip(h_opt, 0, 1)
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    except:
        pass
    
    # Adaptive refinement with simulated annealing
    max_iter = 1000
    step_size = 0.1
    temp = 1.0
    min_temp = 0.001
    decay_rate = 0.995
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb with temperature-dependent noise
        noise = np.random.normal(0, step_size * temp, n_points)
        h_new = best_h + noise
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        elif np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
            best_h = h_new
            best_c5 = c5_new
            
        # Update temperature
        temp = max(min_temp, temp * decay_rate)
    
    return (list(best_h), best_c5, n_points)
