import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create triangular window as starting point
        h_start = 1 - np.abs(np.arange(n_points) / (n_points - 1) - 0.5) * 2
        # Add some noise to break symmetry and improve diversity
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        # Scale to correct integral
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use differential evolution for global search
    def objective(h):
        # Normalize to satisfy integral constraint
        h_normalized = h * (n_points/2) / np.sum(h)
        h_normalized = np.clip(h_normalized, 0, 1)
        return evaluate_erdos_solution(h_normalized, dx)
    
    # Define bounds for each variable (0 to 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Apply differential evolution with constraints
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=300,
            popsize=50,
            mutation=(0.5, 1.0),
            recombination=0.9,
            seed=seed,
            callback=lambda x, convergence: None,
            disp=False
        )
        
        # Check if it improved upon initial
        if de_result.success:
            h_de = de_result.x
            # Re-normalize
            h_de = h_de * (n_points/2) / np.sum(h_de)
            h_de = np.clip(h_de, 0, 1)
            
            c5_de = evaluate_erdos_solution(h_de, dx)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
                
    except Exception as e:
        pass
    
    # Local refinement for top candidates
    top_candidates = []
    num_candidates = 10
    
    # For diversity, create additional candidates by perturbing the best so far
    for _ in range(num_candidates):
        h_pert = best_h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        top_candidates.append(h_pert)
    
    # Add best solution itself
    top_candidates.append(best_h)
    
    # Local optimization for each candidate
    for h_candidate in top_candidates:
        try:
            # Define constraints
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_candidate, method='L-BFGS-B',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-8})
            
            if res.success:
                h_refined = res.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    
        except Exception:
            pass
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
