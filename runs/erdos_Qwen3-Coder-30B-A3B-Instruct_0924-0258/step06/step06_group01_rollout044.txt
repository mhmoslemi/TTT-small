import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on C5 bound
    c_low = 0.0
    c_high = 0.38085
    tolerance = 1e-5
    
    # For feasibility check using CVXOPT
    while (c_high - c_low > tolerance) and (time() - start_time < budget_s * 0.95):
        c_mid = (c_low + c_high) / 2.0
        
        # Build convex feasibility problem
        h_vars = cp.Variable(n_points)
        
        # Sum constraint
        constraints = [cp.sum(h_vars) == n_points / 2]
        
        # Bounds
        constraints.extend([h_vars >= 0, h_vars <= 1])
        
        # Overlap constraints
        # For each k
        for k in range(n_points):
            # Sum of h[i] * (1 - h[(i+k)%n_points])
            expr = 0
            for i in range(n_points):
                expr += h_vars[i] * (1 - h_vars[(i+k) % n_points])
            constraints.append(expr <= c_mid * n_points / 2)
        
        # Dummy objective (minimize 0)
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iterations=500)
            if prob.status == cp.OPTIMAL:
                # Solution is feasible, try to improve
                h_sol = h_vars.value
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                c5_test = evaluate_erdos_solution(h_sol, dx)
                if c5_test < c5_best:
                    h_best = h_sol.copy()
                    c5_best = c5_test
                c_high = c_mid
            else:
                c_low = c_mid
        except Exception:
            c_low = c_mid
            
    # Final refinement using local optimization if needed
    if time() - start_time < budget_s * 0.1:
        try:
            cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                lambda x: evaluate_erdos_solution(x, dx),
                h_best,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'ftol': 1e-9, 'maxiter': 200}
            )
            if res.success:
                h_ref = res.x
                h_ref = np.clip(h_ref, 0, 1)
                h_ref = h_ref * (n_points/2) / np.sum(h_ref)
                c5_ref = evaluate_erdos_solution(h_ref, dx)
                if c5_ref < c5_best:
                    h_best = h_ref
                    c5_best = c5_ref
        except Exception:
            pass
    
    return (list(h_best), c5_best, n_points)
