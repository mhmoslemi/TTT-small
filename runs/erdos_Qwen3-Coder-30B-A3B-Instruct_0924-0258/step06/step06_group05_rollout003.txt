import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary algorithm with adaptive mutation
    population_size = 50
    generations = 1000
    mutation_rate = 0.05
    mutation_decay = 0.95
    min_mutation_rate = 0.01
    no_improve_count = 0
    max_no_improve = 100
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Main evolutionary loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        population_scores = []
        for h in population:
            score = evaluate_erdos_solution(h, dx)
            population_scores.append(score)
        
        # Sort by score
        sorted_indices = np.argsort(population_scores)
        best_in_gen = population_scores[sorted_indices[0]]
        if best_in_gen < best_c5:
            best_c5 = best_in_gen
            best_h = population[sorted_indices[0]].copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Update mutation rate
        if no_improve_count > max_no_improve:
            mutation_rate = max(min_mutation_rate, mutation_rate * mutation_decay)
            no_improve_count = 0
            
        # Select top 20%
        top_k = max(1, population_size // 5)
        selected = [population[sorted_indices[i]] for i in range(top_k)]
        
        # Generate offspring
        offspring = []
        for _ in range(population_size - len(selected)):
            parent = selected[np.random.randint(0, len(selected))]
            noise = np.random.normal(0, mutation_rate, n_points)
            child = parent + noise
            child = np.clip(child, 0, 1)
            # Repair to maintain sum constraint
            child = child * (n_points / 2) / np.sum(child)
            offspring.append(child)
            
        # Form new population
        population = selected + offspring
        
        # Periodic convex relaxation (every 100 generations)
        if gen % 100 == 0 and gen > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add correlation constraints
                k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if 0 < k < n_points:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
    
    # Final local optimization refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
