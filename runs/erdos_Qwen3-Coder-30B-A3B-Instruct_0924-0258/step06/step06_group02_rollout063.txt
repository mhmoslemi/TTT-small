import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        # Create a symmetric structured initialization
        h_start = np.zeros(n_points)
        centers = [0.5, 1.0, 1.5]
        spread = 0.3
        for c in centers:
            idx = int(c / 2.0 * n_points)
            start_idx = max(0, idx - 5)
            end_idx = min(n_points, idx + 5)
            h_start[start_idx:end_idx] = 0.8
            
        # Normalize to ensure integral constraint
        h_start = h_start * (n_points / (2 * np.sum(h_start)))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # SDP relaxation for lower bound
    try:
        h_sdp = cp.Variable(n_points)
        constraints = [
            cp.sum(h_sdp) == n_points / 2.0,
            h_sdp >= 0,
            h_sdp <= 1
        ]
        
        # Add a few key constraints based on sample overlap
        for k in [1, 2, 5, 10, 15]:
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_sdp[:n_points-k], 1 - h_sdp[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_start = np.clip(h_sdp.value, 0, 1)
            h_start = h_start * (n_points / (2 * np.sum(h_start)))
    except Exception:
        pass
    
    # Trust region optimization
    current_tr_radius = 0.2
    iterations = 0
    max_iter = 2000
    
    while iterations < max_iter and (time() - start_time < budget_s * 0.90):
        if time() - start_time > budget_s * 0.90:
            break
        
        # Stochastic perturbation within trust region
        noise = np.random.uniform(-current_tr_radius, current_tr_radius, n_points)
        h_new = best_h + noise
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / (2 * np.sum(h_new)))
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update trust region
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            current_tr_radius = min(current_tr_radius * 1.1, 0.5)
        else:
            current_tr_radius *= 0.9
        
        iterations += 1
    
    # Final refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / (2 * np.sum(h_final)))
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
