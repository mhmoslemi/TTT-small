import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.random.rand(n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    T_min = 1e-5
    alpha = 0.999
    steps_per_temp = 100
    
    T = T0
    h_current = h_start.copy()
    current_c5 = best_c5
    
    while T > T_min and (time() - start_time) < budget_s * 0.95:
        for _ in range(steps_per_temp):
            # Create candidate solution by perturbing two random indices
            h_candidate = h_current.copy()
            
            i, j = random.sample(range(n_points), 2)
            delta = random.uniform(-0.1, 0.1)
            
            # Adjust two indices to maintain sum constraint
            h_candidate[i] += delta
            h_candidate[j] -= delta
            
            # Ensure feasibility
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Re-scale to maintain sum constraint
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Evaluate candidate
            candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept or reject based on simulated annealing criterion
            delta_c5 = candidate_c5 - current_c5
            if delta_c5 < 0 or np.exp(-delta_c5 / T) > random.random():
                h_current = h_candidate.copy()
                current_c5 = candidate_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
                
        # Cool down
        T *= alpha
    
    # Final polishing with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
