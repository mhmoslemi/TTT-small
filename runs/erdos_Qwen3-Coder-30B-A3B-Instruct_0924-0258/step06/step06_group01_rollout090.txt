import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with provided construction or default
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search boundaries
    c_low = 0.0
    c_high = 0.38085  # Safe upper bound
    tolerance = 1e-5
    
    # Iterative binary search with convex relaxation
    while time() - start_time < budget_s * 0.95 and (c_high - c_low) > tolerance:
        c_mid = (c_low + c_high) / 2.0
        
        # Convex feasibility problem
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = []
        constraints.append(cp.sum(h_vars) == n_points / 2)
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        
        # Overlap constraints: h[i](1-h[i+k]) <= c_mid * dx * n_points / 2 for all k
        # This is approximated as bilinear constraints via McCormick relaxation
        # We simplify to using a direct quadratic formulation
        
        # Approximate by looking at full correlation terms
        # Precompute all indices for convolution
        for k in range(n_points):
            # This is a quadratic constraint in practice
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_vars[i] * (1 - h_vars[j])
            constraints.append(expr <= c_mid * n_points / 2)
        
        # Minimize dummy objective (0) to just check feasibility
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, max_iters=500, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_vars.value
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    current_c5 = evaluate_erdos_solution(h_sol, dx)
                    if current_c5 < best_c5:
                        best_h = h_sol.copy()
                        best_c5 = current_c5
                    c_high = c_mid  # Feasible, tighten upper bound
                else:
                    c_low = c_mid  # Infeasible due to solver issues
            else:
                c_low = c_mid  # Infeasible problem
        except Exception:
            c_low = c_mid  # Could not solve reliably
    
    # Final optimization step with NLP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            h_ref = res.x
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (n_points/2) / np.sum(h_ref)
            final_c5 = evaluate_erdos_solution(h_ref, dx)
            if final_c5 < best_c5:
                best_h = h_ref
                best_c5 = final_c5
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
