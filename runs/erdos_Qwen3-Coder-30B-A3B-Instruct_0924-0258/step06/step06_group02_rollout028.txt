import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with focused perturbations
    T = 0.5
    T_min = 1e-4
    cooling_rate = 0.99
    max_steps = 2000
    
    # First phase: simulated annealing
    for i in range(max_steps):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Calculate current overlaps to identify problematic regions
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Focus on middle correlations where overlap is highest
        mid_index = len(corr) // 2
        overlap_indices = np.argsort(corr)[::-1][:max(1, n_points // 10)]
        
        h_new = best_h.copy()
        # Apply small perturbations to top overlap regions
        for idx in overlap_indices:
            if idx < n_points:
                perturb = np.random.choice([-0.1, 0.1])
                h_new[idx] += perturb
                h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Maintain sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        delta = c5_new - best_c5
        
        if delta < 0 or np.exp(-delta / T) > np.random.random():
            best_h = h_new.copy()
            best_c5 = c5_new
            
        T = max(T_min, T * cooling_rate)
    
    # Refinement with L-BFGS
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Use the best solution from annealing
    h_refine = best_h.copy()
    h_refine = np.clip(h_refine, 0, 1)
    h_refine = h_refine * (n_points/2) / np.sum(h_refine)
    
    # Projected gradient descent on feasible region
    for _ in range(100):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Simple gradient approximation using finite differences
        eps = 1e-5
        grad = np.zeros_like(h_refine)
        base_obj = objective(h_refine)
        
        for i in range(n_points):
            h_plus = h_refine.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            obj_plus = objective(h_plus)
            grad[i] = (obj_plus - base_obj) / eps
        
        # Update with projected gradient step
        step_size = 1e-3 * (0.99 ** _)
        h_new = h_refine - step_size * grad
        
        # Apply box constraints and normalization
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        new_obj = objective(h_new)
        if new_obj < best_c5:
            best_h = h_new.copy()
            best_c5 = new_obj
    
    return (list(best_h), best_c5, n_points)
