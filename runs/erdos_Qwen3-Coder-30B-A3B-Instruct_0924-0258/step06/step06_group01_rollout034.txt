import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.random.rand(n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    rho = 1e-4
    max_iter = 1000
    tol = 1e-6
    stagnation_threshold = 50
    
    # Initialize variables
    h = h_start.copy()
    z = h.copy()
    u = np.zeros(n_points)
    
    # ADMM iterations
    stagnation_count = 0
    previous_best = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Solve the primal update problem using CVXPY
        # min (rho/2)||z - h + u||_2^2 + (1/rho)*(max_k h*(1-h_k)) s.t. sum(z) = n_points/2, 0 <= z <= 1
        z_new = cp.Variable(n_points)
        objective = cp.Minimize(
            (rho/2) * cp.sum_squares(z_new - h + u) + 
            (1/rho) * cp.max(cp.convolve(z_new, 1 - z_new, mode='full'))
        )
        constraints = [
            cp.sum(z_new) == n_points/2,
            z_new >= 0,
            z_new <= 1
        ]
        
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS, verbose=False)
            
            if prob.status == 'optimal':
                z = z_new.value
            else:
                z = np.clip(z, 0, 1)
        except:
            z = np.clip(z, 0, 1)
        
        # Update dual variable
        u = u + h - z
        
        # Proximal update for h
        h = z - (1/rho) * (z - h + u)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Evaluate solution
        current_c5 = evaluate_erdos_solution(h, dx)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        if stagnation_count > stagnation_threshold:
            break
    
    # Final refinement with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
