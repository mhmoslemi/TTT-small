import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n):
    """Create a triangular window centered at n/2"""
    x = np.arange(n)
    center = (n - 1) / 2
    # Create triangular shape
    triangle = 1 - np.abs(x - center) / (n / 2)
    return np.clip(triangle, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = triangular_window(n_points)

    # Ensure the sum constraint
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE parameters
    max_iter = 300
    popsize = 50
    mutation = 0.7
    crossover = 0.9
    
    # Store best solution found
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Wrapper for DE to minimize C5
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Mutation and crossover functions
    def mutate(population, idx, F):
        a, b, c = population[np.random.choice(len(population), 3, replace=False)]
        mutant = a + F * (b - c)
        return np.clip(mutant, 0, 1)
    
    def crossover(mutant, target, CR):
        crossover_mask = np.random.random(len(target)) < CR
        if not np.any(crossover_mask):
            crossover_mask[np.random.randint(0, len(target))] = True
        trial = np.where(crossover_mask, mutant, target)
        return trial
    
    # Differential Evolution main loop
    population = [np.copy(h_start)]
    for i in range(1, popsize):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    for gen in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        new_population = []
        
        for i in range(popsize):
            # Create mutant vector
            mutant = mutate(population, i, mutation)
            # Apply crossover
            trial = crossover(mutant, population[i], crossover)
            # Enforce sum constraint
            trial = trial * (n_points / 2) / np.sum(trial)
            
            # Evaluate
            fitness = objective(trial)
            old_fitness = objective(population[i])
            
            # Selection
            if fitness < old_fitness:
                new_population.append(trial)
            else:
                new_population.append(population[i])
        
        population = new_population
        
        # Update best
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    
    # Local refinement with L-BFGS-B
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Refine best solution found
    refined_results = []
    for _ in range(10):  # Test top 10 solutions
        if time() - start_time > budget_s * 0.9:
            break
        idx = np.random.randint(len(population))
        h_candidate = population[idx].copy()
        try:
            res = minimize(objective, h_candidate, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                refined_results.append((h_refined, c5_refined))
        except:
            continue
    
    if refined_results:
        best_refined_h, best_refined_c5 = min(refined_results, key=lambda x: x[1])
        if best_refined_c5 < best_c5:
            best_h = best_refined_h
            best_c5 = best_refined_c5
    
    # Ensure final solution meets all constraints
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
