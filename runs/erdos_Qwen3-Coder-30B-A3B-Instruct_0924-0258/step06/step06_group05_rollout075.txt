import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population
    population_size = 50
    population = [h_start.copy()]
    population_c5 = [best_c5]
    
    # Generate initial population
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        c5 = evaluate_erdos_solution(h, dx)
        population.append(h)
        population_c5.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Evolutionary algorithm
    mutation_sigma = 0.05
    no_improvement_count = 0
    max_no_improvement = 100
    generation = 0
    max_generations = 1000
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Sort population by C5
        sorted_indices = np.argsort(population_c5)
        population = [population[i] for i in sorted_indices]
        population_c5 = [population_c5[i] for i in sorted_indices]
        
        # Keep top 20%
        elite_size = max(1, population_size // 5)
        elite = population[:elite_size]
        elite_c5 = population_c5[:elite_size]
        
        if population_c5[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = population_c5[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            mutation_sigma *= 0.95
            no_improvement_count = 0
            if mutation_sigma < 0.01:
                mutation_sigma = 0.01

        # Create offspring
        offspring = []
        for _ in range(population_size - elite_size):
            # Select parent
            parent_idx = np.random.randint(0, elite_size)
            parent = elite[parent_idx]
            
            # Mutate
            noise = np.random.normal(0, mutation_sigma, n_points)
            h_new = parent + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            offspring.append((h_new, c5_new))

        # Replace population
        population = elite
        population_c5 = elite_c5
        for h, c5 in offspring:
            population.append(h)
            population_c5.append(c5)
        
        # Periodic convex relaxation
        if generation % 100 == 0 and generation > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add constraints for some k values
                k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=300)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
                
        generation += 1

    # Final refinement with local optimization if needed
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
