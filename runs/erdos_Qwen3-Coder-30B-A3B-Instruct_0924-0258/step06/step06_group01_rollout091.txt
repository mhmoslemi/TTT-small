import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Read initial values or create default

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    C_low = 0.0
    C_high = 0.38085
    tolerance = 1e-5
    max_iter = 1000
    
    # Initialize the best solution to the initial one
    current_h = h_start.copy()
    current_c5 = best_c5
    
    iter_count = 0
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        C_mid = (C_low + C_high) / 2.0
        
        # Create CVXPY problem for feasibility check
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add the overlap constraints
        # For each k in range 0 to n_points-1:
        # sum_i h[i] * (1 - h[(i+k)%n_points]) <= C_mid * dx * n_points
        # To solve this, we'll enumerate shifts and that's O(n_points^2) in the feasibility check for a fixed C
        
        # Instead, test for each shift k if the constraint is violated:
        # We do not create the explicit constraints but instead evaluate with a grid search on the row
        # The convex problem will enumerate all shifts... but let's do it manually for faster convergence
        
        # In feasibility test:
        # We evaluate the integral manually: h[i] * (1-h[j]) over all (i,j) such that j=(i+k)%n_points
        # We create constraints for the critical overlaps
        # It's better to just compute it explicitly for feasibility checking
        
        # Simpler approach: vectorized manipulation
        # try a dynamic feasibility check using cvxpy with binary search refinement
        
        h_temp = h_var.value
        if h_temp is None:
            h_temp = np.ones(n_points) * 0.5
        else:
            h_temp = np.clip(h_temp, 0, 1)

        # Just use an intelligent Nd optimization with SLSQP
        max_attempts = 5
        attempt = 0
        success = False
        while attempt < max_attempts and not success and time() - start_time < budget_s * 0.9:

                bounds = [(0, 1) for _ in range(n_points)]

                res = minimize(
                    lambda x: evaluate_erdos_solution(x, dx),
                    h_temp.copy(),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons_sum,
                    options={'ftol': 1e-8, 'maxiter': 200}

                    h_temp = np.clip(h_temp, 0, 1)

                best_c5 = current_c5

        if current_c5 <= C_mid:
            C_high = C_mid
        else:
            C_low = C_mid
            
        if C_high - C_low < tolerance:
            break

    # Final check with detailed optimization

            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
