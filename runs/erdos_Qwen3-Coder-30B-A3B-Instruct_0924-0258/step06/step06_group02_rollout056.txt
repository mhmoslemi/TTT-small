import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        return h
    scale = n / (2 * sum_h)
    h *= scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing with focused perturbations
    T = 0.5
    T_min = 1e-4
    cooling_rate = 0.95
    max_steps = 2000
    step = 0
    
    while T > T_min and step < max_steps and time() - start_time < budget_s * 0.8:
        # Identify indices with high overlap
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        top_indices = np.argsort(corr_full)[-int(n_points*0.15):]
        top_indices = top_indices[top_indices > n_points//2]
        
        # Perturb
        h_new = best_h.copy()
        perturbed = np.zeros(n_points)
        
        for i in range(10):  # Perturb 10 random indices
            idx = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.1, 0.1)
            h_new[idx] += delta
            perturbed[idx] = 1
            
        # Apply perturbations more aggressively to top overlap indices
        for idx in top_indices:
            if np.random.rand() < 0.7:
                delta = np.random.uniform(-0.15, 0.15)
                h_new[idx] += delta
                perturbed[idx] = 1
            
        # Project to feasible set
        h_new = project_feasible(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        delta_c5 = c5_new - best_c5
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Cooling schedule
        if step % 50 == 0:
            T *= cooling_rate
            
        step += 1
        
        if time() - start_time > budget_s * 0.8:
            break
    
    # QCP Refinement with All Shifts
    if time() - start_time < budget_s * 0.95:
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add overlap constraints for all possible shifts
            for k in range(1, n_points):
                if k <= n_points - k:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass
    
    # Final projection and refinement
    try:
        h_final = project_feasible(best_h, n_points)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
