import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available, otherwise initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create triangular window initialization
        h_start = 1 - np.abs(np.arange(n_points) / (n_points-1) - 0.5) * 2
        # Add small random noise
        h_start += np.random.normal(0, 0.02, n_points)
        h_start = np.clip(h_start, 0, 1)
        # Normalize to sum = n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # DE parameters
    popsize = 50
    maxiter = 300
    F = 0.7
    CR = 0.9
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Custom DE callback to track progress
    def de_callback(xk, convergence):
        nonlocal best_h, best_c5
        if time() - start_time > budget_s * 0.8:
            return True
        h = np.array(xk)
        # Ensure constraint ∫h(x)dx = 1
        h = h * (n_points/2) / np.sum(h)
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        return False
    
    # Perform DE
    try:
        result = differential_evolution(
            lambda h: evaluate_erdos_solution(h, dx),
            bounds,
            args=(),
            strategy='best1bin',
            maxiter=maxiter,
            popsize=popsize,
            tol=1e-6,
            mutation=(F,),
            recombination=CR,
            callback=de_callback,
            seed=seed,
            disp=False,
            polish=False
        )
        
        if result.success:
            h_final = np.array(result.x)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Local refinement for top candidates
    num_refinements = 10
    top_candidates = []
    
    # Gather top candidates from DE run (if any)
    if 'result' in locals():
        try:
            if 'top_candidates' not in locals() or len(top_candidates) == 0:
                # Create sample candidates around the final result
                top_candidates = [best_h]
                for _ in range(num_refinements - 1):
                    pert = np.random.normal(0, 0.01, n_points)
                    candidate = best_h + pert
                    candidate = np.clip(candidate, 0, 1)
                    candidate = candidate * (n_points/2) / np.sum(candidate)
                    top_candidates.append(candidate)
        except Exception:
            top_candidates = [best_h]
    
    # Apply L-BFGS-B refinement
    for h_candidate in top_candidates[:num_refinements]:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # L-BFGS-B refinement
            res = minimize(objective, h_candidate, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            
            if res.success:
                h_refined = res.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            continue
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
