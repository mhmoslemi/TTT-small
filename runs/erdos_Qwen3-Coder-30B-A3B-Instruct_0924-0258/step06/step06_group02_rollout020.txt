import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

                sequence.append(val)
            return np.array(sequence)
        
        h_start = halton_like(n_points)
        h_start = (h_start - np.min(h_start)) / (np.max(h_start) - np.min(h_start))

    # Gradient-based optimization
    max_iterations = 1000
    learning_rate = 0.02

        gradient = np.zeros(n_points)

                gradient[:n_points-k] += (1 - best_h[k:])
                # h[i+k] * (1 - h[i]) contributes to gradient[i+k]

        # Apply gradient
        new_h = best_h - learning_rate * gradient

        # Add small noise
        noise = np.random.normal(0, 0.005, n_points)
        candidate_h = best_h + noise
