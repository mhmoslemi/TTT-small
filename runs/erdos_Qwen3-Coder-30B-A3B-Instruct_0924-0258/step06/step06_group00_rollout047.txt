import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate initial candidate
    h_current = h_start.copy()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Multi-start optimization with periodic refinement
    max_restarts = 50
    restart_count = 0
    
    while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        h_restart = h_current.copy()
        
        # Add random perturbation
        perturbation = np.random.normal(0, 0.03, n_points)
        h_restart += perturbation
        h_restart = np.clip(h_restart, 0, 1)
        
        # Normalize
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        # Frequency domain refinement
        try:
            # Apply FFT and filtering
            H = fft(h_restart)
            # Apply a Hamming window to suppress high frequencies (smoothing)
            window = np.hamming(n_points)
            H_filtered = H * window
            h_refined = np.real(ifft(H_filtered))
            
            # Ensure feasibility
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            
            # Local optimization
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            
            if res.success:
                h_candidate = res.x
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    restart_count = 0
                else:
                    restart_count += 1
            else:
                restart_count += 1
        except Exception:
            restart_count += 1
    
    # Final refinement with local optimizer
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 300, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
