import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Homotopy method with convex relaxation
    max_homotopy_steps = 100
    homotopy_step = 0.01
    
    h_current = h_start.copy()
    current_c5 = best_c5
    
    for step in range(max_homotopy_steps):
        if time() - start_time > budget_s * 0.95:
            break
            
        t = step * homotopy_step
        if t > 1.0:
            t = 1.0
            
        try:
            # Convex relaxation with homotopy parameter
            h_opt = cp.Variable(n_points)
            h_target = np.ones(n_points) * (n_points/2) / n_points  # uniform
            
            # Objective: weighted sum of original and regularization
            objective = cp.Minimize(
                t * cp.max(cp.multiply(
                    cp.convolve(h_opt, 1 - h_opt, mode='full')[:n_points],
                    cp.ones(n_points)
                )) + (1 - t) * cp.sum_squares(h_opt - h_target)
            )
            
            # Constraints
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_opt.value
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                if c5_new < current_c5:
                    h_current = h_new.copy()
                    current_c5 = c5_new
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
        except Exception:
            pass
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
