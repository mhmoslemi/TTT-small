import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create sinusoidal starting point
        x = np.linspace(0, 2, 35)
        h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / 35))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (35 / 2) / np.sum(h_start)
        n_points = 35
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search with LP feasibility checks
    low, high = 0.3807, 0.38085
    tolerance = 1e-6
    max_iter = 50
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        c_mid = (low + high) / 2.0
        try:
            # Solve LP feasibility problem
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Sample lags for constraints
            lags_to_check = []
            for k in range(1, n_points - 1):
                lags_to_check.append(k)
            
            # Check constraint sampling
            sampled_lags = lags_to_check[:n_points // 10] if len(lags_to_check) > 10 else lags_to_check
            
            for k in sampled_lags:
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))                        
                constraints.append(corr_sum <= c_mid * dx + 1e-8)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
                high = c_mid
            else:
                low = c_mid
                
        except Exception:
            low = c_mid
    
    # Final refinement with gradient descent
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_fun(h):
            return np.sum(h) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint_fun}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Perform optimization
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 200, 'ftol': 1e-8})
        if result.success:
            best_h = result.x
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    # Ensure constraints are satisfied
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
