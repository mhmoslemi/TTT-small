import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n):
    """Create a triangular window centered at the middle."""
    mid = n // 2
    window = np.zeros(n)
    for i in range(n):
        window[i] = 1 - abs(i - mid) / (mid + 1)
    return window

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 35
    dx = 2.0 / n_points
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    else:
        h_start = triangular_window(n_points)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution
    def objective(h):
        # Renormalize to ensure sum(h) = n_points/2
        h = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE with a larger population and more iterations
    de_maxiter = 300
    de_popsize = 50
    de_strategy = 'best1bin'
    de_tol = 1e-6
    
    try:
        result = differential_evolution(objective, bounds, 
                                       strategy=de_strategy,
                                       maxiter=de_maxiter,
                                       popsize=de_popsize,
                                       tol=de_tol,
                                       seed=seed,
                                       workers=1,
                                       updating='immediate',
                                       disp=False)
        
        if result.success:
            h_de = result.x * (n_points/2) / np.sum(result.x)
            c5_de = evaluate_erdos_solution(h_de, dx)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
                
    except Exception:
        pass
    
    # Local refinement using L-BFGS-B
    if time() - start_time > budget_s * 0.9:
        return (list(best_h), best_c5, n_points)
    
    # Perform local optimization with multiple restarts
    # This is a simple approach: restart from best DE solution
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def objective_limited(h):
        return evaluate_erdos_solution(h, dx)
    
    # Create bounds and constraints for L-BFGS-B
    bounds_lbfgs = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Try several random restarts for better local optimization
    restarts = 3
    for _ in range(restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Use current best as starting point for local search
        x0 = best_h.copy()
        
        # Perturb slightly to avoid local minima at the same point
        perturbation = np.random.normal(0, 0.01, n_points)
        x0 = x0 + perturbation
        x0 = np.clip(x0, 0, 1)
        x0 = x0 * (n_points/2) / np.sum(x0)
        
        try:
            res = minimize(objective_limited, x0, method='L-BFGS-B',
                           bounds=bounds_lbfgs, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6})
            
            if res.success:
                h_local = res.x
                h_local = np.clip(h_local, 0, 1)
                h_local = h_local * (n_points/2) / np.sum(h_local)
                c5_local = evaluate_erdos_solution(h_local, dx)
                if c5_local < best_c5:
                    best_h = h_local
                    best_c5 = c5_local
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
