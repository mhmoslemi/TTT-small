import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

        for i in range(n_points):

            else:
                h_start[i] = 0.7
        # Normalize to sum = n_points / 2
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a more sophisticated optimization approach
    # Gradient-based method with projection to constraints
    max_iterations = 1000
    learning_rate = 0.01

        if time() - start_time > budget_s * 0.90:

        # Compute gradient using finite differences for better stability
        gradient = np.zeros(n_points)

        for j in range(n_points):

            h_temp = np.clip(h_temp, 0, 1)

        learning_rate *= decay

            best_h = h_final

    except:

    return (list(best_h), best_c5, n_points)
