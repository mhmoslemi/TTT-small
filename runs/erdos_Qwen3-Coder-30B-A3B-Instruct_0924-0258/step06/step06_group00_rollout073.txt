import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Create a piecewise constant function that's close to the ideal
        h_start = np.ones(n_points) * 0.5
        # Set up a better initial guess with some structure
        half_idx = n_points // 2
        h_start[:half_idx] = 0.7
        h_start[half_idx:] = 0.3
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Number of restarts
    max_restarts = 50
    restart_count = 0
    
    while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Random perturbation with better control
        h_restart = h_start.copy()
        # Add some heterogeneity
        pert = np.random.normal(0, 0.05, n_points)
        h_restart = h_restart + pert
        # Clamp between [0,1]
        h_restart = np.clip(h_restart, 0, 1)
        # Re-normalize to sum to n_points/2
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        # Local optimization using SLSQP
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0  # Reset restart count
            else:
                restart_count += 1
        except Exception:
            restart_count += 1
        
        # Sometimes just slightly perturb the best and try again
        if restart_count > 0 and restart_count % 5 == 0:
            h_start = best_h + np.random.normal(0, 0.1, n_points) * 0.1
            h_start = np.clip(h_start, 0, 1)
            h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # Final refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
