import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Direct optimization with structured approach
    # Start with multiple random perturbations of the initial solution
    population_size = 100
    population = []
    
    # Generate diverse initial population
    population.append(h_start.copy())
    
    for _ in range(population_size - 1):
        h_perturb = h_start + np.random.normal(0, 0.03, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        # Normalize to correct sum
        h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
        population.append(h_perturb)
    
    # GA parameters
    generations = 200
    elite_size = population_size // 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness (lower C5 is better)
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            
        # Elitism
        elite = population[:elite_size]
        
        # New population
        new_population = elite.copy()
        
        # Selection and reproduction
        while len(new_population) < population_size:
            # Tournament selection
            p1_idx = random.randint(0, population_size // 2)
            p2_idx = random.randint(0, population_size // 2)
            p1 = population[p1_idx]
            p2 = population[p2_idx]
            
            # Crossover
            if random.random() < crossover_rate:
                mask = np.random.random(n_points) < 0.5
                child = np.where(mask, p1, p2)
            else:
                child = p1.copy()
                
            # Mutation
            if random.random() < mutation_rate:
                noise = np.random.normal(0, 0.02, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                
            # Normalize sum
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population[:population_size]
        
        # Periodic local refinement on top performers
        if gen % 10 == 0:
            for i in range(min(10, len(population))):
                h_curr = population[i]
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                try:
                    res = minimize(objective, h_curr.copy(), method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-8})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < fitness[i]:
                            population[i] = h_refined
                            fitness[i] = c5_refined
                except Exception:
                    pass
    
    # Final SLSQP refinement on the best
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
