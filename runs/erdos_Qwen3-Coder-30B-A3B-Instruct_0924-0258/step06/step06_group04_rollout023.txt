import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-start optimization
    n_starts = 5
    for start_idx in range(n_starts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate different initial points
        h_init = h_start.copy()
        if start_idx > 0:
            # Add some variation
            noise = np.random.normal(0, 0.05, n_points)
            h_init = h_init + noise
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points/2) / np.sum(h_init)
        
        # Local optimization with different strategies
        h_current = h_init.copy()
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        # Use a simple gradient-based approach with bounds
        max_iter = 300
        step_size = 0.05
        decay_rate = 0.995
        min_step = 0.0001
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb with gradient-like direction
            # Compute approximate gradient using finite differences
            grad = np.zeros(n_points)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_current.copy()
                h_plus[j] = min(1.0, h_plus[j] + eps)
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                
                h_minus = h_current.copy()
                h_minus[j] = max(0.0, h_minus[j] - eps)
                h_minus = h_minus * (n_points/2) / np.sum(h_minus)
                
                c5_minus = evaluate_erdos_solution(h_minus, dx)
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update step
            h_new = h_current - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                step_size = min(0.1, step_size * 1.05)
            else:
                step_size *= decay_rate
                step_size = max(min_step, step_size)
            
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
        
        # Refinement with convex optimization if we have time
        if time() - start_time < budget_s * 0.8:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample key lags for constraints
                lags = [1, 2, 4, 7, 10, 15]
                for k in lags:
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
    
    # Final check and normalize
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
