import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try multiple strategies with different starting points
    strategies = []
    
    # Strategy 1: Original construction
    strategies.append(("original", h_start.copy()))
    
    # Strategy 2: Piecewise constant with different values
    h_piecewise = np.ones(n_points)
    mid = n_points // 2
    h_piecewise[:mid] = 0.3
    h_piecewise[mid:] = 0.7
    h_piecewise = h_piecewise * (n_points/2) / np.sum(h_piecewise)
    strategies.append(("piecewise", h_piecewise))
    
    # Strategy 3: Sine wave modulation
    x = np.linspace(0, 2, n_points)
    h_sine = 0.5 + 0.3 * np.sin(3 * np.pi * x)
    h_sine = np.clip(h_sine, 0, 1)
    h_sine = h_sine * (n_points/2) / np.sum(h_sine)
    strategies.append(("sine", h_sine))
    
    # Strategy 4: Random construction
    h_random = np.random.random(n_points)
    h_random = h_random * (n_points/2) / np.sum(h_random)
    strategies.append(("random", h_random))
    
    # Strategy 5: Simple step function with peak
    h_step = np.ones(n_points) * 0.5
    h_step[n_points//3:2*n_points//3] = 0.8
    h_step = h_step * (n_points/2) / np.sum(h_step)
    strategies.append(("step", h_step))
    
    # Optimization using multiple strategies
    for strategy_name, h_start in strategies:
        if time() - start_time > budget_s * 0.95:
            break
            
        h_current = h_start.copy()
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Local optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            # SLSQP optimization
            res = minimize(objective, h_current, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 500, 'ftol': 1e-10})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
        
        # Additional local search with random restarts
        max_restarts = 10
        for _ in range(max_restarts):
            if time() - start_time > budget_s * 0.95:
                break
            h_restart = np.random.random(n_points) * 0.2 + 0.4  # Avoid extreme values
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                              bounds=bounds, constraints=cons_sum, 
                              options={'maxiter': 300, 'ftol': 1e-9})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
                
    # One final refinement for best solution
    
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_final, method='SLSQP', 
                      bounds=bounds, constraints=cons_sum, 
                      options={'maxiter': 1000, 'ftol': 1e-12})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
