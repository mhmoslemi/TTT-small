import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available, otherwise create a sinusoidal pattern
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / 2)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search to find better C5
    low = 0.3807
    high = 0.381
    tolerance = 1e-6
    iterations = 0
    max_iter = 50
    
    while iterations < max_iter and high - low > tolerance:
        if time() - start_time > budget_s * 0.95:
            break
            
        c_guess = (low + high) / 2.0
        
        # LP formulation for feasibility check of current guess C
        try:
            h_var = cp.Variable(n_points)
            constraints = [
                cp.sum(h_var) == n_points / 2.0,
                h_var >= 0,
                h_var <= 1
            ]

            # Add overlap constraints for key lags
            for k in range(1, n_points):
                if k <= n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_var[k-n_points+1:], 1 - h_var[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= c_guess * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            
            if prob.status == cp.OPTIMAL:
                high = c_guess
            else:
                low = c_guess
                
        except Exception:
            low = c_guess
        
        iterations += 1
    
    # If we have a feasible solution, refine it
    if high < 0.381:
        try:
            h_refine = cp.Variable(n_points)
            constraints = [
                cp.sum(h_refine) == n_points / 2.0,
                h_refine >= 0,
                h_refine <= 1
            ]
            # Key lags for overlap constraints
            step = max(1, n_points // 10)
            for k in range(1, n_points, step):
                if k <= n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_refine[:n_points-k], 1 - h_refine[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_refine[k-n_points+1:], 1 - h_refine[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= high * dx + 1e-6)

            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            
            if prob.status == cp.OPTIMAL:
                best_h = h_refine.value
                best_c5 = high
        except Exception:
            pass
    
    # Final refinement using gradient-based method
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, 
                          options={'maxiter': 1000, 'ftol': 1e-12})
        if result.success:
            best_h = result.x
            best_c5 = objective(best_h)
    except Exception:
        pass
    
    # Ensure final constraints are met
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
