import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n_points))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Adaptive gradient descent with local refinement
    max_iter = 2000
    step_size = 0.05
    decay_rate = 0.999
    min_step = 0.0001
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Gradient approximation using finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            h_minus = best_h.copy()
            h_minus[j] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update
        h_new = best_h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        
        # Decay step size
        step_size = max(min_step, step_size * decay_rate)
    
    # Phase 2: Exact optimization with quadratic constraints
    try:
        # Sample only key lags to reduce problem size
        lag_indices = np.arange(1, n_points//2, max(1, n_points//20))
        
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraints corresponding to sampled lags
        for k in lag_indices:
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Phase 3: Local search refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_eq(h):
        return np.sum(h) - n_points / 2
    
    def constraint_ineq(h):
        return np.array([1 - h.max(), h.min()])
    
    # Use scipy minimize for final refinement
    try:
        constraints = [
            {'type': 'eq', 'fun': constraint_eq},
            {'type': 'ineq', 'fun': lambda h: 1 - h.max()},
            {'type': 'ineq', 'fun': lambda h: h.min()}
        ]
        
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            constraints=constraints,
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 1000, 'ftol': 1e-8}
        )
        
        if result.success:
            final_h = np.clip(result.x, 0, 1)
            final_h = final_h * (n_points/2) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
    
    # Final check and return
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
