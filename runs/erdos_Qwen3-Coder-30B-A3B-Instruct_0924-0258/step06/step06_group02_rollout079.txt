import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with foveated perturbations
    T = 0.5
    T_min = 1e-4
    cooling_rate = 0.99
    max_steps = 2000
    
    # For temperature annealing
    for step in range(max_steps):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Calculate current overlap to determine where to perturb
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Find indices with highest overlap values (top 10%)
        overlap_indices = np.argsort(corr)[-max(1, n_points//10):]
        # Focus perturbation on these indices
        perturb_indices = np.random.choice(overlap_indices, size=max(1, n_points//20), replace=False)
        
        h_new = best_h.copy()
        # Perturb selected indices
        for idx in perturb_indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_new[idx] += delta
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Normalize to correct sum
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on simulated annealing criterion
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Cool down
        if T > T_min:
            T *= cooling_rate
    
    # Gradient-based refinement using CVXPY
    try:
        # Model with fixed structure from best solution
        h_opt = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(h_opt, 1 - h_opt)))
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, eps=1e-8)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final check and refinement
    try:
        # Project to ensure constraints
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points/2) / np.sum(best_h)
        final_c5 = evaluate_erdos_solution(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
