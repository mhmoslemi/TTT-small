import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        # Structured initial guess based on symmetry and binary-like patterns
        h_start = np.zeros(n_points)
        # Place peaks at symmetric locations
        peaks = [0.4, 0.8, 1.2, 1.6]
        for peak in peaks:
            idx = int(peak / 2.0 * n_points)
            h_start[max(0, idx-3):min(n_points, idx+4)] = 0.6
        h_start = h_start * (n_points / (2 * np.sum(h_start)))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)

    # Trust region adaptive optimization
    tr_radius = 0.3
    max_iter = 1000
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Perturbation within trust region
        noise = np.random.uniform(-tr_radius, tr_radius, n_points)
        h_new = best_h + noise
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize integral constraint exactly
        sum_h = np.sum(h_new)
        if sum_h > 0:
            h_new = h_new * (n_points / (2 * sum_h))
            
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            tr_radius = min(0.8, tr_radius * 1.05)  # Expand trust region
        else:
            tr_radius = max(0.02, tr_radius * 0.9)  # Contract trust region
            
    # Final SDP refinement to tighten bounds
    try:
        h_sdp = cp.Variable(n_points)
        constraints = [
            cp.sum(h_sdp) == n_points / 2.0,
            h_sdp >= 0,
            h_sdp <= 1
        ]
        
        # Compute overlap expression with full correlation
        corr_vars = []
        for k in range(1, n_points):
            if k <= n_points // 2:
                corr_var = cp.sum(cp.multiply(h_sdp[:n_points-k], 1 - h_sdp[k:]))
            else:
                corr_var = cp.sum(cp.multiply(h_sdp[k-n_points+1:], 1 - h_sdp[:n_points-(k-n_points+1)]))
            corr_vars.append(corr_var)
            
        max_overlap = cp.max(cp.vstack(corr_vars))
        prob = cp.Problem(cp.Minimize(max_overlap), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_sdp.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final projection to exact integral constraint
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * (n_points / (2 * sum_h))
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
