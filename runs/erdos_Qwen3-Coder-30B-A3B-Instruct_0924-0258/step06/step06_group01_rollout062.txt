import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    # Instead of genetic algorithm or parseable heuristics, 
    # using projected gradient descent with convexification of the max operation

    v = np.zeros_like(h)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:

        # Compute current overlap gradients

        # We approximate the max operation using smooth max (log-sum-exp)
        # This is an approximation; exact max is not differentiable
        # Here we compute a quadratic surrogate at current h
        # and solve the local problem
        
        # Compute correlations for current h
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Compute the gradient of the overlap w.r.t. h
        # at shift k = max_idx - (2 * n_points - 1) // 2 
        shift = max_idx - (2 * n_points - 1) // 2
        n_corr = len(corr)
        grad = np.zeros_like(h)
        
        # For shift k, derivative is (1 - h_{i+k}) for h_i
        # (item in h with corresponding convolution index)
        # But we need to be careful with the zero padding in correlation
        # We compute the full convolution contribution of this shift
        
        # Construct the correlation kernel and gradient
        # Create shifted version of 1-h
        shifted_h = np.concatenate([np.zeros(shift), 1-h, np.zeros(n_points-1-shift)])
        # Note: correlation computation in numpy gives result of length 2*n-1 for
        # vectors of same size, so idx goes up to 2*n-2
        
        # We just want to compute the gradient of correlation value at max_idx
        # which is the QP (second order Taylor approx of the original)
        # We can approximate the quadratic surrogate
        
        # Compute actual gradient manually for single max term
        # Print correlation indices for debugging
        local_idx = 0
        # Compute local gradient for max overlap
        for i in range(n_points):
            # For correlation index i+shift+n_points-1, we are looking at:
            # Contribution to overlap at shift k = max_idx - (n_points-1)
            # This is in the full convolution output
            corr_idx = i + shift + n_points - 1
            if corr_idx < n_corr and corr_idx >= 0:
                # The derivative of h[i] w.r.t. this overlap is 1 - h[i+shift] 
                # (in the sense of convolution ?)
                if i + shift >= 0 and i + shift < n_points:
                    grad[i] = 1 - h[i+shift]
        
        # Or more precisely - let's compute the exact gradient meaningfully
        # By Loop:

    try:

        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-10, 'maxiter': 1000}
        )
        if res.success:
            h_ref = res.x
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (n_points/2) / np.sum(h_ref)
            final_c5 = evaluate_erdos_solution(h_ref, dx)
            if final_c5 < best_c5:
                best_h = h_ref
                best_c5 = final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
