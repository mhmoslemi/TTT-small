import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization
    max_iter = 5000
    step_size = 0.01
    noise_scale = 0.05
    noise_decay = 0.995
    patience = 0
    max_patience = 100
    
    h_current = best_h.copy()
    current_c5 = best_c5
    
    for idx in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps and find max
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_k = np.argmax(corr)
        max_overlap = corr[max_k] * dx
        
        # Compute the gradient: derivative of sum(h[i]*(1-h[i+k])) w.r.t. h[i]
        gradient = np.zeros(n_points)
        k = max_k - n_points + 1  # k is negative or zero
        
        # Gradient computation for indices where i+k >= 0
        if k >= 0:
            for i in range(n_points - k):
                gradient[i] += 1 - h_current[i + k]
            # For negative k values: i+k < 0, need to consider i-k >= 0 case
            # We already handled the -k terms using i-k >= 0 for i >= k
        else:
            for i in range(n_points):
                if i - k < n_points:
                    gradient[i] -= h_current[i - k]
                
        # Handle the symmetric case
        if k < 0:
            for i in range(n_points):
                if i + k >= 0:
                    gradient[i] -= h_current[i + k]
        
        # Normalize gradient to ensure reasonable step size
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient /= grad_norm
            gradient *= step_size
            
        # update with gradient and clip
        h_new = h_current + gradient
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < current_c5:
            h_current = h_new.copy()
            current_c5 = c5_new
            patience = 0
            
            if c5_new < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_new
        else:
            patience += 1
            
            # Add small noise periodically and repair
            if patience % 20 == 0 or patience >= max_patience:
                noise = np.random.normal(0, noise_scale, n_points)
                h_new = h_current + noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                if c5_new < current_c5:
                    h_current = h_new.copy()
                    current_c5 = c5_new
                    
                    if c5_new < best_c5:
                        best_h = h_current.copy()
                        best_c5 = c5_new
                        
                noise_scale *= noise_decay
                patience = 0
                
    # Final convex optimization refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample a few lags, avoid dense constraints
        k_vals = np.linspace(0.01, 1.99, min(15, n_points//2))
        for k_val in k_vals:
            k = int(np.round(k_val * n_points / 2))
            if k < n_points and k > 0:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final gradient-descent refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
