import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a sinusoidal-like initialization
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(2 * np.pi * x / 2)
        h_start = np.clip(h_start, 0, 1)
    
    # Normalize to sum to n_points / 2
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Perform gradient-based optimization with constraint handling
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
        
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': lambda h: constraint_bounds(h)[n_points:]},
        {'type': 'ineq', 'fun': lambda h: constraint_bounds(h)[:n_points]}
    ]
    
    # Adaptive Nelder-Mead optimization with smart restarts
    max_iter = 300
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perform local optimization with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, 
                            constraints=constraints, options={'maxiter': 10})
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
        except:
            pass
            
        # Random perturbation with re-normalization for exploration
        if time() - start_time > budget_s * 0.8:
            # Increase exploration intensity as time runs out
            noise_scale = 0.01
        else:
            noise_scale = 0.05
            
        noise = np.random.normal(0, noise_scale, n_points)
        h_new = best_h + noise
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            
    # Final QC with an improved solver
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        corr_vals = []
        for k in range(1, n_points):
            if k <= n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            corr_vals.append(corr_sum)
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Normalize and finalize
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
