import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n_points):
    """Create a triangular window function that peaks in the middle."""
    x = np.arange(n_points)
    mid = (n_points - 1) / 2
    window = 1 - np.abs(x - mid) / (n_points / 2)
    return np.clip(window, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = triangular_window(n_points)
    
    dx = 2.0 / n_points
    
    # Normalize initial solution
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # DE parameters
    popsize = 50
    maxiter = 300
    mutation = 0.7
    crossover = 0.9
    
    # Constraint enforcement
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    # Objective function for DE
    def de_objective(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        return evaluate_erdos_solution(h, dx)
    
    # Bounds for DE
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Differential Evolution
    de_result = differential_evolution(
        de_objective,
        bounds,
        seed=seed,
        maxiter=maxiter,
        popsize=popsize,
        mutation=mutation,
        crossover=crossover,
        disp=False,
        callback=lambda x, convergence: None
    )
    
    if de_result.success:
        h_de = normalize(np.clip(de_result.x, 0, 1))
        c5_de = evaluate_erdos_solution(h_de, dx)
        if c5_de < best_c5:
            best_h = h_de
            best_c5 = c5_de
    
    # Refinement with L-BFGS-B
    def lbfgs_objective(h):
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Select best individuals for refinement
    candidates = [best_h]
    for _ in range(10):
        if time() - start_time > budget_s * 0.9:
            break
        h_rand = np.random.random(n_points)
        h_rand = normalize(h_rand)
        candidates.append(h_rand)
    
    # Refine candidates with L-BFGS-B
    for h in candidates:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Ensure h is within bounds
        h = np.clip(h, 0, 1)
        h = normalize(h)
        
        # L-BFGS-B optimization with constraints
        res = minimize(
            lbfgs_objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 200, 'ftol': 1e-8},
            tol=1e-8
        )
        
        if res.success:
            h_refined = normalize(np.clip(res.x, 0, 1))
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = normalize(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
