import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        h_start = np.clip(h_start, 0, 1)
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try a few different optimization approaches
    for i in range(20):
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            # Random perturbation + normalization
            noise = np.random.normal(0, 0.05, n_points)
            h_new = best_h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    
    # Constrained optimization with CVXPY
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add overlap constraints using a subset of lags to reduce problem size
        max_lags = min(20, n_points // 2)
        for k in range(1, max_lags):
            if 2*k < n_points:
                # For direct correlation
                corr = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                # Wrap-around case
                corr = cp.sum(cp.multiply(h_opt[n_points-k:], 1 - h_opt[:k]))
            constraints.append(corr <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final refinement by shooting method with simulated annealing
    try:
        current_h = best_h.copy()
        current_c5 = best_c5
        temp = 0.1
        decay = 0.999
        step_size = 0.01
        
        for i in range(10000):
            if time() - start_time > budget_s * 0.97:
                break

            perturbation = np.random.normal(0, step_size, n_points)

                current_c5 = new_c5
            elif np.random.rand() < np.exp(-(new_c5 - current_c5) / temp):

        pass

    best_h = np.clip(best_h, 0, 1)

    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
