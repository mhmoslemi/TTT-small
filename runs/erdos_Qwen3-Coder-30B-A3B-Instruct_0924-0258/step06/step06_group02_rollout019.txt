import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def check_feasibility(C_candidate, h, n, dx, sample_rate=0.2):
    # Randomly sample shifts to reduce constraints
    k_samples = np.random.choice(range(1, n), size=int(n * sample_rate), replace=False)
    
    # Define decision variables
    h_opt = cp.Variable(n)
    
    # Integral constraint
    constraints = [cp.sum(h_opt) == n / 2]
    
    # Bounds
    constraints += [0 <= h_opt, h_opt <= 1]
    
    # Overlap constraints for sampled shifts
    for k in k_samples:
        corr_sum = cp.sum(cp.multiply(h_opt[:n-k], 1 - h_opt[k:]))
        constraints.append(corr_sum <= C_candidate / dx)
    
    # Formulate and solve problem
    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=cp.ECOS, eps=1e-6)
        return prob.status == cp.OPTIMAL
    except:
        return False

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search for the minimal C5
    low = 0.38
    high = best_c5
    iterations = 20
    
    for _ in range(iterations):
        if time() - start_time > budget_s * 0.95:
            break
        mid = (low + high) / 2
        if check_feasibility(mid, best_h, n_points, dx):
            best_c5 = mid
            high = mid
        else:
            low = mid

    # Refinement with all shifts using the best C5
    try:
        h_final = cp.Variable(n_points)
        constraints = [
            cp.sum(h_final) == n_points / 2,
            h_final >= 0,
            h_final <= 1
        ]
        
        # Constraint for all shifts
        for k in range(1, n_points):
            corr_sum = cp.sum(cp.multiply(h_final[:n_points-k], 1 - h_final[k:]))
            constraints.append(corr_sum <= best_c5 / dx)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_result = h_final.value
        else:
            h_result = best_h.copy()
    except Exception:
        h_result = best_h.copy()
    
    # Final projection to exact integral
    h_result = np.clip(h_result, 0, 1)
    h_result = h_result * (n_points/2) / np.sum(h_result)
    final_c5 = evaluate_erdos_solution(h_result, dx)
    
    return (list(h_result), final_c5, n_points)
