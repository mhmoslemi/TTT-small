import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a sinusoidal-like initialization to avoid local minima
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(2 * np.pi * x / 2)
        h_start = np.clip(h_start, 0, 1)
    
    # Normalize to sum to n_points / 2
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a more robust optimization approach with simulated annealing
    max_iter = 5000
    temp = 1.0
    temp_decay = 0.999
    min_temp = 0.001
    step_size = 0.1
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Simulated annealing perturbation
        noise = np.random.normal(0, step_size, n_points)
        h_new = best_h + noise
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        elif np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
            best_h = h_new
            best_c5 = c5_new
        
        # Cool down
        temp = max(min_temp, temp * temp_decay)
    
    # LP refinement to improve feasibility
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraints for several key lags only to reduce complexity
        lags_to_check = [1, 2, 4, 8, 16]  # Sample a few key lags
        for k in lags_to_check:
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final validation and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
