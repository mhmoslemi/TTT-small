import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Homotopy approach to escape local optima
    max_homotopy_steps = 100
    t_values = np.linspace(0, 1, max_homotopy_steps)
    
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    current_c5 = best_c5
    
    for t in t_values:
        if time() - start_time > budget_s * 0.95:
            break
        
        # Convex relaxation with homotopy parameter t
        try:
            h_opt = cp.Variable(n_points)
            # Regularization term: (1-t)*||h - uniform||^2
            uniform = np.ones(n_points) * 0.5
            reg_term = (1 - t) * cp.sum_squares(h_opt - uniform)
            
            # Objective: t * C5 + (1-t) * regularization
            # But we handle constraint differently for CVXPY
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add a few key constraint samples to make it more robust
            k_vals = np.arange(1, min(10, n_points), 2)
            for k in k_vals:
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(reg_term), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=200)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                # Repair feasibility and compute objective
                h_feasible = np.clip(h_feasible, 0, 1)
                h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                
                if c5_feasible < current_c5:
                    h_current = h_feasible.copy()
                    current_c5 = c5_feasible
                    if c5_feasible < best_c5:
                        best_h = h_feasible.copy()
                        best_c5 = c5_feasible
                        
        except Exception:
            pass
            
        # Small random perturbation for exploration
        if time() - start_time > budget_s * 0.8:
            break 
        noise = np.random.normal(0, 0.01, n_points)
        h_new = h_current + noise
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < current_c5:
            h_current = h_new.copy()
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new

    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
