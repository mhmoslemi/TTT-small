import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Use triangular window initialization
        h_start = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h_start[i] = 1.0 - abs(i - mid) / (mid + 1)
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds and constraints for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Differential Evolution with adaptive parameters
    de_options = {
        'maxiter': 300,
        'popsize': 50,
        'mutation': (0.5, 1),
        'recombination': 0.9,
        'strategy': 'best1bin',
        'tol': 1e-6,
        'atol': 1e-6,
        'updating': 'immediate',
        'workers': 1,
    }
    
    # Optimizer function for DE
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    try:
        # Run differential evolution
        res = differential_evolution(objective, bounds, constraints=({'type': 'eq', 'fun': constraint_sum}), 
                                     options=de_options, seed=seed)
        
        if res.success:
            h_de = res.x
            c5_de = objective(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de

        # Local refinement with L-BFGS-B on top results
        n_refinement = min(10, len(best_h) // 2)
        top_indices = np.argsort([evaluate_erdos_solution(h_start, dx)] + 
                                [evaluate_erdos_solution(np.random.random(n_points) * 0.5, dx) 
                                 for _ in range(n_refinement * 2)])[1:n_refinement+1]
        
        # Refinement phase
        for _ in range(n_refinement):
            if time() - start_time > budget_s * 0.9:
                break
                
            # Pick a good candidate
            temp_h = best_h.copy()
            # Slight random perturbation to get a new sample
            perturb = np.random.normal(0, 0.01, n_points)
            temp_h += perturb
            temp_h = np.clip(temp_h, 0, 1)
            # Normalize
            temp_h = temp_h * (n_points/2) / np.sum(temp_h)
            
            # Local optimization for this candidate
            def local_objective(h):
                return evaluate_erdos_solution(h, dx)
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds_local = [(0, 1) for _ in range(n_points)]
            
            # L-BFGS-B optimization
            try:
                res = minimize(local_objective, temp_h, method='L-BFGS-B', 
                               bounds=bounds_local, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = local_objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
                
            if time() - start_time > budget_s * 0.98:
                break
                
    except Exception as e:
        pass
        
    return (list(best_h), best_c5, n_points)
