import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial setup
    n_start = 10
    n_max = 35
    max_iter_per_level = 100
    sa_freq = 0.1
    
    best_h = None
    best_c5 = float('inf')
    
    # Grid refinement
    n_points = n_start
    while n_points <= n_max and (time() - start_time) < budget_s * 0.95:
        # Initialize h
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                h = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, len(h)), h)
        else:
            x = np.linspace(0, 2, n_points)
            h = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n_points))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        dx = 2.0 / n_points
        
        # Optimization loop
        for i in range(max_iter_per_level):
            if (time() - start_time) > budget_s * 0.95:
                break
                
            # Gradient descent step
            # Compute approximate gradient using finite differences
            grad = np.zeros(n_points)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                c5_minus = evaluate_erdos_solution(h_minus, dx)
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with gradient step
            alpha = 0.01
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Simulated Annealing step occasionally
            if np.random.random() < sa_freq:
                temp = 0.1 * np.exp(-i / 100)
                noise = np.random.normal(0, temp, n_points)
                h_new = h + noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Accept if better, or with probability based on temperature
                c5_old = evaluate_erdos_solution(h, dx)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < c5_old or np.random.random() < np.exp(-(c5_new - c5_old) / temp):
                    h = h_new
            else:
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < evaluate_erdos_solution(h, dx):
                    h = h_new
            
            # Update best solution
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        n_points = min(n_points * 2, n_max)
    
    # Final binary search refinement if needed
    if best_h is not None:
        # Binary search between current bound and target
        c5_lo = best_c5
        c5_hi = 0.3807  # target bound
        tries = 0
        while (c5_hi - c5_lo) > 1e-6 and tries < 20:
            tries += 1
            c5_target = (c5_lo + c5_hi) / 2
            # Try to construct a solution with this target
            try:
                h_opt = cp.Variable(len(best_h))
                constraints = [
                    cp.sum(h_opt) == len(best_h) / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                dx = 2.0 / len(best_h)
                # Add constraints based on the current estimate
                # For simplicity just make sure maximum correlation doesn't exceed
                # the target. Relax this to actually match the specific way C5 is computed.
                corr = np.correlate(best_h, 1 - best_h, mode='full')
                max_corr = np.max(corr)
                # This is a convolution check; we want max_corr*dx <= c5_target
                # But we need to make sure that is properly computed in the constraints
                # We could set up a full inequality for all lags which is expensive,
                # So we focus on the current upper bound.
                # Use a rough approximation: that we claim the maximum measurable correlation
                # is at most c5_target/dx.
                # Actually just solve: max correlation <= target
                # The current solution is best_h so we just verify:
                c5_test = evaluate_erdos_solution(best_h, dx)
                if c5_test <= c5_target:
                    c5_hi = c5_target
                else:
                    c5_lo = c5_target
            except:
                break
        best_c5 = c5_lo
    
    # Final validation
    if best_h is None:
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
        else:
            # Just use a simple square wave
            best_h = np.ones(35) * 0.5
        best_h = best_h * (35 / 2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h, 2.0 / 35)
    
    best_h = np.array(best_h)
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / len(best_h))
    
    return (list(best_h), best_c5, len(best_h))
