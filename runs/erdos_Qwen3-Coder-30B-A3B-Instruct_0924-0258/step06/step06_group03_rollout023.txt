import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        # Initialize with a Hamming window
        h = 0.54 + 0.46 * np.cos(np.pi * (np.arange(n_points) / n_points - 0.5) * 2)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Projected gradient descent
    max_iter = 1000
    eta = 0.01
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Compute the correlation for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        
        # Gradient: derivative of the maximum correlation with respect to h
        # This is the convolution of 1-h with delta function at max_lag_idx
        # For a discrete case, we compute this directly
        grad = np.zeros(n_points)
        corr_len = len(corr)
        for k in range(n_points):
            lag = k - (corr_len // 2)  # lag index
            if 0 <= k < corr_len:
                # derivative of correlation at lag
                if lag <= 0:
                    # h[i] contributes to correlation(k) where k = i + lag
                    # It's 1 - h[i + lag] for terms that should be considered
                    grad[i] += 1 if 0 <= i + lag < n_points else 0
                else:
                    # h[i] contributes to correlation(k) where k = i - lag
                    grad[i] += 1 if 0 <= i - lag < n_points else 0
        
        # Update h
        h_new = h - eta * grad
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Scale to satisfy integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
