import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        h_start = 1 - np.abs(np.arange(n_points) / (n_points - 1) - 0.5) * 2
        h_start = np.clip(h_start + np.random.normal(0, 0.05, n_points), 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # DE parameters
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Use differential evolution for global search
    de_options = {'maxiter': 150, 'popsize': 50, 'mutation': (0.5, 1.0), 
                  'recombination': 0.9, 'seed': seed, 'disp': False}

    remaining_time = budget_s * 0.8
    de_start = time()
    
    # Create initial population around the starting point
    def create_initial_pop():
        pop = []
        for _ in range(50):
            temp_h = h_start + np.random.normal(0, 0.05, n_points)
            temp_h = np.clip(temp_h, 0, 1)
            temp_h = temp_h * (n_points/2) / np.sum(temp_h)
            pop.append(temp_h)
        return pop
    
    # DE optimizations
    try:
        # Use a custom DE with a fixed mut and cross rate
        def de_optimize():
            # Initialize population
            population = create_initial_pop()

                if time() - start_time > budget_s * 0.8:
                    return

                fitness = [objective(ind) for ind in population]

                if fitness[min_fitness_idx] < best_c5:
                    best_h = population[min_fitness_idx].copy()
                    best_c5 = fitness[min_fitness_idx]
                
                # DE selection for next generation
                new_population = []
                
                # Keep elite
                elite_idx = np.argsort(fitness)[:2]
                for i in elite_idx:
                    new_population.append(population[i].copy())
                
                # Generate rest using DE
                for _ in range(48):
                    # Choose three distinct individuals
                    a, b, c = random.sample(range(50), 3)

                    trial = trial * (n_points/2) / np.sum(trial)
                    
                    # Crossover
                    if random.random() < 0.9:

        de_optimize()

        pass

    top_candidates = []

                return evaluate_erdos_solution(x, dx)
            
            try:

                if res.success:
                    refined = res.x
                    refined = refined * (n_points/2) / np.sum(refined)
                    refined_c5 = objective(refined)
                    if refined_c5 < best_c5:
                        best_h = refined
                        best_c5 = refined_c5
            except Exception:
                pass
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
