import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n):
    """Create a triangular window function"""
    half = n // 2
    if n % 2 == 0:
        window = np.concatenate([np.arange(1, half + 1), np.arange(half, 0, -1)])
    else:
        window = np.concatenate([np.arange(1, half + 2), np.arange(half, 0, -1)])
    return window
