import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate multiple initial candidates including structured ones
    def generate_initial_candidates():
        candidates = []
        # 1. Original initialization
        candidates.append(h_start.copy())
        # 2. Random perturbations
        for _ in range(5):
            h_pert = h_start + np.random.normal(0, 0.05, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            candidates.append(h_pert.copy())
        # 3. Alternate pattern
        h_alt = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                h_alt[i] = 0.8
            else:
                h_alt[i] = 0.2
        h_alt = h_alt * (n_points/2) / np.sum(h_alt)
        candidates.append(h_alt)
        return candidates
    
    # DE bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def de_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # DE constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Differential Evolution global optimization
    max_de_iter = 500
    popsize = 100
    maxiter = min(max_de_iter, int(budget_s / 0.5))  # ~0.5s per iter
    
    # Create initial population
    init_pop = generate_initial_candidates()
    
    # Run DE only if we have enough time
    if time() - start_time < budget_s * 0.9:
        try:
            result = differential_evolution(
                de_objective,
                bounds,
                args=(),
                strategy='best1bin',
                maxiter=maxiter,
                popsize=popsize,
                tol=1e-6,
                mutation=(0.5, 1),
                recombination=0.7,
                seed=seed,
                callback=None,
                disp=False,
                polish=True,
                init=init_pop,
                atol=1e-6
            )
            if result.success:
                h_de = result.x
                # Ensure it satisfies constraint
                h_de = h_de * (n_points/2) / np.sum(h_de)
                c5_de = de_objective(h_de)
                if c5_de < best_c5:
                    best_h = h_de
                    best_c5 = c5_de
        except Exception:
            pass
    
    # Local refinement with SLSQP on best from DE or original
    if time() - start_time < budget_s * 0.95:
        h_refine = best_h.copy()
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_points/2) / np.sum(h_refine)
        
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # SLSQP with multiple restarts for better local search
            best_local_h = h_refine.copy()
            best_local_c5 = objective(h_refine)
            
            for _ in range(20):
                if time() - start_time > budget_s * 0.95:
                    break
                h_restart = h_refine + np.random.normal(0, 0.01, n_points)
                h_restart = np.clip(h_restart, 0, 1)
                h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-9})
                
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_local_c5:
                        best_local_h = h_refined
                        best_local_c5 = c5_refined
            
            if best_local_c5 < best_c5:
                best_h = best_local_h
                best_c5 = best_local_c5
        except Exception:
            pass
    
    # Final check
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
