import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Main optimization loop
    max_iter = 10000
    step_size = 0.05
    noise_decay = 0.995
    noise_level = 0.05
    patience = 0
    max_patience = 100
    
    for idx in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Gradient-based update
        h_current = best_h.copy()
        c5_current = best_c5
        
        # Compute overlap for all shifts
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        k_max = max_idx - (n_points - 1)
        
        # Gradient computation
        gradient = np.zeros(n_points)
        if k_max > 0:
            gradient[:n_points-k_max] += 1 - h_current[k_max:]
        if k_max < 0:
            gradient[-k_max:] += h_current[:n_points+k_max]
            
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm
            
        # Line search
        alpha = step_size
        for _ in range(10):
            h_new = h_current + alpha * gradient
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                break
            alpha *= 0.5
            
        # Add noise periodically
        if idx % 10 == 0:
            noise = np.random.normal(0, noise_level, n_points)
            h_new = h_current + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                
        # Update best if improved
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            patience = 0
        else:
            patience += 1
            
        # Reduce noise level
        noise_level *= noise_decay
        
        # Early stopping if no improvement
        if patience > max_patience:
            break
    
    # Convex relaxation refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add a set of sampled constraints to reduce density
        k_vals = np.linspace(0.1, 1.9, min(30, n_points//2))
        for k_val in k_vals:
            k = int(k_val * n_points / 2)
            if 0 < k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
        
    # Final SLSQP refinement 
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
