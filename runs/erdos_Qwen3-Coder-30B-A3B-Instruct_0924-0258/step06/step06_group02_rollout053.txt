import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T = 0.5
    T_final = 1e-4
    cooling_rate = 0.99
    steps = 2000
    
    for i in range(steps):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Calculate current overlap for all shifts to focus perturbations
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        overlap_shifts = corr * dx
        # Focus on larger overlaps
        top_indices = np.argsort(overlap_shifts)[-max(1, n_points//10):]
        
        # Make perturbation
        h_new = best_h.copy()
        perturbed = np.zeros(n_points)
        
        # Perturb top-10% indices
        for idx in top_indices:
            if np.random.rand() < 0.7:  # 70% chance to perturb
                delta = np.random.choice([-0.1, 0.1])
                h_new[idx] += delta
                perturbed[idx] = 1
        
        # Clamp values
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on temperature
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Cooling schedule
        T = max(T_final, T * cooling_rate)
    
    # Gradient Descent refinement
    try:
        # Convert to CVXPY variables
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2.0,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Simplified approach with known optimal solution
        h_refined = best_h.copy()
        step_size = 1e-3
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
                
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            gradient = np.zeros(n_points)
            for i in range(n_points):
                grad_sum = 0
                for k in range(n_points):
                    if k < n_points:
                        if i+k < n_points:
                            grad_sum += (1 - h_refined[i+k])
                gradient[i] = grad_sum * dx
                
            h_refined -= step_size * gradient
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            
            if c5_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_refined
            step_size *= 0.99
            
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
