import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Parameters
    temp_initial = 0.5
    temp_final = 1e-4
    cooling_rate = 0.99
    max_anneal_steps = 2000
    perturbation_magnitude = 0.1
    
    # Simulated Annealing
    temp = temp_initial
    for step in range(max_anneal_steps):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Select indices with highest overlap impact
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        corr_values = corr_full[len(best_h)-1:]
        top_indices = np.argsort(corr_values)[-max(1, n_points // 10):]
        
        # Perturb selected indices
        h_new = best_h.copy()
        for i in top_indices:
            delta = np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
            h_new[i] += delta
        
        # Enforce bounds and normalization
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on temperature
        if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / temp):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Cool down temperature
        temp = max(temp_final, temp * cooling_rate)
    
    # Gradient-Based Refinement
    try:
        h_refined = best_h.copy()
        step_size = 1e-3
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Gradient estimation using finite differences
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                grad[i] = (c5_plus - best_c5) / eps
                
            # Update with gradient descent step
            h_refined = h_refined - step_size * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            
            # Update best if improved
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_refined
            
            step_size *= 0.995  # Decay step size
            
    except Exception:
        pass
    
    # Final validation and cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
