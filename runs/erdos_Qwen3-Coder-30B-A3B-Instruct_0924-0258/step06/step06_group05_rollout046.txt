import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a periodic pattern with roughly half ones
        h_start = np.zeros(n_points)
        ones_count = n_points // 2
        for i in range(ones_count):
            h_start[i * (n_points // ones_count)] = 1
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search with bit flips
    max_iter = 1000
    temperature = 1.0
    cooling_rate = 0.995
    accepted = 0
    stagnation_count = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor by swapping a 0 and 1
        h_new = best_h.copy()
        ones_indices = np.where(h_new > 0.5)[0]
        zeros_indices = np.where(h_new < 0.5)[0]
        
        if len(ones_indices) > 0 and len(zeros_indices) > 0:
            i = np.random.choice(ones_indices)
            j = np.random.choice(zeros_indices)
            h_new[i], h_new[j] = h_new[j], h_new[i]
            
            # Re-scale to keep sum = n_points/2
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept if better or with some probability
            if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / temperature):
                best_h = h_new.copy()
                best_c5 = c5_new
                accepted += 1
                last_improvement = iteration
                stagnation_count = 0
            else:
                stagnation_count += 1
                
            # Annealing
            temperature *= cooling_rate
            
            if stagnation_count > 100:
                temperature = max(0.01, temperature * 0.9)
                stagnation_count = 0

    # Convex refinement every 100 iterations
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample k values for constraints
        k_vals = np.random.choice(range(1, n_points), size=min(20, n_points//3), replace=False)
        for k in k_vals:
            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            h_feasible = np.clip(h_feasible, 0, 1)
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    # Ensure feasible sum and bounds
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
