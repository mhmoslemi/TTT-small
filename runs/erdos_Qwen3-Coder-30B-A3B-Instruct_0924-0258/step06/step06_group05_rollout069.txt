import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive step size
    max_iter = 1000
    patience = 0
    patience_limit = 100
    step_size = 0.05
    decay_factor = 0.99
    
    for idx in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find the maximum overlap shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Compute gradient for the worst-case shift
        gradient = np.zeros(n_points)
        k = max_k - (n_points - 1)  # Convert to shift offset
        
        # Gradient computation based on correlation
        if k >= 0:
            gradient[k:] += -best_h[:-k]
            gradient[:-k] += best_h[k:]
        else:
            k_abs = -k
            gradient[k_abs:] += -best_h[:-k_abs]
            gradient[:-k_abs] += best_h[k_abs:]
            
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm
        
        # Line search
        best_step_size = step_size
        best_h_new = None
        best_c5_new = best_c5
        
        for _ in range(10):  # Max 10 line search steps
            h_new = best_h - best_step_size * gradient
            h_new = np.clip(h_new, 0, 1)
            # Renormalize to correct sum
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5_new:
                best_c5_new = c5_new
                best_h_new = h_new.copy()
                best_step_size *= 1.2  # Increase step if good
            else:
                best_step_size *= 0.5  # Decrease step if bad
                if best_step_size < 1e-6:
                    break
        
        h_new = best_h_new if best_h_new is not None else best_h
        c5_new = best_c5_new
        
        # Adaptive step size update
        step_size *= decay_factor
        
        # Update global best
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience = 0
        else:
            patience += 1
            
        # Random perturbation
        if patience > 20 and np.random.random() < 0.3:
            noise = np.random.normal(0, step_size * 0.2, n_points)
            h_new = best_h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                patience = 0
                
        if patience > patience_limit:
            break
    
    # Convex optimization refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample key shift values
        k_vals = np.hstack([np.arange(1, n_points//2, max(1, n_points//20)), 
                           np.arange(n_points//2, n_points-1, max(1, n_points//20))])
        k_vals = np.unique(k_vals)
        
        for k in k_vals:
            if k < n_points and k > 0:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=300)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
