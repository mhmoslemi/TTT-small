import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization
    resolutions = [35, 40, 50, 60, 80]
    
    # Gradient-based optimization with restarts
    def optimize_at_resolution(h_init, res):
        h_current = h_init.copy()
        dx_current = 2.0 / res
        h_current = h_current * (res/2) / np.sum(h_current)
        
        # SLSQP with restarts
        max_restarts = 20
        restart_count = 0
        local_best = h_current.copy()
        local_c5 = evaluate_erdos_solution(h_current, dx_current)
        
        while restart_count < max_restarts and time() - start_time < budget_s * 0.95:
            # Perturb
            h_perturb = h_current + np.random.normal(0, 0.03, res)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (res/2) / np.sum(h_perturb)
            
            # Local optimization
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx_current)
                
                def constraint_sum(h):
                    return np.sum(h) - res / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(res)]
                res_opt = minimize(objective, h_perturb, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 300, 'ftol': 1e-9})
                if res_opt.success:
                    h_new = res_opt.x
                    c5_new = objective(h_new)
                    if c5_new < local_c5:
                        local_best = h_new
                        local_c5 = c5_new
                        restart_count = 0
                    else:
                        restart_count += 1
                else:
                    restart_count += 1
            except Exception:
                restart_count += 1
        
        return local_best, local_c5
    
    # Main optimization loop over resolutions
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
        
        if res > n_points:
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               best_h)
            h_current = h_interp
        else:
            h_current = best_h[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        # Optimize at this resolution
        new_h, new_c5 = optimize_at_resolution(h_current, res)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            n_points = len(best_h)
            dx = 2.0 / n_points
            
        # Occasionally inject structured approach
        if res % 3 == 0 and time() - start_time < budget_s * 0.90:
            h_structured = np.zeros(res)
            # Build alternating segments for low overlap
            mid = res // 2
            h_structured[:mid] = 1.0
            h_structured[mid:] = 0.0
            h_structured = h_structured * (res/2) / np.sum(h_structured)
            
            new_h_struct, c5_struct = optimize_at_resolution(h_structured, res)
            if c5_struct < best_c5:
                best_h = new_h_struct
                best_c5 = c5_struct
                n_points = len(best_h)
                dx = 2.0 / n_points

    # Final refinement at best resolution
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
