import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available, otherwise create Hamming window
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create Hamming window
        h_start = 0.54 + 0.46 * np.cos(np.pi * (np.arange(n_points) / n_points - 0.5) * 2)
        # Normalize to correct integral
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Projected gradient descent with adaptive step size
    max_iter = 1000
    eta = 0.01
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Find current maximum correlation lag
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag = np.argmax(corr)
        
        # Compute gradient (simplified: finite difference approximation)
        eps = 1e-6
        gradient = np.zeros(n_points)
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            if h_plus[j] > 1: h_plus[j] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = best_h.copy()
            h_minus[j] -= eps
            if h_minus[j] < 0: h_minus[j] = 0
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            
            gradient[j] = (corr_plus[max_lag] - corr_minus[max_lag]) / (2 * eps)
        
        # Update with gradient descent
        new_h = best_h - eta * gradient
        # Project to [0,1]
        new_h = np.clip(new_h, 0, 1)
        # Scale to maintain integral constraint
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate the new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
    
    # Final refinement using SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
