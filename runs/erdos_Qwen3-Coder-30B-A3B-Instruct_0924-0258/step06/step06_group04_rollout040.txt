import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for PSO
    n_particles = 50
    max_iter = 200
    inertia_w = 0.7
    c1 = 1.4
    c2 = 1.4
    vel_max = 0.2
    
    # Initialize particles
    particles = [h_start.copy() for _ in range(n_particles)]
    velocities = [np.random.uniform(-vel_max, vel_max, n_points) for _ in range(n_particles)]
    p_best = [h.copy() for h in particles]
    p_best_scores = [evaluate_erdos_solution(h, dx) for h in particles]
    g_best = best_h.copy()
    g_best_score = best_c5
    
    # Adaptive constraint sampling
    def get_critical_lags(h, num_lags=10):
        corr = np.correlate(h, 1 - h, mode='full')
        lags = np.argmax(np.abs(corr)) - np.arange(len(corr))
        critical_lags = np.argsort(-np.abs(corr))[:num_lags]
        return critical_lags
    
    # PSO iterations
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        critical_lags = get_critical_lags(g_best, num_lags=min(20, n_points // 2))
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia_w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -vel_max, vel_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            score = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best[i] = particles[i].copy()
                p_best_scores[i] = score
                
                # Update global best
                if score < g_best_score:
                    g_best = particles[i].copy()
                    g_best_score = score
                    if g_best_score < best_c5:
                        best_h = g_best.copy()
                        best_c5 = g_best_score
        
        # Periodic SQP refinement
        if iter_count % 10 == 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add constraints for sampled lags
                for k in critical_lags:
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                    constraints.append(corr_sum <= g_best_score * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
    
    # Final local refinement with scipy minimize
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=constraint, options={'maxiter': 100})
        if result.success:
            final_h = result.x
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
    
    # Ensure final constraints hold
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
