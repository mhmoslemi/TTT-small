import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    n_start = 10
    n_max = 35
    best_h = None
    best_c5 = float('inf')
    
    n_points = n_start
    while n_points <= n_max and (time() - start_time) < budget_s * 0.95:
        # Initialize h
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        else:
            x = np.linspace(0, 2, n_points)
            h = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n_points))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        dx = 2.0 / n_points
        
        # Gradient-based refinement
        for iter in range(100):
            if (time() - start_time) > budget_s * 0.95:
                break
            
            # Compute gradient numerically (simplified)
            grad = np.zeros(n_points)
            eps = 1e-4
            h_orig = h.copy()
            
            for i in range(n_points):
                h_pert = h_orig.copy()
                h_pert[i] += eps
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
                c5_plus = evaluate_erdos_solution(h_pert, dx)
                
                h_pert = h_orig.copy()
                h_pert[i] -= eps
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
                c5_minus = evaluate_erdos_solution(h_pert, dx)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with gradient
            alpha = 0.05
            h = h - alpha * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Simple Simulated Annealing step
            if np.random.random() < 0.1:
                temp = 0.05 * np.exp(-iter / 100)
                noise = np.random.normal(0, temp, n_points)
                h_new = h + noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                c5_old = evaluate_erdos_solution(h, dx)
                if c5_new < c5_old or np.random.random() < np.exp(-(c5_new - c5_old) / temp):
                    h = h_new
            
            # Check improvement
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        
        n_points *= 2
    
    # Final binary search for C5
    if best_h is not None:
        dx = 2.0 / len(best_h)
        target_c5 = 0.3807
        min_c5 = 0.3800
        
        # Binary search
        for _ in range(20):
            if (time() - start_time) > budget_s * 0.95:
                break
            test_c5 = (min_c5 + target_c5) / 2
            try:
                # Try CVX approach for tight bound
                h_opt = cp.Variable(len(best_h))
                constraints = [
                    cp.sum(h_opt) == len(best_h) / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Constrain each overlap to be at most test_c5
                for k in range(1, len(best_h)):
                    if k < len(best_h)//2:
                        corr = cp.sum(cp.multiply(h_opt[:len(best_h)-k], 1 - h_opt[k:]))
                    else:
                        corr = cp.sum(cp.multiply(h_opt[k-len(best_h)+1:], 1 - h_opt[:len(best_h)-(k-len(best_h)+1)]))
                    constraints.append(corr <= test_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=200)
                
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < test_c5:
                        target_c5 = test_c5
                        best_c5 = min(best_c5, c5_feasible)
                    else:
                        min_c5 = test_c5
                else:
                    min_c5 = test_c5
            except:
                min_c5 = test_c5
        
        # Ensure final solution satisfies constraints
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (len(best_h) / 2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    if best_h is None:
        # Fallback to simple default
        n_points = 35
        x = np.linspace(0, 2, n_points)
        best_h = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / 10))
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        dx = 2.0 / n_points
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, len(best_h))
