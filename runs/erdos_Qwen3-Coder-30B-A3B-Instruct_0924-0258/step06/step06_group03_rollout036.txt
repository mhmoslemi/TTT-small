import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with parent construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create triangular window as base
        h_start = 1 - np.abs(np.arange(n_points) / (n_points-1) - 0.5) * 2
        # Add noise for diversity
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        # Normalize
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # DE bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints for DE
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    # Objective function for DE
    def de_objective(x):
        # Correct the sum constraint
        x = x * (n_points/2) / np.sum(x)
        return evaluate_erdos_solution(x, dx)
    
    # Use a larger population for better exploration
    popsize = 100 if n_points <= 35 else 50
    
    # Run differential evolution
    for iteration in range(300):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Use differential evolution with constraint handling
        res = differential_evolution(
            de_objective,
            bounds,
            maxiter=20,
            popsize=popsize,
            seed=seed + iteration,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.9,
            disp=False
        )
        
        if res.success:
            x_opt = res.x
            x_opt = x_opt * (n_points/2) / np.sum(x_opt)
            c5_val = evaluate_erdos_solution(x_opt, dx)
            if c5_val < best_c5:
                best_h = x_opt.copy()
                best_c5 = c5_val
    
    # Local refinement with L-BFGS-B
    def lbfgs_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def lbfgs_constraint(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': lbfgs_constraint}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try several local optimizations from top candidates
    top_candidates = []
    for _ in range(10):
        trial = best_h + np.random.normal(0, 0.02, n_points)
        trial = np.clip(trial, 0, 1)
        trial = trial * (n_points/2) / np.sum(trial)
        top_candidates.append(trial)
    
    top_candidates.append(best_h)
    
    for h_trial in top_candidates:
        try:
            res = minimize(
                lbfgs_objective,
                h_trial,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 200, 'ftol': 1e-8}
            )
            if res.success:
                h_refined = res.x
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            continue
    
    # Final safety check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
