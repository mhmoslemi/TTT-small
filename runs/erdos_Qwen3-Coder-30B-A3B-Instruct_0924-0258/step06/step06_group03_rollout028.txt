import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create triangular initialization
        h_start = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h_start[i] = 1.0 - abs(i - mid) / (mid + 1)
        # Normalize
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Function to be minimized
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Constraint function for sum == n_points/2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Differential Evolution parameters
    maxiter = 300
    popsize = 50
    mutation = (0.5, 1)
    recombination = 0.7
    
    # Differential Evolution
    de_result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=maxiter,
        popsize=popsize,
        mutation=mutation,
        recombination=recombination,
        seed=seed,
        atol=1e-8,
        callback=lambda x, convergence: None if time() - start_time > budget_s * 0.8 else None
    )
    
    # Refine with local optimization if successful
    if de_result.success:
        h_de = de_result.x.copy()
        c5_de = objective(h_de)
        if c5_de < best_c5:
            best_h = h_de
            best_c5 = c5_de
    
    # Local refinement for top individuals from DE
    top_individuals = []
    if de_result.success:
        top_individuals.append(de_result.x)
    
    # Add a few random individuals with proper normalization
    for _ in range(5):
        h_rand = np.random.random(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        top_individuals.append(h_rand)
    
    # Refine using L-BFGS-B
    for h_init in top_individuals:
        try:
            # Use bounds and constraints
            res = minimize(
                objective,
                h_init,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 100, 'ftol': 1e-10}
            )
            if res.success:
                h_refined = res.x.copy()
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            continue

    return (list(best_h), best_c5, n_points)
