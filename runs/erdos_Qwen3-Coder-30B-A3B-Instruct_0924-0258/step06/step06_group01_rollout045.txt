import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.random.rand(n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # Simulated Annealing
    T0 = 1.0
    Tmin = 1e-5
    alpha = 0.999
    max_iter_per_T = 100
    
    h_current = h_start.copy()
    current_overlap = evaluate_erdos_solution(h_current, dx)
    best_h = h_current.copy()
    best_overlap = current_overlap
    
    T = T0
    iter_count = 0
    
    while T > Tmin and (time() - start_time) < budget_s * 0.95:
        for _ in range(max_iter_per_T):
            # Generate candidate solution
            h_candidate = h_current.copy()
            
            # Make small perturbations
            i, j = random.sample(range(n_points), 2)
            
            # Perturb two random indices
            delta = random.gauss(0, 0.05)
            h_candidate[i] += delta
            
            # Adjust another element to maintain sum constraint
            adjustment = delta * (n_points / 2) / np.sum(h_current) - delta
            h_candidate[j] -= adjustment
            
            # Ensure bounds and rescale
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Evaluate candidate
            candidate_overlap = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept or reject
            delta = candidate_overlap - current_overlap
            if delta < 0 or np.exp(-delta / T) > np.random.rand():
                h_current = h_candidate.copy()
                current_overlap = candidate_overlap
                
            # Update best solution
            if current_overlap < best_overlap:
                best_h = h_current.copy()
                best_overlap = current_overlap
            
            iter_count += 1
            
        T *= alpha
    
    # Final refinement with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            best_overlap = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_overlap, n_points)
