import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Grid refinement approach with adaptive sampling
    n_start = 10
    n_max = 35
    resolutions = [n_start * (2 ** i) for i in range(10) if n_start * (2 ** i) <= n_max]
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize or interpolate from previous solution
        if n_points == n_start:
            h = np.ones(n_points) * 0.5
        else:
            # Interpolate from previous solution
            h = np.interp(np.linspace(0, 2, n_points), 
                         np.linspace(0, 2, len(best_h)), 
                         best_h)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Hybrid optimization: gradient descent + simulated annealing
        step_size = 0.1
        temp = 0.1
        decay_rate = 0.995
        min_step = 0.001
        
        for iter in range(100):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Gradient descent step
            h_new = h.copy()
            noise = np.random.normal(0, temp, n_points)
            h_new += noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Simulated Annealing step
            if np.random.random() < 0.1:  # 10% chance
                perturbation = np.random.normal(0, temp, n_points)
                h_test = h + perturbation
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points/2) / np.sum(h_test)
                c5_test = evaluate_erdos_solution(h_test, dx)
                c5_curr = evaluate_erdos_solution(h, dx)
                
                if c5_test < c5_curr or np.random.random() < np.exp(-(c5_test - c5_curr) / temp):
                    h = h_test
                    best_c5 = min(best_c5, c5_test)
            
            # Adaptive step size
            step_size = max(min_step, step_size * decay_rate)
            temp = max(0.001, temp * decay_rate)
            
            # Update best solution
            c5_new = evaluate_erdos_solution(h, dx)
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
    
    # Final polish with optimization
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add key constraint from solution
        corr = cp.conv(h_opt, 1 - h_opt)
        corr_max = cp.max(corr)
        constraints.append(corr_max <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception as e:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
