import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.zeros(n_points)
        # Create a structured pattern inspired by symmetric constructions
        centers = np.array([0.5, 1.0, 1.5])
        for c in centers:
            idx = int(c * n_points / 2)
            h_start[max(0, idx-5):min(n_points, idx+5)] = 0.8
        h_start = h_start * (n_points / (2 * np.sum(h_start)))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Trust Region Optimization
    tr_radius = 0.5
    max_iterations = 1000
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Adaptive perturbation within trust region
        perturbation = np.random.uniform(-tr_radius, tr_radius, n_points)
        h_new = best_h + perturbation
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct integral
        sum_h = np.sum(h_new)
        if sum_h > 0:
            h_new = h_new * (n_points / (2 * sum_h))
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            tr_radius = min(0.8, tr_radius * 1.1)
        else:
            tr_radius = max(0.02, tr_radius * 0.9)
    
    # Convex Relaxation for final refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add overlap constraints with current best
        overlap_terms = []
        for k in range(1, n_points):
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            overlap_terms.append(corr_sum)
        
        # Relax the maximum constraint
        max_overlap_var = cp.Variable()
        constraints.append(max_overlap_var >= best_c5 * dx)
        
        # Add constraints for each k
        for i in range(len(overlap_terms)):
            if i < n_points - 1:
                constraints.append(overlap_terms[i] <= max_overlap_var + 1e-8)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            h_feasible = np.clip(h_feasible, 0, 1)
            sum_h = np.sum(h_feasible)
            if sum_h > 0:
                h_feasible = h_feasible * (n_points / (2 * sum_h))
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception as e:
        pass
    
    # Final refinement with trust region
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        sum_h = np.sum(h_final)
        if sum_h > 0:
            h_final = h_final * (n_points / (2 * sum_h))
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception as e:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * (n_points / (2 * sum_h))
    
    return (list(best_h), best_c5, n_points)
