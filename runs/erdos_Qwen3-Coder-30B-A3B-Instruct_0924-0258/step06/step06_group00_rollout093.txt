import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 200
    elite_size = population_size // 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start)
        # Generate diverse perturbations
        for _ in range(population_size - 1):
            h_pert = h_start + np.random.normal(0, 0.1, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            population.append(h_pert)
    else:
        # Random initialization
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points/2) / np.sum(h)
            population.append(h)
    
    # Genetic Algorithm main loop
    no_improvement_count = 0
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= 50:
            break
        
        # Selection
        selected_indices = sorted_indices[:population_size//5]
        selected = [population[i] for i in selected_indices]
        
        # Elitism
        elite = population[:elite_size]
        
        # Create new population
        new_population = elite[:]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Select parents
            parent1 = random.choice(selected)
            parent2 = random.choice(selected)
            
            # Crossover
            if random.random() < crossover_rate:
                crossover_point = random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                    child2[i] += np.random.normal(0, 0.05)
            
            # Clip and normalize
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
    
    # Local refinement with SLSQP on top candidates
    top_candidates = population[:10]
    for i, h_refined in enumerate(top_candidates):
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_final = res.x
                c5_refined = objective(h_final)
                if c5_refined < best_c5:
                    best_h = h_final
                    best_c5 = c5_refined
        except Exception:
            continue
    
    return (list(best_h), best_c5, n_points)
