import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a low-discrepancy sequence for better initialization
        def halton_sequence(n, base=2):
            sequence = []
            for i in range(n):
                x = 0
                f = 1.0 / base
                while i > 0:
                    x += (i % base) * f
                    i //= base
                    f /= base
                sequence.append(x)
            return np.array(sequence)
        
        h_start = halton_sequence(n_points)
        h_start = (h_start - np.min(h_start)) / (np.max(h_start) - np.min(h_start))
        h_start = h_start * 0.5  # Scale to average 0.5
        h_start = h_start * (n_points/2) / np.sum(h_start)  # Normalize to correct integral
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive gradient descent with momentum
    max_iterations = 10000
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute gradient
        gradient = np.zeros(n_points)
        for k in range(1, n_points):
            if k < n_points:
                # Forward terms
                gradient[:n_points-k] += (1 - best_h[k:])
                # Backward terms (for k > 0)
                if k > 0:
                    gradient[k:] -= best_h[:-k]
        
        # Update velocity and parameters
        velocity = momentum * velocity - learning_rate * gradient
        best_h = best_h + velocity
        best_h = np.clip(best_h, 0, 1)
        
        # Project to correct integral
        sum_h = np.sum(best_h)
        if sum_h > 0:
            best_h = best_h * (n_points/2) / sum_h
        
        # Evaluate
        c5_new = evaluate_erdos_solution(best_h, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = best_h.copy()
        
        # Adaptive learning rate
        learning_rate = max(0.0001, learning_rate * 0.9995)
    
    # Final refinement with quadratic optimization
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add some overlap constraints to help convergence
        step = max(1, n_points // 10)
        for k in range(1, n_points, step):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final feasibility check and refinement
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * (n_points/2) / sum_h
    
    c5_final = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), c5_final, n_points)
