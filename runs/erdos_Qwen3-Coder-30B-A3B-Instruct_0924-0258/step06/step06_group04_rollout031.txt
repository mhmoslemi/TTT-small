import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        # Create a sinusoidal-like starting point
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(2 * np.pi * x / (2 / n_points))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Simple gradient descent approach
    max_iter = 1000
    step_size = 0.01
    decay_rate = 0.998
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.7:
            break
            
        # Compute gradient approximations
        h_eval = best_h.copy()
        h_eval = np.clip(h_eval, 0, 1)
        h_eval = h_eval * (n_points/2) / np.sum(h_eval)
        
        # Perturb and evaluate
        noise = np.random.normal(0, step_size, n_points)
        h_new = h_eval + noise
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            
        step_size = max(0.0001, step_size * decay_rate)
    
    # Phase 2: Constraint Sampling with Integer Programming
    try:
        # Define the constraint sampling approach
        n_samples = min(20, n_points // 2)

        h_opt = cp.Variable(n_points)
        
        # Create constraints
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample key lags for constraints
        for k in range(1, n_points//2 + 1):
            # Sample some lags
            if np.random.random() < 0.3:  # 30% chance to sample

                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))

                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        # Solve
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)

                h_feasible = h_opt.value

            pass

    # Final normalization and evaluation

    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
