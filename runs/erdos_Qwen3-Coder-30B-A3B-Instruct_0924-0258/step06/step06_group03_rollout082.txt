import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n_points):
    """Create a triangular window centered at n_points/2"""
    half = n_points // 2
    window = np.array([i + 1 for i in range(half)] + [n_points - i - 1 for i in range(n_points - half)])
    return window / np.sum(window) * (n_points / 2)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = triangular_window(n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    bounds = [(0, 1) for _ in range(n_points)]
    maxiter = 300
    popsize = 50
    mutation = 0.7
    crossover = 0.9
    
    # Define constraint function for sum = n_points/2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    # Use differential_evolution with constraints
    def constrained_objective(x):
        # Normalize to satisfy sum constraint
        x_normalized = x * (n_points / 2) / np.sum(x)
        return evaluate_erdos_solution(x_normalized, dx)
    
    # Perform DE optimization
    de_result = differential_evolution(
        constrained_objective, 
        bounds, 
        maxiter=maxiter, 
        popsize=popsize, 
        mutation=mutation, 
        crossover=crossover,
        seed=seed,
        disp=False,
        tol=1e-8
    )
    
    if de_result.success:
        # Normalize the DE result to meet sum constraint
        h_de = de_result.x * (n_points / 2) / np.sum(de_result.x)
        c5_de = evaluate_erdos_solution(h_de, dx)
        if c5_de < best_c5:
            best_h = h_de.copy()
            best_c5 = c5_de
            
    # Local refinement with L-BFGS-B
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Simple gradient-based refinement
    for _ in range(5):  # Perform refinement on top 10 solutions
        if time() - start_time > budget_s * 0.9:
            break
            
        # Try several local optimizations with different starting points
        for _ in range(10):
            # Add small random perturbation
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            
            try:
                # L-BFGS-B optimization
                res = minimize(
                    objective, 
                    h_perturbed, 
                    method='L-BFGS-B',
                    bounds=[(0, 1) for _ in range(n_points)],
                    constraints={'type': 'eq', 'fun': constraint_sum},
                    options={'maxiter': 100}
                )
                if res.success:
                    c5_refined = objective(res.x)
                    if c5_refined < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_refined
            except:
                continue
    
    # Final check
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
