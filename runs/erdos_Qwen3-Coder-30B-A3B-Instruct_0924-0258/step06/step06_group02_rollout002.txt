import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    temperature = 0.5
    cooling_rate = 0.99
    max_steps = 2000
    
    # Track best solution
    current_h = best_h.copy()
    current_c5 = best_c5
    
    for step in range(max_steps):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Perturb in top regions of highest overlap
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        # Find indices where overlap is among the top 10% of values
        top_indices = np.argsort(corr)[-max(1, n_points//10):]  # top 10% indices
        h_new = current_h.copy()
        
        # Randomly choose some of top indices to modify
        num_modify = max(1, n_points//20)  # modify about 5% of points
        indices_to_modify = np.random.choice(top_indices, size=num_modify, replace=False)
        
        # Modify selected points
        for i in indices_to_modify:
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
            h_new[i] = np.clip(h_new[i], 0, 1)
        
        # Project to correct integral
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        delta_c5 = c5_new - current_c5
        
        # Accept or reject based on simulated annealing criterion
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temperature):
            current_h = h_new
            current_c5 = c5_new
            
        # Update global best if needed
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Cool down
        temperature *= cooling_rate
    
    # Gradient-based refinement using CVXPY
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add overlap constraints for some key indices
        indices_for_constraints = np.linspace(1, n_points-1, min(20, n_points-1), dtype=int)
        for k in indices_for_constraints:
            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
            
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except:
        pass
    
    # Final polishing with local gradient steps
    try:
        h_final = best_h.copy()
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            eps = 1e-3
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h_final.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                
                h_minus = h_final.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points/2) / np.sum(h_minus)
                c5_minus = evaluate_erdos_solution(h_minus, dx)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with gradient
            h_final = h_final - 1e-3 * grad
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            
            # Update best if better
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
                
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
