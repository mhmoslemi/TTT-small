import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 500
    omega = 0.7
    phi1 = phi2 = 1.5
    v_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Seed particles around initial solution
    particles.append(h_start.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    for i in range(1, n_particles):
        # Add noise to the initial solution
        h_perturbed = h_start + np.random.normal(0, 0.1, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        particles.append(h_perturbed)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests
    personal_bests = [p.copy() for p in particles]
    personal_best_scores = [evaluate_erdos_solution(p, dx) for p in particles]
    
    # Global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO iterations
    stagnation_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            v_new = (omega * velocities[i] 
                     + phi1 * r1 * (personal_bests[i] - particles[i])
                     + phi2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            v_new = np.clip(v_new, -v_max, v_max)
            
            # Update position
            pos_new = particles[i] + v_new
            
            # Enforce bounds
            pos_new = np.clip(pos_new, 0, 1)
            
            # Normalize to sum constraint
            pos_new = pos_new * (n_points/2) / np.sum(pos_new)
            
            # Evaluate new position
            score = evaluate_erdos_solution(pos_new, dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_bests[i] = pos_new.copy()
                personal_best_scores[i] = score
                
                # Update global best if needed
                if score < global_best_score:
                    global_best = pos_new.copy()
                    global_best_score = score
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
                
            particles[i] = pos_new.copy()
            velocities[i] = v_new.copy()
            
        # Early termination if no improvement
        if stagnation_count > 100:
            break
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, global_best, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < global_best_score:
                global_best = h_refined
                global_best_score = c5_refined
    except Exception:
        pass
    
    return (list(global_best), global_best_score, n_points)
