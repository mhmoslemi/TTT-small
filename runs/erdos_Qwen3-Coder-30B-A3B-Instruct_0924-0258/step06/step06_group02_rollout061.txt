import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        # Construct a symmetric pattern inspired by root systems
        h_start = np.zeros(n_points)
        centers = np.array([0.5, 1.0, 1.5])  # Balanced centers
        spread = 0.3
        x_vals = np.linspace(0, 2, n_points)
        for c in centers:
            h_start += np.exp(-0.5 * ((x_vals - c) / spread) ** 2)
        h_start = h_start * (n_points / (2 * np.sum(h_start)))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Kolmogorov-Arnold Optimization with adaptive trust regions
    trust_radius = 0.3
    max_iterations = 3000
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Adapt trust region based on performance
        if i % 100 == 0 and i > 0:
            trust_radius = max(0.01, trust_radius * 0.95)
        
        # Generate Gaussian perturbation within trust region
        noise = np.random.normal(0, trust_radius, n_points)
        h_new = best_h + noise
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct integral
        sum_h = np.sum(h_new)
        if sum_h > 0:
            h_new = h_new * (n_points / (2 * sum_h))
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
    
    # Final quadratic constraint refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add most critical overlap constraints
        indices = np.linspace(1, n_points//2, min(10, n_points//4), dtype=int)
        for k in indices:
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final validation step
    try:
        sum_h = np.sum(best_h)
        if sum_h > 0:
            best_h = best_h * (n_points / (2 * sum_h))
        final_c5 = evaluate_erdos_solution(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
