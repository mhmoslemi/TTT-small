import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimized evolutionary search with local refinement
    max_iter = 1000
    population_size = 20
    elite_size = 4
    mutation_rate = 0.1
    crossover_rate = 0.7
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Add initial solution to population
    population[0] = best_h
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new population
        new_population = population[:elite_size]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Selection (tournament)
            idx1 = np.random.randint(0, elite_size)
            idx2 = np.random.randint(0, elite_size)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Crossover
            if np.random.random() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                if np.random.random() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.05)
            
            # Ensure constraints
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
    
    # Final local optimization using scipy
    try:
        # Create a function to be minimized
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        # Constraints for optimization
        def constraint_sum(h):
            return np.sum(h) - n_points/2
        
        def constraint_lower(h):
            return h - 0
        
        def constraint_upper(h):
            return 1 - h
        
        # Bounds for each variable
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                       constraints=[{'type': 'eq', 'fun': constraint_sum},
                                   {'type': 'ineq', 'fun': constraint_lower},
                                   {'type': 'ineq', 'fun': constraint_upper}],
                       options={'maxiter': 100, 'ftol': 1e-6})
        
        if res.success:
            h_opt = res.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points/2) / np.sum(h_opt)
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    
    except Exception:
        pass
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
