import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 500
    omega = 0.7
    phi1 = phi2 = 1.5
    v_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial point as one particle
    h_normalized = h_start * (n_points/2) / np.sum(h_start)
    particles.append(h_normalized.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Generate other particles
    for _ in range(n_particles - 1):
        h_pert = h_normalized + np.random.normal(0, 0.1, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        particles.append(h_pert.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO iteration
    pbest = [particles[i].copy() for i in range(n_particles)]
    pbest_values = [evaluate_erdos_solution(particles[i], dx) for i in range(n_particles)]
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    iter_no_improve = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            v_new = (omega * velocities[i] + 
                    phi1 * r1 * (pbest[i] - particles[i]) + 
                    phi2 * r2 * (gbest - particles[i]))
            
            # Clamp velocity
            v_new = np.clip(v_new, -v_max, v_max)
            
            # Update position
            pos_new = particles[i] + v_new
            
            # Apply constraints
            pos_new = np.clip(pos_new, 0, 1)
            pos_new = pos_new * (n_points/2) / np.sum(pos_new)
            
            particles[i] = pos_new
            velocities[i] = v_new
            
            # Evaluate new position
            new_value = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if new_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = new_value
                
                # Update global best
                if new_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = new_value
                    iter_no_improve = 0
                else:
                    iter_no_improve += 1
            
        # Check for early stopping
        if iter_no_improve > 100:
            break
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, gbest, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
