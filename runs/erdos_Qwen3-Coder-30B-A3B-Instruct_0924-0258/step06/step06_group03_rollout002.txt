import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Structured initialization: triangular window
        h_start = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h_start[i] = 1.0 - abs(i - mid) / (mid + 1)
        # Normalize to satisfy sum constraint
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution for global search
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Add constraint to normalize sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # DE parameters
    maxiter = 300
    popsize = 50
    mutation = 0.7
    crossover = 0.9
    
    # DE with custom bounds handling
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=maxiter,
            popsize=popsize,
            mutation=mutation,
            crossover=crossover,
            seed=seed,
            disp=False,
            tol=1e-6
        )
        if de_result.success:
            h_de = de_result.x
            c5_de = objective(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    except Exception:
        pass
    
    # Refinement with L-BFGS-B
    if time() - start_time < budget_s * 0.9:
        # Define constraints for L-BFGS-B
        def constraint_sum_lbf(h):
            return np.sum(h) - n_points / 2.0
        
        # Design for L-BFGS-B
        constraints = [{'type': 'eq', 'fun': constraint_sum_lbf}]
        bounds_lbgfs = [(0, 1) for _ in range(n_points)]
        
        # Refinement for top individuals from DE
        top_individuals = [best_h]
        
        for h_ref in top_individuals:
            try:
                # L-BFGS-B optimization
                ref_result = minimize(
                    objective,
                    h_ref,
                    method='L-BFGS-B',
                    bounds=bounds_lbgfs,
                    constraints=constraints,
                    options={'maxiter': 200, 'ftol': 1e-9}
                )
                
                if ref_result.success:
                    h_refined = ref_result.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                continue
                
        # Final fine-tune on best solution
        try:
            ref_result = minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds_lbgfs,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-10}
            )
            if ref_result.success:
                h_final = ref_result.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass

    return (list(best_h), best_c5, n_points)
