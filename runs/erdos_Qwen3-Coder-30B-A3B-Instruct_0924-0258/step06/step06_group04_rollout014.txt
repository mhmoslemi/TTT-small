def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

            h_start = np.clip(h_start, 0, 1)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        # Optimization loop
        dx = 2.0 / n_points
        h_current = h_start.copy()
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        # Adaptive gradient descent with simulated annealing
        max_iter = 100
        step_size = 0.1
        temp = 0.1
        decay_rate = 0.99
        
        for _ in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Gradient descent step
            grad = np.zeros(n_points)
            for i in range(n_points):
                corr = np.correlate(h_current, 1 - h_current, mode='full')
                grad[i] = corr[n_points-1+i] - corr[n_points-1-i] if i < n_points//2 else corr[n_points-1+i] - corr[n_points-1-(n_points-1-i)]
            
            h_new = h_current - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)

                    h_current = h_perturb
                    c5_current = c5_perturb
            
            temp *= decay_rate
            step_size *= decay_rate
        
        # Update best
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        # Next resolution
        n_points *= 2
