import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with parent construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create triangular window with random perturbation
        h_start = np.array([1 - abs(i / (n_points-1) - 0.5) * 2 for i in range(n_points)])
        h_start += np.random.normal(0, 0.01, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    popsize = 50
    maxiter = 300
    mutation = 0.7
    crossover = 0.9
    
    # Objective function for DE
    def de_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Constraint function for DE: sum(h) == n_points/2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE optimization loop
    for iteration in range(maxiter):
        if time() - start_time > budget_s * 0.8:
            break
        # Run DE with current bounds and constraints
        try:
            result = differential_evolution(
                de_objective, 
                bounds, 
                maxiter=maxiter, 
                popsize=popsize,
                mutation=mutation,
                crossover=crossover,
                seed=seed,
                disp=False,
                polish=False
            )
            if result.success and not np.isnan(result.fun):
                h_candidate = np.clip(result.x, 0, 1)
                h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except Exception:
            pass
    
    # Local refinement with L-BFGS-B
    try:
        def objective_h(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum_h(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum_h}
        bounds_h = [(0, 1) for _ in range(n_points)]
        
        # Sample top candidates for local refinement
        top_candidates = [best_h]
        for _ in range(10):
            # Random mutation of best
            h_test = best_h.copy() + np.random.normal(0, 0.005, n_points)
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points/2) / np.sum(h_test)
            top_candidates.append(h_test)
        
        # Refine top candidates
        for h_test in top_candidates:
            try:
                res = minimize(objective_h, h_test, method='L-BFGS-B',
                               bounds=bounds_h, constraints=cons_sum,
                               options={'maxiter': 100, 'ftol': 1e-10})
                if res.success and not np.isnan(res.fun):
                    c5_test = res.fun
                    if c5_test < best_c5:
                        best_h = res.x
                        best_c5 = c5_test
            except Exception:
                continue
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
