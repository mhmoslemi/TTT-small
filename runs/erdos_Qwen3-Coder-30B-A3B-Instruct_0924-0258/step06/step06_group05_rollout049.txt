import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create binary representation with roughly half ones
        h_start = np.zeros(n_points)
        ones_count = n_points // 2
        h_start[:ones_count] = 1
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search with bit swaps and convex refinement
    max_iter = 1000
    temperature = 1.0
    cooling_rate = 0.995
    sa_threshold = 0.2  # Accept 20% worse solutions initially
    
    # Track best solution
    best_h_binary = best_h.copy()
    best_c5_binary = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Generate neighbor by swapping two bits
        h_new = best_h_binary.copy()
        indices = np.where(h_new > 0.5)[0]
        if len(indices) == 0:
            indices = np.arange(n_points)
        zeros = np.where(h_new < 0.5)[0]
        if len(zeros) == 0:
            zeros = np.arange(n_points)
        
        if len(indices) > 0 and len(zeros) > 0:
            i = np.random.choice(indices)
            j = np.random.choice(zeros)
            h_new[i] = 0
            h_new[j] = 1
            
            # Ensure sum is correct
            current_sum = np.sum(h_new)
            if abs(current_sum - n_points/2) > 1e-6:
                # Fix by adjusting another element
                if current_sum > n_points/2:
                    # Need to reduce sum
                    k = np.random.choice(np.where(h_new > 0.5)[0])
                    h_new[k] = 0
                else:
                    # Need to increase sum  
                    k = np.random.choice(np.where(h_new < 0.5)[0])
                    h_new[k] = 1
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept if better, or with probability based on temperature
            if c5_new < best_c5_binary or (np.random.rand() < sa_threshold * np.exp(-abs(c5_new - best_c5_binary) / (temperature + 1e-8))):
                best_h_binary = h_new
                best_c5_binary = c5_new
            
            # Anneal temperature
            temperature *= cooling_rate
            if temperature < 1e-6:
                temperature = 1e-6
                
        # Periodic convex refinement
        if iteration % 50 == 0 and iteration > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample k values for constraints
                k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5_binary * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=200)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5_binary:
                        best_h_binary = h_feasible
                        best_c5_binary = c5_feasible
            except Exception:
                pass    
                
        # Update overall best
        if best_c5_binary < best_c5:
            best_h = best_h_binary
            best_c5 = best_c5_binary

    # Final local refinement
    if best_c5_binary < best_c5:
        best_h = best_h_binary
        best_c5 = best_c5_binary

    # Return with normalization if needed
    h_rescaled = best_h.copy()
    h_rescaled = h_rescaled * (n_points/2) / np.sum(h_rescaled) if abs(np.sum(h_rescaled) - n_points/2) > 1e-6 else h_rescaled
    final_c5 = evaluate_erdos_solution(h_rescaled, dx)
    
    return (list(h_rescaled), final_c5, n_points)
