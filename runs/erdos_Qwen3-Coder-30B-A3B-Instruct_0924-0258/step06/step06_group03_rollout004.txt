import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n):
    """Create a triangular window function"""
    center = (n - 1) / 2.0
    window = np.array([max(0, 1 - abs(i - center) / center) for i in range(n)])
    return window

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create structured initialization using triangular window
        h_start = triangular_window(n_points)
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # DE parameters
    popsize = 50
    maxiter = 300
    mutation_factor = 0.7
    crossover_rate = 0.9
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    def de_callback(x, convergence):
        # Update best solution
        if time() - start_time > budget_s * 0.8: 
            return True
        h = np.array(x)
        c5 = objective(h)
        nonlocal best_h, best_c5
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        return False
    
    # DE optimization
    de_result = differential_evolution(
        objective,
        bounds,
        args=(),
        maxiter=maxiter,
        popsize=popsize,
        mutation=(mutation_factor,),
        crossover=crossover_rate,
        seed=seed,
        callback=de_callback,
        disp=False,
        polish=False,
        workers=1
    )
    
    if time() - start_time > budget_s * 0.9:
        return (list(best_h), best_c5, n_points)
    
    # Refinement with L-BFGS-B (on top 10 individuals)
    if de_result.success:
        h_refined = de_result.x
        c5_refined = objective(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    # Local refinement on top individuals
    top_individuals = []
    for _ in range(10):
        # Perturb best solution
        h_pert = best_h + np.random.normal(0, 0.01, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        top_individuals.append(h_pert)
    
    # Add best solution if not already included
    if not any(np.allclose(h, best_h) for h in top_individuals):
        top_individuals.append(best_h.copy())
    
    best_refined = best_h.copy()
    best_refined_c5 = best_c5
    
    for h in top_individuals[:]:
        try:
            res = minimize(objective, h, method='L-BFGS-B', 
                          bounds=bounds, 
                          options={'maxiter': 300, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_refined_c5:
                    best_refined = h_refined
                    best_refined_c5 = c5_refined
        except Exception:
            continue
    
    return (list(best_refined), best_refined_c5, n_points)
