import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    random.seed(seed)
    start_time = time()

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    max_iterations = 10000
    lr = 0.05
    momentum = 0.9
    conv_radius = 1e-3
    stagnation_threshold = 100
    
    # Initialize gradient tracking
    velocity = np.zeros_like(h_start)
    
    # Main optimization loop
    iteration = 0
    stagnation_count = 0
    last_best = best_c5
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        h_current = best_h.copy()
        
        # Compute current overlaps to find max
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        if max_overlap < best_c5:
            best_h = h_current.copy()
            best_c5 = max_overlap
            stagnation_count = 0
            last_best = max_overlap
        else:
            stagnation_count += 1
            if stagnation_count > stagnation_threshold:
                # Perform random perturbation if stuck
                h_current = np.clip(h_current + np.random.normal(0, 0.01, n_points), 0, 1)
                h_current = h_current * (n_points/2) / np.sum(h_current)
                corr = np.correlate(h_current, 1 - h_current, mode='full')
                max_overlap = corr[max_idx] * dx
                if max_overlap < best_c5:
                    best_h = h_current.copy()
                    best_c5 = max_overlap
                    stagnation_count = 0

            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = corr_plus[max_idx] * dx
            
            h_minus = h_current.copy()
            h_minus[i] -= eps
            if h_minus[i] < 0:
                h_minus[i] = 0.0

            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            max_minus = corr_minus[max_idx] * dx
            
            grad[i] = (max_plus - max_minus) / (2 * eps)

    if time() - start_time < budget_s * 0.98:

                    best_c5 = c5_final
        except Exception:
            pass

    return (list(best_h), best_c5, n_points)
