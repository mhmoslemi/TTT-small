import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n):
    """Create a triangular window centered at n/2"""
    x = np.arange(n)
    center = (n - 1) / 2
    return 1 - np.abs(x - center) / (n / 2)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = triangular_window(n_points)
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # DE bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE settings
    popsize = 50
    maxiter = 300
    mutation = 0.7
    recombination = 0.9
    
    def de_objective(h):
        # Ensure normalization
        h_normalized = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h_normalized, dx)
    
    # Differential Evolution with bounds and constraints
    result = differential_evolution(
        de_objective,
        bounds,
        maxiter=maxiter,
        popsize=popsize,
        mutation=mutation,
        recombination=recombination,
        seed=seed,
        disp=False,
        tol=1e-6
    )
    
    # Get best DE solution
    h_de = result.x
    h_de = h_de * (n_points / 2) / np.sum(h_de)
    c5_de = evaluate_erdos_solution(h_de, dx)
    
    # Refinement with L-BFGS-B
    def lbfgs_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Try refining the DE solution
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lbfgs_objective,
            h_de,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons_sum,
            options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-6}
        )
        if res.success:
            h_refined = res.x
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < c5_de:
                h_final = h_refined
                c5_final = c5_refined
            else:
                h_final = h_de
                c5_final = c5_de
        else:
            h_final = h_de
            c5_final = c5_de
    except Exception:
        h_final = h_de
        c5_final = c5_de
    
    # Final check of best solution so far
    best_h = h_final
    best_c5 = c5_final
    
    # If we haven't used much time, try more exploration
    if time() - start_time < budget_s * 0.7:
        # Run a few more iterations of DE with a slightly different setup
        popsize = 20
        maxiter = 100
        mutation = 0.8
        recombination = 0.9
        
        result2 = differential_evolution(
            de_objective,
            bounds,
            maxiter=maxiter,
            popsize=popsize,
            mutation=mutation,
            recombination=recombination,
            seed=seed+1,
            disp=False,
            tol=1e-6
        )
        
        h_de2 = result2.x
        h_de2 = h_de2 * (n_points / 2) / np.sum(h_de2)
        c5_de2 = evaluate_erdos_solution(h_de2, dx)
        
        if c5_de2 < best_c5:
            best_h = h_de2
            best_c5 = c5_de2
            
        # Try some more optimization
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            res2 = minimize(
                lbfgs_objective,
                h_de2,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            if res2.success:
                h_refined2 = res2.x
                h_refined2 = h_refined2 * (n_points / 2) / np.sum(h_refined2)
                c5_refined2 = evaluate_erdos_solution(h_refined2, dx)
                if c5_refined2 < best_c5:
                    best_h = h_refined2
                    best_c5 = c5_refined2
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
