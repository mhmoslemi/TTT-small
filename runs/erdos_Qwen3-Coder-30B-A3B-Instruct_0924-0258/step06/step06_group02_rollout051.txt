import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T = 0.5
    T_min = 1e-4
    alpha = 0.99
    steps = 2000
    
    for i in range(steps):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Temperature cooling
        if T > T_min:
            T *= alpha
        
        # Select indices with high overlap
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_indices = np.argsort(corr)[-max(1, n_points//10):]
        
        # Make a perturbation
        h_new = best_h.copy()
        for idx in max_indices:
            if np.random.random() < 0.5:
                delta = np.random.uniform(-0.1, 0.1)
                h_new[idx] += delta
                h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Re-normalize to maintain sum = n/2
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Compute new overlap
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject the new solution
        if c5_new < best_c5 or np.random.random() < np.exp(-(c5_new - best_c5) / T):
            best_h = h_new.copy()
            best_c5 = c5_new
    
    # Gradient Descent refinement
    try:
        h_refined = best_h.copy()
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Compute gradient
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            grad = np.zeros_like(h_refined)
            
            # Approximate gradient of overlap integral
            for i in range(n_points):
                # Change h[i] and recompute overlap
                h_temp_up = h_refined.copy()
                h_temp_down = h_refined.copy()
                h_temp_up[i] += 1e-4
                h_temp_up = np.clip(h_temp_up, 0, 1)
                h_temp_up = h_temp_up * (n_points/2) / np.sum(h_temp_up)
                h_temp_down[i] -= 1e-4
                h_temp_down = np.clip(h_temp_down, 0, 1)
                h_temp_down = h_temp_down * (n_points/2) / np.sum(h_temp_down)
                
                c5_up = evaluate_erdos_solution(h_temp_up, dx)
                c5_down = evaluate_erdos_solution(h_temp_down, dx)
                grad[i] = (c5_up - c5_down) / (2e-4)
            
            # Update step
            step_size = 1e-3 * (0.99 ** _)
            h_refined = h_refined - step_size * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            
            # Evaluate
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            
            if c5_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (list(best_h), best_c5, n_points)
