import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def feasibility_projection(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points/2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Predefined resolution steps for exploration
    resolutions = [35, 40, 50, 60, 80]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to higher resolution if needed
        if res > n_points:
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = feasibility_projection(h_current, n_points)
        
        # Restart strategy: multiple perturbations with local optimization
        max_restarts = 50
        restart_count = 0
        
        while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
            # Inject structured pattern with some random perturbation
            h_perturbed = h_current.copy()
            
            # Add noise
            noise_std = 0.05
            noise = np.random.normal(0, noise_std, n_points)
            h_perturbed += noise
            
            # Local optimization with bounds
            h_perturbed = feasibility_projection(h_perturbed, n_points)
            
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(objective, h_perturbed, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 1000, 'ftol': 1e-10})
                
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        restart_count = 0
                else:
                    restart_count += 1
            except Exception:
                restart_count += 1
    
    # Final optimization with additional restarts
    for _ in range(10):
        if time() - start_time > budget_s * 0.99:
            break
        h_restart = best_h + np.random.normal(0, 0.01, n_points)
        h_restart = feasibility_projection(h_restart, n_points)
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_restart, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 1000, 'ftol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
