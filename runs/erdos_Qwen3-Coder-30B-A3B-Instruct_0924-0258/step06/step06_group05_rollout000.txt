import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based evolutionary search
    pop_size = 50
    n_generations = 1000
    mutation_rate = 0.05
    patience = 100
    
    population = []
    fitness = []
    
    # Initialize population with diverse candidates
    for _ in range(pop_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Main evolutionary loop
    no_improvement = 0
    for gen in range(n_generations):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Sort by fitness
        idxs = np.argsort(fitness)
        elite = [population[i] for i in idxs[:pop_size//5]]
        elite_fitness = [fitness[i] for i in idxs[:pop_size//5]]
        
        # Update best
        if fitness[idxs[0]] < best_c5:
            best_c5 = fitness[idxs[0]]
            best_h = population[idxs[0]].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Create offspring
        offspring = []
        for i in range(pop_size - len(elite)):
            parent = elite[i % len(elite)]
            noise = np.random.normal(0, mutation_rate, n_points)
            child = parent + noise
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            offspring.append(child)
        
        # Replace population
        population = elite + offspring
        fitness = elite_fitness + [evaluate_erdos_solution(h, dx) for h in offspring]
        
        # Adaptive mutation
        if no_improvement > patience:
            mutation_rate = max(0.01, mutation_rate * 0.95)
            no_improvement = 0
            
        # Convex relaxation every 100 generations
        if gen % 100 == 0 and gen > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
            
                # Use a subset of k values for efficiency
                k_vals = np.linspace(0.1, 1.9, min(30, n_points//2))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=200)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
                
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
