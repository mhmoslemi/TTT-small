import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Homotopy method with convex relaxation
    max_homotopy_steps = 100
    t_vals = np.linspace(0, 1, max_homotopy_steps)
    
    h_current = best_h.copy()
    current_c5 = best_c5
    
    for t in t_vals:
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            # Convex relaxation problem
            h_opt = cp.Variable(n_points)
            # Objective: weighted combination
            obj = t * cp.sum(cp.multiply(h_opt, 1 - h_opt)) + (1 - t) * cp.sum_squares(h_opt - 0.5 * np.ones(n_points))
            
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            prob = cp.Problem(cp.Minimize(obj), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_opt.value
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                if c5_new < current_c5:
                    h_current = h_new.copy()
                    current_c5 = c5_new
                    
                    if c5_new < best_c5:
                        best_h = h_current.copy()
                        best_c5 = c5_new
                        
        except Exception:
            pass
            
        # Local refinement with simplex method
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_current, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-8})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < current_c5:
                    h_current = h_refined
                    current_c5 = c5_refined
                    
                    if c5_refined < best_c5:
                        best_h = h_current.copy()
                        best_c5 = c5_refined
                        
        except Exception:
            pass
            
        # Randomized perturbation with adaptive step size
        if time() - start_time > budget_s * 0.9:
            break
            
        step_size = max(0.001, 0.1 * (1 - t))
        noise = np.random.normal(0, step_size, n_points)
        h_new = h_current + noise
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < current_c5:
            h_current = h_new.copy()
            current_c5 = c5_new
            
            if c5_new < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_new

    return (list(best_h), best_c5, n_points)
