import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO Parameters
    n_particles = 50
    max_iter = 200
    inertia = 0.7
    c1, c2 = 1.4, 1.4
    velocity_max = 0.2
    
    # Initialize PSO
    particles = [h_start.copy() for _ in range(n_particles)]
    velocities = [np.random.uniform(-velocity_max, velocity_max, n_points) for _ in range(n_particles)]
    pbest = [p.copy() for p in particles]
    pbest_vals = [evaluate_erdos_solution(p, dx) for p in particles]
    gbest = best_h.copy()
    gbest_val = best_c5
    
    # PSO main loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample lags for constraint refinement
        if iter_num % 50 == 0:
            corr = np.correlate(gbest, 1 - gbest, mode='full')
            lag_indices = np.argsort(-corr)[:max(10, n_points // 10)]
            lag_indices = lag_indices[lag_indices > 0]  # Exclude zero lag
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (
                inertia * velocities[i] +
                c1 * r1 * (pbest[i] - particles[i]) +
                c2 * r2 * (gbest - particles[i])
            )
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            c5_val = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if c5_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = c5_val
            
            # Update global best
            if c5_val < gbest_val:
                gbest = particles[i].copy()
                gbest_val = c5_val
                
                if gbest_val < best_c5:
                    best_h = gbest.copy()
                    best_c5 = gbest_val
                    
    # SQP refinement with adaptive constraints
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Use dynamic lag constraints
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        lag_indices = np.argsort(-corr)[:max(10, n_points // 10)]
        lag_indices = lag_indices[lag_indices > 0]  # Exclude zero lag
        
        for k in lag_indices:
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
