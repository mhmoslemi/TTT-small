import numpy as np
from scipy.optimize import differential_evolution
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        n_points = 35
        h_start = np.ones(n_points) * 0.5

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Normalize initial
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Scale the objective function to make optimization easier
    def objective(h):
        if time() - start_time > budget_s * 0.95:
            return float('inf')
        h_norm = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h_norm, dx)
    
    # Differential Evolution
    try:
        result = differential_evolution(
            objective,
            bounds,
            maxiter=500,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            disp=False,
            tol=1e-6
        )
        if not np.isinf(result.fun):
            h_final = result.x
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Local refinement with Nelder-Mead if time permits
    try:
        if time() - start_time < budget_s * 0.9:
            # Normalize and bound for local solver
            x0 = np.clip(best_h, 0, 1)

                h_final = h_final * (n_points/2) / np.sum(h_final)

                if c5_final < best_c5:

    except Exception:

    best_h = np.clip(best_h, 0, 1)

    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
