import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search approach with bit flips
    # Convert to binary first
    h_bin = (h_start > 0.5).astype(int)
    target_sum = n_points // 2
    
    # Make sure we have the correct number of 1s
    actual_sum = np.sum(h_bin)
    diff = target_sum - actual_sum
    if diff > 0:
        # Add 1s
        zero_indices = np.where(h_bin == 0)[0]
        for i in range(diff):
            if i < len(zero_indices):
                h_bin[zero_indices[i]] = 1
    elif diff < 0:
        # Remove 1s
        one_indices = np.where(h_bin == 1)[0]
        for i in range(-diff):
            if i < len(one_indices):
                h_bin[one_indices[i]] = 0

    # Binary search loop
    max_iter = 10000
    sa_temp = 1.0
    sa_decay = 0.9995
    accept_prob = 0.2
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Make a random bit flip
        zero_indices = np.where(h_bin == 0)[0]
        one_indices = np.where(h_bin == 1)[0]
        
        if len(zero_indices) == 0 or len(one_indices) == 0:
            continue
        
        i = np.random.choice(zero_indices)
        j = np.random.choice(one_indices)
        
        # Flip the bits
        h_new_bin = h_bin.copy()
        h_new_bin[i] = 1
        h_new_bin[j] = 0
        
        # Recompute h from binary
        h_new = h_new_bin.astype(float)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept with simulated annealing
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            h_bin = h_new_bin.copy()
        else:
            # Accept with probability
            if np.random.rand() < accept_prob:
                h_bin = h_new_bin.copy()
                h_new = h_new * (n_points/2) / np.sum(h_new)
                h_bin = (h_new > 0.5).astype(int)
        
        # Periodic convex refinement
        if iteration % 100 == 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample k values for constraints
                k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    # Rescale to satisfy sum
                    h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible.copy()
                        best_c5 = c5_feasible
            except Exception:
                pass
        
        sa_temp *= sa_decay
        accept_prob = max(0.01, accept_prob * sa_decay)
        
    return (list(best_h), best_c5, n_points)
