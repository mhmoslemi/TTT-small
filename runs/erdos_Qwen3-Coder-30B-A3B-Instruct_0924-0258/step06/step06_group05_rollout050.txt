import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    population_size = 50
    generations = 1000
    mutation_rate = 0.05
    elite_size = int(0.2 * population_size)
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Track best evolutionary solution
    best_evolutionary = best_h.copy()
    best_evolutionary_c5 = best_c5
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_evolutionary_c5:
            best_evolutionary = population[0].copy()
            best_evolutionary_c5 = fitness[0]
            
        # Keep elite
        new_population = population[:elite_size]
        
        # Generate offspring
        for _ in range(population_size - elite_size):
            # Select parent
            parent_idx = np.random.randint(0, elite_size)
            parent = population[parent_idx]
            
            # Mutate
            noise = np.random.normal(0, mutation_rate, n_points)
            child = parent + noise
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            # Occasionally apply convex relaxation
            if np.random.random() < 0.1:
                try:
                    h_opt = cp.Variable(n_points)
                    constraints = [
                        cp.sum(h_opt) == n_points / 2.0,
                        h_opt >= 0,
                        h_opt <= 1
                    ]
                    
                    # Add constraints on correlation sums
                    k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
                    for k_val in k_vals:
                        k = int(k_val * n_points / 2)
                        if k < n_points and k > 0:
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                            constraints.append(corr_sum <= best_evolutionary_c5 * dx + 1e-6)
                    
                    prob = cp.Problem(cp.Minimize(0), constraints)
                    prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                    if prob.status == cp.OPTIMAL:
                        child = np.clip(h_opt.value, 0, 1)
                        child = child * (n_points/2) / np.sum(child)
                except Exception:
                    pass
            
            new_population.append(child)
        
        population = new_population
        
        # Adapt mutation rate
        if gen % 100 == 0 and gen > 0:
            if best_evolutionary_c5 >= best_c5:
                mutation_rate *= 0.95
            mutation_rate = max(mutation_rate, 0.01)
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_evolutionary, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_evolutionary_c5:
                best_evolutionary = h_final
                best_evolutionary_c5 = c5_final
    except Exception:
        pass

    # Return best solution found
    return (list(best_evolutionary), best_evolutionary_c5, n_points)
