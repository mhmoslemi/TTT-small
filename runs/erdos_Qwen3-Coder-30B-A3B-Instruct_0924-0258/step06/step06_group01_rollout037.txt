import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    random.seed(seed)
    start_time = time()

    if 'initial_h_values' in globals():

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with convex relaxation
    max_iterations = 10000
    iteration = 0
    patience = 0
    patience_limit = 100
    learning_rate = 0.1
    momentum = 0.9
    convex_radius = 0.01
    
    h_current = h_start.copy()
    velocity = np.zeros_like(h_current)
    
    # Main optimization loop
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Compute current overlap
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        # Update best if improved
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1
        
        if patience >= patience_limit:
            break
            
        # Compute gradient estimation using finite differences
        epsilon = 1e-4
        grad = np.zeros_like(h_current)
        h_plus = h_current.copy()
        h_minus = h_current.copy()
        
        for i in range(n_points):
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            # Ensure bounds
            h_plus[i] = np.clip(h_plus[i], 0, 1)
            h_minus[i] = np.clip(h_minus[i], 0, 1)
            
            # Rescale to maintain sum constraint
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            # Compute directional derivative estimate
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Restore original values
            h_plus[i] = h_current[i]
            h_minus[i] = h_current[i]
        
        # Update velocity with momentum
        velocity = momentum * velocity - learning_rate * grad
        h_new = h_current + velocity
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_current = h_new
        
        iteration += 1
    
    # Final refinement with local optimization
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        if res.success:
            final_h = res.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points/2) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
