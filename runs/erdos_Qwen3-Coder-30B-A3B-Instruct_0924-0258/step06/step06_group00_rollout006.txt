import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 500
    omega = 0.7
    phi1 = phi2 = 1.5
    v_max = 0.2
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    # Start with initial solution
    particles.append(h_start.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    h_norm = h_start * (n_points/2) / np.sum(h_start)
    score = evaluate_erdos_solution(h_norm, dx)
    pbest.append(h_norm.copy())
    pbest_scores.append(score)
    
    # Generate other particles
    for i in range(1, n_particles):
        h_rand = h_start + np.random.normal(0, 0.1, n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        particles.append(h_rand.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        score = evaluate_erdos_solution(h_rand, dx)
        pbest.append(h_rand.copy())
        pbest_scores.append(score)
        
        if score < best_c5:
            best_h = h_rand.copy()
            best_c5 = score
    
    # Global best tracking
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO iterations
    stagnation_count = 0
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_update = (omega * velocities[i] + 
                         phi1 * r1 * (pbest[i] - particles[i]) + 
                         phi2 * r2 * (gbest - particles[i]))
            # Clip velocity
            vel_update = np.clip(vel_update, -v_max, v_max)
            velocities[i] = vel_update
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            score = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = particles[i].copy()
                
                # Update global best
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    if score < best_c5:
                        best_h = particles[i].copy()
                        best_c5 = score
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                    
        # Early termination if no improvement
        if stagnation_count > 100:
            break
    
    # Final refinement with SLSQP
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
