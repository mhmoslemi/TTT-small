import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO Parameters
    n_particles = 50
    inertia_w = 0.7
    c1 = 1.4
    c2 = 1.4
    max_iter = 200
    v_max = 0.2
    
    # Initialize PSO
    particles = [h_start.copy() for _ in range(n_particles)]
    velocities = [np.random.uniform(-v_max, v_max, n_points) for _ in range(n_particles)]
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [evaluate_erdos_solution(p, dx) for p in particles]
    
    # Global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # Adaptive sampling
    def get_critical_lags(h, num_lags=10):
        corr = np.correlate(h, 1 - h, mode='full')
        lags = np.arange(len(corr))
        # Focus on the middle lags, avoiding very short and very long ones
        mid_lags = lags[len(lags)//2 - 50 : len(lags)//2 + 50]
        # Select lags with highest correlations
        return mid_lags[np.argsort(corr[mid_lags])[-num_lags:]]
    
    # PSO iterations
    for iter_idx in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia_w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to sum constraint
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            score = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = score
            
            # Update global best
            if score < global_best_score:
                global_best = particles[i].copy()
                global_best_score = score
        
        if iter_idx % 20 == 0 and global_best_score < best_c5:
            best_h = global_best.copy()
            best_c5 = global_best_score
    
    # SQP Refinement with Adaptive Constraints
    try:
        h_var = cp.Variable(n_points)
        
        # Basic constraints
        constraints = [
            cp.sum(h_var) == n_points / 2.0,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Adaptive constraint sampling
        critical_lags = get_critical_lags(global_best)
        
        # Add constraints for critical lags
        for lag in critical_lags:
            if lag < n_points:
                # Actual convolutions (geometrically aware)
                if lag <= n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_var[:n_points-lag], 1 - h_var[lag:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_var[lag-n_points+1:], 1 - h_var[:n_points-(lag-n_points+1)]))
                constraints.append(corr_sum <= (best_c5 + 1e-6) * dx)
        
        # Solve SQP
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_var.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception as e:
        pass
    
    # Final refined evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
