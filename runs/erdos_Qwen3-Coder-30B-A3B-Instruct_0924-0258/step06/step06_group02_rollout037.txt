import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        # Create a structured initialization inspired by symmetry and sparse patterns
        h_start = np.zeros(n_points)
        centers = np.array([0.5, 1.0, 1.5])
        for center in centers:
            idx = int(center / 2.0 * n_points)
            h_start[max(0, idx-5):min(n_points, idx+5)] = 0.8
        h_start = h_start * (n_points / (2 * np.sum(h_start)))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Trust region optimization
    tr_radius = 0.2
    max_iterations = 2000
    lambda_merit = 0.1
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.85:
            break
        
        # Adaptive perturbation with trust region
        noise = np.random.uniform(-tr_radius, tr_radius, n_points)
        h_new = best_h + noise
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct integral
        sum_h = np.sum(h_new)
        if sum_h > 0:
            h_new = h_new * (n_points / (2 * sum_h))
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Constraint violation check
        constraint_violation = max(0, np.max(h_new - 1), np.min(h_new), abs(np.sum(h_new) - n_points/2))
        merit_new = c5_new + lambda_merit * constraint_violation
        merit_best = best_c5 + lambda_merit * max(0, np.max(best_h - 1), np.min(best_h), abs(np.sum(best_h) - n_points/2))
        
        # Accept or reject based on merit function
        if merit_new < merit_best:
            best_h = h_new.copy()
            best_c5 = c5_new
            tr_radius = min(0.5, tr_radius * 1.1)  # Expand trust region
        else:
            tr_radius = max(0.01, tr_radius * 0.9)  # Shrink trust region
        
        # Adjustable merit parameter
        if constraint_violation > 1e-4:
            lambda_merit *= 1.2
        else:
            lambda_merit *= 0.95
    
    # Convex relaxation refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Overlap constraints
        for k in range(1, n_points//2):
            if k < n_points//2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final projection to maintain constraints
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * (n_points / (2 * sum_h))
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
