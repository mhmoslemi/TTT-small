import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        # Create symmetric pattern inspired by root systems
        h_start = np.zeros(n_points)
        centers = np.array([0.5, 1.0, 1.5])
        spread = 0.3
        for c in centers:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - c))
            # Create Gaussian-like bumps
            x = np.linspace(0, 2, n_points)
            h_start += 0.8 * np.exp(-0.5 * ((x - c) / spread)**2)
        # Normalize to integral constraint
        h_start = h_start * (n_points / (2 * np.sum(h_start)))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a hybrid refinement method: SDP + Trust Region
    # SDP relaxation for lower bound
    try:
        # Relax to SDP to improve init
        h_sdp = cp.Variable(n_points)
        overlap_terms = []
        for k in range(1, n_points):
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_sdp[:n_points-k], 1 - h_sdp[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_sdp[k-n_points+1:], 1 - h_sdp[:n_points-(k-n_points+1)]))
            overlap_terms.append(corr_sum)
        
        max_overlap = cp.max(cp.vstack(overlap_terms)) * dx
        constraints = [
            cp.sum(h_sdp) == n_points / 2.0,
            h_sdp >= 0,
            h_sdp <= 1
        ]
        prob = cp.Problem(cp.Minimize(max_overlap), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_sdped = h_sdp.value
            c5_sdp = evaluate_erdos_solution(h_sdped, dx)
            if c5_sdp < best_c5:
                best_h = h_sdped
                best_c5 = c5_sdp
    except Exception:
        pass

    # Trust region optimization
    current_tr_radius = 0.1
    lambda_merit = 0.1
    max_iterations = 1000
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.80:
            break
            
        # Perturb within trust region
        noise = np.random.uniform(-current_tr_radius, current_tr_radius, n_points)
        h_new = best_h + noise
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct integral
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        constraint_violation = (
            max(0, np.max(h_new - 1), np.min(h_new), 
                abs(np.sum(h_new) - n_points/2))
        )
        merit_new = c5_new + lambda_merit * constraint_violation
        
        merit_best = best_c5 + lambda_merit * (
            max(0, np.max(best_h - 1), np.min(best_h), 
                abs(np.sum(best_h) - n_points/2))
        )
        
        if merit_new < merit_best:
            best_h = h_new.copy()
            best_c5 = c5_new
            current_tr_radius = min(current_tr_radius * 1.1, 0.5)
        else:
            current_tr_radius = max(current_tr_radius * 0.9, 0.001)
        
        if constraint_violation > 1e-4:
            lambda_merit *= 1.5
        else:
            lambda_merit *= 0.9
            
    # Final refinement
    try:
        h_final = np.clip(best_h, 0, 1)
        sum_h = np.sum(h_final)
        if sum_h > 0:
            h_final = h_final * (n_points/2) / sum_h
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
