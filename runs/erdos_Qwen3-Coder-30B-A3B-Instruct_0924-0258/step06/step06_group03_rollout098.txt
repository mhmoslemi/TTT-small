import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Initialize with a Hamming window
        n_points = 35
        h_start = 0.54 + 0.46 * np.cos(np.pi * (np.arange(n_points) / n_points - 0.5) * 2)
        # Normalize to sum to n_points / 2
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Projected gradient descent
    max_iter = 1000
    eta = 0.01
    
    def project_h(h):
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Scale to correct sum
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Get current maximum correlation index
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag = np.argmax(corr)
        # Compute gradient approximation w.r.t. h at the max lag
        # For simplicity, we just use finite differences
        h_plus = best_h.copy()
        h_minus = best_h.copy()
        epsilon = 1e-4
        h_plus[max_lag] += epsilon
        h_minus[max_lag] -= epsilon
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
        grad = (corr_plus[max_lag] - corr_minus[max_lag]) / (2 * epsilon)
        
        # Update h using projected gradient descent
        h_new = best_h - eta * grad
        h_new = project_h(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Ensure final solution is valid
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
