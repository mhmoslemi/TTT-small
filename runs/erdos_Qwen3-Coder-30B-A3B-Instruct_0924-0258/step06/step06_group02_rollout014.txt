import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.zeros(n_points)
        # Create structured initial solution with high concentration regions
        centers = [0.5, 1.0, 1.5]
        spread = 0.3
        for c in centers:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - c))
            h_start[max(0, idx-4):min(n_points, idx+4)] = 0.7
        h_start = h_start * (n_points / (2 * np.sum(h_start)))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Trust region optimization with adaptive step size
    max_iterations = 1000
    tr_radius = 0.3
    decay_factor = 0.98
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.80:
            break
            
        # Generate perturbation within trust region
        perturbation = np.random.uniform(-tr_radius, tr_radius, n_points)
        h_new = best_h + perturbation
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct integral
        sum_h = np.sum(h_new)
        if sum_h > 0:
            h_new = h_new * (n_points / (2 * sum_h))
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            tr_radius = min(0.5, tr_radius * 1.1)  # Expand trust region
        else:
            tr_radius *= decay_factor  # Contract trust region
    
    # Final convex relaxation refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add bounding constraints for overlap terms
        overlap_terms = []
        for k in range(1, n_points):
            if k <= n_points // 2:
                corr_term = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_term = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            overlap_terms.append(corr_term)
        
        max_overlap = cp.max(cp.vstack(overlap_terms))
        constraints.append(max_overlap <= best_c5 * dx + 1e-5)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final projection and validation
    try:
        best_h = np.clip(best_h, 0, 1)
        sum_h = np.sum(best_h)
        if sum_h > 0:
            best_h = best_h * (n_points / (2 * sum_h))
        final_c5 = evaluate_erdos_solution(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
