def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Determine n_points and initial h
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Iterative approach with convex relaxation
    max_iter = 500
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Relax and solve
        h_current = best_h.copy()
        
        # Create approximate convex quadratic problem
        # This is a simplified convex approximation for demonstration
        # In a real scenario, one would model a quadratic objective
        # Here, we iteratively refine using local search
        
        # Simple optimization: impose bounds and use SLSQP
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Local optimization step
        try:
            res = minimize(
                objective,
                h_current,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            if res.success:
                h_new = res.x
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                c5_val = evaluate_erdos_solution(h_new, dx)
                
                if c5_val < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_val
        except:
            pass
            
        # Occasionally, try some random perturbations to escape local minima
        if iteration % 30 == 0 and iteration > 0:
            h_perturb = h_current + np.random.normal(0, 0.01, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
            c5_val = evaluate_erdos_solution(h_perturb, dx)
            
            if c5_val < best_c5:
                best_h = h_perturb.copy()
                best_c5 = c5_val

    # Final refinement
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass

    return (list(best_h), best_c5, n_points)
