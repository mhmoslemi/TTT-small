import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        return h
    scale = n / (2 * sum_h)
    h *= scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Focused Perturbations
    T = 0.5
    cooling = 0.95
    min_T = 1e-4
    steps = 2000
    
    for i in range(steps):
        if time() - start_time > budget_s * 0.80:
            break
            
        # Identify top indices by overlap
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        # Get indices where overlap is higher than average
        avg_overlap = np.mean(corr_full)
        high_overlap_indices = np.where(corr_full > avg_overlap)[0]
        # Select 15% of highest indices
        selected = np.random.choice(high_overlap_indices, 
                                      size=max(1, len(high_overlap_indices)//7), 
                                      replace=False)
        
        # Perturb selected indices
        h_new = best_h.copy()
        for idx in selected:
            # Small random perturbation
            h_new[idx] += np.random.normal(0, 0.1)
        
        # Project to feasible
        h_new = project_feasible(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        delta = c5_new - best_c5
        
        # Accept with probability
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Cool down
        T = max(min_T, T * cooling)
    
    # QCP refinement with all shifts
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraints for all shifts k=1 to n-1
        for k in range(1, n_points):
            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final refinement with gradient-based optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    try:
        # Use scipy optimization
        x0 = best_h.copy()
        # Ensure constraints
        x0 = project_feasible(x0, n_points)
        
        # Optimize
        result = minimize(objective, x0, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                          options={'maxiter': 100, 'disp': False})
        
        if result.success:
            h_final = project_feasible(result.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
        
    # Ensure final feasibility
    best_h = project_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
