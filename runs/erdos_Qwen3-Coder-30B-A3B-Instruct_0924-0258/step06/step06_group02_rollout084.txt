import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        return h
    scale = n / (2 * sum_h)
    h *= scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n = len(h)
    else:
        n = 70
        h = np.ones(n) * 0.5
    
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing with focused perturbations
    temp = 0.5
    cooling_rate = 0.95
    max_steps = 2000
    
    for i in range(max_steps):
        if time() - start_time > budget_s * 0.80:
            break
            
        # Identify high-overlap regions
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_overlap_indices = np.argsort(corr)[-int(n*0.15):]
        
        h_new = best_h.copy()
        for idx in max_overlap_indices:
            if idx < len(h_new):
                h_new[idx] += np.random.normal(0, 0.1)
        
        # Apply projection
        h_new = project_feasible(h_new, n)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
            best_h = h_new
            best_c5 = c5_new
            
        temp *= cooling_rate
    
    # Solve QCP with all shifts included
    try:
        h_opt = cp.Variable(n)
        constraints = [
            cp.sum(h_opt) == n / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add all shift constraints
        for k in range(1, n):
            if k <= n // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k:], 1 - h_opt[:n-k]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final feasible projection
    try:
        h_final = project_feasible(best_h, n)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n)
