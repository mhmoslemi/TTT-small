import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a triangular window as starting point
        h_start = 1 - np.abs(np.arange(n_points) / (n_points-1) - 0.5) * 2
        # Add noise to break symmetry
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Function to be minimized
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Constraint function
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Try to do better with differential evolution
    try:
        # First try a few DE iterations
        de_result = differential_evolution(
            objective, 
            bounds, 
            constraints=cons_sum,
            seed=seed,
            maxiter=100,
            popsize=50,
            atol=1e-8,
            rtol=1e-8
        )
        if de_result.success:
            h_de = de_result.x
            c5_de = objective(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    except Exception:
        pass
    
    # Refine with local optimization
    try:
        # Use multiple starting points for local refinement
        best_local_h = best_h
        best_local_c5 = best_c5
        
        for i in range(5):
            # Use a slightly perturbed version as starting point
            h_perturbed = best_h + np.random.normal(0, 0.02, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            
            # L-BFGS-B for local refinement
            res = minimize(objective, h_perturbed, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons_sum,
                          options={'maxiter': 200, 'ftol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_local_c5:
                    best_local_h = h_refined
                    best_local_c5 = c5_refined
        
        if best_local_c5 < best_c5:
            best_h = best_local_h
            best_c5 = best_local_c5
            
    except Exception:
        pass
    
    # Final check for any improvement using constrained optimization
    try:
        # Provided values might be better, so just in case do one final pass
        res = minimize(objective, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons_sum,
                      options={'maxiter': 200, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Verify constraints and return
    final_sum = np.sum(best_h)
    if abs(final_sum - n_points/2.0) > 1e-6:
        best_h = best_h * (n_points/2.0) / final_sum
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
