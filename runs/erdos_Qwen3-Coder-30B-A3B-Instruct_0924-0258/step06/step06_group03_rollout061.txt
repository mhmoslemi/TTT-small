import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create triangular window with noise
        h_start = 1 - np.abs(np.arange(n_points) / (n_points-1) - 0.5) * 2
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # Helper to maintain constraints
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Objective function
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # DE bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # DE with custom parameters
    de_kwargs = {
        'bounds': bounds,
        'maxiter': 300,
        'popsize': 50,
        'mutation': (0.5, 1.0),
        'recombination': 0.9,
        'tol': 1e-6,
        'seed': seed,
        'workers': 1,
        'disp': False
    }
    
    # DE optimization
    try:
        de_result = differential_evolution(objective, **de_kwargs)
        if de_result.success:
            h_de = repair_h(de_result.x)
            c5_de = objective(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    except:
        pass
    
    # Refinement - Local search on top candidates
    candidates = []
    candidates.append(best_h)
    for _ in range(9):  # Get 10 candidates to refine
        h_rand = np.random.random(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        h_rand = np.clip(h_rand, 0, 1)
        candidates.append(h_rand)
    
    # Local optimization for top candidates
    for h in candidates:
        try:
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = repair_h(res.x)
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            continue
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
