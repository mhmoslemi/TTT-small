import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create a triangular window with some noise
        h_start = 1 - np.abs(np.arange(n_points) / (n_points-1) - 0.5) * 2
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # DE parameters
    popsize = 50
    maxiter = 300
    mutation_factor = 0.7
    crossover_rate = 0.9
    
    # bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def de_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def de_callback(x, convergence):
        nonlocal best_h, best_c5
        c5 = de_objective(x)
        if c5 < best_c5:
            best_h = x.copy()
            best_c5 = c5
    
    # DE optimization
    for i in range(maxiter):
        if time() - start_time > budget_s * 0.8:
            break
        try:
            result = differential_evolution(
                de_objective,
                bounds,
                strategy='best1bin',
                maxiter=1,
                popsize=popsize,
                mutation=(mutation_factor,),
                recombination=crossover_rate,
                seed=seed + i,
                callback=de_callback,
                disp=False
            )
            if result.success:
                c5 = evaluate_erdos_solution(result.x, dx)
                if c5 < best_c5:
                    best_h = result.x.copy()
                    best_c5 = c5
        except:
            pass

    # Local refinement with L-BFGS-B on top candidates
    top_candidates = []
    for _ in range(10):
        h_candidate = best_h.copy()
        h_candidate += np.random.normal(0, 0.01, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        top_candidates.append(h_candidate)
    
    # Add best solution to candidates
    top_candidates.append(best_h.copy())
    
    # Refine each candidate with L-BFGS-B
    for h_candidate in top_candidates:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_candidate, method='L-BFGS-B',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100})
            
            if res.success:
                c5_candidate = objective(res.x)
                if c5_candidate < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_candidate
        except:
            pass
                
    return (list(best_h), best_c5, n_points)
