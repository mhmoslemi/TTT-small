import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use Halton sequence for better distribution
        def halton_sequence(n, base=2):
            sequence = []
            for i in range(n):
                f = 1
                r = 0
                while i > 0:
                    f = f / base
                    r = r + (i % base) * f
                    i = i // base
                sequence.append(r)
            return np.array(sequence)
        
        h_start = halton_sequence(n_points)
        h_start = (h_start - np.min(h_start)) / (np.max(h_start) - np.min(h_start))
        h_start = h_start * 0.5
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive learning rate
    max_iterations = 1000
    learning_rate = 0.01
    decay = 0.999
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute gradient analytically for total overlap
        gradient = np.zeros(n_points)
        for k in range(1, n_points):
            # For each shift k, compute gradient contribution
            if k < n_points:
                # h[i] contributes to gradient through terms (1-h[i+k])
                gradient[:n_points-k] += (1 - best_h[k:])
                # and also through terms -h[i-k] where valid
                if k > 0:
                    gradient[k:] -= best_h[:n_points-k]
        
        # Update step
        new_h = best_h - learning_rate * gradient
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(new_h, dx)
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
            
        # Decay learning rate
        learning_rate *= decay
        
        # Check for early stopping based on improvement
        if i > 100 and abs(c5_new - best_c5) < 1e-8:
            break
    
    # Final refinement with quadratic programming
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraints: total overlap must be within current best
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_overlap = np.max(corr) * dx
        
        # Add a constraint on maximum overlap
        for k in range(1, n_points):
            if k < n_points/2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k:], 1 - h_opt[:n_points-k]))
            constraints.append(corr_sum <= max_overlap * 1.01)
            
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
