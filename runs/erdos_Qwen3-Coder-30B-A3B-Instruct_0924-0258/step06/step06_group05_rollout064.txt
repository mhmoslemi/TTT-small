import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    if 'initial_h_values' in globals():
        for _ in range(n_particles):
            pert = np.random.normal(0, 0.05, n_points)
            h = h_start + pert
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
            val = evaluate_erdos_solution(h, dx)
            pbest.append(h.copy())
            pbest_vals.append(val)
    else:
        for _ in range(n_particles):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points/2) / np.sum(h)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
            val = evaluate_erdos_solution(h, dx)
            pbest.append(h.copy())
            pbest_vals.append(val)
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO iterations
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Cooling schedule for simulated annealing
        T = 1.0 / (1.0 + iter_count * 0.005)
        
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            current_val = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
            
            # Update global best
            if current_val < gbest_val:
                gbest = particles[i].copy()
                gbest_val = current_val
                
            # Simulated annealing with probability of accepting worse solutions
            if current_val > gbest_val and np.random.random() < np.exp(-(current_val - gbest_val) / T):
                # Accept worse solution with some probability
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                
                if current_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_val
        
        # Periodic convex relaxation every 50 iterations
        if iter_count % 50 == 0 and iter_count > 0:
            try:
                # Solve QCP
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample k values
                k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= gbest_val * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < gbest_val:
                        gbest = h_feasible
                        gbest_val = c5_feasible
            except Exception:
                # If QCP doesn't work, proceed anyway
                pass
    
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, gbest, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            final_h = res.x
            final_c5 = objective(final_h)
            if final_c5 < gbest_val:
                gbest = final_h
                gbest_val = final_c5
    except Exception:
        pass
    
    return (list(gbest), gbest_val, n_points)
