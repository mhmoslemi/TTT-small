import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        return h
    scale = n / (2 * sum_h)
    h *= scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Multi-resolution approach
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n = len(h)
    else:
        n = 70
        h = np.ones(n) * 0.5
    
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Annealing with temperature schedule
    T = 0.5
    cooling_rate = 0.95
    min_T = 1e-4
    max_steps = 2000
    
    for i in range(max_steps):
        if time() - start_time > budget_s * 0.80:
            break
            
        # Identify regions with high overlap
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Get indices where overlapping is high
        k_offset = len(best_h) - 1
        overlaps = corr[k_offset:]  # positive shifts only
        top_indices = np.argsort(overlaps)[-int(0.15 * len(overlaps)):]
        
        # Perturb selected indices with step size that decreases with temperature
        h_new = best_h.copy()
        epsilon = T * 0.1
        
        # Randomly adjust positions based on overlap sensitivity
        for idx in np.random.choice(top_indices, size=min(5, len(top_indices)), replace=False):
            if np.random.rand() > 0.5:
                h_new[idx] += epsilon
            else:
                h_new[idx] -= epsilon
                
        # Project to constraints
        h_new = project_feasible(h_new, n)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept with simulated annealing probability
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Cooling schedule
        T = max(min_T, T * cooling_rate)
    
    # QCP refine with all shifts
    try:
        h_opt = cp.Variable(n)
        constraints = [
            cp.sum(h_opt) == n / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add all correlation constraints
        for k in range(1, n):
            left_len = n - k
            corr_sum = cp.sum(cp.multiply(h_opt[:left_len], 1 - h_opt[k:]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final projection to ensure feasibility
    final_h = project_feasible(best_h, n)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    # Ensure integral correctness
    assert abs(np.sum(final_h) - n/2) < 1e-5, "Integral constraint violated"
    
    return (list(final_h), final_c5, n)
