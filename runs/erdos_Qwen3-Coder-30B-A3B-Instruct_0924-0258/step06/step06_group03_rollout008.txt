import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n_points):
    """Create a triangular window as initialization"""
    half = n_points // 2
    if n_points % 2 == 0:
        window = np.concatenate([np.arange(1, half + 1), np.arange(half, 0, -1)])
    else:
        window = np.concatenate([np.arange(1, half + 2), np.arange(half, 0, -1)])
    return window / np.sum(window) * (n_points / 2)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = triangular_window(n_points)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Differential Evolution for global search
    de_options = {'maxiter': 300, 'popsize': 50, 'strategy': 'best1bin', 'seed': seed}
    
    # Add some diversity to initial population
    pop = [h_start.copy()]
    for _ in range(9):
        pert = np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_start + pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        pop.append(h_pert)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # DE with custom population
    def de_objective(x):
        return objective(x)
    
    # Run DE for global exploration
    try:
        de_res = differential_evolution(de_objective, bounds, seed=seed, maxiter=300, popsize=50, 
                                       strategy='best1bin', mutation=(0.5, 1), recombination=0.7)
        if de_res.success:
            h_de = de_res.x
            c5_de = objective(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    except Exception:
        pass
    
    # Local refinement with L-BFGS-B
    # First, create a smoother version to avoid local minima artifacts
    refined_h = best_h.copy()
    
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, refined_h, method='L-BFGS-B', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final check with different optimizer
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
