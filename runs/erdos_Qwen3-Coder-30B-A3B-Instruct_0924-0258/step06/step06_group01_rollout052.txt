import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: sum(h) = n_points/2, h[i] in [0,1]"""
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * (n_points/2) / sum_h
    else:
        h = np.ones_like(h) * (n_points/2) / n_points
    return h

    max_iterations = 500
    iteration = 0
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Compute current correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)

        # Compute gradient estimates for the correlation objective
        # Approximate gradient via finite differences
        
        h_new = h.copy()

        for i in range(n_points):
            # Perturb h[i] and see change in objective
            eps = 1e-5
            h_plus = h.copy()
            h_minus = h.copy()
            
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Project to feasible region
            h_plus = project_to_feasible(h_plus, n_points)

            return evaluate_erdos_solution(h_vec, dx)
        
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
