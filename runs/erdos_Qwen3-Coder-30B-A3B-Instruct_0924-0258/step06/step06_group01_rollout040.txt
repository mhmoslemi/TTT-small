import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    C_low = 0.0
    C_high = 0.38085
    tolerance = 1e-5
    max_iterations = 100
    
    def is_feasible(C):
        # Create variables
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        
        # Add overlap constraints for all shifts
        for k in range(n_points):
            # Compute overlap for shift k
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h[i] * (1 - h[j])
            constraints.append(overlap_expr <= C * n_points / 2)
        
        # Solve with CVXPY
        try:
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.CVXOPT, max_iters=500, verbose=False)
            return prob.status == 'optimal'
        except:
            return False

    # Binary search loop
    iteration = 0
    while (C_high - C_low > tolerance) and (time() - start_time < budget_s * 0.95) and iteration < max_iterations:
        C_mid = (C_low + C_high) / 2
        if is_feasible(C_mid):
            C_high = C_mid
            # Refine the solution using local optimization
            try:
                h_ref = best_h.copy()
                def objective(x):
                    return evaluate_erdos_solution(x, dx)
                
                cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(
                    objective,
                    h_ref,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons_sum,
                    options={'ftol': 1e-8, 'maxiter': 100}
                )
                if res.success:
                    h_ref = res.x
                    h_ref = np.clip(h_ref, 0, 1)
                    h_ref = h_ref * (n_points/2) / np.sum(h_ref)
                    ref_c5 = evaluate_erdos_solution(h_ref, dx)
                    if ref_c5 < best_c5:
                        best_h = h_ref.copy()
                        best_c5 = ref_c5
            except:
                pass
        else:
            C_low = C_mid
        iteration += 1

    # Final refinement using local optimization
    try:
        def objective(x):
            return evaluate_erdos_solution(x, dx)
        
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass

    return (list(best_h), best_c5, n_points)
