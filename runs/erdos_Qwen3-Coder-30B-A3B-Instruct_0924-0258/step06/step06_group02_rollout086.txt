import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        # Create a symmetric pattern
        h_start = np.zeros(n_points)
        centers = [0.5, 1.0, 1.5]
        spread = 0.3
        for c in centers:
            idx = int(c * n_points / 2)
            for i in range(max(0, idx-4), min(n_points, idx+4)):
                h_start[i] = max(h_start[i], 1.0 - abs(i - idx) * 0.2)
        h_start = h_start * (n_points / (2 * np.sum(h_start)))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # SDP relaxation for lower bound
    try:
        h_sdp = cp.Variable(n_points)
        constraints = [
            cp.sum(h_sdp) == n_points / 2.0,
            h_sdp >= 0,
            h_sdp <= 1
        ]
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_start = h_sdp.value
    except Exception:
        pass
    
    # Trust Region Optimization
    current_tr_radius = 0.3
    lambda_merit = 0.05
    patience = 0
    max_patience = 20
    
    for i in range(2000):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Perturb within trust region
        noise = np.random.uniform(-current_tr_radius, current_tr_radius, n_points)
        h_new = best_h + noise
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct integral
        sum_h = np.sum(h_new)
        if sum_h > 0:
            h_new = h_new * (n_points / (2 * sum_h))
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        constraint_violation = max(0, np.max(h_new - 1), np.min(h_new), abs(np.sum(h_new) - n_points/2))
        merit_new = c5_new + lambda_merit * constraint_violation
        
        merit_best = best_c5 + lambda_merit * max(0, np.max(best_h - 1), np.min(best_h), abs(np.sum(best_h) - n_points/2))
        
        if merit_new < merit_best:
            best_h = h_new.copy()
            best_c5 = c5_new
            current_tr_radius = min(current_tr_radius * 1.1, 0.5)
            patience = 0
        else:
            current_tr_radius = max(current_tr_radius * 0.8, 0.01)
            patience += 1
            
        if patience > max_patience:
            patience = 0
            current_tr_radius = min(current_tr_radius * 1.1, 0.5)
    
    # Final refinement with optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraintIntegral(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraintIntegral}]
    
    try:
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, 
                          options={'maxiter': 200, 'ftol': 1e-8})
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            ref_sum = np.sum(refined_h)
            if ref_sum > 0:
                refined_h = refined_h * (n_points / (2 * ref_sum))
            c5_refined = evaluate_erdos_solution(refined_h, dx)
            if c5_refined < best_c5:
                best_h = refined_h
                best_c5 = c5_refined
    except:
        pass
    
    # Final feasibility check and normalization
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * (n_points / (2 * sum_h))
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
