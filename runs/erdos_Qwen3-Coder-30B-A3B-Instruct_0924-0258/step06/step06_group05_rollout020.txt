import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    
    # Start with binary construction
    if np.allclose(h_start, h_start.round()):
        h_binary = h_start.round()
    else:
        # Convert to binary with correct sum
        h_binary = np.zeros(n_points)
        ones_count = int(n_points / 2)
        h_binary[:ones_count] = 1
        np.random.shuffle(h_binary)
    
    best_c5 = evaluate_erdos_solution(h_binary, dx)
    
    # Binary search with bit swaps and local refinement
    max_iter = 1000
    temp = 1.0
    temp_decay = 0.995
    
    # Get binary solution from initial_h_values
    h_binary = np.zeros(n_points)
    ones_count = int(n_points / 2)
    h_binary[:ones_count] = 1
    # Shuffle to make it non-trivial
    np.random.shuffle(h_binary)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random bit swap
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if i != j:
            h_new = h_binary.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept with simulated annealing probability
            if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / temp):
                h_binary = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_binary.copy()
        
        # Periodic convex refinement
        if iteration % 100 == 0 and iteration > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample k values for constraints
                k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-8)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
                
        temp *= temp_decay
    
    return (list(best_h), best_c5, n_points)
