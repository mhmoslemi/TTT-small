import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a structured pattern to avoid local minima
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO Parameters
    n_particles = 50
    inertia = 0.7
    cognitive = 1.4
    social = 1.4
    velocity_max = 0.2
    
    # Initialize PSO
    positions = np.array([h_start.copy() for _ in range(n_particles)])
    velocities = np.random.uniform(-velocity_max, velocity_max, (n_particles, n_points))
    personal_best_positions = positions.copy()
    personal_best_values = np.array([evaluate_erdos_solution(p, dx) for p in positions])
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best_positions[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO iterations
    max_iter = 200
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (personal_best_positions[i] - positions[i]) +
                social * r2 * (global_best - positions[i])
            )
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            positions[i] += velocities[i]
            positions[i] = np.clip(positions[i], 0, 1)
            
            # Normalize to sum constraint
            positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
            
            # Evaluate
            current_c5 = evaluate_erdos_solution(positions[i], dx)
            if current_c5 < personal_best_values[i]:
                personal_best_values[i] = current_c5
                personal_best_positions[i] = positions[i].copy()
                
                if current_c5 < global_best_value:
                    global_best_value = current_c5
                    global_best = positions[i].copy()
    
    # Update best solution from PSO
    best_h = global_best.copy()
    best_c5 = global_best_value
    
    # Refinement using SQP with adaptive constraints
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample key lags for constraints (adaptive)
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        lags = np.arange(len(corr_full))
        # Use strongest lags for constraints
        strongest_indices = np.argsort(corr_full)[-10:]
        strongest_lags = lags[strongest_indices]
        
        for k in strongest_lags:
            # Adjust for valid indices in convolution
            if k < len(best_h):
                corr_sum = cp.sum(cp.multiply(h_opt[:len(best_h)-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-len(best_h)+1:], 1 - h_opt[:len(best_h)-(k-len(best_h)+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
        
    # Final refinement using scipy minimize on the best candidate
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    # Final normalization to sum constraint
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
