import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        return h
    scale = n / (2 * sum_h)
    h *= scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Focused Perturbations
    T = 0.5
    cooling_factor = 0.95
    max_steps = 2000
    patience = 100
    no_improve_count = 0
    
    for i in range(max_steps):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Focus on high-overlap regions
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Find indices where overlap is high (top 15%)
        top_indices = np.argsort(corr)[-int(0.15 * len(corr)):]
        
        h_new = best_h.copy()
        # Modify top overlap indices with perturbation
        for k in top_indices:
            if k < len(h_new):
                h_new[k] += np.random.normal(0, 0.1)
        
        h_new = project_feasible(h_new, n_points)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > patience:
                # Restart with new random point
                h_new = np.random.uniform(0, 1, n_points)
                h_new = project_feasible(h_new, n_points)
                c5_new = evaluate_erdos_solution(h_new, dx)

                    best_h = h_new.copy()
                    best_c5 = c5_new
                no_improve_count = 0
        
        T *= cooling_factor
    
    # QCP refinement with all shifts
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add overlap constraints for all shifts
        for k in range(1, n_points):
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible

        # Use the best solution so far as starting point

        # Create a reasonable initial approximation
        h_final = h_init.copy()
        h_final = project_feasible(h_final, n_points)
        c5_final = evaluate_erdos_solution(h_final, dx)
        
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    # Ensure final solution is feasible
    best_h = project_feasible(best_h, n_points)
    c5_final = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), c5_final, n_points)
