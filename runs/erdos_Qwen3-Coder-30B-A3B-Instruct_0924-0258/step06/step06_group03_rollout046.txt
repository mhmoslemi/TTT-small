import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create Hamming window-based initialization
        h_start = 0.54 + 0.46 * np.cos(np.pi * (np.arange(n_points) / n_points - 0.5) * 2)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Projected gradient descent with fixed step size
    max_iter = 1000
    step_size = 0.01
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate current objective and find the maximizing lag
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient with respect to the lag that maximizes overlap
        # The correlation is equivalent to a convolution with flipped kernel
        # Gradient of overlap at k is -2 * h * (convolve(h, 1-h, mode='full')[k])
        # Gradient w.r.t. h is -2 * (1-h) * delta_{k}(convolution with 1-h)
        # In simplified finite difference: gradient = -2 * (1-h[0:n_points-k]) where k = max_lag
        # For effectively the gradient from the right side of the correlation
        # actually: correlation(k) = sum h[i] * (1 - h[i+k])
        # grad_h[i] = - (1 - h[i + k]) where k = max_lag
        
        # Now get the actual gradient computation for corr_max
        # Gradient analysis based on derivative of correlation
        # d/dh_i corr(h, 1-h)[k] = d/dh_i sum_j h[j] * (1 - h[j+k]) \\
        # = (1 - h[i+k]) = 0 when i+k is out of range
        gradient = np.zeros_like(best_h)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] = -(1 - best_h[i + max_lag])
        else:
            # max_lag < 0
            for i in range(-max_lag, n_points):
                gradient[i] = -(1 - best_h[i + max_lag])
        
        # Update with gradient descent
        h_new = best_h - step_size * gradient
        
        # Project to valid space: clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final validation
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) - n_points / 2.0) < 1e-10
    
    return (list(best_h), best_c5, n_points)
