import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 200
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = population_size // 10
    tournament_size = population_size // 5
    
    # Initialize population
    def create_individual():
        h = np.random.normal(0.5, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        population.append(create_individual())
    
    # GA main loop
    no_improvement_count = 0
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Evaluate fitness
        fitness = []
        for ind in population:
            c5 = evaluate_erdos_solution(ind, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Early termination
        if no_improvement_count > 50:
            break
        
        # Selection (tournament)
        selected = []
        for _ in range(population_size):
            tournament = random.sample(range(population_size), tournament_size)
            winner = min(tournament, key=lambda i: fitness[i])
            selected.append(population[winner].copy())
        
        # Crossover and Mutation
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected, 2)
            
            if random.random() < crossover_rate:
                # Uniform crossover
                mask = np.random.random(n_points) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                for i in range(n_points):
                    if random.random() < mutation_rate:
                        child[i] += np.random.normal(0, 0.05)
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                
                if len(new_population) < population_size:
                    new_population.append(child)
        
        population = new_population
    
    # Local refinement with SLSQP on best individuals
    top_candidates = population[:elite_size]
    for candidate in top_candidates:
        if time() - start_time > budget_s * 0.95:
            break
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, candidate, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            continue
    
    return (list(best_h), best_c5, n_points)
