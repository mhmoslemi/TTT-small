import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.random.rand(n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    rho = 1e-4
    max_iter = 1000
    tol = 1e-6
    
    h = h_start.copy()
    z = h.copy()
    u = np.zeros(n_points)
    iteration = 0
    
    # ADMM iterations
    while time() - start_time < budget_s * 0.95 and iteration < max_iter:
        # Primal update: solve convex subproblem
        # minimize (rho/2) * ||z - h + u||_2^2 + (1/rho) * C5(h)
        # subject to sum(z) = n_points/2, 0 <= z <= 1

        # We'll approximate the C5 term by linearizing it
        # This is a simplified approximation for computational tractability
        # In practice, exact treatment would require solving the max-overlap problem
        # For now, we'll focus on simple convex relaxation
        
        # Solve using CVXPY
        try:
            z_var = cp.Variable(n_points)
            objective = cp.Minimize(
                cp.sum_squares(z_var - h + u) * rho / 2 + 
                cp.sum(cp.multiply(1 - h, z_var))  # Simplified linearization
            )
            constraints = [
                cp.sum(z_var) == n_points / 2,
                z_var >= 0,
                z_var <= 1
            ]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                z = z_var.value
            else:
                z = np.clip(h + u, 0, 1)
                z = z * (n_points/2) / np.sum(z)
        except:
            # Fallback to simpler update
            z = np.clip(h + u, 0, 1)
            z = z * (n_points/2) / np.sum(z)
        
        # Dual update
        u = u + h - z
        
        # Proximal update on h
        h = z - (1/rho) * (1 - 2 * z)  # gradient of linearized term  
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Check convergence
        if np.linalg.norm(h - z) < tol:
            break
            
        iteration += 1
        
        # Update best solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    # Final refinement with local optimization
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
