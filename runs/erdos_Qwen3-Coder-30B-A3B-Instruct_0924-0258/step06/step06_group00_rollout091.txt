import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for GA
    population_size = 100
    generations = 200
    elite_size = population_size // 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = population_size // 5
    
    # Initialize population
    population = []
    population.append(h_start.copy())
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # GA loop
    no_improvement_count = 0
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= 50:
            break
        
        # Selection (tournament)
        new_population = population[:elite_size]
        while len(new_population) < population_size:
            # Tournament selection
            selected = []
            for _ in range(tournament_size):
                idx = random.randint(0, population_size - 1)
                selected.append((idx, fitness[idx]))
            selected.sort(key=lambda x: x[1])
            parent_idx = selected[0][0]
            parent = population[parent_idx]
            new_population.append(parent.copy())
        
        # Crossover and mutation
        for i in range(elite_size, population_size, 2):
            if random.random() < crossover_rate and i + 1 < population_size:
                parent1 = new_population[i]
                parent2 = new_population[i + 1]
                
                # Uniform crossover
                mask = np.random.random(n_points) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
                
                new_population[i] = child1
                new_population[i + 1] = child2
            else:
                # Just copy existing
                new_population[i] = population[i].copy()
                if i + 1 < population_size:
                    new_population[i + 1] = population[i + 1].copy()
        
        # Mutation
        for i in range(elite_size, population_size):
            if random.random() < mutation_rate:
                h = new_population[i]
                noise = np.random.normal(0, 0.05, n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                new_population[i] = h
        
        population = new_population
    
    # Local refinement on top candidates
    for i in range(min(10, len(population))):
        h_refined = population[i].copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
