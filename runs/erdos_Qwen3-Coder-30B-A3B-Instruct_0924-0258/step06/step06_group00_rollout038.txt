import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)

    # DE bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE parameters
    popsize = 100
    maxiter = 500
    mutation_factor = 0.5
    crossover_prob = 0.7
    
    def de_objective(h):
        # Normalize to integral constraint
        h_normalized = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h_normalized, dx)
    
    # Generate diverse initial population
    initial_pop = [h_start.copy()]
    for _ in range(popsize - 1):
        # Random perturbations
        h_perturbed = h_start + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        initial_pop.append(h_perturbed)
    
    # Differential Evolution with multiple independent runs
    de_runs = 5
    for run_idx in range(de_runs):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Shuffle the initial population
        random.shuffle(initial_pop)
        
        # Run DE with custom setup
        result = differential_evolution(
            de_objective,
            bounds,
            maxiter=maxiter,
            popsize=popsize,
            mutation=(mutation_factor, 0.9),
            recombination=crossover_prob,
            seed=seed + run_idx,
            disp=False,
            polish=True
        )
        
        # Normalize result
        h_opt = result.x * (n_points / 2) / np.sum(result.x)
        c5_opt = evaluate_erdos_solution(h_opt, dx)
        
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Local refinement with SLSQP for best solution
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    h_refined = best_h.copy()
    h_refined = np.clip(h_refined, 0, 1)
    h_refined = h_refined * (n_points/2) / np.sum(h_refined)
    
    try:
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
