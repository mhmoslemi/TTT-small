import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        h_start = 0.5 + 0.3 * np.sin(np.pi * x)
        h_start = np.clip(h_start, 0, 1)
        # Normalize to ensure sum(h) = n_points / 2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based refinement
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0
    
    def constraint_bounds(h_vec):
        return np.concatenate([h_vec, 1 - h_vec])
    
    # Use a robust optimization strategy: alternating between gradient steps and refinement
    max_iter = 500
    step_size = 0.05
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try a gradient-based step using a simple gradient approximation
        h_grad = best_h.copy()
        # Compute approximate gradient using finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        
        # Perturb each component slightly to estimate gradient
        for j in range(n_points):
            h_plus = h_grad.copy()
            h_minus = h_grad.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            # Ensure bounds
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            
            grad[j] = (objective(h_plus) - objective(h_minus)) / (2 * eps)

        h_new = h_grad - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)

        c5_new = evaluate_erdos_solution(h_new, dx)

            best_h = h_new
            best_c5 = c5_new

        if np.random.random() < 0.1:
            noise = np.random.normal(0, step_size, n_points)
            h_rand = best_h + noise
            h_rand = np.clip(h_rand, 0, 1)
            h_rand = h_rand * (n_points/2) / np.sum(h_rand)
            c5_rand = evaluate_erdos_solution(h_rand, dx)

        constraints = [

        for k in range(1, n_points):

            if k < n_points // 2:

        prob = cp.Problem(cp.Minimize(0.0), constraints)

    # Final normalization and validation

    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
