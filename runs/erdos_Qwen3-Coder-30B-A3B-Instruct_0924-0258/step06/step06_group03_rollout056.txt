import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create structured initial guess based on triangular window
        h_start = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h_start[i] = max(0, 1.0 - abs(i - mid) / (mid + 1))
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Create constraints dictionary
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Differential Evolution
    de_result = differential_evolution(
        objective, 
        bounds, 
        maxiter=300, 
        popsize=50, 
        tol=1e-8,
        constraints=cons,
        seed=seed,
        disp=False
    )
    
    # Update best solution from DE
    if de_result.success:
        h_de = de_result.x
        c5_de = objective(h_de)
        if c5_de < best_c5:
            best_h = h_de.copy()
            best_c5 = c5_de
    
    # Local refinement with L-BFGS-B
    if time() - start_time < budget_s * 0.9:
        try:
            # Create bounds and constraints for L-BFGS
            bounds_lbfgs = [(0, 1) for _ in range(n_points)]
            
            # Refine using local optimizer
            for _ in range(10):  # Try multiple starting points
                # Random perturbation of best solution
                h_init = best_h + np.random.normal(0, 0.01, n_points)
                h_init = np.clip(h_init, 0, 1)
                h_init = h_init * (n_points/2) / np.sum(h_init)
                
                # Optimize with L-BFGS-B
                res = minimize(
                    objective, 
                    h_init,
                    method='L-BFGS-B',
                    bounds=bounds_lbfgs,
                    constraints=cons,
                    options={'maxiter': 100, 'ftol': 1e-9},
                    callback=None
                )
                
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
