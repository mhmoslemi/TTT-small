import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Handle initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create triangular window with constraints
        h_start = 1 - np.abs(np.linspace(0, 1, n_points) - 0.5) * 2
        # Normalize to satisfy constraint sum(h) = n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    max_iter = 5000
    T = 0.5
    alpha = 0.99
    min_T = 1e-6
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # SA main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Generate candidate solution
        delta = np.random.normal(0, T, n_points)
        candidate_h = current_h + delta
        
        # Apply convex combination to maintain constraint
        sum_old = np.sum(current_h)
        sum_new = np.sum(candidate_h)
        if sum_new > 0:
            # Scaling to maintain integral constraint
            scaled_h = candidate_h * (sum_old / sum_new)
            # Clip and re-normalize
            scaled_h = np.clip(scaled_h, 0, 1)
            final_h = scaled_h * (n_points/2) / np.sum(scaled_h)
        else:
            final_h = current_h.copy()
            
        candidate_c5 = evaluate_erdos_solution(final_h, dx)
        
        # Metropolis acceptance criterion
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            current_h = final_h.copy()
            current_c5 = candidate_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Cool down
        T = max(min_T, T * alpha)
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
