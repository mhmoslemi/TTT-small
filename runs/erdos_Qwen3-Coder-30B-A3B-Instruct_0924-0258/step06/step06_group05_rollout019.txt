import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with random perturbations
    max_iter = 10000
    step_size = 0.01
    noise_scale = 0.05
    noise_decay = 0.99
    iter_no_improve = 0
    max_no_improve = 100
    
    h_current = best_h.copy()
    current_c5 = best_c5
    
    for idx in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find the maximum overlap shift
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        
        # Compute gradient of the correlation at k_max
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                gradient[i] += 1 - h_current[i + k_max]
            if i - k_max >= 0:
                gradient[i] -= h_current[i - k_max]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm * step_size
            
        # Take gradient step
        h_new = h_current - gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < current_c5:
            h_current = h_new.copy()
            current_c5 = c5_new
            iter_no_improve = 0
            
            if c5_new < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_new
        else:
            iter_no_improve += 1
            
        # Add noise periodically
        if idx % 30 == 0:
            noise = np.random.normal(0, noise_scale, n_points)
            h_new = h_current + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < current_c5:
                h_current = h_new.copy()
                current_c5 = c5_new
                
            noise_scale *= noise_decay
        
        # Check for early termination
        if iter_no_improve > max_no_improve:
            break
    
    # Final refinement with convex optimization
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample k values for constraints
        k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
        for k_val in k_vals:
            k = int(k_val * n_points / 2)
            if k < n_points and k > 0:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
