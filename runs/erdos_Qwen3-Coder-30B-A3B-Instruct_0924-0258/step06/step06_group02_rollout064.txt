import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        # Initialize with symmetric pattern inspired by root systems
        h_start = np.zeros(n_points)
        centers = np.array([0.6, 1.0, 1.4])  # Empirically derived centers
        spread = 0.4
        for c in centers:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - c))
            h_start[max(0, idx-4):min(n_points, idx+4)] = 0.8  # High concentration regions
        # Normalize to have integral = 1
        h_start = h_start * (n_points / (2 * np.sum(h_start)))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # SDP relaxation for lower bound
    try:
        h_sdp = cp.Variable(n_points)
        constraints = [
            cp.sum(h_sdp) == n_points / 2.0,
            h_sdp >= 0,
            h_sdp <= 1
        ]
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, verbose=False, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_start = h_sdp.value
    except Exception:
        pass
    
    # Trust Region Optimization with Merit Function
    current_tr_radius = 0.3
    lambda_merit = 0.1
    last_improvement = 0
    
    for iteration in range(2000):
        if time() - start_time > budget_s * 0.85:
            break
        
        # Generate perturbation within trust region
        perturbation = np.random.uniform(-current_tr_radius, current_tr_radius, n_points)
        h_new = best_h + perturbation
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct integral
        sum_h = np.sum(h_new)
        if sum_h > 0:
            h_new = h_new * (n_points / (2 * sum_h))
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        constraint_violation = max(0, np.max(h_new - 1), np.min(h_new), abs(np.sum(h_new) - n_points/2))
        merit_new = c5_new + lambda_merit * constraint_violation
        
        # Baseline merit
        constraint_violation_best = max(0, np.max(best_h - 1), np.min(best_h), abs(np.sum(best_h) - n_points/2))
        merit_best = best_c5 + lambda_merit * constraint_violation_best
        
        # Accept or reject
        if merit_new < merit_best:
            best_h = h_new.copy()
            best_c5 = c5_new
            current_tr_radius = min(0.5, current_tr_radius * 1.1)
            last_improvement = iteration
        else:
            current_tr_radius = max(0.01, current_tr_radius * 0.9)
            
        # Adapt lambda_merit
        if constraint_violation > 1e-4:
            lambda_merit *= 1.3
        else:
            lambda_merit *= 0.9
    
    # Final refinement with projected gradient or another local method
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        sum_h = np.sum(h_final)
        if sum_h > 0:
            h_final = h_final * (n_points / (2 * sum_h))
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    # Ensure feasibility and return
    h_final = np.clip(best_h, 0, 1)
    sum_h = np.sum(h_final)
    if sum_h > 0:
        h_final = h_final * (n_points / (2 * sum_h))
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
