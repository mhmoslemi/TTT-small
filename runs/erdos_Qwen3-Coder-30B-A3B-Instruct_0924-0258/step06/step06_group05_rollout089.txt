import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    inertia_decay = 0.995
    temp = 1.0
    temp_decay = 0.995
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO iteration
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update inertia weight
        w *= inertia_decay
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (pbest[i] - particles[i])
                           + c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair and bound
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            curr_val = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if curr_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = curr_val
                
                # Update global best
                if curr_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = curr_val
                    
            # Simulated annealing acceptance
            if curr_val > gbest_val:
                prob = np.exp((gbest_val - curr_val) / temp)
                if np.random.rand() < prob:
                    pbest[i] = particles[i].copy()
                    pbest_vals[i] = curr_val
                    if curr_val < gbest_val:
                        gbest = particles[i].copy()
                        gbest_val = curr_val
            
            # Update temperature
            temp *= temp_decay
            
        # Convex relaxation every 50 iterations
        if iteration % 50 == 0 and iteration > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample k values for constraints, reducing density
                k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)  # to index space
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= gbest_val * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < gbest_val:
                        gbest = h_feasible
                        gbest_val = c5_feasible
            except Exception:
                pass
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, gbest, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < gbest_val:
                gbest = h_refined
                gbest_val = c5_refined
    except Exception:
        pass
    
    return (list(gbest), gbest_val, n_points)
