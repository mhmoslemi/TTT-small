import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, norm
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.random.rand(n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / n_points)
    
    # ADMM parameters
    rho = 1e-4
    max_iter = 1000
    tol = 1e-6
    iteration = 0
    
    h = h_start.copy()
    z = h.copy()
    u = np.zeros(n_points)
    
    # Keep track of best solution found
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    
    # Inner ADMM loop with CVXOPT
    while iteration < max_iter and time() - start_time < budget_s * 0.95:
        # Step 1: Solve the convex subproblem for z
        # minimize (rho/2) * ||z - (h - u)||_2^2 + (1/rho) * max_k int h(x)(1-h(x+k)) dx
        # subject to sum(z) = n_points/2, 0 <= z[i] <= 1
        
        # This is a complex constraint involving the correlation term
        # We'll use a simpler approach: solve the convex relaxation using cvxpy
        try:
            z_var = Variable(n_points)
            # Simple convex objective with constraints
            objective = Minimize(cp_sum((z_var - (h - u))**2) / 2)
            constraints = [
                cp_sum(z_var) == n_points / 2,
                z_var >= 0,
                z_var <= 1
            ]
            prob = Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                z_new = z_var.value
                # Ensure exact feasibility
                z_new = np.clip(z_new, 0, 1)
                z_new = z_new * (n_points/2) / np.sum(z_new)
                z = z_new.copy()
            else:
                # fallback: simple projection
                z = np.clip(h - u, 0, 1)
                z = z * (n_points/2) / np.sum(z)
                
        except:
            # fallback: simple projection
            z = np.clip(h - u, 0, 1)
            z = z * (n_points/2) / np.sum(z)
        
        # Step 2: Update dual variables
        u = u + h - z
        
        # Step 3: Update h
        h = z - (1/rho) * (z - (h - u))
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Check convergence
        if np.linalg.norm(h - z) < tol:
            break
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        iteration += 1
    
    # Final refinement with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(
            lambda x: evaluate_erdos_solution(x, 2.0 / n_points),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
