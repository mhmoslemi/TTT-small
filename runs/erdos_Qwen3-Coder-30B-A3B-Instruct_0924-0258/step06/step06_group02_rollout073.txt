import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a structured pattern
        h_start = np.ones(n_points) * 0.5
        # Add some variability
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive learning rate
    max_iterations = 1000
    learning_rate = 0.02
    decay_factor = 0.995
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute gradient for total overlap
        gradient = np.zeros(n_points)
        for k in range(1, n_points):
            # Forward differences for valid indices
            if k < n_points:
                gradient[:n_points-k] += (1 - best_h[k:])
            if k < n_points:
                gradient[k:] -= best_h[:-k]
        
        # Apply gradient and project
        new_h = best_h - learning_rate * gradient
        new_h = np.clip(new_h, 0, 1)
        # Scale to meet constraint sum(h) = n/2
        scale_factor = (n_points/2) / np.sum(new_h)
        new_h = new_h * scale_factor
        
        # Evaluate new point
        c5_new = evaluate_erdos_solution(new_h, dx)
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
            
        # Decay learning rate
        learning_rate *= decay_factor
    
    # Final refinement using a local search
    try:
        # Try local perturbations
        perturbation_magnitude = 0.01
        for _ in range(1000):
            if time() - start_time > budget_s * 0.95:
                break
            # Create small random perturbation
            perturbation = np.random.normal(0, perturbation_magnitude, n_points)
            temp_h = best_h + perturbation
            temp_h = np.clip(temp_h, 0, 1)
            # Normalize
            temp_h = temp_h * (n_points/2) / np.sum(temp_h)
            
            c5_temp = evaluate_erdos_solution(temp_h, dx)
            if c5_temp < best_c5:
                best_h = temp_h.copy()
                best_c5 = c5_temp
    except Exception:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
