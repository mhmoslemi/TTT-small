import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a structured starting point to better explore
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 200
    w = 0.7  # inertia weight
    c1 = 1.4  # cognitive parameter
    c2 = 1.4  # social parameter
    v_max = 0.2
    
    # Initialize particles
    particles = [h_start.copy() for _ in range(n_particles)]
    velocities = [np.random.uniform(-v_max, v_max, n_points) for _ in range(n_particles)]
    personal_best = [p.copy() for p in particles]
    personal_best_values = [evaluate_erdos_solution(p, dx) for p in particles]
    global_best = best_h.copy()
    global_best_value = best_c5
    
    # Adaptive constraint sampling
    constraint_lags = list(range(1, n_points // 2))
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) +
                           c2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Clip to [0, 1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_c5 < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = current_c5
                
                # Update global best
                if current_c5 < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = current_c5
                    best_h = particles[i].copy()
                    best_c5 = current_c5

    # Perform SQP refinement to tighten the solution
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Use a subset of the critical lags from the best solution
        corr = np.correlate(global_best, 1 - global_best, mode='full')
        critical_indices = np.argsort(-corr)[1:11]  # Top 10 lags excluding k=0
        for k in critical_indices:
            # Convert index back to lag (handles circular correlation)
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                k_corr = k - n_points + 1
                corr_sum = cp.sum(cp.multiply(h_opt[k_corr:], 1 - h_opt[:n_points-k_corr]))
            constraints.append(corr_sum <= global_best_value * dx + 1e-8)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
