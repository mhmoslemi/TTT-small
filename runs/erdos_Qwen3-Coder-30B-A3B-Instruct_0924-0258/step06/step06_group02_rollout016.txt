import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def check_feasibility(C_candidate, h_start, n, dx, sample_rate=0.2):
    k_samples = np.random.choice(range(1, n), size=max(1, int(n * sample_rate)), replace=False)
    
    h_opt = cp.Variable(n)
    constraints = [cp.sum(h_opt) == n / 2]
    constraints += [0 <= h_opt, h_opt <= 1]
    
    for k in k_samples:
        if k < n:
            corr_sum = cp.sum(cp.multiply(h_opt[:n-k], 1 - h_opt[k:]))
        else:
            corr_sum = cp.sum(cp.multiply(h_opt[k-n+1:], 1 - h_opt[:n-(k-n+1)]))
        constraints.append(corr_sum <= C_candidate * dx)
    
    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=cp.ECOS, eps=1e-6)
        return prob.status == cp.OPTIMAL
    except:
        return False

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search for tighter C5
    low = 0.38
    high = best_c5
    iterations = 20
    
    for _ in range(iterations):
        if time() - start_time > budget_s * 0.8:
            break
        mid = (low + high) / 2
        if check_feasibility(mid, h_start, n_points, dx):
            best_c5 = mid
            high = mid
        else:
            low = mid
    
    # Refinement with full constraints
    try:
        h_refine = cp.Variable(n_points)
        constraints = [cp.sum(h_refine) == n_points / 2.0]
        constraints += [0 <= h_refine, h_refine <= 1]
        
        for k in range(1, n_points):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_refine[:n_points-k], 1 - h_refine[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_refine[k-n_points+1:], 1 - h_refine[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx)
            
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_optimal = h_refine.value
            c5_optimal = evaluate_erdos_solution(h_optimal, dx)
            if c5_optimal < best_c5:
                best_h = h_optimal
                best_c5 = c5_optimal
    except Exception:
        pass
    
    # Final projection to exact integral
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * (n_points / (2 * sum_h))
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
