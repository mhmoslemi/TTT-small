import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use sinusoidal pattern as initialization
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)
        h_start = np.clip(h_start, 0, 1)
    
    # Normalize to sum to n_points/2
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # Gradient-based optimization with box constraints
    def objective(h):
        return evaluate_erdos_solution(h, 2.0/n_points)
    
    def constraint_sum(h):
        return np.sum(h) - n_points/2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    max_iter = 200
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Optimize with scipy
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                         options={'maxiter': 5, 'ftol': 1e-6})
        if result.success:
            h_new = result.x
            # Project to feasible region
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, 2.0/n_points)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Try to solve a relaxed optimization problem with CVXPY
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraints for key lags (relaxed approach)
        for k in range(1, n_points//2):  # Only a subset of lags
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * 2.0/n_points + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, 2.0/n_points)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    
    except Exception:
        pass

    # Final refinement
    try:
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                         options={'maxiter': 10, 'ftol': 1e-8})
        if result.success:
            h_final = result.x
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, 2.0/n_points)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, 2.0/n_points)
    
    return (list(best_h), final_c5, n_points)
