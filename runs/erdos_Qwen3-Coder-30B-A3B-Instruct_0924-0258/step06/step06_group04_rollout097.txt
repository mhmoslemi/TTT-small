import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Local search with perturbations
    max_iter = 1000
    step_size = 0.05
    decay_rate = 0.995
    min_step = 0.0001
    
    def project_and_normalize(h_in):
        h_proj = np.clip(h_in, 0, 1)
        h_proj = h_proj * (n_points/2) / np.sum(h_proj)
        return h_proj
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate perturbation
        noise = np.random.normal(0, step_size, n_points)
        h_new = best_h + noise
        h_new = project_and_normalize(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        
        # Adaptive step size
        step_size = max(min_step, step_size * decay_rate)
        
        # Occasionally do a stochastic restart
        if i % 100 == 0 and time() - start_time < budget_s * 0.9:
            h_restart = np.random.rand(n_points) * 0.5
            h_restart = project_and_normalize(h_restart)
            c5_restart = evaluate_erdos_solution(h_restart, dx)
            if c5_restart < best_c5:
                best_h = h_restart
                best_c5 = c5_restart
    
    # Convex relaxation refinement using CVXPY
    try:
        # Use the best solution so far as starting point for convex optimization
        h_feasible = cp.Variable(n_points)
        constraints = [
            cp.sum(h_feasible) == n_points / 2.0,
            h_feasible >= 0,
            h_feasible <= 1
        ]
        
        # Sample a subset of constraints for efficiency
        sample_indices = np.linspace(0, n_points-1, min(20, n_points), dtype=int)
        for k in sample_indices:
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_feasible[:n_points-k], 1 - h_feasible[k:]))
                # Add a small margin to avoid numerical issues
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
            else:
                corr_sum = cp.sum(cp.multiply(h_feasible[k-n_points+1:], 1 - h_feasible[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        # Use a faster solver like ECOS for this subproblem
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_solution = h_feasible.value
            # Project and normalize
            h_solution = np.clip(h_solution, 0, 1)
            h_solution = h_solution * (n_points/2) / np.sum(h_solution)
            c5_solution = evaluate_erdos_solution(h_solution, dx)
            if c5_solution < best_c5:
                best_h = h_solution
                best_c5 = c5_solution
    except Exception:
        pass
    
    # Final refinement with scipy minimize
    try:
        # Define objective for scipy minimize
        def obj(h_in):
            h_in = np.clip(h_in, 0, 1)
            h_in = h_in * (n_points/2) / np.sum(h_in)
            return evaluate_erdos_solution(h_in, dx)
        
        # Constraints
        def constraint_sum(h_in):
            return np.sum(h_in) - n_points/2
        
        constraints = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use L-BFGS-B for this part
        result = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, 
                         constraints=constraints, options={'maxiter': 100})
        if result.success:
            h_final = np.clip(result.x, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
