import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Sinusoidal initialization
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization
    max_iter = 5000
    step_size = 0.01
    tolerance = 1e-6
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient approximation
        h_current = best_h.copy()
        h_plus = h_current.copy()
        h_minus = h_current.copy()
        epsilon = 1e-4
        
        # Compute finite differences for gradient
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus[j] += epsilon
            h_minus[j] -= epsilon
            
            # Normalize
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[j] = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Reset
            h_plus[j] = h_current[j]
            h_minus[j] = h_current[j]
        
        # Update
        h_new = h_current - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            
        # Adaptive step sizing
        if i % 100 == 0 and i > 0:
            step_size *= 0.99
    
    # Final optimization with CVXPY to enforce constraints
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add overlap constraints for a subset of major lags
        lags = [1, 2, 3, 4, 5]
        for k in lags:
            if k < n_points:
                # Compute appropriate correlation sum
                corr_sum = 0
                for i in range(n_points - k):
                    corr_sum += h_opt[i] * (1 - h_opt[i + k])
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        # Add all lags for more precision if not too large
        if n_points <= 20:
            for k in range(1, n_points):
                corr_sum = 0
                for i in range(n_points - k):
                    corr_sum += h_opt[i] * (1 - h_opt[i + k])
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final normalization and evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
