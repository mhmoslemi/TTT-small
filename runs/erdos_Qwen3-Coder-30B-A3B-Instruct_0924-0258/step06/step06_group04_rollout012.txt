import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Grid refinement with increasing resolution
    resolutions = [10, 20, 35]  # Starting from coarse to fine
    for n_res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate or create initial guess
        if n_res <= n_points:
            h_init = best_h[:n_res]
        else:
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_res)
            h_init = np.interp(x_new, x_old, best_h)
        
        # Normalize to sum to n_points/2
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_res / 2) / np.sum(h_init)
        
        # Simulated Annealing with gradient-like steps
        h_current = h_init.copy()
        c5_current = evaluate_erdos_solution(h_current, 2.0 / n_res)
        
        # Adaptive parameters
        temp = 0.1
        alpha = 0.05
        decay = 0.995
        min_temp = 1e-6
        iter_per_temp = 20
        
        for i in range(100):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb using random step
            perturbation = np.random.normal(0, temp, n_res)
            h_new = h_current + perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_res / 2) / np.sum(h_new)
            
            # Evaluate new solution
            c5_new = evaluate_erdos_solution(h_new, 2.0 / n_res)
            
            # Accept or reject based on Metropolis criterion
            if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / temp):
                h_current = h_new
                c5_current = c5_new
            
            # Update temperature
            temp = max(min_temp, temp * decay)
            
            # Update best solution
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
            
        # Final clip and normalize
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Binary search to tighten bound
    target = 0.3807
    lower = best_c5
    upper = max(0.3807, best_c5 + 0.001)
    
    for _ in range(20):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (lower + upper) / 2
        try:
            # Build feasibility problem using CVXPY
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add overlap constraints
            for k in range(1, n_points):
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    break
                constraints.append(corr_sum <= mid * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            
            if prob.status == cp.OPTIMAL:
                lower = mid
            else:
                upper = mid
        except Exception:
            upper = mid
    
    best_c5 = (lower + upper) / 2
    
    return (list(best_h), best_c5, n_points)
