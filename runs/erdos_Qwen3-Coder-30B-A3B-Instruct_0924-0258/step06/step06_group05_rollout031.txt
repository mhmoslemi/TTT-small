import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Homotopy method to escape local optima
    max_homotopy_steps = 100
    homotopy_steps = 0
    
    h_current = best_h.copy()
    current_c5 = best_c5
    
    # Adaptive homotopy parameters
    t_min, t_max = 0.0, 1.0
    t = t_min
    
    try:
        while t < t_max and homotopy_steps < max_homotopy_steps:
            if time() - start_time > budget_s * 0.95:
                break
                
            # Solve relaxed problem with homotopy parameter t
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Use original objective with weight t
            corr_terms = []
            for k in range(1, n_points):
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    corr_terms.append(corr_sum)
            
            obj_term = cp.max(cp.hstack(corr_terms)) if corr_terms else 0
            relaxed_obj = t * obj_term + (1 - t) * cp.sum(cp.square(h_opt - 0.5))
            
            prob = cp.Problem(cp.Minimize(relaxed_obj), constraints)
            
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=300)
                
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    h_feasible = np.clip(h_feasible, 0, 1)
                    h_feasible = h_feasible * (n_points / 2) / np.sum(h_feasible)
                    
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    
                    if c5_feasible < current_c5:
                        h_current = h_feasible.copy()
                        current_c5 = c5_feasible
                        
                        if c5_feasible < best_c5:
                            best_h = h_feasible.copy()
                            best_c5 = c5_feasible
                            
                    # Adaptive homotopy step size
                    t = min(t_max, t + 0.01 * (1 - t))
                    homotopy_steps += 1
                else:
                    # Fallback to best solution if optimization fails
                    t = min(t_max, t + 0.01)
                    homotopy_steps += 1
                    
            except:
                # Fallback
                t = min(t_max, t + 0.01)
                homotopy_steps += 1
                
    except Exception as e:
        pass
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Ensure final solution has correct constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
