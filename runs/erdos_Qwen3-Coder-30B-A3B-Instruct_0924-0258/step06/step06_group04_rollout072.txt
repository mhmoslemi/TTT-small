import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive Stochastic Descent with better hyperparameters
    max_iter = 1000
    step_size = 0.1
    decay_rate = 0.995
    min_step = 0.001
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb
        noise = np.random.normal(0, step_size, n_points)
        h_new = best_h + noise
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        
        # Decay step size
        step_size = max(min_step, step_size * decay_rate)
    
    # Additional refinement using Nelder-Mead
    try:
        def objective(h_vec):
            h = np.clip(h_vec, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            return evaluate_erdos_solution(h, dx)
        
        # Optimize with constraints
        constraints = [
            {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2},
            {'type': 'ineq', 'fun': lambda h: h}, 
            {'type': 'ineq', 'fun': lambda h: 1 - h}
        ]
        
        result = minimize(
            objective, 
            best_h, 
            method='SLSQP',
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success:
            h_refined = np.clip(result.x, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
