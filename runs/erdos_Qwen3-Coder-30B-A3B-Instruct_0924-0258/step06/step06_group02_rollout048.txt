import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        def halton_sequence(n, base=2):
            sequence = []
            for i in range(n):
                f = 1
                r = 0
                while i > 0:
                    f /= base
                    r += (i % base) * f
                    i //= base
                sequence.append(r)
            return np.array(sequence)
        
        h_start = halton_sequence(n_points)
        h_start = (h_start - h_start.min()) / (h_start.max() - h_start.min())
        h_start = h_start * 0.5  # Scale to average 0.5
        h_start = h_start * (n_points/2) / np.sum(h_start)  # Ensure sum h = n_points/2
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive step size
    iterations = 10000
    learning_rate = 0.01
    
    for i in range(iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute gradient more carefully
        gradient = np.zeros(n_points)

            # For each shift k, add contribution to gradient
            if k < n_points:
                # Gradient contribution from h[i] due to overlap with h[i+k]
                gradient[:n_points-k] += (1 - best_h[k:])
                # Gradient contribution from h[i+k] due to overlap with h[i]
                gradient[k:] -= best_h[:n_points-k]
        
        # Update step with learning rate
        new_h = best_h - learning_rate * gradient
        
        # Apply constraints
        new_h = np.clip(new_h, 0, 1)  # Keep in [0,1]

        # Decrease learning rate over time
        learning_rate *= 0.9995
    
    # Final refinement with a simple local search
    for _ in range(100):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Small random perturbations
        pert = np.random.normal(0, 0.001, n_points)

        if test_c5 < best_c5:

            best_c5 = test_c5
