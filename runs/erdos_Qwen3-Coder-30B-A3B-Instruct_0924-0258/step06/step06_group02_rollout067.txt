import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def check_feasibility(C_candidate, h_start, n, dx):
    # Randomly sample shifts
    k_samples = np.random.choice(range(1, n), size=max(1, int(n * 0.2)), replace=False)
    
    # Define decision variables
    h_opt = cp.Variable(n)
    
    # Integral constraint
    constraints = [cp.sum(h_opt) == n / 2]
    
    # Bounds
    constraints += [0 <= h_opt, h_opt <= 1]
    
    # Overlap constraints for sampled shifts
    for k in k_samples:
        if k < n:
            corr_sum = cp.sum(cp.multiply(h_opt[:n-k], 1 - h_opt[k:]))
        else:
            corr_sum = cp.sum(cp.multiply(h_opt[n-k:], 1 - h_opt[:k]))
        constraints.append(corr_sum <= C_candidate * dx)
    
    # Formulate and solve problem
    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=cp.ECOS, eps=1e-6)
        return prob.status == cp.OPTIMAL
    except:
        return False

    # Binary search for tight bound
    low = 0.38
    high = best_c5

        if time() - start_time > budget_s * 0.8:
            break

            high = mid

    # Final refinement using all constraints but limited solver
    try:

        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_final = h_opt.value
            c5_final = evaluate_erdos_solution(h_final, dx)
