import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def check_feasibility(C_candidate, h_init, n, dx):
    # Randomly sample shifts
    k_samples = np.random.choice(range(1, n), size=max(1, int(n * 0.2)), replace=False)
    
    # Define decision variables
    h_opt = cp.Variable(n)
    
    # Integral constraint
    constraints = [cp.sum(h_opt) == n / 2]
    
    # Bounds
    constraints += [0 <= h_opt, h_opt <= 1]
    
    # Overlap constraints for sampled shifts
    for k in k_samples:
        if k < n:
            corr_sum = cp.sum(cp.multiply(h_opt[:n-k], 1 - h_opt[k:]))
        else:
            corr_sum = cp.sum(cp.multiply(h_opt[k-n+1:], 1 - h_opt[:n-(k-n+1)]))
        constraints.append(corr_sum <= C_candidate / dx)
    
    # Formulate and solve problem
    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=cp.ECOS, eps=1e-6)
        return prob.status == cp.OPTIMAL
    except:
        return False

def run(seed=42, budget_s=600, **kwargs):

        n_points = len(h_start)

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search in [0.38, 0.38085]
    low = 0.38
    high = 0.38085
    iterations = 20
    
    for _ in range(iterations):
        if time() - start_time > budget_s * 0.85:
            break
            
        mid = (low + high) / 2
        feasible = check_feasibility(mid, best_h, n_points, dx)
        
        if feasible:
            best_c5 = mid
            high = mid
        else:
            low = mid
            
        # Check if we're getting close to the target
        if high - low < 1e-6:
            break
    
    # Final refinement with all constraints using the best C5 found
    try:
        h_final = cp.Variable(n_points)
        constraints = [
            cp.sum(h_final) == n_points / 2.0,
            h_final >= 0,
            h_final <= 1
        ]
        
        # All overlap constraints
        for k in range(1, n_points):
            corr_sum = cp.sum(cp.multiply(h_final[:n_points-k], 1 - h_final[k:]))
            constraints.append(corr_sum <= best_c5 / dx)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_final.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Deep polishing with simulated annealing
    try:
        # Clone best solution
        h_current = best_h.copy()
        c5_current = best_c5
        temp = 1.0
        decay = 0.99
        steps = 1000
        
        for step in range(steps):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Random perturbation
            noise = np.random.normal(0, 0.01 * temp, n_points)
            h_new = h_current + noise

            # Project to correct integral
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)

            temp *= decay
            
        best_h = h_current
        best_c5 = c5_current
    except Exception:
        pass
