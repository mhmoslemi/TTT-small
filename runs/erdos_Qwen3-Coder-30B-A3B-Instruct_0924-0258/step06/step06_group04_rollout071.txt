import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Precompute correlation indices for efficiency
    corr_indices = []
    for k in range(1, n_points):
        if k <= n_points // 2:
            corr_indices.append((slice(None, n_points-k), slice(k, None)))
        else:
            corr_indices.append((slice(k-n_points+1, None), slice(None, n_points-(k-n_points+1))))
    
    # Optimization using gradient-based method with constraints
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_integral(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy minimize with trust-constr for better convergence
    constraints = [{'type': 'eq', 'fun': constraint_integral}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Adaptive optimization approach with restarts
    max_iter = 1000
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try several restarts with random perturbations
        for _ in range(3):
            # Random initial perturbation
            perturbation = np.random.normal(0, 0.05, n_points)
            h_trial = best_h + perturbation
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points/2) / np.sum(h_trial)
            
            try:
                res = minimize(
                    objective, 
                    h_trial,
                    method='trust-constr',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 50, 'verbose': 0}
                )
                if res.success:
                    c5_trial = objective(res.x)
                    if c5_trial < best_c5:
                        best_h = res.x
                        best_c5 = c5_trial
            except:
                continue
    
    # Final refinement with CVXPY for better constraint satisfaction
    try:
        h_opt = cp.Variable(n_points)
        constraints_cp = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Global constraint with reduced set
        full_corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_correlation = np.max(full_corr) * dx
        constraints_cp.append(cp.max(cp.multiply(h_opt, 1 - h_opt)) <= max_correlation + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints_cp)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final check and normalize
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
