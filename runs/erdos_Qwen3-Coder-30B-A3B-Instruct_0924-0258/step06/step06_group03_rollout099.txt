import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Initialize with Hamming window
        n_points = 35
        h_start = 0.54 + 0.46 * np.cos(np.pi * (np.arange(n_points) / n_points - 0.5) * 2)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Projected gradient descent
    max_iter = 1000
    learning_rate = 0.01
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Find the maximum correlation lag
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Approximated gradient using finite differences
            h_plus = best_h.copy()
            h_plus[i] += 1e-5
            if h_plus[i] <= 1:
                c_plus = evaluate_erdos_solution(h_plus, dx)
            else:
                c_plus = np.inf
            
            h_minus = best_h.copy()
            h_minus[i] -= 1e-5
            if h_minus[i] >= 0:
                c_minus = evaluate_erdos_solution(h_minus, dx)
            else:
                c_minus = np.inf
            
            if np.isfinite(c_plus) and np.isfinite(c_minus):
                gradient[i] = (c_plus - c_minus) / (2e-5)
            else:
                gradient[i] = 0
                
        # Update with learning rate
        new_h = best_h - learning_rate * gradient
        
        # Project new_h back to feasible region
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
