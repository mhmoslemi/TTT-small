import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    inertia_decay = 0.995
    temperature = 1.0
    temp_decay = 0.995
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_c5.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # PSO iterations
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update inertia
        w = max(0.4, w * inertia_decay)
        
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) +
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Repair solution to maintain sum constraint
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            current_c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = current_c5
            
            # Simulated annealing
            if current_c5 > gbest_c5:
                prob_accept = np.exp(-(current_c5 - gbest_c5) / temperature)
                if np.random.random() < prob_accept:
                    pbest[i] = particles[i].copy()
                    pbest_c5[i] = current_c5
                    
                    if current_c5 < gbest_c5:
                        gbest = particles[i].copy()
                        gbest_c5 = current_c5
        
        # Annealing temperature decay
        temperature *= temp_decay
        
        # Periodic convex relaxation
        if iter_num % 50 == 0 and iter_num > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample k values for constraints
                k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)  # to index space
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= gbest_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < gbest_c5:
                        gbest = h_feasible
                        gbest_c5 = c5_feasible
            except Exception:
                pass
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, gbest, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < gbest_c5:
                best_h = h_refined
                best_c5 = c5_refined
        else:
            best_h = gbest.copy()
            best_c5 = gbest_c5
    except Exception:
        best_h = gbest.copy()
        best_c5 = gbest_c5
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
