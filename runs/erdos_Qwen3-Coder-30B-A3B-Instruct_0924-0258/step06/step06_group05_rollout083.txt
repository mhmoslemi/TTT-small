import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Homotopy strategy: gradually deform the problem
    max_homotopy_steps = 100
    t_values = np.linspace(0, 1, max_homotopy_steps)
    
    h_current = h_start.copy()
    c5_current = best_c5
    
    for t in t_values:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convex relaxation: minimize t*C5 + (1-t)*||h - h_target||^2
        h_target = np.ones(n_points) * 0.5  # uniform target initially
        # Update target based on current h to make it more adaptive
        if t > 0.1:  # Only change target when we're past the initial stage
            h_target = h_current.copy()
        h_target = h_target * (n_points / 2) / np.sum(h_target)  # renormalize
        
        try:
            h_opt = cp.Variable(n_points)
            # Regularization term: (1-t) * ||h - h_target||^2
            reg_term = (1 - t) * cp.sum_squares(h_opt - h_target)
            # Objective: t * C5 + regularization
            # We'll linearize C5 as constraint for the QP solver
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Sample constraints to reduce density
            num_k_sample = min(50, n_points // 2)
            k_sample = np.random.choice(n_points-1, num_k_sample, replace=False)
            k_sample = k_sample + 1   # Valid k values in [1, n_points-1]
            # Add constraints on overlaps
            for k in k_sample:
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    # Introduce a proxy to represent overlap constraint
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)
            
            # Goal is to minimize t*C5 + (1-t)*||h - h_target||^2, but without 
            # explicit objective in convex form, so use QP-style of replacing
            # the C5 part with a constraint optimization variant
            # This is approximate in the full implementation
            
            prob = cp.Problem(cp.Minimize(reg_term), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                h_feasible = np.clip(h_feasible, 0, 1)
                h_feasible = h_feasible * (n_points / 2) / np.sum(h_feasible)
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < c5_current:
                    h_current = h_feasible.copy()
                    c5_current = c5_feasible
                    
                if c5_feasible < best_c5:
                    best_h = h_feasible.copy()
                    best_c5 = c5_feasible
            else:
                # Fallback to previous if optimization fails
                pass
                
        except Exception:
            # Fallback to previous h if QP fails
            pass

    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try multiple random restarts
        for _ in range(10):
            if time() - start_time > budget_s * 0.95:
                break
            h_random = h_current + np.random.normal(0, 0.05, n_points)
            h_random = np.clip(h_random, 0, 1)
            h_random = h_random * (n_points / 2) / np.sum(h_random)
            res = minimize(objective, h_random, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
