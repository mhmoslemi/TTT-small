import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a triangular window as starting point
        h_start = 1 - np.abs(np.arange(n_points) / (n_points - 1) - 0.5) * 2
        # Perturb slightly
        h_start += np.random.normal(0, 0.02, n_points)
        # Ensure it satisfies constraint sum(h) = n_points/2
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Number of DE iterations
    max_iter = 300
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points/2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # DE parameters
    popsize = 50
    mutation = 0.7
    crossover = 0.9
    
    # Run DE
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Use differential evolution for global search
        de_result = differential_evolution(
            objective, 
            bounds, 
            strategy='best1bin', 
            maxiter=1, 
            popsize=popsize,
            mutation=mutation,
            crossover=crossover,
            seed=seed + i,
            disp=False
        )
        
        if de_result.success:
            # Ensure constraint satisfaction
            h_candidate = np.clip(de_result.x, 0, 1)
            h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
            
            # Evaluate this candidate
            c5_candidate = objective(h_candidate)
            
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
                
    # Refinement with local optimization
    top_candidates = []
    
    # Collect top candidates from the current best
    for _ in range(10):
        h_trial = np.clip(best_h + np.random.normal(0, 0.02, n_points), 0, 1)
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        top_candidates.append(h_trial)
    
    # Use L-BFGS-B for local refinement
    for h_trial in top_candidates:
        try:
            res = minimize(
                objective,
                h_trial,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-10}
            )
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
