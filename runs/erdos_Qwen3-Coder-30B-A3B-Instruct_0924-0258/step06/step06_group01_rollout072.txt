import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.random.rand(n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    Tf = 1e-5
    alpha = 0.999
    iterations_per_T = 100
    T = T0
    
    h_current = h_start.copy()
    current_c5 = best_c5
    
    while T > Tf and (time() - start_time) < budget_s * 0.95:
        for _ in range(iterations_per_T):
            # Create candidate solution
            h_candidate = h_current.copy()
            
            # Perturb two random indices
            i, j = random.sample(range(n_points), 2)
            delta = random.gauss(0, 0.05)
            
            # Update both indices
            h_candidate[i] += delta
            h_candidate[j] -= delta
            
            # Ensure sum constraint
            sum_h = np.sum(h_candidate)
            scale = (n_points/2) / sum_h
            h_candidate = h_candidate * scale
            
            # Clamp to [0,1]
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Evaluate candidate
            candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept or reject
            delta_c5 = candidate_c5 - current_c5
            if delta_c5 < 0 or np.exp(-delta_c5 / T) > random.random():
                h_current = h_candidate.copy()
                current_c5 = candidate_c5
                
            # Update best
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
        
        # Decrease temperature
        T *= alpha
    
    # Final polishing with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
