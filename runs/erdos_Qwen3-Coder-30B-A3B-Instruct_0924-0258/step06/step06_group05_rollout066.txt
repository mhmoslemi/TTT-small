import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Stochastic gradient descent with adaptive step sizes
    max_iter = 5000
    step_size = 0.05
    patience = 0
    max_patience = 100
    
    for idx in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient direction for maximum overlap
        corr_vals = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr_vals)
        max_val = corr_vals[k_max]
        
        # Construct gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                gradient[i] += (1 - best_h[i + k_max])
            if i - k_max >= 0:
                gradient[i] -= best_h[i - k_max]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm
        
        # Line search to find best step
        best_step = 0
        best_val = max_val
        for alpha in np.logspace(-4, 0, 20):
            h_new = best_h - alpha * step_size * gradient
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            new_val = evaluate_erdos_solution(h_new, dx)
            if new_val < best_val:
                best_val = new_val
                best_step = alpha
        
        # Apply step
        if best_step > 0:
            h_new = best_h - best_step * step_size * gradient
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            best_h = h_new
            best_c5 = best_val
            patience = 0
        else:
            patience += 1
        
        # Random perturbation
        if idx % 50 == 0 and patience > 10:
            noise = np.random.normal(0, step_size * 0.1, n_points)
            h_new = best_h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            new_val = evaluate_erdos_solution(h_new, dx)
            if new_val < best_c5:
                best_h = h_new
                best_c5 = new_val
                patience = 0
        
        # Reduce step size if no improvement for a while
        if patience > 50:
            step_size *= 0.95
            patience = 0
            
        # Early stopping if no improvement
        if patience > max_patience:
            break
            
    # Convex relaxation refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add a few constraint rows based on sample lags
        lags = np.arange(1, min(15, n_points))
        for lag in lags:
            if lag < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-lag], 1 - h_opt[lag:]))
                # This will be used as a constraint on the actual maximum
                # Add a constraint with margin to allow small violations
                constraints.append(corr_sum <= best_c5 * dx + 1e-5)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-7, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
        
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
