import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 200
    elite_size = population_size // 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    
    # Add initial solution
    pop_h = h_start.copy()
    pop_h = np.clip(pop_h, 0, 1)
    pop_h = pop_h * (n_points/2) / np.sum(pop_h)
    population.append(pop_h)
    
    # Add noise perturbations
    for _ in range(population_size - 1):
        noise = np.random.normal(0, 0.1, n_points)
        h_perturbed = h_start + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        population.append(h_perturbed)
    
    # Helper for constraint satisfaction
    def satisfy_constraints(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Main GA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness (lower is better)
        fitness_scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness_scores.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if fitness_scores[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness_scores[0]
        
        # Start new population with elite
        new_population = population[:elite_size]
        
        # Tournament selection + crossover + mutation
        while len(new_population) < population_size:
            # Tournament selection
            tournament_size = 5
            selected_indices = []
            for _ in range(2):
                tournament = random.sample(range(len(population)), tournament_size)
                winner_idx = min(tournament, key=lambda i: fitness_scores[i])
                selected_indices.append(winner_idx)
            
            parent1 = population[selected_indices[0]]
            parent2 = population[selected_indices[1]]
            
            # Crossover
            if random.random() < crossover_rate:
                crossover_point = random.randint(0, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                if random.random() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.05)
            
            child1 = satisfy_constraints(child1)
            child2 = satisfy_constraints(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
    
    # Local refinement on top candidates
    for i in range(elite_size):
        h_candidate = population[i].copy()
        h_refined = h_candidate.copy()
        h_refined = satisfy_constraints(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    # Final evaluation
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
