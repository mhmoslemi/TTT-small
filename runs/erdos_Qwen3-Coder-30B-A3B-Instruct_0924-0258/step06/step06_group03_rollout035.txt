import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create triangular window as initial guess
        h_start = 1 - np.abs(np.arange(n_points) / (n_points-1) - 0.5) * 2
        # Perturb with small random noise
        h_start += np.random.normal(0, 0.01, n_points)
        h_start = np.clip(h_start, 0, 1)
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use differential evolution with custom bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Define the constraint manually for DE
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # We'll use a penalty method in DE's objective
    def penalty_objective(h):
        penalty = 0
        sum_violation = abs(np.sum(h) - n_points / 2.0)
        if sum_violation > 1e-6:
            penalty += 1000 * sum_violation
            
        return objective(h) + penalty
    
    # DE parameters with early termination
    maxiter = 300
    popsize = 50
    
    # Use DE for global search
    for iter_num in range(maxiter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Run DE with bounds and a fixed number of iterations
        result = differential_evolution(
            penalty_objective,
            bounds,
            maxiter=10,
            popsize=popsize,
            seed=seed + iter_num,
            disp=False
        )
        
        if result.success:
            h_candidate = result.x
            # Ensure it meets constraints
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
            
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    
    # Local refinement with L-BFGS-B
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    try:
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try 10 random points for local refinement
        for _ in range(10):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Start with current best and add small random noise
            h_start_refine = best_h + np.random.normal(0, 0.005, n_points)
            h_start_refine = np.clip(h_start_refine, 0, 1)
            h_start_refine = h_start_refine * (n_points/2) / np.sum(h_start_refine)
            
            res = minimize(objective, h_start_refine, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons,

                h_refined = h_refined * (n_points/2) / np.sum(h_refined)

                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
