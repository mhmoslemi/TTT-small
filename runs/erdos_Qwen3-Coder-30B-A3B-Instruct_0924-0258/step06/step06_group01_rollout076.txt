def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points/2 and 0 <= h[i] <= 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Simple rescaling approach
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * (target_sum / n_points)
    
    # Ensure bounds still satisfied
    h = np.clip(h, 0, 1)
    
    # Fine tune to exactly match target sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argsort(h)
        for i in range(len(idx)):
            j = idx[i]
            if h[j] + diff <= 1:
                h[j] += diff
                break
            else:
                diff -= (1 - h[j])
                h[j] = 1
                if abs(diff) < 1e-10:
                    break
        else:
            # If still need to adjust, make some small changes somewhere
            for i in range(len(h)):
                if h[i] - diff >= 0:
                    h[i] -= diff
                    break
                else:
                    diff += h[i]
                    h[i] = 0
    return h
