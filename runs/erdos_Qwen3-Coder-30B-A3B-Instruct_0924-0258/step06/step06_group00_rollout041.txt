import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 200
    elite_size = population_size // 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    def normalize(h):
        return h * (n_points/2) / np.sum(h)
    
    def create_individual():
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.1, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        return normalize(h)
    
    population = [create_individual() for _ in range(population_size)]
    population_fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
    
    # GA main loop
    no_improvement_count = 0
    generation = 0
    
    while time() - start_time < budget_s * 0.95 and generation < generations:
        # Sort population by fitness
        sorted_indices = np.argsort(population_fitness)
        population = [population[i] for i in sorted_indices]
        population_fitness = [population_fitness[i] for i in sorted_indices]
        
        # Check if we improved
        if population_fitness[0] < best_c5:
            best_c5 = population_fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count > 50:
            break
            
        # Select top individuals
        elite = population[:elite_size]
        
        # Create new generation
        new_population = elite[:]
        
        while len(new_population) < population_size:
            # Tournament selection
            tournament_size = 5
            parent1_idx = min(random.sample(range(population_size), tournament_size),
                              key=lambda i: population_fitness[i])
            parent2_idx = min(random.sample(range(population_size), tournament_size),
                              key=lambda i: population_fitness[i])
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if random.random() < crossover_rate:
                crossover_point = random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                    child2[i] += np.random.normal(0, 0.05)
            
            # Bounds and normalization
            child1 = np.clip(child1, 0, 1)
            child1 = normalize(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = normalize(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:population_size]
        population_fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
        
        generation += 1
    
    # Local refinement on top individuals
    top_indices = np.argsort(population_fitness)[:10]
    for idx in top_indices:
        h_candidate = population[idx].copy()
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_candidate, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            continue
    
    return (list(best_h), best_c5, n_points)
