```python
import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def feasible_h(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points/2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution search with different intensities
    resolutions = [35, 40, 50, 60, 80, 100]
    restarts_per_resolution = [20, 15, 10, 8, 5, 3]
    noise_levels = [0.03, 0.025, 0.02, 0.015, 0.01, 0.005]
    
    for res_idx, res in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = feasible_h(h_current, n_points)
        
        max_restarts = restarts_per_resolution[res_idx]
        noise_level = noise_levels[res_idx]
        
        # Predefined structured patterns as additional restarts
        patterns = [
            np.concatenate([np.ones(n_points//3), np.zeros(n_points - n_points//3)]),
            np.concatenate([np.zeros(n_points//3), np.ones(n_points - n_points//3)]),
            np.ones(n_points) * 0.5,
            np.concatenate([np.ones(n_points//2), np.zeros(n_points - n_points//2)])
        ]
        
        best_in_resolution = float('inf')
        h_best_in_resolution = h_current.copy()
        
        for restart in range(max_restarts):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Choose restart strategy: structured pattern or perturbation
            if restart % 3 == 0 and restart > 0:
                # Use structured pattern
                h_restart = patterns[restart % len(patterns)]
            else:
                # Random perturbation
                h_restart = h_current + np.random.normal(0, noise_level, n_points)
            
            h_restart = feasible_h(h_restart, n_points)
            
            # SLSQP local optimization
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 500, 'ftol': 1e-9})
                
                if res.success:
                    h_refined = res.x
                    h_refined = feasible_h(h_refined, n_points)
                    c5_refined = objective(h_refined)
                    if c5_refined < best_in_resolution:
                        best_in_resolution = c5_refined
                        h_best_in_resolution = h_refined
            except Exception:
                # If optimization fails, try another approach
                try:
                    h_restart = h_restart + np.random.normal(0, noise_level/2, n_points)
                    h_restart = feasible_h(h_restart, n_points)
                except:
                    pass
        
        # Update best over all resolutions so far
        if best_in_resolution < best_c5:
            best_c5 = best_in_resolution
            best_h = h_best_in_resolution.copy()
        
        h_start = h_best_in_resolution.copy()
    
    # Final aggressive refinement
    try:
        h_final = feasible_h(best_h.copy(), n_points)
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Multiple local SLSQP tries
        for _ in range(10):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_try = h_final + np.random.normal(0, 0.01, n_points)
            h_try = feasible_h(h_try, n_points)
            
            res = minimize(objective, h_try, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-10})
            
            if res.success:
                h_refined = res.x
                h_refined = feasible_h(h_refined, n_points)
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
```
