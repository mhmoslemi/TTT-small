import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a hybrid approach: frequency domain smoothing + local optimization
    max_iter = 1000
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1
        
        # Create a smoother version via FFT-based filtering
        h_smooth = h_start.copy()
        h_smooth = np.clip(h_smooth, 0, 1)
        h_smooth = h_smooth * (n_points/2) / np.sum(h_smooth)
        
        # Apply frequency domain filtering to smooth the function
        # This can help reduce high frequency oscillations that contribute to overlap
        fft_h = np.fft.fft(h_smooth)
        # Apply a Hann window (low-pass filter) in frequency domain
        window = np.hanning(n_points)
        # For real signals, we should consider the symmetric nature
        window = np.concatenate([window[:n_points//2], window[:n_points//2][::-1]])
        fft_filtered = fft_h * window
        h_filtered = np.real(np.fft.ifft(fft_filtered))
        
        # Ensure the filtered function is within bounds
        h_filtered = np.clip(h_filtered, 0, 1)
        h_filtered = h_filtered * (n_points/2) / np.sum(h_filtered)
        
        # Perturb slightly to avoid local minima
        h_perturbed = h_filtered + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Local optimization on perturbed version
        try:
            def objective(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            def constraint_sum(h_vec):
                return np.sum(h_vec) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Try multiple local minimizers
            methods = ['SLSQP', 'L-BFGS-B']
            for method in methods:
                try:
                    res = minimize(objective, h_perturbed, 
                                   method=method, 
                                   bounds=bounds, 
                                   constraints=cons_sum, 
                                   options={'maxiter': 200, 'ftol': 1e-9})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                            break
                except Exception:
                    continue
        except Exception:
            pass
            
        # Set h_start for next iteration to be best found so far
        h_start = best_h.copy()
        
        # Periodically print progress
        if iter_count % 100 == 0:
            print(f"Iteration {iter_count}, C5 = {best_c5:.6f}")
    
    return (list(best_h), best_c5, n_points)
