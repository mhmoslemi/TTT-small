import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n):
    """Create a triangular window function."""
    x = np.arange(n)
    center = (n - 1) / 2
    triangle = 1 - np.abs(x - center) / center
    return np.maximum(triangle, 0)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = triangular_window(n_points)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # First improve with differential evolution
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint function for sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Set up constraints properly for differential evolution
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # DE optimization - use a smaller population to be fast
    result_de = differential_evolution(
        objective, 
        bounds, 
        constraints=constraints,
        seed=seed,
        maxiter=200,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        disp=False
    )
    
    # Extract best result from DE
    best_h = result_de.x.copy()
    best_c5 = result_de.fun
    
    # Refine with local optimization
    # Time remaining for local refinement with 100 iterations max
    time_remaining = budget_s - (time() - start_time)
    max_local_iter = min(100, max(10, int(time_remaining * 10)))
    
    try:
        # Use L-BFGS-B for better local refinement
        result_local = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': max_local_iter},
            tol=1e-8
        )
        if result_local.success:
            h_refined = result_local.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final validation and correction
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    # Ensure final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
