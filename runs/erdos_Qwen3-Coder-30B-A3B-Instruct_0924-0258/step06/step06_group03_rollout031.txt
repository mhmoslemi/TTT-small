def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx

def triangular_window(n_points):
    """Create a triangular window centered at n_points/2"""
    half = n_points // 2
    window = np.zeros(n_points)
    window[:half] = np.linspace(0, 1, half)
    window[half:] = np.linspace(1, 0, n_points - half)
    return window
