def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')

    return max_overlap

def project_onto_constraints(h, n_points):
    """Project h onto the feasible region: 0 <= h[i] <= 1, sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Scale to sum to n_points/2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones_like(h) * (n_points / 2) / n_points
    
    return h
