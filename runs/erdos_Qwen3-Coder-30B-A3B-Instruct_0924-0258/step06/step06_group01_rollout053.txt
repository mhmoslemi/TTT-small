import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points/2 and h[i] in [0,1]"""
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scale_factor = (n_points/2) / current_sum
        h = h * scale_factor
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with existing construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Better initialization with alternating pattern
        h_start = np.ones(n_points)
        h_start[::2] = 0.3
        h_start[1::2] = 0.7
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use CVXOPT solver for better numerical stability
    # Initial projection
    h_start = project_to_feasible(h_start, n_points)
    
    # Iterative improvement using convex relaxation approach
    max_iter = 500
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create a new candidate using optimization
        h_current = best_h.copy()
        h_new = h_current.copy()
        
        # We'll use a simple heuristic first - try local search around good points
        # Make small random modifications
        for _ in range(3):
            idx = np.random.randint(0, n_points)
            # Small perturbation
            h_new[idx] += np.random.normal(0, 0.02)
            
        # Project to feasible region
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        # Regular SLSQP optimization every few iterations
        if iteration % 10 == 0:
            try:
                cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(
                    lambda x: evaluate_erdos_solution(x, dx),
                    best_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons_sum,
                    options={'ftol': 1e-9, 'maxiter': 100}
                )
                if res.success:
                    h_ref = res.x
                    h_ref = project_to_feasible(h_ref, n_points)
                    new_c5 = evaluate_erdos_solution(h_ref, dx)
                    if new_c5 < best_c5:
                        best_h = h_ref
                        best_c5 = new_c5
            except Exception:
                pass
    
    # Final check
    # Try to optimize to get better solution
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        if res.success:
            h_final = res.x
            h_final = project_to_feasible(h_final, n_points)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
