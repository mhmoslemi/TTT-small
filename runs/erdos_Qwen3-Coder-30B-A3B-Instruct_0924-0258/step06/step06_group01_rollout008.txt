from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    z = h.copy()

        # For now, we will compute a local update and then do a more complete proximal 
        # that solves a similar quadratic problem with the constraints

        # Instead just compute a more careful projection:

            # Approximate the optimal overlap minimization in the ADMM ascent pass
