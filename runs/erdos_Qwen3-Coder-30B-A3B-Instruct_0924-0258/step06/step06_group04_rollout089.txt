import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Grid refinement parameters
    n_start = 10
    n_max = 35
    n_points = n_start
    dx = 2.0 / n_points
    
    # Best solution found
    best_h = None
    best_c5 = float('inf')
    
    while n_points <= n_max:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize h
        if 'initial_h_values' in globals() and n_points == 10:
            h = np.array(initial_h_values)
        else:
            # Sinusoidal initialization
            x = np.linspace(0, 2, n_points)
            h = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2.0 / n_points))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Optimization loop
        max_iter = 200
        step_size = 0.1
        temp = 0.1
        decay_rate = 0.99
        sa_rate = 0.1  # 10% of iterations use SA
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Gradient step (simplified steepest descent)
            # Compute gradient using finite differences
            h_new = h.copy()
            eps = 1e-4
            grad = np.zeros_like(h)
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                c5_minus = evaluate_erdos_solution(h_minus, dx)
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with gradient
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Apply simulated annealing with probability
            if np.random.random() < sa_rate:
                noise = np.random.normal(0, temp, n_points)
                h_candidate = h + noise
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                c5_current = evaluate_erdos_solution(h, dx)
                
                if c5_candidate < c5_current or np.random.random() < np.exp(-(c5_candidate - c5_current) / temp):
                    h_new = h_candidate
            
            # Evaluate new solution
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            
            h = h_new
            step_size = max(1e-6, step_size * decay_rate)
            temp *= decay_rate
        
        # Double resolution
        n_points *= 2
        dx = 2.0 / n_points
    
    # Final binary search refinement
    if best_h is not None:
        try:
            h_opt = cp.Variable(len(best_h))
            constraints = [
                cp.sum(h_opt) == len(best_h) / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible.copy()
                    best_c5 = c5_feasible
        except Exception:
            pass
    
    # Final check to make sure constraints are met
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (len(best_h) / 2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
    # If we didn't find a better solution, use default
    if best_h is None:
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
        else:
            n_points = 35
            best_h = np.ones(n_points) * 0.5
            dx = 2.0 / n_points
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, len(best_h))
