import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Grid refinement approach
    resolutions = [10, 20, 35]
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize h for this resolution
        if n_points == 10:
            h = np.ones(10) * 0.5
        elif n_points == 20:
            # Interpolate from previous solution
            if len(h_start) == 10:
                h = np.interp(np.linspace(0, 2, 20), np.linspace(0, 2, 10), h_start)
            else:
                h = np.ones(20) * 0.5
        else:  # n_points == 35
            # Interpolate from previous solution, replace with sinusoidal if needed
            if len(h_start) == 20:
                h = np.interp(np.linspace(0, 2, 35), np.linspace(0, 2, 20), h_start)
            elif len(h_start) == 10:
                h = np.interp(np.linspace(0, 2, 35), np.linspace(0, 2, 10), h_start)
            else:
                # Use sinusoidal pattern
                x = np.linspace(0, 2, 35)
                h = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / 35))
                h = np.clip(h, 0, 1)
                
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)

        # More intensive optimization for final resolution
        if n_points == 35:
            # Gradient-based optimization
            for i in range(100):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Perturb with noise and keep within feasible region
                noise = np.random.normal(0, 0.01, n_points)
                h_new = h + noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                h = h_new

    # Final binary search refinement
    try:
        # Build a more refined version using CVXPY to improve solution
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add some constraints involving correlation terms
        corr_constraint_limit = best_c5 * dx + 1e-6
        correlations = []
        for k in range(1, n_points):
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            correlations.append(corr_sum)
            constraints.append(corr_sum <= corr_constraint_limit)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final correction
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Ensure we don't exceed time limit
    while time() - start_time > budget_s * 0.95:
        break
    
    return (list(best_h), best_c5, n_points)
