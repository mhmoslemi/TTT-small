import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available, otherwise start with a periodic pattern
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create a piecewise constant function with alternating high/low values
        h_start = np.ones(n_points)
        half = n_points // 2
        h_start[:half] = 0.7
        h_start[half:] = 0.3
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution grid
    resolutions = [40, 45, 50, 55, 60]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Optimization with multiple local search strategies
        max_restarts = 50
        for i in range(max_restarts):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Random restart with different strategies
            strategy = i % 4
            if strategy == 0:
                h_restart = np.clip(h_current + np.random.normal(0, 0.05, n_points), 0, 1)
            elif strategy == 1:
                # Perturbation around median
                h_pert = np.random.uniform(0, 1, n_points)
                h_pert = np.sort(h_pert)
                h_restart = np.interp(np.linspace(0, 1, n_points), h_pert, h_pert)
            elif strategy == 2:
                # Sigmoid-based restart
                x = np.linspace(0, 2, n_points) - 1
                h_restart = 1 / (1 + np.exp(-5 * x))
                noise = np.random.normal(0, 0.03, n_points) 
                h_restart = np.clip(h_restart + noise, 0, 1)
            else: # strategy == 3
                # Step function restart
                h_restart = np.clip(h_current + np.random.normal(0, 0.1, n_points), 0, 1)
                
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            # SLSQP local optimization
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Try different optimization methods
            method_choices = ['Nelder-Mead', 'Powell']
            for method in method_choices:
                try:
                    res = minimize(objective, h_restart, method=method,
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 300, 'ftol': 1e-8})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                            break  # Found improvement, stop trying other methods
                except:
                    continue
                if time() - start_time > budget_s * 0.95:
                    break
    
    # Final SLSQP refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
