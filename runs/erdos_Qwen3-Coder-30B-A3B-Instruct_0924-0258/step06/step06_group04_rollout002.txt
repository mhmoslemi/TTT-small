import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a sinusoidal pattern
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)
        h_start = np.clip(h_start, 0, 1)
    
    # Normalize to sum to n_points / 2
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    max_iter = 1000
    step_size = 0.05
    min_step = 0.0001
    decay_rate = 0.995
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb with gradient-based noise
        gradient = np.zeros(n_points)
        h_eval = best_h.copy()
        h_eval = np.clip(h_eval, 0, 1)
        h_eval = h_eval * (n_points/2) / np.sum(h_eval)
        
        # Estimate gradient via finite differences
        eps = 1e-4
        for j in range(n_points):
            h_plus = h_eval.copy()
            h_plus[j] += eps
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            h_minus = h_eval.copy()
            h_minus[j] -= eps
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            
            gradient[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Apply gradient with adaptive step size
        h_new = best_h - step_size * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        
        # Decay step size
        step_size = max(min_step, step_size * decay_rate)
    
    # Final quadratic programming refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add overlap constraints at key lags
        lags_to_check = list(range(1, min(n_points//2 + 1, 15)))
        for k in lags_to_check:
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
                
    except Exception:
        pass
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
