import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a triangular window as starting point, with some noise
        h_start = 1 - np.abs(np.arange(n_points) / (n_points-1) - 0.5) * 2
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    DE_bounds = [(0, 1) for _ in range(n_points)]
    DE_popsize = 50
    DE_maxiter = 300
    DE_workers = 1
    
    # Run DE for initial optimization
    def de_objective(h):
        # Ensure the sum constraint
        h = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    # DE for exploration
    de_result = differential_evolution(
        de_objective,
        DE_bounds,
        popsize=DE_popsize,
        maxiter=DE_maxiter,
        seed=seed,
        workers=DE_workers,
        disp=False,
        atol=1e-8,
        tol=1e-8
    )
    
    # Check if DE found a better solution
    if de_result.success:
        h_de = de_result.x * (n_points/2) / np.sum(de_result.x)
        c5_de = evaluate_erdos_solution(h_de, dx)
        if c5_de < best_c5:
            best_h = h_de
            best_c5 = c5_de
    
    # Local refinement with L-BFGS-B on top candidates
    top_candidates = []
    # Generate diverse candidates near DE solution
    num_candidates = 10
    for _ in range(num_candidates):
        candidate = de_result.x + np.random.normal(0, 0.02, n_points)
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * (n_points/2) / np.sum(candidate)
        top_candidates.append(candidate)
    
    # Add the DE solution itself
    top_candidates.append(de_result.x * (n_points/2) / np.sum(de_result.x))
    top_candidates = [c for c in top_candidates if all(0<=x<=1 for x in c)]
    
    def refine_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def refine_constraint(h):
        return np.sum(h) - n_points/2.0
    
    cons = {'type': 'eq', 'fun': refine_constraint}
    bounds = [(0, 1) for _ in range(n_points)]

    # Refine each candidate
    for cand in top_candidates:
        if time() - start_time > budget_s * 0.8:
            break
        try:
            res = minimize(
                refine_objective,
                cand,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            continue
    
    return (list(best_h), best_c5, n_points)
