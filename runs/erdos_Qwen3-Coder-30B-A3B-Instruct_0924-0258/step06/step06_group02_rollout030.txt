import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        return h
    scale = n / (2 * sum_h)
    h *= scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_c5 = float(best_c5)
    
    # Simulated Annealing with focused perturbations
    T = 0.5
    cooling_rate = 0.95
    min_T = 1e-4
    max_iter = 2000
    perturb_frac = 0.15
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Calculate current overlap for all shifts
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        overlap_per_shift = corr[:n_points-1]
        
        # Focus on top percentage of problematic shifts
        num_perturb = max(1, int(len(overlap_per_shift) * perturb_frac))
        idxs = np.argsort(overlap_per_shift)[-num_perturb:]
        
        # Perturb selected elements
        h_new = best_h.copy()
        for idx in idxs:
            delta = np.random.uniform(-0.1, 0.1)
            h_new[idx] += delta
        h_new = project_feasible(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        elif np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Cooling schedule
        T = max(min_T, T * cooling_rate)
    
    # QCP Refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraints for all shifts
        # Optimized: only add non-redundant constraints.
        # Only check up to n_points//2 to avoid repetition.
        for k in range(1, n_points//2 + 1):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final refinement with local optimization
    try:
        # Use simple gradient descent starting from best solution
        h_current = best_h.copy()
        h_current = project_feasible(h_current, n_points)
        
        for _ in range(50):  # fixed number of steps
            if time() - start_time > budget_s * 0.95:
                break
            # Use a custom local update for constraint violation (not gradient)
            # Just ensure feasibility
            h_current = project_feasible(h_current, n_points)
            c5_current = evaluate_erdos_solution(h_current, dx)
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current

    except Exception:
        pass
    
    # Final validation and return
    try:
        h_final = project_feasible(best_h, n_points)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), float(best_c5), n_points)
