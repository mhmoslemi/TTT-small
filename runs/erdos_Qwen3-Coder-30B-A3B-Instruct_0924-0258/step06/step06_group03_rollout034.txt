import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n_points):
    """Create triangular window function centered at n_points/2"""
    half = n_points // 2
    if n_points % 2 == 0:
        # Even case
        window = np.concatenate([np.arange(1, half + 1), np.arange(half, 0, -1)])
    else:
        # Odd case
        window = np.concatenate([np.arange(1, half + 2), np.arange(half, 0, -1)])
    return window / np.sum(window) * (n_points / 2)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = triangular_window(n_points)
    
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Set DE bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define objective for DE
    def de_objective(h):
        # Normalize to satisfy constraint sum(h) = n_points/2
        h_norm = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h_norm, dx)
    
    # Perform differential evolution
    de_result = differential_evolution(
        de_objective,
        bounds,
        maxiter=300,
        popsize=50,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        disp=False,
        polish=False
    )
    
    # Final refinement with L-BFGS-B
    if de_result.success:
        h_de = de_result.x
        h_de = h_de * (n_points / 2) / np.sum(h_de)
        de_c5 = evaluate_erdos_solution(h_de, dx)
        
        if de_c5 < best_c5:
            best_h = h_de.copy()
            best_c5 = de_c5
    
    # Local refinement with L-BFGS-B for the best solution
    def lbfgs_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def lbfgs_constraint(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': lbfgs_constraint}
    bounds_lbfgs = [(0, 1) for _ in range(n_points)]
    
    # Try refining repeatedly with different initializations
    for _ in range(5):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Random perturbation around best solution
        h_init = best_h + np.random.normal(0, 0.01, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        try:
            res = minimize(
                lbfgs_objective,
                h_init,
                method='L-BFGS-B',
                bounds=bounds_lbfgs,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-10}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except:
            continue
    
    # Double-check final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
