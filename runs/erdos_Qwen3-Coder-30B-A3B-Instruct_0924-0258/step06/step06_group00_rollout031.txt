import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 500
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Start with initial solution
    particles.append(h_start.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Generate other particles
    for _ in range(n_particles - 1):
        h_new = h_start + np.random.normal(0, 0.1, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        particles.append(h_new)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Personal bests
    p_best = [p.copy() for p in particles]
    p_best_values = [evaluate_erdos_solution(p, dx) for p in particles]
    
    # Global best
    g_best_idx = np.argmin(p_best_values)
    g_best = p_best[g_best_idx].copy()
    g_best_value = p_best_values[g_best_idx]
    
    # PSO iteration
    stagnation_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (w * velocities[i] + 
                 c1 * r1 * (p_best[i] - particles[i]) + 
                 c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            new_pos = particles[i] + v
            
            # Enforce constraints
            new_pos = np.clip(new_pos, 0, 1)
            new_pos = new_pos * (n_points/2) / np.sum(new_pos)
            
            # Evaluate
            new_c5 = evaluate_erdos_solution(new_pos, dx)
            
            # Update personal best
            if new_c5 < p_best_values[i]:
                p_best[i] = new_pos.copy()
                p_best_values[i] = new_c5
            
            # Update global best
            if new_c5 < g_best_value:
                g_best = new_pos.copy()
                g_best_value = new_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
                
            # Update particle
            particles[i] = new_pos
            velocities[i] = v
            
        # Early termination based on stagnation
        if stagnation_count > 100:
            break
    
    # Final refinement with SLSQP
    try:
        h_final = g_best.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
