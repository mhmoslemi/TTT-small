import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with available starting solution
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search bounds
    c_low = 0.0
    c_high = 0.38085
    tolerance = 1e-5
    max_iterations = 1000
    
    # Solver settings
    max_iter = 500
    
    # Binary search loop
    iteration = 0
    while (c_high - c_low > tolerance) and (time() - start_time < budget_s * 0.95) and iteration < max_iterations:
        c_mid = (c_low + c_high) / 2.0
        
        # Create CVXPY variables
        h = cp.Variable(n_points)
        
        # Objective - dummy (we just want feasibility)
        objective = cp.Minimize(0)
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Add overlap constraints for all shifts
        for k in range(n_points):
            # Compute overlap for shift k
            overlap_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_sum += h[i] * (1 - h[j])
            
            # This needs to be <= c_mid * n_points/2  (scaled dx factor)
            constraints.append(overlap_sum <= c_mid * n_points / 2.0)
        
        # Formulate and solve problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, max_iters=max_iter, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_solution = h.value
                h_projected = np.clip(h_solution, 0, 1)
                h_projected = h_projected * (n_points / 2) / np.sum(h_projected)
                current_c5 = evaluate_erdos_solution(h_projected, dx)
                
                if current_c5 < best_c5:
                    best_h = h_projected.copy()
                    best_c5 = current_c5
                
                if current_c5 <= c_mid:
                    c_high = c_mid
                else:
                    c_low = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid
        
        iteration += 1
    
    # Final refinement with local optimization
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
