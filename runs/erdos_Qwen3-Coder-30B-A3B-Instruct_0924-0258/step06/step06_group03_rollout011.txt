import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with best construction or create a new one
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a triangular window-like structure with small random noise
        n_points = 35
        x = np.linspace(0, 1, n_points)
        h_start = 1 - np.abs(x - 0.5) * 2
        h_start += np.random.normal(0, 0.02, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # Step 1: Differential Evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    # DE optimization
    best_c5 = float('inf')
    best_h = h_start.copy()
    
    # Use differential evolution for global exploration
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            args=(),
            strategy='best1bin',
            maxiter=int(budget_s * 0.6 / 10),
            popsize=50,
            mutation=(0.5, 1.0),
            recombination=0.9,
            seed=seed,
            callback=lambda x, convergence: None,
            disp=False,
            polish=False
        )
        if de_result.success:
            h_de = de_result.x
            c5_de = objective(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    except:
        pass
    
    # Step 2: Local refinement using L-BFGS-B
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    try:
        # Project the DE result to satisfy constraints
        h_projected = project_h(best_h)
        c5_projected = objective(h_projected)
        if c5_projected < best_c5:
            best_h = h_projected
            best_c5 = c5_projected
            
        # Local refinement with constraints
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds_lbfgs = [(0, 1) for _ in range(n_points)]
        
        result = minimize(
            objective,
            h_projected,
            method='L-BFGS-B',
            bounds=bounds_lbfgs,
            constraints=cons,
            options={'maxiter': 200}
        )
        
        if result.success:
            h_refined = project_h(result.x)
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    # Final verification
    best_h = project_h(best_h)
    best_c5 = objective(best_h)
    
    return (list(best_h), best_c5, n_points)
