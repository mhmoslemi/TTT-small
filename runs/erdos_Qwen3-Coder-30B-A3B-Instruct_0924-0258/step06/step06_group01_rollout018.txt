import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the provided construction, or a uniform one
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    C_low = 0.0
    C_high = 0.38085
    tolerance = 1e-5
    max_iterations = 1000
    
    def is_feasible(C):
        """Check if there exists a feasible h for a given C."""
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add overlap constraints for all shifts
        for k in range(n_points):
            # Manually compute the correlation sum for shift k
            # h[i] * (1 - h[(i+k) % n_points])
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_vars[i] * (1 - h_vars[j])
            constraints.append(expr <= C * n_points / 2)
        
        # Dummy objective (not used for feasibility)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, max_iters=500, verbose=False)
            return prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INFEASIBLE
        except:
            return False
    
    def optimize_with_constraints(h_init):
        """Refine a solution using constrained optimization."""
        dx = 2.0 / n_points
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        try:
            res = minimize(
                lambda x: evaluate_erdos_solution(x, dx),
                h_init,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'ftol': 1e-8, 'maxiter': 200}
            )
            if res.success:
                return np.clip(res.x, 0, 1) * (n_points/2) / np.sum(res.x)
        except:
            pass
        return h_init
    
    # Binary search on the upper bound C_5
    iterations = 0
    while (C_high - C_low > tolerance) and (time() - start_time < budget_s * 0.95) and iterations < max_iterations:
        C_mid = (C_low + C_high) / 2
        if is_feasible(C_mid):
            C_high = C_mid
        else:
            C_low = C_mid
        iterations += 1
        
        # Every few iterations, try to improve using optimization
        if iterations % 10 == 0:
            # Reset with best solution so far
            h_test = optimize_with_constraints(best_h)
            c5_test = evaluate_erdos_solution(h_test, dx)
            if c5_test < best_c5:
                best_h = h_test
                best_c5 = c5_test
            
            # Early termination if we're close to target
            if C_high - C_low < 1e-4 and best_c5 < 0.3808:
                break
    
    # Refine the final solution
    final_h = optimize_with_constraints(best_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
