import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create periodic pattern
        h_start = np.zeros(n_points)
        ones_count = n_points // 2
        for i in range(ones_count):
            h_start[i * (n_points // ones_count)] = 1
        if n_points % 2 == 1:
            h_start[n_points//2] = 1
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search approach
    max_iter = 1000
    T = 1.0
    decay = 0.995
    
    # Bit swap sampling with simulated annealing
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor by swapping two random bits
        h_new = best_h.copy()
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
        
        # Ensure binary
        h_new = np.round(h_new).clip(0, 1)
        
        # Rebalance to sum constraint
        current_sum = np.sum(h_new)
        target_sum = n_points / 2.0
        if abs(current_sum - target_sum) > 1e-6:
            # Find indices to shift
            ones = np.where(h_new > 0.5)[0]
            zeros = np.where(h_new < 0.5)[0]
            if len(ones) > target_sum and len(zeros) > 0:
                # Shift one 1 to 0
                i1 = np.random.choice(ones)
                i2 = np.random.choice(zeros)
                h_new[i1] = 0
                h_new[i2] = 1
            elif len(ones) < target_sum and len(ones) > 0:
                # Shift one 0 to 1
                i1 = np.random.choice(zeros)
                i2 = np.random.choice(ones)
                h_new[i1] = 1
                h_new[i2] = 0

        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept with probability
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        T *= decay
        
        # Periodic QCP refinement
        if i % 100 == 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample k values for constraints
                k_vals = np.linspace(0.05, 1.95, min(15, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)  # to index space
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=300)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass

    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
