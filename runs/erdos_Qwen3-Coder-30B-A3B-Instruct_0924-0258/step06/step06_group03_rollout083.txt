import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 40
        # Create triangular window initialization
        h_start = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h_start[i] = i / mid
            h_start[n_points - 1 - i] = i / mid
        h_start[mid] = 1.0
        # Normalize to satisfy constraint
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds for differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define objective function for DE
    def de_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Function to enforce constraint sum(h) = n_points / 2
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Differential Evolution parameters
    de_maxiter = 300
    popsize = 50
    
    # DE with bounds
    try:
        res_de = differential_evolution(
            de_objective,
            bounds,
            maxiter=de_maxiter,
            popsize=popsize,
            seed=seed,
            disp=False,
            tol=1e-6
        )
        if res_de.success:
            h_de = normalize(res_de.x)
            c5_de = de_objective(h_de)
            if c5_de < best_c5:
                best_h = h_de.copy()
                best_c5 = c5_de
    except Exception:
        pass
    
    # Local refinement with L-BFGS-B
    try:
        # Use a subset of top individuals for local refinement
        refiner_count = min(10, popsize)
        candidates = []
        
        # Sample from the current best
        sample_size = min(100, popsize)
        for _ in range(sample_size):
            h_sample = h_start.copy()
            for i in range(n_points):
                if np.random.rand() < 0.2:  # Small mutation
                    h_sample[i] += np.random.normal(0, 0.01)
            h_sample = np.clip(h_sample, 0, 1)
            h_sample = normalize(h_sample)
            candidates.append(h_sample)
            
        # Evaluate candidates
        candidates_eval = []
        for h_test in candidates:
            c5_val = evaluate_erdos_solution(h_test, dx)
            candidates_eval.append((c5_val, h_test))
            
        candidates_eval.sort(key=lambda x: x[0])
        top_candidates = [h for _, h in candidates_eval[:refiner_count]]
        
        for h_candidate in top_candidates:
            # Local refinement with L-BFGS-B
            def objective(h_refined):
                return evaluate_erdos_solution(h_refined, dx)
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds_ref = [(0, 1) for _ in range(n_points)]
            
            res_ref = minimize(
                objective,
                h_candidate,
                method='L-BFGS-B',
                bounds=bounds_ref,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if res_ref.success:
                h_refined = normalize(res_ref.x)
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    
    except Exception:
        pass
    
    # Ensure final normalization and constraint checking
    best_h = normalize(best_h)
    if abs(np.sum(best_h) - n_points / 2.0) > 1e-6:
        best_h = normalize(best_h)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
