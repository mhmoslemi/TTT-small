import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Homotopy method approach
    max_homotopy_steps = 100
    t_min = 0.0
    t_max = 1.0
    
    # Initialize homotopy parameter
    t_vals = np.linspace(t_min, t_max, max_homotopy_steps)
    current_h = best_h.copy()
    current_c5 = best_c5
    
    for i, t in enumerate(t_vals):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Use a convex relaxation approach to solve for current t
        try:
            # Create variable
            h_opt = cp.Variable(n_points)
            # Objective: weighted combination of original objective and regularization
            # Approximate the maximum overlap with a smooth function
            # We approximate this as minimizing a sum of squared differences from uniform
            objective = cp.Minimize(
                t * cp.maximum(cp.Variable(1), 0) + 
                (1-t) * cp.sum_squares(h_opt - np.ones(n_points) * 0.5)
            )
            
            # Define constraints
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Solve for the current t value
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            
            if prob.status == cp.OPTIMAL:
                new_h = h_opt.value
                new_c5 = evaluate_erdos_solution(new_h, dx)
                if new_c5 < current_c5:
                    current_h = new_h.copy()
                    current_c5 = new_c5
                    
        except Exception:
            pass
        
        # Also try a few local search steps
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            # Stochastic perturbation
            noise = np.random.normal(0, 0.01, n_points)
            h_new = current_h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < current_c5:
                current_h = h_new.copy()
                current_c5 = c5_new
                
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
