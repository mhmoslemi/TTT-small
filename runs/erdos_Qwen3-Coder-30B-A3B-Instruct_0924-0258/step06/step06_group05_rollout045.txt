import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search with simulated annealing
    current_h = h_start.copy()
    current_c5 = best_c5
    
    iter_count = 0
    temp = 1.0
    temp_decay = 0.995
    max_iter = 1000
    
    def fast_evaluate(h_vec):
        # Optimized evaluation without overhead
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Bit-flip neighborhood search
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1
        # Generate candidate by flipping two random bits
        h_new = current_h.copy()
        i, j = np.random.randint(0, n_points, 2)
        if i != j:
            h_new[i] = 1 - h_new[i]
            h_new[j] = 1 - h_new[j]
        
        # Ensure sum constraint (exact count)
        ones_old = np.sum(current_h)
        ones_new = np.sum(h_new)
        if abs(ones_new - ones_old) > 0.5:
            # Need adjustment to fix sum
            if ones_new > ones_old:
                zeros = np.where(h_new == 0)[0]
                if len(zeros) > 0:
                    idx = np.random.choice(zeros)
                    h_new[idx] = 1
            else:
                ones = np.where(h_new == 1)[0]
                if len(ones) > 0:
                    idx = np.random.choice(ones)
                    h_new[idx] = 0
            
        c5_new = fast_evaluate(h_new)
        
        # Acceptance criterion
        if c5_new < current_c5 or np.random.rand() < np.exp(-(c5_new - current_c5) / temp):
            current_h = h_new.copy()
            current_c5 = c5_new
            
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
                
        temp *= temp_decay
        
        # Periodically refine with QCP
        if iter_count % 100 == 0 and time() - start_time < budget_s * 0.9:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample k values for constraints
                k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if 0 < k < n_points:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = fast_evaluate(h_feasible)
                    if c5_feasible < best_c5:
                        best_h = h_feasible.copy()
                        best_c5 = c5_feasible
                        
            except Exception:
                pass
    
    # Final optimization with scipy
    try:
        def objective(h):
            return fast_evaluate(h)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
