import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a smooth pattern
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing approach
    temperature = 1.0
    cooling_rate = 0.9995
    max_iterations = 10000
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Perturb the solution
        h_new = best_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05)
        h_new[idx] += delta
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain integral constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject the move
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / temperature):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Cool down
        temperature *= cooling_rate
    
    # Final refinement using gradient info
    try:
        # Gradient descent with step size based on current best
        learning_rate = 0.001
        for _ in range(1000):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute gradient numerically
            gradient = np.zeros(n_points)
            for i in range(n_points):
                grad_sum = 0
                for k in range(1, n_points):
                    if i + k < n_points:
                        grad_sum += (1 - best_h[i + k])
                    if i - k >= 0:
                        grad_sum -= best_h[i - k]
                gradient[i] = grad_sum
                
            # Update with gradient
            new_h = best_h - learning_rate * gradient
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points/2) / np.sum(new_h)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(new_h, dx)
            if c5_new < best_c5:
                best_h = new_h
                best_c5 = c5_new
            learning_rate *= 0.999
            
    except:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
