import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_initialization(n):
    """Create a triangular window function"""
    h = np.zeros(n)
    mid = n // 2
    for i in range(n):
        if i <= mid:
            h[i] = i / mid
        else:
            h[i] = (n - i) / (n - mid)
    return h
