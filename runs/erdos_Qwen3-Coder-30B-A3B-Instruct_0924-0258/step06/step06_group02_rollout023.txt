import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Generate a low-discrepancy sequence
        def halton_sequence(n, base=2):
            sequence = []
            for i in range(n):
                x = 0
                f = 1.0 / base
                while i > 0:
                    x += (i % base) * f
                    i //= base
                    f /= base
                sequence.append(x)
            return np.array(sequence)
        
        h_start = halton_sequence(n_points)
        h_start = (h_start - h_start.min()) / (h_start.max() - h_start.min())
        h_start = h_start * 0.5  # Scale to average ~0.5
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive learning rate
    max_iterations = 1000
    learning_rate = 0.01
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.90:
            break
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for k in range(1, n_points):
            # For each shift k, compute contribution to gradient
            if k < n_points:
                # h[i] appears in term h[i](1-h[i+k])
                # Gradient w.r.t. h[i] is (1-h[i+k]) 
                gradient[:n_points-k] += (1 - best_h[k:])
                # h[i] also appears in term h[i-k](1-h[i]) for i >= k
                # Gradient w.r.t. h[i] is -h[i-k]
                if k < n_points:
                    gradient[k:] -= best_h[:n_points-k]
        
        # Update with gradient
        new_h = best_h - learning_rate * gradient
        
        # Clip to feasible range and renormalize
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate new point
        c5_new = evaluate_erdos_solution(new_h, dx)
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
        
        # Update learning rate
        learning_rate *= 0.995
    
    # Final refinement with scipy minimize (if we can afford it)
    try:
        # We only execute full optimization if we still have time
        if time() - start_time < budget_s * 0.95:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            # Constraint for integral
            def constraint_integral(h):
                return np.sum(h) - n_points / 2
            
            cons = {'type': 'eq', 'fun': constraint_integral}
            bounds = [(0, 1) for _ in range(n_points)]
            
            result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                              options={'maxiter': 100, 'ftol': 1e-8}, tol=1e-8)
            
            if result.success:
                refined_h = result.x
                refined_h = np.clip(refined_h, 0, 1)
                refined_h = refined_h * (n_points/2) / np.sum(refined_h)
                c5_refined = evaluate_erdos_solution(refined_h, dx)
                if c5_refined < best_c5:
                    best_h = refined_h.copy()
                    best_c5 = c5_refined
    except:
        pass
    
    # Final safety check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
