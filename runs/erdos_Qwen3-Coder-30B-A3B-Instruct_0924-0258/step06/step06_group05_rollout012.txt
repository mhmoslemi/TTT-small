import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    inertia_decay = 0.995
    max_iter = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0.3, 0.7, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        c5 = evaluate_erdos_solution(h, dx)
        pbest.append(h.copy())
        pbest_c5.append(c5)
    
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # PSO iteration
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2), np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) +
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Repair to maintain constraint
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
                
                if c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = c5
        
        # Annealing: allow worse solutions occasionally
        if np.random.random() < np.exp(-gbest_c5 / (0.001 + iter_num / 100.0)):
            # perturb global best
            perturbed = gbest + np.random.normal(0, 0.02, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (n_points/2) / np.sum(perturbed)
            c5_new = evaluate_erdos_solution(perturbed, dx)
            if c5_new < gbest_c5:
                gbest = perturbed.copy()
                gbest_c5 = c5_new
        
        # Periodic QCP refinement every 50 iterations
        if iter_num % 50 == 0 and iter_num > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1,
                ]
                
                k_vals = np.linspace(0.2, 1.8, min(20, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= gbest_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_feasible = np.array(h_opt.value).flatten()
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < gbest_c5:
                        gbest = h_feasible
                        gbest_c5 = c5_feasible
            except Exception:
                pass
        
        # Decay inertia
        w *= inertia_decay
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, gbest, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < gbest_c5:
                best_h = h_refined
                best_c5 = c5_refined
        else:
            best_h = gbest
            best_c5 = gbest_c5
    except Exception:
        best_h = gbest
        best_c5 = gbest_c5
        
    return (list(best_h), best_c5, n_points)
