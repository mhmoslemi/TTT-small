def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    def create_initial_population(pop_size, n):
        population = []
        # Include base solution
        h_base = h_start.copy()
        h_base = h_base * (n/2) / np.sum(h_base)
        population.append(h_base)

        for i in range(pop_size - 1):
            if i % 3 == 0:
                # Random noise
                h = h_base + np.random.normal(0, 0.05, n)
            elif i % 3 == 1:
                # Block pattern
                h = np.ones(n) * 0.5
                half_n = n // 2
                h[:half_n] = 0.8
                h[half_n:] = 0.2
            else:
                # Alternating pattern
                h = np.ones(n) * 0.3
                h[::2] = 0.7
                
            # Normalize and clip
            h = np.clip(h, 0, 1)
            h = h * (n/2) / np.sum(h)
            population.append(h)
            
        return population

        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    population_size = 100
    max_iter = 500
    mutation_factor = 0.5
    crossover_prob = 0.7

        if time() - start_time > budget_s * 0.95:

                mutant = population[idx1] + mutation_factor * (population[idx2] - population[idx3])

                trial = np.copy(population[i])

                        trial[j] = mutant[j]

                        best_h = h_final

                break
    
    return (list(best_h), best_c5, n_points)
