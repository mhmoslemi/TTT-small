import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create a triangular window initialization
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1) * 2)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    bounds = [(0, 1) for _ in range(n_points)]
    maxiter = 300
    popsize = 50
    mutation = 0.7
    recombination = 0.9
    
    # DE optimization
    def de_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Run DE
    de_result = differential_evolution(
        de_objective, 
        bounds, 
        args=(),
        maxiter=maxiter,
        popsize=popsize,
        mutation=mutation,
        recombination=recombination,
        seed=seed,
        callback=lambda x, convergence: None,
        polish=False,
        disp=False
    )
    
    if de_result.success:
        h_de = de_result.x
        c5_de = de_objective(h_de)
        if c5_de < best_c5:
            best_h = h_de
            best_c5 = c5_de
    
    # Local refinement with L-BFGS-B
    try:
        def lbfgsb_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        # Ensure initial feasibility
        h_refined = np.clip(best_h, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        # L-BFGS-B refinement
        lbfgsb_result = minimize(
            lbfgsb_objective,
            h_refined,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10},
            callback=lambda x: None
        )
        
        if lbfgsb_result.success:
            h_final = lbfgsb_result.x
            c5_final = lbfgsb_objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
                
    except Exception as e:
        pass
    
    # Final guard against numerical issues
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
