import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Set up PSO parameters
    n_particles = 50
    max_iter = 1000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    inertia_decay = 0.995
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.rand(n_points) * 0.5 + 0.25
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        c5 = evaluate_erdos_solution(h, dx)
        pbest.append(h.copy())
        pbest_c5.append(c5)
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # PSO iterations
    temp = 1.0
    temp_decay = 0.995
    
    for iter_idx in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Update inertia
        w = max(0.3, w * inertia_decay)
        
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Repair constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
                
                # Update global best
                if c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = c5
        
        # Simulated annealing step
        if np.random.random() < 0.01:
            temp *= temp_decay
            for i in range(n_particles):
                if np.random.random() < np.exp(-(pbest_c5[i] - gbest_c5) / (temp + 1e-10)):
                    # Accept worse solution temporarily
                    h_new = pbest[i] + np.random.normal(0, 0.02, n_points)
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    if c5_new < pbest_c5[i]:
                        pbest[i] = h_new.copy()
                        pbest_c5[i] = c5_new
                        if c5_new < gbest_c5:
                            gbest = h_new.copy()
                            gbest_c5 = c5_new

        # Occasionally solve QCP relaxation
        if iter_idx % 50 == 0 and iter_idx > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]

                k_vals = np.linspace(1, n_points-1, min(20, n_points//2))
                for k in k_vals.astype(int):
                    if k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= gbest_c5 * dx + 1e-6)

                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                    if prob.status == cp.OPTIMAL:
                        h_feasible = h_opt.value
                        c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                        if c5_feasible < gbest_c5:
                            gbest = h_feasible
                            gbest_c5 = c5_feasible
                except:
                    pass
            except Exception:
                pass
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, gbest, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < gbest_c5:
                gbest = h_refined
                gbest_c5 = c5_refined
    except Exception:
        pass
    
    return (list(gbest), gbest_c5, n_points)
