import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create triangular window as starting point
        h_start = 1 - np.abs(np.arange(n_points) / (n_points - 1) - 0.5) * 2
        # Add small random perturbation
        h_start += np.random.normal(0, 0.02, n_points)
        h_start = np.clip(h_start, 0, 1)
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # DE bounds: each variable is constrained between 0 and 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # DE parameters
    population_size = 50
    max_generations = 300
    mutation_factor = 0.7
    crossover_rate = 0.9
    
    # DE optimization loop
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Run differential evolution for 1 generation
        try:
            result = differential_evolution(
                objective,
                bounds,
                maxiter=1,
                popsize=population_size,
                mutation=(mutation_factor, mutation_factor),
                recombination=crossover_rate,
                seed=seed + gen,
                disp=False,
                polish=False
            )
            
            if result.success and not np.isnan(result.fun):
                h_candidate = result.x
                # Ensure sum constraint
                h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
                h_candidate = np.clip(h_candidate, 0, 1)
                
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                    
        except Exception:
            pass
    
    # Final refinement with L-BFGS-B
    try:
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds_opt = [(0, 1) for _ in range(n_points)]
        result = minimize(objective, best_h, method='L-BFGS-B', 
                          bounds=bounds_opt, constraints=cons_sum, 
                          options={'maxiter': 200, 'ftol': 1e-10})
        if result.success and not np.isnan(result.fun):
            h_refined = result.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
