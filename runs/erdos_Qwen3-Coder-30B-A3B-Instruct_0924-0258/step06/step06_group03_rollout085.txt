import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create a triangular window and add some noise
        h_start = 1 - np.abs(np.arange(n_points) / (n_points - 1) - 0.5) * 2
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Objective function with proper constraints
    def objective(h):
        # Ensure h sums to n_points/2
        h = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    # Set up DE bounds for constraints
    def de_constraint(x):
        return np.sum(x) - n_points / 2.0
    
    constraints = {'type': 'eq', 'fun': de_constraint}
    
    # DE optimization
    de_res = differential_evolution(
        objective, 
        bounds, 
        seed=seed, 
        maxiter=1000, 
        popsize=50,
        mutation=(0.5, 1.0),
        recombination=0.9,
        constraints=constraints,
        disp=False,
        tol=1e-6
    )
    
    # Check if DE found a better solution
    if de_res.success:
        h_de = de_res.x
        h_de = h_de * (n_points/2) / np.sum(h_de)
        c5_de = evaluate_erdos_solution(h_de, dx)
        if c5_de < best_c5:
            best_h = h_de
            best_c5 = c5_de
    
    # Local refinement with L-BFGS-B
    def lbfgs_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def lbfgs_constraint(h):
        return np.sum(h) - n_points / 2.0
    
    # Refine top solutions from DE
    top_candidates = [best_h]
    
    for _ in range(10):
        if time() - start_time > budget_s * 0.9:
            break
        try:
            # Perturb best solution
            perturbed = best_h + np.random.normal(0, 0.02, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (n_points/2) / np.sum(perturbed)
            
            # Local optimization
            res = minimize(
                lbfgs_objective,
                perturbed,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if res.success:
                h_refined = res.x
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            continue
    
    # Final check with more refinement
    try:
        res = minimize(
            lbfgs_objective,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        
        if res.success:
            h_final = res.x
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
