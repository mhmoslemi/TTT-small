import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    # Simulated Annealing parameters
    T0 = 1.0
    Tmin = 1e-5
    alpha = 0.999
    iter_per_T = 100
    T = T0
    
    # Initialize best solution
    current_overlap = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_overlap = current_overlap
    
    # Annealing loop
    while T > Tmin and (time() - start_time) < budget_s * 0.95:
        for _ in range(iter_per_T):
            # Create new candidate solution
            h_candidate = h.copy()
            
            # Randomly select two indices to perturb
            i, j = np.random.choice(n_points, 2, replace=False)
            
            # Generate perturbation
            delta = np.random.normal(0, 0.05)
            
            # Apply to both indices with constraint adjustment
            h_candidate[i] += delta
            h_candidate[j] -= delta
            
            # Ensure the sum constraint is maintained approximately
            # Adjust the last element to maintain sum
            adjusted_sum = np.sum(h_candidate)
            h_candidate[-1] += (n_points/2 - adjusted_sum)
            
            # Clamp values to [0, 1]
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Ensure the sum is exactly n_points/2
            h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
            
            # Evaluate overlap
            candidate_overlap = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept or reject based on simulated annealing criteria
            delta_overlap = candidate_overlap - current_overlap
            if delta_overlap < 0 or np.exp(-delta_overlap / T) > np.random.rand():
                h = h_candidate.copy()
                current_overlap = candidate_overlap
                
                # Update best solution
                if current_overlap < best_overlap:
                    best_h = h.copy()
                    best_overlap = current_overlap
        
        # Decrease temperature
        T *= alpha
    
    # Final refinement with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,

            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_overlap = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_overlap, n_points)
