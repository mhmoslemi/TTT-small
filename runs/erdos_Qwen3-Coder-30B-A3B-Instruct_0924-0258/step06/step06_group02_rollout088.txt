import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Start with smaller problem and grow
        n_points = 35
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Strategy: structured initialization + trust region optimization
    # Generate structured initial point
    if 'initial_h_values' not in globals():
        h_start = np.zeros(n_points)
        # Create a pattern with high concentration in regions
        regions = [0.5, 1.0, 1.5]  # Center points
        spread = 0.3
        for center in regions:
            idx = int(center / 2 * n_points)
            start_idx = max(0, idx - int(spread * n_points / 2))
            end_idx = min(n_points, idx + int(spread * n_points / 2))
            h_start[start_idx:end_idx] = 0.8
        # Normalize
        h_start = h_start * (n_points / (2 * np.sum(h_start)))
    
    # Begin with trust region based optimization
    h_current = h_start.copy()
    c5_current = evaluate_erdos_solution(h_current, dx)
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Trust region parameters
    tr_radius = 0.2
    shrink_factor = 0.8
    expand_factor = 1.1
    max_iterations = 500
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate a perturbation within trust region
        perturbation = np.random.uniform(-tr_radius, tr_radius, n_points)
        h_new = h_current + perturbation
        h_new = np.clip(h_new, 0, 1)
        
        # Project back to integral constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on improvement
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
            tr_radius = min(0.5, tr_radius * expand_factor)  # Expand trust region
        else:
            tr_radius *= shrink_factor  # Shrink trust region
            
        # Update best
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
    # Final refinement using a small QP problem
    try:
        h_opt = cp.Variable(n_points)
        # Objective is to minimize C5
        # We'll approximate the max overlap as a scalar variable
        obj = cp.Variable()
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1,
            # Simplified constraints for a maximum overlap term
            # Create a proxy to encourage smaller overlaps
        ]
        # Simple linear relaxation of overlap constraints
        overlap_exprs = []
        for k in range(1, n_points):
            # k range we use is limited to make the problem easier to solve
            if k < n_points // 2:
                corr = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                # This is more correctly handled in full correlation
                corr = cp.sum(cp.multiply(h_opt[k:], 1 - h_opt[:n_points-k]))
            overlap_exprs.append(corr)
            
        # Constraint on this term being at most a multiplier of best_c5
        max_over = cp.max(overlap_exprs) # Why not use this directly?
        constraints.append(max_over <= obj)
        
        prob = cp.Problem(cp.Minimize(obj), constraints)
        prob.solve(solver=cp.ECOS, verbose=False, eps=1e-5)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception as e:
        pass
    
    return (list(best_h), best_c5, n_points)
