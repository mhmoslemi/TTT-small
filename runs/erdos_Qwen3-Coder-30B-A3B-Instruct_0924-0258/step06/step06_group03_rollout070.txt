import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        # Use Hamming window as initialization
        h = 0.54 + 0.46 * np.cos(np.pi * (np.arange(n_points) / n_points - 0.5) * 2)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Projected gradient descent
    max_iter = 1000
    eta = 0.01
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute current maximal correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_lag = max_idx - (len(h) - 1)
        
        # Compute gradient approximation
        # For small changes, we can approximate gradient using finite differences
        # But let's use the correlation structure directly
        # Simply shift by -max_lag to get the relevant part
        shifted_h = np.roll(h, max_lag)
        
        # Gradient w.r.t. h for current lag (conceptual - this is the direction of steepest descent)
        # The gradient of correlation is the convolution with -1
        grad = np.correlate(shifted_h, -1 * np.ones_like(h), mode='full')
        
        # We're interested in the gradient at max_lag
        if max_lag >= 0:
            grad_h = grad[max_lag:max_lag + len(h)]
        else:
            grad_h = np.concatenate([grad[max_lag:], np.zeros(-max_lag)])
            grad_h = grad_h[:len(h)]
            
        # Update step
        h_new = h - eta * grad_h
        
        # Project to feasible set [0,1] and ensure sum = n_points / 2
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        current_c5 = evaluate_erdos_solution(h, dx)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
