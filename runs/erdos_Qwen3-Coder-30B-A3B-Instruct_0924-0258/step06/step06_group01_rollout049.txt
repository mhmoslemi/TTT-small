import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    Tmin = 1e-5
    alpha = 0.999
    max_iter_per_T = 100
    
    T = T0
    current_h = h.copy()
    current_c5 = best_c5
    
    # SA loop
    while T > Tmin and time() - start_time < budget_s * 0.95:
        for _ in range(max_iter_per_T):
            # Generate neighbor: perturb two random indices
            i, j = random.sample(range(n_points), 2)
            h_new = current_h.copy()
            
            # Small random perturbation
            delta = random.gauss(0, 0.05)
            h_new[i] += delta
            
            # Adjust j to maintain sum constraint
            adjustment = -delta * (n_points / 2) / np.sum(current_h)
            h_new[j] += adjustment
            
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to maintain sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Compute new overlap
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject
            delta_c5 = new_c5 - current_c5
            if delta_c5 < 0 or np.exp(-delta_c5 / T) > random.random():
                current_h = h_new.copy()
                current_c5 = new_c5
            
            # Update best
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        T *= alpha
    
    # Final refinement with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
