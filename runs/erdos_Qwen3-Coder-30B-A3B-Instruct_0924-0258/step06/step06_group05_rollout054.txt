import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary algorithm with population-based search
    population_size = 50
    num_generations = 1000
    mutation_rate = 0.05
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Track best solution
    best_index = np.argmin([evaluate_erdos_solution(h, dx) for h in population])
    best_h = population[best_index].copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Evolution loop
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Evaluate population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Selection: keep top 20%
        sorted_indices = np.argsort(fitness)
        elite_indices = sorted_indices[:population_size // 5]
        
        # Create offspring
        offspring = []
        for idx in elite_indices:
            # Add mutation
            noise = np.random.normal(0, mutation_rate, n_points)
            child = population[idx] + noise
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            offspring.append(child.copy())
        
        # Replace population with elite + offspring
        new_population = [population[i] for i in elite_indices]
        while len(new_population) < population_size:
            new_population.append(offspring[np.random.randint(len(offspring))])
        population = new_population[:population_size]
        
        # Update best solution
        current_best_idx = np.argmin([evaluate_erdos_solution(h, dx) for h in population])
        current_best_h = population[current_best_idx]
        current_best_c5 = evaluate_erdos_solution(current_best_h, dx)
        
        if current_best_c5 < best_c5:
            best_h = current_best_h.copy()
            best_c5 = current_best_c5
        
        # Adaptive mutation rate
        if generation % 100 == 0:
            mutation_rate *= 0.95
            mutation_rate = max(mutation_rate, 0.01)
    
    # Final QCP refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample k values for constraints
        k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
        for k_val in k_vals:
            k = int(k_val * n_points / 2)  # to index space
            if k < n_points and k > 0:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
        
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
