import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on a discrete space
    def binary_to_real(h_bin):
        # Convert binary vector to real vector with same sum constraint
        h_real = h_bin.astype(float)
        h_real = h_real * (n_points/2) / np.sum(h_real)
        return h_real

    def make_binary_h(n_points, ones_count):
        # Create a binary vector with exactly 'ones_count' ones
        h = np.zeros(n_points)
        h[:ones_count] = 1
        np.random.shuffle(h)
        return h

    # Initial binary representation
    ones_count = int(n_points // 2)
    h_bin = make_binary_h(n_points, ones_count)
    h_real = binary_to_real(h_bin)
    current_c5 = evaluate_erdos_solution(h_real, dx)
    
    if current_c5 < best_c5:
        best_h = h_real.copy()
        best_c5 = current_c5

    # Annealing parameters
    temp = 1.0
    decay = 0.995
    min_temp = 1e-6
    max_iter = 10000
    
    # Temporarily enable不大
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break

        # Select two random positions to flip
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        h_new_bin = h_bin.copy()
        h_new_bin[i1] = 1 - h_new_bin[i1]  # Flip
        h_new_bin[i2] = 1 - h_new_bin[i2]  # Flip

        # Ensure sum constraint (should be maintained)
        if np.sum(h_new_bin) == ones_count:
            h_new_real = binary_to_real(h_new_bin)
            new_c5 = evaluate_erdos_solution(h_new_real, dx)
            delta = new_c5 - current_c5

            # Accept or reject
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h_bin = h_new_bin
                h_real = h_new_real
                current_c5 = new_c5

                if new_c5 < best_c5:
                    best_h = h_real.copy()
                    best_c5 = new_c5

        temp *= decay
        if temp < min_temp:
            temp = min_temp

    # Periodic convex refinement
    for i in range(1000):
        if time() - start_time > budget_s * 0.99:
            break
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]

            # Sample a few lag values
            lag_vals = np.linspace(0.1, 1.9, min(20, n_points//3))
            for k_val in lag_vals:
                k = int(k_val * n_points / 2)
                if k < n_points and k > 0:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)

            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=200)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible

                # Reinitialize binary from real solution for next iteration
                h_bin = np.round(h_feasible).astype(int)
                h_bin = np.clip(h_bin, 0, 1)
                h_real = binary_to_real(h_bin)
                current_c5 = evaluate_erdos_solution(h_real, dx)
        except Exception:
            pass
        if time() - start_time > budget_s * 0.98:
            break

    # Final local refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
