import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create a triangular-shaped initial guess
        h_start = 1 - np.abs(np.arange(n_points) / (n_points - 1) - 0.5) * 2
        # Add some noise
        h_start += np.random.normal(0, 0.02, n_points)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Ensure initial solution satisfies constraint
    h_start = h_start * (n_points/2) / np.sum(h_start)
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # DE optimization
    try:
        de_result = differential_evolution(
            objective, 
            bounds, 
            args=(),
            maxiter=300,
            popsize=50,
            mutation=(0.5, 1.0),
            recombination=0.9,
            seed=seed,
            callback=lambda x, convergence: None,
            disp=False
        )
        if de_result.success:
            h_de = de_result.x
            h_de = np.clip(h_de, 0, 1)
            h_de = h_de * (n_points/2) / np.sum(h_de)
            c5_de = evaluate_erdos_solution(h_de, dx)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    except Exception:
        pass
    
    # Local refinement on the best solution found by DE
    if time() - start_time > budget_s * 0.8:
        return (list(best_h), best_c5, n_points)
    
    # Try local refinement with L-BFGS-B
    try:
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='L-BFGS-B', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        
        if res.success:
            h_refined = res.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
