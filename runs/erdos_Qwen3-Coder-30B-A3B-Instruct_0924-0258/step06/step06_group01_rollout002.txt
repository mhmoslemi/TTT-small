import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Get initial construction or create one
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a simple initial guess
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different number of points to get better results
    candidates = [35, 50, 40, 30]
    
    for n_points in candidates:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_current = h_start.copy()
        h_current = h_current * (n_points/2) / np.sum(h_current)
        h_current = np.clip(h_current, 0, 1)
        
        # Iterative refinement using optimization
        for iteration in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Try to refine using scipy minimize
            try:
                cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(
                    lambda x: evaluate_erdos_solution(x, dx),
                    h_current,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons_sum,
                    options={'ftol': 1e-9, 'maxiter': 100}
                )
                
                if res.success:
                    h_ref = res.x
                    h_ref = np.clip(h_ref, 0, 1)
                    h_ref = h_ref * (n_points/2) / np.sum(h_ref)
                    current_c5 = evaluate_erdos_solution(h_ref, dx)
                    
                    if current_c5 < best_c5:
                        best_h = h_ref.copy()
                        best_c5 = current_c5
                    h_current = h_ref.copy()
                else:
                    # Fall back to gradient-free method with small steps
                    step_size = 0.01
                    for i in range(n_points):
                        h_temp = h_current.copy()
                        h_temp[i] += step_size
                        h_temp = np.clip(h_temp, 0, 1)
                        h_temp = h_temp * (n_points/2) / np.sum(h_temp)
                        current_c5 = evaluate_erdos_solution(h_temp, dx)
                        if current_c5 < best_c5:
                            best_h = h_temp.copy()
                            best_c5 = current_c5
                        h_temp[i] -= 2*step_size
                        h_temp = np.clip(h_temp, 0, 1)
                        h_temp = h_temp * (n_points/2) / np.sum(h_temp)
                        current_c5 = evaluate_erdos_solution(h_temp, dx)
                        if current_c5 < best_c5:
                            best_h = h_temp.copy()
                            best_c5 = current_c5
                    h_current = best_h.copy()
                
            except Exception:
                break
    
    # Final refinement
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
