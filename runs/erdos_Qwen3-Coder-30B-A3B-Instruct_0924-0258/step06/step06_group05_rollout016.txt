import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Early termination tracking
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Gradient descent optimization with perturbations
    max_iter = int(budget_s * 10)  # Roughly 10 steps per second
    step_size = 0.05
    
    for idx in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Identify k that maximizes overlap
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        
        # Compute gradient of S_k_max
        gradient = np.zeros(n_points)
        k = k_max - (n_points - 1)  # convert from full correlation index
        
        # Compute derivative of h[i] * (1 - h[i + k])
        # For valid indices
        if k >= 0:
            for i in range(n_points - k):
                gradient[i] += 1 - best_h[i + k]
            for i in range(k, n_points):
                gradient[i] -= best_h[i - k]
        else:
            k_abs = -k
            for i in range(n_points - k_abs):
                gradient[i] += 1 - best_h[i + k_abs]
            for i in range(k_abs, n_points):
                gradient[i] -= best_h[i - k_abs]
        
        # Normalize gradient for stable step
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm * step_size
        
        # Update with gradient
        h_new = best_h + gradient
        
        # Add random noise occasionally
        if idx % 10 == 0:
            noise = np.random.normal(0, step_size * 0.1, n_points)
            h_new += noise
            
        # Keep feasible
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to meet constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept improvement
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            # Random restart
            h_new = np.random.rand(n_points)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            no_improvement_count = 0

    # Convex relaxation step
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample k values for constraints
        k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
        for k_val in k_vals:
            k = int(k_val * n_points / 2)
            if k < n_points and k > 0:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
