import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Grid refinement
    resolutions = []
    n_current = 10
    while n_current <= 35:
        resolutions.append(n_current)
        n_current *= 2
        if n_current > 35:
            resolutions.append(35)
            break
    
    # Refine each resolution
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize
        if n_points == len(h_start):
            h = h_start.copy()
        else:
            # Interpolate from previous
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_start)), h_start)
            # Sinusoidal initialization
            x = np.linspace(0, 2, n_points)
            h = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2.0 / n_points))
            h = np.clip(h, 0, 1)
        
        # Scale to satisfy constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Optimization loop
        max_iter = 100
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Stochastic gradient descent
            h_new = h.copy()
            noise = np.random.normal(0, 0.01, n_points)
            h_new += noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_h = h_new
                best_c5 = c5_new
            
            # Simulated Annealing
            if i % 10 == 0:
                temp = 1.0 / (1.0 + i * 0.01)
                if np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
                    h = h_new
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
        
        # Final correction
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
        
        # Reference point for next resolution
        h_start = best_h.copy()
    
    # Binary search for tighter bound
    c5_target = 0.3807
    c5_lo, c5_hi = best_c5, c5_target
    
    for _ in range(20):
        if time() - start_time > budget_s * 0.95:
            break
        c5_mid = (c5_lo + c5_hi) / 2
        try:
            # LP relaxation to check feasibility
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add constraint for max overlap at most c5_mid
            corr_sum = cp.sum(cp.multiply(h_opt, 1 - h_opt))
            constraints.append(corr_sum <= c5_mid * dx + 1e-8)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                c5_lo = c5_mid
            else:
                c5_hi = c5_mid
        except Exception:
            c5_hi = c5_mid
    
    final_c5 = c5_lo
    
    # Final check to ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
