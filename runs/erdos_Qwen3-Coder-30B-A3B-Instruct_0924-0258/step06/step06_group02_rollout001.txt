import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T0 = 0.5
    Tf = 1e-4
    cooling_rate = 0.99
    max_steps = 2000
    
    T = T0
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Track recent improvements
    recent_improvements = 0
    improvement_threshold = 50
    
    for step in range(max_steps):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Select indices with highest current overlap
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        # Get indices where overlap is maximal (excluding trivial case)
        k_indices = np.argsort(corr)[-int(n_points*0.1):] 
        k_indices = k_indices[k_indices > 0]  # Avoid trivial 0 shift
        
        # Apply perturbation to top overlap indices
        h_new = current_h.copy()
        modified = False
        
        for i in k_indices:
            if np.random.random() < 0.7:  # 70% chance to modify
                perturb = np.random.choice([-0.1, 0.1])
                h_new[i] += perturb
                modified = True
        
        if not modified:
            # If no modification, randomly perturb
            i = np.random.randint(0, n_points)
            perturb = np.random.choice([-0.1, 0.1])
            h_new[i] += perturb
        
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Acceptance criterion
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            recent_improvements = 0
        else:
            recent_improvements += 1
            if np.random.random() < math.exp(-(c5_new - current_c5) / T):
                current_h = h_new.copy()
                current_c5 = c5_new
        
        # Cooling
        if step % 10 == 0 and T > Tf:
            T *= cooling_rate
        
        # Restart if no improvement for a while
        if recent_improvements >= improvement_threshold:
            current_h = best_h.copy()
            current_c5 = best_c5
            T = T0
            recent_improvements = 0
    
    # Gradient-based refinement
    try:
        # Convert to continuous problem suitable for optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        # Initialize with our best solution
        h_refined = best_h.copy()
        
        # Perform local optimization
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        # Try gradient-based optimization
        result = minimize(objective, h_refined, method='SLSQP', 
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-6})
        
        if result.success:
            h_final = result.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final check
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    # Ensure final constraints
    if np.abs(np.sum(best_h) - n_points/2) > 1e-6:
        best_h = best_h * (n_points/2) / np.sum(best_h)
        
    return (list(best_h), best_c5, n_points)
