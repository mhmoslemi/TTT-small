import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def clamp_and_normalize(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points/2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach with better restart strategy
    resolutions = [30, 35, 40, 50, 60, 80]
    max_restarts = 25
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate
            h_interp = np.interp(np.linspace(0, 2, res), 
                               np.linspace(0, 2, n_points), 
                               h_start)
            h_current = h_interp
        else:
            h_current = h_start[:res].copy()
            if res < n_points:
                h_current = np.pad(h_current, (0, n_points - res), 'constant')
            else:
                h_current = h_current[:n_points]
        
        n_points = len(h_current)
        dx = 2.0 / n_points
        
        # Normalize to correct integral
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Mix of random and structured restarts
        restart_count = 0
        while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
            # Try several restart strategies
            strategy = restart_count % 4
            if strategy == 0:
                # Pure random
                h_restart = np.random.rand(n_points) * 0.5
            elif strategy == 1:
                # Perturb current best
                h_restart = h_current + np.random.normal(0, 0.03, n_points)
            elif strategy == 2:
                # Block structure
                h_restart = np.random.rand(n_points) * 0.5
                for i in range(0, n_points, 5):
                    if i + 5 <= n_points:
                        mean_val = np.mean(h_current[i:i+5])
                        h_restart[i:i+5] = mean_val + (np.random.rand(5) - 0.5) * 0.2
            else:
                # Add noise based on previous best
                noise = np.random.normal(0, 0.05, n_points)
                h_restart = h_current + noise
                
            h_restart = clamp_and_normalize(h_restart, n_points)
            
            # Local optimization using SLSQP
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res_opt = minimize(objective, h_restart, method='SLSQP', 
                                bounds=bounds, constraints=cons_sum, 
                                options={'maxiter': 300, 'ftol': 1e-10})
                if res_opt.success:
                    h_refined = res_opt.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        restart_count = 0
                else:
                    restart_count += 1
            except Exception:
                restart_count += 1
    
    # Final refinement at highest resolution
    try:
        h_final = best_h.copy()
        h_final = clamp_and_normalize(h_final, n_points)
        h_final = np.clip(h_final, 0, 1)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
