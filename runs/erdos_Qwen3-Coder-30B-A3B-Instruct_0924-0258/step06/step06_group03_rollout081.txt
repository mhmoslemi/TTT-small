import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a triangular window as initialization
        h_start = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h_start[i] = max(0, 1 - abs(i - mid) / (mid + 1))
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    # Use bounds for each variable and a custom constraint
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Objective function for DE (constraints handled by penalty)
    def de_objective(h):
        # Ensure normalization constraint
        h_sum = np.sum(h)
        if abs(h_sum - n_points/2) > 1e-6:
            # Add penalty for violating the constraint
            penalty = 1000 * abs(h_sum - n_points/2)
        else:
            penalty = 0
        
        return evaluate_erdos_solution(h, dx) + penalty
    
    # Run differential evolution
    de_result = differential_evolution(
        de_objective,
        bounds,
        maxiter=300,
        popsize=50,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        disp=False,
        worker_threads=1
    )
    
    # Use DE best solution if available
    if de_result.success:
        h_DE = de_result.x
        c5_DE = evaluate_erdos_solution(h_DE, dx)
        if c5_DE < best_c5:
            best_h = h_DE
            best_c5 = c5_DE
    
    # Local refinement with L-BFGS-B
    def refine_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def refine_constraint(h):
        return np.sum(h) - n_points / 2.0
    
    # Perform local refinement only for a short time
    remaining_time = budget_s * 0.9 - (time() - start_time)
    if remaining_time > 1.0:
        # Refine the best solution we have so far
        cons = {'type': 'eq', 'fun': refine_constraint}
        bounds_refine = [(0, 1) for _ in range(n_points)]
        
        try:
            # Run several local optimizations from different starting points
            for _ in range(5):
                if time() - start_time > budget_s * 0.95:
                    break
                h_start_local = best_h + np.random.normal(0, 0.05, n_points)
                h_start_local = np.clip(h_start_local, 0, 1)
                h_start_local = h_start_local * (n_points/2) / np.sum(h_start_local)
                
                res = minimize(
                    refine_objective,
                    h_start_local,
                    method='L-BFGS-B',
                    bounds=bounds_refine,
                    constraints=cons,
                    options={'maxiter': 100}
                )
                
                if res.success:
                    h_refined = res.x
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
        except Exception:
            pass

    return (list(best_h), best_c5, n_points)
