import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search for the minimum C5
    c_low = 0.0
    c_high = 0.38085
    tolerance = 1e-5
    
    # Function to check if a given C is achievable
    def is_feasible(C):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points/2,
            h >= 0,
            h <= 1
        ]
        
        # Add overlap constraints for all possible shifts
        for k in range(n_points):
            # Compute the overlap for shift k
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h[i] * (1 - h[j])
            constraints.append(overlap_expr <= C * n_points / 2)
            
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=500)
            return prob.status == cp.OPTIMAL
        except:
            return False

    # Binary search
    while time() - start_time < budget_s * 0.95 and (c_high - c_low) > tolerance:
        c_mid = (c_low + c_high) / 2
        if is_feasible(c_mid):
            c_high = c_mid
            # Try to refine solution
            h_refined = cp.Variable(n_points)
            constraints = [
                cp.sum(h_refined) == n_points/2,
                h_refined >= 0,
                h_refined <= 1
            ]
            
            # Add overlap constraints
            for k in range(n_points):
                overlap_expr = 0
                for i in range(n_points):
                    j = (i + k) % n_points
                    overlap_expr += h_refined[i] * (1 - h_refined[j])
                constraints.append(overlap_expr <= c_mid * n_points / 2)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_sol = h_refined.value
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    new_c5 = evaluate_erdos_solution(h_sol, dx)
                    if new_c5 < best_c5:
                        best_h = h_sol.copy()
                        best_c5 = new_c5
            except:
                pass
        else:
            c_low = c_mid
    
    # Final optimization using local method
    def obj_func(h):
        return evaluate_erdos_solution(h, dx)
        
    cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use best solution from binary search as starting point
    h_init = best_h.copy()
    try:
        res = minimize(
            obj_func,
            h_init,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            h_final = res.x
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
