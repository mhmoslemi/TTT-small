import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 35
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    T_min = 1e-5
    alpha = 0.999
    iterations_per_T = 100
    
    T = T0
    current_h = h.copy()
    current_c5 = best_c5
    
    # Main Simulated Annealing Loop
    while T > T_min and (time() - start_time) < budget_s * 0.95:
        for _ in range(iterations_per_T):
            # Create neighbor solution
            h_candidate = current_h.copy()
            
            # Select two random indices
            i, j = random.sample(range(n_points), 2)
            
            # Perturb values
            delta = random.uniform(-0.1, 0.1)
            h_candidate[i] += delta
            
            # Adjust the other value to maintain the sum constraint
            # We want sum(h) to remain constant so we adjust h[j] by -delta * (h[i] / sum(h))
            # But that wouldn't work because we are perturbing two values.
            # So we adjust h[j] such that sum(h) stays the same.
            # We adjust j appropriately to compensate for the change in i
            sum_diff = h_candidate[i] - current_h[i]
            h_candidate[j] -= sum_diff
            
            # Clip and normalize
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Evaluate candidate
            candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept or reject based on Metropolis criterion
            delta_e = candidate_c5 - current_c5
            if delta_e < 0 or np.exp(-delta_e / T) > random.random():
                current_h = h_candidate
                current_c5 = candidate_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

        # Decrease temperature
        T *= alpha

    # Final refinement with SLSQP
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
