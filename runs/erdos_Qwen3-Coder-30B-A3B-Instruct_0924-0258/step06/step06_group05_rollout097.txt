import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize binary helper
    def to_binary(h_vec):
        # Convert to binary (0 or 1) based on threshold
        return (h_vec > 0.5).astype(int)
        
    def rescale_binary(h_bin):
        # Rescale binary vector to have correct sum
        current_sum = np.sum(h_bin)
        target_sum = n_points / 2.0
        if abs(current_sum - target_sum) < 1e-6:
            return h_bin
        diff = int(target_sum - current_sum)
        if diff > 0:
            # Add ones
            zero_indices = np.where(h_bin == 0)[0]
            if len(zero_indices) >= diff:
                for i in range(diff):
                    h_bin[zero_indices[i]] = 1
        elif diff < 0:
            # Remove ones
            one_indices = np.where(h_bin == 1)[0]
            if len(one_indices) >= -diff:
                for i in range(-diff):
                    h_bin[one_indices[i]] = 0
        return h_bin
    
    def binary_evaluate(h_bin):
        # Use binary values for exact evaluation
        h_float = h_bin.astype(float)
        return evaluate_erdos_solution(h_float, dx)
    
    # Start with binary conversion of best
    h_binary = to_binary(best_h)
    h_binary = rescale_binary(h_binary)
    current_c5 = binary_evaluate(h_binary)
    
    if current_c5 < best_c5:
        best_h = h_binary.astype(float)
        best_c5 = current_c5

    # Binary search with simulated annealing
    max_iter = 10000
    temperature = 10.0
    cooling_rate = 0.995
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate new candidate by swapping two random positions
        h_new = h_binary.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Ensure sum constraint
        h_new = rescale_binary(h_new)
        
        # Evaluate new solution
        c5_new = binary_evaluate(h_new)
        
        # Accept if better, or with some probability if worse
        if c5_new < current_c5 or np.random.rand() < np.exp((current_c5 - c5_new) / temperature):
            h_binary = h_new
            current_c5 = c5_new
            
            if c5_new < best_c5:
                best_h = h_binary.astype(float)
                best_c5 = c5_new
                last_improvement = iteration

        # Cooling schedule
        if iteration % 10 == 0:
            temperature *= cooling_rate
        
        # Periodic convex refinement
        if iteration % 100 == 0 and iteration > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                # Sample a few lags
                k_vals = np.linspace(1, n_points-1, min(20, n_points//2)).astype(int)
                for k in k_vals:
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=300)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
        
        # Early stopping if stagnation
        if iteration - last_improvement > 2000:
            break

    # Final local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
