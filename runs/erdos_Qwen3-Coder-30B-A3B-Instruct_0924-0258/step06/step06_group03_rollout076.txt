import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def triangular_window(n):
    # Create a triangular window function
    x = np.arange(n)
    peak = (n - 1) / 2
    h = 1 - np.abs(x - peak) / peak
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = triangular_window(n_points)
    
    # Normalize to satisfy constraint
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # Objective function
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Constraint function: sum must equal n_points/2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial best solution
    best_h = h_start.copy()
    best_c5 = objective(best_h)
    
    # Let's perform a global optimization first
    # Using Differential Evolution as it's good for high-dimensional nonlinear problems
    de_bounds = [(0, 1) for _ in range(n_points)]
    
    # Add normalization to the constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Run differential evolution
    try:
        de_result = differential_evolution(
            objective,
            de_bounds,
            constraints=constraints,
            seed=seed,
            maxiter=500,
            popsize=30,
            mutation=(0.5, 1),
            recombination=0.7,
            updating='deferred',
           workers=1,
            disp=False
        )
        
        if de_result.success and de_result.fun < best_c5:
            best_h = de_result.x.copy()
            best_c5 = de_result.fun
    except Exception:
        pass
    
    # Local refinement using L-BFGS-B
    if time() - start_time < budget_s * 0.8:
        try:
            def refined_objective(h):
                return evaluate_erdos_solution(h, dx)
            
            # Use scipy minimize with L-BFGS-B
            result = minimize(
                refined_objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-10},
                callback=None
            )
            
            if result.success and result.fun < best_c5:
                best_h = result.x.copy()
                best_c5 = result.fun
        except Exception:
            pass
    
    # Final check and return
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
