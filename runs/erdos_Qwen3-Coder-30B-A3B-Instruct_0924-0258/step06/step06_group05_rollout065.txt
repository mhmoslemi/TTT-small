import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def compute_gradients(h, k_max, n_points):
    """Compute gradient of correlation at shift k_max"""
    grad = np.zeros(n_points)
    for i in range(n_points):
        if i + k_max < n_points:
            grad[i] += (1 - h[i + k_max])
        if i - k_max >= 0:
            grad[i] -= h[i - k_max]
    return grad

def line_search(h, grad, dx, max_iter=10):
    """Backtracking line search"""
    alpha = 1.0
    c5_current = evaluate_erdos_solution(h, dx)
    for _ in range(max_iter):
        h_new = h + alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (len(h) / 2) / np.sum(h_new)
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < c5_current:
            return h_new, c5_new, alpha
        alpha *= 0.5
    return h, c5_current, alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with line search
    max_iter = 1000
    
    for idx in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Identify maximum overlap shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        k_max = k_max - (n_points - 1)  # Convert to actual shift
        
        if k_max == 0:
            k_max = 1
            
        # Compute gradient
        grad = compute_gradients(best_h, abs(k_max), n_points)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
            
        # Line search
        h_new, c5_new, _ = line_search(best_h, grad, dx)
        
        # Random perturbation with decreasing probability
        if idx % 50 == 0 and np.random.rand() < 0.3:
            noise = np.random.normal(0, 0.05, n_points)
            h_new = best_h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            
        # Update best solution
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
    
    # Convex relaxation
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraints for several random shifts
        k_vals = np.random.choice(range(1, n_points//2), min(15, n_points//3), replace=False)
        for k_val in k_vals:
            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k_val], 1 - h_opt[k_val:]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
        
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
