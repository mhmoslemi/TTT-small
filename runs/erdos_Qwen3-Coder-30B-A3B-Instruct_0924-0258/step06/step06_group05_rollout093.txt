import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with adaptive step size
    max_iter = 10000
    decay_rate = 0.999
    step_size_base = 0.01
    patience = 0
    patience_limit = 100
    
    for idx in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find current maximum overlap index
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        
        # Compute gradient of S_{k_max}
        gradient = np.zeros(n_points)
        k = k_max - (n_points - 1)  # Convert full correlation index to shift
        
        # Gradient computation
        for i in range(n_points):
            if i + k >= 0 and i + k < n_points:
                gradient[i] += (1 - best_h[i + k])
            if i - k >= 0 and i - k < n_points and k != 0:
                gradient[i] -= best_h[i - k]
        
        # Normalize gradient and take step
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 1e-10:
            gradient = gradient / grad_norm
        
        step_size = step_size_base * (decay_rate ** idx)
        h_new = best_h + step_size * gradient
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept improvement
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience = 0
        else:
            patience += 1
            
        # Random perturbations to escape local minima
        if idx % 50 == 0 and patience > 10:
            noise = np.random.normal(0, 0.005, n_points)
            h_new = best_h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                patience = 0
            else:
                # Revert to previous value
                patience = 0
                
        if patience > patience_limit:
            break
    
    # Final QCP refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample k values for constraints, reducing density
        k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
        for k_val in k_vals:
            k = int(k_val * n_points / 2)  # to index space
            if k < n_points and k > 0:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
        
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
