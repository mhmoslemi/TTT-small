import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # ADMM parameters
    rho = 1e-4
    max_iter = 1000
    tol = 1e-6
    iter_count = 0
    
    # Variables for ADMM
    z = h.copy()
    u = np.zeros(n_points)
    
    # Main ADMM loop
    while time() - start_time < budget_s * 0.9 and iter_count < max_iter:
        # Primal update: minimize the augmented Lagrangian
        # We approximate this by solving the convex subproblem
        try:
            # Create optimization variables
            z_new = cp.Variable(n_points)
            # Objective with penalty term
            # We minimize quadratic penalty + scaled correlation term
            corr_term = 0
            # For each k, compute sum(h[i] * (1 - h[i+k]))
            # Approximate with correlation-like computation
            # Due to complexity, just solve a simpler version
            h_copy = z.copy()
            
            # Compute references for gradient
            corr_full = np.correlate(h_copy, 1 - h_copy, mode='full')
            max_corr = np.max(corr_full)
            
            # For simplicity, handle in numpy with SLSQP after small number of ADMM steps
            # Reduction of ADMM constraint handling
            def obj_func(z_vec):
                corr_vals = np.correlate(z_vec, 1 - z_vec, mode='full')
                return -np.max(corr_vals) * dx
            
            # Equality constraint
            cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                obj_func,
                z,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            
            if res.success:
                z = res.x
            else:
                z = z.copy()  # fallback to previous
            
            # Update h
            h = z - (1/rho) * (z - h + u)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            
            # Update dual variable
            u = u + h - z

            # Check for convergence
            primal_residual = np.linalg.norm(h - z)
            if primal_residual < tol:
                break
                
            iter_count += 1

            # Update best
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

        except Exception:
            break
    
    # Final refinement
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
