import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Direct optimization with constrained minimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    # Constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try multiple random restarts
    n_restarts = 20
    for restart in range(n_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Start from perturbed version of best solution
        h_init = best_h + np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points/2) / np.sum(h_init)
        
        try:
            res = minimize(objective, h_init, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                c5_new = objective(res.x)
                if c5_new < best_c5:
                    best_h = res.x
                    best_c5 = c5_new
        except:
            pass
    
    # Local search around best
    if time() - start_time < budget_s * 0.9:
        for i in range(1000):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb
            noise = np.random.normal(0, 0.02, n_points)
            h_new = best_h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
