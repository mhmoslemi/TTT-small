import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search approach
    h_binary = (h_start > 0.5).astype(int)
    h_binary = h_binary.astype(float)
    h_binary = h_binary * (n_points/2) / np.sum(h_binary)
    
    # Initial binary with better distribution
    ones_count = n_points // 2
    h_binary = np.zeros(n_points)
    h_binary[:ones_count] = 1.0
    np.random.shuffle(h_binary)
    
    # Ensure sum constraint
    current_sum = np.sum(h_binary)
    if abs(current_sum - n_points/2) > 1e-6:
        # Adjust to match exactly
        diff = int(n_points/2 - current_sum)
        if diff > 0:
            # Need to set some 0s to 1s
            zero_indices = np.where(h_binary == 0)[0]
            for i in range(diff):
                if i < len(zero_indices):
                    h_binary[zero_indices[i]] = 1.0
        elif diff < 0:
            # Need to set some 1s to 0s
            one_indices = np.where(h_binary == 1)[0]
            for i in range(-diff):
                if i < len(one_indices):
                    h_binary[one_indices[i]] = 0.0
        current_sum = np.sum(h_binary)
    
    h_current = h_binary.copy()
    current_c5 = evaluate_erdos_solution(h_current, dx)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_current.copy()
    
    # Bit-flip local search
    iter_count = 0
    max_iters = 10000
    temp = 1.0
    temp_decay = 0.999
    min_temp = 1e-6
    
    for iter_count in range(max_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor by swapping two random bits
        idx1 = np.random.randint(0, n_points)
        idx2 = np.random.randint(0, n_points)
        
        if idx1 == idx2:
            continue

        new_h = h_current.copy()
        new_h[idx1] = 1 - new_h[idx1]  # Flip bit
        new_h[idx2] = 1 - new_h[idx2]
        
        # Maintain sum exactly
        new_sum = np.sum(new_h)
        if abs(new_sum - n_points/2) > 1e-6:
            # Only accept if sum is correct within tolerance
            continue
        
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept with probability based on temperature
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / temp):
            h_current = new_h.copy()
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5

        temp = max(temp * temp_decay, min_temp)
        
    # Convex refinement with sampled constraints
    try:
        h_opt = cp.Variable(n_points)
        
        # Constraint: sum = n_points/2
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample a few lags for the constraint
        k_vals = np.linspace(1, n_points-1, min(20, n_points//3), dtype=int)
        for k in k_vals:
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-8)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-8, max_iters=200)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
        
    # Final refinement using continuous optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Ensure the solution is valid
    final_sum = np.sum(best_h)
    if abs(final_sum - n_points/2) > 1e-6:
        # Rescale
        best_h = best_h * (n_points / 2) / final_sum
    
    return (list(best_h), best_c5, n_points)
