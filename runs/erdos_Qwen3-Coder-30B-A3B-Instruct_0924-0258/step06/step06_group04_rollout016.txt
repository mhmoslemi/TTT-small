```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a smooth pattern to prevent local minima
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO Parameters
    n_particles = 50
    n_iter = 200
    w = 0.7  # inertia weight
    c1, c2 = 1.4, 1.4  # acceleration coefficients
    v_max = 0.2  # max velocity
    
    # Initialize particles
    particles = np.random.rand(n_particles, n_points) * 0.5 + 0.25  # [0.25, 0.75]
    particles = np.clip(particles, 0, 1)
    for i in range(n_particles):
        particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
    
    velocities = np.random.rand(n_particles, n_points) * 0.1 - 0.05  
    pbest = particles.copy()
    pbest_scores = np.array([evaluate_erdos_solution(p, dx) for p in particles])
    
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # Adaptive constraint sampling
    lag_samples = None
    constraint_sampling_freq = 50
    
    for iter_num in range(n_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update constraints every few iterations
        if iter_num % constraint_sampling_freq == 0:
            # Sample lags where overlap is high (include lags with large correlation)
            corr_full = np.correlate(gbest, 1 - gbest, mode='full')
            # Sample the top 30 lags with highest correlation
            top_lag_indices = np.argsort(corr_full)[-30:][::-1]
            lag_samples = top_lag_indices[:15]
            # Adjust for negative lags that wrap around
            lag_samples = lag_samples[lag_samples < n_points]
        
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            # Renormalize to satisfy sum constraint
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            score = evaluate_erdos_solution(particles[i], dx)
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = particles[i].copy()
                if score < gbest_score:
                    gbest_score = score
                    gbest = particles[i].copy()
        
        # Update global best
        gbest_idx = np.argmin(pbest_scores)
        if pbest_scores[gbest_idx] < gbest_score:
            gbest = pbest[gbest_idx].copy()
            gbest_score = pbest_scores[gbest_idx]
    
    # Post-processing with SQP
    try:
        h_sqp = cp.Variable(n_points)
        constraints = [
            cp.sum(h_sqp) == n_points / 2.0,
            h_sqp >= 0,
            h_sqp <= 1
        ]
        
        # Use critical lags from PSO
        if lag_samples is not None:
            for k in lag_samples:
                if k == 0:  # This case is identity, skip
                    continue
                if k < n_points:
                    # For lag k, compute correlation over valid region
                    corr_sum = cp.sum(cp.multiply(h_sqp[:n_points-k], 1 - h_sqp[k:]))
                else:
                    # Wraparound case
                    corr_sum = cp.sum(cp.multiply(h_sqp[k-n_points+1:], 1 - h_sqp[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= gbest_score * dx + 1e-6)
        
        # Minimize a penalty objective
        obj = cp.Minimize(0)
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_sqp.value
            # Recompute c5
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < gbest_score:
                best_h = h_feasible
                best_c5 = c5_feasible
            else:
                best_h = gbest
                best_c5 = gbest_score
        else:
            # fallback to best found so far
            best_h = gbest
            best_c5 = gbest_score
    except Exception:
        best_h = gbest
        best_c5 = gbest_score
    
    # Final normalization and evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
```
