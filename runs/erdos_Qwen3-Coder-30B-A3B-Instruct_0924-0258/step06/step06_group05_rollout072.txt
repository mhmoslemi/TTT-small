import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Start with binary representation
    h_binary = (h_start > 0.5).astype(int)
    h_current = h_binary.astype(float)
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h_current, dx)
    best_h = h_current.copy()
    
    # Annealing parameters
    temp = 1.0
    decay = 0.995
    min_temp = 1e-6
    
    # Binary search with swap operations
    max_iter = 1000
    iter_count = 0
    
    while iter_count < max_iter and time() - start_time < budget_s * 0.95:
        iter_count += 1
        # Generate neighbor via bit swap
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        h_new = h_current.copy()
        h_new[idx1] = 1 - h_new[idx1]
        h_new[idx2] = 1 - h_new[idx2]
        
        # Ensure sum constraint is maintained
        current_sum = np.sum(h_current)
        new_sum = np.sum(h_new)
        diff = new_sum - current_sum
        
        if abs(diff) > 1e-6:  # Adjust sum
            # Find indices to flip to maintain sum
            if diff > 0:
                to_flip = np.where(h_new == 0)[0]
            else:
                to_flip = np.where(h_new == 1)[0]
            
            if len(to_flip) > 0:
                flip_idx = np.random.choice(to_flip)
                h_new[flip_idx] = 1 - h_new[flip_idx]
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5:
            h_current = h_new.copy()
            best_c5 = c5_new
            best_h = h_current.copy()
        else:
            # Simulated annealing
            p = np.exp(-(c5_new - best_c5) / temp)
            if np.random.rand() < p:
                h_current = h_new.copy()
        
        # Temperature decay
        temp = max(min_temp, temp * decay)
        
        # Periodic convex optimization refinement
        if iter_count % 100 == 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample a few k values for constraints
                k_vals = [int(k * n_points / 2) for k in np.linspace(0.1, 1.0, 10) if 
                         int(k * n_points / 2) < n_points and int(k * n_points / 2) > 0]
                
                for k in k_vals:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
    
    # Final optimization pass with scipy
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Ensure valid sum
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
