import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Homotopy-based approach to escape local optima
    max_homotopy_steps = 100
    homotopy_step = 0.01
    t = 0.0
    
    for step in range(max_homotopy_steps):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Solve relaxed problem
        try:
            h_opt = cp.Variable(n_points)
            # Regularized objective: t * C5 + (1-t) * ||h - h_init||^2
            c5_value = cp.Variable()
            
            # Add constraints for the overlap term
            k_vals = np.linspace(0.1, 1.9, min(15, n_points//4))
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add the overlap constraints
            for k in k_vals:
                k_idx = int(k * n_points / 2)
                if k_idx < n_points and k_idx > 0:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k_idx], 1 - h_opt[k_idx:]))
                    constraints.append(corr_sum <= c5_value * dx + 1e-8)
            
            # Regularization term
            h_init = best_h
            reg_term = cp.sum_squares(h_opt - h_init) * (1 - t)
            
            prob = cp Problem(cp.Minimize(t * c5_value + reg_term), constraints)
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=300)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_opt.value
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except Exception:
            pass
            
        # Increment t for homotopy
        t = min(1.0, t + homotopy_step)
        
        # Try local refinement with small random perturbations
        if step % 5 == 0:
            try:
                noise = np.random.normal(0, 0.01, n_points)
                h_test = best_h + noise
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points/2) / np.sum(h_test)
                c5_test = evaluate_erdos_solution(h_test, dx)
                if c5_test < best_c5:
                    best_h = h_test.copy()
                    best_c5 = c5_test
            except Exception:
                pass
    
    # Final refinement with constrained optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
