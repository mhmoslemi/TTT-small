import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with alternating projections
    max_iter = 10000
    lr = 0.1
    momentum = 0.9
    conv_radius = 0.01
    
    # Initialize
    h = h_start.copy()
    v = np.zeros_like(h)
    
    # Create a helper function to project onto feasible set
    def project_h(h_proj):
        h_proj = np.clip(h_proj, 0, 1)
        h_sum = np.sum(h_proj)
        h_proj = h_proj * (n_points/2) / h_sum
        return h_proj
    
    # Optimization loop
    for i in range(max_iter):
        # Time check
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlap and best shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        if max_overlap < best_c5:
            best_h = h.copy()
            best_c5 = max_overlap
            
        # Gradient estimation (simplified gradient of the correlation at max)
        # This is a semi-derivative approach for non-differentiable parts
        # Estimate gradient as finite difference for simplicity
        grad = np.zeros_like(h)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = corr_plus[max_idx] * dx
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = project_h(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            max_minus = corr_minus[max_idx] * dx
            
            grad[j] = (max_plus - max_minus) / (2 * eps)
        
        # Update with momentum
        v = momentum * v - lr * grad
        h = h + v
        
        # Project
        h = project_h(h)
    
    # Final polishing
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            final_c5 = evaluate_erdos_solution(h, dx)
            if final_c5 < best_c5:
                best_h = h.copy()
                best_c5 = final_c5
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
