import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based evolutionary algorithm
    population_size = 50
    mutation_rate = 0.05
    generations = 1000
    elite_size = population_size // 5
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Main evolutionary loop
    no_improvement_count = 0
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive mutation rate
        if no_improvement_count > 100:
            mutation_rate *= 0.95
            no_improvement_count = 0
        mutation_rate = max(mutation_rate, 0.01)
        
        # Selection: keep top elite
        new_population = population[:elite_size]
        
        # Create offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent_idx = np.random.choice(elite_size, 2, replace=False)
            parent1, parent2 = population[parent_idx[0]], population[parent_idx[1]]
            
            # Crossover (averaging with some noise)
            alpha = np.random.random()
            offspring = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            noise = np.random.normal(0, mutation_rate, n_points)
            offspring += noise
            offspring = np.clip(offspring, 0, 1)
            
            # Repair to satisfy constraint
            offspring = offspring * (n_points/2) / np.sum(offspring)
            
            new_population.append(offspring)
            
        population = new_population
        
        # Periodic convex relaxation
        if gen % 50 == 0:
            try:
                # Formulate QCQP
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample k values for better coverage
                k_vals = np.linspace(0.05, 1.95, min(30, n_points//2))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if 0 < k < n_points:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=300)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    h_feasible = np.clip(h_feasible, 0, 1)
                    h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
