import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Use Hamming window as starting point
        n_points = 35
        h = 0.54 + 0.46 * np.cos(np.pi * (np.arange(n_points) / n_points - 0.5) * 2)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    # Projected gradient descent
    max_iter = 1000
    eta = 0.01
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate current correlation to get the maximum lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient with respect to max lag
        # For simplicity, we approximate the gradient numerically
        # by perturbing one element at a time
        grad = np.zeros(n_points)
        eps = 1e-6
        for j in range(n_points):
            h_pert = h.copy()
            h_pert[j] += eps
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
            grad[j] = (corr_pert[max_lag_idx] - corr[max_lag_idx]) / eps
        
        # Update h using gradient descent
        h_new = h - eta * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check if new solution is better
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new
    
    # Final refinement with SLSQP if time allows
    if time() - start_time < budget_s * 0.9:
        try:
            def objective(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            def constraint_sum(h_vec):
                return np.sum(h_vec) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
