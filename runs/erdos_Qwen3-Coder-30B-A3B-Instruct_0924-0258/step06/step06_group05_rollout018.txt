import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with random perturbations
    max_iter = 5000
    step_size = 0.02
    noise_decay = 0.995
    noise_level = 0.05
    
    for idx in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Identify worst overlap
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        max_corr = corr[k_max]
        
        # Compute gradient for this shift
        grad = np.zeros(n_points)
        if k_max < n_points:
            grad[:n_points-k_max] += (1 - best_h[k_max:])
        if k_max > 0:
            grad[k_max:] += best_h[:-k_max]
        
        # Normalize gradient and take step
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
        h_new = best_h + step_size * grad
        
        # Add noise periodically
        if idx % 100 == 0:
            h_new += np.random.normal(0, noise_level, n_points)
            noise_level *= noise_decay
            
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Occasionally refine with convex optimization
        if idx % 50 == 0 and idx > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample a few key lags for constraints
                k_vals = np.linspace(0.1, 1.9, min(15, n_points//4))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
                
    # Final local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
