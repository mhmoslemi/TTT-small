import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T = 0.5
    T_min = 1e-4
    cooling_rate = 0.99
    steps = 2000
    
    for step in range(steps):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Perturb the solution
        h_new = best_h.copy()
        
        # Identify indices with highest overlap values
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_indices = np.argsort(corr)[-int(n_points/10):]
        
        # Modify the top 10% of indices
        for i in max_indices[-int(n_points/20):]:
            # Select a random index from top 10%
            idx = np.random.randint(n_points)
            delta = np.random.choice([-0.1, 0.1])
            h_new[idx] += delta
            h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Ensure integral constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject with probability
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Cooling schedule
        T = max(T_min, T * cooling_rate)
    
    # Gradient-Based Refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Solve with constrained optimization
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        # Try with scipy.optimize.minimize
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                       constraints=constraints, options={'maxiter': 100})
        if res.success:
            h_refined = res.x
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final verification
    try:
        h_final = np.clip(best_h, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
