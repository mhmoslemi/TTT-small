import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary approach: work with integer-valued configs
    def to_binary(h):
        return (h > 0.5).astype(int) 
    
    def from_binary(b):
        return b.astype(float)
    
    def valid_config(b):
        return abs(np.sum(b) - n_points / 2.0) < 1e-6
    
    # Initialize binary configuration
    h_bin = to_binary(best_h)
    while not valid_config(h_bin):
        # Perturb until valid
        flip_idx = np.random.randint(0, n_points)
        h_bin[flip_idx] = 1 - h_bin[flip_idx]
    
    # Evaluate initial
    best_c5_bin = evaluate_erdos_solution(from_binary(h_bin), dx)
    
    # Simulated Annealing with Bit Swaps
    temp_start = 0.5
    temp_decay = 0.995
    temp_min = 1e-4
    
    sa_iter = 0
    max_sa_iter = 20000
    
    current_c5 = best_c5_bin
    current_bin = h_bin.copy()
    
    # Track best
    best_bin = current_bin.copy()
    best_c5 = current_c5
    
    # Refinement frequency
    refine_every = 100
    
    while sa_iter < max_sa_iter and (time() - start_time) < budget_s * 0.95:
        # Temperature
        temp = max(temp_start * (temp_decay ** sa_iter), temp_min)
        
        # Generate neighbor by flipping 2 random bits
        i0, i1 = np.random.choice(n_points, 2, replace=False)
        
        # Flip
        candidate_bin = current_bin.copy()
        candidate_bin[i0] = 1 - candidate_bin[i0]
        candidate_bin[i1] = 1 - candidate_bin[i1]
        
        # Check constraint satisfaction
        if not valid_config(candidate_bin):
            continue
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(from_binary(candidate_bin), dx)
        
        # Accept or reject
        if c5_candidate < current_c5 or np.random.rand() < np.exp(-(c5_candidate - current_c5) / temp):
            current_c5 = c5_candidate
            current_bin = candidate_bin.copy()
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_bin = current_bin.copy()
        
        sa_iter += 1
        
        # Periodic refinement with QCP
        if sa_iter % refine_every == 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Random sampling of lag parameters
                k_vals = np.unique(np.round(np.linspace(0.05, 1.95, min(30, n_points // 3)) * n_points / 2).astype(int))
                k_vals = k_vals[(k_vals > 0) & (k_vals < n_points)]
                
                for k in k_vals:
                    if k >= n_points:
                        continue
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500, verbose=False)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_c5 = c5_feasible
                        best_h = h_feasible
            except Exception:
                pass
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
    except Exception:
        pass

    # Ensure sum constraint
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
