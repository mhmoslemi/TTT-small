import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create triangular window as starting solution
        h_start = 1 - np.abs(np.arange(n_points) / (n_points - 1) - 0.5) * 2
        # Perturb slightly
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        # Normalize
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    population_size = 50
    max_generations = 300
    mutation_factor = 0.7
    crossover_prob = 0.9
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Define constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Normalize h to satisfy constraint
    def normalize_h(h):
        h = np.clip(h, 0, 1)
        return h * (n_points / 2) / np.sum(h)
    
    # Differential Evolution optimization
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.8:
            break
            
        # DE is quite slow, so we're sampling less frequently
        if gen % 10 != 0:
            continue
            
        # We'll try to create a population
        pop = []
        for _ in range(population_size):
            # Create initial population around current best
            h = best_h + np.random.normal(0, 0.05, n_points)
            h = normalize_h(h)
            pop.append(h)
            
        # Evaluate and get the best
        fitness_scores = [objective(h) for h in pop]
        best_idx = np.argmin(fitness_scores)
        if fitness_scores[best_idx] < best_c5:
            best_h = pop[best_idx].copy()
            best_c5 = fitness_scores[best_idx]
    
    # Refinement with L-BFGS-B
    def refine_h(h):
        # Ensure constraint
        h = normalize_h(h)
        
        # Define constraint
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100})
            if res.success:
                h_refined = res.x
                h_refined = normalize_h(h_refined)
                c5_refined = objective(h_refined)
                return h_refined, c5_refined
        except:
            pass
        return h, objective(h)
    
    # Take top candidates from recent generation and refine
    top_candidates = []
    for _ in range(10):
        h = best_h + np.random.normal(0, 0.02, n_points)
        h = normalize_h(h)
        top_candidates.append(h)
    
    refined_candidates = []
    for candidate in top_candidates:
        if time() - start_time > budget_s * 0.9:
            break
        h_refined, c5_refined = refine_h(candidate)
        refined_candidates.append((h_refined, c5_refined))
    
    # Update best if refinement improved it
    if refined_candidates:
        best_refined = min(refined_candidates, key=lambda x: x[1])
        if best_refined[1] < best_c5:
            best_h = best_refined[0]
            best_c5 = best_refined[1]
    
    return (list(best_h), best_c5, n_points)
