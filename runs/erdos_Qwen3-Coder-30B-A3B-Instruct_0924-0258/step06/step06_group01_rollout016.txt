import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on C5
    C_low = 0.0
    C_high = 0.38085
    tolerance = 1e-5
    max_iterations = 1000
    
    # Try binary search with convex optimization feasibility check
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        C_mid = (C_low + C_high) / 2.0
        
        # Convex feasibility check using CVXPY
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Overlap constraints for all shifts
        for k in range(n_points):
            # Compute the correlation sum
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_var[i] * (1 - h_var[j])
            constraints.append(expr <= C_mid * n_points / 2)
        
        # Dummy objective (feasibility check)
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=500)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_solution = h_var.value
                h_solution = np.clip(h_solution, 0, 1)
                h_solution = h_solution * (n_points / 2) / np.sum(h_solution)
                current_c5 = evaluate_erdos_solution(h_solution, dx)
                
                if current_c5 < best_c5:
                    best_h = h_solution.copy()
                    best_c5 = current_c5
                    
                C_high = C_mid
            else:
                C_low = C_mid
        except:
            C_low = C_mid
            
        if C_high - C_low < tolerance:
            break
    
    # Final refinement with optimization
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass

    return (list(best_h), best_c5, n_points)
