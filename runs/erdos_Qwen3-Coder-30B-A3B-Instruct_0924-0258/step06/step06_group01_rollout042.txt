import numpy as np
from cvxpy import *
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    # Binary search setup
    C_low = 0.0
    C_high = 0.38085
    tolerance = 1e-5
    
    best_h = h.copy()
    best_C = evaluate_erdos_solution(h, dx)
    
    # Helper to create and solve convex feasibility problem
    def check_feasibility(C):
        h_var = Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            sum(h_var) == n_points / 2
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            # The overlap term for shift k
            expr = 0
            for i in range(n_points):
                expr += h_var[i] * (1 - h_var[(i + k) % n_points])
            constraints.append(expr <= C * n_points / 2)
        
        prob = Problem(Minimize(0), constraints)
        try:
            prob.solve(solver=CVXOPT, verbose=False, max_iters=500)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                return True, h_var.value
        except:
            pass
        return False, None
    
    # Binary search loop
    while time() - start_time < budget_s * 0.9 and C_high - C_low > tolerance:
        C_mid = (C_low + C_high) / 2.0
        feasible, h_new = check_feasibility(C_mid)
        
        if feasible:
            C_high = C_mid
            if h_new is not None:
                # Project to [0,1] and normalize
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                current_C = evaluate_erdos_solution(h_new, dx)
                if current_C < best_C:
                    best_C = current_C
                    best_h = h_new.copy()
        else:
            C_low = C_mid
    
    # Final correction using local optimization
    try:
        from scipy.optimize import minimize
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if res.success:
            h_ref = res.x
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (n_points/2) / np.sum(h_ref)
            final_C = evaluate_erdos_solution(h_ref, dx)
            if final_C < best_C:
                best_C = final_C
                best_h = h_ref.copy()
    except:
        pass
    
    return (list(best_h), best_C, n_points)
