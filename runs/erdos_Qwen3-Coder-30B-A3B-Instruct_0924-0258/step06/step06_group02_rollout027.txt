import numpy as np
from time import time

from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with hexagonal proposal
    T = 0.5
    T_min = 1e-4
    cool_rate = 0.99
    steps = 2000
    
    h_current = best_h.copy()
    c5_current = best_c5
    
    # Track improvement
    no_improvement_count = 0
    max_no_improvement = 100
    
    for i in range(steps):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Focused perturbation on top overlap regions
        corr_current = np.correlate(h_current, 1 - h_current, mode='full')
        # Add a bit of bias toward regions with high overlap
        # This creates a preference for changes in areas that were previously problematic
        k_indices = np.argsort(corr_current)[-n_points//10:]  # top 10%
        
        # Choose which indices to modify
        indices_to_modify = np.random.choice(k_indices, size=max(1, n_points//20), replace=False)
        
        # Create new candidate h
        h_new = h_current.copy()
        for idx in indices_to_modify:
            # Add small random perturbation to selected indices
            perturb = np.random.uniform(-0.1, 0.1)
            h_new[idx] += perturb
            
        # Keep within bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize to fix sum to n_points / 2
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Compute overlap for new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
            no_improvement_count = 0
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
        else:
            # Probabilistically accept even if worse
            delta = c5_new - c5_current
            if np.random.rand() < np.exp(-delta / T):
                h_current = h_new.copy()
                c5_current = c5_new
            no_improvement_count += 1
            
        # Cooling schedule
        T = max(T_min, T * cool_rate)
        
        if no_improvement_count >= max_no_improvement:
            # Reset if stuck
            h_current = best_h.copy()
            c5_current = best_c5
            no_improvement_count = 0
    
    # Perform gradient refinement
    def obj_func(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
        
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Square loss around best solution
    def soft_constraint(h_vec):
        return np.square(constraint_sum(h_vec))
        
    # Use projected gradient descent (simple approach)
    h_refined = best_h.copy()
    n_iter = 100
    alpha = 1e-3
    
    for _ in range(n_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Compute gradient (approximate, but sufficient to move in right direction)
        eps = 1e-5
        grad = np.zeros(n_points)
        base_val = obj_func(h_refined)
        
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            val_plus = obj_func(h_plus)
            grad[i] = (val_plus - base_val) / eps
        
        # Update with gradient and project onto feasible set
        h_refined -= alpha * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        # Check for better solution
        new_val = obj_func(h_refined)
        if new_val < best_c5:
            best_c5 = new_val
            best_h = h_refined.copy()
            
    # Final evaluation and return
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
