import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization
    n_particles = 50
    max_iter = 200
    w = 0.7  # Inertia weight
    c1 = 1.4  # Cognitive parameter
    c2 = 1.4  # Social parameter
    v_max = 0.2  # Max velocity
    
    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_c5 = []
    
    for _ in range(n_particles):
        h = h_start.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        c5 = evaluate_erdos_solution(h, dx)
        personal_best.append(h.copy())
        personal_best_c5.append(c5)
    
    global_best = best_h.copy()
    global_best_c5 = best_c5
    
    # Adaptive lag sampling
    def get_critical_lags(h, dx, num_lags=10):
        corr = np.correlate(h, 1 - h, mode='full')
        # Get indices of largest correlations
        indices = np.argpartition(-np.abs(corr), num_lags)[:num_lags]
        return indices
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update each particle
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            v = w * v + c1 * r1 * (personal_best[i] - h) + c2 * r2 * (global_best - h)
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Update personal best
            if c5_new < personal_best_c5[i]:
                personal_best[i] = h_new.copy()
                personal_best_c5[i] = c5_new
                
                # Check global best
                if c5_new < global_best_c5:
                    global_best = h_new.copy()
                    global_best_c5 = c5_new
            
            # Update particle
            particles[i] = h_new
            velocities[i] = v
            
        # Adaptive constraint sampling
        if iter_num % 20 == 0:
            # Update critical lags based on the best current solution
            critical_lags = get_critical_lags(global_best, dx, num_lags=15)
            
    # Final SQP refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraints based on critical lags
        for k in critical_lags:
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= global_best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < global_best_c5:
                global_best = h_feasible
                global_best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final check
    global_best = np.clip(global_best, 0, 1)
    global_best = global_best * (n_points/2) / np.sum(global_best)
    final_c5 = evaluate_erdos_solution(global_best, dx)
    
    return (list(global_best), final_c5, n_points)
