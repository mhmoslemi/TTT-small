```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h to satisfy sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if abs(sum_h) > 1e-10:
        h = h * (n_points / 2) / sum_h
    else:
        h = np.ones(n_points) * 0.5
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    
    # Initialize particles
    particles = [h_start.copy() for _ in range(n_particles)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
    personal_best = [p.copy() for p in particles]
    personal_best_c5 = [evaluate_erdos_solution(p, dx) for p in particles]
    
    global_best = best_h.copy()
    global_best_c5 = best_c5
    
    T = 1.0
    T_min = 1e-6
    alpha = 0.995
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] +
                           c1 * r1 * (personal_best[i] - particles[i]) +
                           c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair to feasible space
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if c5 < personal_best_c5[i]:
                personal_best[i] = particles[i].copy()
                personal_best_c5[i] = c5

                # Update global best
                if c5 < global_best_c5:
                    global_best = particles[i].copy()
                    global_best_c5 = c5
                    if global_best_c5 < best_c5:
                        best_h = global_best.copy()
                        best_c5 = global_best_c5
                        
        # Simulated annealing acceptance
        if np.random.random() < np.exp(-best_c5 / T):
            # Randomly perturb global best
            perturb = np.random.normal(0, 0.05, n_points)
            candidate = global_best + perturb
            candidate = repair_h(candidate, n_points)
            candidate_c5 = evaluate_erdos_solution(candidate, dx)
            if candidate_c5 < global_best_c5:
                global_best = candidate.copy()
                global_best_c5 = candidate_c5
                if global_best_c5 < best_c5:
                    best_h = global_best.copy()
                    best_c5 = global_best_c5
        
        # Update temperature
        T = max(T_min, T * alpha)
        
        # Convex relaxation every 50 iterations
        if iteration % 50 == 0 and iteration > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Sample k values for constraints
                k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
                for k_val in k_vals:
                    k = int(k_val * n_points / 2)
                    if k < n_points and k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
    
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final repair and validation
    best_h = repair_h(best_h, n_points)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
```
