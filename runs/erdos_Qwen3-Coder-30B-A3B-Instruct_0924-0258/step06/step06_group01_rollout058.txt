import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # ADMM parameters
    rho = 1e-4
    max_iter = 1000
    tol = 1e-6
    last_improvement = 0
    
    z = h.copy()
    u = np.zeros(n_points)
    
    # ADMM iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Primal update: solve convex subproblem
        # min (rho/2)||z - h + u||_2^2 + (1/rho)*C5_approx(z)
        # s.t. sum(z) = n_points/2, 0 <= z[i] <= 1
        
        # Use cvxpy for convex relaxation
        z_var = cp.Variable(n_points)
        
        # Calculate approximate objective (using current h as a fixed point)
        # We will minimize the maximum correlation more precisely using a quadratic approximation
        objective = cp.Minimize(
            cp.sum_squares(z_var - h + u) + 
            (1/rho) * cp.max(cp.multiply(h, 1 - h) @ cp.kron(cp.eye(n_points), cp.ones((1, 1))) + 
                             cp.kron(cp.ones((1, 1)), cp.multiply(h, 1 - h)) @ cp.eye(n_points))
        )
        
        constraints = [cp.sum(z_var) == n_points/2, z_var >= 0, z_var <= 1]
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=100)
            z_new = z_var.value
        except:
            # fallback to simple projection
            z_new = z.copy() + u - h
            z_new = np.clip(z_new, 0, 1)
            z_new = z_new * (n_points/2) / np.sum(z_new)
        
        # Dual update
        u = u + h - z_new
        
        # Update h for next iteration
        h = z_new.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Check convergence
        if np.linalg.norm(h - z_new) < tol:
            break
            
        # Evaluate and track best
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            last_improvement = iteration
            
        # Stop if no improvement for many iterations
        if iteration - last_improvement > 50:
            break
            
    # Final refine with SLSQP if time permits
    if time() - start_time < budget_s * 0.98:
        try:
            cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                lambda x: evaluate_erdos_solution(x, dx),
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'ftol': 1e-9, 'maxiter': 200}
            )
            if res.success:
                best_h = res.x
                best_h = np.clip(best_h, 0, 1)
                best_h = best_h * (n_points/2) / np.sum(best_h)
                best_c5 = evaluate_erdos_solution(best_h, dx)
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
