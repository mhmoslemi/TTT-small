import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a low-discrepancy sequence for initialization
        def halton_sequence(n, base=2):
            sequence = []
            for i in range(n):
                x = 0
                f = 1.0 / base
                while i > 0:
                    x += (i % base) * f
                    i //= base
                    f /= base
                sequence.append(x)
            return np.array(sequence)
        
        h_start = halton_sequence(n_points)
        h_start = (h_start - np.min(h_start)) / (np.max(h_start) - np.min(h_start))
        h_start = h_start * 0.5  # Scale to average 0.5
        h_start = h_start * (n_points / 2) / np.sum(h_start)  # Ensure sum h = n/2
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization
    max_iterations = 1000
    learning_rate = 0.01
    decay = 0.999
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute gradient of total overlap
        gradient = np.zeros(n_points)
        for k in range(1, n_points):
            # Forward differences
            if k < n_points:
                gradient[:n_points-k] += (1 - best_h[k:])
            # Backward differences  
            if k < n_points:
                gradient[k:] -= best_h[:-k]
        
        # Update with gradient
        new_h = best_h - learning_rate * gradient
        new_h = np.clip(new_h, 0, 1)
        
        # Project to correct integral
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(new_h, dx)
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
        
        # Decay learning rate
        learning_rate *= decay
    
    # Convex optimization refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraints that should be satisfied
        # The full correlation is expensive, so we sample key ones
        for k in range(1, n_points // 2):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final check and normalization
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * (n_points / (2 * sum_h))
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
