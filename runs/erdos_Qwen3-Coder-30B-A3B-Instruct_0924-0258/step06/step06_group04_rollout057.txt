import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a sinusoidal pattern for better symmetry
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 / n_points))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Adaptive stochastic descent with multiple restarts
    max_iter = 300
    step_size = 0.1
    decay_rate = 0.99
    min_step = 0.001
    restart_count = 0
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Perturb
        noise = np.random.normal(0, step_size, n_points)
        h_new = best_h + noise
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        
        # Decay step size
        step_size = max(min_step, step_size * decay_rate)
        
        # Periodic restarts
        if i % 50 == 0 and i > 0:
            restart_count += 1
            if restart_count < 3:
                # Add more noise for restart
                noise = np.random.normal(0, 0.2, n_points)
                h_new = best_h + noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new

    # Phase 2: Constraint sampling refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Adaptive constraint sampling with large step sizes
        step = max(1, n_points // 8)  # Sample lags more sparsely
        sampled_lags = list(range(1, n_points // 2, step))
        
        # Add initial constraints
        for k in sampled_lags:
            # Direct correlation
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-7)
        
        # Try a slightly modified approach with fixed C value
        # This lets us do a small binary search on C if needed
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible

    except Exception:
        pass

    # Phase 3: Local refinement with gradient descent
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def gradient(h):
        # Numerical gradient
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad
    
    # This phase attempts local refinement
    try:
        h_refined = best_h.copy()
        c5_refined = best_c5
        
        # Gradient descent for a few iterations
        learning_rate = 0.01
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            grad = gradient(h_refined)
            h_new = h_refined - learning_rate * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < c5_refined:
                h_refined = h_new
                c5_refined = c5_new
            else:
                break
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    except Exception:
        pass

    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
