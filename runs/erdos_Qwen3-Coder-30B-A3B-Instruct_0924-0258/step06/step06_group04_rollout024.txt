import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use fewer iterations but smarter optimization
    max_iter = 300
    step_size = 0.05
    min_step = 0.001
    
    # Try multiple random starts to avoid local optima
    for _ in range(5):
        if time() - start_time > budget_s * 0.98:
            break
            
        h_current = h_start.copy()
        # Slight randomness for start
        noise = np.random.normal(0, 0.01, n_points)
        h_current = np.clip(h_current + noise, 0, 1)
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Gradient-based update using finite differences
            c5_current = evaluate_erdos_solution(h_current, dx)
            
            # Compute gradient via finite differences
            grad = np.zeros(n_points)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_current.copy()
                h_plus[j] = min(1.0, h_current[j] + eps)
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                grad[j] = (c5_plus - c5_current) / eps
            
            # Update with gradient descent
            h_new = h_current - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Accept new point if better
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                
            # Decrease step size
            step_size = max(min_step, step_size * 0.995)
    
    # Final quadratic programming refinement if time allows
    try:
        if time() - start_time < budget_s * 0.9:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add key constraints for lags
            lags_to_check = [1, 2, 3, 5, 7, 10, 15, 20]
            for k in lags_to_check:
                if k < n_points:
                    # Adjust for zero-padding in correlation
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final normalization and verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
