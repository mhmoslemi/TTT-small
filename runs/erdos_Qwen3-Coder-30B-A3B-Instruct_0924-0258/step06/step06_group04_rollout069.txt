```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a structured pattern
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO Parameters
    n_particles = 50
    n_iter = 200
    inertia_w = 0.7
    c1 = 1.4
    c2 = 1.4
    velocity_limit = 0.2
    
    # Initialize PSO
    particles = [h_start.copy() for _ in range(n_particles)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
    pbest = [p.copy() for p in particles]
    pbest_scores = [evaluate_erdos_solution(p, dx) for p in particles]
    gbest = best_h.copy()
    gbest_score = best_c5
    
    # PSO Loop
    for iter_num in range(n_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample critical lags for better constraints
        def get_critical_lags(h, num=10):
            corr = np.correlate(h, 1 - h, mode='full')
            # Get indices of top overlaps
            top_indices = np.argsort(corr)[-num:][::-1]
            # Convert to actual lag values
            return [i - (n_points - 1) for i in top_indices]
            
        critical_lags = get_critical_lags(best_h, 10)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (
                inertia_w * velocities[i] +
                c1 * r1 * (pbest[i] - particles[i]) +
                c2 * r2 * (gbest - particles[i])
            )
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            
            # Update position
            particles[i] += velocities[i]
            
            # Boundary handling and normalization
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            score = evaluate_erdos_solution(particles[i], dx)
            
            # Update particle best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                
                # Update global best
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    best_h = particles[i].copy()
                    best_c5 = score
        
        # Periodic refinement with SQP
        if iter_num % 20 == 0 and iter_num > 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add constraints for critical lags
                for k in critical_lags:
                    start_idx = max(0, k)
                    end_idx = min(n_points, n_points + k)
                    if k >= 0:
                        corr_sum = cp.sum(cp.multiply(
                            h_opt[start_idx:end_idx], 
                            1 - h_opt[k:end_idx+k]
                        ))
                    else:
                        corr_sum = cp.sum(cp.multiply(
                            h_opt[:end_idx], 
                            1 - h_opt[-k:end_idx-k]
                        ))
                    constraints.append(corr_sum <= gbest_score * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < gbest_score:
                        gbest = h_feasible
                        gbest_score = c5_feasible
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
    
    # Final normalization and evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
```
