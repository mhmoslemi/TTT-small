import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initial setup
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    h_start = np.clip(h_start, 0, 1)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization parameters
    max_iter = 10000
    momentum = 0.9
    learning_rate = 0.05
    patience = 100
    
    # For gradient estimation
    prev_grad = np.zeros(n_points)
    best_improvement = 0
    no_improvement_count = 0
    
    # Optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient approximation
        h_current = best_h.copy()
        eps = 1e-6
        grad = np.zeros(n_points)
        
        # Finite difference gradient estimation
        for i in range(n_points):
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            
            val_plus = evaluate_erdos_solution(h_plus, dx)
            val_minus = evaluate_erdos_solution(h_minus, dx)
            grad[i] = (val_plus - val_minus) / (2 * eps)
        
        # Update with momentum
        velocity = momentum * prev_grad + (1 - momentum) * grad
        prev_grad = velocity
        
        # Take step
        h_new = h_current - learning_rate * velocity
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if improvement
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improvement_count = 0
            best_improvement = new_c5
        else:
            no_improvement_count += 1
            
        # Early stopping
        if no_improvement_count > patience:
            break
            
    # Final refinement with local optimizer
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        if res.success:
            final_h = res.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points/2) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass

    # Ensure the final solution meets constraints
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)

    return (list(final_h), final_c5, n_points)
