import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with projected updates
    max_iter = 10000
    learning_rate = 0.1
    momentum = 0.9
    conv_radius = 0.01
    patience = 100
    patience_counter = 0
    prev_best = float('inf')
    
    # Initialize velocity for momentum
    velocity = np.zeros(n_points)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlap and gradient estimation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Compute a convex relaxation of the non-convex term
        # We'll compute a quadratic approximation of the overlap term
        # around the current solution
        # For each shift k, we compute a local quadratic approximation
        # Here use the gradient of max overlap with respect to h
        h_grad = np.zeros(n_points)
        for k in range(n_points):
            # This is a simplified approximation - in practice might want exact derivative
            # But for efficiency, we will approximate the gradient by finite differences
            h_plus = best_h.copy()
            h_plus[k] += conv_radius
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            overlap_plus = np.max(corr_plus) * dx
            
            # Compute gradient approximation
            h_grad[k] = (overlap_plus - max_overlap) / conv_radius
        
        # Apply momentum
        velocity = momentum * velocity - learning_rate * h_grad
        new_h = best_h + velocity
        
        # Project back to feasible set:
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate the new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            patience_counter = 0
            prev_best = best_c5
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            break
            
    # Local refinement with scipy minimize
    try:
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda x: evaluate_erdos_solution(x, dx),
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons_sum,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
