import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a sinusoidal-like pattern
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(2 * np.pi * x / 2)
        h_start = np.clip(h_start, 0, 1)
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on C5 bound
    low, high = 0.3807, 0.381
    tolerance = 1e-6
    max_iterations = 50
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        C = (low + high) / 2.0
        
        # Formulate LP
        h_opt = cp.Variable(n_points)
        
        # Constraint 1: h in [0,1]
        bounds = [h_opt >= 0, h_opt <= 1]
        
        # Constraint 2: sum = 1
        normalization = [cp.sum(h_opt) == n_points / 2.0]
        
        # Constraint 3: overlap constraints
        constraints = bounds + normalization
        
        # For all possible lags
        for k in range(n_points):
            if k == 0:
                continue
            lag_expr = 0
            if k < n_points:
                lag_expr = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                lag_expr = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(lag_expr <= C * dx + 1e-8)
        
        # Solve problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
                high = C
            else:
                low = C
        except Exception:
            low = C
    
    # Final optimization with gradient descent
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # project onto feasible set
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # use a simple gradient descent to refine
    h_current = best_h.copy()
    step_size = 0.01
    decay = 0.999
    max_iter = 5000
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # gradient estimate
        grad = np.zeros_like(h_current)
        for k in range(1, n_points):
            # approximate gradients of correlations
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            corr = corr * dx
            max_k_idx = np.argmax(corr)
            if k == max_k_idx:
                # compute w.r.t to h_current
                if k < n_points:
                    for i in range(n_points - k):
                        grad[i] += (1 - h_current[i+k])
                else:
                    for i in range(k - n_points + 1, n_points):
                        grad[i] -= (1 - h_current[i - k + n_points - 1])
        
        # descent step
        h_new = h_current - step_size * grad
        h_new = project(h_new)
        c5_new = objective(h_new)
        
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            
        h_current = h_new
        step_size *= decay

    # Final correction
    final_h = project(best_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
