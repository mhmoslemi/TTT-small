import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Homotopy method to escape local minima
    max_homotopy_steps = 100
    homotopy_step = 1.0 / max_homotopy_steps
    
    # Start with a simple initial point
    h_current = h_start.copy()
    current_c5 = best_c5
    
    # Track best solution found
    best_iter_h = h_current.copy()
    best_iter_c5 = current_c5
    
    for t in np.linspace(0, 1, max_homotopy_steps):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Define the homotopy problem
        try:
            h_opt = cp.Variable(n_points)
            # Objective combines original term with relaxation
            corr = cp.sum(cp.multiply(h_opt[:-1], 1 - h_opt[1:]))
            # Add regularization to the original solution
            regularization = cp.sum_squares(h_opt - np.ones(n_points) * 0.5)
            
            # Minimize convex relaxation of the problem
            objective = cp.Minimize(
                t * corr + (1 - t) * regularization
            )
            
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < current_c5:
                    h_current = h_feasible.copy()
                    current_c5 = c5_feasible
                    
                if c5_feasible < best_iter_c5:
                    best_iter_h = h_feasible.copy()
                    best_iter_c5 = c5_feasible
                    
        except Exception as e:
            pass
        
        # Local improvement using scipy
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # SLSQP with random perturbations
            hpert = h_current + np.random.normal(0, 0.01, n_points)
            hpert = np.clip(hpert, 0, 1)
            hpert = hpert * (n_points/2) / np.sum(hpert)
            
            res = minimize(objective, hpert, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-8})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                
                if c5_refined < current_c5:
                    h_current = h_refined.copy()
                    current_c5 = c5_refined
                    
                if c5_refined < best_iter_c5:
                    best_iter_h = h_refined.copy()
                    best_iter_c5 = c5_refined
                    
        except Exception as e:
            pass
        
        # Update global best
        if best_iter_c5 < best_c5:
            best_h = best_iter_h.copy()
            best_c5 = best_iter_c5
    
    # Final correction to ensure sum constraint
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
