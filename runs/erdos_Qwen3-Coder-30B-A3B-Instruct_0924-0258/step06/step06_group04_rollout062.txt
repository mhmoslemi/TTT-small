import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Grid refinement from small to large
    resolutions = [10, 20, 35]  # Fixed set to control computation
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize for current resolution
        if res <= len(h_start):
            h_current = h_start[:res].copy()
            h_current = h_current * (res/2) / np.sum(h_current)
        else:
            # Interpolate from previous resolution
            x_orig = np.linspace(0, 2, len(h_start))
            x_new = np.linspace(0, 2, res)
            h_current = np.interp(x_new, x_orig, h_start)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (res/2) / np.sum(h_current)
        
        n_points_current = res
        dx_current = 2.0 / n_points_current
        
        # Simulated Annealing + Gradient Descent Hybrid
        T = 0.1
        alpha = 0.1
        decay_rate = 0.99
        max_iter = 200
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Gradient descent step
            h_new = h_current.copy()
            noise = np.random.normal(0, T, n_points_current)
            h_candidate = h_new + noise
            
            # Project to feasible region
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points_current/2) / np.sum(h_candidate)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_candidate, dx_current)
            
            # Accept or reject
            if c5_new < best_c5:
                h_current = h_candidate
                best_c5 = c5_new
            elif np.random.rand() < np.exp(-(c5_new - best_c5) / T):
                h_current = h_candidate
                
            # Cooling and step size decay
            T *= decay_rate
            alpha *= decay_rate
            
        # CVXPY refinement on final step if solution improves
        try:
            h_opt = cp.Variable(n_points_current)
            constraints = [
                cp.sum(h_opt) == n_points_current / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Sample key lags for constraints
            step = max(1, n_points_current // 10)
            for k in range(1, n_points_current, step):
                if k < n_points_current // 2:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points_current-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points_current+1:], 1 - h_opt[:n_points_current-(k-n_points_current+1)]))
                constraints.append(corr_sum <= best_c5 * dx_current + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx_current)
                if c5_feasible < best_c5:
                    h_current = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass
        
        # Update best solution
        best_h = h_current
        dx = dx_current
        n_points = n_points_current
    
    # Final evaluation
    best_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), best_c5, n_points)
