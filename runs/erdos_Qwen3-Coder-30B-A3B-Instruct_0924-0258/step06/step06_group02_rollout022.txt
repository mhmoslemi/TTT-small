import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a structured initialization for better starting point
        # Create a smooth profile using sine wave pattern
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization
    max_iterations = 1000
    learning_rate = 0.01
    decay = 0.995
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Compute gradient
        gradient = np.zeros(n_points)
        for k in range(1, n_points):
            # Contribution from h[i] * (1 - h[i+k])
            if k < n_points:
                gradient[:n_points-k] += (1 - best_h[k:])
            # Contribution from (1 - h[i]) * h[i-k] 
            if k < n_points:
                gradient[k:] -= best_h[:n_points-k]
        
        # Update step
        new_h = best_h - learning_rate * gradient
        new_h = np.clip(new_h, 0, 1)
        
        # Maintain integral constraint
        sum_h = np.sum(new_h)
        if sum_h > 0:
            new_h = new_h * (n_points / 2) / sum_h
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
        
        # Decay learning rate
        learning_rate *= decay
    
    # Final validation and cleanup
    final_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(final_h)
    if sum_h > 0:
        final_h = final_h * (n_points / 2) / sum_h
    
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
