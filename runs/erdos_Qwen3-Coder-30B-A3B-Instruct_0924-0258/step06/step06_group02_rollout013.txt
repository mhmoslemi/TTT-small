import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with structured approach
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.zeros(n_points)
        # Create symmetric structure inspired by root systems
        centers = np.array([0.5, 1.0, 1.5])
        for c in centers:
            idx = int(c / 2 * n_points)
            h_start[max(0, idx-5):min(n_points, idx+5)] = 0.8
        # Normalize to correct integral
        h_start = h_start * (n_points / (2 * np.sum(h_start)))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Constraint bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimizer with adaptive trust region
    def trust_region_optimization():
        nonlocal best_h, best_c5
        tr_radius = 0.3
        min_tr_radius = 1e-6
        max_tr_radius = 1.0
        max_iter = 1000
        iter_count = 0
        
        while iter_count < max_iter and time() - start_time < budget_s * 0.95:
            if time() - start_time > budget_s * 0.90:
                break
                
            iter_count += 1
            
            # Sample perturbation within trust region
            noise = np.random.normal(0, tr_radius, n_points)
            h_new = best_h + noise
            
            # Clip to bounds
            h_new = np.clip(h_new, 0, 1)
            
            # Project to correct integral
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                # Expand trust region
                tr_radius = min(tr_radius * 1.1, max_tr_radius)
            else:
                # Contract trust region
                tr_radius = max(tr_radius * 0.9, min_tr_radius)
                
    try:
        trust_region_optimization()
    except:
        pass
    
    # SDP refinement attempt (simplified)
    try:
        # Solve a simpler SDP relaxation
        h_sdp = cp.Variable(n_points)
        constraints = [
            cp.sum(h_sdp) == n_points / 2,
            h_sdp >= 0,
            h_sdp <= 1
        ]
        
        # Work with the highest overlaps by approximating
        # This is a rough approximation for that term
        corr_parts = []
        for k in range(1, min(10, n_points//2)):
            corr_term = cp.sum(cp.multiply(h_sdp[:n_points-k], 1 - h_sdp[k:]))
            corr_parts.append(corr_term)
        
        if corr_parts:
            max_corr = cp.max(cp.vstack(corr_parts)) 
            prob = cp.Problem(cp.Minimize(max_corr * dx), constraints)
            prob.solve(solver=cp.ECOS, verbose=False, eps=1e-5)
            if prob.status == cp.OPTIMAL:
                h_refine = h_sdp.value
                c5_refine = evaluate_erdos_solution(h_refine, dx)
                if c5_refine < best_c5:
                    best_h = h_refine.copy()
                    best_c5 = c5_refine
    except:
        pass
    
    # Final Project
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * (n_points / (2 * sum_h))
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
