import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the given construction if available, otherwise create a triangular one
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 40
        # Create triangular window
        h_start = 1 - np.abs(np.arange(n_points) / (n_points - 1) - 0.5) * 2
        # Add small random perturbation
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # Ensure the starting solution meets constraints
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # DE parameters
    popsize = 50
    maxiter = 300
    mutation_rate = 0.7
    crossover_rate = 0.9
    bounds = [(0, 1) for _ in range(n_points)]

    def de_objective(h):
        return evaluate_erdos_solution(h, dx)

    # DE bounds constraint function
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    # Create a wrapper for differential evolution that respects constraints
    def de_fitness(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points/2) / np.sum(x)
        return de_objective(x)

    # Evaluate initial solution
    current_best = best_c5

    # Begin DE loop
    for iteration in range(maxiter):
        if time() - start_time > budget_s * 0.8:
            break

        # Use differential evolution with bounds
        try:
            result = differential_evolution(
                de_fitness,
                bounds,
                popsize=popsize,
                maxiter=1,
                mutation=(mutation_rate,),
                recombination=crossover_rate,
                seed=seed + iteration,
                disp=False
            )
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                c5_new = de_objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass

    # Local refinement using L-BFGS-B on top 10 solutions
    try:
        top_candidates = []
        for _ in range(10):
            h_trial = h_start.copy()
            h_trial += np.random.normal(0, 0.01, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points/2) / np.sum(h_trial)
            top_candidates.append(h_trial)
        
        # Evaluate candidates
        fitness_scores = []
        for h in top_candidates:
            fit = evaluate_erdos_solution(h, dx)
            fitness_scores.append(fit)
        
        # Refine best candidates
        best_idx = np.argmin(fitness_scores)
        best_candidate = top_candidates[best_idx]
        
        def objective_lbfgs(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum_lbfgs(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum_lbfgs}
        bounds_lbfgs = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective_lbfgs, best_candidate, method='L-BFGS-B', 
                       bounds=bounds_lbfgs, constraints=cons_sum, 
                       options={'maxiter': 100})
        
        if res.success:
            h_refined = res.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            c5_refined = objective_lbfgs(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception as e:
        pass

    return (list(best_h), best_c5, n_points)
