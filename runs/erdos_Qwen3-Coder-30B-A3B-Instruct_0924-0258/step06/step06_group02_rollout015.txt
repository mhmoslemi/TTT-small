import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def check_feasibility(C_candidate, n, dx, h_start=None):
    if h_start is None:
        h_start = np.ones(n) * 0.5
    
    # Randomly sample shifts
    k_samples = np.random.choice(range(1, n), size=max(10, int(n * 0.2)), replace=False)
    
    # Define decision variables
    h_opt = cp.Variable(n)
    
    # Integral constraint
    constraints = [cp.sum(h_opt) == n / 2]
    
    # Bounds
    constraints += [0 <= h_opt, h_opt <= 1]
    
    # Overlap constraints for sampled shifts
    for k in k_samples:
        if k < n:
            corr_sum = cp.sum(cp.multiply(h_opt[:n-k], 1 - h_opt[k:]))
        else:
            corr_sum = cp.sum(cp.multiply(h_opt[k-n+1:], 1 - h_opt[:n-(k-n+1)]))
        constraints.append(corr_sum <= C_candidate / dx)
    
    # Formulate and solve problem
    prob = cp.Problem(cp.Minimize(0), constraints)
    prob.solve(solver=cp.ECOS, eps=1e-6)
    
    return prob.status == cp.OPTIMAL

def run(seed=42, budget_s=600, **kwargs):

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search approach to find tight upper bound
    low = 0.38
    high = 0.38085
    best_c5 = high
    iterations = 20
    sample_rate = 0.2
    
    for _ in range(iterations):
        if time() - start_time > budget_s * 0.85:
            break
            
        mid = (low + high) / 2
        feasible = check_feasibility(mid, n_points, dx, h_start)
        
        if feasible:
            best_c5 = mid
            high = mid
        else:
            low = mid
    
    # Final refinement with all shifts and ECOS
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add overlap constraints for all k
        for k in range(1, n_points):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final polishing with local search
    try:
        h_test = best_h.copy()
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points/2) / np.sum(h_test)
        c5_test = evaluate_erdos_solution(h_test, dx)
        if c5_test < best_c5:
            best_h = h_test
            best_c5 = c5_test
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
