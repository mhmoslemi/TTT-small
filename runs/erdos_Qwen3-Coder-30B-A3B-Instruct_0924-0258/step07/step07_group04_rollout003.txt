import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_h(n_points):
    x = np.linspace(0, 2, n_points)
    # Use a periodic-like initialization that's more structured
    h_initial = 0.5 * (1 + np.sin(2 * np.pi * x))
    h_initial = np.clip(h_initial, 0, 1)
    normalization = (n_points / 2) / np.sum(h_initial)
    h_initial *= normalization
    return h_initial

    start_time = time()

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_size = 20
    tabu_list = []
    max_iter = 1000
    T = 1.0
    alpha = 0.998
    stagnation_limit = 50
    
    h_current = h_start.copy()
    current_score = best_c5
    
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by perturbing a random subset of points
        indices = np.random.choice(n_points, size=max(1, int(n_points * 0.15)), replace=False)
        h_candidate = h_current.copy()
        
        for i in indices:
            # Perturb by a random amount between -15% and +15%
            perturb = np.random.uniform(-0.15, 0.15)
            h_candidate[i] += h_candidate[i] * perturb
            
        # Ensure bounds
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Renormalize to correct integral
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        
        # Evaluate candidate
        candidate_score = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject based on simulated annealing
        if candidate_score < current_score:
            # Always accept better solutions
            h_current = h_candidate.copy()
            current_score = candidate_score
            if candidate_score < best_c5:
                best_h = h_candidate.copy()
                best_c5 = candidate_score
                stagnation_count = 0
        else:
            # Probabilistically accept worse solutions
            delta = candidate_score - current_score
            if np.random.rand() < np.exp(-delta / T):
                h_current = h_candidate.copy()
                current_score = candidate_score
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Update temperature
        T = max(0.001, T * alpha)
        
        # Update tabu list
        tabu_list.append(h_current.copy())
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
        
        # Early stopping if stagnation
        if stagnation_count > stagnation_limit:
            break
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)

            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP',

                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
