import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_h(n_points):
    x = np.linspace(0, 2, n_points)
    h_initial = 0.5 * (1 + np.sin(2 * np.pi * x))
    h_initial = np.clip(h_initial, 0, 1)
    normalization = (n_points / 2) / np.sum(h_initial)
    h_initial *= normalization
    return h_initial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = initialize_h(n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.998
    max_iter = 10000
    tabu_size = 20
    tabu_list = []
    
    h_current = h_start.copy()
    c5_current = best_c5
    
    stagnation_count = 0
    max_stagnation = 500
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate
        candidate_idx = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        h_candidate = h_current.copy()
        
        # Perturb selected indices
        for i in candidate_idx:
            perturbation = np.random.uniform(-0.15, 0.15)
            h_candidate[i] += perturbation
            
        # Clip and normalize
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        delta = c5_candidate - c5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_current = h_candidate
            c5_current = c5_candidate
            
            # Update tabu list
            tabu_list.append(h_candidate.copy())
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
                
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Cooling schedule
        T *= alpha
        
        # Stagnation check
        if stagnation_count > max_stagnation:
            # Random restart with small perturbation
            h_current = best_h + np.random.normal(0, 0.01, n_points)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            c5_current = evaluate_erdos_solution(h_current, dx)
            stagnation_count = 0
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
