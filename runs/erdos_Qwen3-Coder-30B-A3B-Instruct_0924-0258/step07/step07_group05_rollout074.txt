import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Basis function construction
    m = 10  # Number of basis functions
    basis_functions = []
    
    # Create piecewise linear basis functions with compact support
    for i in range(m):
        basis = np.zeros(n_points)
        start_idx = int(i * n_points / m)
        end_idx = int((i + 1) * n_points / m)
        mid_idx = (start_idx + end_idx) // 2
        
        if end_idx - start_idx > 1:
            # Create triangular pulse
            for j in range(start_idx, end_idx):
                if j < mid_idx:
                    basis[j] = (j - start_idx) / (mid_idx - start_idx)
                else:
                    basis[j] = (end_idx - j) / (end_idx - mid_idx)
        
        basis_functions.append(basis)
    
    basis_matrix = np.array(basis_functions).T
    coefficients = np.random.uniform(0.1, 0.9, m)
    norm_factor = np.sum(basis_matrix @ coefficients)
    coefficients *= (n_points / 2) / norm_factor
    
    # Trust-region optimization
    max_iter = 5000
    trust_radius = 0.1
    patience = 0
    max_patience = 50
    perturbation_magnitude = 0.05
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Construct h from basis
        h_current = basis_matrix @ coefficients
        h_current = np.clip(h_current, 0, 1)
        current_sum = np.sum(h_current)
        if abs(current_sum - n_points / 2) > 1e-6:
            h_current = h_current * (n_points / 2) / current_sum
            
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        # Update best solution
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            patience = 0
        else:
            patience += 1
            
        # Adaptive perturbation
        if patience > max_patience:
            coefficients += np.random.normal(0, perturbation_magnitude, m)
            perturbation_magnitude *= 0.5
            patience = 0
            
        # Gradient approximation and update using trust-region
        grad = np.zeros(m)
        epsilon = 1e-6
        
        for i in range(m):
            coef_perturbed = coefficients.copy()
            coef_perturbed[i] += epsilon
            h_perturbed = basis_matrix @ coef_perturbed
            h_perturbed = np.clip(h_perturbed, 0, 1)
            sum_perturbed = np.sum(h_perturbed)
            if abs(sum_perturbed - n_points / 2) > 1e-6:
                h_perturbed = h_perturbed * (n_points / 2) / sum_perturbed
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            grad[i] = (c5_perturbed - c5_current) / epsilon
        
        # Update coefficients with simple trust-region step
        try:
            # Solve trust region subproblem: min ||g^T d + 1/2 d^T B d||, ||d|| <= trust_radius
            direction = -grad
            norm_dir = np.linalg.norm(direction)
            if norm_dir > trust_radius:
                direction *= trust_radius / norm_dir
            coefficients += direction
            coefficients = np.clip(coefficients, 0, 10)
        except:
            pass
            
        # For better convergence, occasionally ensure feasibility
        if iteration % 100 == 0:
            h_working = basis_matrix @ coefficients
            h_working = np.clip(h_working, 0, 1)
            sum_working = np.sum(h_working)
            if abs(sum_working - n_points / 2) > 1e-6:
                coefficients *= (n_points / 2) / sum_working

    # Final refinement with local optimization using scipy
    try:
        def objective(c):
            h_test = basis_matrix @ c
            h_test = np.clip(h_test, 0, 1)
            sum_h = np.sum(h_test)
            if abs(sum_h - n_points / 2) > 1e-6:
                h_test = h_test * (n_points / 2) / sum_h
            return evaluate_erdos_solution(h_test, dx)
        
        def constraint_sum(c):
            h_test = basis_matrix @ c
            h_test = np.clip(h_test, 0, 1)
            return np.sum(h_test) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 10) for _ in range(m)]
        
        res = minimize(objective, coefficients, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            coefficients = res.x
            h_final = basis_matrix @ coefficients
            h_final = np.clip(h_final, 0, 1)
            sum_final = np.sum(h_final)
            if abs(sum_final - n_points / 2) > 1e-6:
                h_final = h_final * (n_points / 2) / sum_final
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
