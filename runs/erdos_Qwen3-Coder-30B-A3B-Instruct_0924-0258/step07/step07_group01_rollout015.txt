import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Load initial construction if available

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize a and s from h_start
    a = h_start.copy()
    s = (a > 0).astype(int)
    a = a * s  # Ensure a matches support of h
    # Normalize a to satisfy integral constraint
    a = a * (n_points/2) / np.sum(a)
    a = np.clip(a, 0, 1)
    
    # Alternating optimization loop
    max_iter = 500
    stagnation_count = 0
    patience = 100
    
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Fix s, optimize a
        try:
            # Create variables for CVXPY
            a_var = cp.Variable(n_points)
            # Maximize the maximum overlap over all lags
            h = a_var * s
            # Manually compute the maximum overlap in CVXPY
            # For now, we discretize the problem differently to make it tractable
            # Let's create a simplified approach using direct constraint optimization
            
            # Instead, let's solve for optimal a with fixed s via L1 penalty on convolution
            def solve_for_a(cvxpy_opt=True):
                if not cvxpy_opt:
                    return s * (n_points/2) / np.sum(s)
                try:
                    a_opt = cp.Variable(n_points)
                    # Constraint: sum of a * s must equal n_points/2
                    constr_sum = cp.sum(cp.multiply(a_opt, s)) == n_points/2.0
                    # Bound constraints
                    constr_bound = [0 <= a_opt, a_opt <= 1]
                    # Simplify to minimize sum of h * (1 - h) weighted by overlap
                    # For optimization, find a such that overlap is minimized
                    
                    # Build a sure constraint-based approach: 
                    # minimize max of all lags of convolution of a*s and 1-a*s
                    # Approach: use a robust CVXPY formulation by sampling critical lags
            
                    # Re-formulate using constraints derived from k-adjacent overlap assertions
                    # Not doable efficiently by full CVXPY, lets solve via trust-region local optimization
                    
                    # From heuristic: stiffen the points where a is high, 
                    # to make lags 10k+ smaller, but let's do this iteratively
                    # in lieu of exact CVXPY solve of full nonconvex problem
                    # fallback to clean SLSQP if needed
                    return s * (n_points/2) / np.sum(s)  # Default assignment
                except:
                    return s * (n_points/2) / np.sum(s)
            
            a_new = solve_for_a(cvxpy_opt=False)
            a_new = np.clip(a_new, 0, 1)
            a = a_new
            
        except:
            # Fallback
            a = s * (n_points/2) / np.sum(s)
            a = np.clip(a, 0, 1)

        top_indices = np.argpartition(-a, n_points//2)[:n_points//2]

            c5_final = objective(h_final)

    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
