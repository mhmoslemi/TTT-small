import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    convex_interval = 50
    max_generations = 1000
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Main GA loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Selection (tournament)
        new_population = []
        for _ in range(pop_size):
            # Tournament selection
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            best_index = tournament_indices[np.argmin(tournament_fitness)]
            new_population.append(population[best_index].copy())
        
        # Crossover and mutation
        for i in range(0, pop_size, 2):
            if np.random.rand() < crossover_rate and i+1 < pop_size:
                # Uniform crossover
                parent1 = new_population[i]
                parent2 = new_population[i+1]
                mask = np.random.rand(n_points) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
                
                # Mutation
                if np.random.rand() < mutation_rate:
                    mutation_indices = np.random.choice(n_points, 
                                                       int(mutation_rate * n_points))
                    child1[mutation_indices] += np.random.normal(0, 0.03, len(mutation_indices))
                
                if np.random.rand() < mutation_rate:
                    mutation_indices = np.random.choice(n_points, 
                                                       int(mutation_rate * n_points))
                    child2[mutation_indices] += np.random.normal(0, 0.03, len(mutation_indices))
                
                # Repair constraints
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points/2) / np.sum(child1)
                
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points/2) / np.sum(child2)
                
                new_population[i] = child1
                new_population[i+1] = child2
            else:
                # Mutation only
                if np.random.rand() < mutation_rate:
                    mutation_indices = np.random.choice(n_points, 
                                                       int(mutation_rate * n_points))
                    new_population[i][mutation_indices] += np.random.normal(0, 0.03, len(mutation_indices))
                    new_population[i] = np.clip(new_population[i], 0, 1)
                    new_population[i] = new_population[i] * (n_points/2) / np.sum(new_population[i])
        
        # Replace population
        population = new_population
        
        # Update best solution
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        
        # Convex relaxation every convex_interval generations
        if generation % convex_interval == 0 and generation > 0:
            try:
                # Get top candidates for convex relaxation
                sorted_indices = np.argsort(fitness)
                top_candidates = [population[i] for i in sorted_indices[:min(5, len(population))]]
                
                for h_candidate in top_candidates:
                    # Try to improve with convex optimization
                    h_opt = cp.Variable(n_points)
                    constraints = [
                        cp.sum(h_opt) == n_points / 2.0,
                        h_opt >= 0,
                        h_opt <= 1
                    ]
                    
                    # Add constraints with some k values
                    k_vals = np.linspace(0.05, 1.95, min(15, n_points//3))
                    for k_val in k_vals:
                        k = int(k_val * n_points / 2)
                        if k < n_points and k > 0:
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                    
                    prob = cp.Problem(cp.Minimize(0), constraints)
                    prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                    if prob.status == cp.OPTIMAL:
                        h_feasible = h_opt.value
                        if np.all(h_feasible >= 0) and np.all(h_feasible <= 1):
                            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                            if c5_feasible < best_c5:
                                best_h = h_feasible
                                best_c5 = c5_feasible
            except Exception:
                pass
                
        # Occasionally add diversity
        if generation % 100 == 0:
            for i in range(0, pop_size, 5):
                if np.random.rand() < 0.3:
                    h_new = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points/2) / np.sum(h_new)
                    population[i] = h_new
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
