import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a more sophisticated initial guess
        x = np.linspace(0, 2, n_points)
        # Smoother version of previous design
        h_start = 0.5 * (1 + np.sin(np.pi * x))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # More targeted optimization using scipy minimize
    # with constraints and bounds
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
    ]
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try multiple random restarts
    for restart in range(10):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Start from perturbed version of best
        h_perturbed = best_h + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        try:
            result = minimize(
                objective,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if result.success:
                h_opt = result.x
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
        except:
            continue
    
    # Final refinement using convex optimization approach
    try:
        # Gate the number of terms for efficiency
        n_terms = min(10, n_points // 2)
        if n_terms < 2:
            n_terms = 2
            
        # Create basis matrix
        x_space = np.linspace(0, 2, n_points)
        basis = np.zeros((n_points, n_terms))
        for i in range(n_terms):
            basis[:, i] = np.cos(np.pi * i * x_space / 2)
        
        # Create convex problem for coefficients
        c = cp.Variable(n_terms)
        h_cvx = basis @ c
        
        # Overlap constraint
        corr_terms = []
        # Approximate max correlation with a feasible one
        # We consider a few leading lag values
        for k in range(min(10, n_points//2)):
            if k == 0:
                continue  # Skip zero lag as it's not in the sum
            else:
                # Manually compute
                corr_sum = cp.sum(cp.multiply(h_cvx[:n_points-k], 1 - h_cvx[k:]))
                corr_terms.append(corr_sum)
        
        # Using sum of all terms with good maximum approximation
        max_term = cp.max(corr_terms) if corr_terms else cp.sum(h_cvx[:n_points//2])
        
        constraints = [
            cp.sum(h_cvx) == n_points / 2.0,
            h_cvx >= 0,
            h_cvx <= 1
        ]
        
        # Minimize max correlation (as proxy for final C5)
        prob = cp.Problem(cp.Minimize(max_term), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_cvx.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
        
        # Add additional refinement if any

    except Exception:
        pass

    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
