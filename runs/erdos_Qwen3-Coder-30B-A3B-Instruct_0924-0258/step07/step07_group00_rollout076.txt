import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Main optimization loop
    max_iter = 10000
    current_iter = 0
    step_size = 0.01
    
    while time() - start_time < budget_s * 0.95 and current_iter < max_iter:
        # Evaluate all overlaps to find the maximum
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        max_lag = max_idx - (len(h_current) - 1)
        
        # Compute gradient for the worst-case lag
        grad = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                grad[i] += 1 - 2 * h_current[i + max_lag]
        else:
            for i in range(-max_lag, n_points):
                grad[i] += 1 - 2 * h_current[i + max_lag]
        
        # Gradient descent step
        h_new = h_current - step_size * grad
        
        # Clip to feasible bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if improvement or if within tolerance
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h_current = h_new.copy()
            step_size = min(0.05, step_size * 1.1)  # Increase step size if successful
        else:
            # Decrease step size and try again
            step_size = max(1e-6, step_size * 0.5)
            h_current = h_new.copy()
        
        current_iter += 1
        
        # Occasionally re-evaluate best solution
        if current_iter % 100 == 0:
            c5_eval = evaluate_erdos_solution(best_h, dx)
            if c5_eval < best_c5:
                best_c5 = c5_eval
    
    # Final refinement using scipy optimizer
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-10})
        if res.success:
            final_h = res.x
            final_c5 = objective(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
