import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create periodic initial function for better starting point
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    # Parameters
    max_iter = 10000
    T = 0.5  # Initial temperature
    alpha = 0.995  # Cooling rate
    T_min = 1e-5
    
    # Tabu list
    tabu_list = []
    tabu_size = 20
    
    # Perturbation parameters
    perturbation_size = 0.1
    min_perturbation_size = 0.05
    
    # Stagnation counter
    stagnation_count = 0
    max_stagnation = 50
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate solution by perturbing 15% of points
        candidate_h = current_h.copy()
        indices_to_change = np.random.choice(n_points, size=int(0.15 * n_points), replace=False)
        
        for idx in indices_to_change:
            perturbation = np.random.uniform(-perturbation_size, perturbation_size)
            candidate_h[idx] += perturbation
            
        # Ensure bounds
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Normalize to satisfy integral constraint
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        
        # Evaluate the candidate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Accept or reject the candidate
        if candidate_c5 < current_c5:
            # Always accept if better
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            stagnation_count = 0
        else:
            # Accept with probability based on temperature
            delta = candidate_c5 - current_c5
            acceptance_prob = np.exp(-delta / T)
            if np.random.random() < acceptance_prob:
                current_h = candidate_h.copy()
                current_c5 = candidate_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Update best if found
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        
        # Manage tabu list
        tabu_list.append(current_h.copy())
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
            
        # Decrease temperature
        T = max(T * alpha, T_min)
        
        # Adaptive perturbation size
        if T < 0.1:
            perturbation_size = max(min_perturbation_size, perturbation_size * 0.99)
        
        # Stagnation check
        if stagnation_count > max_stagnation:
            # Restart with more diverse solution
            # Perturb the best solution
            restart_h = best_h.copy()
            for i in range(n_points):
                if np.random.random() < 0.1:  # 10% chance to perturb each point
                    restart_h[i] += np.random.uniform(-0.1, 0.1)
            restart_h = np.clip(restart_h, 0, 1)
            restart_h = restart_h * (n_points / 2) / np.sum(restart_h)
            current_h = restart_h.copy()
            current_c5 = evaluate_erdos_solution(current_h, dx)
            stagnation_count = 0
            T *= 0.5  # Increase temperature to allow more exploration
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
