import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm parameters
    pop_size = 100
    max_generations = 250
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = 10
    tournament_size = 5
    
    # Initialize population
    population = []
    fitness = []
    
    # Create initial population
    for i in range(pop_size):
        if i == 0 and 'initial_h_values' in globals():
            h = h_start.copy()
        else:
            # Random construction
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
        f = evaluate_erdos_solution(h, dx)
        fitness.append(f)
    
    # Hall of fame for top solutions
    hall_of_fame = []
    hall_fitness = []
    
    # EA loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        # Update hall of fame
        for i in range(min(elite_size, len(sorted_pop))):
            if len(hall_of_fame) < elite_size:
                hall_of_fame.append(sorted_pop[i].copy())
                hall_fitness.append(sorted_fit[i])
            else:
                if sorted_fit[i] < hall_fitness[-1]:
                    hall_of_fame[-1] = sorted_pop[i].copy()
                    hall_fitness[-1] = sorted_fit[i]
                    hall_fitness.sort()
                    
        # Select parents for reproduction
        selected_indices = []
        for i in range(pop_size):
            # Tournament selection
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[j] for j in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected_indices.append(winner_idx)
            
        # Create new population
        new_population = []
        new_fitness = []
        
        # Elitism - keep best solutions
        for i in range(elite_size):
            new_population.append(sorted_pop[i].copy())
            new_fitness.append(sorted_fit[i])
            
        # Generate offspring
        while len(new_population) < pop_size:
            # Select parents
            parent1_idx, parent2_idx = np.random.choice(selected_indices, 2, replace=False)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
                
            # Mutation
            if np.random.rand() < mutation_rate:
                # Gaussian mutation
                mutation_strength = 0.1 * (1 - generation / max_generations)
                noise = np.random.normal(0, mutation_strength, n_points)
                child += noise
                # Clip and renormalize
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                
            # Check if it's a bit-flip mutation
            if np.random.rand() < 0.15 and len(hall_of_fame) > 0:
                # Select from hall of fame
                elite = hall_of_fame[np.random.randint(len(hall_of_fame))]
                # Flip some values
                flip_idx = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                child[flip_idx] = elite[flip_idx]
                # Renormalize
                child = child * (n_points/2) / np.sum(child)
                
            # Add to new population
            new_population.append(child)
            new_fitness.append(evaluate_erdos_solution(child, dx))
            
        # Replace population
        population = new_population[:pop_size]
        fitness = new_fitness[:pop_size]
        
        # Track best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
    
    # Final refinement using constrained optimization
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraints to match current best (but don't enforce too tightly)
        # Use a sample of recent best correlations
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Take top lags for constraint feasibility
        for i, corr_val in enumerate(corr):
            if i < 5 or np.random.rand() < 0.1:
                k = i - (n_points - 1)
                # Extract correlation corresponding to lag k
                if k < 0:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points+k], 1 - h_opt[-k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k:], 1 - h_opt[:n_points-k]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-8)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=2000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
