import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with constraints
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def constraint_bounds(h):
        return np.concatenate([h, 1.0 - h])
    
    cons = [{'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': lambda h: constraint_bounds(h)}]
    
    # Use L-BFGS-B with constraints
    result = minimize(objective, best_h, method='L-BFGS-B', 
                      bounds=[(0, 1) for _ in range(n_points)],
                      constraints=cons, 
                      options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9})
    
    if result.success:
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points/2) / np.sum(h_opt)
        c5_opt = evaluate_erdos_solution(h_opt, dx)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    # Post-processing: try small PSO refinement for further improvement
    if time() - start_time > budget_s * 0.9:
        return (list(best_h), best_c5, n_points)
    
    # PSO refinement
    pop_size = 20
    max_iter = 50
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    velocity_max = 0.1
    
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for i in range(pop_size):
        h = best_h + np.random.uniform(-0.05, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
        c5 = evaluate_erdos_solution(h, dx)
        pbest.append(h.copy())
        pbest_c5.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        for i in range(pop_size):
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            cognitive_force = cognitive * r1 * (pbest[i] - particles[i])
            social_force = social * r2 * (best_h - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive_force + social_force
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to constraint
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
                
                if c5 < best_c5:
                    best_h = particles[i].copy()
                    best_c5 = c5
    
    # Final refinement using convex optimization
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraints from top lags
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        top_lags = np.argsort(corr)[-5:]
        for k in top_lags:
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
