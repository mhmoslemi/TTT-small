import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    num_generations = 200
    num_parents = population_size // 5
    mutation_rate = 0.3
    stagnation_limit = 50
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    best_fitness_history = []
    stagnation_count = 0
    
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness_scores = []
        for h in population:
            fitness = evaluate_erdos_solution(h, dx)
            fitness_scores.append(fitness)
        
        # Track best
        current_best_idx = np.argmin(fitness_scores)
        current_best_fitness = fitness_scores[current_best_idx]
        best_fitness_history.append(current_best_fitness)
        
        if current_best_fitness < best_c5:
            best_c5 = current_best_fitness
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > stagnation_limit:
            break
            
        # Selection (take top 20%)
        selected_indices = np.argsort(fitness_scores)[:num_parents]
        selected_population = [population[i] for i in selected_indices]
        
        # Crossover and mutation to create new population
        new_population = selected_population.copy()
        while len(new_population) < population_size:
            # Tournament selection for two parents
            parent1 = selected_population[np.random.randint(0, len(selected_population))]
            parent2 = selected_population[np.random.randint(0, len(selected_population))]
            
            # Arithmetic crossover
            child = (parent1 + parent2) / 2.0
            
            # Mutation
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child = child + noise
                child = np.clip(child, 0, 1)
                
            # Normalize
            child = child * (n_points/2) / np.sum(child)
            new_population.append(child)
            
        population = new_population[:population_size]
    
    # Local refinement with SLSQP on top 5 solutions
    fitness_scores = [evaluate_erdos_solution(h, dx) for h in population]
    top_indices = np.argsort(fitness_scores)[:5]
    
    for idx in top_indices:
        h_refined = population[idx].copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
