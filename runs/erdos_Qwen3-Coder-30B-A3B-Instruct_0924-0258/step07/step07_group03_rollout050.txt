import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based optimization
    max_iter = 2000
    lr = 0.1
    patience = 50
    no_improve_count = 0
    
    # Random restarts
    for restart in range(5):
        if time() - start_time > budget_s * 0.8:
            break
            
        h_current = h_start.copy()
        current_c5 = best_c5
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute correlations
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_k = np.argmax(corr)
            max_val = corr[max_k]
            
            # Compute gradient for the worst case
            grad = np.zeros_like(h_current)
            # Compute gradient contribution from lag k
            lag = max_k - (n_points - 1)  # Convert full correlation index to lag
            if lag >= 0:
                # h_current[i] contributes to correlation with 1-h_current[i+lag]
                for i in range(n_points - lag):
                    grad[i] += (1 - h_current[i + lag])
            else:
                # h_current[i] contributes to correlation with 1-h_current[i+lag]
                lag_abs = -lag
                for i in range(n_points - lag_abs):
                    grad[i + lag_abs] += (1 - h_current[i])
            
            # Gradient descent step (try different learning rates)
            h_new = h_current - lr * grad
            
            # Project to feasible region
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Compute new value
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            if new_c5 < current_c5:
                h_current = h_new.copy()
                current_c5 = new_c5
                no_improve_count = 0
                if current_c5 < best_c5:
                    best_h = h_current.copy()
                    best_c5 = current_c5
            else:
                no_improve_count += 1
                if no_improve_count > patience:
                    lr *= 0.5
                    no_improve_count = 0
                    if lr < 1e-6:
                        break
    
    # Phase 2: Final refinement with L-BFGS
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='L-BFGS-B', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
