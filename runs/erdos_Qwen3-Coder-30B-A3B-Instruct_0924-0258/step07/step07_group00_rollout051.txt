import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)  # Ensure sum constraint
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Gradient descent parameters
    max_iter = 10000
    step_size = 0.01
    tolerance = 1e-9
    
    iter_count = 0
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all overlaps and find max
        corr = np.correlate(h_current, 1 - h_current, mode='full') 
        max_idx = np.argmax(corr)
        max_lag = max_idx - (n_points - 1)  # Convert to actual lag
        
        # Compute gradient w.r.t. h for the max lag
        # The overlap for lag k is sum(h[i] * (1 - h[i+k]))
        gradient = np.zeros(n_points)
        
        if max_lag >= 0:
            # h[i] * (1 - h[i + max_lag]) for i = 0, ..., n_points - max_lag - 1
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_current[i + max_lag]
            # h[i + max_lag] * (1 - h[i]) for i = 0, ..., max_lag - 1  
            for i in range(max_lag):
                gradient[i + max_lag] += -h_current[i]
        else:
            # max_lag is negative, so shift is positive
            # h[i] * (1 - h[i + max_lag]) where max_lag < 0
            # i + max_lag >= 0, so i >= -max_lag
            max_lag_abs = -max_lag
            for i in range(max_lag_abs, n_points):
                gradient[i] += 1 - h_current[i - max_lag_abs]
            # h[i + max_lag] * (1 - h[i]) where i + max_lag < 0, i.e. i < -max_lag  
            for i in range(max_lag_abs):
                gradient[i] += -h_current[i + max_lag_abs]
        
        # Apply step
        h_new = h_current - step_size * gradient
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)  # Renormalize to sum to n_points/2
        
        # Check constraint satisfaction
        if abs(np.sum(h_new) - n_points/2) > 1e-6:
            step_size *= 0.5
            if step_size < 1e-10:
                break
            continue
            
        # Compute new overlap
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject the step
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Update current solution
        h_current = h_new.copy()

        iter_count += 1
        
    # Final refinement using scipy minimize with constraints
    if time() - start_time < budget_s * 0.9:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
