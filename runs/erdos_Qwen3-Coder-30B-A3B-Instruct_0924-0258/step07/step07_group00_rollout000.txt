import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient-based optimization with focus on maximum overlap
    max_iter = 10000
    step_size = 0.01
    stagnation_count = 0
    max_stagnation = 50
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        # If we've improved, update best
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                break
        
        # Compute gradient for the maximum overlap lag
        # For lag k, the gradient w.r.t. h[i] is (1 - h[i+k])
        # But we need to handle boundary conditions carefully
        
        # Find the maximal overlap and apply gradient
        max_lag = max_lag_idx - (n_points - 1)  # Correct for full correlation
        
        # Gradient calculation for the specific lag
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            # h[i] contributes 1-h[i+lag] to the overlap
            # gradient[i] = -h[i+lag] for i in range(n_points - lag)
            for i in range(n_points - max_lag):
                gradient[i] -= (1 - h[i + max_lag])
        else:
            # max_lag < 0, so h[i] contributes 1-h[i-lag] to the overlap
            # gradient[i] = -h[i-lag] for i in range(-lag, n_points)
            for i in range(-max_lag, n_points):
                gradient[i] -= (1 - h[i + max_lag])
        
        # Apply gradient descent step
        h_new = h - step_size * gradient
        
        # Clip to feasible range [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if step caused improvement
        h = h_new
        
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
