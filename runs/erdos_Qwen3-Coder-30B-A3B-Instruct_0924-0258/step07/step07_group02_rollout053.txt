import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def repair_solution(h, n_points):
    # Clip to [0,1] and renormalize
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5  # Default uniform
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 50
    generations = 200
    mutation_rate = 0.05
    crossover_rate = 0.7
    elite_size = 5
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = best_h.copy()
        else:
            # Random perturbation of best solution
            noise = np.random.normal(0, mutation_rate * 0.5, n_points)
            h = best_h + noise
        h = repair_solution(h, n_points)
        population.append(h)
    
    # EA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update global best
        if sorted_fitness[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = sorted_fitness[0]
        
        # Selection: tournament
        new_population = []
        for i in range(pop_size):
            # Keep top elitists
            if i < elite_size:
                new_population.append(sorted_population[i].copy())
            else:
                # Tournament selection
                tourn_size = 5
                tourn_indices = np.random.choice(pop_size, tourn_size, replace=False)
                tourn_fitness = [fitness[i] for i in tourn_indices]
                winner = tourn_indices[np.argmin(tourn_fitness)]
                new_population.append(population[winner].copy())
        
        # Crossover and mutation
        for i in range(elite_size, pop_size):
            if np.random.rand() < crossover_rate and i+1 < pop_size:
                # Blend crossover
                parent1 = new_population[i]
                parent2 = new_population[i+1]
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
                new_population[i] = child
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_rate * 0.5, n_points)
                new_population[i] += noise
            new_population[i] = repair_solution(new_population[i], n_points)
        
        population = new_population
        
        # Reduce mutation rate
        mutation_rate = max(0.005, mutation_rate * 0.98)
    
    # Final SQP refinement of top solutions
    try:
        top_solutions = sorted([(evaluate_erdos_solution(h, dx), h) for h in population])[:10]
        for _, h_test in top_solutions:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add constraint on top lags to help convergence
            corr = np.correlate(h_test, 1 - h_test, mode='full')
            top_lags = np.argsort(corr)[-5:][::-1]
            for k in top_lags:
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
    except Exception:
        pass
    
    # Final verification
    best_h = repair_solution(best_h, n_points)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
