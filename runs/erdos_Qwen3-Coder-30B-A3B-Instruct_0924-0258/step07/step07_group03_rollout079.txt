import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based optimization
    max_iter = 1000
    lr = 0.1
    patience = 50
    no_improve_count = 0
    
    h_current = h_start.copy()
    target_sum = n_points / 2.0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute all correlations to find the worst case
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        
        # Compute gradient for the worst case lag
        grad = np.zeros_like(h_current)
        lag = max_idx - (len(h_current) - 1)
        
        # Construct the gradient by considering the contribution of
        # each point to the correlation at the max index
        for i in range(len(h_current)):
            j = i + lag
            if 0 <= j < len(h_current):
                grad[i] += (1 - h_current[j])
            j = i - lag
            if 0 <= j < len(h_current):
                grad[i] += (1 - h_current[j])
        
        # Apply gradient step
        h_new = h_current - lr * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        scale_factor = target_sum / np.sum(h_new)
        h_new = h_new * scale_factor
        
        # Evaluate
        current_c5 = evaluate_erdos_solution(h_new, dx)
        
        if current_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = current_c5
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        h_current = h_new.copy()
        
        if no_improve_count > patience:
            lr *= 0.5
            no_improve_count = 0
            
            if lr < 1e-6:
                break
    
    # Phase 2: Final local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='L-BFGS-B', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
