import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def refine_with_SLSQP(h_init, dx, n_points):
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1)] * n_points
    
    result = minimize(objective, h_init, method='SLSQP',
                      constraints=cons, bounds=bounds,
                      options={'maxiter': 500, 'ftol': 1e-9})
    if result.success:
        h_refined = result.x
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2.0) / np.sum(h_refined)
        return h_refined
    return h_init

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2.0) / np.sum(h)
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Adaptive gradient descent with momentum
    eta = 0.01
    momentum = 0.9
    v = np.zeros(n_points)
    no_improvement_count = 0
    max_no_improvement = 100
    
    max_iterations = 1000
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlaps and gradient
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        max_k = np.argmax(overlaps)
        
        # Approximate gradient based on the maximum overlap constraint
        gradient = np.zeros(n_points)
        for i in range(n_points):
            for k in range(-n_points + 1, n_points):
                j = (i + k) % n_points
                if np.abs(k) == max_k:
                    gradient[i] += (h[j] - 1) * dx
        
        # Add momentum
        v = momentum * v - eta * gradient
        h = h + v
        
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
        
        # Update best solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
            eta = max(eta, 0.01)
        else:
            eta = max(eta * 0.5, 1e-6)
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final refinement with SLSQP
    best_h = refine_with_SLSQP(best_h, dx, n_points)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
