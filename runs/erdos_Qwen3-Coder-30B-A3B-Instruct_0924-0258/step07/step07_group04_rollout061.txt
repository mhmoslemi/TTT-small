import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a better initial guess with periodic function
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))  # 5 cycles
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated annealing with tabu search
    # Parameters
    T = 0.5
    alpha = 0.995
    T_min = 1e-5
    tabu_size = 20
    max_iter = 1000
    stagnation_limit = 50
    
    # Tabu list to store recent solutions
    tabu_list = []
    
    # Perturbation size that decreases over time
    perturbation_size = 0.1
    
    stagnation_count = 0
    iteration = 0
    
    while time() - start_time < budget_s * 0.95:
        if iteration >= max_iter:
            break
            
        # Generate candidate solution
        candidate_h = best_h.copy()
        
        # Select indices to modify (15% of points)
        indices_to_modify = np.random.choice(n_points, size=int(0.15 * n_points), replace=False)
        
        # Perturb selected points
        for i in indices_to_modify:
            # Add random perturbation
            delta = np.random.uniform(-perturbation_size, perturbation_size)
            candidate_h[i] += delta
        
        # Clip values to [0,1]
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Normalize to maintain integral constraint
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(candidate_h, dx)
        
        # Check if candidate is in tabu list
        is_tabu = any(np.allclose(candidate_h, h_tabu, atol=1e-6) for h_tabu in tabu_list)
        
        # Accept or reject based on SA criteria
        if c5_candidate < best_c5:
            # Always accept better solution
            best_h = candidate_h.copy()
            best_c5 = c5_candidate
            stagnation_count = 0
        elif not is_tabu:
            # Accept with probability based on temperature
            delta = c5_candidate - best_c5
            if delta > 0 and np.random.random() < np.exp(-delta / T):
                best_h = candidate_h.copy()
                best_c5 = c5_candidate
                stagnation_count = 0
        else:
            # If tabu, still might accept if it's significantly better
            delta = c5_candidate - best_c5
            if delta < -0.001 and np.random.random() < 0.5:  # 50% chance if significantly better
                best_h = candidate_h.copy()
                best_c5 = c5_candidate
                stagnation_count = 0
        
        # Add to tabu list
        tabu_list.append(best_h.copy())
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
        
        # Update so far
        iteration += 1
        stagnation_count += 1
        
        # Apply cooling
        T = max(T * alpha, T_min)
        perturbation_size = max(0.05, perturbation_size * alpha)
        
        # Restart if stagnation
        if stagnation_count > stagnation_limit:
            # Add random perturbation to best solution
            noise = np.random.normal(0, 0.05, n_points)
            best_h = best_h + noise
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            stagnation_count = 0
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final validation
    final_sum = np.sum(best_h)
    assert abs(final_sum - n_points/2) < 1e-6, "Sum constraint not met"
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "Bounds not met"
    assert best_c5 == evaluate_erdos_solution(best_h, dx), "Evaluation mismatch"
    
    return (list(best_h), best_c5, n_points)
