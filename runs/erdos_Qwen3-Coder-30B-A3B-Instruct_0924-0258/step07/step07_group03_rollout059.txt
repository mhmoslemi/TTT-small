import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient Descent with adaptive learning rate
    h = h_start.copy()
    learning_rate = 0.1
    max_iter = 1000
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute correlation and find worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        l_max = np.argmax(corr)
        current_c5 = corr[l_max] * dx

        # Compute gradient of correlation at worst lag
        # This is approximated by finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (corr_plus[l_max] - corr[l_max]) / eps
            
        # Update with gradient
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Perturb occasionally to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            idxs = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            h[idxs] += np.random.uniform(-0.1, 0.1, len(idxs))
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            
        # Adaptive learning rate
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            last_improvement = iteration
        else:
            if iteration - last_improvement > 100:
                learning_rate *= 0.5
                last_improvement = iteration
                
    # Phase 2: SQP Refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)

        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0

        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]

        res = minimize(objective, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 200, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
