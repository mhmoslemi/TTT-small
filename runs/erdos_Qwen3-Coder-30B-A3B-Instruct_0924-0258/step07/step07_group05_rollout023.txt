import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Basis function approach
    m = 10  # number of basis functions
    # Create piecewise linear basis functions
    x = np.linspace(0, 2, n_points)
    basis = np.zeros((n_points, m))
    
    # Create triangular basis functions
    for i in range(m):
        center = 2.0 * i / (m - 1) if m > 1 else 1.0
        width = 0.4
        basis[:, i] = np.maximum(0, 1 - np.abs(x - center) / width)
    
    # Normalize each basis function
    for i in range(m):
        basis[:, i] = basis[:, i] / np.max(basis[:, i]) if np.max(basis[:, i]) > 0 else basis[:, i]
    
    # Initial coefficients
    try:
        coeffs = np.linalg.lstsq(basis, h_start, rcond=None)[0]
    except:
        coeffs = np.random.rand(m) * 0.5
    coeffs = np.clip(coeffs, 0, 1)
    
    # Optimize the coefficients
    def objective(c):
        h = basis @ c
        h = np.clip(h, 0, 1)
        # Enforce integral constraint
        h = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(c):
        h = basis @ c
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return np.sum(h) - n_points/2
    
    def constraint_bounds(c):
        h = basis @ c
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return np.concatenate([h, 1 - h])
    
    # Trust region optimization
    prev_c = coeffs.copy()
    prev_c5 = objective(coeffs)
    tr_radius = 0.1
    patience = 0
    max_patience = 50
    perturbation_magnitude = 0.05
    
    for iter_num in range(10000):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Update current solution
        h = basis @ coeffs
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Optimization step
        # Generate direction using a simple gradient approximation
        eps = 1e-4
        grad = np.zeros(m)
        for i in range(m):
            c_plus = coeffs.copy()
            c_plus[i] += eps
            h_plus = basis @ c_plus
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            c_minus = coeffs.copy()
            c_minus[i] -= eps
            h_minus = basis @ c_minus
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Apply trust region update
        delta = -grad
        if np.linalg.norm(delta) > tr_radius:
            delta = delta * tr_radius / np.linalg.norm(delta)
        
        coeffs_new = coeffs + delta
        # Project to feasible region
        coeffs_new = np.clip(coeffs_new, 0, 1) 
        
        # Ensure integral constraint for new solution
        h_new = basis @ coeffs_new
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < current_c5:
            coeffs = coeffs_new
            tr_radius *= 2.0
            patience = 0
        else:
            tr_radius *= 0.5
            patience += 1
        
        # Perturbation if stuck
        if patience > max_patience:
            coeffs += np.random.normal(0, perturbation_magnitude, m)
            coeffs = np.clip(coeffs, 0, 1)
            perturbation_magnitude *= 0.5
            patience = 0
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        h_refined = np.clip(np.array(best_h), 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            best_h = res.x
            best_c5 = objective(best_h)
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
