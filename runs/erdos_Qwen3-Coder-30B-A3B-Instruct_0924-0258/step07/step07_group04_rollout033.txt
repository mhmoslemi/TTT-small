import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create periodic initialization
        n_points = 20
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multiresolution approach - start with coarse, refine
    resolutions = [20, 40, 80]
    current_n = 20
    current_h = h_start.copy()
    
    for n in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to new resolution if needed
        if n > current_n:
            x_old = np.linspace(0, 2, current_n)
            x_new = np.linspace(0, 2, n)
            current_h = np.interp(x_new, x_old, current_h)
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n / 2) / np.sum(current_h)
            current_n = n
            
        # Local optimization with simulated annealing-like approach
        T = 0.1
        cooling_rate = 0.995
        max_iter = 200 if n == 80 else 150 if n == 40 else 100
        
        for it in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Choose random points to perturb (10-20%)
            n_perturb = max(1, min(n // 5, n // 2))
            indices = np.random.choice(n, n_perturb, replace=False)
            
            h_candidate = current_h.copy()
            
            # Perturb selected points
            for i in indices:
                # Adaptive perturbation size - larger initially
                perturb_size = T * 0.2
                delta = np.random.uniform(-perturb_size, perturb_size)
                h_candidate[i] += delta
            
            # Clip to bounds
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Normalize
            h_candidate = h_candidate * (n / 2) / np.sum(h_candidate)
            
            # Evaluate
            c5_candidate = evaluate_erdos_solution(h_candidate, 2.0 / n)
            
            # Accept with Metropolis criterion
            if c5_candidate < best_c5 or np.random.random() < np.exp(-(c5_candidate - best_c5) / T):
                current_h = h_candidate.copy()
                if c5_candidate < best_c5:
                    best_h = current_h.copy()
                    best_c5 = c5_candidate
                    # Reset temperature if we got a good improvement
                    T = 0.1
            
            # Cool temperature
            T *= cooling_rate
            
    # Final SLSQP refinement
    try:
        final_h = best_h.copy()
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (current_n / 2) / np.sum(final_h)
        
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / current_n)
        
        def constraint_sum(h):
            return np.sum(h) - current_n / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(current_n)]
        res = minimize(objective, final_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            final_h = res.x
            final_c5 = objective(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, current_n)
