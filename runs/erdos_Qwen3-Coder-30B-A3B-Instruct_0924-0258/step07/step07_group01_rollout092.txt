import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize a (amplitude) and s (support)
    if 'initial_h_values' in globals():
        a = np.ones(n_points) * 0.5
        s = np.array(initial_h_values) > 0.001
        # Normalize to sum = n_points/2
        active_sum = np.sum(a * s)
        if active_sum > 0:
            a *= (n_points/2) / active_sum
        else:
            a = np.ones(n_points) * (n_points/2) / n_points
            s = np.ones(n_points, dtype=bool)
    else:
        s = np.ones(n_points, dtype=bool)
        a = np.ones(n_points) * 0.5
    
    # Ensure sum = n_points/2
    active_sum = np.sum(a * s)
    if active_sum > 0:
        a *= (n_points/2) / active_sum
    else:
        a = np.ones(n_points) * (n_points/2) / n_points
    
    h_current = a * s
    current_c5 = evaluate_erdos_solution(h_current, dx)
    if current_c5 < best_c5:
        best_h = h_current.copy()
        best_c5 = current_c5
    
    # Alternating optimization loop
    stagnation_count = 0
    max_stagnation = 100
    
    # Function to compute bounds for CVXPY
    def compute_overlap_bound(a_vals, s_vals, dx):
        # Set up CVXPY problem
        n = len(a_vals)
        # Create the actual h vector
        h = a_vals * s_vals
        # Convolution is not directly supported, so we compute max overlap manually
        # This is a simplified approach: we approximate the overlap
        # using a fast approximation for small n.
        overlap = 0.0
        for lag in range(n):
            # Compute overlap for specific lag for indicated part of h
            corr_val = 0.0
            for i in range(n):
                if s_vals[i] and (i + lag < n) and s_vals[i + lag]:
                    corr_val += h[i] * (1 - h[i + lag])
            if corr_val > overlap:
                overlap = corr_val
        return overlap * dx
    
    for it in range(5000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Step 1: Fix s, optimize a (with sum and bounds constraints)
        # Set up optimization problem
        a_cvx = cp.Variable(n_points)
        h_cvx = cp.multiply(a_cvx, s)
        # Objective: minimize max overlap
        # We compute this approximately for this iteration
        
        # But, let's just compute the current max overlap for evaluating a step
        h_temp = a * s
        current_overlap = evaluate_erdos_solution(h_temp, dx)
        
        # We can approximate the effect of variation in a using gradients from 
        # iterative contraction

        # Instead, perform local optimization on a (to improve overlap)
        # Use a heuristic refinement: raise a where it's flat and lower where it is sharp
        
        # We approximate and simply offload to SLSQP
        def obj_slsqp(a_vals):
            h_vals = a_vals * s
            h_vals = np.clip(h_vals, 0, 1)
            overlap = evaluate_erdos_solution(h_vals, dx)
            return overlap
        
        def constraint_sum(a_vals):
            return np.sum(a_vals * s) - n_points / 2.0
        
        # Perform SLSQP optimization
        try:
            res = minimize(obj_slsqp, a.copy(), method='SLSQP', 
                           bounds=[(0, 1) for _ in range(n_points)], 
                           constraints={'type': 'eq', 'fun': constraint_sum},
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                a = res.x
                h_new = a * s
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new * s)
                a = h_new / s
                a = np.clip(a, 0, 1)
        except Exception:
            pass
            
        # Step 2: Fix a, optimize s (set s to top K points by a)
        # Find top K entries (half of total)
        k = n_points // 2
        indices = np.argpartition(a, -k)[-k:]
        new_s = np.zeros(n_points, dtype=bool)
        new_s[indices] = True
        
        # Ensure sum = n_points/2
        sum_a = np.sum(a[new_s])
        if sum_a > 0:
            a = a * (n_points/2) / sum_a
        a = np.clip(a, 0, 1)
        
        # Normalization for s should be correct
        s = new_s
        
        h_current = a * s
        new_c5 = evaluate_erdos_solution(h_current, dx)
        
        if new_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = new_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count >= max_stagnation:
                break
    
    # Final refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
