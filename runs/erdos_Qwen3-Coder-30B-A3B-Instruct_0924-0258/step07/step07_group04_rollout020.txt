import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a sine-based starting point for better exploration
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(3 * np.pi * x)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe style optimization with iterative coordinate updates
    max_iter = 1000
    gamma = 0.1
    stagnation_limit = 100
    stagnation_count = 0
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Compute correlation matrix
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(corr) - 1) // 2
        
        # Create direction: negative gradient approximation
        # Approximate gradient of negative overlap w.r.t. h
        direction = np.zeros_like(h)
        # Correlation of (1-h) with h shifted
        for i in range(n_points):
            # We consider how changing h[i] affects the overlap
            # For minimal overlap we want to reduce correlation peak
            if max_lag >= 0:
                if i + max_lag < n_points:
                    direction[i] = -(1 - h[i + max_lag])
                else:
                    # Wrap around
                    direction[i] = -1 + h[i + max_lag - n_points]
            else:
                # max_lag < 0
                if i >= -max_lag:
                    direction[i] = -(1 - h[i + max_lag])
                else:
                    # Wrap around from end
                    direction[i] = -1 + h[i + max_lag + n_points]
            
        # Apply Frank-Wolfe step
        h_new = h + gamma * direction
        
        # Project back to feasible region: [0,1] and sum==1
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Update h for next iteration
        h = h_new.copy()
        
        # Occasionally perturb to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.02, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            
        if stagnation_count > stagnation_limit:
            break
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        # Use best_h as starting point for refinement
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
