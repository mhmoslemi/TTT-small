import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based refinement with perturbations
    max_iter = 1000
    learning_rate = 0.1
    patience = 50
    
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    patience_counter = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation to find maximum overlap lag
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_corr_val = corr[max_lag_idx]
        
        # Compute gradient
        grad = np.zeros_like(h_current)
        # For the worst lag, compute gradient contribution
        # We want to reduce this specific correlation contribution
        # Which corresponds to the pattern of h_i * (1 - h_{i+k})
        # Let's compute affected positions
        n_corr = len(corr)
        lag = max_lag_idx - (n_points - 1)  # lag is -n_points+1 to n_points-1
        
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                grad[i] += (1 - 2 * h_current[i]) * (1 - h_current[j])
        
        # Apply gradient descent
        h_new = h_current - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience_counter = 0
            learning_rate = 0.1  # Reset learning rate
        else:
            patience_counter += 1
            if patience_counter > patience:
                learning_rate *= 0.5
                patience_counter = 0
        
        # Update current solution
        h_current = h_new
        
        # Occasionally perturb
        if iteration % 50 == 0:
            pert_idx = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            perturbation = np.random.uniform(-0.1, 0.1, len(pert_idx))
            h_current[pert_idx] += perturbation
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Phase 2: SQP refinement 
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    try:
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use best solution as starting point for SQP
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
