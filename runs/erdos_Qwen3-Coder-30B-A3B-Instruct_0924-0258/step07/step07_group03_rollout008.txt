import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based descent with perturbations
    h = h_start.copy()
    learning_rate = 0.1
    max_iter = 1000
    perturb_freq = 50
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute overlap for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        l_max = np.argmax(corr)
        current_c5 = corr[l_max] * dx
        
        # Compute gradient with respect to the worst-lag
        # For lag l_max, the gradient
        # For correlation h * (1-h) at lag k, 
        # gradient w.r.t. h[i] is (1-h[i+k]) for i such that i+k is in valid range
        # Since h* (1-h) in full mode, we use a symmetric approach
        # Here we implement a direction that tries to reduce the worst overlap
        # The gradient direction for a particular lag involves shifting h and 1-h
        # gradient[i] = contribution to corr at l_max from h[i]
        dh = np.zeros_like(h)
        # This is to improve the worst correlation value
        # Since it's a convolution, we compute the gradient of the correlation at l_max
        
        # Simplified gradient estimate: we look at when a shift contributes to the high overlap
        # The correlation at lag l_max is sum_{i} h[i] * (1-h[i+l_max])
        # So gradient w.r.t. h[j] is -1 * (if j + l_max in range) or 1 * (if j-l_max in range) 
        # Actually, let's compute the full gradient to get a better update direction
        
        # Reconstruct a proper gradient:
        # corr[l_max] = \sum_{i} h[i] (1 - h[i + l_max])
        # d corr[l_max] / dh[j] = 1 - 2 * h[j + l_max] if (j + l_max) within bounds
        # But this is not correct either due to full convolution nature.
        # Consider that correlate gives h*inv_h with full mode, i.e., for each k, sum_{i} h[i] * (1-h[i+k])
        # Gradient w.r.t h[j] = sum_{k} of d/dh[j] of sum_{i} h[i] * (1-h[i+k]) 
        # = sum_{k: i+k=j} (1−2*h[j]) + sum_{i: j=i+k} (-h[i])  
        #                     = (1−2*h[j]) * (number of possible i such that i+k=j) - sum_{i such that j = i+k} h[i]
        # Let's think specially.
        # The gradient of the marginal element of correlation at lag k, for element j:
        # The contribution is sum_{i} h[i] (1-h[i+k]) ... so we take jth element
        # d/dh[j] = -h[j+k] if j+k valid, and +1 if i=j, so it's -h[j+k] + (not necessarily 1 but it's -
        # It's simpler with finite differences calculation:
        
        # For direction, use finite differences approximation        
        #Only focus on the dominant overlap term, and adjust the gradient direction
        # Find all positions such the overlap relates to the max lag correction
        # Let's recalculate the gradient properly based on support
        # Gradient for [i], element of h:
        # For correlation at l_max, from the full convolution
        # corr[l_max] = sum_{i} h[i] (1 - h[i + l_max])
        # For i = 0 to n_points-1:
        # Only valid terms: if i + l_max >= 0 and i + l_max < n_points (which depends on n_points, l_max)
        # That translates to i from max(0, -l_max) to min(n_points-1, n_points-1-l_max)
        
        # Compute a gradient pointing away from the highest cross-correlation.
        # Actual finite difference: 
        # Compute h_new = h + step_size * direction
        # If the effect causes the max to go down, take into account that
        # But this is not viable for gradient approximation
        
        # Simplified approach: directly compute gradient towards reducing the l_max overlap
        # Suppose we have a correlation at l_max = max, i.e. diff = x_i (1-x_{i+l_max})
        # We calculate how changing h[k] affects that sum value, which is standard 
        # When we are near the max, it helps to diffuse the values.
        # This simulates reducing the max lags sum (approximately):
        
        # Dictionary approach to better track contribution for ΔC = Σ h[i] * (1 - h[i+l_max])
        # dC / dh[j] = Σ_i d/dh[j] (h[i] * (1 - h[i + l_max]))
        # = Σ_i d/dh[j] (h[i]) * (1 - h[i + l_max]) + h[i] * d/dh[j] (1 - h[i + l_max])
        # For j=index, expect j = i or j = i + l_max  
        # If j = i, then d/dh[j] h[i] = 1, and d/dh[j] h[i+l_max] = 0
        # If j = i+l_max, then d/dh[j] h[i] = 0, and d/dh[j] h[i+l_max] = 1
        # So △C/△h[j] = 1*(1-h[j+l_max]) + h[j] * (-1) = 1 - h[j+l_max] - h[j]
        # But this has indexing issues - we define:
        # For any fixed element h[j], let's count how many terms of the correlation contributes to it.
        
        # Better idea: step in direction that reduces the maximum correlation:
        # If we have sum (h[i] * (1 - h[i + l_max])) :
        # The contribution to the derivative of term at index j = 1 - h[j] - h[j+l_max] 
        # (Note: this depends on indices if they are valid,)
        # So let's compute it properly over possible indices,
        # and make the change direction to reduce the sum.
        
        # Since the constraint is fixing ∑h = n_points/2, we fix total sum by renormalizing.
        
        # Let's do a different approach.
        # For a fixed l_max:
        # corr[l_max] = sum_i h[i] * (1 - h[i + l_max])
        dh = np.zeros(n_points)
        # Compute gradient where i + l_max is within bounds
        for i in range(n_points):
            if i + l_max < n_points:
                dh[i] -= (1 - 2 * h[i + l_max])
            if i - l_max >= 0:
                dh[i] += h[i - l_max]
        
        # Normalize: [step should be adjusted by current value]
        # Adaptive update based on scale of h
        # Can use a smarter update size
        h_new = h - learning_rate * dh
        # Clip and re-scale
        h_new = np.clip(h_new, 0, 1)
        # Normalize to sum to target
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Test the improvement
        c5_trial = evaluate_erdos_solution(h_new, dx)
        if c5_trial < current_c5:
            h = h_new.copy()
            if c5_trial < best_c5:
                best_h = h.copy()
                best_c5 = c5_trial
            stagnation_count = 0
        else:
            stagnation_count += 1
            # If stagnation, reduce learning rate
            if stagnation_count > 20:
                learning_rate *= 0.9
                if learning_rate < 1e-6:
                    break
                    
        if stagnation_count > 100:
            break
        
        # Perturb the solution periodically
        if iteration % perturb_freq == 0 and iteration > 0:
            # Add noise to a random subset
            to_perturb = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
            h[to_perturb] += np.random.uniform(-0.1, 0.1, to_perturb.shape)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
    
    # Phase 2: SQP optimization refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Start with the best found so far
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12, 'disp': False})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
