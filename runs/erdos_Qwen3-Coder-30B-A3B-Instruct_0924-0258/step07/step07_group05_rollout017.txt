import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Benders decomposition approach
    max_iter = 5000
    convergence_tol = 1e-6
    
    # Perturbation parameters
    mag = 0.05
    decay = 0.95
    
    # Track convergence
    prev_c5 = float('inf')
    
    # Initialize constraints list
    constraints_list = []
    
    # Main Benders loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all possible shifts
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Only consider overlapping parts where both functions are defined
        overlaps = corr[len(best_h)-1:]  # Only the valid overlapping parts
        
        # Find worst overlap
        k_worst = np.argmax(overlaps)
        max_overlap = overlaps[k_worst] * dx
        
        # Add constraint: overlap at k_worst should not exceed current best
        constraints_list.append((k_worst, max_overlap))
        
        # Solve convex relaxation with maintained constraints
        try:
            h_opt = cp.Variable(n_points)
            obj = cp.Minimize(0)  # dummy objective
            constraints = [cp.sum(h_opt) == n_points / 2.0,
                           h_opt >= 0,
                           h_opt <= 1]
            
            # Add all constraints
            for k, bound in constraints_list:
                if k < n_points:
                    # For a shift k, consider the overlap from h[i] to h[i+k]
                    # This is the sum over i of h[i] * (1 - h[i+k])
                    if k > 0:
                        # Compute correlations for this shift
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        constraints.append(corr_sum <= bound + 1e-6)
                        
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                h_feasible = np.clip(h_feasible, 0, 1)
                h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
                    prev_c5 = best_c5
        except:
            pass
            
        # Perturb h slightly
        perturb = mag * np.random.randn(n_points)
        best_h = best_h + perturb
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points/2) / np.sum(best_h)
        
        # Update convergence check
        new_c5 = evaluate_erdos_solution(best_h, dx)
        if abs(new_c5 - prev_c5) < convergence_tol:
            break
        prev_c5 = new_c5
        
        # Reduce perturbation magnitude
        mag *= decay
        
    # Final polishing with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
