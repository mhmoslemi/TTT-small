import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coevolutionary algorithm
    pop_size = 30
    max_iter = 300
    mut_rate = 0.1
    cross_rate = 0.8
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Random perturbation of best solution
            h = best_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Main coevolution loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all individuals
        for i in range(pop_size):
            c5 = evaluate_erdos_solution(population[i], dx)
            fitness[i] = c5
        
        # Selection: keep top 50%
        sorted_indices = np.argsort(fitness)
        elite_indices = sorted_indices[:pop_size//2]
        elite_pop = [population[i] for i in elite_indices]
        elite_fit = [fitness[i] for i in elite_indices]
        
        # Generate offspring
        offspring = []
        
        for i in range(pop_size - len(elite_pop)):
            # Tournament selection
            parent1_idx = np.random.choice(elite_indices)
            parent2_idx = np.random.choice(elite_indices)
            
            # Crossover
            if np.random.rand() < cross_rate:
                point = np.random.randint(1, n_points-1)
                child = np.concatenate([population[parent1_idx][:point], 
                                       population[parent2_idx][point:]])
            else:
                child = population[parent1_idx].copy()
            
            # Mutation
            if np.random.rand() < mut_rate:
                mut_idx = np.random.randint(n_points)
                child[mut_idx] = np.clip(child[mut_idx] + np.random.normal(0, 0.08), 0, 1)
            
            # Renormalize
            child = child * (n_points/2) / np.sum(child)
            offspring.append(child)
        
        # Replace population
        population = elite_pop + offspring
        fitness = elite_fit + [evaluate_erdos_solution(p, dx) for p in offspring]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
    
    # Final hybrid optimization with bounds
    try:
        # Use a smaller subset for convex optimization to get tighter bounds
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Create a hard constraint based on best current solution
        h_test = best_h.copy()
        corr = np.correlate(h_test, 1 - h_test, mode='full')
        max_k_idx = np.argmax(corr)
        max_lag = max_k_idx - (n_points - 1)
        
        # Approximate the maximum overlap constraint
        # Use lags around the maximum
        top_lags = np.argsort(corr)[-15:]
        for k in top_lags:
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final verification cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
