import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Differential Evolution optimization
    de_bounds = [(0, 1) for _ in range(n_points)]
    
    # DE with custom bounds enforcement
    def de_objective(h):
        h_clipped = np.clip(h, 0, 1)
        h_normalized = h_clipped * (n_points/2) / np.sum(h_clipped)
        return evaluate_erdos_solution(h_normalized, dx)
    
    try:
        # DE with custom bounds handling
        de_result = differential_evolution(
            de_objective,
            de_bounds,
            maxiter=500,
            popsize=50,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed,
            workers=1,
            updating='immediate',
            disp=False
        )
        
        if de_result.success:
            h_de = np.clip(de_result.x, 0, 1)
            h_de = h_de * (n_points/2) / np.sum(h_de)
            c5_de = evaluate_erdos_solution(h_de, dx)
            
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
                
    except Exception as e:
        pass
    
    # Local refinement with SLSQP
    try:
        def constraint_sum_slqsp(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum_slqsp}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(
            objective,

            method='SLSQP', 
            bounds=bounds, 
            constraints=cons_sum, 
            options={'maxiter': 1000, 'ftol': 1e-10}
        )
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
                
    except Exception as e:
        pass
    
    # Final polishing with a few more DE iterations on top of best solution

        # Run a few more DE iterations around the current best solution
        def fine_tune_objective(h):
            h_clipped = np.clip(h, 0, 1)
            h_normalized = h_clipped * (n_points/2) / np.sum(h_clipped)
            return evaluate_erdos_solution(h_normalized, dx)

    return (list(final_h), final_c5, n_points)
