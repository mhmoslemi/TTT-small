import numpy as np
from scipy.optimize import minimize
from time import time
from scipy.spatial.distance import pdist, squareform
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def repair_h(h, n_points):
    """Repair h vector to satisfy constraints"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - n_points/2) > 1e-10:
        h = h * (n_points/2) / h_sum
    return h

def crossover(p1, p2, alpha=None):
    """Arithmetic crossover"""
    if alpha is None:
        alpha = np.random.uniform(0.2, 0.8)
    offspring = alpha * p1 + (1 - alpha) * p2
    return offspring

def mutate(individual, mutation_rate=0.2, std=0.1):
    """Gaussian mutation with adaptive std"""
    mask = np.random.random(len(individual)) < mutation_rate
    noise = np.random.normal(0, std, len(individual))
    individual = individual + mask * noise
    return individual

def compute_entropy(fitnesses):
    """Compute entropy of fitness distribution"""
    freqs = np.bincount(fitnesses.astype(int))
    probs = freqs / len(fitnesses)
    probs = probs[probs > 0]
    return -np.sum(probs * np.log(probs))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5  # Default initialization
        
    dx = 2.0 / n_points
    
    # Initial population
    pop_size = 100
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = np.random.uniform(0, 1, n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Evolutionary Algorithm
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Hall of fame
    hall_of_fame = [(best_h.copy(), best_c5)]
    
    patience = 0
    max_patience = 80
    generation = 0
    max_generations = 300
    
    while generation < max_generations and time() - start_time < budget_s * 0.95:
        generation += 1
        
        # Evaluate population
        fitnesses = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitnesses.append(c5)
        fitnesses = np.array(fitnesses)
        
        # Update hall of fame
        for h, c5 in zip(population, fitnesses):
            if c5 < hall_of_fame[0][1]:
                hall_of_fame.append((h.copy(), c5))
                hall_of_fame.sort(key=lambda x: x[1])
                hall_of_fame = hall_of_fame[:10]
        
        # Update best solution
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitnesses[best_idx]
            patience = 0
        else:
            patience += 1
            
        if patience >= max_patience:
            break
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        elite = [population[i] for i in sorted_indices[:pop_size//10]]
        
        # Create new population
        new_pop = elite[:]
        while len(new_pop) < pop_size:
            # Selection
            idx1, idx2 = np.random.choice(sorted_indices[:pop_size//2], 2, replace=False)
            parent1, parent2 = population[idx1], population[idx2]
            
            # Crossover
            if np.random.random() < 0.7 * (1 - generation/max_generations):
                offspring = crossover(parent1, parent2)
            else:
                offspring = parent1.copy()
            
            # Mutation
            std = max(0.01, 0.1 * (1 - generation/max_generations)**2)
            offspring = mutate(offspring, std=std)
            
            # Repair
            offspring = repair_h(offspring, n_points)
            
            new_pop.append(offspring)
            
        population = new_pop[:pop_size]
    
    # Final L-BFGS refinement on top solutions
    top_solutions = [h for h, _ in hall_of_fame[:5]]
    for h in top_solutions:
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except:
            continue

    # Final verification and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
