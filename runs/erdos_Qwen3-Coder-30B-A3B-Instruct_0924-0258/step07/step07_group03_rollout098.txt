import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T0 = 0.1
    cooling_rate = 0.001
    max_iter = 1000
    perturbation_fraction = 0.15
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb solution
        num_perturb = int(n_points * perturbation_fraction)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        delta = np.random.uniform(-0.1, 0.1, num_perturb)
        
        candidate_h = current_h.copy()
        candidate_h[indices] += delta
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Normalize
        candidate_h = candidate_h * (n_points/2) / np.sum(candidate_h)
        
        # Evaluate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Accept with probability
        T = T0 * np.exp(-cooling_rate * iteration)
        if candidate_c5 < current_c5 or np.random.rand() < np.exp(-(candidate_c5 - current_c5) / T):
            current_h = candidate_h
            current_c5 = candidate_c5
            stagnation_count = 0
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            stagnation_count += 1
            
        if stagnation_count > 100:
            break
    
    # Gradient Descent refinement
    max_iter_gd = 500
    learning_rate = 0.1
    stagnation_gd = 0
    
    h_gd = best_h.copy()
    current_c5_gd = best_c5
    
    for iteration in range(max_iter_gd):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Find max overlap lag
        corr = np.correlate(h_gd, 1 - h_gd, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient for the max overlap lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                gradient[i] = (1 - h_gd[i + k_max]) * dx
            if i - k_max >= 0:
                gradient[i] += (1 - h_gd[i - k_max]) * dx
                
        # Update
        h_new = h_gd - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < current_c5_gd:
            h_gd = h_new
            current_c5_gd = new_c5
            stagnation_gd = 0
            if current_c5_gd < best_c5:
                best_h = h_gd.copy()
                best_c5 = current_c5_gd
        else:
            stagnation_gd += 1
            if stagnation_gd > 50:
                learning_rate /= 2
                stagnation_gd = 0
                
        if learning_rate < 1e-8:
            break
    
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
