import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population
    pop_size = 50
    population = []
    fitness = []
    
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Genetic algorithm parameters
    crossover_rate = 0.7
    mutation_rate = 0.2
    tournament_size = 5
    elite_size = 5
    
    generation = 0
    stagnation_count = 0
    last_best = best_c5
    
    while time() - start_time < budget_s * 0.95:
        generation += 1
        
        # Sort population by fitness
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]
        fitness = [fitness[i] for i in sorted_idx]
        
        # Check for stagnation
        if abs(last_best - best_c5) < 1e-8:
            stagnation_count += 1
        else:
            last_best = best_c5
            stagnation_count = 0
            
        if stagnation_count > 20:
            # Restart with better random solutions
            for i in range(pop_size):
                if np.random.rand() < 0.3:
                    h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.1, n_points)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points/2) / np.sum(h)
                    population[i] = h
                    fitness[i] = evaluate_erdos_solution(h, dx)
                    if fitness[i] < best_c5:
                        best_h = h.copy()
                        best_c5 = fitness[i]
            stagnation_count = 0
        
        # Create new population
        new_population = []
        new_fitness = []
        
        # Elite retention
        for i in range(elite_size):
            new_population.append(population[i].copy())
            new_fitness.append(fitness[i])
        
        # Generate rest through selection, crossover, mutation
        while len(new_population) < pop_size:
            # Tournament selection
            idx1 = tournament_selection(population, fitness, tournament_size)
            idx2 = tournament_selection(population, fitness, tournament_size)
            
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child1, child2 = uniform_crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                child1 = gaussian_mutation(child1, 0.03)
            if np.random.rand() < mutation_rate:
                child2 = gaussian_mutation(child2, 0.03)
            
            # Repair and evaluate
            child1 = clip_and_repair(child1, n_points, dx)
            child2 = clip_and_repair(child2, n_points, dx)
            
            c5_1 = evaluate_erdos_solution(child1, dx)
            c5_2 = evaluate_erdos_solution(child2, dx)
            
            new_population.extend([child1, child2])
            new_fitness.extend([c5_1, c5_2])
            
            # Update best
            if c5_1 < best_c5:
                best_h = child1.copy()
                best_c5 = c5_1
            if c5_2 < best_c5:
                best_h = child2.copy()
                best_c5 = c5_2
        
        # Trim to exact population size
        population = new_population[:pop_size]
        fitness = new_fitness[:pop_size]
        
        # Convex relaxation every 30 generations
        if generation % 30 == 0:
            try:
                # Pick top performers
                top_idx = np.argsort(fitness)[:min(10, len(fitness))]
                for i in top_idx:
                    h_candidate = population[i]
                    h_opt = cp.Variable(n_points)
                    constraints = [
                        cp.sum(h_opt) == n_points / 2.0,
                        h_opt >= 0,
                        h_opt <= 1
                    ]
                    
                    # Sample k values for constraints
                    k_vals = np.linspace(0.05, 1.95, min(15, n_points//5))
                    for k_val in k_vals:
                        k = int(k_val * n_points / 2) if int(k_val * n_points / 2) < n_points else n_points - 1
                        if k < n_points and k > 0:
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                    
                    prob = cp.Problem(cp.Minimize(0), constraints)
                    try:
                        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=300)
                        if prob.status == cp.OPTIMAL:
                            h_feasible = h_opt.value
                            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                            if c5_feasible < best_c5:
                                best_h = h_feasible
                                best_c5 = c5_feasible
                    except Exception:
                        pass
            except Exception:
                pass
    
    # Final refinement with scipy
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)

def tournament_selection(population, fitness, tournament_size):
    """Select an individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitness)]
    return winner_idx

def uniform_crossover(parent1, parent2):
    """Perform uniform crossover"""
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def gaussian_mutation(individual, mutation_strength):
    """Add Gaussian noise to individual"""
    noise = np.random.normal(0, mutation_strength, len(individual))
    mutated = individual + noise
    return mutated

def clip_and_repair(h, n_points, dx):
    """Clip values and repair sum constraint"""
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    h = h * (n_points/2) / sum_h
    return h
