import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Gradient-based optimization
    max_iter = 500
    step_size = 0.01
    tolerance = 1e-8
    decay_rate = 0.95
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current overlaps
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for the maximum overlap lag
        gradient = np.zeros(n_points)
        for j in range(n_points):
            k = (j + l_max) % n_points
            gradient[j] = -2 * dx * (1 - h[k] - h[j])
        
        # Update step
        h_new = h - step_size * gradient
        
        # Enforce bounds and sum constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check improvement
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            step_size = min(0.01, step_size * 1.1)
        else:
            step_size *= decay_rate
            
        h = h_new
        
        # Early stopping when step gets too small
        if step_size < 1e-7:
            break
    
    # Final refinement with L-BFGS-B
    try:
        def objective_and_gradient(h_vec):
            h = h_vec.reshape(n_points)
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            max_overlap = np.max(overlaps)
            l_max = np.argmax(overlaps)
            
            gradient = np.zeros(n_points)
            for i in range(n_points):
                j = (i + l_max) % n_points
                gradient[i] = -2 * dx * (1 - h[j] - h[i])
            
            return max_overlap, gradient.flatten()
        
        from scipy.optimize import fmin_l_bfgs_b
        h_flat = best_h.flatten()
        bounds = [(0, 1) for _ in range(n_points)]
        h_refined, _, _ = fmin_l_bfgs_b(objective_and_gradient, h_flat, 
                                        bounds=bounds, maxiter=100)
        
        h_refined = h_refined.reshape(n_points)
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        refined_c5 = evaluate_erdos_solution(h_refined, dx)
        if refined_c5 < best_c5:
            best_h = h_refined
            best_c5 = refined_c5
            
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
