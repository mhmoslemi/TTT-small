import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    mut_rate = 0.2
    cross_rate = 0.7
    tourney_size = 5
    elite_size = 5
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
        fit = evaluate_erdos_solution(h, dx)
        fitness.append(fit)
        if fit < best_c5:
            best_h = h.copy()
            best_c5 = fit
    
    # Main GA loop
    gen = 0
    stagnation = 0
    max_stagnation = 50
    
    while time() - start_time < budget_s * 0.95:
        gen += 1
        if stagnation > max_stagnation:
            # Restart with better solutions
            population = []
            fitness = []
            for i in range(pop_size):
                if 'initial_h_values' in globals():
                    h = h_start + np.random.normal(0, 0.1, n_points)
                else:
                    h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.1, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                population.append(h.copy())
                fit = evaluate_erdos_solution(h, dx)
                fitness.append(fit)
                if fit < best_c5:
                    best_h = h.copy()
                    best_c5 = fit
            stagnation = 0
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            stagnation = 0
        else:
            stagnation += 1
            
        # Select parents using tournament
        parents = []
        for _ in range(pop_size):
            tourney = np.random.choice(pop_size, tourney_size, replace=False)
            best_in_tourney = min(tourney, key=lambda i: fitness[i])
            parents.append(population[best_in_tourney].copy())
        
        # Create new population
        new_population = population[:elite_size]  # Elitism
        
        while len(new_population) < pop_size:
            if np.random.rand() < cross_rate:
                # Crossover
                p1, p2 = np.random.choice(pop_size, 2, replace=False)
                alpha = np.random.rand()
                child = alpha * population[p1] + (1 - alpha) * population[p2]
            else:
                # Clone
                child = population[np.random.randint(pop_size)].copy()
            
            # Mutation
            if np.random.rand() < mut_rate:
                mut_indices = np.random.choice(n_points, int(mut_rate * n_points), replace=False)
                for idx in mut_indices:
                    child[idx] += np.random.normal(0, 0.03)
                child = np.clip(child, 0, 1)
                # Re-scale to satisfy constraint
                child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
            # Compute fitness
            fit = evaluate_erdos_solution(child, dx)
            fitness.append(fit)
        
        # Keep only pop_size individuals
        population = new_population[:pop_size]
        fitness = fitness[:pop_size]
        
        # Occasionally run convex relaxation
        if gen % 30 == 0:
            for i in range(min(10, pop_size)):
                try:
                    h_opt = cp.Variable(n_points)
                    constraints = [
                        cp.sum(h_opt) == n_points / 2.0,
                        h_opt >= 0,
                        h_opt <= 1
                    ]
                    
                    # Add some overlap constraints from current individuals
                    for j in range(5):
                        idx = np.random.randint(0, pop_size)
                        h_test = population[idx]
                        corr = np.correlate(h_test, 1 - h_test, mode='full')
                        max_corr = np.max(corr) * dx
                        # Make sure one overlap is under this bound
                        k = np.argmax(corr)
                        k_pos = k - (n_points - 1)
                        k_abs = min(max(0, k_pos), n_points - 1)
                        if k_abs < n_points - 1:
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k_abs-1], 1 - h_opt[k_abs+1:]))
                            constraints.append(corr_sum <= max_corr * 1.1 + 1e-6)
                    
                    prob = cp.Problem(cp.Minimize(0), constraints)
                    prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=200)
                    if prob.status == cp.OPTIMAL:
                        h_feasible = h_opt.value
                        c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                        if c5_feasible < best_c5:
                            best_h = h_feasible
                            best_c5 = c5_feasible
                except Exception:
                    continue
    
    # Final local refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try multiple random restarts for local refinement
        for _ in range(5):
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
