import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 100
    max_generations = 300
    elite_size = pop_size // 10
    mutation_rate = 0.2
    crossover_rate = 0.7
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Add initial solution to population
    population[0] = h_start.copy()
    
    # Track hall of fame
    hall_of_fame = [(h_start.copy(), best_c5)]
    
    # Evolutionary loop
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update hall of fame
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            hall_of_fame.append((best_h.copy(), best_c5))
            hall_of_fame = sorted(hall_of_fame, key=lambda x: x[1])[:elite_size]
        
        # Create new population
        new_population = [population[0].copy()]  # Keep best
        
        # Elitism
        for i in range(1, elite_size):
            if i < len(population):
                new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_rate and len(population) >= 2:
                # Tournament selection
                idx1 = np.random.randint(0, pop_size // 2)
                idx2 = np.random.randint(0, pop_size // 2)
                p1 = population[idx1]
                p2 = population[idx2]
                
                # Arithmetic crossover
                alpha = np.random.rand()
                child = alpha * p1 + (1 - alpha) * p2
                
                # Repair
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
            else:
                # Mutation only
                idx = np.random.randint(0, pop_size // 2)
                child = population[idx].copy()
                
                # Adaptive mutation - rate decreases over time
                mutation_intensity = 0.1 * (1 - gen / max_generations)
                mutation_indices = np.random.choice(n_points, 
                                                   int(n_points * mutation_rate), 
                                                   replace=False)
                for i in mutation_indices:
                    child[i] += np.random.normal(0, mutation_intensity)
                
                # Repair
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:pop_size]
    
    # Final refinement with SLSQP
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add a constraint that the max correlation is at most best_c5
        h_test = best_h.copy()
        corr = np.correlate(h_test, 1 - h_test, mode='full')
        max_k_idx = np.argmax(corr)
        max_lag = max_k_idx - (n_points - 1)
        sample_lags = np.sort(np.argsort(corr)[-10:])  # top 10 lags
        
        # Add a relaxation over all constraints found to be active
        # by evaluating the current best solution
        
        for k in sample_lags:
            if k < n_points:  # k is the lag typically 
                l = k
                if l < n_points:
                    # calculate correlation for this lag
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-l], 1 - h_opt[l:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[l-n_points+1:], 1 - h_opt[:n_points-(l-n_points+1)]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass
    except Exception:
        pass
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
