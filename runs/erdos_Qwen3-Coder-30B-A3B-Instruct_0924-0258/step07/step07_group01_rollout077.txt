import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize problem
    h_vars = cp.Variable(n_points)
    constraints = [h_vars >= 0, h_vars <= 1]
    constraints.append(cp.sum(h_vars) == n_points / 2.0)
    
    # Problem with iterative tightening
    max_lag = 2 * n_points - 1
    corr_full = np.zeros(max_lag)
    h_vals = h_start.copy()
    
    for iteration in range(20):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        max_corr = np.max(corr) * dx
        
        # CVXPY optimization for one step
        objective = cp.Minimize(cp.max(cp.multiply(corr, dx)))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if prob.status == cp.OPTIMAL:
                h_new = h_vars.value
                if h_new is not None:
                    h_vals = np.clip(h_new, 0, 1)
                    # Adjust to sum constraint
                    h_vals = h_vals * (n_points / 2.0) / np.sum(h_vals)
                    c5_val = evaluate_erdos_solution(h_vals, dx)
                    if c5_val < best_c5:
                        best_h = h_vals.copy()
                        best_c5 = c5_val
        except Exception:
            pass
        
        # Local refinement with SLSQP
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_vals, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
            
        h_vals = best_h.copy()  # Use best so far for next iteration
    
    return (list(best_h), best_c5, n_points)
