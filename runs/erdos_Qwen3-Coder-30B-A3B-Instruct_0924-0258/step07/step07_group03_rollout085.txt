import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Convex relaxation using SDP
    max_sdp_time = budget_s * 0.7
    sdp_start = time()
    
    try:
        # Try multiple solvers for SDP
        solvers_to_try = ['ECOS', 'SCS', 'MOSEK']
        h_sdp = None
        best_sdp_c5 = float('inf')
        
        for solver in solvers_to_try:
            if time() - sdp_start > max_sdp_time:
                break
                
            try:
                # Create variables
                h = cp.Variable(n_points, name='h')
                
                # Constraints for convex relaxation
                constraints = [
                    h >= 0,
                    h <= 1,
                    cp.sum(h) == n_points / 2.0
                ]
                
                # Objective: minimize max correlation
                # This is a relaxed version of the true objective
                # We can use a relaxation approach for now
                t = cp.Variable()
                constraints.append(t >= 0)
                
                # Try to solve with a simpler relaxation first
                prob = cp.Problem(cp.Minimize(t), constraints)
                prob.solve(solver=solver, verbose=False, max_iters=1000)
                
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    h_sol = h.value
                    if h_sol is not None:
                        # Check constraint satisfaction
                        if np.all(h_sol >= 0) and np.all(h_sol <= 1):
                            c5_val = evaluate_erdos_solution(h_sol, dx)
                            if c5_val < best_sdp_c5:
                                best_sdp_c5 = c5_val
                                h_sdp = h_sol.copy()
                                
            except Exception:
                continue
            
        if h_sdp is not None:
            # If we got a good SDP result, use it as starting point
            h_start = h_sdp
            best_c5 = evaluate_erdos_solution(h_start, dx)
            best_h = h_start.copy()
            
    except Exception:
        pass
    
    # Phase 2: Local optimization with SLSQP
    max_local_time = budget_s * 0.3
    local_start = time()
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Multi-start local optimization
    num_starts = 20 if len(h_start) <= 50 else 10
    for _ in range(num_starts):
        if time() - local_start > max_local_time:
            break
            
        # Random perturbations around current best
        if np.random.rand() < 0.5:
            # Perturb the current best
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        else:
            # Random initial guess
            h_perturbed = np.random.uniform(0, 1, n_points)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        try:
            res = minimize(objective, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-10})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    
        except Exception:
            continue
    
    # Final validation check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
