import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    n_particles = 50
    max_iter = 2000
    inertia_range = [0.5, 0.9]
    step_size_range = [0.1, 0.5]
    T_start = 1.0
    cooling_rate = 0.995
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        c5 = evaluate_erdos_solution(h, dx)
        pbest.append(h.copy())
        pbest_c5.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # PSO with simulated annealing
    T = T_start
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Adaptive parameters
        inertia = inertia_range[0] + (inertia_range[1] - inertia_range[0]) * (1 - iter_num / max_iter)
        step_size = step_size_range[0] + (step_size_range[1] - step_size_range[0]) * (1 - iter_num / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            2.0 * r1 * (pbest[i] - particles[i]) + 
                            2.0 * r2 * (best_h - particles[i]))
            
            # Update position
            particles[i] += velocities[i] * step_size
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Repair constraint
            sum_h = np.sum(particles[i])
            if abs(sum_h - n_points/2) > 1e-6:
                particles[i] = particles[i] * (n_points/2) / sum_h
            
            # Evaluate
            c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
                
                if c5 < best_c5:
                    best_h = particles[i].copy()
                    best_c5 = c5
            
            # Simulated annealing
            elif c5 > best_c5 and np.random.rand() < np.exp(-(c5 - best_c5) / T):
                # Accept worse solution
                pass
        
        # Update temperature
        T *= cooling_rate
        
        # Occasionally refine with local optimization
        if iter_num % 100 == 0 and iter_num > 0:
            try:
                # Simple local refinement
                h_refined = best_h.copy()
                for k in range(n_points):
                    # Test small changes
                    h_test = h_refined.copy()
                    h_test[k] = min(1, max(0, h_test[k] + 0.01))
                    h_test *= (n_points/2) / np.sum(h_test)
                    c5_test = evaluate_erdos_solution(h_test, dx)
                    if c5_test < best_c5:
                        h_refined = h_test
                        best_c5 = c5_test
                        
                best_h = h_refined
            except Exception:
                pass
    
    # Final convex relaxation with reduced number of constraints
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add key constraints to be important for the objective
        k_vals = np.arange(1, min(n_points-1, 30))
        if len(k_vals) > 0:
            for k in k_vals:
                if k < n_points:
                    # Add constraints for non-zero lags
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
