import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct initial point as piecewise constant function
        h_start = np.ones(n_points) * 0.5
        # Add small random variation for diversity
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with bounds
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Scale up for numerical stability
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Early convergence
    max_iter = 1000
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Use minimize with L-BFGS-B
        result = minimize(objective, best_h.copy(), method='L-BFGS-B', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 10, 'ftol': 1e-9})
        
        if result.success:
            h_new = result.x
            # Ensure sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        else:
            # Fallback - random perturbation
            h_perturb = best_h + np.random.normal(0, 0.02, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
            c5_perturb = evaluate_erdos_solution(h_perturb, dx)
            if c5_perturb < best_c5:
                best_h = h_perturb.copy()
                best_c5 = c5_perturb
    
    return (list(best_h), best_c5, n_points)
