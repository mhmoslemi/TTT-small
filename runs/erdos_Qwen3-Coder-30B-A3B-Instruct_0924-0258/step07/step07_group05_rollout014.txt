    max_overlap = np.max(corr) * dx

def compute_gradient(h, dx, active_k):
    n = len(h)
    grad = np.zeros(n)
    
    # For each active k, compute gradient contribution
    for k in active_k:
        # Gradient of h_i * (1 - h_{i+k}) w.r.t. h_i is (1 - h_{i+k})
        # and w.r.t. h_{i+k} is -h_i
        if k < n:
            # Part 1: -h_i contribution to h_{i+k}
            grad[k:] -= h[:-k] * dx
            # Part 2: (1 - h_{i+k}) contribution to h_i
            grad[:-k] += (1 - h[k:]) * dx
    return grad
