import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based descent with adaptive learning rate
    h = h_start.copy()
    learning_rate = 0.1
    max_iter = 1000
    stagnation_count = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute correlation and find worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        # Compute gradient w.r.t. h using finite differences concept
        # For each position, the contribution to max overlap depends on convolution
        gradient = np.zeros_like(h)
        # The gradient calculation for the worst lag
        # For correlation h * (1-h), we can approximate gradient:
        for i in range(n_points):
            # Sum over all positions where convolution contributes
            contrib = 0.0
            for k in range(n_points):
                j = i - k
                if 0 <= j < n_points:
                    contrib += (1 - 2 * h[j]) * (1 - h[k])
            gradient[i] = contrib * dx
        
        # Update h
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        
        # Ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Random perturbations to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            perturb_idx = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            h[perturb_idx] += np.random.uniform(-0.1, 0.1, len(perturb_idx))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Check if improved
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_count = 0
            last_improvement = iteration
        else:
            stagnation_count += 1
            if stagnation_count > 50:
                learning_rate *= 0.5
                stagnation_count = 0
        
        if iteration - last_improvement > 200:
            break
    
    # Phase 2: Use SQP to refine further
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
