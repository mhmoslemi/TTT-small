```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a better initial vector using sine wave with some randomness
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA Parameters
    pop_size = 100
    max_gen = 250
    elite_size = 10
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    
    # Create initial population
    population = []
    fitness = []
    
    # Start with the best solution (and its clone)
    population.append(best_h.copy())
    fitness.append(best_c5)
    
    # Fill rest with random diverse solutions
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
        fitness.append(evaluate_erdos_solution(h, dx))
    
    gen = 0
    no_improvement = 0
    mutation_variance = 0.1
    
    while gen < max_gen and time() - start_time < budget_s * 0.95:
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Track best so far
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= 80:
            break
            
        # Selection: Tournament
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create new population
        new_population = []
        
        # Keep elites
        elite_indices = np.argsort(fitness)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring through crossover + mutation
        while len(new_population) < pop_size:
            # Select parents
            p1_idx, p2_idx = np.random.choice(pop_size, 2, replace=False)
            p1, p2 = selected[p1_idx], selected[p2_idx]
            
            # Crossover (arithmetic blending)
            if np.random.rand() < crossover_rate:
                alpha = np.random.uniform(0.2, 0.8)
                child1 = alpha * p1 + (1 - alpha) * p2
                child2 = (1 - alpha) * p1 + alpha * p2
            else:
                child1, child2 = p1.copy(), p2.copy()
            
            # Mutation (Gaussian + bit-flip)
            # Gaussian mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_variance, n_points)
                child1 += noise
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points/2) / np.sum(child1)
                
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_variance, n_points)
                child2 += noise
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points/2) / np.sum(child2)
                
            # Bit-flip mutation
            if np.random.rand() < mutation_rate:
                num_bits = int(mutation_rate * n_points)
                bit_indices = np.random.choice(n_points, num_bits, replace=False)
                for idx in bit_indices:
                    child1[idx] = 1 - child1[idx]
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points/2) / np.sum(child1)
                
            if np.random.rand() < mutation_rate:
                num_bits = int(mutation_rate * n_points)
                bit_indices = np.random.choice(n_points, num_bits, replace=False)
                for idx in bit_indices:
                    child2[idx] = 1 - child2[idx]
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points/2) / np.sum(child2)
                
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
                
        population = new_population[:pop_size]
        gen += 1
        
        # Adaptive mutation
        if gen % 50 == 0 and mutation_variance > 0.01:
            mutation_variance *= 0.9
    
    # Final local refinement using L-BFGS-B
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_func(h):
        return np.sum(h) - n_points / 2

    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Try to improve top candidates with local optimization
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    top_indices = np.argsort(fitness)[:5]
    
    for idx in top_indices:
        h_init = population[idx]
        try:
            result = minimize(
                objective,
                h_init,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints=constraints,
                options={'ftol': 1e-8, 'gtol': 1e-8}
            )
            
            if result.success:
                h_refined = result.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined.copy()
        except:
            pass
    
    # Final verification and CVXPY refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Objective is to minimize the max correlation
        # Add a rough constraint that the current best is acceptable
        # Always allow constraint that matches our current best
        h_test = best_h.copy()
        corr = np.correlate(h_test, 1 - h_test, mode='full')
        max_k_idx = np.argmax(corr)
        max_lag = max_k_idx - (n_points - 1)
        # Add all lags but make sure the maximum doesn't exceed our bound
        sample_lags = np.sort(np.argsort(corr)[-5:])
        
        for k in sample_lags:
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
            
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)

    return (list(best_h), best_c5, n_points)
```
