import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        # Create a smooth starting function
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe style optimization with coordinate-wise updates
    max_iter = 1000
    gamma = 0.1  # Learning rate
    perturb_freq = 50
    stagnation_limit = 100
    
    h_current = best_h.copy()
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current state
        current_c5 = evaluate_erdos_solution(h_current, dx)
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Periodic perturbation to escape local minima
        if iteration % perturb_freq == 0 and iteration > 0:
            perturbation = np.random.normal(0, 0.02, n_points)
            h_current = h_current + perturbation
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
            continue
            
        # Compute correlation to find worst-case overlap
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        
        # Find the lag that gives maximum correlation
        lag = max_idx - (n_points - 1)
        
        # Frank-Wolfe direction: use the gradient-like direction
        # The negative gradient direction is (1-h) - diag(max_lag_correlation)
        # But since this is discrete, approximate by local perturbation
        
        # Use the lag info to compute average of (1-h) shifted by lag
        # this gives us a rough gradient estimate
        gradient_direction = np.zeros(n_points)
        
        # For positive lag, shift 1-h to the right
        if lag >= 0:
            # Add 1-h shifted right (lag positions)
            gradient_direction[lag:] += (1 - h_current[:-lag] if lag > 0 else (1 - h_current))
        else:
            # Otherwise lag is negative, shift to the left
            gradient_direction[:lag] += (1 - h_current[-lag:] if lag < 0 else (1 - h_current))
        
        # Normalize gradient direction for unit step size
        gradient_norm = np.linalg.norm(gradient_direction)
        if gradient_norm > 0:
            gradient_direction /= gradient_norm
        
        # Apply step in this direction
        h_new = h_current + gamma * gradient_direction
        # Project back to [0,1] and maintain sum constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        h_current = h_new

    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use best solution so far as starting point
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
