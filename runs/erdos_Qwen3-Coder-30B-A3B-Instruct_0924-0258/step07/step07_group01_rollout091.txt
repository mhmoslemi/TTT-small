import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize variables for alternating optimization
    a = np.ones(n_points) * 0.5  # amplitude
    s = np.ones(n_points)        # support (binary)
    if 'initial_h_values' in globals():
        # Initialize s from the existing construction
        s = (h_start > 1e-6).astype(float)
        # Set a to the values where s=1
        a = np.where(s > 0.5, h_start, 0.0)
    else:
        # Initialize support evenly
        mid = n_points // 2
        s[:mid] = 0
        s[mid:] = 1
        a = np.where(s > 0.5, 0.5, 0.0)
    
    # Ensure sum constraint
    current_sum = np.sum(a * s)
    a = a * (n_points / 2) / current_sum
    
    current_c5 = evaluate_erdos_solution(a * s, dx)
    if current_c5 < best_c5:
        best_h = (a * s).copy()
        best_c5 = current_c5
    
    # Alternate between optimizing amplitude and support
    max_iter = 500
    patience = 0
    stagnation_limit = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Step 1: Fix s, optimize a
        # Using CVXPY to solve the convex optimization problem
        try:
            a_new = cp.Variable(n_points)
            # Objective: minimize maximum overlap (simplified approach)
            # We'll use a simpler constraint-based approach
            s_active = s > 0.5
            x_sum = cp.sum(cp.multiply(a_new, s_active))
            constraints = [
                x_sum == n_points / 2,
                a_new >= 0,
                a_new <= 1
            ]
            
            # Try to define the overlap term
            # This is a simplified version for convex optimization
            # For now just enforce constraints and re-eval
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == 'optimal':
                current_a = a_new.value
                if np.all(current_a >= 0) and np.all(current_a <= 1):
                    a = current_a.copy()
                else:
                    # fallback to gradient based method
                    a = a * (n_points/2) / (np.sum(a * s) + 1e-10)
            else:
                # fallback to projection
                a = a * (n_points/2) / (np.sum(a * s) + 1e-10)
        except:
            # Fallback to simple re-normalization
            a = a * (n_points/2) / (np.sum(a * s) + 1e-10)
        
        # Step 2: Fix a, optimize s (top-k sparsification)
        a_proj = np.abs(a.copy())  # Ensure non-negative for selection
        indices = np.argsort(a_proj)[::-1][:n_points//2]
        s_new = np.zeros(n_points)
        s_new[indices] = 1.0

        current_sum = np.sum(a * s_new)

    h_refined = best_h.copy()

    h_refined = h_refined * (n_points/2) / np.sum(h_refined)

        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
