import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 100
    max_generations = 250
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = 10
    
    # Population initialization
    population = []
    fitness = []
    
    # Add initial solution as elite
    population.append(best_h.copy())
    fitness.append(best_c5)
    
    # Initialize population
    for i in range(pop_size - 1):
        # Mix between random and Latin hypercube sampling
        if i < pop_size // 2:
            h = np.random.rand(n_points)
        else:
            # Latin hypercube sampling
            h = np.linspace(0, 1, n_points) + np.random.rand(n_points) / n_points
            h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Main EA loop
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitness[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = sorted_fitness[0]
        
        # Selection: tournament + best
        new_population = []
        new_fitness = []
        
        # Keep elite
        for i in range(elite_size):
            new_population.append(sorted_population[i])
            new_fitness.append(sorted_fitness[i])
        
        # Tournament selection
        tournament_size = 3
        while len(new_population) < pop_size:
            # Tournament selection for parent 1
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [sorted_fitness[i] for i in tournament_indices]
            parent1_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Tournament selection for parent 2 (different from parent 1 with small prob)
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [sorted_fitness[i] for i in tournament_indices]
            parent2_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Crossover with probability
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = alpha * sorted_population[parent1_idx] + (1 - alpha) * sorted_population[parent2_idx]
            else:
                child = sorted_population[parent1_idx].copy()
            
            # Mutation
            mutated = False
            if np.random.rand() < mutation_rate:
                # Add Gaussian noise
                noise = np.random.normal(0, 0.05, n_points)
                child = child + noise
                mutated = True
            
            # Bit-flip mutation
            if np.random.rand() < 0.1 and mutated:
                flip_indices = np.random.choice(n_points, int(0.1 * n_points), replace=False)
                for idx in flip_indices:
                    child[idx] = 1 - child[idx]
            
            # Repair and normalize
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
            c5 = evaluate_erdos_solution(child, dx)
            new_fitness.append(c5)
        
        population = new_population
        fitness = new_fitness
    
    # Final local refinement
    try:
        # Try to refine top 5 solutions with optimization
        top_indices = np.argsort([evaluate_erdos_solution(p, dx) for p in population])[:5]
        for i in top_indices:
            h0 = population[i]
            
            # Define optimization variables
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add constraints based on settings of best overlap
            h_test = h0.copy()
            corr = np.correlate(h_test, 1 - h_test, mode='full')
            top_lags = np.argsort(corr)[-10:][::-1]
            
            # Add some constraints for active lags
            for k in top_lags[:5]:
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= np.max(corr) * dx + 1e-8)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
    except Exception:
        pass

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
