import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
        h_start += np.random.uniform(-0.05, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multiresolution approach
    resolutions = [20, 40, 80]
    current_h = h_start.copy()
    current_n = n_points
    
    for n_res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to new resolution if needed
        if n_res > current_n:
            x_old = np.linspace(0, 2, current_n)
            x_new = np.linspace(0, 2, n_res)
            current_h = np.interp(x_new, x_old, current_h)
            current_n = n_res
            
        # Local refinement with simulated annealing
        dx_new = 2.0 / current_n
        
        # Annealing parameters
        temp = 1.0
        cooling_rate = 0.995
        min_temp = 1e-6
        max_iter = 300 if n_res == 80 else 150 if n_res == 40 else 100
        
        for it in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate candidate
            candidate = current_h.copy()
            # Mutate 10-20% of points
            n_mutate = max(1, int(0.15 * current_n))
            indices = np.random.choice(current_n, n_mutate, replace=False)
            
            for idx in indices:
                perturbation = np.random.uniform(-temp * 0.3, temp * 0.3) * candidate[idx]
                candidate[idx] += perturbation
                candidate[idx] = np.clip(candidate[idx], 0, 1)
            
            # Normalize
            candidate = candidate * (current_n/2) / np.sum(candidate)
            
            # Evaluate
            c5_candidate = evaluate_erdos_solution(candidate, dx_new)
            
            # Accept or reject
            if c5_candidate < best_c5:
                current_h = candidate.copy()
                best_c5 = c5_candidate
                if c5_candidate < best_c5:
                    best_h = current_h.copy()
                    best_c5 = c5_candidate
            else:
                # Accept with probability
                delta = c5_candidate - best_c5
                if np.random.rand() < np.exp(-delta / temp):
                    current_h = candidate.copy()
            
            temp = max(min_temp, temp * cooling_rate)
    
    # Final polishing with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (current_n/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx_new)
        
        def constraint_sum(h):
            return np.sum(h) - current_n / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(current_n)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, current_n)
