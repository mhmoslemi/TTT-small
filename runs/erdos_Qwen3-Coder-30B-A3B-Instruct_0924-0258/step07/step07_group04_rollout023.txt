import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe with adaptive step size
    max_iter = 1000
    gamma = 0.1
    
    h_current = h_start.copy()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    stagnation_count = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation and find maximum
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        # Focus on meaningful lags (symmetric)
        max_lag = max_idx - (len(corr) - 1) // 2
        # Limit lag range
        max_lag = np.clip(max_lag, -50, 50)
        
        # Create direction vector to move towards minimizing max correlation
        # This is a simplified direction: we want to go against the direction
        # that increases the max correlation
        # Since correlation is h * (1 - h_shift), let's compute the gradient available in the current concept
        direction = np.zeros(n_points)
        for i in range(n_points):
            # Approximate gradient of correlation w.r.t. h[i]
            if i + max_lag >= 0 and i + max_lag < n_points:
                # The derivative of correlation b/w h[i] and (1-h[i+lag]) wrt h[i] includes -1
                # But the correlation depends on h[i] and h[i+lag] in a more complex way
                # Approximate direction using finite differences
                direction[i] = -1.0 if max_lag >= 0 else 1.0
        
        # Simple Frank-Wolfe update using direction
        h_new = h_current - gamma * direction
        h_new = np.clip(h_new, 0, 1)
        # Normalize to correct sum
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if improvement
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = iteration
            gamma = max(0.01, gamma * 0.99)  # Adaptive step size
        else:
            gamma *= 0.99  # Small decay to prevent large steps
        
        # Stagnation checks
        if iteration - last_improvement > 100:
            # Perturb current solution
            h_current += np.random.normal(0, 0.02, n_points)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
        else:
            h_current = h_new
            
        # Random perturbation occasionally to escape local optima
        if iteration % 50 == 0 and iteration > 0:
            h_current += np.random.normal(0, 0.01, n_points)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
            
        # Periodic check for convergence
        if iteration % 100 == 0 and iteration > 0:
            if abs(evaluate_erdos_solution(h_current, dx) - best_c5) < 1e-6:
                stagnation_count += 1
            else:
                stagnation_count = 0
            if stagnation_count > 2:
                break

    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
