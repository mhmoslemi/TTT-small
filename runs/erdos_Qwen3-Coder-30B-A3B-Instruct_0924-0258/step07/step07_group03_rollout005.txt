import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient Descent with adaptive learning rate
    max_iter = 1000
    learning_rate = 0.1
    stagnation_count = 0
    last_improvement = 0
    
    h = h_start.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_value = corr[max_lag_idx]
        
        # Compute gradient contribution from worst correlation
        # The gradient with respect to h[i] accounts for how 
        # h[i] affects the overlap at the worst lag
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            # At lag corresponding to max_value, shift by i, so 1-h[i] contributes to overlap
            # The gradient of overlap wrt h[i] is (1 - 2*h[i]) at the lag
            # This is a simplification, but effective
            gradient[i] = -2 * h[i]
            
        # Apply gradient with learning rate
        h = h + learning_rate * gradient
        
        # Project back to feasible region [0,1]
        h = np.clip(h, 0, 1)
        
        # Scale to meet sum constraint
        h = h * (n_points/2) / np.sum(h)
        
        # Evaluate
        current_c5 = evaluate_erdos_solution(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Adapt learning rate based on performance
        if iteration - last_improvement > 50:
            learning_rate *= 0.9
            last_improvement = iteration
            
        # Random perturbations to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            noise_indices = np.random.choice(len(h), size=int(len(h)*0.1), replace=False)
            h[noise_indices] += np.random.uniform(-0.05, 0.05, len(noise_indices))
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
    
    # Phase 2: Local optimization (SQP)
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
