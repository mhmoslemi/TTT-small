import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        # Initialize with periodic function
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))  # 5 cycles over [0,2]
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
        # Add small random perturbations
        h_start += np.random.uniform(-0.05, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different resolutions
    resolutions = [20, 40, 80]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            h_new = np.interp(x_new, x_orig, best_h)
            h_new = np.clip(h_new, 0, 1)
            # Normalize
            h_new = h_new * (res/2) / np.sum(h_new)
            n_points = res
            dx = 2.0 / n_points
            current_h = h_new
        else:
            current_h = best_h
            
        # Simulated Annealing with adaptive perturbation
        current_c5 = evaluate_erdos_solution(current_h, dx)
        
        # Parameters
        T = 0.5  # Initial temperature
        T_min = 1e-6  # Minimum temperature
        alpha = 0.99  # Cooling rate
        max_iter = 100 if res == 20 else (200 if res == 40 else 300)
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate candidate solution
            h_candidate = current_h.copy()
            
            # Select random points to modify (10-20%)
            n_modify = max(1, min(n_points//5, int(0.15 * n_points)))
            indices = np.random.choice(n_points, n_modify, replace=False)
            
            # Perturb
            delta = T * 0.3  # Perturbation size dependent on temperature
            for idx in indices:
                h_candidate[idx] += np.random.uniform(-delta, delta)
            
            # Enforce bounds and normalize
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
            
            # Evaluate
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept or reject
            if c5_candidate < current_c5:
                current_h = h_candidate
                current_c5 = c5_candidate
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
            else:
                # Accept with probability
                delta_c = c5_candidate - current_c5
                if np.random.rand() < np.exp(-delta_c / T):
                    current_h = h_candidate
                    current_c5 = c5_candidate
                    
            # Cool down
            if T > T_min:
                T *= alpha
                
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
