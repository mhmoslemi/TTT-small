import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Better initial construction based on literature and geometric intuition
        x = np.linspace(0, 2, n_points)
        h_start = np.zeros(n_points)
        # Place a bump in the middle
        center = n_points // 2
        for i in range(n_points):
            dist = abs(i - center)
            h_start[i] = max(0, 1 - dist * 0.1)
        # Normalize
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 100
    max_generations = 300
    elite_size = pop_size // 10
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Random samples
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evaluation history
    fitness_history = []
    no_improvement_count = 0
    last_best = best_c5
    
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness_scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness_scores.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Check for improvement
        if fitness_scores[0] < best_c5:
            best_c5 = fitness_scores[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count > 80:
            break
            
        # Select elite
        elite = population[:elite_size]
        
        # Create new population
        new_population = elite[:]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Select parents
            parent1 = elite[np.random.randint(0, elite_size)]
            parent2 = elite[np.random.randint(0, elite_size)]
            
            # Crossover
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation (adaptive)
            mutation_rate = 0.2
            mut_strength = 0.1 * (1 - gen / max_generations * 0.5)
            if np.random.rand() < 0.7:  # crossover rate
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.normal(0, mut_strength)
            
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:pop_size]
    
    # Final SQP refinement on top solutions
    top_solutions = []
    top_scores = []
    
    for h in population[:5]:
        c5 = evaluate_erdos_solution(h, dx)
        if len(top_solutions) < 5:
            top_solutions.append(h.copy())
            top_scores.append(c5)
        elif c5 < max(top_scores):
            idx = np.argmax(top_scores)
            top_solutions[idx] = h.copy()
            top_scores[idx] = c5
    
    for h in top_solutions:
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
