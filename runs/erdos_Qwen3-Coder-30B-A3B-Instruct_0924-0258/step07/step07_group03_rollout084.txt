import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Phase 1: Gradient-based refinement with adaptive learning rate
    max_iter = 1000
    learning_rate = 0.1
    last_improvement = 0
    patience = 50
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap for all lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        c5_current = corr[max_lag_idx] * dx
        
        # Compute gradient (approximated using finite differences)
        epsilon = 1e-4
        grad = np.zeros(n_points)
        h_plus = h_current.copy()
        h_minus = h_current.copy()
        
        for i in range(n_points):
            h_plus[i] = min(1.0, h_current[i] + epsilon)
            h_minus[i] = max(0.0, h_current[i] - epsilon)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[i] = (corr_plus[max_lag_idx] - corr_minus[max_lag_idx]) / (2 * epsilon) * dx
            
            # Reset for next iteration
            h_plus[i] = h_current[i]
            h_minus[i] = h_current[i]
        
        # Update with gradient
        h_new = h_current - learning_rate * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check if this is an improvement
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = iteration
            learning_rate = 0.1  # Reset learning rate on improvement
        
        # Update current solution
        h_current = h_new
        
        # Adaptive learning rate decay
        if iteration - last_improvement > patience:
            learning_rate *= 0.8
            last_improvement = iteration
            
        # Random perturbations
        if iteration % 50 == 0:
            pert_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            for i in pert_indices:
                h_current[i] += np.random.uniform(-0.1, 0.1)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Phase 2: SQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use best solution so far as starting point
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final validation to ensure constraints are met
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
