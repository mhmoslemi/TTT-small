import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a better initial construction
        x = np.linspace(0, 2, n_points)
        # Construct more carefully with fewer steps for better conditioning
        h_start = 0.5 * (1 + np.sin(np.pi * x))
        h_start = np.clip(h_start, 0, 1)
        # Normalize to correct integral
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use L-BFGS with bounds for refinement
    max_iter = 500
    method = 'L-BFGS-B'
    
    # Optimization loop with time control
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb solution slightly for diversity
        if i > 0:
            noise_scale = 0.05
            noise = np.random.normal(0, noise_scale, n_points)
            h_pert = best_h + noise
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        else:
            h_pert = best_h.copy()
            
        # Define bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective for scipy minimize 
        def obj(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
            
        # Constraints
        def integral_constraint(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        constraints = {'type': 'eq', 'fun': integral_constraint}
        
        # Minimize using L-BFGS-B
        res = minimize(obj, h_pert, method=method, bounds=bounds, constraints=constraints, 
                       options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-6})
        
        if res.success:
            h_new = res.x
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
    
    # Finally, try with CVXPY to see if we can improve further
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Objective: maximize the overlap at the maximum lag
        corr = cp.Variable(2 * n_points - 1)
        # Add constraint that each correlation term is below the bound
        for i in range(2 * n_points - 1):
            k = i - (n_points - 1)
            if k >= 0:
                # correlation from h[0:n-k] * (1 - h[k:n])
                corr[i] = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                # correlation from h[-k:n] * (1 - h[0:n+k])
                corr[i] = cp.sum(cp.multiply(h_opt[-k:], 1 - h_opt[:n_points+k]))
        
        # Maximize the maximum correlation
        max_corr = cp.max(corr)
        prob = cp.Problem(cp.Minimize(max_corr * dx), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final normalization checks
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
