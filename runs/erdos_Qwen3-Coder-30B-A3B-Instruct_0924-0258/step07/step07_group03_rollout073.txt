import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T0 = 0.1
    cooling_rate = 0.001
    max_iter = 10000
    perturb_fraction = 0.15
    
    h_current = h_start.copy()
    c5_current = best_c5
    
    stagnation_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Perturb solution
        h_new = h_current.copy()
        n_perturb = int(n_points * perturb_fraction)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to satisfy constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Annealing acceptance
        T = T0 * np.exp(-cooling_rate * iteration)
        if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current)/T):
            h_current = h_new.copy()
            c5_current = c5_new
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        if stagnation_count > 1000:
            break
    
    # Gradient Descent refinement
    max_iter_gd = 500
    learning_rate = 0.1
    
    h_refined = best_h.copy()
    c5_refined = best_c5
    
    # Loop until no improvement for a while
    no_improvement_count = 0
    for iteration in range(max_iter_gd):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find the k that maximizes the overlap
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient for the k_max case
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] = (1 - h_refined[i + k_max]) * dx
            if i - k_max >= 0:
                grad[i] += (1 - h_refined[i - k_max]) * dx
        
        # Update h
        h_new = h_refined - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < c5_refined:
            h_refined = h_new.copy()
            c5_refined = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 50:
            learning_rate /= 2.0
            no_improvement_count = 0
            
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            best_h = res.x
            best_c5 = objective(best_h)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
