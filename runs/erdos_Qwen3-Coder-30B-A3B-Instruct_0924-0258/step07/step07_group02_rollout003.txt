import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm parameters
    pop_size = 60
    generations = 250
    mutation_rate = 0.05
    crossover_rate = 0.7
    elite_size = 5
    
    # Initialize population
    population = []
    fitness_scores = []
    
    # Add elite solutions
    population.append(h_start.copy())
    fitness_scores.append(best_c5)
    
    # Generate initial random population
    for _ in range(pop_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        c5 = evaluate_erdos_solution(h, dx)
        fitness_scores.append(c5)
    
    # Main EA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if fitness_scores[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness_scores[0]
        
        # Selection: tournament selection
        new_population = [population[0]]  # Elitism
        
        # Create offspring
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_size = 3
            parent1_idx = min(random.sample(range(pop_size), tournament_size), 
                              key=lambda i: fitness_scores[i])
            parent2_idx = min(random.sample(range(pop_size), tournament_size), 
                              key=lambda i: fitness_scores[i])
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if random.random() < crossover_rate:
                # Uniform crossover
                mask = np.random.random(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            if random.random() < mutation_rate:
                # Gaussian mutation
                noise = np.random.normal(0, mutation_rate * 0.5, n_points)
                child += noise
                # Repair
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        # Replace population
        population = new_population[:pop_size]
        fitness_scores = []
        
        # Evaluate new fitness
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness_scores.append(c5)
        
        # Decrease mutation rate
        mutation_rate = max(0.005, mutation_rate * 0.98)
    
    # Final refinement using local optimization
    top_candidates = population[:10]
    for candidate in top_candidates:
        # L-BFGS-B refinement
        try:
            def objective(h):
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points/2
            
            from scipy.optimize import Bounds
            bounds = Bounds(0, 1)
            result = minimize(objective, candidate, method='L-BFGS-B', 
                            bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, 
                            options={'maxiter': 200})
            
            if result.success:
                h_refined = np.clip(result.x, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
