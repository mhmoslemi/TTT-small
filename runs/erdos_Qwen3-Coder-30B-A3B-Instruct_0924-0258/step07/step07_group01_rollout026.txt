import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize using the provided starting point
    h_current = h_start.copy()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)

    # CVXPY optimization approach
    for iteration in range(20):  # Limit iterations to stay within time budget
        if time() - start_time > budget_s * 0.95:
            break
            
        # Ensure we're starting from a valid point
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        try:
            # Create CVXPY variables
            h_var = cp.Variable(n_points)
            
            # Define the maximum correlation constraint
            # For each valid lag in the cross-correlation
            # We want to minimize max_lag correlation(h, 1-h, lag) * dx
            # This is a bit tricky because we need to model the maximum of correlations
            
            # Let's approximate the maximum by constraining a scalar variable
            # to be larger than any of the correlations
            max_overlap_upper = cp.Variable()
            
            # We need to compute the correlations manually to build constraints
            # So we'll compute a sequence of constraint inequalities
            
            # Get the full correlation (this is from numpy for verification)
            corr_full = np.correlate(h_current, 1 - h_current, mode='full')
            
            # Create constraints for each valid lag
            constraints = [max_overlap_upper >= corr_full[i] * dx 
                          for i in range(len(corr_full))]
            
            # Add integral constraint
            constraints.append(cp.sum(h_var) == n_points / 2)
            
            # Add bound constraints
            constraints.extend([h_var[i] >= 0 for i in range(n_points)])
            constraints.extend([h_var[i] <= 1 for i in range(n_points)])
            
            # Objective function: minimize max_overlap_upper
            objective = cp.Minimize(max_overlap_upper)
            prob = cp.Problem(objective, constraints)
            
            # Solve the problem
            prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                h_new = np.array(h_new)
                h_new = np.clip(h_new, 0, 1)
                # Ensure sum constraint is exactly met
                h_new = h_new * (n_points/2) / np.sum(h_new)
                
                # Compute overlap
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h_current = h_new.copy()
                else:
                    # Local refinement if improvement is not seen
                    h_current = h_new.copy()
            else:
                # If the optimization failed, try with SLSQP
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                # Use current h for the local search
                res = minimize(objective, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-10})
                
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    h_refined = np.clip(h_refined, 0, 1)
                    h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        h_current = h_refined.copy()
                else:
                    # If SLSQP fails too, continue to next iteration
                    h_current = h_current * (n_points/2) / np.sum(h_current)
        except Exception as e:
            # If anything fails, fall back to a basic refinement
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-8})
                
                if res.success:
                    h_refined = res.x
                    h_refined = np.clip(h_refined, 0, 1)
                    h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        h_current = h_refined.copy()
                else:
                    h_current = h_current * (n_points/2) / np.sum(h_current)
            except:
                h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Final refinement using SLSQP
    def objective_final(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum_final(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum_final = {'type': 'eq', 'fun': constraint_sum_final}
    bounds_final = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective_final, best_h, method='SLSQP', 
                       bounds=bounds_final, constraints=cons_sum_final, 
                       options={'maxiter': 300, 'ftol': 1e-12})
        
        if res.success:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
