import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    max_iter = 1000
    T0 = 0.1
    c = 0.001
    perturbation_fraction = 0.1
    
    h_current = h_start.copy()
    c5_current = best_c5
    
    stagnation_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb solution
        indices = np.random.choice(n_points, int(n_points * perturbation_fraction), replace=False)
        h_new = h_current.copy()
        
        # Add small random perturbations
        for idx in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_new[idx] += delta
            
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to maintain sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Temperature
        T = T0 * np.exp(-c * iteration / max_iter)
        
        # Accept or reject
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
            stagnation_count = 0
        else:
            # Probabilistically accept worse solution
            if np.random.rand() < np.exp(-(c5_new - c5_current) / T):
                h_current = h_new.copy()
                c5_current = c5_new
                stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Update best solution
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        if stagnation_count > 100:
            break
    
    # Local refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
