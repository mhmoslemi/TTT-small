import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Initialize with a periodic function
        x = np.linspace(0, 2, 35)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (35 / 2) / np.sum(h_start)
        h_start *= normalization
        n_points = 35
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    tabu_list = []
    tabu_size = 20
    T = 0.5  # Initial temperature
    alpha = 0.995  # Cooling rate
    T_min = 1e-5
    max_iter = 1000
    stagnation_limit = 50
    stagnation_count = 0
    perturbation_size = 0.1
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate solution
        candidate_h = current_h.copy()
        indices = np.random.choice(n_points, size=int(0.15 * n_points), replace=False)
        for i in indices:
            # Perturb by small amount
            perturb = np.random.uniform(-perturbation_size, perturbation_size)
            candidate_h[i] += perturb
        
        # Ensure bounds
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Normalize to satisfy integral constraint
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        
        # Evaluate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Accept or reject
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                stagnation_count = 0
            
            # Update tabu list
            tabu_list.append(current_h.copy())
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        else:
            stagnation_count += 1
            
        # Cooling schedule
        T = max(T * alpha, T_min)
        
        # Stagnation check and restart
        if stagnation_count > stagnation_limit:
            # Perturb best solution slightly
            restart_h = best_h + np.random.normal(0, 0.02, n_points)
            restart_h = np.clip(restart_h, 0, 1)
            restart_h = restart_h * (n_points / 2) / np.sum(restart_h)
            current_h = restart_h.copy()
            current_c5 = evaluate_erdos_solution(current_h, dx)
            stagnation_count = 0
    
    # Final refinement with trust-constr
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h.copy(), method='trust-constr', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'gtol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
