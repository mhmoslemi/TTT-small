import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create periodic initialization
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / (2 * np.sum(h_start)))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    T = 0.5
    T_min = 1e-5
    alpha = 0.995
    tabu_list = []
    tabu_size = 20
    stagnation_count = 0
    max_stagnation = 50
    
    # Perturbation sizes
    perturb_start = 0.1
    perturb_end = 0.05
    perturb_factor = (perturb_end / perturb_start) ** (1.0 / 1000)
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    iteration = 0
    max_iter = 1000
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iter:
        perturb_size = perturb_start * (perturb_factor ** iteration)
        
        # Generate candidate solution
        candidate_h = current_h.copy()
        indices = np.random.choice(n_points, size=int(0.15 * n_points), replace=False)
        
        for i in indices:
            delta = np.random.uniform(-perturb_size, perturb_size)
            candidate_h[i] += delta
            
        # Clip and normalize
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_h = candidate_h * (n_points / (2 * np.sum(candidate_h)))
        
        # Evaluate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5:
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            if candidate_c5 < best_c5:
                best_h = candidate_h.copy()
                best_c5 = candidate_c5
                stagnation_count = 0
        else:
            # Probabilistic acceptance
            delta = candidate_c5 - current_c5
            acceptance_prob = math.exp(-delta / T)
            if np.random.random() < acceptance_prob:
                current_h = candidate_h.copy()
                current_c5 = candidate_c5
                stagnation_count = 0
        
        # Update tabu list
        if len(tabu_list) >= tabu_size:
            tabu_list.pop(0)
        tabu_list.append(current_h.copy())
        
        # Check stagnation
        stagnation_count += 1
        if stagnation_count >= max_stagnation:
            # Restart with perturbed best solution
            restart_h = best_h.copy()
            noise = np.random.normal(0, 0.05, n_points)
            restart_h += noise
            restart_h = np.clip(restart_h, 0, 1)
            restart_h = restart_h * (n_points / (2 * np.sum(restart_h)))
            current_h = restart_h
            current_c5 = evaluate_erdos_solution(current_h, dx)
            stagnation_count = 0
                
        T = max(T * alpha, T_min)
        iteration += 1
    
    # Final refinement with SLSQP 
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
