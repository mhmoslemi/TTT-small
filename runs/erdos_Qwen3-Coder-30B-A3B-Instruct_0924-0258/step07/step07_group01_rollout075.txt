import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try CVXPY-based optimization
    max_iter = 20
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create CVXPY variables
        h_var = cp.Variable(n_points, name='h')
        # Objective: minimize max overlap (approximated)
        # We'll approximate the max overlap constraint
        # For now, we'll just correct the sum
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points/2]
        
        # Try to minimize using CVXPY (we use a simpler approach)
        try:
            # This is a simplified version - just solve with sum constraint
            prob = cp.Problem(cp.Minimize(cp.sum(h_var)), constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_cvx = h_var.value
                h_cvx = np.clip(h_cvx, 0, 1)
                # Re-scale to meet integral constraint exactly
                h_cvx = h_cvx * (n_points/2) / np.sum(h_cvx)
                c5_cvx = evaluate_erdos_solution(h_cvx, dx)
                
                if c5_cvx < best_c5:
                    best_c5 = c5_cvx
                    best_h = h_cvx.copy()
        except:
            pass
            
        # Local refinement using SLSQP
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        h_refined = best_h.copy()
        try:
            res = minimize(objective, h_refined, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
        # Random perturbation
        h_new = best_h + np.random.normal(0, 0.02, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    return (list(best_h), best_c5, n_points)
