import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Convex relaxation using SDP
    # Try to solve a relaxed version using CVXPY
    
    # Try different SDP solvers
    solvers = ['ECOS', 'SCS']
    for solver in solvers:
        try:
            # Create variables
            h_relax = cp.Variable(n_points)
            Z = cp.Variable((n_points, n_points), symmetric=True)
            t = cp.Variable()
            
            # Constraints for SDP relaxation
            constraints = [
                cp.constraints.SOC(t, cp.norm(h_relax, 2)),
                Z >> cp.outer(h_relax, h_relax),
                cp.trace(Z) == n_points / 2.0,
                h_relax >= 0,
                h_relax <= 1
            ]
            
            # Objective: minimize t
            prob = cp.Problem(cp.Minimize(t), constraints)
            
            # Try to solve with solver
            prob.solve(solver=solver, verbose=False, time_limit=300)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_relax.value
                if h_sol is not None:
                    # Normalize to correct sum
                    h_sol = h_sol * (n_points / 2.0) / np.sum(h_sol)
                    h_sol = np.clip(h_sol, 0, 1)
                    c5_val = evaluate_erdos_solution(h_sol, dx)
                    if c5_val < best_c5:
                        best_h = h_sol.copy()
                        best_c5 = c5_val
                break  # Break on success
        except Exception:
            continue  # Try next solver
    
    # Phase 2: Local refinement with SLSQP
    elapsed = time() - start_time
    if elapsed < budget_s * 0.9:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
