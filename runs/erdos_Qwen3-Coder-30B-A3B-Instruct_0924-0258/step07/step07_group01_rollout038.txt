import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a simple piecewise constant function
        h_start = np.zeros(n_points)
        # Place two intervals of length 0.5 each, symmetrically
        dx = 2.0 / n_points
        # Set first interval from index 10 to 20 (approx 0.65 to 1.15) 
        # and second from index 20 to 30 (approx 1.15 to 1.65)
        # This ensures we get 0.5 + 0.5 = 1 units of integral
        h_start[10:20] = 1.0  # covers x ~ [0.65, 1.15]
        h_start[20:30] = 1.0  # covers x ~ [1.15, 1.65]
        # Scale to unit integral  
        h_start = h_start * (n_points / 2.0) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simple gradient-free local search
    max_iters = 5000
    for it in range(max_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try small perturbations of each point
        h_trial = best_h.copy()
        idx = np.random.randint(0, n_points)
        pert = np.random.uniform(-0.05, 0.05)
        h_trial[idx] += pert
        h_trial[idx] = np.clip(h_trial[idx], 0, 1)
        
        # Renormalize
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        
        c5_trial = evaluate_erdos_solution(h_trial, dx)
        if c5_trial < best_c5:
            best_h = h_trial
            best_c5 = c5_trial
    
    # Final evaluation to be sure
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
