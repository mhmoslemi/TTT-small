import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization
    h = h_start.copy()
    learning_rate = 0.01
    no_improvement_count = 0
    max_no_improvement = 200
    iteration = 0
    
    while time() - start_time < budget_s * 0.95:
        # Compute the full correlation to find the maximum overlap lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # The actual lag is the offset from the center
        # np.correlate gives lags from -(n-1) to (n-1) in full mode
        lag = max_k - (n_points - 1)
        
        # Compute gradient for that specific lag
        # h[i] is modified based on influence from h[(i+lag)%n_points] and h[i]
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + lag >= 0 and i + lag < n_points:
                gradient[i] = 2 * (1 - h[i + lag] - h[i])
            else:
                # Handle boundary conditions by using modulo
                idx = (i + lag) % n_points
                gradient[i] = 2 * (1 - h[idx] - h[i])
        
        # Update with gradient descent
        h_new = h - learning_rate * gradient
        
        # Clip to [0,1] and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject update
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
            # Increase learning rate slightly if we improve
            learning_rate = min(learning_rate * 1.05, 0.1)
        else:
            h = h_new.copy()
            no_improvement_count += 1
            # Reduce learning rate if no improvement
            learning_rate *= 0.95
            
        # Reset learning rate occasionally to escape local minima
        if no_improvement_count >= 100:
            learning_rate = 0.01
            no_improvement_count = 0
            
        iteration += 1
        
        # Break early if no improvement for a while
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final refinement using scipy minimize
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
