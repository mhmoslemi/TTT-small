import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    random.seed(seed)
    start_time = time()
    
    # Load or create initial solution

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 100
    num_generations = 200
    elite_size = population_size // 5
    mutation_rate = 0.3
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.1, n_points)
        else:
            h = np.random.random(n_points) * 0.5 + 0.25
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    stagnation_count = 0
    best_overall_c5 = best_c5
    
    # GA loop
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if fitness[0] < best_overall_c5:
            best_overall_c5 = fitness[0]
            best_h = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > 50:
            break
            
        # Create next generation
        new_population = population[:elite_size].copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if np.random.random() < crossover_rate and len(new_population) + 1 < population_size:
                # Tournament selection
                parent1_idx, parent2_idx = np.random.choice(elite_size, 2, replace=False)
                parent1 = population[parent1_idx]
                parent2 = population[parent2_idx]
                
                # Arithmetic crossover
                alpha = np.random.random()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
                
                # Mutation
                for i in range(n_points):
                    if np.random.random() < mutation_rate:
                        child1[i] += np.random.normal(0, 0.05)
                    if np.random.random() < mutation_rate:
                        child2[i] += np.random.normal(0, 0.05)
                        
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points/2) / np.sum(child1)
                
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points/2) / np.sum(child2)
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Random mutation
                idx = np.random.randint(elite_size)
                child = population[idx].copy()
                for i in range(n_points):
                    if np.random.random() < mutation_rate:
                        child[i] += np.random.normal(0, 0.05)
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                new_population.append(child)
        
        population = new_population
    
    # Final SLSQP refinement on top solutions
    top_solutions = population[:5] if len(population) >= 5 else population
    
    for h_refined in top_solutions:

        h_refined = h_refined * (n_points/2) / np.sum(h_refined)

            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            continue
    
    return (list(best_h), best_c5, n_points)
