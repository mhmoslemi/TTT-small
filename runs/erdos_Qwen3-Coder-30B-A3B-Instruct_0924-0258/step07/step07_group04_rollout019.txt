import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a structured starting point: sawtooth shape
        h_start = np.linspace(0.1, 0.9, n_points)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    
    # Normalize to constraint
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds for differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Objective function with penalty term for constraint violation
    def objective(h):
        # Violation of integral constraint
        constraint_violation = abs(np.sum(h) - n_points / 2.0)
        penalty = 1000 * constraint_violation**2
        return evaluate_erdos_solution(h, dx) + penalty
    
    # Differential Evolution with adaptive parameters
    max_iter = 500
    popsize = 50
    mutation_rate = 0.5
    crossover_rate = 0.9
    
    # DE optimization
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Adaptive DE parameters
        if iteration > 100:
            mutation_rate = max(0.1, 0.5 * (1 - iteration / max_iter))
            crossover_rate = 0.9
            
        # Run DE
        result = differential_evolution(
            objective,
            bounds,
            maxiter=10,
            popsize=popsize,
            mutation=mutation_rate,
            recombination=crossover_rate,
            seed=seed + iteration,
            disp=False
        )
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    
    # Final SLSQP refinement
    try:
        def objective_slsqp(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds_slsqp = [(0, 1) for _ in range(n_points)]
        res = minimize(objective_slsqp, best_h, method='SLSQP', 
                       bounds=bounds_slsqp, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective_slsqp(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
