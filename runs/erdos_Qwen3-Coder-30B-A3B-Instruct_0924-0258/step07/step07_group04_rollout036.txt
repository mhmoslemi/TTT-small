import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Initialize with a better starting point
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))  # 5 cycles over [0,2]
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    T = 0.5
    alpha = 0.995
    T_min = 1e-5
    tabu_list = []
    tabu_size = 20
    stagnation_count = 0
    max_stagnation = 50
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Adaptive perturbation size
    perturb_start = 0.1
    perturb_end = 0.05
    iter_count = 0
    max_iter = 1000
    
    def perturb_solution(h, perturb_size):
        h_new = h.copy()
        # Randomly select indices to perturb
        indices = np.random.choice(n_points, size=max(1, int(0.15 * n_points)), replace=False)
        for i in indices:
            delta = np.random.uniform(-perturb_size, perturb_size)
            h_new[i] += delta
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        # Normalize to satisfy integral constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        return h_new
    
    def is_tabu(h, tabu_list):
        # Check if this configuration is in tabu list
        for past_h in tabu_list:
            if np.allclose(h, past_h, rtol=1e-5):
                return True
        return False
    
    iter_count = 0
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Determine perturbation size (decreases over time)
        progress = iter_count / max_iter
        perturb_size = perturb_start + (perturb_end - perturb_start) * (1 - progress)
        
        # Generate candidate solution
        candidate_h = perturb_solution(current_h, perturb_size)
        
        # Check if candidate is tabu
        if is_tabu(candidate_h, tabu_list):
            # If tabu, still allow with low probability
            accept_prob = 0.05  # This encourages escaping tabu
        else:
            accept_prob = 1.0
        
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5 or (accept_prob > np.random.random() and np.random.random() < np.exp(-(candidate_c5 - current_c5) / T)):
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            
            # Update tabu list
            tabu_list.append(current_h.copy())
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
        
        # Cooling schedule
        T = max(T * alpha, T_min)
        
        # Restart mechanism if stagnation
        if stagnation_count > max_stagnation:
            # Perturb best solution to escape local minima
            restart_perturb = 0.1
            current_h = perturb_solution(best_h, restart_perturb)
            current_c5 = evaluate_erdos_solution(current_h, dx)
            stagnation_count = 0
            
        iter_count += 1
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final validation
    final_sum = np.sum(best_h)
    if abs(final_sum - n_points / 2.0) > 1e-6:
        # Re-normalize
        best_h = best_h * (n_points/2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
