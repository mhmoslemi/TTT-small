import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based refinement with dynamic learning rate
    max_iterations = 1000
    step_size = 0.01
    
    h_current = h_start.copy()
    last_improvement = 0
    improvement_threshold = 50
    
    iteration = 0
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        # Compute all correlations to find the maximum
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = np.max(corr) * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h_current.copy()
            last_improvement = iteration
            
        # Calculate gradient for the maximum overlap lag
        # The gradient is the partial derivative of the overlap w.r.t. h[i]
        # For lag k, gradient[i] = 2 * (1 - h[i+k] - h[i]) / n_points
        grad = np.zeros(n_points)
        for i in range(n_points):
            k = max_lag_idx - n_points + 1  # Convert correlation index to lag
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 2 * (1 - h_current[j] - h_current[i]) / n_points
                
        # Update step
        h_new = h_current - step_size * grad
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if update is beneficial
        h_current = h_new
        
        # Adjust step size if needed
        if iteration > 0 and iteration % 100 == 0:
            new_c5 = evaluate_erdos_solution(h_current, dx)
            if new_c5 >= best_c5:
                step_size *= 0.5
                if step_size < 1e-6:
                    break
                    
        iteration += 1
        
        # Early stopping if no improvement for a while
        if iteration - last_improvement > improvement_threshold:
            break
    
    # Final optimization using scipy
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
