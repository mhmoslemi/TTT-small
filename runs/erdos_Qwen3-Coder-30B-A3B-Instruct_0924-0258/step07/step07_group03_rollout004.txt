import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based descent
    max_iter = 1000
    learning_rate = 0.1
    stagnation_threshold = 50
    stagnation_count = 0
    
    h_current = h_start.copy()
    target_sum = n_points / 2.0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute overlaps for all lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag = np.argmax(corr)
        max_value = corr[max_lag]
        
        # Compute gradient with respect to h for the max overlapping lag
        gradient = np.zeros(n_points)
        # For lag k, we need indices where h[i] and (1-h[i+k]) overlap
        # When i+k < n_points, contributions are from h[i] and (1-h[i+k])
        # Correlation computes sum_over_i h[i] * (1 - h[i+k]) 
        # Gradient wrt h[i] is the derivative of sum_k h[i] * (1 - h[i+k])
        # That's -sum_k (1 - h[i+k]) for all valid k
        # For fixed i, sum for k where i+k is valid
        # i ranges from 0 to n_points -1, k from 0 to n_points-1,
        # but we shift by max_lag = (n_points-1) - k, so k = (n_points-1) - max_lag
        phase_shift = max_lag - (n_points - 1) 
        # That's negative, the correct lag
        
        # Compute gradient more efficiently
        for i in range(n_points):
            for k in range(n_points):
                j = i + k
                if j < n_points:  # Valid overlap, h[i] and (1-h[j]) are multiplied
                    # Actual weight sharing with moving target lag
                    gradient[i] -= (1 - h_current[j])
        
        # Apply step and ensure feasibility
        h_new = h_current - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        
        # Re-scale: sum must be target_sum
        sum_h_new = np.sum(h_new)
        if sum_h_new > 0:
            h_new *= target_sum / sum_h_new
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Check if improvement
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        h_current = h_new.copy()
        
        if stagnation_count >= stagnation_threshold:
            learning_rate *= 0.5
            stagnation_count = 0

    # Phase 2: Refinement with L-BFGS-B
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='L-BFGS-B', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
