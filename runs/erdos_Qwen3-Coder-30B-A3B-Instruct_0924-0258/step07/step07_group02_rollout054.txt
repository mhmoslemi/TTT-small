import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm parameters
    pop_size = 50
    max_generations = 300
    mutation_rate = 0.05
    crossover_rate = 0.7
    elite_size = 5
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = np.random.uniform(0.1, 0.9, n_points)
        h = h * (n_points/2) / np.sum(h)  # Normalize to sum = n_points/2
        h = np.clip(h, 0, 1)
        population.append(h)
    
    def repair(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def crossover(p1, p2):
        if random.random() > crossover_rate:
            return p1.copy()
        alpha = random.random()
        child = alpha * p1 + (1 - alpha) * p2
        return child
    
    def mutate(h, rate):
        if random.random() > rate:
            return h.copy()
        noise = np.random.normal(0, rate * 0.2, len(h))
        h = h + noise
        return repair(h)
    
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    # EA main loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = [fitness(h) for h in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best
        if fitnesses[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitnesses[0]
        
        # Selection: top 50% + random
        selected_indices = sorted_indices[:pop_size//2]
        for _ in range(pop_size - len(selected_indices)):
            selected_indices.append(random.randint(0, pop_size-1))
        
        # Create new population
        new_population = []
        # Keep elite
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        while len(new_population) < pop_size:
            a, b = random.sample(selected_indices, 2)
            child = crossover(population[a], population[b])
            child = mutate(child, mutation_rate)
            new_population.append(child)
        
        population = new_population
        
        # Update mutation rate (linear decay)
        mutation_rate = max(0.005, 0.05 - generation * (0.045 / max_generations))
    
    # Final refinement with scipy
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    # Use best solution as starting point
    h0 = best_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
    ]
    
    try:
        result = minimize(
            objective,
            h0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
