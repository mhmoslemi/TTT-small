import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    elite_size = 5
    max_generations = 1000
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Genetic algorithm loop
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness for all
        for i in range(pop_size):
            fitness[i] = evaluate_erdos_solution(population[i], dx)
        
        # Sort population by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new population
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(population[i].copy())

        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitness, tournament_size)
            parent2 = tournament_selection(population, fitness, tournament_size)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            mutate(child1, mutation_rate)
            mutate(child2, mutation_rate)
            
            # Ensure feasibility
            child1 = repair_constraints(child1, n_points, dx)
            child2 = repair_constraints(child2, n_points, dx)

            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population
    
    # Final convex relaxation

            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,

            prob = cp.Problem(cp.Minimize(0), constraints)

                best_c5 = c5_refined

    return (list(best_h), best_c5, n_points)

def tournament_selection(population, fitness, tournament_size):
    selected_indices = np.random.choice(len(population), tournament_size)
    best_idx = selected_indices[np.argmin([fitness[i] for i in selected_indices])]
    return population[best_idx].copy()

def crossover(parent1, parent2):
    # Uniform crossover
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(individual, mutation_rate):
    for i in range(len(individual)):
        if np.random.rand() < mutation_rate:
            individual[i] += np.random.normal(0, 0.03)
    return individual

def repair_constraints(h, n_points, dx):
    h = np.clip(h, 0, 1)
    h = h * (n_points/2) / np.sum(h)
    return h
