import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Basis decomposition approach
    # Use 10 basis functions (triangular) to represent h
    
    # Create triangular basis functions
    def create_basis(n, m):
        # Create m triangular basis functions over [0, 2]
        basis = np.zeros((n, m))
        x = np.linspace(0, 2, n)
        for i in range(m):
            # Center basis function at i * 2 / (m-1) for m > 1, else at 1
            center = i * 2.0 / (m - 1) if m > 1 else 1.0
            # Width of each basis function: 0.4
            width = 0.4
            
            # Triangular pulse
            for j in range(n):
                dist = abs(x[j] - center)
                if dist <= width:
                    basis[j, i] = 1 - dist / width
        return basis
    
    # Create basis
    basis = create_basis(n_points, 10)
    
    # Initial coefficients - use least-squares to project h_start onto basis
    try:
        # Ensure h_start is feasible
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
        
        # Project h_start onto the basis
        coeff = np.linalg.lstsq(basis, h_start, rcond=None)[0]
        coeff = np.maximum(coeff, 0)  # Keep coefficients non-negative
        
        # Update h with current coefficients
        h_current = np.clip(basis @ coeff, 0, 1)
        h_current = h_current * (n_points/2) / np.sum(h_current)
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            
    except:
        # Fallback to the original h_start
        coeff = np.ones(10) * 0.1
        current_c5 = best_c5

    # Trust-region optimization of coefficients
    max_iter = 2000
    trust_radius = 0.1
    radius_factor = 2.0
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap for current h
        h_current = np.clip(basis @ coeff, 0, 1)
        h_current = h_current * (n_points/2) / np.sum(h_current)
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        # Update best
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
        
        # Optimization step
        # Calculate gradient and Hessian by finite differences
        try:
            # Update coefficients using trust-region step
            # Simple gradient descent on a small subset of overlap terms
            delta = np.random.normal(0, trust_radius, 10)
            new_coeff = coeff + delta
            
            # Ensure positive coefficients
            new_coeff = np.maximum(new_coeff, 0)
            
            # Try a step
            h_new = np.clip(basis @ new_coeff, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            # Accept if improvement
            if new_c5 < current_c5:
                coeff = new_coeff
                trust_radius = min(trust_radius * radius_factor, 1.0)
            else:
                trust_radius = max(trust_radius / radius_factor, 1e-6)
                
            # Occasionally use viability checks with optimization
            if iter_num % 10 == 0:
                try:
                    # Simple downscale on large coefficients to avoid negatives at boundaries
                    new_coeff = np.maximum(coeff, 0)
                    # Refine with full optimization for subset of basis functions
                    # Since we have sparse basis, just take the dominate one
                    h_new = np.clip(basis @ new_coeff, 0, 1)
                    h_new = h_new * (n_points/2) / np.sum(h_new)
                    new_c5 = evaluate_erdos_solution(h_new, dx)
                    if new_c5 < current_c5:
                        coeff = new_coeff
                except:
                    pass
                
        except Exception:
            pass
    
    # Final refinement if time allows
    if time() - start_time < budget_s * 0.9:
        try:
            def objective(c):
                h = np.clip(basis @ c, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(c):
                h = np.clip(basis @ c, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                return np.sum(h) - n_points / 2.0
                
            bounds = [(None, None) for _ in range(10)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            res = minimize(objective, coeff, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = np.clip(basis @ res.x, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
        finally:
            pass

    return (list(best_h), best_c5, n_points)
