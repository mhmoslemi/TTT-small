import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with adaptive learning rate
    h = h_start.copy()
    learning_rate = 0.01
    max_iter = 1000
    patience = 50
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current overlap
        overlap = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= patience:
            break
            
        # Compute gradient for the most constraining lag
        # For lag k=0, the gradient of h[i] is 2*(1-h[i]-h[(i+0)%n_points])
        # The actual gradient with the most constraining lag
        # Note: correlation is computed as [h[0],h[1],...,h[n-1],0,0]
        # So we need to compute the difference in that indexing
        
        # We'll compute gradient w.r.t. all positions using correlation index k_max
        # The lag k_max corresponds to:
        # for i in range(n_points):
        #   grad[i] += 2*(h[(i + k_max) % n_points] + h[i] - 1)
        # But we need to consider how correlate handles indices
        # np.correlate(h, 1-h, mode='full') gives [k = -n+1,...,n-1]
        # so there are 2*n-1 elements
        # k=0 means h[i] pairs with (1-h[i]), k=-k0 means h[i] pairs with (1-h[i+k0]) etc.
        # Indices in the correlation are:
        # for i in range(2*n_points - 1):
        #   correlate[i] = sum(h[j] * (1 - h[j + i-n+1]) for j in range(n_points) if 0 <= j+i-n+1 < n_points)
        # But easier approach: use 1-h as a kernel, so it's a convolution
        # The key is that the contribution of h[j] to result at lag k is h[j] * (1-h[j+k])
        # So gradient for h[i] is sum over all valid k where j=i: 2*(1-h[(i+k)%n_points])
        
        # Drawback: we iterate over all possible lags, since prior iter we compute k_max,
        # but the computation is: for position i and shift k_max resulting in max overlap:
        # gradient[i] = 2 * (1 - h[i] - h[(i + k_max) % n_points])  
        # (this is actual change in sum when h[i] is changed)
        
        # Better approach: go directly from the full correlation. It's more theoretically base.
        # Manually compute, since update based on the maximal slack:
        # Hard cast to make k_index in range
        k_max = max_k - (n_points - 1)  # 0 -> lag is 0, n_points-1 -> lag is 0, 2*n_points-2 -> lag = 0
        # But I should match how correlation interprets the lags
        # In np.correlate(h, 1-h, mode='full'):  
        # index 0 corresponds to shift k = -n+1 
        # index n-1 corresponds to shift k = 0
        # index 2*n-2 corresponds to shift k = n-1
        # So k_index = index - (n-1)  
        # So for k = -n+1 => index = 0
        # for k = 0 => index = n-1  
        # for k = n-1 => index = 2*n-2
        
        k_max = k_max  # This is the true shift index for the correlation
        # Gradient calculation is differently intuitive:
        # for each i, gradient_i is derivative with respect to h_i of sum over all k of 
        # h_i * (1 - h_{i+k}) for overlapping pairs
        # In our one pair contribution:
        # For i and k_max shift, we get 1 * (1-h_{i+k_max}) + h_{i+k_max} * 1 = 1 - h_{i+k_max} + h_{i+k_max}
        # Wait, each pair (h[i], h[i+k]) has the contribution to correlation at lag k: h[i] * (1 - h[i+k])
        # So the change of total when we change h[i]:
        # Sum over all k such that i+k is valid (which means k can be any integer so long as (i+k)%n is valid)
        # But we're only interested in when h[i+k] is part of a term with h[i]
        # So it is a linear function from h[i] to correlation value, where we have:
        # contribution = h[i] * (1 - h[i+k_max]) for this k and pair
        # and derivative w.r.t. h[i] is (1 - h[i+k_max])
        # For k is constant for the whole optimization iteration, since the limiting shift is fixed in this iteration
        
        # But the full computation is actually more complex. 
        # What I'll do is compute numerically using finite differences approx to validate
        # But I want sensible formula.
        # When lag k_max is fixed:
        # Total correlation = sum over all i of h[i] * (1 - h[(i + k_max) % n_points]) 
        # Differentiate w.r.t. h[j]: 
        # If j = i then derivative = 1 - h[(i + k_max) % n_points]
        # If j = (i+k_max) % n_points then derivative = -h[i]  
        # Otherwise it's 0
        # Which means gradient explanation simply:
        #  ∂/∂h[j] correlation = (1 - h[(j + k_max) % n_points]) - h[(j - k_max) % n_points]
        # ref: The derivative w.r.t. h_i of h_i * (1 - h_{i+k}) is
        # (1 - h_{i+k}) + h_i * 0 = 1 - h_{i+k}
        # And if the formula is a double sum this will contribute for terms where:
        # 1. h_i appears in the sum as h_i= h_{i+k}, so 1 channel 
        # 2. h_{i+k} appears in the sum as h_i, so -h_{i+k} channel
        # So truly: the gradient[i] = 1 - h[(i + k_max) % n_points] - h[(i - k_max) % n_points]
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # For h[i], the contribute term is h[i] * (1 - h[(i+k_max) % n_points])
            # So gradient is (1 - h[(i+k_max) % n_points]) 
            # Also consider the contribution to other positions.
            # In full sum, for fixed i and j, h[i] contributes to term h[i] * (1 - h[j])
            # So we have the j term contributes, etc.
            # So the gradient with respect to h[i] of the whole sum:
            # This means that in the sum of prod h[i] * (1 - h[i+k]) over all valid i,k:
            # d/dh[i] = 1 - h[(i + k_max) % n_points] for k_max corresponding to max
            # Wait the derivative is taken w.r.t. h[i] of total overlap.
            # For 1-lag correction it'd be:
            # correlation = sum_i h[i] * (1 - h[i+1]) 
            # then d/dh[i] = 1 - h[i+1] - h[i-1] (see previous formula analysis)
            # But we're testing the matched pair constraint. 
            # The overlaps are full correlation. 
            # min_h[i] = max(0, h[i] - lr * gradient[i])
            # So total overlap at lag k is: \sum_i h[i] * (1 - h[i+k])
            # For h_i, gradient w.r.t. h_i = 1 - h[i+k_max] (if k_max = k)
            # But also, for term h[i] * (1 - h[i+k]) we also include the term h[i+k] * (1 - h[i+k+k])
            # So the evolution of h_i derivative seems much more involved.  
            # That's okay, let's just move forward with simpler valid implementation
            
        if k_max == 0:
            # For k_max = 0, i.e., self-correlation. 
            # i - 0 = i, so same, no shifting when k=0
            # it makes little difference, this would be a recursive case
            # Let's do the in-place straightforward case for simplicity
            gradient = np.zeros(n_points)
            for i in range(n_points):
                gradient[i] = 1 - h[(i + k_max) % n_points] - h[(i - k_max) % n_points]
        else:
            # Manually.compute gradient. It's 1 - h[i+k_max] - h[i-k_max]
            gradient = np.zeros(n_points)
            for i in range(n_points):
                gradient[i] = 1 - h[(i + k_max) % n_points] - h[(i - k_max) % n_points]
        
        # Update h
        h = h - learning_rate * gradient
        
        # Clip h to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Adapt learning rate
        if np.linalg.norm(gradient) < 1e-12:
            learning_rate *= 0.9  # Subtle decay 
        elif np.any(np.isnan(h)):
            learning_rate *= 0.5
        elif max_overlap < best_c5:
            learning_rate *= 1.1  # accelerates if improving often
        else:
            learning_rate *= 0.95
            
        if learning_rate < 1e-6:
            learning_rate = 1e-6
            
    # Final local optimization with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
