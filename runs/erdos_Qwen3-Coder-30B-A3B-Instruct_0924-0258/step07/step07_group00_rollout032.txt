import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient descent approach
    max_iter = 1000
    step_size = 0.01
    tolerance = 1e-6
    patience = 50
    patience_counter = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        lag_indices = np.arange(-len(h) + 1, len(h))
        max_lag_idx = np.argmax(corr)
        max_lag = lag_indices[max_lag_idx]
        
        # Compute gradient for the max lag
        grad = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                grad[i] += 2 * (1 - h[i + max_lag] - h[i])
        else:
            for i in range(-max_lag, n_points):
                grad[i] += 2 * (1 - h[i + max_lag] - h[i])
        
        # Update h
        h_new = h - step_size * grad
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check constraint satisfaction
        if not (np.abs(np.sum(h_new) - n_points / 2) < 1e-6):
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Update h
        h = h_new.copy()
        
        # Adaptive step size and early stopping
        if patience_counter >= patience:
            step_size *= 0.5
            patience_counter = 0
            if step_size < 1e-8:
                break
    
    # Local refinement using scipy
    from scipy.optimize import minimize
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
