from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    # Center around 1.0 to allow proper overlap checking in [0,2]
    # a = 0.75, b = 1.25 to place 0.5-wide intervals
    a = 0.75
    b = 1.25
    
    h_constructed = np.zeros(n_points)
    # Map continuous positions to discrete samples
    for i in range(n_points):
        x = i * dx
        if a <= x < a + 0.5 or b <= x < b + 0.5:
            h_constructed[i] = 1.0
    
    # Normalize area to 1
    current_area = np.sum(h_constructed) * dx
    h_constructed = h_constructed * (1.0 / current_area)
    
    # Compute overlap
    c5_bound = evaluate_erdos_solution(h_constructed, dx)

    # This is a more refined version based on the structure
    best_h = h_constructed.copy()

    def objective(h):
        return evaluate_erdos_solution(h, dx)

        return np.sum(h) * dx - 1.0

    from scipy.optimize import minimize
    
    # Relaxing constraints slightly for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    try:

        h_initial = h_constructed.copy()
