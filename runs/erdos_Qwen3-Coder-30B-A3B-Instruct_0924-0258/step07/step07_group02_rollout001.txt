import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def repair_solution(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def crossover(parent1, parent2, alpha=0.5):
    return alpha * parent1 + (1 - alpha) * parent2

def mutate(h, mutation_rate, n_points):
    noise = np.random.normal(0, mutation_rate, n_points)
    h_new = h + noise
    return repair_solution(h_new, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a simple construction
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm parameters
    pop_size = 50
    generations = 200
    mutation_rate = 0.05
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.05, n_points)
        h = repair_solution(h, n_points)
        population.append(h)
    
    # EA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        best_indices = sorted_indices[:pop_size//4]
        
        # Update best solution
        if fitness[sorted_indices[0]] < best_c5:
            best_h = population[sorted_indices[0]].copy()
            best_c5 = fitness[sorted_indices[0]]
        
        # Create new population
        new_population = [population[i] for i in best_indices]  # Elite
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = population[np.random.choice(best_indices)]
            parent2 = population[np.random.choice(best_indices)]
            
            child = crossover(parent1, parent2)
            child = mutate(child, mutation_rate, n_points)
            new_population.append(child)
        
        population = new_population[:pop_size]
        
        # Adaptive mutation rate
        if gen % 20 == 0 and gen > 0:
            mutation_rate *= 0.8
            if mutation_rate < 0.005:
                mutation_rate = 0.005
    
    # Final refinement with L-BFGS
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final repair and verification
    best_h = repair_solution(best_h, n_points)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
