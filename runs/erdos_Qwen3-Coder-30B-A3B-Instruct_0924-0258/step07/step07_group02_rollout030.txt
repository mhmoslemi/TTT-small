import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm approach
    pop_size = 100
    generations = 300
    elite_size = pop_size // 10
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Track best in hall of fame
    hall_of_fame = []
    for h in population:
        fitness = evaluate_erdos_solution(h, dx)
        hall_of_fame.append((h.copy(), fitness))
    
    hall_of_fame.sort(key=lambda x: x[1])
    hall_of_fame = hall_of_fame[:elite_size]
    
    # Mutation parameters
    initial_mutation_rate = 0.1
    mutation_decay = 0.95
    crossover_rate = 0.7
    crossover_decay = 0.98
    
    # Main EA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitnesses = []
        for h in population:
            fitness = evaluate_erdos_solution(h, dx)
            fitnesses.append(fitness)
        
        # Update hall of fame
        for i, h in enumerate(population):
            if fitnesses[i] < hall_of_fame[-1][1]:
                hall_of_fame.append((h.copy(), fitnesses[i]))
                hall_of_fame.sort(key=lambda x: x[1])
                hall_of_fame = hall_of_fame[:elite_size]
        
        # Selection: tournament
        new_pop = []
        for i in range(pop_size):
            # Tournament selection
            idxs = np.random.choice(pop_size, 3, replace=False)
            best_idx = idxs[np.argmin([fitnesses[j] for j in idxs])]
            new_pop.append(population[best_idx].copy())
        
        # Crossover and mutation
        for i in range(0, pop_size, 2):
            if np.random.rand() < crossover_rate:
                # Arithmetic crossover
                alpha = np.random.rand()
                h1, h2 = new_pop[i], new_pop[i+1]
                child1 = alpha * h1 + (1 - alpha) * h2
                child2 = (1 - alpha) * h1 + alpha * h2
                new_pop[i] = child1
                new_pop[i+1] = child2
            
            # Mutation
            if np.random.rand() < initial_mutation_rate:
                mut_strength = max(0.01, initial_mutation_rate * (mutation_decay ** gen))
                for j in range(n_points):
                    if np.random.rand() < 0.2:  # 20% of elements
                        new_pop[i][j] += np.random.normal(0, mut_strength)
                        new_pop[i+1][j] += np.random.normal(0, mut_strength)
                # Repair
                new_pop[i] = np.clip(new_pop[i], 0, 1)
                new_pop[i] = new_pop[i] * (n_points/2) / np.sum(new_pop[i])
                new_pop[i+1] = np.clip(new_pop[i+1], 0, 1)
                new_pop[i+1] = new_pop[i+1] * (n_points/2) / np.sum(new_pop[i+1])
        
        population = new_pop
        
        # Adaptive parameters
        initial_mutation_rate = max(0.01, initial_mutation_rate * mutation_decay)
        crossover_rate = max(0.3, crossover_rate * crossover_decay)
        
        # Early stopping
        if gen > 0 and gen % 50 == 0:
            recent_improvement = False
            if len(hall_of_fame) >= 2:
                recent_improvement = hall_of_fame[0][1] < hall_of_fame[-1][1]
            if not recent_improvement:
                break
    
    # Final selection from hall of fame
    best_h = hall_of_fame[0][0]
    best_c5 = hall_of_fame[0][1]
    
    # Final SQP refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
