import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from copy import deepcopy

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    elite_size = 5
    max_generations = 2000
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if 'initial_h_values' in globals() and i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fit = evaluate_erdos_solution(h, dx)
        fitness.append(fit)
        if fit < best_c5:
            best_h = h.copy()
            best_c5 = fit
    
    # GA iterations
    generation = 0
    stagnation_count = 0
    prev_best = best_c5
    
    while generation < max_generations and time() - start_time < budget_s * 0.95:
        generation += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track stagnation
        if abs(fitness[0] - prev_best) < 1e-8:
            stagnation_count += 1
        else:
            stagnation_count = 0
        prev_best = fitness[0]
        
        # If stagnation, restart with new random population
        if stagnation_count > 50:
            population = []
            fitness = []
            for i in range(pop_size):
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points/2) / np.sum(h)
                population.append(h)
                fit = evaluate_erdos_solution(h, dx)
                fitness.append(fit)
                if fit < best_c5:
                    best_h = h.copy()
                    best_c5 = fit
            stagnation_count = 0
        
        # Create new population
        new_population = []
        new_fitness = []
        
        # Elitism: keep the best
        for i in range(elite_size):
            new_population.append(population[i].copy())
            new_fitness.append(fitness[i])
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            p1_idx = tournament_select(population, fitness, tournament_size)
            p2_idx = tournament_select(population, fitness, tournament_size)
            
            if np.random.rand() < crossover_rate:
                child1, child2 = crossover(population[p1_idx], population[p2_idx], n_points)
            else:
                child1, child2 = population[p1_idx].copy(), population[p2_idx].copy()
            
            # Mutation
            child1 = mutate(child1, mutation_rate, n_points)
            child2 = mutate(child2, mutation_rate, n_points)
            
            # Repair constraints
            child1 = repair_constraint(child1, n_points, dx)
            child2 = repair_constraint(child2, n_points, dx)
            
            # Evaluate children
            fit1 = evaluate_erdos_solution(child1, dx)
            fit2 = evaluate_erdos_solution(child2, dx)
            
            new_population.extend([child1, child2])
            new_fitness.extend([fit1, fit2])
            
            # Update best
            if fit1 < best_c5:
                best_h = child1.copy()
                best_c5 = fit1
            if fit2 < best_c5:
                best_h = child2.copy()
                best_c5 = fit2
        
        # Trim to exact population size
        population = new_population[:pop_size]
        fitness = new_fitness[:pop_size]
        
        # Periodic convex relaxation (every 50 generations)
        if generation % 50 == 0 and generation > 0:
            try:
                for i in range(min(10, len(population))):
                    h_candidate = population[i]
                    # Use convex relaxation for refinement
                    h_refined = convex_refine(h_candidate, n_points, dx, best_c5)
                    if h_refined is not None:
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
            except Exception:
                pass
    
    # Final local optimization on best solution
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)

def tournament_select(population, fitness, k):
    selected_indices = np.random.choice(len(population), k)
    best_idx = selected_indices[np.argmin([fitness[i] for i in selected_indices])]
    return best_idx

def crossover(parent1, parent2, n_points):
    # Uniform crossover
    mask = np.random.rand(n_points) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(individual, mutation_rate, n_points):
    mutated = individual.copy()
    num_mutation = int(n_points * mutation_rate)
    mutation_indices = np.random.choice(n_points, num_mutation, replace=False)
    for idx in mutation_indices:
        mutated[idx] += np.random.normal(0, 0.03)
    return mutated

def repair_constraint(individual, n_points, dx):
    # Ensure sum constraint
    individual = np.clip(individual, 0, 1)
    current_sum = np.sum(individual)
    if abs(current_sum - n_points/2) > 1e-6:
        individual = individual * (n_points/2) / current_sum
    return individual

def convex_refine(h_candidate, n_points, dx, current_bound):
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraint to potentially improve solution
        # Ensure that C5 is not worse than current bound
        # or at least a substantial improvement over current best
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            return h_feasible
    except Exception:
        pass
    return None
