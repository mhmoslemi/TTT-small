import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based optimization
    max_iter_phase1 = int(budget_s * 0.8 / 0.01)  # Rough estimate for iterations
    learning_rate = 0.1
    stagnation_counter = 0
    last_best = best_c5
    
    # Random restarts
    restarts = 5
    for restart in range(restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        h = h_start.copy()
        h *= (n_points / 2) / np.sum(h)  # Ensure constraint
        h = np.clip(h, 0, 1)
        
        for i in range(max_iter_phase1):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute all correlations to find maximum
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            max_val = corr[max_idx]
            
            # Compute gradient for the worst lag
            # Gradient for element at index j under lag k: 
            # for forward: -1 for j+k, for backward: 1 for j-k
            grad = np.zeros_like(h)
            for j in range(len(h)):
                # Contribution from h[j] to corr[max_idx]
                # Find k that corresponds to max_idx in correlation
                k = max_idx - len(h) + 1  # lag value
                if 0 <= j + k < len(h):
                    grad[j] -= 1
                if 0 <= j - k < len(h) and k != 0:
                    grad[j] += 1
                    
            # Apply gradient and line search
            h_new = h - learning_rate * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new solution
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                stagnation_counter = 0
            else:
                stagnation_counter += 1
                
            if stagnation_counter > 50:
                learning_rate *= 0.5
                stagnation_counter = 0
                
            h = h_new
            
            # Random perturbation
            if i % 100 == 0 and i > 0:
                perturbation = np.random.normal(0, 0.01, len(h))
                h = h + perturbation
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
    
    # Phase 2: Local refinement with L-BFGS-B
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try different starting points
    for _ in range(3):
        if time() - start_time > budget_s * 0.95:
            break
        try:
            start_h = best_h + np.random.normal(0, 0.01, len(best_h))
            start_h = np.clip(start_h, 0, 1)
            start_h = start_h * (n_points / 2) / np.sum(start_h)
            
            res = minimize(objective, start_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    # Final check
    best_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), best_c5, n_points)
