from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)

    max_iter = 1000
    lr = 0.01
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradients via finite differences
        h_current = best_h.copy()
        corr_current = np.correlate(h_current, 1 - h_current, mode='full')
        max_k_idx = np.argmax(corr_current)
        max_lag = max_k_idx - (n_points - 1)
        
        # Sample a few key lags for gradient estimation
        sample_k_indices = []
        for k in range(n_points//2 - 5, n_points//2 + 5):
            if 0 <= k < n_points:
                sample_k_indices.append(k)
        
        # For each sample point, compute approximate gradient
        grad = np.zeros(n_points)
        epsilon = 1e-4
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += epsilon

        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Reduce learning rate
        lr *= 0.999
        
        # Check for convergence
        if iter_num % 50 == 0 and iter_num > 0:

            pass
    
    # Final optimization via constrained convex optimization
    try:
        # Use a more structured convex optimization
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Compute the maximum overlap directly
        prob = cp.Problem(cp.Minimize(0), constraints)

    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
