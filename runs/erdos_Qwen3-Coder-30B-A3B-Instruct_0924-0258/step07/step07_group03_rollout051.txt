import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based optimization with adaptive learning rate
    max_iter = 1000
    learning_rate = 0.1
    patience = 0
    patience_limit = 50
    
    # Initialize history
    h_current = h_start.copy()
    current_c5 = best_c5
    
    # Gradient descent
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute all overlaps
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        max_lag = max_idx - (n_points - 1)
        
        # Compute gradient for the worst-case lag
        gradient = np.zeros(n_points)
        # Forward difference approximation around max_lag
        for i in range(n_points):
            # Calculate contribution to the overlap at lag max_lag
            if i + max_lag >= 0 and i + max_lag < n_points:
                gradient[i] += (1 - h_current[i + max_lag])
            if i - max_lag >= 0 and i - max_lag < n_points:
                gradient[i] += h_current[i - max_lag]
        
        # Gradient descent step
        h_new = h_current - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to satisfy constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Update
        if new_c5 < current_c5:
            h_current = h_new
            current_c5 = new_c5
            patience = 0
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
        else:
            patience += 1
            if patience >= patience_limit:
                learning_rate *= 0.5
                patience = 0
                
    # Phase 2: Local refinement with L-BFGS-B
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Create constraints
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try several restarts
    restarts = 5
    for i in range(restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Slightly perturb best solution
        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            res = minimize(objective, h_perturbed, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 1000, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
