import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_h(n_points):
    x = np.linspace(0, 2, n_points)
    h_initial = 0.5 * (1 + np.sin(2 * np.pi * x))
    h_initial = np.clip(h_initial, 0, 1)
    normalization = (n_points / 2) / np.sum(h_initial)
    h_initial *= normalization
    return h_initial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 50
        h_start = initialize_h(n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.998
    max_iter = 10000
    tabu_size = 20
    tabu_list = []
    
    # Main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate
        h_candidate = best_h.copy()
        indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        for i in indices:
            perturb = np.random.uniform(-0.15, 0.15)
            h_candidate[i] += perturb
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        delta = c5_candidate - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            
            # Tabu list update
            if len(tabu_list) >= tabu_size:
                tabu_list.pop(0)
            tabu_list.append(h_candidate.copy())
            
        # Annealing
        T *= alpha
        
        # Check stagnation
        if iteration % 100 == 0 and iteration > 0:
            if abs(np.diff([evaluate_erdos_solution(h, dx) for h in tabu_list[-5:]]).mean()) < 1e-10:
                T = max(T * 0.9, 1e-6)
    
    # Final SLSQP refinement
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
