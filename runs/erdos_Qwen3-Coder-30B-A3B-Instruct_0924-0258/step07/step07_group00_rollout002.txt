import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient-based iterative improvement
    max_iter = 2000
    step_size = 0.01
    tolerance = 1e-9
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h) - 1)
        
        # Compute gradient for this specific lag
        grad = np.zeros_like(h)
        for j in range(len(h)):
            k = (j + max_lag) % len(h)
            grad[j] = 1 - 2 * h[k]
        
        # Perform gradient step
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Compute new overlap value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Update h
        h = h_new.copy()
        
        # Reduce step size on constraint violation or no improvement
        if (np.sum(h) < n_points/2 - 1e-6 or 
            np.sum(h) > n_points/2 + 1e-6 or 
            c5_new >= best_c5):
            step_size *= 0.5
            if step_size < 1e-6:
                break
        
    return (list(best_h), best_c5, n_points)
