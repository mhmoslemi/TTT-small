import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

    dx = 2.0 / n_points
    
    # Try a simple two-interval construction
    # Set h = 1 on [a, a+0.5] and [b, b+0.5] where a = 0.65, b = 0.85
    a = 0.65
    b = 0.85
    h_candidate = np.zeros(n_points)
    
    # Find indices corresponding to intervals
    start_a = int(a / dx)
    end_a = int((a + 0.5) / dx)
    start_b = int(b / dx)
    end_b = int((b + 0.5) / dx)
    
    # Clamp indices to valid range
    start_a = max(0, min(start_a, n_points - 1))
    end_a = max(0, min(end_a, n_points))
    start_b = max(0, min(start_b, n_points - 1))
    end_b = max(0, min(end_b, n_points))
    
    h_candidate[start_a:end_a] = 1.0
    h_candidate[start_b:end_b] = 1.0
    
    # Normalize to have integral = 1
    integral = np.sum(h_candidate) * dx
    h_candidate = h_candidate * (1.0 / integral)
    
    c5_candidate = evaluate_erdos_solution(h_candidate, dx)
    
    # For small improvement, perform gradient descent on selected points
    # Only optimize indices that are 0 < h < 1
    indices = np.where((h_candidate > 0) & (h_candidate < 1))[0]
    if len(indices) == 0:
        return (list(h_candidate), c5_candidate, n_points)
    
    best_h = h_candidate.copy()
    best_c5 = c5_candidate
    
    # Gradient descent with fixed step size on allowed indices
    lr = 0.01
    max_iter = 1000
    tolerance = 1e-8
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Randomly select an index to perturb
        idx = np.random.choice(indices)
        
        # Perturb
        h_new = best_h.copy()
        h_new[idx] += lr * np.random.uniform(-1, 1)
        h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Re-normalize
        integral = np.sum(h_new) * dx
        h_new = h_new * (1.0 / integral)
        
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            
            # Recompute indices if they changed
            indices = np.where((best_h > 0) & (best_h < 1))[0]
        
        # Break if no improvement
        if i > 10 and abs(best_c5 - c5_candidate) < tolerance:
            break
    
    return (list(best_h), best_c5, n_points)
