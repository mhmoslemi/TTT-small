import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary algorithm implementation
    pop_size = 50
    generations = 100
    elite_size = 5
    mutation_rate = 0.1
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Generate random feasible solution
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
    
    # Find initial best
    best_idx = np.argmin(fitness)
    if fitness[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    # Evolutionary loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        elite_fitness = [fitness[i] for i in sorted_indices[:elite_size]]
        
        # Create new population
        new_population = elite.copy()
        new_fitness = elite_fitness.copy()
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = sorted_indices[np.random.randint(0, pop_size//2)]
            parent2_idx = sorted_indices[np.random.randint(0, pop_size//2)]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Blend crossover
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
            
            # Ensure feasibility
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
            c5 = evaluate_erdos_solution(child, dx)
            new_fitness.append(c5)
        
        population = new_population[:pop_size]
        fitness = new_fitness[:pop_size]
        
        # Update global best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
    
    # Post-processing refinement using gradient-based method
    try:
        h_refined = cp.Variable(n_points)
        constraints = [
            cp.sum(h_refined) == n_points / 2.0,
            h_refined >= 0,
            h_refined <= 1
        ]
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_refined.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final clip and renormalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
