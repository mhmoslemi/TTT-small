import numpy as np
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Start with a tight initial bound
    max_overlap_initial = 0.38085
    tighten_factor = 0.995
    max_iter = 50
    
    # Try to solve with MILP approach
    h_opt = None
    c5_opt = float('inf')
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # McCormick envelopes approach
        # Create optimization variables
        h = cp.Variable(n_points, name='h')
        max_overlap = cp.Variable(name='max_overlap')
        
        # Constraints for h
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Add McCormick envelope constraints for bilinear terms
        # h(i) * (1 - h(i + k)) terms
        # Add overlap constraint: max_overlap >= correlation_k
        # We will use a simpler tight constraint approach:
        # Basic constraint with max_overlap as upper bound
        # We want to minimize max_overlap subject to bilinear approximations
        # For now simplify into a convex upper bound
        # But since we're trying to minimize, we'll do SLSQP refinement
        
        # Since we can't do full-MM in a simple solver for all terms,
        # let's refine the existing solution via SLSQP
        # But first try the direct MILP approach for a stricter solver
        
        # So we establish a whole optimization problem to bound the maximum
        # We'll optimize with a small number of constraints and variable
        # Every iteration do a better tightening based on last optimal
        
        # Committing to refine with local search
        
        # Check if we have a better current solution from previous iteration
        h_refine = best_h.copy()
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_points/2) / np.sum(h_refine)
        
        # Define optimization objective:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        # Use generic SLSQP to refine existing solution
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Refinement with SLSQP starting from current best
        try:
            # Adjust bounds for SLSQP with constraints
            res = minimize(objective, h_refine, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception as e:
            pass
        
        # Final test of current best if it's still reasonable
        if best_c5 < max_overlap_initial:
            max_overlap_initial = best_c5 * tighten_factor
        
        # Occasionally solve with stricter bound
        if iteration % 5 == 0:
            try:
                # Use a controlled variable approach to old-school tightening
                h_refine = np.clip(best_h, 0, 1)
                h_refine = h_refine * (n_points/2) / np.sum(h_refine)
                res = minimize(objective, h_refine, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-9})
                if res.success:
                    h_final = res.x
                    c5_final = objective(h_final)
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
            except Exception:
                pass
    
    # Final refinement
    try:
        h_refine = best_h.copy()
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_points/2) / np.sum(h_refine)
        res = minimize(objective, h_refine, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
