import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        # Use a structured starting point: sine wave with low amplitude
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.1 * np.sin(2 * np.pi * x)
        h_start = np.clip(h_start, 0, 1)
    
    # Normalize to sum(h) = n_points / 2
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Objective function for differential evolution
    def de_objective(h):
        # Normalize to correct sum
        h = h * (n_points / 2) / np.sum(h)
        # Clip to bounds
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h, dx)
    
    # Bounds for differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Differential evolution with adaptive control
    de_options = {
        'popsize': 50,
        'maxiter': 1000,
        'mutation': (0.5, 1.0),
        'recombine': 0.9,
        'tol': 1e-6,
        'strategy': 'best1bin'
    }
    
    # Perform differential evolution
    try:
        result = differential_evolution(
            de_objective, 
            bounds, 
            **de_options,
            seed=seed,
            maxiter=1000,
            disp=False
        )
        if result.success and result.fun < best_c5:
            best_h = np.clip(result.x * (n_points / 2) / np.sum(result.x), 0, 1)
            best_c5 = result.fun
    except Exception:
        pass
    
    # Local refinement with SLSQP using best result from DE
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(
            objective, 
            best_h.copy(),
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons_sum, 
            options={'maxiter': 1000, 'ftol': 1e-10},
            callback=lambda x: None
        )
        if res.success:
            h_final = res.x
            h_final = np.clip(h_final * (n_points / 2) / np.sum(h_final), 0, 1)
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final validation and return
    final_h = np.clip(best_h * (n_points / 2) / np.sum(best_h), 0, 1)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
