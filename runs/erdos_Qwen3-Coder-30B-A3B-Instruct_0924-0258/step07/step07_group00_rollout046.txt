import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def refine_with_SLSQP(h_init, dx, n_points):
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1)] * n_points
    
    result = minimize(objective, h_init, method='SLSQP',
                      constraints=cons, bounds=bounds,
                      options={'maxiter': 500, 'ftol': 1e-9})
    if result.success:
        h_refined = result.x
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2.0) / np.sum(h_refined)
        return h_refined
    return h_init

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()

    h = h * (n_points / 2.0) / np.sum(h)
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Main optimization loop with gradient descent
    eta = 0.01
    max_iterations = 1000
    no_improvement_count = 0
    max_no_improvement = 100
    
    for iter in range(max_iterations):
        # Compute all overlaps
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # If no improvement for a while, break
        if no_improvement_count >= max_no_improvement:
            break
            
        if iter % 50 == 0:
            eta = 0.01
            
        # Compute gradient more carefully
        gradient = np.zeros(n_points)
        for k in range(-n_points + 1, n_points):
            for i in range(n_points):
                j = (i + k) % n_points
                gradient[i] += (h[j] - 1)
        
        gradient = gradient * dx * 2
        
        # Update using gradient descent
        h_new = h - eta * gradient
        
        # Ensure bounds and sum constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        # Evaluate new
        current_c5 = evaluate_erdos_solution(h_new, dx)

            no_improvement_count = 0
            eta = max(eta, 0.01)

            no_improvement_count += 1
            
        h = h_new

    best_h = refine_with_SLSQP(best_h, dx, n_points)
    c5_bound = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), c5_bound, n_points)
