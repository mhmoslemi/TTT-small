import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Structured initial guess: sine wave with small amplitude
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.2 * np.sin(2 * np.pi * x)
        h_start = np.clip(h_start, 0, 1)
    
    # Normalize to correct integral
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint for the integral
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Lagrangian objective with penalty for constraint violation
    def lagrangian_objective(h, penalty_weight=1e3):
        overlap = evaluate_erdos_solution(h, dx)
        constraint_violation = abs(constraint_sum(h))
        return overlap + penalty_weight * constraint_violation
    
    # Differential Evolution optimizer
    try:
        # Run DE with adaptive termination based on time budget
        max_iter = 400
        current_iter = 0
        de_result = None
        
        while current_iter < max_iter and time() - start_time < budget_s * 0.8:
            # Adaptive penalty weight based on iterations
            penalty_weight = 1e3 * (1 + current_iter / max_iter * 10)
            
            # Perform DE with current penalty
            de_result = differential_evolution(
                lambda h: lagrangian_objective(h, penalty_weight),
                bounds,
                maxiter=20,
                popsize=50,
                mutation=(0.5, 1.0),
                recombination=0.9,
                seed=seed,
                disp=False,
                workers=1
            )
            
            if de_result.success:
                h_candidate = de_result.x
                # Re-normalize to correct integral
                h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
            current_iter += 20
    except Exception as e:
        pass
    
    # Local refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds_slsqp = [(0, 1) for _ in range(n_points)]
        
        # Use best solution from DE as initial guess for SLSQP
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        res = minimize(objective, h_refined, method='SLSQP',
                       bounds=bounds_slsqp, constraints=cons_sum,
                       options={'maxiter': 500, 'ftol': 1e-10})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final validation
    h_validated = best_h.copy()
    h_validated = np.clip(h_validated, 0, 1)
    h_validated = h_validated * (n_points / 2) / np.sum(h_validated)
    c5_validated = evaluate_erdos_solution(h_validated, dx)
    
    return (list(h_validated), c5_validated, n_points)
