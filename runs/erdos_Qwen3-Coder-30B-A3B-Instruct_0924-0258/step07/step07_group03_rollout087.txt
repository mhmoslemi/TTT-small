import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convex relaxation for initial solution
    def convex_relaxation(h0, n_points, dx):
        # Variable for h
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2.0
        ]
        
        # Objective: Minimize maximum correlation
        # This is a bit tricky since max is not convex.
        # We'll use a different approach: minimize sum of all correlations
        # which should be similar in practice
        
        # Try to find a global min via convex relaxation
        # Use proxy: lower bound for the maximum overlap
        # Create matrix variable for h*h^T to get convex relaxation
        Z = cp.Variable((n_points, n_points), PSD=True)
        constraints.append(Z == h @ h.T)
        
        # Bound the entries of Z
        for i in range(n_points):
            constraints.append(Z[i, i] <= 1)
        
        # Objective approximation: minimize trace of Z 
        # This will be a very loose relaxation, but provides a feasible starting point
        # Better approach: use constraint programming for finding a good h
        try:
            # Use simpler iterative approach:
            return h0.copy()
        except:
            return h0.copy()
    
    # Use convex relaxation to improve starting point
    try:
        h_convex = convex_relaxation(h_start, n_points, dx)
        h_init = h_convex
        if np.all(h_init >= 0) and np.all(h_init <= 1):
            h_init = h_init * (n_points/2) / np.sum(h_init)
            c5_convex = evaluate_erdos_solution(h_init, dx)
            if c5_convex < best_c5:
                best_h = h_init.copy()
                best_c5 = c5_convex
    except:
        h_init = h_start.copy()
    
    # Global optimization with different method
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try several optimizers with different starting points
    
    # Local search using L-BFGS-B
    try:
        res1 = minimize(objective, h_init, method='L-BFGS-B', 
                       bounds=bounds, options={'ftol': 1e-10, 'gtol': 1e-10})
        if res1.success:
            h_lbfgs = res1.x
            c5_lbfgs = objective(h_lbfgs)
            if c5_lbfgs < best_c5:
                best_h = h_lbfgs.copy()
                best_c5 = c5_lbfgs
    except:
        pass
    
    # Local search using SLSQP 
    try:
        res2 = minimize(objective, h_init, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res2.success:
            h_slsqp = res2.x
            c5_slsqp = objective(h_slsqp)
            if c5_slsqp < best_c5:
                best_h = h_slsqp.copy()
                best_c5 = c5_slsqp
    except:
        pass
    
    # Try more random restarts
    n_restarts = 10
    for restart in range(n_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbed start
        h_random = np.random.uniform(0, 1, n_points)
        h_random = h_random * (n_points/2) / np.sum(h_random)
        
        try:
            res = minimize(objective, h_random, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 500, 'ftol': 1e-8})
            if res.success:
                h_trial = res.x
                c5_trial = objective(h_trial)
                if c5_trial < best_c5:
                    best_h = h_trial.copy()
                    best_c5 = c5_trial
        except:
            pass
    
    # Final improvement using pure local search
    try:
        res_final = minimize(objective, best_h, method='SLSQP', 
                            bounds=bounds, constraints=cons_sum,
                            options={'maxiter': 1000, 'ftol': 1e-12})
        if res_final.success:
            h_final = res_final.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    except:
        pass
    
    # Ensure feasibility one more time 
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
