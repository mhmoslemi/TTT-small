import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Balanced ramp function as starting point
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(np.pi * x / 1.0))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)

    # Evolutionary algorithm parameters
    pop_size = 50
    max_generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 5

    # Initialize population
    population = []
    for _ in range(pop_size):
        if np.random.rand() < 0.3:  # 30% chance to use initial solution with noise
            noise = np.random.normal(0, 0.05, n_points)
            h = best_h + noise
        else:
            h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)

    def repair(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h

    # Evolution loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitness[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = sorted_fitness[0]
        
        # Selection: keep top 50%
        top_half = sorted_population[:pop_size//2]
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())
            
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            idx1 = np.random.choice(len(top_half))
            idx2 = np.random.choice(len(top_half))
            parent1 = top_half[idx1]
            parent2 = top_half[idx2]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
            
            # Repair and add to population
            child = repair(child)
            new_population.append(child)
        
        population = new_population
        
        # Gradually reduce mutation rate
        mutation_rate = max(0.01, mutation_rate * 0.99)

    # Final L-BFGS refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_func(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_func}
        
        # Try refinement for top 10 candidates
        top_candidates = []
        fitness_vals = []
        for h in population[:10]:
            c5 = evaluate_erdos_solution(h, dx)
            top_candidates.append(h)
            fitness_vals.append(c5)
            
        # Refine the best candidates
        for i, h in enumerate(top_candidates):
            if fitness_vals[i] < best_c5:
                try:
                    res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                                 constraints=constraint, options={'maxiter': 500})
                    if res.success:
                        refined_h = res.x
                        refined_h = np.clip(refined_h, 0, 1)
                        refined_h = refined_h * (n_points/2) / np.sum(refined_h)
                        refined_c5 = evaluate_erdos_solution(refined_h, dx)
                        if refined_c5 < best_c5:
                            best_h = refined_h
                            best_c5 = refined_c5
                except:
                    pass
    except Exception:
        pass

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
