def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Augmented Lagrangian method
    max_iter = 5000
    tol = 1e-6
    
    # Initialize parameters
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)

    mu = 100.0     # Penalty parameter
    
    # Track history for convergence
    prev_c5 = float('inf')
    
    def compute_gradient(h, active_k):
        """Compute gradient of max overlap with respect to h"""
        n = len(h)
        grad = np.zeros(n)
        # For each active k, compute gradient of correlation
        for k in active_k:
            if k < n:
                # Gradient of h_i * (1 - h_{i+k}) w.r.t. h_i is (1 - h_{i+k})
                # and w.r.t. h_{i+k} is -h_i
                for i in range(n - k):
                    grad[i] += (1 - h[i+k])
                for i in range(n - k):
                    grad[i+k] -= h[i]
        # Normalize by n * dx
        grad = grad / (n * dx)
        return grad
    
    def get_active_k(h):
        """Get indices k where overlap is maximal"""
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        # Find all k values that give the maximum overlap
        active_k = []
        for i, val in enumerate(corr):
            if abs(val - max_val) < 1e-8:
                # Convert index to actual lag k (0-based indexing in correlate)
                k = i - (n_points - 1)  # This makes k range from -n+1 to n-1
                if k >= 0:
                    active_k.append(k)
        return active_k

    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current max overlap and active k values
        c5 = evaluate_erdos_solution(h, dx)
        active_k = get_active_k(h)

            break
        prev_c5 = c5

        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        try:
            grad = compute_gradient(h, active_k)
        except:
            # fallback gradient calculation if needed
            grad = np.zeros_like(h)

        grad = np.clip(grad, -1, 1)
        
        # Try updating with a step
        step_size = 0.05  # Start with a fixed step size
        h_new = h - step_size * grad
        
        # Project h_new to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check improvement
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept update if improvement
        if c5_new <= c5:

            # Update penalty parameter (adaptive)
            mu *= 0.99  # Mild decay
        else:
            # Reject update, reduce penalty parameter
            mu *= 1.1
            
        # Project h again to ensure constraints
        h = np.clip(h, 0, 1)

            return evaluate_erdos_solution(h, dx)

            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
