import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a starting point
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimizer settings
    max_iter = 100
    pop_size = 20

    # Initialize population with perturbations
    population = []
    for _ in range(pop_size):
        h_perturb = h_start + np.random.normal(0, 0.02, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
        population.append(h_perturb)
    
    # Main optimization loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness_scores.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        top_fitness = fitness_scores[sorted_indices[0]]
        if top_fitness < best_c5:
            best_h = population[sorted_indices[0]].copy()
            best_c5 = top_fitness
            
        # Select top individuals
        top_indices = sorted_indices[:pop_size//2]
        top_pop = [population[i] for i in top_indices]
        
        # Generate new population
        new_pop = top_pop[:]
        while len(new_pop) < pop_size:
            # Crossover
            parent1, parent2 = np.random.choice(top_pop, size=2, replace=False)
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            # Mutation
            noise = np.random.normal(0, 0.01, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            new_pop.append(child)
            
        population = new_pop[:pop_size]
    
    # Final refinement using scipy minimize with bounds
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_integral(h):
        return np.sum(h) - n_points / 2.0
        
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': constraint_integral},
    ]
    
    try:
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-6}
        )
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass

    # Final check
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
