import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based local optimization with SLSQP
    max_iter = 2000
    gamma = 0.1  # Learning rate
    tolerance = 1e-10
    last_improvement = 0
    improvement_threshold = 50
    
    h_current = best_h.copy()
    current_c5 = best_c5
    
    # Local optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_current, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': tolerance})
            
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                
                if c5_new < current_c5:
                    h_current = h_new
                    current_c5 = c5_new
                    last_improvement = 0
                    if current_c5 < best_c5:
                        best_h = h_current.copy()
                        best_c5 = current_c5
                else:
                    last_improvement += 1
                    
                if last_improvement > improvement_threshold:
                    break
            else:
                # Fallback: perturb current solution
                h_new = h_current + np.random.normal(0, 0.01, n_points)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                c5_new = objective(h_new)
                
                if c5_new < current_c5:
                    h_current = h_new
                    current_c5 = c5_new
                    last_improvement = 0
                    if current_c5 < best_c5:
                        best_h = h_current.copy()
                        best_c5 = current_c5
                else:
                    last_improvement += 1
                    
        except:
            # Fallback: random perturbation
            h_new = h_current + np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = objective(h_new)
            
            if c5_new < current_c5:
                h_current = h_new
                current_c5 = c5_new
                last_improvement = 0
                if current_c5 < best_c5:
                    best_h = h_current.copy()
                    best_c5 = current_c5
            else:
                last_improvement += 1
            
            if last_improvement > improvement_threshold:
                break
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
