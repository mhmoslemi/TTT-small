import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try a different approach: use sequential quadratic programming
    # combined with heuristic refinement
    
    # Start with better initial solution
    h_current = h_start.copy()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Quadratic programming approach with bounds
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    n_starts = 5

        if time() - start_time > budget_s * 0.9:
            break

        h_perturbed = h_current + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        try:
            res = minimize(objective, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 300, 'ftol': 1e-10})
            if res.success:
                h_candidate = res.x
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate

                break

            h_samp = np.clip(h_samp, 0, 1)
            
            try:
                res = minimize(objective, h_samp, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-10})
                if res.success:
                    h_candidate = res.x
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_grid_c5:

        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)

                final_h = res.x

        except Exception:

    # Ensure final result is valid
    best_h = np.clip(best_h, 0, 1)

    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
