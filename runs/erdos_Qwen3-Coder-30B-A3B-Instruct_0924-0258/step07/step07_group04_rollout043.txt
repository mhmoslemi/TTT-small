import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create structured initial guess
        h_start = 0.5 + 0.1 * np.sin(np.linspace(0, 2*np.pi, n_points))
        h_start = np.clip(h_start, 0, 1)
    
    # Normalize to correct integral
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds for differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints for the optimization
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Wrapper for differential evolution
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Differential evolution optimization
    de_options = {
        'maxiter': 500,
        'popsize': 50,
        'tol': 1e-6,
        'init': 'latinhypercube',
        'recombination': 0.9,
        'strategy': 'best1bin'
    }
    
    try:
        result = differential_evolution(
            objective, 
            bounds, 
            constraints={'type': 'eq', 'fun': constraint_sum},
            **de_options,
            seed=seed,
            disp=False
        )
        
        if result.success:
            h_de = result.x
            c5_de = objective(h_de)
            
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
                
    except Exception:
        pass
    
    # Local refinement using SLSQP
    try:
        def objective_slsqp(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum_slsqp(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum_slsqp}
        bounds_slsqp = [(0, 1) for _ in range(n_points)]
        
        refined_result = minimize(
            objective_slsqp,
            best_h,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons_sum,
            options={'maxiter': 1000, 'ftol': 1e-10}
        )
        
        if refined_result.success:
            h_final = refined_result.x
            c5_final = objective_slsqp(h_final)
            
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
                
    except Exception:
        pass
    
    # Final check and enforce constraint
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    c5_final = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), c5_final, n_points)
