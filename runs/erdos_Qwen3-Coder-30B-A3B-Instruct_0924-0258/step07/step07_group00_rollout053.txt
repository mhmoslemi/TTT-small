import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
        h = h * (n_points/2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    # Precompute full correlation indices
    full_len = 2 * n_points - 1
    lags = np.arange(-n_points + 1, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient-based optimization with focus on max lag
    max_iter = 10000
    step_size = 0.01
    tol = 1e-9
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = lags[max_lag_idx]
        max_overlap = corr[max_lag_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Compute gradient w.r.t. h for the max lag
        # For lag k, d/dh_i of overlap integral is (1 - h[i+k]) if k >= 0 and h[i-k] if k < 0
        grad = np.zeros(n_points)
        if max_lag >= 0:
            # h[i] contributes to overlap with h[i+max_lag] term
            for i in range(n_points - max_lag):
                grad[i] += (1 - h[i + max_lag])
        else:
            # h[i] contributes to overlap with h[i+max_lag] term (i+max_lag < 0 means we look at the other side)
            for i in range(-max_lag, n_points):
                grad[i] -= h[i + max_lag]
        
        # SGD update
        h_new = h - step_size * grad
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy constraint sum(h) = n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if the improvement is small, then reduce step size
        h = h_new
        
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
