from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    gamma = 0.1
    stagnation_count = 0
    
    h_current = h_start.copy()

    h_current = h_current * (n_points/2) / np.sum(h_current)

            break
            
        # Compute correlations
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag = np.argmax(corr)
        
        # Create direction vector for Frank-Wolfe
        # We approximate with a simple gradient-like update
        s = np.zeros_like(h_current)
        s[0] = 1.0  # Simple direction towards decreasing overlap
        
        # Projected gradient step (Frank-Wolfe style)
        h_new = h_current - gamma * s
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to correct sum
        h_new = h_new * (n_points/2) / np.sum(h_new)

        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:

                best_c5 = c5_perturbed
                stagnation_count = 0
            
        # Update current point
        h_current = h_new.copy()
        
        if stagnation_count > 100:

            return evaluate_erdos_solution(h, dx)
