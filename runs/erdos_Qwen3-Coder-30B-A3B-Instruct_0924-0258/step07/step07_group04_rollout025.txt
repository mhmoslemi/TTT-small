import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_h(n_points):
    x = np.linspace(0, 2, n_points)
    h_initial = 0.5 * (1 + np.sin(2 * np.pi * x))
    h_initial = np.clip(h_initial, 0, 1)
    normalization = (n_points / 2) / np.sum(h_initial)
    h_initial *= normalization
    return h_initial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 50
        h_start = initialize_h(n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Annealing parameters
    T = 1.0
    alpha = 0.998
    tabu_size = 20
    tabu_list = []
    max_iter = 10000
    
    # Simulated annealing with tabu search
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by perturbing a random subset of points
        candidate = best_h.copy()
        indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        perturbation_magnitude = 0.1 * T
        
        for i in indices:
            delta = np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
            candidate[i] += delta
        
        # Ensure bounds
        candidate = np.clip(candidate, 0, 1)
        
        # Renormalize
        candidate = candidate * (n_points/2) / np.sum(candidate)
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(candidate, dx)
        
        # Check tabu list
        candidate_tuple = tuple(candidate)
        if candidate_tuple in tabu_list:
            # Accept with low probability
            if np.random.random() > 0.1:
                continue
        
        # Accept or reject based on simulated annealing
        if c5_candidate < best_c5:
            # Accept better solution
            best_h = candidate.copy()
            best_c5 = c5_candidate
            # Add to tabu list
            tabu_list.append(candidate_tuple)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        else:
            # Accept worse solution with probability
            delta = c5_candidate - best_c5
            if np.random.random() < np.exp(-delta / T):
                best_h = candidate.copy()
                best_c5 = c5_candidate
                tabu_list.append(candidate_tuple)
                if len(tabu_list) > tabu_size:
                    tabu_list.pop(0)
        
        # Cool down
        T = max(0.001, T * alpha)
    
    # Final SLSQP refinement
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
