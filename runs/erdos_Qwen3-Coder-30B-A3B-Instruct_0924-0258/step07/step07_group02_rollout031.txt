import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm with DE-style mutation
    pop_size = 100
    max_gen = 300
    mutation_factor = 0.8
    crossover_rate = 0.7
    elite_size = 10
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Random initial population
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
        # Normalize to exact constraint
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # EA loop
    gen = 0
    no_improvement_count = 0
    last_best = best_c5
    
    while gen < max_gen and time() - start_time < budget_s * 0.95:
        gen += 1
        
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        
        # Check for stagnation
        if abs(last_best - best_c5) < 1e-6:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            last_best = best_c5
            
        if no_improvement_count > 80:
            break
            
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            idxs = np.random.choice(pop_size, 3, replace=False)
            best_idx = idxs[np.argmin([fitness[i] for i in idxs])]
            selected.append(population[best_idx].copy())
        
        # Crossover and mutation
        new_pop = []
        for i in range(0, pop_size, 2):
            # Some individuals don't mutate to maintain diversity
            if np.random.rand() < crossover_rate:
                # DE-style mutation
                a, b, c = selected[np.random.choice(pop_size, 3, replace=False)]
                trial = a + mutation_factor * (b - c)
                # Crossover
                mask = np.random.rand(n_points) < 0.7
                child1 = np.where(mask, trial, selected[i])
                child2 = np.where(mask, trial, selected[i+1]) if i+1 < pop_size else selected[i]
            else:
                child1 = selected[i].copy()
                child2 = selected[i+1].copy() if i+1 < pop_size else selected[i].copy()
            
            # Repair to constraints
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_pop.append(child1)
            if i+1 < pop_size:
                new_pop.append(child2)
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        for i in range(elite_size):
            new_pop[-(elite_size - i)] = population[elite_indices[i]].copy()
            
        population = new_pop
        
        # Adaptive mutation factor
        mutation_factor = 0.8 * (1 - gen / max_gen * 0.5)
        if mutation_factor < 0.1:
            mutation_factor = 0.1

    # Final SQP refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]

        # Add key constraints from best solution
        h_test = best_h.copy()
        corr = np.correlate(h_test, 1 - h_test, mode='full')
        top_lags = np.argsort(corr)[-10:]
        for k in top_lags:
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
