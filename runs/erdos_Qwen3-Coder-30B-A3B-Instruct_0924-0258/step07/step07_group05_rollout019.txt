from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Benders decomposition parameters
    max_iter = 5000
    convergence_tol = 1e-6
    perturb_mag = 0.05
    perturb_decay = 0.95
    constraint_freq = 50
    
    # Initialize
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure constraint satisfaction
    prev_c5 = float('inf')
    
    # Main Benders loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all possible shifts
        corr_full = np.correlate(h, 1 - h, mode='full')
        # Only consider relevant shifts (for which we have full correlation)
        overlap_vals = corr_full[len(h)-1:] * dx  # This gives actual overlaps for k = 0 to n-1
        
        # Find worst-case shift (max overlap)
        max_overlap_idx = np.argmax(overlap_vals)
        max_overlap_val = overlap_vals[max_overlap_idx]
        
        # Store the best solution so far
        if max_overlap_val < best_c5:
            best_h = h.copy()
            best_c5 = max_overlap_val
        
        # Add constraint for worst overlap
        # Constraint: sum_{i=0}^{n-1-k} h_i * (1 - h_{i+k}) <= max_overlap_val
        # We are directly modeling this at the constraint level
        
        # Convex relaxation - reformulate as a feasibility problem
        # But rather than adding constraints, we'll directly solve the total problem
        # and dominate with improvement
        
        # Generate new candidate via perturbation
        h_new = h + perturb_mag * np.random.randn(n_points)

                return evaluate_erdos_solution(h_candidate, dx)

                return np.sum(h_candidate) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_new, method='SLSQP',

            else:
                h = h_new

            return evaluate_erdos_solution(h, dx)
