import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient-based optimization with focus on worst lag
    max_iter = 10000
    step_size = 0.01
    tolerance = 1e-9
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps (correlations) for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        # The correlation length is 2*n_points - 1, centered at n_points-1
        # We want to maximize h(i) * (1 - h(i+k)) for all k
        # So for each lag k, we have:
        # h[0]*1-h[k] + h[1]*1-h[k+1] + ... + h[n_points-1-k]*1-h[n_points-1]
        # which is significantly complicated to compute directly
        
        # More direct approach
        gap_corr = np.correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(gap_corr)
        max_overlap = np.max(gap_corr) * dx
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Compute gradient wrt. the current worst lag
        # This part is simplified - we'll compute the gradient manually
        # Our gradient direction should be approximately:
        # For the lag where correlation is max, we compute finite difference
        # for each element in h
        # And we update using gradient descent for that maximum location 
        
        # Do fixed number of gradient steps (small number)
        for step in range(20):
            # Compute the correlation at all lags
            gaps = np.correlate(h, 1-h, mode='full')

            max_lag = np.argmax(gaps)
            # Extract the specific correlation from that lag
            # For the l-th lag, we compute:
            # sum_h_i * (1 - h_{i+l})
            # Also the contribution to the Jacobian of max_lag
            # We optimize one lag at a time, focusing on maximum
            
            # Let's compute a more direct approximation of gradient
            # For a fixed lag l, the contribution to the lag l overlap

            # Its partial derivative w.r.t. h[i] is (1 - h[i+l]) 
            # and w.r.t. h[i+l] is -h[i]
            
            # Gradient approximation:

            grad = np.zeros(n_points)

            lag_start = max_lag - (n_points - 1)

                if i + lag_start >= 0 and i + lag_start < n_points:

                    grad[i] += 1 - h[i + lag_start]

            # Normalize gradient
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 0:

            # Update h (gradient descent)
            h_new = h - step_size * grad

        best_c5 = final_c5

    return (list(best_h), best_c5, n_points)
