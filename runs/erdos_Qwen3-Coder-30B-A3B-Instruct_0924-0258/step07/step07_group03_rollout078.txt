import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient Descent with adaptive learning rate
    max_iter = 2000
    learning_rate = 0.1
    patience = 50
    patience_counter = 0
    
    h_current = best_h.copy()
    current_c5 = best_c5
    
    # Phase 1: Gradient Descent
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute all overlaps
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag = np.argmax(corr)
        max_overlap = corr[max_lag] * dx
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Contribution to overlap at max_lag
            # Lag k corresponds to correlation[i] = sum(h[j] * (1 - h[j+k]))
            lag = max_lag - n_points + 1  # Equivalent lag
            if lag < 0:  # h[j+k] exists for j < n_points + lag
                for j in range(n_points + lag):
                    if j + lag >= 0:  # Only contribute if valid index
                        if i == j:
                            grad[i] -= 1  # From h[j]
                        if i == j + lag:
                            grad[i] -= 1  # From 1-h[j+k]
            else:  # h[j] exists for j >= lag
                for j in range(lag, n_points):
                    if i == j:
                        grad[i] -= 1  # From h[j]
                    if i == j - lag:
                        grad[i] -= 1  # From 1-h[j-k]

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm * learning_rate
        else:
            grad = np.zeros_like(grad)

        # Update
        h_new = h_current - grad
        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)
        if sum_h > 0:
            h_new = h_new * (n_points / 2) / sum_h
        else:
            h_new = np.ones(n_points) * 0.5

        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < current_c5:
            h_current = h_new
            current_c5 = new_c5
            patience_counter = 0
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
        else:
            patience_counter += 1
            if patience_counter > patience:
                learning_rate *= 0.5
                patience_counter = 0
    
    # Phase 2: Final L-BFGS refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='L-BFGS-B', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Random restarts
    for _ in range(5):
        if time() - start_time > budget_s * 0.95:
            break
        h_random = np.random.uniform(0, 1, n_points)
        h_random = h_random * (n_points / 2) / np.sum(h_random)
        c5_random = evaluate_erdos_solution(h_random, dx)
        if c5_random < best_c5:
            best_h = h_random
            best_c5 = c5_random
    
    return (list(best_h), best_c5, n_points)
