import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))  # 5 cycles over [0,2]
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
        # Add small random perturbation
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multiresolution refinement
    resolutions = [20, 40, 80]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        n_points = res
        dx = 2.0 / n_points
        
        # Interpolate from previous solution if needed
        if res > len(best_h):
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_new, x_old, best_h)
            h_start = np.clip(h_interp, 0, 1)
            h_start = h_start * (n_points/2) / np.sum(h_start)
        else:
            h_start = best_h[:n_points].copy()
            h_start = h_start * (n_points/2) / np.sum(h_start)
        
        # Annealing parameters
        max_iter = 200 if res == 80 else 150 if res == 40 else 100
        perturbation_size = 0.15
        T = 1.0
        T_decay = 0.99
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Select random points to perturb
            n_perturb = max(1, int(0.2 * n_points))
            indices = np.random.choice(n_points, n_perturb, replace=False)
            
            # Generate candidate
            h_candidate = best_h.copy()
            for idx in indices:
                delta = np.random.uniform(-perturbation_size, perturbation_size) * h_candidate[idx]
                h_candidate[idx] += delta
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
            
            # Evaluate
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept or reject
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
            else:
                # Accept with probability
                delta_c5 = c5_candidate - best_c5
                if np.random.rand() < np.exp(-delta_c5 / T):
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                    
            # Update perturbation and temperature
            if iteration % 50 == 0 and iteration > 0:
                perturbation_size = max(0.05, perturbation_size * 0.9)
            T = max(0.01, T * T_decay)
    
    # Final refinement with SLSQP
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
