import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_h(n_points):
    x = np.linspace(0, 2, n_points)
    h_initial = 0.5 * (1 + np.sin(2 * np.pi * x))
    h_initial = np.clip(h_initial, 0, 1)
    normalization = (n_points / 2) / np.sum(h_initial)
    h_initial *= normalization
    return h_initial

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    T = 1.0                # Initial temperature
    alpha = 0.998          # Cooling rate
    tabu_size = 20
    tabu_list = []
    max_iter = 10000
    
    # Perturbation parameters
    perturb_min = 0.05
    perturb_max = 0.15
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by perturbing a random subset of indices
        indices_to_perturb = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        h_candidate = best_h.copy()
        
        # Apply perturbations
        for idx in indices_to_perturb:
            perturb_factor = np.random.uniform(perturb_min, perturb_max)
            if np.random.random() < 0.5:
                h_candidate[idx] *= (1 + perturb_factor)
            else:
                h_candidate[idx] *= (1 - perturb_factor)
        
        # Clip to bounds
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Renormalize
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Tabu check
        candidate_tuple = tuple(h_candidate.round(6))
        if candidate_tuple in tabu_list:
            # Accept with low probability
            if np.random.random() > 0.9:
                pass
            else:
                continue
                
        # Accept or reject
        delta = c5_candidate - best_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            
            # Add to tabu list
            if len(tabu_list) >= tabu_size:
                tabu_list.pop(0)
            tabu_list.append(candidate_tuple)
            
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
        
        # Cool down
        T *= alpha
    
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 500, 'ftol': 1e-12})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
