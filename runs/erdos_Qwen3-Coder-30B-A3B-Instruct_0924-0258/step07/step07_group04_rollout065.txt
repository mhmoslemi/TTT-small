import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a smooth construction that has good properties
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.2 * np.sin(3 * np.pi * x)  # Sine wave to balance
        h_start = np.clip(h_start, 0, 1)
    
    # Normalize to correct integral
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Set up DE bounds (0 to 1 for all variables)
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        # Add penalty for constraint violations
        penalty = 0
        if np.sum(h) < n_points/2 - 1e-8:
            penalty += 1e6 * (n_points/2 - np.sum(h))
        if np.sum(h) > n_points/2 + 1e-8:
            penalty += 1e6 * (np.sum(h) - n_points/2)
        
        # Compute C5 with constraints
        c5 = evaluate_erdos_solution(h, dx)
        return c5 + penalty

    # Differential Evolution setup
    de_options = {
        'strategy': 'best1bin',
        'maxiter': 400,
        'popsize': 50,
        'tol': 1e-6,
        'mutation': (0.5, 1.0),
        'recombination': 0.9,
        'polish': False
    }
    
    # DE optimization
    try:
        result = differential_evolution(
            objective, 
            bounds,
            **de_options,
            seed=seed,
            callback=lambda x, convergence: None,
            maxiter=de_options['maxiter']
        )
        
        if result.success:
            h_de = result.x
            # Normalize to sum constraint
            h_de = h_de * (n_points/2) / np.sum(h_de)
            c5_de = evaluate_erdos_solution(h_de, dx)

            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
                
    except Exception as e:
        pass
    
    # Local refinement with SLSQP around the best found so far
    try:
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        def objective_slsqp(h):
            return evaluate_erdos_solution(h, dx)
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds_slsqp = [(0, 1) for _ in range(n_points)]
        
        # Re-normalize best_h for localization
        h_refined = best_h * (n_points/2) / np.sum(best_h)
        
        res = minimize(
            objective_slsqp, 
            h_refined,
            method='SLSQP',
            bounds=bounds_slsqp, 
            constraints=cons_sum, 
            options={'maxiter': 500, 'ftol': 1e-10}
        )
        
        if res.success:
            h_final = res.x
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
                
    except Exception as e:
        pass
    
    # Ensure final solution is valid
    valid_h = np.clip(best_h, 0, 1)
    valid_h = valid_h * (n_points/2) / np.sum(valid_h)
    final_c5 = evaluate_erdos_solution(valid_h, dx)
    
    return (list(valid_h), final_c5, n_points)
