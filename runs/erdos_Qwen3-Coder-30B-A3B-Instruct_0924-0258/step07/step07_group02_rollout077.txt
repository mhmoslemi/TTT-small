import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initial construction
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary algorithm parameters
    pop_size = 60
    max_gen = 200
    mutation_rate = 0.1
    crossover_rate = 0.7
    elitism_rate = 0.1
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    def repair(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Main EA loop
    for gen in range(max_gen):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]
        fitness = [fitness[i] for i in sorted_idx]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Select elite
        elite_size = max(1, int(elitism_rate * pop_size))
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite[:]
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = population[np.random.randint(0, pop_size//2)]
            parent2 = population[np.random.randint(0, pop_size//2)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
            
            # Repair
            child = repair(child)
            new_population.append(child)
        
        population = new_population
        # Adaptive mutation
        mutation_rate = max(0.01, mutation_rate * 0.995)
    
    # Final L-BFGS refinement on top solutions
    top_solutions = population[:10]
    for h in top_solutions:
        # Create helper function for optimization
        def objective(h_vec):
            h_clipped = np.clip(h_vec, 0, 1)
            h_normalized = h_clipped * (n_points/2) / np.sum(h_clipped)
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points/2
        
        try:
            result = minimize(
                objective, 
                h.copy(), 
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': constraint_func},
                options={'maxiter': 100, 'ftol': 1e-9}
            )
            
            if result.success:
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points/2) / np.sum(h_opt)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except:
            pass
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
