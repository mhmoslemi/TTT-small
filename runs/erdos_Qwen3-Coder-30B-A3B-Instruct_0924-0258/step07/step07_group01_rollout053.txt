import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize CVXPY variables
    h = cp.Variable(n_points)
    
    # Create a preliminary version for simpler optimization
    h_current = h_start.copy()
    
    # Iterative refinement using CVXPY
    max_cvx_iterations = 20
    for iteration in range(max_cvx_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Fix the problem size for CVXPY
        try:
            # Create a dual formulation to handle the max operator
            # Add a variable that captures the maximum overlap
            max_overlap = cp.Variable()
            
            # Constraint: max_overlap >= (h * (1-h_shifted)) for all shifts
            constraints = [h >= 0, h <= 1]
            # Need to build the correlations manually to avoid circular dependencies
            
            # Instead, use direct constraint solving approach
            h_refined = h_current.copy()
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            
            # Use SLSQP for refinement
            def objective(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            def constraint_sum(h_vec):
                return np.sum(h_vec) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-10})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
            else:
                h_final = h_refined
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
                    
            h_current = h_final.copy()
            
        except Exception:
            # fallback to simple refinement
            h_current = h_refined.copy()
            
    return (list(best_h), best_c5, n_points)
