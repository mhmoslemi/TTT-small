import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T0 = 0.1
    cooling_rate = 0.001
    max_iter = 1000
    perturbation_fraction = 0.15
    
    h_current = h_start.copy()
    c5_current = best_c5
    T = T0
    
    stagnant_count = 0
    stagnation_limit = 100
    
    for _ in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create new candidate
        h_new = h_current.copy()
        indices = np.random.choice(n_points, int(n_points * perturbation_fraction), replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-0.1, 0.1)
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        delta = c5_new - c5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_current = h_new
            c5_current = c5_new
            stagnant_count = 0
        else:
            stagnant_count += 1
            
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        T *= np.exp(-cooling_rate)
        
        if stagnant_count >= stagnation_limit:
            break
    
    # Gradient Descent refinement
    h_refined = best_h.copy()
    learning_rate = 0.1
    max_gd_iter = 500
    gd_stagnant = 0
    gd_stagnation_limit = 50
    
    for _ in range(max_gd_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Compute correlation and find k_max
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        k_max = np.argmax(corr)
        max_corr = corr[k_max]
        
        # Compute gradient specific to k_max
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                gradient[i] += (1 - h_refined[i + k_max]) * dx
            if i - k_max >= 0:
                gradient[i] += (1 - h_refined[i - k_max]) * dx
                
        # Update
        h_refined = h_refined - learning_rate * gradient
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        c5_new = evaluate_erdos_solution(h_refined, dx)
        if c5_new < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_new
            gd_stagnant = 0
        else:
            gd_stagnant += 1
            if gd_stagnant >= gd_stagnation_limit:
                learning_rate *= 0.5
                gd_stagnant = 0
                
    # Final SLSQP optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
