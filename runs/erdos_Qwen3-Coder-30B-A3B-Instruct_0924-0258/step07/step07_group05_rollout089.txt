def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def compute_gradient(h, dx):

    grad = np.zeros(n)
    # Compute all correlations and find the maximum
    corr = np.correlate(h, 1 - h, mode='full')
    max_k = np.argmax(corr)
    max_val = corr[max_k]
    
    # For the maximum lag k, compute gradient contribution
    k = max_k - (n - 1)
    if k >= 0:
        # h[i] * (1 - h[i+k]) for i = 0 to n-1-k
        for i in range(n - k):
            grad[i] += (1 - h[i + k]) * dx
    else:
        # h[i] * (1 - h[i+k]) for i = -k to n-1
        k_abs = -k
        for i in range(k_abs, n):
            grad[i] += (1 - h[i - k_abs]) * dx
            
    return grad
