import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization
    max_iter = 1000
    step_size = 0.01
    stagnation_limit = 50
    stagnation_count = 0
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # normalize to sum = n_points/2
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(corr)
        max_overlap = corr[max_lag]
        
        # Compute gradient for the most constraining lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # The contribution to the overlap at lag max_lag
            j = (i + max_lag - (n_points - 1)) % n_points
            gradient[i] = 2.0 * (1 - h[j] - h[i])
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm
        
        # Update
        h_new = h - step_size * gradient
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better, otherwise reduce step size
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            stagnation_count = 0
        else:
            stagnation_count += 1
            step_size *= 0.9
            
        if stagnation_count > stagnation_limit:
            # Random restart
            h = np.random.rand(n_points)
            h = h * (n_points/2) / np.sum(h)
            stagnation_count = 0
            
        # Occasionally refine with scipy
        if iteration % 50 == 0 and iteration > 0:
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(objective, h, method='SLSQP', 
                              bounds=bounds, constraints=cons_sum, 
                              options={'maxiter': 100, 'ftol': 1e-9})
                
                if res.success:
                    h = res.x
                    h = np.clip(h, 0, 1)
                    h = h * (n_points/2) / np.sum(h)
                    c5_refined = objective(h)
                    if c5_refined < best_c5:
                        best_c5 = c5_refined
                        best_h = h.copy()
            except Exception:
                pass
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h, method='SLSQP', 
                      bounds=bounds, constraints=cons_sum, 
                      options={'maxiter': 300, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
