import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Initialize with periodic function for better starting point
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    T = 0.5  # Initial temperature
    alpha = 0.995  # Cooling rate
    T_min = 1e-5  # Minimum temperature
    max_iter = 1000
    tabu_list = []
    tabu_size = 20
    stagnation_limit = 50
    stagnation_count = 0
    perturbation_size = 0.1
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Helper function to ensure constraints are maintained
    def ensure_constraints(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Tabu list management
    def is_tabu(h, tabu_list, tolerance=1e-5):
        for past_h in tabu_list:
            if np.allclose(h, past_h, atol=tolerance):
                return True
        return False
    
    def update_tabu(h, tabu_list, tabu_size):
        if len(tabu_list) >= tabu_size:
            tabu_list.pop(0)
        tabu_list.append(h.copy())
        return tabu_list
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate solution
        candidate_h = current_h.copy()
        # Select random indices to perturb
        indices_to_perturb = np.random.choice(n_points, size=int(0.15 * n_points), replace=False)
        
        for i in indices_to_perturb:
            # Perturb by small random value
            delta = np.random.uniform(-perturbation_size, perturbation_size)
            candidate_h[i] += delta
        
        # Ensure constraints
        candidate_h = ensure_constraints(candidate_h)
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Accept or reject based on simulated annealing
        if candidate_c5 < current_c5:
            # Always accept better solution or if it's not tabu
            if not is_tabu(candidate_h, tabu_list):
                current_h = candidate_h.copy()
                current_c5 = candidate_c5
                stagnation_count = 0
                # Update best solution if improved
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
        else:
            # Accept worse solution with probability based on temperature
            if not is_tabu(candidate_h, tabu_list):
                delta = candidate_c5 - current_c5
                acceptance_prob = math.exp(-delta / T)
                if np.random.random() < acceptance_prob:
                    current_h = candidate_h.copy()
                    current_c5 = candidate_c5
                    stagnation_count = 0
                else:
                    stagnation_count += 1

        # Update tabu list
        if not is_tabu(current_h, tabu_list):
            tabu_list = update_tabu(current_h, tabu_list, tabu_size)
        
        # Temperature update
        T = max(T * alpha, T_min)
        
        # Stagnation handling
        if stagnation_count >= stagnation_limit:
            # Reset with some perturbation
            current_h = best_h.copy()
            perturbation = np.random.normal(0, 0.05, n_points)
            current_h += perturbation
            current_h = ensure_constraints(current_h)
            current_c5 = evaluate_erdos_solution(current_h, dx)
            stagnation_count = 0
            # Adjust perturbation size
            perturbation_size = max(0.02, perturbation_size * 0.9)
            
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
