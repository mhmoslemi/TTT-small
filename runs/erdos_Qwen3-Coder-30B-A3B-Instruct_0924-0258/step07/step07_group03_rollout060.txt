import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convex relaxation approach: use SDP to get initial feasible point
    # Phase 1: SDP relaxation
    try:
        # Try to solve a convex relaxation
        h_relaxed = cp.Variable(n_points, name='h')
        Z = cp.Variable((n_points, n_points), name='Z', symmetric=True)
        
        # Constraints for convex relaxation
        constraints = [
            cp.trace(Z) <= n_points/2,  # sum(h) <= n_points/2
            Z >> 0,  # Z is positive semidefinite
            Z[cp.arange(n_points), cp.arange(n_points)] == h_relaxed,  # diagonal elements
            Z[cp.arange(n_points), cp.arange(n_points)] >= 0,  # h >= 0
        ]
        
        # This is a simplified approach, just use a 
        # simple linearization or projection method
        
        # Simple projection + local refinement approach
        h_refined = h_start.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        c5_current = evaluate_erdos_solution(h_refined, dx)
        
        if c5_current < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_current

        # Phase 2: Local optimization with SLSQP
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            # Try quadratic programming first
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except:
            pass
            
        # Try second SLSQP run with different starting point
        if time() - start_time > budget_s * 0.9:
            return (list(best_h), best_c5, n_points)
            
        # Random perturbation with constraint
        h_perturbed = h_refined.copy() + np.random.normal(0, 0.001, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        try:
            res = minimize(objective, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except:
            pass
        
        # Exhaustive local search on a few random samples
        if time() - start_time > budget_s * 0.9:
            return (list(best_h), best_c5, n_points)
            
        # Try a few more local optimizations
        for _ in range(10):
            if time() - start_time > budget_s * 0.95:
                break
            h_trial = np.random.uniform(0, 1, n_points)
            h_trial = h_trial * (n_points/2) / np.sum(h_trial)
            try:
                res = minimize(objective, h_trial, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 200, 'ftol': 1e-9})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = c5_refined
            except:
                continue
    except Exception as e:
        pass
    
    return (list(best_h), best_c5, n_points)
