import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Gradient computation
    def compute_gradient(h, dx):
        # Compute overlap integrals and find active k
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Find all k that contribute to the maximum
        # Considering the structure of correlation
        n = len(h)
        # For sparse correlation, compute only relevant terms
        # Active k is around max_idx - n + 1 to max_idx
        active_k = []
        for k in range(1, n):
            if max_idx >= k and max_idx < 2 * n - 1 - k:
                active_k.append(k)
        
        if not active_k:
            active_k = [1, 2] if n > 2 else [1]
            
        # Compute gradient for the active lags
        gradient = np.zeros(n)
        for k in active_k:
            # Gradient w.r.t. h_i: (1 - h_{i+k}) + (1 - h_{i-k}) for i such that i+k < n
            # Use correlation structure to compute only necessary contributions
            # h_i * (1 - h_{i+k}) for all i where it contributes
            if k < n:
                for i in range(n - k):
                    gradient[i] += 1 - h[i + k]
            
        return gradient / n
    
    # Initial h
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Optimization parameters
    max_iter = 5000
    lr = 0.05
    decay = 0.95
    patience = 50
    patience_counter = 0
    best_obj = float('inf')
    
    # Main optimization loop
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current objective and gradient
        current_obj = objective(h_current)
        grad = compute_gradient(h_current, dx)
        
        # Update step with gradient
        # Project onto feasible region
        h_new = h_current - lr * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to satisfy constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        new_obj = objective(h_new)
        
        if new_obj < current_obj:
            h_current = h_new.copy()
            lr = min(0.5, lr * 1.05)  # Adaptive learning rate
            if new_obj < best_obj:
                best_obj = new_obj
                best_h = h_current.copy()
                best_c5 = new_obj
                patience_counter = 0
            else:
                patience_counter += 1
        else:
            lr = lr * decay
            patience_counter += 1
            
        # Stopping condition
        if patience_counter >= patience:
            break
            
        # Progress log (periodic)
        if i % 100 == 0:
            pass
    
    # Final polishing with scipy minimize
    try:
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use just the best current solution as start
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
