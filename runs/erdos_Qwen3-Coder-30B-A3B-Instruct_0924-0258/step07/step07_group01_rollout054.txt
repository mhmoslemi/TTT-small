import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try CVXPY optimization approach
    max_iter = 20
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create CVXPY variables
        h_var = cp.Variable(n_points)
        
        # Objective: minimize max overlap (upper bound approach)
        # We'll use a two-step approach:
        # 1. Relax the problem to find boundaries
        # 2. Refine with local optimization
        
        # Setup initial solution
        h_init = best_h.copy()
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points/2) / np.sum(h_init)
        
        # Try CVXPY solution
        try:
            # Create constraints
            constr = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points/2]
            
            # Define overlapping correlation manually in terms of h_var
            # The maximum overlap is max_lag np.correlate(h, 1-h, mode='full')[lag]
            # We can set up these maximums as constraints and minimize over them
            
            # For now, just try an SLSQP refinement
            def objective(h_vals):
                return evaluate_erdos_solution(h_vals, dx)
                
            def constraint_sum(h_vals):
                return np.sum(h_vals) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-10})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except Exception:
            pass
        
        # Perform local SLSQP refinement on current best
        try:
            h_local = best_h.copy()
            h_local = np.clip(h_local, 0, 1)
            h_local = h_local * (n_points/2) / np.sum(h_local)
            
            def objective(h_vals):
                return evaluate_erdos_solution(h_vals, dx)
            
            def constraint_sum(h_vals):
                return np.sum(h_vals) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_local, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-10})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except Exception:
            pass
        
        # Progressively relaxing the constraints to converge
        if it % 5 == 0 and it > 0:
            # Utility function to project back onto regularized set
            h_proj = best_h.copy()
            h_proj = np.clip(h_proj, 0, 1)
            h_proj = h_proj * (n_points/2) / np.sum(h_proj)
            best_h = h_proj.copy()
    
    return (list(best_h), best_c5, n_points)
