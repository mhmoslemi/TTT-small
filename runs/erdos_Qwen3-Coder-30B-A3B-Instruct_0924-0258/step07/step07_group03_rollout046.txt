import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with adaptive cooling
    T = 0.1
    cooling_rate = 0.001
    max_iter = 1000
    perturbation_fraction = 0.15
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    stagnation_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb h
        h_new = current_h.copy()
        n_perturb = int(n_points * perturbation_fraction)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        perturbations = np.random.uniform(-0.1, 0.1, n_perturb)
        h_new[indices] += perturbations
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        delta = new_c5 - current_c5
        T = 0.1 * np.exp(-cooling_rate * iteration)
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > 100:
            break
    
    # Gradient Descent refinement
    def compute_gradient(h, k_max):
        grad = np.zeros_like(h)
        n = len(h)
        for i in range(n):
            if i + k_max < n:
                grad[i] += (1 - h[i + k_max]) * dx
            if i - k_max >= 0:
                grad[i] += (1 - h[i - k_max]) * dx
        return grad
    
    # Find the lag with max overlap
    def get_k_max(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.argmax(corr)
    
    learning_rate = 0.1
    gd_iter = 500
    for iteration in range(gd_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        k_max = get_k_max(best_h)
        grad = compute_gradient(best_h, k_max)
        
        # Update
        new_h = best_h - learning_rate * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
        else:
            # Reduce learning rate if no improvement
            learning_rate *= 0.99
            if learning_rate < 1e-8:
                break
    
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
