import numpy as np
from scipy.optimize import minimize
from time import time
from scipy.spatial.distance import cdist

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define basis functions: piecewise linear triangular pulses
    m = 10  # number of basis functions
    basis_centers = np.linspace(0.1, 1.9, m)
    basis_widths = 0.4 * np.ones(m)
    basis_matrix = np.zeros((n_points, m))
    
    x_vals = np.linspace(0, 2, n_points)
    for i in range(m):
        # Create triangular basis function
        basis_matrix[:, i] = np.maximum(0, 1 - np.abs(x_vals - basis_centers[i]) / basis_widths[i])
    
    # Normalize basis matrix to have unit L2 norm
    basis_norms = np.sqrt(np.sum(basis_matrix**2, axis=0)) + 1e-12
    basis_matrix = basis_matrix / basis_norms
    
    # Initial coefficients
    coeffs = np.linalg.lstsq(basis_matrix, h_start, rcond=None)[0]
    coeffs = np.clip(coeffs, -1, 1)
    
    # Optimizer parameters
    max_iter = 2000
    trust_radius = 0.1
    radius_factor = 2.0
    eval_count = 0
    
    def compute_h_from_coeffs(c):
        return basis_matrix @ c
    
    def compute_overlap_from_coeffs(c):
        h = compute_h_from_coeffs(c)
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h, dx)
    
    def compute_constraint_violation(c):
        h = compute_h_from_coeffs(c)
        h = np.clip(h, 0, 1)
        violation = np.abs(np.sum(h) - n_points/2.0)
        return violation
    
    def trust_region_subproblem(c0, grad, hess, trust_radius):
        def obj(delta):
            c = c0 + delta
            h = compute_h_from_coeffs(c)
            h = np.clip(h, 0, 1)
            corr = np.correlate(h, 1 - h, mode='full')
            max_overlap = np.max(corr) * dx
            return max_overlap
            
        def grad_obj(delta):
            # Simplified gradient estimate via finite differences
            eps = 1e-6
            g = np.zeros_like(delta)
            for i in range(len(delta)):
                delta_plus = delta.copy()
                delta_plus[i] += eps
                delta_minus = delta.copy()
                delta_minus[i] -= eps
                g[i] = (obj(delta_plus) - obj(delta_minus)) / (2 * eps)
            return g
        
        # Simple line search for trust region step
        c_step = c0 - 0.1 * grad
        if np.linalg.norm(c_step - c0) > trust_radius:
            c_step = c0 + trust_radius * (c_step - c0) / np.linalg.norm(c_step - c0)
        
        return c_step
    
    # Trust-region optimization loop
    c = coeffs.copy()
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute objective and gradient approximation
        h = compute_h_from_coeffs(c)
        h = np.clip(h, 0, 1)
        current_c5 = evaluate_erdos_solution(h, dx)
        constraint_violation = compute_constraint_violation(c)
        
        # Update best solution
        if current_c5 < best_c5 and constraint_violation < 1e-6:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros(m)

            c_plus[i] += eps
            c_minus = c.copy()
            c_minus[i] -= eps

        new_h = new_h * (n_points / 2.0) / sum_h

        c_new = np.clip(c_new, -1, 1)
        
        # Update
        c = c_new.copy()
        
        # Update trust region radius
        if current_c5 <= best_c5:
            trust_radius *= radius_factor
        else:
            trust_radius /= radius_factor
        trust_radius = np.clip(trust_radius, 1e-4, 1.0)
        
        # Periodic recovery from stagnation
        if iteration > 0 and iteration % 100 == 0:
            if current_c5 >= best_c5:
                # Perturb coefficients
                perturb_mag = 0.05 * (1.0 - iteration / max_iter)
                c += np.random.normal(0, perturb_mag, m)
                c = np.clip(c, -1, 1)
    
    # Final refinement if needed
    def objective(c):
        h = compute_h_from_coeffs(c)
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h, dx)
    
    def constraint_fun(c):
        h = compute_h_from_coeffs(c)
        h = np.clip(h, 0, 1)
        return np.sum(h) - n_points / 2.0
    
    try:
        bounds = [(-1, 1) for _ in range(m)]
        cons = {'type': 'eq', 'fun': constraint_fun}
        res = minimize(objective, coeffs, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            final_h = compute_h_from_coeffs(res.x)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
