import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Get initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create periodic initial guess
        n_points = 20
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive perturbation scheme with multiresolution
    max_resolutions = [20, 40, 80]
    
    for n_points in max_resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to new resolution
        if n_points > len(h_start):
            x_old = np.linspace(0, 2, len(h_start))
            x_new = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_new, x_old, h_start)
            h_start = np.clip(h_interp, 0, 1)
        else:
            h_interp = np.linspace(0, 2, n_points)
            x_old = np.linspace(0, 2, len(h_start))
            h_start = np.interp(h_interp, x_old, h_start)
        
        # Ensure sum constraint
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        # Annealing parameters
        temp = 1.0
        temp_decay = 0.99
        perturbation_size = 0.15
        
        iterations = 200 if n_points == 80 else 150 if n_points == 40 else 100
        
        # Tabu list for preventing cycling
        tabu_list = []
        tabu_size = 20
        
        # Main optimization loop
        for i in range(iterations):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Select random subset of points to perturb
            n_perturb = max(1, int(0.15 * n_points))
            indices = np.random.choice(n_points, n_perturb, replace=False)
            
            h_candidate = h_start.copy()
            
            # Apply perturbations
            for idx in indices:
                delta = np.random.uniform(-perturbation_size, perturbation_size) * h_candidate[idx]
                h_candidate[idx] += delta
            
            # Enforce bounds
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Normalize
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Evaluate
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept or reject
            if c5_candidate < best_c5:
                h_start = h_candidate.copy()
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
            else:
                # Accept with probability based on temperature
                delta_c5 = c5_candidate - best_c5
                if np.random.random() < np.exp(-delta_c5 / temp):
                    h_start = h_candidate.copy()
            
            # Update temperature and perturbation size
            if i % 50 == 0:
                temp *= temp_decay
                perturbation_size *= 0.95
            
            # Tabu management
            tabu_list.append(h_candidate.copy())
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
    
    # Final SLSQP refinement
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
