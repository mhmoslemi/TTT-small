import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 200
    mutation_rate = 0.3
    selection_rate = 0.2
    stagnation_limit = 50
    
    # Create initial population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    best_fitness_history = []
    stagnation_count = 0
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Track best in this generation
        gen_best_idx = np.argmin(fitness)
        gen_best = population[gen_best_idx]
        gen_best_fitness = fitness[gen_best_idx]
        
        if gen_best_fitness < best_c5:
            best_h = gen_best.copy()
            best_c5 = gen_best_fitness
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        best_fitness_history.append(gen_best_fitness)
        
        if stagnation_count >= stagnation_limit:
            break
            
        # Selection (keep top 20%)
        selected_indices = np.argsort(fitness)[:int(population_size * selection_rate)]
        selected = [population[i] for i in selected_indices]
        
        # Create new population through crossover and mutation
        new_population = selected.copy()
        
        while len(new_population) < population_size:
            # Select two parents
            parent1 = random.choice(selected)
            parent2 = random.choice(selected)
            
            # Crossover
            alpha = random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child = child + noise
            
            # Clamp to [0,1]
            child = np.clip(child, 0, 1)
            
            # Normalize to correct sum
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:population_size]
    
    # Local refinement with SLSQP for top candidates
    top_candidates = []
    top_fitness = []
    
    for h in population:
        fit = evaluate_erdos_solution(h, dx)
        top_candidates.append(h)
        top_fitness.append(fit)
    
    # Sort candidates
    sorted_indices = np.argsort(top_fitness)
    top_candidates = [top_candidates[i] for i in sorted_indices[:5]]
    
    for h_refined in top_candidates:
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_refined, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 200, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
