import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based descent
    max_iter = 1000
    lr = 0.1
    patience = 50
    best_iter = 0
    
    h_current = h_start.copy()
    current_c5 = best_c5
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute all overlaps and find worst case
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        max_lag = max_idx - (len(h_current) - 1)
        
        # Compute gradient for the worst overlap
        grad = np.zeros_like(h_current)
        for j in range(len(h_current)):
            # Forward contribution 
            if j + max_lag >= 0 and j + max_lag < len(h_current):
                grad[j] += (1 - h_current[j + max_lag])
            # Backward contribution
            if j - max_lag >= 0 and j - max_lag < len(h_current):
                grad[j] += (1 - h_current[j - max_lag])
        
        # Line search to find best step size
        step_size = lr
        new_h = h_current - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        # Scale to correct sum
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < current_c5:
            h_current = new_h.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
                best_iter = i
        else:
            # Reduce learning rate if no improvement
            lr *= 0.5
            if lr < 1e-6:
                lr = 1e-6
                
        # Early stopping condition
        if i - best_iter > patience:
            break
    
    # Phase 2: Local refinement with L-BFGS-B
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        # Try multiple random restarts for better local minimizer
        best_local = best_h.copy()
        best_local_c5 = best_c5
        
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Slightly perturb current best
            h_perturb = best_local + np.random.normal(0, 0.01, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            res = minimize(objective, h_perturb, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons_sum,
                          options={'maxiter': 500, 'ftol': 1e-10})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_local_c5:
                    best_local = h_refined
                    best_local_c5 = c5_refined
        
        if best_local_c5 < best_c5:
            best_h = best_local
            best_c5 = best_local_c5
            
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
