from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        s = (h_start > 1e-6).astype(int)

        current_sum = np.sum(a * s)
        a = a * (n_points/2) / current_sum if current_sum > 0 else a
    else:
        s = np.zeros(n_points)
        # Place active support in the middle roughly
        center = n_points // 2
        support_width = n_points // 2
        s[center - support_width//2 : center + support_width//2] = 1
        s = s.astype(int)
    
    max_iter = 1000
    stagnation_count = 0
    last_improvement = 0
    
    def ensure_feasible(a, s):
        # Ensure a in [0,1]
        a = np.clip(a, 0, 1)

        s = (s > 0.5).astype(int)
        # Repair the sum constraint
        h = a * s
        current_sum = np.sum(h)
        if current_sum > 0:
            a = a * (n_points/2) / current_sum
        return a, s

        h = a * s
        return evaluate_erdos_solution(h, dx)
        
    # Alternating optimization loop
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break

        # Set up CVXPY problem for a:
        # minimize max_lag conv(a*s, 1-a*s) = max_lag corr(a*s, 1-a*s)

        h_initial = a * s

            h_initial = a * s

        scale_factor = target_sum / total_sum if total_sum > 0 else 1.0

            h_var = a_var * s  # element-wise product
            # Modify the sum constraint
            constraints = [cp.sum(h_var) == target_sum, 
                           a_var >= 0, a_var <= 1]
            objective = cp.Minimize(cp.sum(cp.multiply(s, a_var)))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-6)

            last_improvement += 1

            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_refined, method='SLSQP',

        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
