import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T0 = 0.1
    c = 0.001
    max_iter = 1000
    perturbation_fraction = 0.1
    
    # Initialize current solution
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Track best solution
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Annealing loop
    stagnation_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create perturbed solution
        perturbed_h = current_h.copy()
        
        # Select indices to perturb
        n_perturb = max(1, int(n_points * perturbation_fraction))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Perturb selected values
        for idx in indices:
            perturbed_h[idx] += np.random.uniform(-0.1, 0.1)
            
        # Clip values to [0, 1]
        perturbed_h = np.clip(perturbed_h, 0, 1)
        
        # Scale to maintain sum constraint
        scale_factor = (n_points / 2) / np.sum(perturbed_h)
        perturbed_h *= scale_factor
        
        # Evaluate perturbed solution
        perturbed_c5 = evaluate_erdos_solution(perturbed_h, dx)
        
        # Compute temperature
        t = iteration / max_iter
        T = T0 * np.exp(-c * t)
        
        # Accept or reject
        if perturbed_c5 < current_c5:
            current_h = perturbed_h.copy()
            current_c5 = perturbed_c5
            stagnation_count = 0
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(perturbed_c5 - current_c5) / T):
                current_h = perturbed_h.copy()
                current_c5 = perturbed_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        
        # Early stopping
        if stagnation_count > 100:
            break
    
    # Local optimization with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
