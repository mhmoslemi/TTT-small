import numpy as np
from scipy.optimize import minimize
from time import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Load initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize alternating optimization
    # Decompose h = a * s where s is binary support and a is amplitude
    s = np.zeros(n_points, dtype=int)
    if 'initial_h_values' in globals():
        # Initialize s from the initial_h_values
        s = (h_start > 1e-6).astype(int)
    else:
        # Initialize s to half the points uniformly
        s = np.zeros(n_points, dtype=int)
        s[:n_points//2] = 1
        
    # Initialize a (amplitude) on support
    a = np.zeros(n_points)
    a[s == 1] = np.sum(h_start) / np.sum(s)  # Normalize sum
    
    # Ensure feasibility
    h = a * s
    h = h * (n_points/2) / np.sum(h)
    a = h * s
    a = a / (np.sum(s) + 1e-10)  # Prevent division by zero
    # Reset a on support
    a[s == 1] = h[s == 1]
    
    # Alternating optimization loop
    max_iter = 300
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Fix s, optimize a
        # Optimize over a (subject to constraints) = minimize max overlap
        # Objective: C5 = max_lag (convolution(h, 1-h, lag) * dx)
        # To compute max overlap, we want to solve the inner maximization.
        # Using CVXPY to tackle the convex relaxation of this discrete optimization.
        
        # Different approaches:
        # Linear programming approach with constraint handling for the full convolution
        # We'll solve a specific instance using CVXPY where the problem becomes tractable
        
        # However, since direct convex formulation does not trivially exist,
        # we will instead use SLSQP to solve for a, alternating with support update.
        
        # Rather than solving full convex version, we do by sweep:
        # For fixed support s, multiple tries for a
        h_opt = a * s
        h_opt = h_opt * (n_points/2) / np.sum(h_opt)
        c5_opt = evaluate_erdos_solution(h_opt, dx)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            
        # Next: fix a and update s
        # Choose top_k entries of a (which are not zero), set s to 1 there
        # Enforce sum_h = 1 (equiv. sum(a*s) = 1)
        
        # Top indices by amplitude
        top_indices = np.argsort(a)[::-1][:n_points//2]
        s_new = np.zeros(n_points, dtype=int)
        s_new[top_indices] = 1
        # Adjust a to respect sum constraint assuming s_new
        h_now = a * s_new
        scale = (n_points/2) / np.sum(h_now)
        a_new = a * scale
        h_new = a_new * s_new
        h_new = h_new * (n_points/2) / np.sum(h_new)
        c5_test = evaluate_erdos_solution(h_new, dx)
        
        if c5_test < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_test
            
        a = a_new.copy()
        s = s_new.copy()
        h = h_new.copy()
        
        # slight improvement by enforcing hard constraint:
        # project a to lie in [0,1]
        a = np.clip(a, 0, 1)
        
        # If no improvement, break early
        if iteration > 0 and iteration % 10 == 0:
            # Simple periodic check
            pass
            
        # Add termination check for stagnation
        stagnation_count += 1
        if stagnation_count > 100:
            break
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
