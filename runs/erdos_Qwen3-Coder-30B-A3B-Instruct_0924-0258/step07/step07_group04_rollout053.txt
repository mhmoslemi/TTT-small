import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_h(n_points):
    x = np.linspace(0, 2, n_points)
    h_initial = 0.5 * (1 + np.sin(2 * np.pi * x))
    h_initial = np.clip(h_initial, 0, 1)
    normalization = (n_points / 2) / np.sum(h_initial)
    h_initial *= normalization
    return h_initial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        n_points = len(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    T = 1.0
    alpha = 0.998
    tabu_size = 20
    tabu_list = []
    
    current_h = h_start.copy()
    current_c5 = best_c5
    stagnation_counter = 0
    max_stagnation = 100
    
    # Main optimization loop
    while time() - start_time < budget_s * 0.95:
        # Generate candidate move
        candidate_h = current_h.copy()
        
        # Select random indices to perturb
        n_perturb = max(1, int(0.15 * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Perturb selected values
        for i in indices:
            perturbation = np.random.uniform(-0.15, 0.15)
            candidate_h[i] += perturbation
        
        # Clip to bounds
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Normalize to maintain integral constraint
        candidate_h = candidate_h * (n_points/2) / np.sum(candidate_h)
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5:
            current_h = candidate_h
            current_c5 = candidate_c5
            if candidate_c5 < best_c5:
                best_h = candidate_h
                best_c5 = candidate_c5
            stagnation_counter = 0
        else:
            # Probabilistic acceptance (Simulated Annealing)
            delta = candidate_c5 - current_c5
            if np.random.random() < np.exp(-delta / T):
                current_h = candidate_h
                current_c5 = candidate_c5
            stagnation_counter += 1
        
        # Update temperature
        T *= alpha
        
        # Tabu list management
        if len(tabu_list) >= tabu_size:
            tabu_list.pop(0)
        tabu_list.append(current_h.copy())
        
        # Check for stagnation
        if stagnation_counter >= max_stagnation:
            # Restart with new random solution
            current_h = initialize_h(n_points)
            current_c5 = evaluate_erdos_solution(current_h, dx)
            stagnation_counter = 0
        
        # Occasionally adjust perturbation size if we're making progress
        if stagnation_counter % 20 == 0 and T > 0.01:
            T *= 0.95
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP',

        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
