import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        h_start = np.ones(n_points) * 0.5
        h_start[5:15] = 1.0
        h_start[20:30] = 0.0
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 100
    generations = 250
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = 10
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if np.random.rand() < 0.5:
            h = np.random.rand(n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    population[0] = best_h.copy()  # Elite init
    
    # EA loop
    no_improvement_count = 0
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if sorted_fitness[0] < best_c5:
            best_h = sorted_pop[0].copy()
            best_c5 = sorted_fitness[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 80:
            break
            
        # Selection - top 20% automatically advance
        elites = sorted_pop[:elite_size]
        selected = sorted_pop[:pop_size//5]
        
        # Tournament selection
        new_pop = elites.copy()
        
        while len(new_pop) < pop_size:
            # Binary tournament
            i1, i2 = np.random.choice(len(selected), 2, replace=False)
            if fitness[i1] < fitness[i2]:
                parent1 = selected[i1]
            else:
                parent1 = selected[i2]
                
            i1, i2 = np.random.choice(len(selected), 2, replace=False)
            if fitness[i1] < fitness[i2]:
                parent2 = selected[i1]
            else:
                parent2 = selected[i2]
                
            # Crossover
            if np.random.rand() < crossover_rate:
                # Arithmetic blend
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
            
            # Mutation 
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    # Gaussian perturbation
                    child[i] += np.random.normal(0, 0.1)
                    
            # Ensure feasibility
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_pop.append(child)
            
        population = new_pop[:pop_size]
        
    # Final SQP refinement on best solutions
    top_candidates = []
    fitness = []
    for h in population:
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
        
    sorted_indices = np.argsort(fitness)
    for i in range(min(5, len(sorted_indices))):
        top_candidates.append(population[sorted_indices[i]])
    
    for h in top_candidates:
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            # Use a strong constraint approach
            for k in range(n_points // 4):
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
