import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize current solution
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Benders' decomposition parameters
    max_iter = 5000
    perturbation_magnitude = 0.05
    perturbation_decay = 0.95
    convergence_tolerance = 1e-6
    
    # Iterative refinement using Benders decomposition
    prev_c5 = float('inf')
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find the worst shift k by computing all overlaps
        # This is key to Benders approach
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        k_vals = np.arange(len(corr))
        overlaps = corr * dx
        
        # Find the maximum overlap and corresponding k
        max_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_idx]
        k_worst = k_vals[max_idx] - (len(current_h) - 1)  # Convert index to actual shift
        
        # Update the best solution if this is better
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = current_h.copy()
        
        # Add constraint to master problem using current best (by Benders' idea)
        # We only add constraints based on maximum currents
        # To construct a convex inner approximation:
        # We have the current h, we know it violates some constraint at shift k_worst
        # Let's add support constraints from this k
        try:
            # Create a new master problem with current solution
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Constraint from the worst shift we detected
            if k_worst >= 0:
                # For shift k, overlap = sum over i of h[i] * (1 - h[i + k])
                if k_worst < n_points:
                    # The overlap corresponds to convolution term
                    val = cp.sum(cp.multiply(h[:n_points-k_worst], 1-h[k_worst:]))
                    constraints.append(val <= best_c5 * dx + 1e-8)
            else:
                # Handle negative k case (shifts in other direction)
                k_abs = -k_worst
                if k_abs < n_points:
                    val = cp.sum(cp.multiply(h[k_abs:], 1-h[:n_points-k_abs]))
                    constraints.append(val <= best_c5 * dx + 1e-8)
            
            # Solve the master problem
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            if prob.status == cp.OPTIMAL:
                h_new = h.value
                # Check that we got a valid solution
                if h_new is not None:
                    # Perturb slightly to escape local minima
                    h_perturbed = h_new + perturbation_magnitude * np.random.randn(n_points)
                    
                    # Clip values
                    h_perturbed = np.clip(h_perturbed, 0, 1)
                    
                    # Rescale to maintain constraint
                    sum_h = np.sum(h_perturbed)
                    if sum_h > 0:
                        h_perturbed = h_perturbed * (n_points / 2.0) / sum_h

                    # Evaluate new solution
                    c5_new = evaluate_erdos_solution(h_perturbed, dx)
                    current_h = h_perturbed
                    current_c5 = c5_new
            else:
                # Fallback to MIP in case of failure
                pass
                
        except Exception as e:
            pass
            
        # Reduce perturbation magnitude
        perturbation_magnitude *= perturbation_decay
        
        # Early stopping based on convergence
        if abs(prev_c5 - current_c5) < convergence_tolerance:
            break
        prev_c5 = current_c5
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
