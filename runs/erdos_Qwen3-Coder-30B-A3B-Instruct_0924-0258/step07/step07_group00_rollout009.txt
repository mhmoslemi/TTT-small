import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient-based optimization
    max_iter = 1000
    step_size = 0.01
    no_improvement_limit = 50
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx]
        
        # Compute gradient for this lag
        # The overlap at lag k is sum h[i] * (1 - h[i+k])
        # Gradient w.r.t. h[i] = 1 - h[i+k] - h[i]
        # For lag k=0 to n_points-1, index is offset by n_points-1
        lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        if lag >= 0:
            for i in range(n_points - lag):
                grad[i] += 1.0 - h[i + lag] - h[i]
            # Add g for the terms that have 1-h[i+k] but no h[i]
            for i in range(n_points - lag):
                grad[i + lag] -= h[i]
        else:
            # lag is negative
            for i in range(n_points + lag):
                grad[i] += 1.0 - h[i - lag] - h[i]
            for i in range(n_points + lag):
                grad[i - lag] -= h[i]
        
        # Apply gradient descent
        h_new = h - step_size * grad * dx
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= no_improvement_limit:
            step_size *= 0.5
            no_improvement_count = 0
            if step_size < 1e-6:
                break
        
        h = h_new.copy()
    
    # Final local refinement with SLSQP if within time
    if time() - start_time < budget_s * 0.95:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-12})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
