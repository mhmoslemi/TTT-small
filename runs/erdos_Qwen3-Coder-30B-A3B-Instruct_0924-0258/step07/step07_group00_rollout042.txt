import numpy as np
from scipy.optimize import fmin_l_bfgs_b
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points/2) / np.sum(h)
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Gradient-based optimization
    def objective_and_gradient(h_vec):
        h = h_vec.reshape(-1)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + l_max) % len(h)
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        return max_overlap, gradient.flatten()
    
    # Optimization loop
    max_iter = 200
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # L-BFGS-B optimization
        try:
            h_flat, c5_val, _ = fmin_l_bfgs_b(
                objective_and_gradient, 
                h.flatten(),
                bounds=[(0, 1) for _ in range(len(h))],
                maxiter=50,
                iprint=-1
            )
            h = h_flat.reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
        except:
            pass
    
    # Final verification with gradient-based refinement
    try:
        h_flat, c5_val, _ = fmin_l_bfgs_b(
            objective_and_gradient,
            best_h.flatten(),
            bounds=[(0, 1) for _ in range(len(best_h))],
            maxiter=100,
            iprint=-1
        )
        h_final = h_flat.reshape(-1)
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    except:
        pass
            
    return (list(best_h), best_c5, n_points)
