import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    pop_size = 50
    num_generations = 1000
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Track best solution found
    best_c5_history = []
    
    for gen in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            
        # Track best in this generation
        gen_best = min(fitness)
        if gen_best < best_c5:
            best_c5 = gen_best
            best_h = population[fitness.index(gen_best)].copy()
        
        best_c5_history.append(best_c5)
        
        # Selection: tournament
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size)
            winner = min(tournament, key=lambda i: fitness[i])
            selected.append(population[winner].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[min(i+1, pop_size-1)]
            
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            mutate1 = np.random.rand(n_points) < mutation_rate
            mutate2 = np.random.rand(n_points) < mutation_rate
            if np.any(mutate1):
                child1[mutate1] += np.random.normal(0, 0.03, np.sum(mutate1))
            if np.any(mutate2):
                child2[mutate2] += np.random.normal(0, 0.03, np.sum(mutate2))
                
            # Repair constraints
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        # Keep only valid population size
        population = new_population[:pop_size]
        
        # Occasionally improve with convex relaxation
        if gen % 50 == 0 and gen > 0:
            try:
                # Sort by fitness and take best
                sorted_indices = np.argsort(fitness)[:pop_size//10]
                for idx in sorted_indices:
                    h_candidate = population[idx]
                    # Simple convex relaxation: keep the constraints
                    h_opt = cp.Variable(n_points)
                    constraints = [
                        cp.sum(h_opt) == n_points / 2.0,
                        h_opt >= 0,
                        h_opt <= 1
                    ]
                    
                    # Convex relaxation constraint for some k values to speed up
                    for k in range(1, min(10, n_points)):
                        if k < n_points:
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                    
                    prob = cp.Problem(cp.Minimize(0), constraints)
                    prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=300)
                    if prob.status == cp.OPTIMAL:
                        h_feasible = h_opt.value
                        c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                        if c5_feasible < best_c5:
                            best_h = h_feasible
                            best_c5 = c5_feasible
            except Exception:
                pass
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 100, 'ftol': 1e-8})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
                
    return (list(best_h), best_c5, n_points)
