import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm parameters
    pop_size = 100
    max_generations = 250
    elite_size = 10
    
    # Initialize population
    population = []
    fitness = []
    
    # Generate initial population
    for i in range(pop_size):
        if i == 0 and 'initial_h_values' in globals():
            # Use the initial construction as the first individual
            h = h_start.copy()
        else:
            # Generate random individuals
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        fit = evaluate_erdos_solution(h, dx)
        fitness.append(fit)
    
    # Track hall of fame
    hall_of_fame = []
    for i in range(min(elite_size, len(population))):
        idx = np.argmin(fitness)
        hall_of_fame.append((population[idx].copy(), fitness[idx]))
        population.pop(idx)
        fitness.pop(idx)
    
    # Evolutionary algorithm loop
    no_improvement_count = 0
    
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        # Keep top individuals
        selected_pop = sorted_pop[:pop_size//4]
        selected_fit = sorted_fit[:pop_size//4]
        
        # Add elite
        for h, fit in hall_of_fame:
            if len(selected_pop) < pop_size:
                selected_pop.append(h.copy())
                selected_fit.append(fit)
        
        # Fill rest with new individuals via crossover and mutation
        while len(selected_pop) < pop_size:
            # Select parents
            parent1_idx = np.random.randint(0, len(selected_pop))
            parent2_idx = np.random.randint(0, len(selected_pop))
            p1 = selected_pop[parent1_idx]
            p2 = selected_pop[parent2_idx]
            
            # Crossover
            alpha = np.random.uniform(0.2, 0.8)
            child = alpha * p1 + (1 - alpha) * p2
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            # Mutation: Gaussian noise and bit-flip
            if np.random.rand() < 0.7:  # 70% chance of mutation
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                
            if np.random.rand() < 0.3:  # 30% chance of bit-flip
                flip_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                for idx in flip_indices:
                    child[idx] = 1 - child[idx]
                child = child * (n_points/2) / np.sum(child)
            
            selected_pop.append(child)
            selected_fit.append(evaluate_erdos_solution(child, dx))
        
        # Update hall of fame
        for i in range(min(elite_size, len(selected_pop))):
            idx = np.argmin(selected_fit)
            new_h = selected_pop[idx].copy()
            new_fit = selected_fit[idx]
            if len(hall_of_fame) == 0 or new_fit < hall_of_fame[-1][1]:
                inserted = False
                for j in range(len(hall_of_fame)):
                    if new_fit < hall_of_fame[j][1]:
                        hall_of_fame.insert(j, (new_h, new_fit))
                        inserted = True
                        break
                if not inserted and len(hall_of_fame) < elite_size:
                    hall_of_fame.append((new_h, new_fit))
                if len(hall_of_fame) > elite_size:
                    hall_of_fame.pop()
            else:
                if len(hall_of_fame) < elite_size:
                    hall_of_fame.append((new_h, new_fit))
        
        # Update best solution
        min_fit = min(selected_fit)
        if min_fit < best_c5:
            best_c5 = min_fit
            best_h = selected_pop[np.argmin(selected_fit)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Terminate if no improvement in many generations
        if no_improvement_count > 80:
            break
            
        population = selected_pop
        fitness = selected_fit
    
    # Final refinement using L-BFGS-B on best individuals
    try:
        # Try our best solution first
        h_test = best_h.copy()
        c5_refined = evaluate_erdos_solution(h_test, dx)
        
        # Local optimization for top individual
        def obj(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            return evaluate_erdos_solution(h, dx)
            
        # Use cvxpy for feasibility
        def refine_with_cvxpy(h_start):
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    return h_feasible, c5_feasible
            except:
                pass
            return h_start, evaluate_erdos_solution(h_start, dx)
        
        # Try refined version
        h_feasible, c5_feasible = refine_with_cvxpy(best_h)
        if c5_feasible < best_c5:
            best_h = h_feasible
            best_c5 = c5_feasible
    
    except Exception as e:
        pass
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
