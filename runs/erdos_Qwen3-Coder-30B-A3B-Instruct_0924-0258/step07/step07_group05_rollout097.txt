import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Basis function approach with 10 triangular basis functions
    m = 10  # number of basis functions
    
    # Create piecewise linear basis functions
    def create_basis_functions(n, m):
        # Place basis functions at locations
        bases = []
        x = np.linspace(0, 2, n)
        centers = np.linspace(0, 2, m+2)[1:-1]
        
        for i in range(m):
            # Triangular basis functions
            basis = np.maximum(0, 1 - np.abs((x - centers[i]) / (2.0/m)))
            bases.append(basis)
        return np.array(bases).T
    
    # Create initial basis
    basis = create_basis_functions(n_points, m)
    
    # Initialize coefficients
    coeffs = np.random.rand(m) * 0.5
    h_current = basis @ coeffs
    h_current = np.clip(h_current, 0, 1)
    # Normalize
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Optimization loop
    max_iter = 10000
    patience = 0
    patience_limit = 50
    trust_radius = 0.1
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update basis for current solution
        h_current = basis @ coeffs
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1
            
        # Check for recovery
        if patience > patience_limit:
            coeffs = coeffs + np.random.normal(0, 0.05, m)
            patience = 0
            trust_radius = min(0.1, trust_radius * 2)
        
        # Compute loss gradient direction using finite differences
        eps = 1e-5
        grad = np.zeros(m)
        for i in range(m):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = basis @ coeffs_plus
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            loss_plus = evaluate_erdos_solution(h_plus, dx)
            
            coeffs_minus = coeffs.copy()
            coeffs_minus[i] -= eps
            h_minus = basis @ coeffs_minus
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            loss_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (loss_plus - loss_minus) / (2 * eps)
        
        # Trust region step
        try:
            # Simple gradient descent step with trust region
            step = - grad * 0.01
            
            # Constrain step size
            step_norm = np.linalg.norm(step)
            if step_norm > trust_radius:
                step = step * trust_radius / step_norm
                
            new_coeffs = coeffs + step
            
            # Projection to keep within bounds (0,1) for initial coefficients
            new_coeffs = np.clip(new_coeffs, 0, 1)
            
            coeffs = new_coeffs
            
            # Ensure constraint ∫h dx = 1
            h_test = basis @ coeffs
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points/2) / np.sum(h_test)
            c5_test = evaluate_erdos_solution(h_test, dx)
            
            # Accept if better
            if c5_test < current_c5:
                coeffs = new_coeffs
                # Reduce trust region
                trust_radius = max(1e-6, trust_radius * 0.9)
            else:
                trust_radius = min(0.1, trust_radius * 1.1)
        except Exception as e:
            # If trust region step fails, just perturb slightly
            coeffs = coeffs + np.random.normal(0, 0.02, m)
            coeffs = np.clip(coeffs, 0, 1)
            trust_radius = max(1e-6, trust_radius * 0.9)
    
    # Final refinement with local optimization
    try:
        def objective(c):
            h = basis @ c
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(c):
            h = basis @ c
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(m)]
        
        # Initial guess for coefficients
        res = minimize(objective, coeffs, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8})
        
        if res.success:
            coeffs_final = res.x
            h_final = basis @ coeffs_final
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = objective(coeffs_final)
            
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception as e:
        # In case of optimization failure, return current best 
        pass
    
    # Final check that solution meets constraints
    h_final = basis @ coeffs
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    if c5_final < best_c5:
        best_h = h_final
        best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
