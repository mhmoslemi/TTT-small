import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Benders decomposition parameters
    max_iter = 5000
    convergence_tol = 1e-6
    perturbation_mag = 0.05
    perturbation_decay = 0.95
    
    h_current = h_start.copy()
    c5_current = best_c5
    
    # Initialize constraints tracking
    constraints_added = []
    
    # Main Benders loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Shifts range from 0 to n_points-1
        # For each shift k, the overlap is at index k (0-indexed corresponds to shift k)
        overlaps = corr[:n_points] * dx  # only compute first n_points for k values 0 to n_points-1
        
        # Find the maximum overlap and the corresponding shift
        max_overlap = np.max(overlaps)
        worst_k_idx = np.argmax(overlaps)
        
        # Add constraint for worst k
        constraints_added.append((worst_k_idx, max_overlap))
        
        # Convex relaxation
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add Benders constraints
            for k_idx, max_val in constraints_added:
                if k_idx < n_points:
                    # Compute overlap using the constraint form
                    overlap_expr = cp.sum(cp.multiply(h_opt[:n_points-k_idx], 1 - h_opt[k_idx:]))
                    constraints.append(overlap_expr <= max_val + 1e-8)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_opt.value
                # Evaluate new solution
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                h_current = h_new.copy()
                c5_current = c5_new
                
                # Perturb h_current slightly
                h_current += perturbation_mag * np.random.randn(n_points)
                h_current = np.clip(h_current, 0, 1)
                # Rescale to satisfy the constraint
                h_current = h_current * (n_points/2) / np.sum(h_current)
                
                # Decrease perturbation magnitude
                perturbation_mag *= perturbation_decay
                
        except Exception:
            # Fallback to direct updating
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
            c5_current = evaluate_erdos_solution(h_current, dx)
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
