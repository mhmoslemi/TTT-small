import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        n_points = len(h_start)

        # Construct step function with two intervals
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place two intervals of length 0.5 each
        # Let positions be [0.65, 1.15] and [0.85, 1.35]  (length 0.5 each)
        # This corresponds to approximately 17.5 points each
        i_start1 = int(0.65 / dx)
        i_end1 = int(1.15 / dx)
        i_start2 = int(0.85 / dx)
        i_end2 = int(1.35 / dx)
        # Safe way to set intervals manually for 35 points
        h_start[i_start1:i_end1] = 1
        h_start[i_start2:i_end2] = 1
        
        # Normalize to ensure integral = 1
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try gradient-based refinement 
    from scipy.optimize import minimize
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Vary the interval locations to improve solution
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Perturb interval positions slightly to improve result
    for perturb in np.linspace(-0.03, 0.03, 10):
        h_trial = np.zeros(n_points)
        # Try placing intervals around (0.65+perturb) and (0.85+perturb)
        i_start1 = max(0, int((0.65 + perturb) / dx))
        i_end1 = min(n_points, int((1.15 + perturb) / dx))
        i_start2 = max(0, int((0.85 + perturb) / dx))

        h_trial[i_start2:i_end2] = 1
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        c5_trial = evaluate_erdos_solution(h_trial, dx)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
    
    # Final optimization using SLSQP
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
