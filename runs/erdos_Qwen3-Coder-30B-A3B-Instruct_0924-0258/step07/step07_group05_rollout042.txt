import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Benders decomposition parameters
    max_iter = 5000
    convergence_tol = 1e-6
    perturbation_mag = 0.05
    perturbation_decay = 0.95
    
    # Initialize
    h_current = best_h.copy()
    c5_current = best_c5
    previous_c5 = c5_current + 1.0
    
    # Store constraints
    constraints = []
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Identify worst-case overlap
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_indices = np.arange(len(corr))
        max_k = k_indices[np.argmax(corr)]
        max_overlap = corr[max_k] * dx
        
        # Add constraint
        if max_overlap > c5_current * 1.001:  # Only add if significant
            # Add a constraint that limits overlap at current k
            # This will be approximated in the convex relaxation
            constraints.append((max_k, max_overlap))
        
        # Convex relaxation for master problem
        try:
            h_opt = cp.Variable(n_points)
            constraints_cvx = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add Benders cuts for constraints that are active
            for k, bound in constraints:
                if k < n_points:  # Valid shift
                    # Linear approximation around current h
                    c = np.convolve(h_current, 1 - h_current, mode='full')
                    c = c * dx
                    
                    # For specific k, add constraint
                    k_idx = int(k * n_points / 2)
                    if k_idx < len(c):
                        # Approximate the shift by adding constraint
                        constraints_cvx.append(cp.sum(cp.multiply(h_opt[:n_points-k_idx], 
                                                                  1 - h_opt[k_idx:])) <= bound)
            
            prob = cp.Problem(cp.Minimize(0), constraints_cvx)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            if prob.status == cp.OPTIMAL:
                h_new = h_opt.value
                # The relaxation may not respect the minimum overlapping exactly
                # but constraint approach will help to make h better
                h_current = h_new
            else:
                # Fallback to previous h
                pass
        except Exception:
            pass
        
        # Perturb current solution to escape local minimum
        h_perturbed = h_current + perturbation_mag * np.random.randn(n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Rescale to maintain the constraint
        sum_h = np.sum(h_perturbed)
        if abs(sum_h - n_points/2) > 1e-6:
            h_perturbed = h_perturbed * (n_points/2) / sum_h
        
        # Evaluate perturbed
        c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
        if c5_perturbed < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_perturbed
            
        # Update h_current for next iteration
        h_current = h_perturbed.copy()
        c5_current = best_c5
        perturbation_mag *= perturbation_decay
        
        # Check convergence
        if abs(c5_current - previous_c5) < convergence_tol:
            break
        previous_c5 = c5_current
    
    # Final polishing with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
