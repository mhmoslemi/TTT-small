import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm parameters
    pop_size = 50
    max_generations = 200
    elite_size = 5
    mutation_rate = 0.05
    crossover_rate = 0.7
    
    # Initialize population with mutations of best solution
    population = []
    for _ in range(pop_size):
        h = best_h + np.random.normal(0, mutation_rate, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Track hall of fame
    hall_of_fame = []
    for h in population:
        c5 = evaluate_erdos_solution(h, dx)
        hall_of_fame.append((h.copy(), c5))
    hall_of_fame.sort(key=lambda x: x[1])
    
    # EA loop
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update hall of fame
        for i in range(min(elite_size, len(sorted_population))):
            h = sorted_population[i]
            c5 = sorted_fitness[i]
            if len(hall_of_fame) < elite_size:
                hall_of_fame.append((h.copy(), c5))
            elif c5 < hall_of_fame[-1][1]:
                hall_of_fame[-1] = (h.copy(), c5)
        hall_of_fame.sort(key=lambda x: x[1])
        
        # Selection and reproduction
        new_population = []
        
        # Keep elite
        for h, _ in hall_of_fame:
            new_population.append(h.copy())
            
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = np.random.choice(len(sorted_population[:pop_size//2]))
            parent2_idx = np.random.choice(len(sorted_population[:pop_size//2]))
            
            parent1 = sorted_population[parent1_idx]
            parent2 = sorted_population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Blend crossover
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Gaussian mutation
                child += np.random.normal(0, mutation_rate, n_points)
            
            # Repair to constraint and feasibility
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:pop_size]
        
        # Update best solution
        current_best = sorted_population[0]
        current_c5 = sorted_fitness[0]
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
            
        # Adaptive parameters
        mutation_rate = max(0.005, mutation_rate * 0.98)
        crossover_rate = max(0.3, crossover_rate * 0.99)
    
    # Final refinement using L-BFGS
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Try to improve top few solutions with local optimization
    for i in range(min(10, len(hall_of_fame))):
        h_init = hall_of_fame[i][0]
        # Local optimization
        try:
            res = minimize(objective, h_init, method='L-BFGS-B', 
                          bounds=[(0, 1) for _ in range(n_points)],
                          constraints={'type': 'eq', 'fun': constraint_sum},
                          options={'maxiter': 500})
            if res.success:
                refined_h = res.x
                refined_h = np.clip(refined_h, 0, 1)
                refined_h = refined_h * (n_points/2) / np.sum(refined_h)
                refined_c5 = evaluate_erdos_solution(refined_h, dx)
                if refined_c5 < best_c5:
                    best_h = refined_h.copy()
                    best_c5 = refined_c5
        except:
            pass
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
