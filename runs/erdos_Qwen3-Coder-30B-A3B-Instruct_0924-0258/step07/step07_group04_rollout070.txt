import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe with adaptive learning rate
    max_iter = 1000
    gamma = 0.1
    tolerance = 1e-6
    stagnation_count = 0
    
    h_current = h_start.copy()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Main Frank-Wolfe iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag = np.argmax(corr)
        max_val = corr[max_lag]
        
        # Compute direction (Frank-Wolfe step)
        # We want to move towards minimizing the maximum correlation
        # Approximate the gradient direction
        s = np.zeros(n_points)
        for i in range(n_points):
            # The gradient direction as (1 - h) - direction that reduces max correlation
            # Simplified: approximate with local gradients
            s[i] = 1 - h_current[i]  # This makes the objective decrease
        
        # Frank-Wolfe step
        h_new = h_current + gamma * s
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to correct integral
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new point
        c5_new = evaluate_erdos_solution(h_new, dx)

        if c5_new < best_c5:

            return evaluate_erdos_solution(h, dx)
