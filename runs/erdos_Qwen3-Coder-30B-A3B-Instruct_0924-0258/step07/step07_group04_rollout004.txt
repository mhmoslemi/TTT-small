import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_h(n_points):
    x = np.linspace(0, 2, n_points)
    h_initial = 0.5 * (1 + np.sin(2 * np.pi * x))
    h_initial = np.clip(h_initial, 0, 1)
    normalization = (n_points / 2) / np.sum(h_initial)
    h_initial *= normalization
    return h_initial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 50
        h_start = initialize_h(n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    T = 1.0
    alpha = 0.998
    tabu_size = 20
    tabu_list = []
    max_iter = 10000
    stagnation_limit = 500
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate solution
        candidate_h = current_h.copy()
        
        # Perturb 15% of points with random changes
        indices_to_change = np.random.choice(n_points, size=int(0.15 * n_points), replace=False)
        for i in indices_to_change:
            # Perturb by up to ±15%
            perturbation = np.random.uniform(-0.15, 0.15)
            candidate_h[i] += perturbation
            # Clip to [0, 1]
            candidate_h[i] = np.clip(candidate_h[i], 0, 1)
        
        # Renormalize to meet constraint
        candidate_h = candidate_h * (n_points/2) / np.sum(candidate_h)
        
        # Check if solution is in tabu list (simple hashing for avoiding repetition)
        candidate_hash = tuple(np.round(candidate_h, decimals=6))
        if candidate_hash in tabu_list:
            # Accept with low probability to escape tabu
            if np.random.random() > 0.9:
                pass
            else:
                continue
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Accept or reject
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            
            # Update tabu list
            tabu_list.append(candidate_hash)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Cooling schedule
        T = max(1e-6, T * alpha)
        
        if stagnation_count > stagnation_limit:
            # Reset if stagnating, with more randomness
            T = 1.0
            stagnation_count = 0
            # Try a slightly different initialization
            current_h = initialize_h(n_points) * 0.9 + np.random.random(n_points) * 0.2
            current_h = current_h * (n_points/2) / np.sum(current_h)
            current_c5 = evaluate_erdos_solution(current_h, dx)
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
