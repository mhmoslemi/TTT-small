import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Gradient-based optimization with focus on maximum overlap lag
    max_iter = 10000
    step_size = 0.01
    tolerance = 1e-9
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_index = np.argmax(corr)
        max_lag = max_lag_index - (n_points - 1)
        
        # Compute gradient for this specific lag
        # For h[i] * (1 - h[i+lag]), gradient is:
        # d/dh[i] = (1 - h[i+lag]) if i+lag < n_points else 0
        # d/dh[i+lag] = -h[i] if i+lag < n_points else 0
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + max_lag >= 0 and i + max_lag < n_points:
                gradient[i] += 1 - h_current[i + max_lag]
            if i - max_lag >= 0 and i - max_lag < n_points:
                gradient[i - max_lag] -= h_current[i]
        
        # Normalize gradient for safe step
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm
        
        # Try step
        h_new = h_current - step_size * gradient
        
        # Ensure feasibility (bounds and sum constraint)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if improvement or within tolerance
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Update current solution with some damping
        h_current = 0.9 * h_current + 0.1 * h_new
        
        if np.abs(c5_new - best_c5) < tolerance:
            step_size *= 0.5
            if step_size < 1e-8:
                break
    
    return (list(best_h), best_c5, n_points)
