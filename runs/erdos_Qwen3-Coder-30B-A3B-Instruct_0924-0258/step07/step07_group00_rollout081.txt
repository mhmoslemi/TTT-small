import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent approach
    max_iter = 1000
    step_size = 0.01
    patience = 50
    no_improvement = 0
    
    h = h_start.copy()
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(corr)
        max_overlap = corr[max_lag] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            break
            
        # Compute gradient for the maximum overlap lag
        # The maximum overlap occurs at lag k = max_lag - (n_points - 1)
        k = max_lag - (n_points - 1)
        
        # Gradient of overlap integral w.r.t. h[i]
        # ∂/∂h[i] ∫ h(x)(1 - h(x+k)) dx = 1 - h[i+k] - h[i]
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1 - h[j] - h[i]
        
        # Update step
        h_new = h - step_size * grad
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Local refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
