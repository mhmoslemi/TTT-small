import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with sinusoidal construction
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + 0.5 * np.sin(2 * np.pi * x / (2.0 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm parameters
    pop_size = 50
    generations = 80
    mutation_rate = 0.05
    crossover_rate = 0.7
    
    # Initialize population
    population = []
    fitness_scores = []
    
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Add small random perturbation to initial solution
            noise = np.random.normal(0, 0.05, n_points)
            h = h_start + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness = evaluate_erdos_solution(h, dx)
        fitness_scores.append(fitness)
    
    # Hall of fame
    hall_of_fame = [(h.copy(), f) for h, f in zip(population, fitness_scores)]
    hall_of_fame.sort(key=lambda x: x[1])
    hall_of_fame = hall_of_fame[:10]  # Keep top 10
    
    # EA Loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness_scores[i] for i in sorted_indices]
        
        # Keep top 20% as elites
        elites = sorted_pop[:pop_size // 5]
        
        # Generate new population through crossover and mutation
        new_population = elites.copy()
        
        while len(new_population) < pop_size:
            # Tournament selection for parents
            parent1 = sorted_pop[random.randint(0, pop_size//2)]
            parent2 = sorted_pop[random.randint(0, pop_size//2)]
            
            if random.random() < crossover_rate:
                # Blend crossover
                alpha = random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                # Clone parent
                child = parent1.copy()
            
            # Mutation
            if random.random() < mutation_rate:
                # Gaussian mutation
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                
            new_population.append(child)
            
        # Evaluate new population
        fitness_scores = []
        for h in new_population:
            fitness = evaluate_erdos_solution(h, dx)
            fitness_scores.append(fitness)
            
            # Update hall of fame
            if fitness < hall_of_fame[-1][1]:
                hall_of_fame.append((h.copy(), fitness))
                hall_of_fame.sort(key=lambda x: x[1])
                hall_of_fame = hall_of_fame[:-1]
        
        # Update best solution
        if min(fitness_scores) < best_c5:
            best_idx = np.argmin(fitness_scores)
            best_h = new_population[best_idx].copy()
            best_c5 = fitness_scores[best_idx]
        
        population = new_population[:pop_size]
        fitness_scores = fitness_scores[:pop_size]
        
        # Adaptive mutation rate
        if gen % 10 == 0 and gen > 0:
            mutation_rate = max(0.005, mutation_rate * 0.8)
    
    # Final refinement using L-BFGS-B on top solutions
    top_solutions = [h for h, f in hall_of_fame]
    best_refined = best_h
    best_refined_c5 = best_c5
    
    for h in top_solutions:
        try:
            # Create constraints for optimization
            def objective(h_vals):
                return evaluate_erdos_solution(h_vals, dx)
            
            def constraint_sum(h_vals):
                return np.sum(h_vals) - n_points / 2
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]

            # L-BFGS-B optimization
            result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100})
            if result.success:
                refined_h = result.x
                refined_c5 = evaluate_erdos_solution(refined_h, dx)
                if refined_c5 < best_refined_c5:
                    best_refined = refined_h
                    best_refined_c5 = refined_c5
        except:
            pass
    
    # Final fix-up
    best_refined = np.clip(best_refined, 0, 1)
    best_refined = best_refined * (n_points/2) / np.sum(best_refined)
    best_refined_c5 = evaluate_erdos_solution(best_refined, dx)
    
    return (list(best_refined), best_refined_c5, n_points)
