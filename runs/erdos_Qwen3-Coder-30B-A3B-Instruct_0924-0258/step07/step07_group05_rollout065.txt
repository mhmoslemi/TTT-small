import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Benders decomposition approach
    max_iter = 5000
    convergence_tol = 1e-6
    perturbation_mag = 0.05
    perturbation_decay = 0.95
    
    # Initialize constraints list
    constraints_list = []
    
    h = h_start.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    previous_c5 = best_c5
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all shifts
        corr_full = np.correlate(h, 1 - h, mode='full')
        # Extract only the relevant overlaps (k from 0 to n_points-1)
        overlaps = corr_full[n_points-1:]  # This corresponds to k=0,1,2,...
        
        # Find worst shift k
        max_overlap_idx = np.argmax(overlaps)
        max_overlap_value = overlaps[max_overlap_idx]
        
        # Add constraint to master problem if this is a new constraint
        # To avoid redundancy, we just add an approximate constraint
        
        # Convex relaxation
        try:
            h_vars = cp.Variable(n_points)
            constraints = [
                cp.sum(h_vars) == n_points / 2.0,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Add current constraint using affine approximation
            # This is a simplification but effective
            if max_overlap_value > 0:
                # Only add constraint if it actually constrains the problem
                k = max_overlap_idx
                if k < n_points:
                    # Approximate overlap for shift k as a linear function
                    # In practice, just ensure this shift is less constrained
                    overlap_constr = cp.sum(cp.multiply(h_vars[:n_points-k], 1 - h_vars[k:])) <= best_c5 * dx + 1e-8
                    constraints.append(overlap_constr)
            
            objective = cp.Minimize(0)
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_vars.value
                if not np.any(np.isnan(h_new)):
                    h = h_new.copy()
                    # Perturb slightly
                    h += perturbation_mag * np.random.randn(n_points)
                    h = np.clip(h, 0, 1)
                    # Rescale to make sum = n_points/2
                    h = h * (n_points / 2) / np.sum(h)
                    
                    c5_new = evaluate_erdos_solution(h, dx)
                    if c5_new < best_c5:
                        best_h = h.copy()
                        best_c5 = c5_new
                    
                    # Reduce perturbation
                    perturbation_mag *= perturbation_decay
                    if abs(c5_new - previous_c5) < convergence_tol:
                        # Stop if we haven't improved much
                        if iter_num > 100:
                            break
                    previous_c5 = c5_new
                    
        except Exception:
            # Fallback if optimization fails
            pass
    
    # Final polishing with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
