import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try to improve with convex relaxation
    if n_points <= 100:  # Only attempt for smaller problems
        try:
            # Attempt convex relaxation with CVXPY
            h_var = cp.Variable(n_points)
            
            # Simple bound constraint
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points / 2.0
            ]
            
            # Relaxation to get heuristic solution
            objective = cp.Minimize(cp.sum(h_var))
            prob = cp.Problem(objective, constraints)
            
            # Solve using ECOS solver
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
            
            if prob.status == cp.OPTIMAL:
                h_relaxed = h_var.value
                # Ensure the constraints are satisfied
                h_relaxed = np.clip(h_relaxed, 0, 1)
                h_relaxed = h_relaxed * (n_points/2) / np.sum(h_relaxed)
                val = evaluate_erdos_solution(h_relaxed, dx)
                if val < best_c5:
                    best_h = h_relaxed.copy()
                    best_c5 = val
                    
        except Exception:
            pass
    
    # Local optimization with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Try another local search if time allows
    if time() - start_time < budget_s * 0.8:
        try:
            # Try random restarts
            for _ in range(5):
                if time() - start_time > budget_s * 0.95:
                    break
                h_restart = np.random.uniform(0, 1, n_points)
                h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 500, 'ftol': 1e-10})
                if res.success:
                    h_restart = res.x
                    c5_restart = objective(h_restart)
                    if c5_restart < best_c5:
                        best_h = h_restart
                        best_c5 = c5_restart
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
