import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T0 = 0.1
    cooling_rate = 0.001
    max_iter = 10000
    perturbation_fraction = 0.15
    
    # Initialize
    h_current = h_start.copy()
    c5_current = best_c5
    T = T0
    stagnation_count = 0
    max_stagnation = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb h
        h_new = h_current.copy()
        n_perturb = int(n_points * perturbation_fraction)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-0.1, 0.1)
        
        # Clip
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        delta = c5_new - c5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_current = h_new
            c5_current = c5_new
            stagnation_count = 0
            
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
        else:
            stagnation_count += 1
            
        T = T0 * np.exp(-cooling_rate * iteration)
        
        if stagnation_count >= max_stagnation:
            break
    
    # Gradient Descent refinement
    h_refined = best_h.copy()
    c5_refined = best_c5
    learning_rate = 0.1
    gd_iter = 500
    patience = 50
    best_gd_c5 = c5_refined
    gd_patience = 0
    
    for i in range(gd_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Find k_max
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient for k_max
        grad = np.zeros(n_points)
        n_corr = len(corr)
        for j in range(n_points):
            if j + k_max < n_points:
                grad[j] += (1 - h_refined[j + k_max]) * dx
            if j - k_max >= 0:
                grad[j] += (1 - h_refined[j - k_max]) * dx
        
        # Update
        h_refined = h_refined - learning_rate * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_refined, dx)
        
        if new_c5 < best_gd_c5:
            best_gd_c5 = new_c5
            gd_patience = 0
            if new_c5 < best_c5:
                best_h = h_refined.copy()
                best_c5 = new_c5
        else:
            gd_patience += 1
            if gd_patience >= patience:
                learning_rate *= 0.5
                gd_patience = 0
                
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
