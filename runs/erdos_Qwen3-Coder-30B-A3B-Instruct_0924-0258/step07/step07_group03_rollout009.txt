import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based descent with adaptive learning rate
    h = h_start.copy()
    learning_rate = 0.1
    max_iter = 1000
    patience = 50
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute current overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        # Compute gradient with respect to current worst overlap
        # Simple approach: gradient is 1 - 2*h where contribution is high
        gradient = np.zeros_like(h)
        lag_offset = max_lag_idx - (n_points - 1)
        for i in range(n_points):
            j = i - lag_offset
            if 0 <= j < n_points:
                gradient[i] += (1 - 2 * h[i]) * (1 - h[j])
        
        # Apply gradient update
        h_new = h - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to meet constraint ∫ h dx = 1 (sum(h) * dx = 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check improvement
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improvement_count = 0
            learning_rate = 0.1  # reset learning rate on improvement
        else:
            no_improvement_count += 1
            if no_improvement_count >= patience:
                learning_rate *= 0.5  # reduce learning rate
                no_improvement_count = 0
                
        h = h_new.copy()
        
        # Occasionally perturb the solution
        if iteration % 50 == 0 and iteration > 0:
            perturb_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            h[perturb_indices] += np.random.uniform(-0.1, 0.1, len(perturb_indices))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Phase 2: Local optimization with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use best solution from phase 1 as starting point
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
