import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA Parameters
    pop_size = 100
    generations = 250
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = 10
    tournament_size = 5
    
    # Initialize population
    population = []
    fitness = []
    
    # Elite solution from initial
    population.append(h_start.copy())
    fitness.append(best_c5)
    
    # Generate rest of population
    for i in range(1, pop_size):
        if i < pop_size // 2:
            # Random initialization
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        else:
            # Latin hypercube sampling approach
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            # For better spreading, apply a small random shift
            h = np.roll(h, i % n_points)
            h = h * (n_points/2) / np.sum(h)
        
        population.append(h)
        f = evaluate_erdos_solution(h, dx)
        fitness.append(f)
        
        if f < best_c5:
            best_h = h.copy()
            best_c5 = f

    # Evolutionary Algorithm
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        elite = population[:elite_size]
        elite_fitness = fitness[:elite_size]
        
        # Update global best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection with tournament
        new_pop = elite.copy()
        while len(new_pop) < pop_size:
            # Tournament selection
        
            # For tournament, select indexes randomly, then pick best
            selected_indices = np.random.choice(range(pop_size), size=tournament_size, replace=False)
            selected_fitness = [fitness[i] for i in selected_indices]
            winner_idx = selected_indices[np.argmin(selected_fitness)]
            
            # Get the winner
            parent1 = population[winner_idx]
            
            # Another parent
            selected_indices = np.random.choice(range(pop_size), size=tournament_size, replace=False)
            selected_fitness = [fitness[i] for i in selected_indices]
            winner_idx = selected_indices[np.argmin(selected_fitness)]
            parent2 = population[winner_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                # Inherit parent 1
                child = parent1.copy()
                
            # Mutation
            if np.random.rand() < mutation_rate:
                # Gaussian mutation
                variance = max(0.001, 0.1 * (1 - gen / generations))  # Adaptive variance
                noise = np.random.normal(0, variance, n_points)
                child = child + noise
                child = np.clip(child, 0, 1)
            
            # Normalize to constraint
            child = child * (n_points/2) / np.sum(child)
            
            # Repair feasibility
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            if len(new_pop) < pop_size:
                new_pop.append(child)
                new_fitness = evaluate_erdos_solution(child, dx)
                if new_fitness < best_c5:
                    best_h = child.copy()
                    best_c5 = new_fitness
        
        population = new_pop
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Local optimization refinement
    try:
        # Apply local refinement on best solutions
        top_candidates = min(5, len(population))
        for i in range(top_candidates):
            h0 = population[i]
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            # Use L-BFGS-B for local refinement
            bounds = [(0, 1) for _ in range(n_points)]
            result = minimize(obj, h0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 50})
            
            if result.success:
                h_refined = result.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
    
    except Exception as e:
        pass

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
        
    return (list(best_h), best_c5, n_points)
