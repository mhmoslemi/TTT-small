import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a smooth starting point
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try a few different optimization strategies
    strategies = []
    
    # Strategy 1: Simple gradient descent with projection
    def strategy_1():
        h = best_h.copy()
        lr = 0.01
        for i in range(5000):
            if time() - start_time > budget_s * 0.95:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            max_k_idx = np.argmax(corr)
            # Gradient step in the direction that reduces the maximum correlation
            grad = np.zeros_like(h)
            # Compute gradient for the worst-case correlation
            k = max_k_idx - (n_points - 1)
            if k >= 0:
                grad[:n_points-k] += (1 - h[k:]) - h[:n_points-k]
            else:
                grad[-k:] += (1 - h[:n_points+k]) - h[-k:]
            h -= lr * grad
            # Project to feasible set
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_h[:] = h
                best_c5 = c5
        return best_h, best_c5

    # Strategy 2: SQP with artificial constraint relaxation
    def strategy_2():
        h = best_h.copy()
        # Use cvxpy for quadratic relaxation
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add a few key constraints based on the best h so far
            corr = np.correlate(h, 1 - h, mode='full')
            max_k_idx = np.argmax(corr)
            max_lag = max_k_idx - (n_points - 1)  # actual lag
            
            # Add key lag constraints
            top_lags = np.argsort(corr)[-10:]
            for k in top_lags:
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass
        return h, best_c5
    
    # Strategy 3: Particle swarm optimization hybrid
    def strategy_3():
        pop_size = 30
        max_iter = 100
        inertia = 0.7
        cognitive = 1.4
        social = 1.4
        velocity_max = 0.15
        
        particles = []
        velocities = []
        pbest = []
        pbest_c5 = []
        
        for i in range(pop_size):
            noise = np.random.uniform(-0.1, 0.1, n_points)
            h = best_h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            particles.append(h)
            velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
            c5 = evaluate_erdos_solution(h, dx)
            pbest.append(h.copy())
            pbest_c5.append(c5)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        
        for iter_num in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            for i in range(pop_size):
                # Update velocity
                r1, r2 = np.random.rand(2, n_points)
                cognitive_force = cognitive * r1 * (pbest[i] - particles[i])
                social_force = social * r2 * (best_h - particles[i])
                velocities[i] = inertia * velocities[i] + cognitive_force + social_force
                velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
                
                c5 = evaluate_erdos_solution(particles[i], dx)
                
                if c5 < pbest_c5[i]:
                    pbest[i] = particles[i].copy()
                    pbest_c5[i] = c5
                    
                    if c5 < best_c5:
                        best_h = particles[i].copy()
                        best_c5 = c5
                        
        return best_h, best_c5
    
    # Run the strategies
    strategies.append(strategy_1)
    strategies.append(strategy_2)
    strategies.append(strategy_3)
    
    for i, strategy in enumerate(strategies):
        if time() - start_time > budget_s * 0.95:
            break
        try:
            h, c5 = strategy()
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
        except Exception:
            continue
    
    # Final refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final verification step
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
