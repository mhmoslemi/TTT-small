import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize
    h = best_h.copy()
    mu = 100.0
    alpha = 0.1
    max_iter = 5000
    tol = 1e-6
    
    # Trust region parameters
    trust_radius = 0.1
    trust_radius_factor = 1.1
    
    # Gradient-based augmented Lagrangian
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current C5 and gradient
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Compute gradient w.r.t. active constraint
        active_k = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        
        # Gradient for h_i term
        for i in range(n_points):
            if i + active_k >= 0 and i + active_k < n_points:
                grad[i] += (1 - h[i + active_k])
            if i - active_k >= 0 and i - active_k < n_points:
                grad[i] += h[i - active_k]
        
        # Augmented Lagrangian gradient
        lagrangian_grad = grad + mu * (h - 1) * (np.sum(h) - n_points/2)
        
        # Line search for step size
        step_size = alpha
        for _ in range(10):
            h_new = h - step_size * lagrangian_grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < max_overlap:
                h = h_new
                break
            step_size *= 0.5
        
        # Update penalty
        constraint_violation = abs(np.sum(h) - n_points/2)
        if constraint_violation > 1e-4:
            mu *= 2
        else:
            mu *= 0.9
            
        # Check for convergence
        if abs(max_overlap - best_c5) < tol:
            pass
        else:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Update trust region
        if iter_num % 50 == 0 and iter_num > 0:
            trust_radius *= trust_radius_factor
            if trust_radius > 0.5:
                trust_radius = 0.5
                
    # Final polishing with trust-region method
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final verification
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
