import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point
        h_start = np.ones(n_points) * 0.5
        h_start[5:15] = 1.0
        h_start[20:30] = 0.0
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Basin hopping with local opt
    n_iter = 0
    max_iter = 150
    
    while n_iter < max_iter and (time() - start_time) < budget_s * 0.95:
        # Random perturbation
        h_pert = best_h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        # Local search with L-BFGS
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint(h):
            return np.sum(h) - n_points/2
        
        cons = {'type': 'eq', 'fun': constraint}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, h_pert, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass
            
        # Global perturbation with probabilistic model
        step_sizes = [0.05, 0.03, 0.01]
        for step in step_sizes:
            if np.random.rand() < 0.3:
                # Small random perturbation
                h_new = best_h + np.random.uniform(-step, step, n_points)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        
        n_iter += 1
        
        # Occasionally do more thorough local refinement
        if n_iter % 20 == 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add max overlap constraint
                h_test = best_h.copy()
                corr = np.correlate(h_test, 1 - h_test, mode='full')
                max_k_idx = np.argmax(corr)
                max_lag = max_k_idx - (n_points - 1)
                
                # Constraint for highest overlap lag
                sample_lags = np.argsort(corr)[-15:]
                for k in sample_lags:
                    if k < n_points // 2:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass

    # Final verification and refinement
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
