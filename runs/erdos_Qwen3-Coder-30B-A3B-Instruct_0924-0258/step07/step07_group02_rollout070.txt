import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct good starting point based on known structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization parameters
    max_iter = 100
    lr = 0.01
    decay = 0.95
    
    # Gradient-based local search with random perturbations
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Simple gradient descent step
        h_current = best_h.copy()
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (n_points - 1)
        
        # Compute gradient approximate
        h_copy = h_current.copy()
        h_copy += 1e-5
        
        corr_p = np.correlate(h_copy, 1 - h_copy, mode='full')
        corr_n = np.correlate(h_current, 1 - h_current, mode='full')
        
        grad = (corr_p - corr_n) / 1e-5
        
        # approximate gradient with respect to a single index
        index = np.argmax(np.abs(grad)) - (n_points - 1)
        if index < 0:
            index = index + n_points
        if index >= n_points:
            index = index - n_points
            
        # Apply update
        h_new = h_current.copy()
        h_new[index] += lr * grad[index + n_points - 1]
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Random perturbations
        if i % 5 == 0:
            noise_mag = 0.02 * (1 - (i / max_iter))
            h_perturb = h_current + np.random.normal(0, noise_mag, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
            c5_perturb = evaluate_erdos_solution(h_perturb, dx)
            if c5_perturb < best_c5:
                best_h = h_perturb.copy()
                best_c5 = c5_perturb
                
        lr *= decay

    # Final SQP refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
