import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial guess
        h_start = np.ones(n_points) * 0.5
        # Add some structure to avoid symmetric local minima
        x = np.linspace(0, 2, n_points)
        h_start = 0.3 + 0.4 * np.sin(2 * np.pi * x) + 0.1 * np.random.random(n_points)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds for differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint for sum(h) = n/2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    # Lagrangian with barrier on constraints
    def lagrangian_with_barrier(x, penalty=1e3):
        # Evaluate the original objective
        c5_val = evaluate_erdos_solution(x, dx)
        
        # Add barrier penalty for constraint violations
        sum_violation = abs(constraint_sum(x))
        penalty_term = penalty * sum_violation ** 2
        
        # Penalize boundaries
        boundary_penalty = 0
        for i in range(n_points):
            if x[i] <= 1e-6:
                boundary_penalty += 1e6 * (1e-6 - x[i])**2
            elif x[i] >= 1 - 1e-6:
                boundary_penalty += 1e6 * (x[i] - (1 - 1e-6))**2
                
        return c5_val + penalty_term + boundary_penalty
    
    # Differential Evolution parameters
    de_options = {
        'maxiter': 500,
        'popsize': 50,
        'tol': 1e-6,
        ' mutation': (0.5, 1),
        'recombination': 0.9,
        'seed': seed,
        'strategy': 'best1bin'
    }
    
    # Store best solution from DE
    final_h = best_h.copy()
    final_c5 = best_c5
    
    # Optimization loop with timeout
    try:
        # DE phase
        de_result = differential_evolution(
            lagrangian_with_barrier,
            bounds,
            **de_options,
            callback=lambda x, convergence: None,
            maxiter=500
        )
        
        if not de_result.success:
            # Fall back to whatever we have
            h_fitted = final_h
            c5_fitted = final_c5
        else:
            h_fitted = de_result.x.copy()
            # Ensure feasibility
            h_fitted = np.clip(h_fitted, 0, 1)
            h_fitted = h_fitted * (n_points/2) / np.sum(h_fitted)
            c5_fitted = evaluate_erdos_solution(h_fitted, dx)
            if c5_fitted < final_c5:
                final_h = h_fitted
                final_c5 = c5_fitted
    except Exception as e:
        pass
    
    # Local refinement with SLSQP if needed
    if time() - start_time < budget_s * 0.9:
        try:
            def obj_to_minimize(h):
                return evaluate_erdos_solution(h, dx)
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            bounds_slsqp = [(0, 1) for _ in range(n_points)]
            res = minimize(
                obj_to_minimize, 
                final_h, 
                method='SLSQP',
                bounds=bounds_slsqp,
                constraints=cons_sum,
                options={'maxiter': 1000, 'ftol': 1e-12, 'eps': 1e-8}
            )
            
            if res.success:
                h_refined = res.x
                # Ensure normalization
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_new = evaluate_erdos_solution(h_refined, dx)
                
                if c5_new < final_c5:
                    final_h = h_refined
                    final_c5 = c5_new
        except Exception:
            pass

    return (list(final_h), final_c5, n_points)
