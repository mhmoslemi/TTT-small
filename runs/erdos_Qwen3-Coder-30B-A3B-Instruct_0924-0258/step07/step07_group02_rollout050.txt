import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct starting point with a more structured approach
        x = np.linspace(0, 2, n_points)
        # Try a piecewise linear construction
        h_start = np.zeros(n_points)
        # Create a step-like function that is 1 on [0, 0.5) and 0 on [0.5, 2]
        # Then scale to integral = 1
        h_start[x < 0.5] = 1.0
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm with mutation and crossover
    pop_size = 50
    max_iter = 200
    mutation_rate = 0.05
    crossover_rate = 0.7
    elite_size = 5
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i < len(h_start):
            h = h_start.copy()
            # Add small random noise to diversity
            noise = np.random.normal(0, 0.02, n_points)
            h += noise
            h = np.clip(h, 0, 1)
        else:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # EA iterations
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Mutation and crossover
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            tourn_size = 5
            indices = np.random.choice(pop_size, tourn_size, replace=False)
            tourn_fitness = [fitness[i] for i in indices]
            parent1_idx = indices[np.argmin(tourn_fitness)]
            parent2_idx = indices[np.random.choice([i for i in indices if i != parent1_idx])]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    # Gaussian mutation
                    noise = np.random.normal(0, 0.03, n_points)
                    child += noise
                    child = np.clip(child, 0, 1)
                    # Normalize
                    child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
                if len(new_population) >= pop_size:
                    break
        
        population = new_population
        
        # Decrease mutation rate
        mutation_rate = max(0.005, mutation_rate * 0.99)
    
    # Final refinement with L-BFGS-B
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Evaluate top solutions
    top_solutions = population[:10]
    best_final = best_h.copy()
    best_final_c5 = best_c5
    
    for h in top_solutions:
        # Refine with L-BFGS-B
        try:
            # Define bounds
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraints)
            if res.success:
                refined_h = res.x
                refined_c5 = evaluate_erdos_solution(refined_h, dx)
                if refined_c5 < best_final_c5:
                    best_final = refined_h.copy()
                    best_final_c5 = refined_c5
        except:
            pass
    
    # Final verification
    best_final = np.clip(best_final, 0, 1)
    best_final = best_final * (n_points/2) / np.sum(best_final)
    best_final_c5 = evaluate_erdos_solution(best_final, dx)
    
    return (list(best_final), best_final_c5, n_points)
