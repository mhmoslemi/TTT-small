import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Gradient-based optimization with adaptive step size
    max_iterations = 500
    step_size = 0.01
    tolerance = 1e-6
    stagnation_count = 0
    max_stagnation = 20
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for the max overlap lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Update step
        h_new = h - step_size * gradient
        
        # Apply constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Update if improvement
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
            stagnation_count = 0
            step_size = min(0.01, step_size * 1.2)

            stagnation_count += 1

    from scipy.optimize import minimize
    
    def objective_and_gradient(h_vec):
        h = h_vec.reshape(-1)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        return max_overlap, gradient.flatten()
    
    # Prepare for L-BFGS-B optimization
    h_refined = best_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with bounds
    try:
        result = minimize(
            lambda x: objective_and_gradient(x)[0],
            h_refined,
            method='L-BFGS-B',

            refined_c5 = evaluate_erdos_solution(h_refined, dx)
            if refined_c5 < best_c5:
                best_h = h_refined
                best_c5 = refined_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
