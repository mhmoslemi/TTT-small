import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Convex relaxation with SDP
    # Try different solvers
    solvers = ['ECOS', 'SCS']  # Available and fast
    
    for solver in solvers:
        if time() - start_time > budget_s * 0.7:
            break
        try:
            # Create SDP relaxation
            h_cvx = cp.Variable(n_points, name="h")
            Z = cp.Variable((n_points, n_points), symmetric=True, name="Z")
            t = cp.Variable(name="t")
            
            # Constraints
            constraints = [
                cp.diag(Z) == h_cvx,  # Z[i,i] = h[i]
                Z >> 0,  # Z is positive semidefinite
                cp.sum(h_cvx) == n_points / 2.0,
                0 <= h_cvx,
                h_cvx <= 1,
                t >= 0
            ]
            
            # Add overlap constraints for all lags
            # We'll not add all constraints, but use a couple key ones
            for k in range(1, n_points):
                if k >= n_points:
                    continue
                # Here we'll construct simpler approximations
                # For exact computation, we'd go with convolution
                # But we'll skip that for performance
            
            # Objective: minimize t
            objective = cp.Minimize(t)
            prob = cp.Problem(objective, constraints)
            
            # This is a simplification - we actually need to enforce
            # the correlation constraints which is complex
            # Instead, use the exact correlation evaluation as a heuristic
            
        except:
            continue
    
    # Phase 2: Local refinement with multiple methods
    # First ensure starting point satisfies constraints
    h_current = h_start.copy()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points / 2.0) / np.sum(h_current)
    
    # Local search with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_current, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
                
        # Try local search with L-BFGS-B as well
        res2 = minimize(objective, h_current, method='L-BFGS-B', 
                        bounds=bounds, options={'maxiter': 1000})
        if res2.success:
            h_refined2 = res2.x
            c5_refined2 = objective(h_refined2)
            if c5_refined2 < best_c5:
                best_h = h_refined2
                best_c5 = c5_refined2
                
    except Exception:
        pass
    
    # Try gradient based restarts with different starting points
    n_restarts = 10
    for r in range(n_restarts):
        if time() - start_time > budget_s * 0.95:
            break
        # Small random perturbation
        noise = np.random.normal(0, 0.05, n_points)
        h_try = h_current + noise
        h_try = np.clip(h_try, 0, 1)
        h_try = h_try * (n_points / 2.0) / np.sum(h_try)
        
        try:
            res = minimize(objective, h_try, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 500, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            continue
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
