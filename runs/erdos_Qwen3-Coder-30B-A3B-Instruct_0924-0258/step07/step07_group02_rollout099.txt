from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Grid search for initial good points
    grid_size = 20
    grid_points = np.linspace(0.01, 0.99, grid_size)
    best_c5_grid = best_c5
    best_h_grid = best_h.copy()
    
    for i in range(grid_size):
        h_test = np.ones(n_points) * grid_points[i]
        h_test = h_test * (n_points/2) / np.sum(h_test)
        c5 = evaluate_erdos_solution(h_test, dx)
        if c5 < best_c5_grid:
            best_c5_grid = c5
            best_h_grid = h_test.copy()
    
    best_h = best_h_grid.copy()
    best_c5 = best_c5_grid
    
    # Simple evolutionary search with local refinement
    population_size = 40
    max_generations = 100

    fitness = []

        h = best_h + np.random.normal(0, 0.02, n_points)

        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))

    sorted_indices = np.argsort(fitness)
    population = [population[i] for i in sorted_indices]

            mutation_strength = 0.03 * (1 - gen/max_generations)

            h_child = h_child * (n_points/2) / np.sum(h_child)
            
            new_population.append(h_child)

        new_fitness = []

            new_fitness.append(evaluate_erdos_solution(h, dx))
        
        # Combine and select best

        fitness = new_fitness

        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]

        # Use the best solution so far as starting point

    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
