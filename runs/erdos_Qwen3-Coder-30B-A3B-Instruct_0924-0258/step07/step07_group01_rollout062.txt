import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create simple step function with two intervals
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place two intervals of width 0.5 each
        # Interval 1: [0.65, 1.15], Interval 2: [1.35, 1.85]
        # These are approximated to sample points
        a_start = 0.65
        b_start = 1.35
        interval_width = 0.5
        
        # Find indices for first interval
        a_idx = int(a_start / dx)
        b_idx = int(b_start / dx)
        a_width = int(interval_width / dx)
        
        # Make sure we're not going out of bounds
        a_end = min(a_idx + a_width, n_points)
        b_end = min(b_idx + a_width, n_points)
        
        # Ensure total integral is 1 (sum * dx = 1)
        h_start[a_idx:a_end] = 1
        h_start[b_idx:b_end] = 1
        
        # Adjust to make sure integral matches exactly
        current_sum = np.sum(h_start)
        if current_sum > 0:
            h_start = h_start * (n_points / 2) / current_sum
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try several variants manually to avoid numerical noise
    attempts = []
    # Try different placements: shift the intervals
    for shift in np.linspace(-0.1, 0.1, 5):
        h_test = np.zeros(n_points)
        a = 0.65 + shift
        b = 1.35 + shift
        a_idx = max(0, min(int(a / dx), n_points - 1))
        b_idx = max(0, min(int(b / dx), n_points - 1))
        a_width = int(0.5 / dx)
        a_end = min(a_idx + a_width, n_points)
        b_end = min(b_idx + a_width, n_points)
        if a_end > a_idx and b_end > b_idx:
            h_test[a_idx:a_end] = 1
            h_test[b_idx:b_end] = 1
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            c5 = evaluate_erdos_solution(h_test, dx)
            attempts.append((h_test, c5))
    
    # Also try just one interval with higher density
    h_single = np.zeros(n_points)
    a = 0.5
    a_idx = int(a / dx)
    a_width = int(0.5 / dx)
    a_end = min(a_idx + a_width, n_points)
    if a_end > a_idx:
        h_single[a_idx:a_end] = 1
        h_single = h_single * (n_points / 2) / np.sum(h_single)
        c5 = evaluate_erdos_solution(h_single, dx)
        attempts.append((h_single, c5))
    
    # Try symmetric two interval placement that might be better
    h_sym = np.zeros(n_points)
    # Place intervals at center locations that allow more balanced overlap
    a_pos = 0.5
    b_pos = 1.5
    a_idx = int(a_pos / dx)
    b_idx = int(b_pos / dx)
    a_width = int(0.5 / dx)
    a_end = min(a_idx + a_width, n_points)
    b_end = min(b_idx + a_width, n_points)

        h_sym[a_idx:a_end] = 1
        h_sym[b_idx:b_end] = 1
        h_sym = h_sym * (n_points / 2) / np.sum(h_sym)
        c5 = evaluate_erdos_solution(h_sym, dx)
        attempts.append((h_sym, c5))
    
    # Incorporate all variants
    for h_try, c5_try in attempts:
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
    
    # Try final refinement with a simpler optimization
    # Just small perturbations of the best found solution
    h_refined = best_h.copy()
    for i in range(min(50, n_points)):
        idx = i % n_points
        # Try small adjustment up/down
        h_perturbed = h_refined.copy()
        if h_refined[idx] < 1.0:
            h_perturbed[idx] = min(1.0, h_refined[idx] + 0.02)
        else:
            h_perturbed[idx] = max(0.0, h_refined[idx] - 0.02)
        
        # Readjust sum to 1
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        c5_new = evaluate_erdos_solution(h_perturbed, dx)
        if c5_new < best_c5:
            best_h = h_perturbed
            best_c5 = c5_new
            
        if time() - start_time > budget_s * 0.95:
            break
    
    return (list(best_h), best_c5, n_points)
