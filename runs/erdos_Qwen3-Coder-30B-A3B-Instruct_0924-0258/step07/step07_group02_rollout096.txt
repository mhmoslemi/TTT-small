import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Local optimization using scipy
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_integral(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    # Bound constraints for scipy
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_integral}]
    
    # Try multiple local search strategies
    for i in range(5):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize new random point near current best
        noise = np.random.normal(0, 0.05, n_points)
        h_trial = best_h + noise
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        
        # Local optimization
        try:
            result = minimize(objective, h_trial, method='SLSQP', bounds=bounds, 
                            constraints=constraints, options={'ftol': 1e-8, 'maxiter': 500})
            if result.success:
                c5_trial = result.fun
                if c5_trial < best_c5:
                    best_h = result.x.copy()
                    best_c5 = c5_trial
        except:
            pass
            
        # Random perturbation
        try:
            noise = np.random.normal(0, 0.02, n_points)
            h_trial = best_h + noise
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points/2) / np.sum(h_trial)
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
        except:
            pass
    
    # Final convex optimization refinement via CVXPY
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Collect active constraints via empirical analysis
        h_test = best_h.copy()
        corr = np.correlate(h_test, 1 - h_test, mode='full')
        # Use top 20% of lags with highest correlations
        top_lags = np.argsort(corr)[-n_points//5:]
        
        for k in top_lags:
            if k < n_points:
                corr_term = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_term = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_term <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final verification and cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
