import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with periodic function
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    T = 0.5
    alpha = 0.995
    T_min = 1e-5
    tabu_list = []
    tabu_size = 20
    max_iter = 1000
    stagnation_limit = 50
    stagnation_count = 0
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Adaptive perturbation size
        perturbation_size = 0.1 * (0.5 ** (iteration / 100))
        
        # Select random indices to perturb
        num_perturb = max(1, int(0.15 * n_points))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        # Create candidate solution
        candidate_h = current_h.copy()
        for i in indices:
            delta = np.random.uniform(-perturbation_size, perturbation_size)
            candidate_h[i] += delta
        
        # Enforce bounds
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Normalize to meet sum constraint
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Acceptance probability
        if candidate_c5 < current_c5:
            accepted = True
        else:
            # Accept with probability based on temperature
            delta = candidate_c5 - current_c5
            prob = math.exp(-delta / T)
            accepted = np.random.random() < prob
        
        if accepted:
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Tabu list management
        if len(tabu_list) >= tabu_size:
            tabu_list.pop(0)
        tabu_list.append(current_h.copy())
        
        # Temperature update
        T = max(T * alpha, T_min)
        
        # Stagnation restart
        if stagnation_count > stagnation_limit:
            # Perturb the best solution
            restart_h = best_h.copy()
            for i in range(n_points):
                if np.random.random() < 0.1:
                    delta = np.random.uniform(-0.1, 0.1)
                    restart_h[i] += delta
            restart_h = np.clip(restart_h, 0, 1)
            restart_h = restart_h * (n_points / 2) / np.sum(restart_h)
            current_h = restart_h.copy()
            current_c5 = evaluate_erdos_solution(current_h, dx)
            stagnation_count = 0
        
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
