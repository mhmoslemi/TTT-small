import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize
    h = best_h.copy()
    max_iter = 5000
    convergence_tol = 1e-6
    alpha = 0.1
    mu = 100.0
    lambda_multiplier = 0.0
    trust_radius = 0.1
    
    # Optimization loop
    prev_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlap and gradient
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Gradient computation via finite differences
        grad = np.zeros_like(h)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1.0
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (corr_plus[max_idx] - corr[max_idx]) * dx / eps
        
        # ALM update step
        lagrange_grad = grad + lambda_multiplier + mu * (h - 1)
        
        # Trust region step
        step = -alpha * lagrange_grad
        new_h = h + step
        
        # Project onto feasible set
        new_h = np.clip(new_h, 0, 1)
        scale = (n_points/2) / np.sum(new_h)
        new_h = new_h * scale
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept step if improvement
        if new_c5 < best_c5:
            h = new_h
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # Reject step (more conservative than previous attempts)
            pass
            
        # Update Lagrange multiplier
        lambda_multiplier = mu * (np.sum(h) - n_points/2)
        
        # Update penalty parameter
        if abs(np.sum(h) - n_points/2) > 1e-4 or new_c5 > best_c5:
            mu *= 2.0
        else:
            mu *= 0.9
            
        # Adaptive trust region
        if abs(new_c5 - prev_c5) < convergence_tol and iteration > 100:
            trust_radius *= 0.9
        else:
            trust_radius *= 1.05
            
        trust_radius = min(trust_radius, 0.5)
        alpha = min(alpha, trust_radius)
        
        prev_c5 = new_c5
        
        # Periodic refinement
        if iteration % 100 == 0:
            try:
                def objective(h_vec):
                    return evaluate_erdos_solution(h_vec, dx)
                
                def constraint_sum(h_vec):
                    return np.sum(h_vec) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(objective, h, method='SLSQP',

                               options={'maxiter': 20, 'ftol': 1e-6})
                if res.success:
                    h = res.x
                    c5_refined = objective(h)
                    if c5_refined < best_c5:
                        best_h = h.copy()
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final polishing
    try:
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP',

            final_c5 = objective(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
