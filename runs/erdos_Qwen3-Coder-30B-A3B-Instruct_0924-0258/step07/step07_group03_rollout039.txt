import numpy as np
from scipy.optimize import minimize
from time import time
try:
    import cvxpy as cp
except ImportError:
    cp = None

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Convex relaxation approach using SDP
    # Only proceed if cvxpy is available
    if cp is not None and budget_s > 100:
        try:
            remaining_time = budget_s - (time() - start_time)
            if remaining_time > 50:
                # Create a more structured starting solution based on the initial one
                h_trial = h_start.copy()
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points/2) / np.sum(h_trial)
                
                # Try SDP relaxation
                h_sdp = h_trial.copy()
                
                # Use CVXPY to solve relaxation
                h_var = cp.Variable(n_points)
                Z = cp.Variable((n_points, n_points), symmetric=True)
                
                # Constraints
                constraints = [
                    cp.trace(Z) <= n_points / 2.0,
                    Z >> 0,  # Positive semidefinite
                    Z[0, 0] == cp.matmul(h_var, h_var),
                    Z[0, 1] == cp.matmul(h_var, h_var),
                    cp.diag(Z) == h_var,
                ]
                
                # Objective: minimize t
                t = cp.Variable()
                objective = cp.Minimize(t)
                
                # Formulate problem
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.SCS, verbose=False, time_limit=min(100, remaining_time/2))
                
                # Attempt to extract a good heuristic solution
                if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                    # Try to get a feasible solution from this
                    h_sdp = np.clip(h_var.value, 0, 1)
                    # Normalize it to sum to n_points/2
                    h_sdp = h_sdp * (n_points/2) / np.sum(h_sdp)
                    if np.all(h_sdp >= 0) and np.all(h_sdp <= 1) and abs(np.sum(h_sdp) - n_points/2) < 1e-6:
                        c5_sdp = evaluate_erdos_solution(h_sdp, dx)
                        if c5_sdp < best_c5:
                            best_h = h_sdp.copy()
                            best_c5 = c5_sdp
        except Exception:
            pass
    
    # Phase 2: Local optimization refinement with Basin Hopping-style iterations
    remaining_time = budget_s - (time() - start_time)
    if remaining_time < 10:
        return (list(best_h), best_c5, n_points)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    consistency_tol = 1e-10
    for _ in range(50):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation
        h_perturbed = best_h + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Local optimization
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 300, 'ftol': consistency_tol})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except Exception:
            pass
            
        # Try more direct local search
        h_new = best_h.copy()
        for i in range(n_points):
            if time() - start_time > budget_s * 0.95:
                break
            trial_h = h_new.copy()
            # Simple gradient descent step
            step_size = 0.01
            trial_h[i] += step_size
            if trial_h[i] > 1.0:
                trial_h[i] = 1.0
            trial_h = trial_h * (n_points/2) / np.sum(trial_h)
            c5_new = evaluate_erdos_solution(trial_h, dx)
            if c5_new < best_c5:
                best_h = trial_h.copy()
                best_c5 = c5_new
            
    return (list(best_h), best_c5, n_points)
