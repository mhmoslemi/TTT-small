import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient-based optimization approach
    max_iter = 10000
    step_size = 0.01
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        max_val = corr[max_lag_idx]
        
        # Compute gradient for the most constraining lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Compute derivative of overlap integral w.r.t. h[i]
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * h[i] - (1 - 2 * h[j])
            else:
                grad[i] = 1 - 2 * h[i]
        
        # Update with gradient descent
        h_new = h - step_size * grad
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if this improves the solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new.copy()
        
        # Adjust step size cautiously
        if iteration % 100 == 0 and iteration > 0:
            step_size *= 0.99
        
        # Early exit if improvement is insignificant
        if abs(best_c5 - c5_new) < 1e-10:
            break
    
    return (list(best_h), best_c5, n_points)
