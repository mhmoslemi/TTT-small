import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Periodic initialization
        n_points = 20
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
        h_start += np.random.uniform(-0.05, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multiresolution approach
    resolutions = [20, 40, 80]
    current_n = n_points
    current_h = h_start.copy()
    
    # Gradually refine
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
        if res <= current_n:
            # Interpolate to higher resolution
            x_orig = np.linspace(0, 2, current_n)
            x_new = np.linspace(0, 2, res)
            current_h = np.interp(x_new, x_orig, current_h)
            current_n = res
        
        # Adaptive perturbation with simulated annealing
        max_iter = 200 if res == 80 else 150 if res == 40 else 100
        T = 1.0
        T_min = 1e-4
        alpha = 0.99
        tabu_size = 20
        tabu_list = []
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb randomly
            h_candidate = current_h.copy()
            indices = np.random.choice(current_n, size=max(1, current_n // 5), replace=False)
            perturbation = np.random.uniform(-0.15, 0.15, len(indices))
            
            for i, idx in enumerate(indices):
                h_candidate[idx] += perturbation[i]
            
            # Clip and renormalize
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (current_n / 2) / np.sum(h_candidate)
            
            # Evaluate
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept or reject
            if c5_candidate < best_c5:
                current_h = h_candidate.copy()
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
            else:
                # Accept with probability based on temperature
                delta = c5_candidate - best_c5
                if delta > 0 and np.random.rand() < np.exp(-delta / T):
                    current_h = h_candidate.copy()
            T = max(T_min, T * alpha)
            if T < T_min:
                break

    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - current_n / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(current_n)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    return (list(best_h), best_c5, current_n)
