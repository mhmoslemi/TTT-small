import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        # Try a piecewise constant function that's more localized
        h_start = np.zeros(n_points)
        h_start[:n_points//3] = 0.8
        h_start[n_points//3:2*n_points//3] = 0.2
        h_start[2*n_points//3:] = 0.6
        
        # Scale to meet constraint
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Mutate the best solution with small random perturbations
            h = best_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            # Renormalize
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
    
    best_fitness = min(fitness)
    if best_fitness < best_c5:
        best_h = population[np.argmin(fitness)].copy()
        best_c5 = best_fitness
    
    # Evolution loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]
        fitness = [fitness[i] for i in sorted_idx]

        elite_size = pop_size // 5
        new_pop = population[:elite_size]
        fitness = fitness[:elite_size]
        
        # Generate offspring
        while len(new_pop) < pop_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(elite_size, 2, replace=False)
            parent1, parent2 = population[idx1], population[idx2]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    mut_idx = np.random.randint(n_points)
                    child[mut_idx] += np.random.normal(0, 0.05)
                    child[mut_idx] = np.clip(child[mut_idx], 0, 1)
                    
                # Renormalize
                child = child * (n_points/2) / np.sum(child)
                new_pop.append(child)
        
        # Evaluate new population
        population = new_pop[:pop_size]
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Update global best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:

            best_c5 = fitness[current_best_idx]

    try:

        top_solutions = population[:10]
        best_of_top = min(top_solutions, key=lambda x: evaluate_erdos_solution(x, dx))

        def obj(h_vec):
            h = np.array(h_vec)
            return evaluate_erdos_solution(h, dx)

        lb = [0.0] * n_points
        ub = [1.0] * n_points
        
        # Try to refine around best solution
        result = minimize(
            fun=obj,
            x0=best_of_top,
            method='L-BFGS-B',
            bounds=list(zip(lb, ub)),
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success:
            refined_h = np.array(result.x)
            # Renormalize
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
