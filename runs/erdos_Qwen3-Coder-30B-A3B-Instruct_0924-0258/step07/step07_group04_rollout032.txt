import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        # Periodic initialization with 5 cycles
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
        # Add small random perturbations
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive multiresolution approach
    resolutions = [20, 40, 80]
    current_n = 20
    current_h = h_start.copy()
    
    # For each resolution
    for r, n in enumerate(resolutions):
        if r > 0:
            # Interpolate from previous resolution
            x_old = np.linspace(0, 2, current_n)
            x_new = np.linspace(0, 2, n)
            current_h = np.interp(x_new, x_old, current_h)
            # Add random perturbations
            current_h += np.random.normal(0, 0.05, n)
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n/2) / np.sum(current_h)
        
        current_n = n
        dx = 2.0 / current_n
        
        # Cooling schedule parameters
        temp = 1.0
        temp_decay = 0.99
        temp_min = 1e-6
        iteration = 0
        stagnation_count = 0
        max_stagnation = 100
        
        while time() - start_time < budget_s * 0.95:
            if iteration >= 300:
                break
                
            # Selection of points to perturb: 10-20%
            n_perturb = max(1, min(current_n//5, current_n//2))
            indices = np.random.choice(current_n, n_perturb, replace=False)
            
            # Create candidate
            h_candidate = current_h.copy()
            # Perturb selected points
            for i in indices:
                delta = np.random.normal(0, 0.15)
                h_candidate[i] += delta
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (current_n/2) / np.sum(h_candidate)
            
            # Evaluate
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept with probability
            if c5_candidate < current_h.max() or np.random.random() < np.exp(-(c5_candidate - current_h.max()) / temp):
                current_h = h_candidate
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
            
            # Update temperature
            if iteration % 50 == 0:
                temp = max(temp_min, temp * temp_decay)
            
            iteration += 1
            
            if stagnation_count > max_stagnation:
                break
    
    # Final polishing with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
