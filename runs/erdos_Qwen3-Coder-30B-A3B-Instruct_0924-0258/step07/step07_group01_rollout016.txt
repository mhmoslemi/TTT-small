from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    # Alternating optimization
    max_iter = 5000
    stagnation_limit = 100
    stagnation_count = 0

    a = np.ones(n_points) * 0.5

    s[:n_points//2] = 1

    h = a * s
    h_sum = np.sum(h)

    # Normalize s
    h = a * s
    h_sum = np.sum(h)
    
    for it in range(max_iter):
        current_time = time()
        if current_time - start_time > budget_s * 0.95:
            break
            
        # Fix s, optimize a
        # Problem: minimize max overlap under constraints
        try:
            # Compute overlap function ones

                h = a * s
                return evaluate_erdos_solution(h, dx)

        pass

    return (list(best_h), best_c5, n_points)
