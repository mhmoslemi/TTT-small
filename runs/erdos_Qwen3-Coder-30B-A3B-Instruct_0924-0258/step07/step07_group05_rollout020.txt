import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Basis function approach
    n_basis = 10
    # Create piecewise linear basis functions (triangular pulses)
    basis_funcs = []
    x_grid = np.linspace(0, 2, n_points)
    
    # Initialize basis functions with triangular pulses
    for i in range(n_basis):
        basis = np.zeros(n_points)
        center = i * 2.0 / (n_basis - 1) if n_basis > 1 else 1.0
        width = 0.4
        for j in range(n_points):
            x = x_grid[j]
            dist = abs(x - center)
            if dist <= width:
                basis[j] = 1.0 - dist / width
        basis_funcs.append(basis)
    
    # Normalize basis functions and ensure they sum to 1 for normalization
    basis_matrix = np.array(basis_funcs).T  # shape: (n_points, n_basis)
    
    # Initial coefficients - initialize to match the starting function
    initial_coeffs = np.linalg.lstsq(basis_matrix, h_start, rcond=None)[0]
    current_coeffs = initial_coeffs.copy()
    
    # Use trust region with bounds
    max_iter = 5000
    trust_radius = 0.1
    patience = 0
    min_patience = 50
    perturbation_magnitude = 0.05
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Compute h from current coefficients
        h_current = basis_matrix @ current_coeffs
        h_current = np.clip(h_current, 0, 1)
        
        # Apply sum constraint
        sum_h = np.sum(h_current)
        h_current = h_current * (n_points / 2) / sum_h
        
        # Evaluate
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            patience = 0
        else:
            patience += 1
            
        # Perturbation if no improvement
        if patience > min_patience:
            # Add noise to coefficients
            noise = np.random.normal(0, perturbation_magnitude, n_basis)
            current_coeffs += noise
            perturbation_magnitude *= 0.8
            patience = 0
            
        # Trust region optimization step
        def objective(coeffs):
            h_eval = basis_matrix @ coeffs
            h_eval = np.clip(h_eval, 0, 1)
            h_eval = h_eval * (n_points / 2) / np.sum(h_eval)
            return evaluate_erdos_solution(h_eval, dx)
        
        def gradient(coeffs):
            # Numerical gradient approximation
            eps = 1e-6
            grad = np.zeros_like(coeffs)
            for i in range(len(coeffs)):
                coeffs_plus = coeffs.copy()
                coeffs_plus[i] += eps
                coeffs_minus = coeffs.copy()
                coeffs_minus[i] -= eps
                grad[i] = (objective(coeffs_plus) - objective(coeffs_minus)) / (2 * eps)
            return grad
            
        # Take a step in the gradient direction within trust region
        grad = gradient(current_coeffs)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            step = -grad / grad_norm * trust_radius
            # Ensure within trust region
            step_norm = np.linalg.norm(step)
            if step_norm > trust_radius:
                step = step / step_norm * trust_radius
            new_coeffs = current_coeffs + step
        else:
            new_coeffs = current_coeffs
            
        # Evaluate new point
        h_new = basis_matrix @ new_coeffs
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_coeffs = new_coeffs
            trust_radius *= 1.2  # expand trust region
        else:
            trust_radius *= 0.5  # shrink trust region
            
        # Projection to feasible region
        current_coeffs = np.clip(current_coeffs, -10, 10)  # bound coefficients
    
    # Final refinement with Nelder-Mead
    try:
        def objective_to_minimize(coeffs):
            h_eval = basis_matrix @ coeffs
            h_eval = np.clip(h_eval, 0, 1)
            h_eval = h_eval * (n_points / 2) / np.sum(h_eval)
            return evaluate_erdos_solution(h_eval, dx)
            
        res = minimize(objective_to_minimize, current_coeffs, 
                       method='Nelder-Mead', options={'maxiter': 100, 'fatol': 1e-8})
        if res.success:
            h_final = basis_matrix @ res.x
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
