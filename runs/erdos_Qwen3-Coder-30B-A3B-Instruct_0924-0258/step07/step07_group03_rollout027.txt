import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient Descent with Line Search
    max_iter = 1000
    lr = 0.1
    patience = 0
    patience_limit = 50
    
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute overlaps for all lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Gradient computation for the worst lag
        # The gradient of max_overlap w.r.t. h[i]
        # For lag k, the derivative contributes to h[i] at overlap position
        # Let's do it in a more direct way - consider specific lag and compute gradient
        k_max = max_idx - (len(h_current) - 1)  # convert to actual lag
        # Compute gradient contributions for each h[i]
        gradient = np.zeros_like(h_current)
        for i in range(len(h_current)):
            pos = i + k_max
            if 0 <= pos < len(h_current):
                gradient[i] += (1 - h_current[pos]) * dx
            pos = i - k_max
            if 0 <= pos < len(h_current):
                gradient[i] += h_current[pos] * dx
        
        # Update with gradient descent
        h_new = h_current - lr * gradient
        
        # Project to feasible region (0-1 and sum constraint)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Line search: try a few step sizes
        best_h_try = h_current
        best_val = max_overlap
        for step_size in [0.01, 0.05, 0.1, 0.2, 0.5]:
            h_try = h_current - step_size * gradient
            h_try = np.clip(h_try, 0, 1)
            h_try = h_try * (n_points/2) / np.sum(h_try)
            val = evaluate_erdos_solution(h_try, dx)
            if val < best_val:
                best_val = val
                best_h_try = h_try.copy()
        
        # Update h
        h_current = best_h_try.copy()
        
        # Check improvement
        if best_val < best_c5:
            best_c5 = best_val
            best_h = h_current.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > patience_limit:
            lr *= 0.5
            patience = 0
            if lr < 1e-6:
                break
    
    # Phase 2: Local refinement with L-BFGS-B
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='L-BFGS-B', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
