import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Benders decomposition approach
    max_iter = 5000
    convergence_tol = 1e-6
    perturbation_mag = 0.05
    perturbation_decay = 0.95
    
    # Initialize
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Keep track of constraints
    constraints_list = []
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all k
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        overlaps = corr * dx
        max_overlap_idx = np.argmax(overlaps)
        max_overlap_val = overlaps[max_overlap_idx]
        
        # Find k from the index of max overlap (this is a full correlation)
        # k ranges from 0 to n_points-1 (but shifted properly)
        # The full correlation has length 2*n_points - 1
        # We want the actual shift k that caused this
        k = max_overlap_idx - (n_points - 1)
        
        # Make sure k is valid
        if k < 0:
            k = -k
        elif k > 0:
            k = k
            
        # Add constraint if not already present
        constraint_key = (k, max_overlap_val)
        if constraint_key not in constraints_list:
            constraints_list.append(constraint_key)
        
        # Solve convex relaxation with added constraints
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add previously identified constraints (approximated with linear cuts)
            if len(constraints_list) > 0:
                for (k_val, bound) in constraints_list:
                    if k_val < n_points and k_val >= 0:
                        # We approximate by a linearization at current h
                        lower_bound = max_overlap_val - 1e-8
                        s = np.zeros(n_points)
                        for i in range(n_points - k_val):
                            s[i] += h_current[i] * (1 - h_current[i + k_val])
                        # For new h: sum h_i * (1 - h_{i+k}) 
                        # Linearization of this wrt h_current for small changes
                        # Approximate by upper bound for now to be safe
                        if k_val < n_points:
                            # Constraint: sum_{i=0}^{n-1-k} h[i] (1 - h[i+k]) <= bound * dx
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k_val], 1 - h_opt[k_val:]))
                            constraints.append(corr_sum <= bound * dx)
            
            # Minimize objective: just a dummy objective
            obj = cp.Minimize(0)
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_opt.value
                h_current = h_new
        except Exception as e:
            pass
        
        # Apply perturbation to escape local optima
        if perturbation_mag > 1e-8:
            noise = np.random.randn(n_points) * perturbation_mag
            h_current += noise
            h_current = np.clip(h_current, 0, 1)
            # Rescale
            h_current = h_current * (n_points/2) / np.sum(h_current)
            perturbation_mag *= perturbation_decay
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_current, dx)
        if new_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = new_c5
    
    # Final polishing with gradient-based method
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h.copy(), method='SLSQP', 
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 200, 'ftol': 1e-8})
        
        if res.success:
            polished_h = res.x
            polished_c5 = objective(polished_h)
            if polished_c5 < best_c5:
                best_h = polished_h
                best_c5 = polished_c5
    except Exception:
        pass
    
    # Final verification
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
