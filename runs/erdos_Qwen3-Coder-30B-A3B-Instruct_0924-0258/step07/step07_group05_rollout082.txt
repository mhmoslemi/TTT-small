import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    elite_size = 5
    max_generations = 1000
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # GA iterations
    stagnation_counter = 0
    last_best = best_c5
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Track best
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_idx = fitness.index(min_fitness)
            best_h = population[best_idx].copy()
            best_c5 = min_fitness
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        
        # Check for stagnation
        if stagnation_counter > 50:
            # Reseed population
            for i in range(pop_size):
                if 'initial_h_values' in globals():
                    h = h_start + np.random.normal(0, 0.05, n_points)
                else:
                    h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                population[i] = h
        
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size)
            best_in_tournament = min(tournament, key=lambda i: fitness[i])
            selected.append(population[best_in_tournament].copy())
        
        # Crossover and mutation
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Selection
            parents = np.random.choice(pop_size, 2, replace=False)
            
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, selected[parents[0]], selected[parents[1]])
                child2 = np.where(mask, selected[parents[1]], selected[parents[0]])
            else:
                child1 = selected[parents[0]].copy()
                child2 = selected[parents[1]].copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    mut_indices = np.random.choice(n_points, int(mutation_rate * n_points))
                    for idx in mut_indices:
                        child[idx] += np.random.normal(0, 0.03)
                    child = np.clip(child, 0, 1)
                    # Maintain sum constraint
                    child = child * (n_points/2) / np.sum(child)
                    new_population.append(child)
                    if len(new_population) >= pop_size:
                        break
        
        population = new_population[:pop_size]
        
        # Periodic convex relaxation
        if generation % 50 == 0 and generation > 0:
            try:
                # Test top individuals
                sorted_indices = np.argsort(fitness)[:5]  # Top 5
                for idx in sorted_indices:
                    h_candidate = population[idx].copy()
                    h_opt = cp.Variable(n_points)
                    constraints = [
                        cp.sum(h_opt) == n_points / 2.0,
                        h_opt >= 0,
                        h_opt <= 1
                    ]
                    
                    # Add constraints using sampled k values
                    k_vals = np.linspace(0.05, 1.95, min(20, n_points//3))
                    for k_val in k_vals:
                        k = int(k_val * n_points / 2)
                        if k < n_points and k > 0:
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                            constraints.append(corr_sum <= (best_c5 + 1e-6) * dx)
                    
                    prob = cp.Problem(cp.Minimize(0), constraints)
                    prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                    if prob.status == cp.OPTIMAL:
                        h_feasible = h_opt.value
                        c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                        if c5_feasible < best_c5:
                            best_h = h_feasible
                            best_c5 = c5_feasible
            except Exception:
                pass
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
