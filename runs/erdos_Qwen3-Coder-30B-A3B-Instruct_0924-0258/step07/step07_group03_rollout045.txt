import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # SA parameters
    T0 = 0.1
    cooling_rate = 0.001
    max_iter = 1000
    perturbation_fraction = 0.15
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Early termination counter
    stagnation_count = 0
    max_stagnation = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb h
        perturb_indices = np.random.choice(n_points, int(n_points * perturbation_fraction), replace=False)
        new_h = current_h.copy()
        delta = np.random.uniform(-0.1, 0.1, len(perturb_indices))
        new_h[perturb_indices] += delta
        new_h = np.clip(new_h, 0, 1)
        
        # Normalize
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Calculate temperature and acceptance probability
        T = T0 * np.exp(-cooling_rate * iteration)
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            current_h = new_h
            current_c5 = new_c5
            stagnation_count = 0
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            stagnation_count += 1
            
        if stagnation_count >= max_stagnation:
            break
    
    # Gradient descent refinement
    h_refined = best_h.copy()
    learning_rate = 0.1
    gd_max_iter = 500
    gd_stagnation_count = 0
    gd_stagnation_limit = 50
    prev_c5 = best_c5
    
    for iteration in range(gd_max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find the most problematic lag
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        k_max = np.argmax(corr)
    
        # Compute gradient for the k_max lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] = (1 - h_refined[i + k_max]) * dx
            if i - k_max >= 0:
                grad[i] += (1 - h_refined[i - k_max]) * dx
                
        # Update h
        h_refined -= learning_rate * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_refined, dx)
        
        if new_c5 < prev_c5:
            prev_c5 = new_c5
            gd_stagnation_count = 0
        else:
            gd_stagnation_count += 1
            
        if gd_stagnation_count >= gd_stagnation_limit:
            learning_rate *= 0.5
            gd_stagnation_count = 0
            if learning_rate < 1e-6:
                break
    
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
