import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize parameters
    max_iter = 5000
    convergence_tol = 1e-6
    alpha = 0.1
    mu = 100.0
    trust_radius = 1.0
    
    # Gradient descent with adaptive penalty
    h_current = best_h.copy()
    prev_c5 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current C5 and active overlap
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        # Compute gradient - approximate via finite differences
        eps = 1e-6
        grad = np.zeros_like(h_current)
        h_plus = h_current.copy()
        
        for i in range(n_points):
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1.0
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            grad[i] = (c5_plus - current_c5) / eps
            h_plus[i] = h_current[i]
        
        # Update with projected gradient descent
        h_new = h_current - alpha * grad
        
        # Project onto feasible set [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to satisfy constraint
        sum_h = np.sum(h_new)
        if abs(sum_h - n_points/2) > 1e-8:
            h_new = h_new * (n_points/2) / sum_h
            
        # Compute new objective
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if it improves
        if new_c5 < current_c5:
            h_current = h_new
            current_c5 = new_c5
            
            # Update best solution
            if new_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = new_c5
                
            # Increase step size with trust region
            trust_radius = min(trust_radius * 1.1, 1.0)
            alpha = min(alpha * 1.1, 0.5)
        else:
            # Decrease step size and try again with smaller trust region
            alpha *= 0.5
            trust_radius *= 0.5
            
        # Check convergence
        if abs(prev_c5 - current_c5) < convergence_tol:
            break
            
        prev_c5 = current_c5
        
        # Occasionally update Lagrange multiplier (penalty)
        if iteration % 100 == 0:
            sum_h = np.sum(h_current)
            mu = max(mu * 0.9, 1.0)
    
    # Final polishing with stricter constraints
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final check of constraints
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points/2) > 1e-6:
        best_h = best_h * (n_points/2) / sum_h
    
    return (list(best_h), best_c5, n_points)
