import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    mut_rate = 0.2
    cross_rate = 0.7
    elite_size = 5
    tournament_size = 5
    conv_freq = 50
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Genetic algorithm iterations
    for gen in range(1000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        new_pop = population[:elite_size]
        new_fit = fitness[:elite_size]
        
        # Tournament selection and reproduction
        while len(new_pop) < pop_size:
            # Tournament selection
            parent1_idx = min(np.random.choice(pop_size, tournament_size, replace=False), key=lambda i: fitness[i])
            parent2_idx = min(np.random.choice(pop_size, tournament_size, replace=False), key=lambda i: fitness[i])
            
            if np.random.rand() < cross_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, population[parent1_idx], population[parent2_idx])
            else:
                child = population[parent1_idx].copy()
            
            # Mutation
            if np.random.rand() < mut_rate:
                mutate_positions = np.random.choice(n_points, int(mut_rate * n_points), replace=False)
                for pos in mutate_positions:
                    child[pos] += np.random.normal(0, 0.03)
            
            # Repair constraint
            child = np.clip(child, 0, 1)
            sum_h = np.sum(child)
            if abs(sum_h - n_points/2) > 1e-6:
                child = child * (n_points/2) / sum_h
            
            c5 = evaluate_erdos_solution(child, dx)
            new_pop.append(child)
            new_fit.append(c5)
            
            if c5 < best_c5:
                best_h = child.copy()
                best_c5 = c5
        
        population = new_pop
        fitness = new_fit
        
        # Convex relaxation every conv_freq generations
        if gen % conv_freq == 0 and gen > 0:
            try:
                # Get top solutions
                top_indices = np.argsort(fitness)[:10]
                for idx in top_indices:
                    h_test = population[idx]
                    # Create convex problem
                    h_opt = cp.Variable(n_points)
                    constraints = [
                        cp.sum(h_opt) == n_points / 2.0,
                        h_opt >= 0,
                        h_opt <= 1
                    ]
                    
                    # Sample a few constraints
                    sample_k = np.linspace(1, n_points-1, min(15, n_points//2), dtype=int)
                    for k in sample_k:
                        if k < n_points:
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                    
                    prob = cp.Problem(cp.Minimize(0), constraints)
                    prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                    if prob.status == cp.OPTIMAL:
                        h_feasible = h_opt.value
                        c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                        if c5_feasible < best_c5:
                            best_h = h_feasible
                            best_c5 = c5_feasible
            except Exception:
                pass
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
