import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T0 = 0.1
    cooling_rate = 0.001
    max_iter = 1000
    perturbation_fraction = 0.15
    stagnation_limit = 100
    
    h_current = h_start.copy()
    c5_current = best_c5
    T = T0
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Perturb
        n_perturb = int(n_points * perturbation_fraction)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        h_new = h_current.copy()
        delta = np.random.uniform(-0.1, 0.1, n_perturb)
        h_new[indices] += delta
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        delta_c5 = c5_new - c5_current
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h_current = h_new
            c5_current = c5_new
            
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        T = T0 * np.exp(-cooling_rate * iteration)
        
        if stagnation_count > stagnation_limit:
            break
    
    # Gradient Descent refinement
    h_refined = best_h.copy()
    c5_refined = best_c5
    learning_rate = 0.1
    max_gd_iter = 500
    gd_stagnation = 0
    gd_stagnation_limit = 50
    
    for _ in range(max_gd_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap for all lags to find k_max
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient (only for the k_max case)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                gradient[i] = (1 - h_refined[i + k_max]) * dx
            if i - k_max >= 0:
                gradient[i] += (1 - h_refined[i - k_max]) * dx
                
        # Update h
        h_new = h_refined - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < c5_refined:
            h_refined = h_new
            c5_refined = c5_new
            gd_stagnation = 0
        else:
            gd_stagnation += 1
            if gd_stagnation >= gd_stagnation_limit:
                learning_rate /= 2.0
                gd_stagnation = 0
                if learning_rate < 1e-6:
                    break
    
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < c5_refined:
                h_refined = h_final
                c5_refined = c5_final
    except Exception:
        pass
    
    return (list(h_refined), c5_refined, n_points)
