import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a different starting point based on a smooth construction
        x = np.linspace(0, 2, n_points)
        # Use a smooth sine-like function with higher frequency content
        h_start = 0.5 * (1 + np.sin(3 * np.pi * x / 2))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a hybrid optimization: local search + random sampling
    opt_iters = 500
    for iter_num in range(opt_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbations around best solution
        perturb = np.random.normal(0, 0.05, n_points)
        h_proposal = best_h + perturb
        h_proposal = np.clip(h_proposal, 0, 1)
        h_proposal = h_proposal * (n_points/2) / np.sum(h_proposal)
        
        c5_proposal = evaluate_erdos_solution(h_proposal, dx)
        if c5_proposal < best_c5:
            best_h = h_proposal.copy()
            best_c5 = c5_proposal
            
        # Occasionally do a more aggressive local search
        if iter_num % 10 == 0:
            # Gradient descent with small steps
            h_current = best_h.copy()
            for _ in range(5):
                grad = np.zeros_like(h_current)
                # Compute gradient numerically
                eps = 1e-4
                base_corr = np.correlate(h_current, 1 - h_current, mode='full')
                base_max_k = np.argmax(base_corr)
                
                for i in range(n_points):
                    h_test = h_current.copy()
                    h_test[i] += eps
                    h_test = np.clip(h_test, 0, 1)
                    h_test = h_test * (n_points/2) / np.sum(h_test)
                    test_corr = np.correlate(h_test, 1 - h_test, mode='full')
                    grad[i] = (test_corr[base_max_k] - base_corr[base_max_k]) / eps
                
                # Update step with adaptive learning rate
                learning_rate = 0.1 / (1 + iter_num / 100)
                h_current = h_current - learning_rate * grad
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points/2) / np.sum(h_current)
                
                current_c5 = evaluate_erdos_solution(h_current, dx)
                if current_c5 < best_c5:
                    best_h = h_current.copy()
                    best_c5 = current_c5
    
    # Final SQP refinement
    try:
        # Create a disciplined convex program
        h_opt = cp.Variable(n_points)
        obj = cp.Variable()
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Estimate dominant correlations
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        top_k = np.argsort(corr)[-10:]
        
        # Add constraints for main correlations
        for k in top_k:
            # Compute the correlation contribution
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= obj)
        
        prob = cp.Problem(cp.Minimize(obj), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
