import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient-based optimization
    max_iter = 1000
    step_size = 0.01
    tolerance = 1e-6
    patience = 50
    no_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        # Update best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            break
            
        # Compute gradient for the maximum overlap lag
        lag = max_lag_idx - (n_points - 1)
        grad = np.zeros(n_points)
        
        # Gradient calculation with proper indexing
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                grad[i] = 2 * (1 - h[j] - h[i])
            else:
                grad[i] = 0
                
        # Apply gradient descent
        h_new = h - step_size * grad
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to meet constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if we made progress; if not, reduce step size
        h = h_new
        new_c5 = evaluate_erdos_solution(h, dx)
        
        if abs(new_c5 - max_overlap) < tolerance:
            step_size *= 0.5
            if step_size < 1e-8:
                break
    
    # Final local optimization with scipy
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
