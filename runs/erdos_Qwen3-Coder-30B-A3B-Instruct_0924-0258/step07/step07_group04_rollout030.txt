import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        # Periodic initialization
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))  # 5 cycles
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multiresolution search
    resolutions = [20, 40, 80]
    current_n = 20
    current_h = h_start.copy()
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > current_n:
            # Interpolate to higher resolution
            x_old = np.linspace(0, 2, current_n)
            x_new = np.linspace(0, 2, res)
            current_h = np.interp(x_new, x_old, current_h)
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (res / 2) / np.sum(current_h)
            current_n = res
        else:
            pass  # Same resolution, continue with current_h
        
        dx = 2.0 / current_n
        # Annealing variables
        T = 0.1
        T_min = 1e-6
        alpha = 0.995
        iter_count = 0
        max_iter_per_res = 200 if res == 80 else 150 if res == 40 else 100
        tabu_list = []
        tabu_size = 20
        
        for iteration in range(max_iter_per_res):
            if time() - start_time > budget_s * 0.95:
                break
                
            iter_count += 1
            # Select random indices for perturbation (10-20%)
            n_perturb = max(1, int(0.15 * current_n))
            indices = np.random.choice(current_n, size=n_perturb, replace=False)
            
            # Create candidate
            h_candidate = current_h.copy()
            perturbation_size = max(0.05, 0.15 * (T / 0.1))
            
            for i in indices:
                # Add perturbation
                delta = np.random.uniform(-perturbation_size, perturbation_size) * h_candidate[i]
                h_candidate[i] += delta
            
            # Boundary conditions
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (current_n / 2) / np.sum(h_candidate)
            
            # Check tabu
            candidate_tuple = tuple(h_candidate.round(6))
            if candidate_tuple in tabu_list:
                continue
            
            # Evaluate
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept criterion
            if c5_candidate < best_c5:
                current_h = h_candidate.copy()
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
                
                # Update tabu list
                tabu_list.append(candidate_tuple)
                if len(tabu_list) > tabu_size:
                    tabu_list.pop(0)
            else:
                # Metropolis criterion
                delta = c5_candidate - best_c5
                if delta > 0 and np.random.random() < np.exp(-delta / T):
                    current_h = h_candidate.copy()
                    # Update tabu list
                    tabu_list.append(candidate_tuple)
                    if len(tabu_list) > tabu_size:
                        tabu_list.pop(0)
            
            # Decrease temperature
            if iter_count % 50 == 0:
                T = max(T_min, T * alpha)
    
    # Final refinement with SLSQP
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (current_n / 2) / np.sum(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - current_n / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(current_n)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, current_n)
