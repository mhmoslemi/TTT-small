import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient-based optimization with focus on max lag
    max_iter = 10000
    step_size = 0.01
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        # Identify the lag that gives maximum overlap
        # The middle index corresponds to lag 0
        mid = len(corr) // 2
        lag = max_lag_idx - mid
        
        # Compute gradient for this particular lag
        # For overlap at lag k: int h(x)(1-h(x+k)) dx
        # Gradient w.r.t. h_i is (1 - h[i+k] - h[i])
        grad = np.zeros(n_points)
        for j in range(n_points):
            k = j + lag
            if 0 <= k < n_points:
                grad[j] = 1 - h[k] - h[j]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
            
        # Gradient descent step
        h_new = h - step_size * grad
        
        # Project back to feasible region [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate and update if improvement
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new.copy()
            step_size = 0.01  # reset step size on improvement
        else:
            # Reduce step size on failure to improve
            step_size *= 0.99
            
        # Prevent step_size from becoming too small
        step_size = max(step_size, 1e-6)
    
    return (list(best_h), best_c5, n_points)
