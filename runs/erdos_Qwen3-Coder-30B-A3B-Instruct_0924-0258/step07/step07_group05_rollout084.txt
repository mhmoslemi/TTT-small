import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    pop_size = 50
    generations = 1000
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0.1, 0.9, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
    
    # GA iterations
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Elite selection
        elite = population[:elite_size]
        
        # Create new population
        new_pop = elite.copy()
        
        # Tournament selection and crossover
        while len(new_pop) < pop_size:
            # Tournament selection
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            parent1_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            parent2_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate_idx = np.random.choice(n_points, int(mutation_rate * n_points))
                for idx in mutate_idx:
                    child1[idx] += np.random.normal(0, 0.03)
                child1 = np.clip(child1, 0, 1)
                # Repair constraint
                child1 = child1 * (n_points/2) / np.sum(child1)
            
            if np.random.rand() < mutation_rate:
                mutate_idx = np.random.choice(n_points, int(mutation_rate * n_points))
                for idx in mutate_idx:
                    child2[idx] += np.random.normal(0, 0.03)
                child2 = np.clip(child2, 0, 1)
                # Repair constraint
                child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_pop.append(child1)
            if len(new_pop) < pop_size:
                new_pop.append(child2)
        
        population = new_pop
    
    # Convex relaxation for final refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample some k values for constraints
        k_vals = np.linspace(0.05, 1.95, min(20, n_points//4))
        for k_val in k_vals:
            k = int(k_val * n_points / 2)
            if k < n_points and k > 0:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
