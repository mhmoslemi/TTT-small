import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_h(n_points):
    x = np.linspace(0, 2, n_points)
    h_initial = 0.5 * (1 + np.sin(2 * np.pi * x))
    h_initial = np.clip(h_initial, 0, 1)
    normalization = (n_points / 2) / np.sum(h_initial)
    h_initial *= normalization
    return h_initial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = initialize_h(n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search + Simulated Annealing approach
    tabu_list = []
    tabu_size = 20
    max_iter = 5000
    T = 1.0
    alpha = 0.998
    stagnation_limit = 100
    stagnation_count = 0
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by perturbing a random subset
        num_perturb = max(1, int(0.15 * n_points))
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        
        candidate_h = current_h.copy()
        for i in indices:
            # Perturb by a random factor between 0.85 and 1.15
            perturbation_factor = np.random.uniform(0.85, 1.15)
            candidate_h[i] *= perturbation_factor
            
        # Clip values
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Normalize
        candidate_h = candidate_h * (n_points/2) / np.sum(candidate_h)
        
        # Evaluate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Check if in tabu list
        tabu_candidate = tuple(np.round(candidate_h, decimals=6))
        in_tabu = tabu_candidate in tabu_list
      
        # Accept condition
        if candidate_c5 < current_c5 or not in_tabu or np.random.random() < math.exp(-abs(candidate_c5 - current_c5) / T):
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            
            # Update tabu list
            if len(tabu_list) >= tabu_size:
                tabu_list.pop(0)
            tabu_list.append(tabu_candidate)
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Cool down
        T *= alpha
        
        # Check for stagnation
        if stagnation_count > stagnation_limit:
            break
    
    # Final refinement
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
