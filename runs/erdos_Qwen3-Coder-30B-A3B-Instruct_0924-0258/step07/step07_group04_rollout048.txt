import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe optimization with adaptive step size
    h = h_start.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points/2) / np.sum(h)
    
    gamma = 0.1
    max_iter = 2000
    stagnation_count = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlations
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(corr)
        max_val = corr[max_lag]
        
        # Gradient direction
        # We want to decrease the overlap, so direction = (1 - h) 
        # but we need to project to ensure we're moving toward a better solution
        s = (1 - h)
        # Project to gradient descent direction, but backed by a max lag correction
        # This makes a simple Frank-Wolfe correction: move in gradient direction with step gamma

        # Update h
        h_new = h - gamma * s
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            last_improvement = iteration
        else:
            h = h_new.copy()
            # Reduce step size if no improvement
            gamma = max(0.01, gamma * 0.99)
            
        # Periodic perturbation to escape local minima
        if iteration % 50 == 0 and iteration > 0:

            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                last_improvement = iteration
        
        # Check for stagnation
        if iteration - last_improvement > 100:
            break
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try multiple random starting points for SLSQP
        best_slsqp = best_c5

            # Small random perturbation
            h_pert = h_refined + np.random.normal(0, 0.01, n_points)

            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            
            res = minimize(objective, h_pert, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-12})
            if res.success:
                score = objective(res.x)
                if score < best_slsqp:
                    best_slsqp = score
                    best_slsqp_h = res.x.copy()
        
        if best_slsqp < best_c5:
            best_h = best_slsqp_h
            best_c5 = best_slsqp

    except Exception as e:
        pass
    
    return (list(best_h), best_c5, n_points)
