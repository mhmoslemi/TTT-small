import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with focus on max overlap
    max_iter = 10000
    step_size = 0.01
    decay_factor = 0.5
    min_step_size = 1e-6
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps and find the one with maximum value
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient for this shift (k = max_lag)
        grad = np.zeros(n_points)
        for i in range(n_points):
            k = max_lag
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1 - 2 * best_h[i] - 2 * best_h[j]
        
        # Update step and apply gradient descent  
        h_new = best_h - step_size * grad
        
        # Project to feasible region [0,1] and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check feasibility of new solution
        if np.sum(h_new) < 0.99 * (n_points / 2) or np.sum(h_new) > 1.01 * (n_points / 2):
            step_size *= decay_factor
            if step_size < min_step_size:
                break
            continue
            
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            step_size = min(0.05, step_size * 1.2)  # Increase step size on improvement
        else:
            step_size *= decay_factor  # Reduce step size on no improvement
            if step_size < min_step_size:
                break
    
    # Final local refinement with optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 2000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final checks
    if not (0 <= np.min(best_h) and np.max(best_h) <= 1):
        best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
