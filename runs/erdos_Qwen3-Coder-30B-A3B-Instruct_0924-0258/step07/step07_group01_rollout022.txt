import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 200
    mutation_rate = 0.3
    elite_size = population_size // 5  # Top 20% as elite
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0.2, 0.8, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    stagnation_count = 0
    for generation in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > 50:
            break
            
        # Select elite
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite[:]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if random.random() < crossover_rate and len(new_population) + 1 < population_size:
                # Tournament selection
                parent1 = tournament_selection(population, fitness, 3)
                parent2 = tournament_selection(population, fitness, 3)
                
                # Arithmetic crossover
                alpha = random.random()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
                
                # Mutation
                if random.random() < mutation_rate:
                    mutate_gene(child1, 0.1)
                if random.random() < mutation_rate:
                    mutate_gene(child2, 0.1)
                
                # Ensure constraints
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points/2) / np.sum(child1)
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points/2) / np.sum(child2)
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Random new individual
                h = np.random.uniform(0.2, 0.8, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                new_population.append(h)
        
        population = new_population
    
    # Local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Refine top solutions
    for i in range(min(5, len(population))):
        h_refined = population[i].copy()
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h_refined, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 200, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)

def tournament_selection(population, fitness, tournament_size):
    tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_index]

def mutate_gene(h, strength):
    for i in range(len(h)):
        if random.random() < 0.1:  # 10% chance to mutate each gene
            h[i] += np.random.normal(0, strength)
