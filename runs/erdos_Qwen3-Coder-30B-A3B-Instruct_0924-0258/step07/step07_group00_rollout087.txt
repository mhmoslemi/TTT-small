import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with adaptive learning rate
    h = h_start.copy()
    learning_rate = 0.01
    no_improvement_count = 0
    max_no_improvement = 50
    
    iterations = 0
    max_iterations = 1000
    
    while time() - start_time < budget_s * 0.95 and iterations < max_iterations:
        # Compute correlations for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_index = np.argmax(corr)
        max_lag = max_lag_index - (len(h) - 1)
        
        # Compute gradient
        gradient = np.zeros(len(h))
        for i in range(len(h)):
            j = (i + max_lag) % len(h)
            gradient[i] = 2 * (1 - h[j] - h[i])
        
        # Update
        h_new = h - learning_rate * gradient
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Adjust learning rate
        if no_improvement_count > 0:
            learning_rate *= 0.5
            if learning_rate < 1e-6:
                learning_rate = 1e-6
        else:
            learning_rate = 0.01
            
        h = h_new
        iterations += 1
        
        # Periodic refinement with SLSQP
        if iterations % 100 == 0 and iterations > 0:
            try:
                def objective(h_vec):
                    return evaluate_erdos_solution(h_vec, dx)
                
                def constraint_sum(h_vec):
                    return np.sum(h_vec) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(objective, h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100})
                
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                h = h_refined if 'h_refined' in locals() else h
            except:
                pass
    
    # Final SLSQP refinement
    try:
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000})
        
        if res.success:
            final_h = res.x
            final_c5 = objective(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
