import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with adaptive learning rate
    h = h_start.copy()
    eta = 0.01
    no_improvement_count = 0
    max_no_improvement = 200
    step_count = 0
    max_steps = 2000
    
    while time() - start_time < budget_s * 0.95 and step_count < max_steps:
        # Compute correlations for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        
        # Compute gradient for the max overlapping lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_k - (n_points - 1)) % n_points
            grad[i] = 2 * (1 - h[j] - h[i])
        
        # Update with gradient
        h_new = h - eta * grad
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new

            no_improvement_count = 0
            eta = min(0.05, eta * 1.1)  # Increase learning rate

        if step_count % 100 == 0:

                    return evaluate_erdos_solution(h, dx)

                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(objective, h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-9})

                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:

                pass

            return evaluate_erdos_solution(h, dx)

            return np.sum(h) - n_points / 2.0

        if res.success:

    except Exception:
