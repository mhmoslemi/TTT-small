import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        # Create a good starting point

        # Try to make a function that's mostly 1 for a short interval, 
        # then drops to 0, but smooth
        h_start = np.zeros(n_points)
        mid = n_points // 2
        # Create a smooth version at the center
        for i in range(n_points):
            # Smooth transition
            if i < mid:
                h_start[i] = 0.5 + 0.5 * np.sin(np.pi * i / mid)
            else:
                h_start[i] = 0.5 - 0.5 * np.sin(np.pi * (i - mid) / (n_points - mid))
        h_start = np.clip(h_start, 0, 1)
        
        # Normalize to correct sum
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a hybrid of genetic algorithm and local optimization
    # Genetic algorithm parameters
    pop_size = 100
    generations = 300
    elite_size = 10
    mutation_rate = 0.1
    crossover_rate = 0.7
    
    # Create initial population, ensuring sum constraint
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Random perturbation with better normalization
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Evaluation function for population
    def evaluate_population(pop):
        return [evaluate_erdos_solution(h, dx) for h in pop]
    
    # Repair function to maintain constraints
    def repair(h):
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        h = h * scale
        return h
    
    # Main optimization loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = evaluate_population(population)
        
        # Track best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        new_pop = sorted_pop[:elite_size]
        
        # Generate offspring
        while len(new_pop) < pop_size:
            # Selection (tournament)
            tourn_size = 3
            idx1 = random.sample(range(elite_size), 1)[0]
            idx2 = random.sample(range(elite_size), 1)[0] 
            parent1 = sorted_pop[idx1]
            parent2 = sorted_pop[idx2]
            
            # Crossover
            if random.random() < crossover_rate:
                alpha = random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
            
            # Mutation
            if random.random() < mutation_rate:
                for i in range(n_points):
                    if random.random() < 0.1:  # 10% of elements
                        child[i] += random.gauss(0, 0.05)
                
            # Repair
            child = repair(child)
            
            new_pop.append(child)
        
        # Replace population
        population = new_pop[:pop_size]
    
    # Final refinement with scipy minimize using L-BFGS-B on best solution
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        def constraint_bounds(h):
            return np.concatenate([h, 1 - h])
        
        # Constraints: sum(h) = n_points/2, 0 <= h <= 1 
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Optimize with better starting point (best from GA)
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                         options={'maxiter': 200, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_h = repair(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    # Final repair and validation
    best_h = repair(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
