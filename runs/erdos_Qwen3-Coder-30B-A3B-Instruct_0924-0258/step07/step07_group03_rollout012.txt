import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try convex relaxation approach as primary method
    try:
        # Define optimization variables
        h_var = cp.Variable(n_points)
        Z = cp.Variable((n_points, n_points), symmetric=True)
        
        # Objective: minimize maximum correlation
        t = cp.Variable()
        
        # Constraints for convex relaxation
        constraints = [
            cp.bmat([
                [t * np.ones((n_points, n_points)) + Z, Z],
                [Z, t * np.ones((n_points, n_points)) - Z]
            ]) >> 0,
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Convex optimization
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve with MOSEK or fallback
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, time_limit=0.8 * budget_s)
        except:
            try:
                prob.solve(solver=cp.ECOS, verbose=False, time_limit=0.8 * budget_s)
            except:
                prob.solve(solver=cp.SCS, verbose=False, time_limit=0.8 * budget_s)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_relaxed = h_var.value
            h_relaxed = np.maximum(0, np.minimum(1, h_relaxed))
            h_relaxed = h_relaxed * (n_points / 2) / np.sum(h_relaxed)
            c5_relaxed = evaluate_erdos_solution(h_relaxed, dx)
            if c5_relaxed < best_c5:
                best_h = h_relaxed
                best_c5 = c5_relaxed
                
    except Exception as e:
        pass

    # Local refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined

    except Exception as e:
        pass

    # Try a few more local refinements if time permits
    elapsed = time() - start_time
    if elapsed < 0.9 * budget_s:
        for _ in range(5):
            # Random restart from clipped noisy points
            h_noisy = best_h + np.random.normal(0, 0.01, n_points)
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = h_noisy * (n_points / 2) / np.sum(h_noisy)
            
            try:
                res = minimize(objective, h_noisy, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 500, 'ftol': 1e-10})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass

    return (list(best_h), best_c5, n_points)
