import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a more sophisticated initial guess
        x = np.linspace(0, 2, n_points)
        # Use piecewise constant function with peak near center
        h_start = np.zeros(n_points)
        # Place some mass near center and edges
        center_idx = n_points // 2
        h_start[center_idx-2:center_idx+3] = 0.8
        h_start[0:3] = 0.4
        h_start[-3:] = 0.4
        h_start = np.clip(h_start, 0, 1)
        # Normalize to sum = n_points/2
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary algorithm for global exploration
    pop_size = 100
    generations = 300
    mutation_rate = 0.2
    crossover_rate = 0.7
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Track hall of fame
    hall_of_fame = []
    for i in range(min(10, len(population))):
        hall_of_fame.append((population[i].copy(), evaluate_erdos_solution(population[i], dx)))
    hall_of_fame.sort(key=lambda x: x[1])
    
    # EA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update hall of fame
        for i in range(min(len(sorted_population), 10)):
            h = sorted_population[i]
            c5 = sorted_fitness[i]
            found = False
            for j, (h_fame, c5_fame) in enumerate(hall_of_fame):
                if np.allclose(h, h_fame) and abs(c5 - c5_fame) < 1e-8:
                    found = True
                    break
            if not found and len(hall_of_fame) < 10:
                hall_of_fame.append((h.copy(), c5))
                hall_of_fame.sort(key=lambda x: x[1])
            elif not found and c5 < hall_of_fame[-1][1]:
                hall_of_fame[-1] = (h.copy(), c5)
                hall_of_fame.sort(key=lambda x: x[1])
        
        # Early stopping check
        if gen > 0 and gen % 50 == 0:
            best_in_gen = sorted_fitness[0]
            if best_in_gen >= best_c5 - 1e-6:
                # No improvement in last 50 gens
                pass
                
        # Selection (tournament with size 3)
        def tournament_select():
            idxs = np.random.choice(pop_size, 3, replace=False)
            return sorted_population[np.argmin([fitness[i] for i in idxs])]
        
        # New population
        new_population = [sorted_population[0].copy()]  # Keep best
        
        while len(new_population) < pop_size:
            # Crossover
            if np.random.rand() < crossover_rate and len(new_population) < pop_size - 1:
                parent1 = tournament_select()
                parent2 = tournament_select()
                
                # Arithmetic crossover
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
                
                # Repair
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points / 2) / np.sum(child1)
                
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
                
                new_population.append(child1)
                if len(new_population) < pop_size:
                    new_population.append(child2)
            else:
                # Mutation
                parent = tournament_select()
                child = parent.copy()
                
                # Select random points to flip
                num_to_change = max(1, int(mutation_rate * n_points))
                indices = np.random.choice(n_points, num_to_change, replace=False)
                
                # Apply Gaussian mutation
                for i in indices:
                    noise = np.random.normal(0, 0.1)
                    child[i] += noise
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                
                new_population.append(child)
        
        # Update population
        population = new_population
    
    # Find the best solution from all generations
    for h in hall_of_fame:
        if h[1] < best_c5:
            best_h = h[0]
            best_c5 = h[1]
    
    # Final L-BFGS refinement using scipy
    try:
        def objective(h_v):
            h = h_v.copy()
            corr = np.correlate(h, 1 - h, mode='full')
            return -np.max(corr) * dx
        
        def constraint_sum(h_v):
            return np.sum(h_v) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        # Use best solution from EA as starting point
        h_init = best_h.copy()
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        result = minimize(
            objective, 
            h_init, 
            method='L-BFGS-B',
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        
        if result.success:
            h_refined = np.clip(result.x, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception as e:
        pass

    # Final normalization and verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
