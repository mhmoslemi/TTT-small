import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        # Periodic initialization with 5 cycles over [0,2]
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 0.4))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
        # Add small random perturbations
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multiresolution approach
    resolutions = [20, 40, 80]
    current_n = 20
    h_current = h_start.copy()
    
    # For each resolution
    for res_idx, target_n in enumerate(resolutions):
        # Interpolate if needed
        if target_n > current_n:
            x_orig = np.linspace(0, 2, current_n)
            x_new = np.linspace(0, 2, target_n)
            h_current = np.interp(x_new, x_orig, h_current)
            current_n = target_n
        elif target_n < current_n:
            # Downsample
            h_current = h_current[::current_n//target_n]
            current_n = target_n
        
        # Perturb to avoid local minima
        h_current += np.random.normal(0, 0.01, current_n)
        h_current = np.clip(h_current, 0, 1)
        normalization = (current_n / 2) / np.sum(h_current)
        h_current *= normalization
        
        # Main loop with simulated annealing approach
        max_iter = 100 if res_idx < len(resolutions) - 1 else 300
        temperature = 0.2
        temperature_decay = 0.98
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Select a few random points to perturb
            n_perturb = max(3, current_n // 10)
            indices = np.random.choice(current_n, size=n_perturb, replace=False)
            
            # Copy current solution and perturb
            h_candidate = h_current.copy()
            for i in indices:
                # Add perturbation in [0.05, 0.15] range, scaled by current value
                perturb = np.random.uniform(-0.1, 0.1) * h_candidate[i]
                h_candidate[i] += perturb
            h_candidate = np.clip(h_candidate, 0, 1)
            normalization = (current_n / 2) / np.sum(h_candidate)
            h_candidate *= normalization
            
            # Evaluate
            c5_candidate = evaluate_erdos_solution(h_candidate, 2.0 / current_n)
            
            # Accept or reject based on temperature
            if c5_candidate < best_c5:
                h_current = h_candidate.copy()
                best_c5 = c5_candidate
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
            elif np.random.rand() < np.exp(-(c5_candidate - best_c5) / temperature):
                h_current = h_candidate.copy()
            
            temperature *= temperature_decay
            
        dx = 2.0 / current_n
    
    # Final SLSQP refinement on the best solution found
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (current_n/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / current_n)
        
        def constraint_sum(h):
            return np.sum(h) - current_n / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(current_n)]
        
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, current_n)
