import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a structured approach: sine wave with low amplitude
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.2 * np.sin(np.pi * x)
        h_start = np.clip(h_start, 0, 1)
    
    # Normalize initial guess
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Lagrangian function definition
    def lagrangian(h, penalty_weight=1000.0):
        # Main objective
        overlap = evaluate_erdos_solution(h, dx)
        # Penalty for constraint violation
        constraint_violation = abs(constraint_sum(h))
        return overlap + penalty_weight * constraint_violation
    
    # DE parameters
    popsize = 50
    maxiter = 1000
    mutation_rate = 0.5
    crossover_rate = 0.9
    
    # DE sampling bounds
    lower_bounds = [0.0] * n_points
    upper_bounds = [1.0] * n_points
    
    # DE with adaptive parameters
    try:
        # Perform DE
        de_result = differential_evolution(
            lagrangian,
            bounds,
            strategy='best1bin',
            maxiter=maxiter,
            popsize=popsize,
            mutation=(0.5, 1.0),
            recombination=crossover_rate,
            seed=seed,
            disp=False,
            polish=False
        )
        
        if de_result.success:
            h_de = de_result.x.copy()
            # Ensure normalization
            h_de = h_de * (n_points/2) / np.sum(h_de)
            c5_de = evaluate_erdos_solution(h_de, dx)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
                
    except Exception:
        pass
    
    # Local refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds_slsqp = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds_slsqp, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10}, 
                       callback=lambda x: None if time() - start_time > budget_s * 0.95 else None)
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
