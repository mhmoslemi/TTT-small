import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with adaptive learning rate
    max_iter = 5000
    lr = 0.01
    patience = 50
    patience_counter = 0
    prev_c5 = best_c5 + 1.0
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Compute gradient estimation
        h_current = best_h.copy()
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        corr_full = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr_full)
        max_lag = max_idx - (n_points - 1)
        
        # Simple gradient approximation
        grad = np.zeros(n_points)
        for i in range(n_points):
            idx_shift = i + max_lag
            if 0 <= idx_shift < n_points:
                grad[i] += 1 - 2 * h_current[i]
        
        # Update
        h_new = h_current - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience_counter = 0
            lr = min(0.1, lr * 1.05)  # Adaptive learning rate
        else:
            patience_counter += 1
            if patience_counter >= patience:
                lr *= 0.5
                patience_counter = 0
                # Add noise to escape local minima
                noise = np.random.normal(0, 0.001, n_points)
                best_h = np.clip(best_h + noise, 0, 1)
                best_h = best_h * (n_points/2) / np.sum(best_h)
    
    # Final refinement using convex optimization
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Sample key lags for constraints
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        sample_lags = np.argsort(corr_full)[-15:]  # top 15 lags
        
        for k_idx in sample_lags:
            if k_idx < n_points:
                # Compute correlation for lag k
                corr_k = cp.sum(cp.multiply(h_opt[:n_points-k_idx], 1 - h_opt[k_idx:]))
                constraints.append(corr_k <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=2000)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
        
    except Exception:
        pass
    
    # Final verification and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
