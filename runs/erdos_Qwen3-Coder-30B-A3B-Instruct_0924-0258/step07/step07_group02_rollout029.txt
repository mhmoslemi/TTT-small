import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a balanced approach for initialization
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm parameters
    pop_size = 50
    max_iter = 200
    mutation_rate = 0.05
    crossover_rate = 0.7
    elite_size = 5
    tournament_size = 5
    max_time = budget_s * 0.95
    
    # Initialize population
    population = []
    fitness = []
    
    # Create initial population
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Add some diversity with perturbations
            noise = np.random.normal(0, 0.05, n_points)
            h = h_start + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
    
    # Sort by fitness
    sorted_indices = np.argsort(fitness)
    population = [population[i] for i in sorted_indices]
    fitness = [fitness[i] for i in sorted_indices]
    
    if fitness[0] < best_c5:
        best_h = population[0].copy()
        best_c5 = fitness[0]
    
    # Evolutionary loop
    for gen in range(max_iter):
        if time() - start_time > max_time:
            break
            
        # Selection: tournament
        new_population = []
        elite = population[:elite_size]
        new_population.extend(elite)
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Create offspring via crossover with another parent
            if np.random.rand() < crossover_rate:
                idx2 = np.random.choice(pop_size, tournament_size)
                idx2 = idx2[np.argmin([fitness[i] for i in idx2])]
                # Blend crossover
                alpha = np.random.rand()
                child = alpha * population[winner_idx] + (1 - alpha) * population[idx2]
            else:
                child = population[winner_idx].copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Adaptive mutation with decreasing variance
                mut_strength = max(0.005, mutation_rate * (1 - gen / max_iter))
                noise = np.random.normal(0, mut_strength, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            
            # Repair
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        # Evaluate new population
        new_fitness = []
        for h in new_population:
            c5 = evaluate_erdos_solution(h, dx)
            new_fitness.append(c5)
        
        # Replace population
        population = new_population
        fitness = new_fitness
        
        # Update best
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_h = population[min_idx].copy()
            best_c5 = fitness[min_idx]
        
        # Adapt mutation rate
        if gen > 0 and gen % 20 == 0:
            mutation_rate *= 0.95
    
    # Final refinement using scipy
    def obj_func(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Sample top solutions for refinement
    top_candidates = population[:10]
    best_refined = best_h.copy()
    best_refined_c5 = best_c5
    
    for h_refine in top_candidates:
        if time() - start_time > max_time:
            break
            
        # Use L-BFGS-B for refinement
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            result = minimize(obj_func, h_refine, method='L-BFGS-B', 
                             bounds=bounds, options={'maxiter': 100})
            if result.success:
                refined_h = result.x
                refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
                refined_c5 = evaluate_erdos_solution(refined_h, dx)
                if refined_c5 < best_refined_c5:
                    best_refined = refined_h.copy()
                    best_refined_c5 = refined_c5
        except:
            pass
    
    # Final check and return
    if best_refined_c5 < best_c5:
        best_h = best_refined
        best_c5 = best_refined_c5
    
    return (list(best_h), best_c5, n_points)
