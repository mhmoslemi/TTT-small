import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based iterative descent
    max_iter = 10000
    learning_rate = 0.1
    patience = 100
    no_improve_count = 0
    
    h = h_start.copy()
    target_sum = n_points / 2.0
    
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        max_k_idx = np.argmax(corr)
        max_corr = corr[max_k_idx]
        
        # Compute gradient for the worst-case lag
        # Correlation of h with (1-h) at lag k_max
        # dC/dh[i] = 1 - h[i-k_max] for i where the term is non-zero
        grad = np.zeros(n_points)
        if max_k_idx < n_points:
            # Forward correlation
            for i in range(max_k_idx, n_points):
                grad[i] += 1.0
                if i - max_k_idx >= 0:
                    grad[i] -= 1.0
        else:
            # Backward correlation
            k = max_k_idx - n_points + 1
            for i in range(n_points):
                if i + k < n_points:
                    grad[i] += 1.0
                if i >= k:
                    grad[i] -= 1.0
        
        # Apply learning rate and update
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to correct sum
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * target_sum / current_sum
            
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        if no_improve_count > patience:
            learning_rate *= 0.5
            no_improve_count = 0
            if learning_rate < 1e-6:
                break
                
        h = h_new
    
    # Phase 2: Final refinement with L-BFGS-B
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='L-BFGS-B', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Fallback: random restarts
    restarts = 5
    for i in range(restarts):
        if time() - start_time > budget_s * 0.95:
            break
        h_restart = np.random.uniform(0, 1, n_points)
        h_restart = h_restart * target_sum / np.sum(h_restart)
        c5_restart = evaluate_erdos_solution(h_restart, dx)
        if c5_restart < best_c5:
            best_c5 = c5_restart
            best_h = h_restart.copy()
    
    return (list(best_h), best_c5, n_points)
