import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient Descent with adaptive learning rate
    h_current = h_start.copy()
    learning_rate = 0.1
    max_iter = 1000
    patience = 0
    min_improvement = 1e-8
    last_loss = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute overlap and identify worst lag
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        current_c5 = corr[max_lag_idx] * dx
        
        # Compute gradient with respect to h
        # This is an approximation; actual gradient would require careful indexing
        # We take a simpler approach: compute the effect of small changes at each point
        gradient = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad_at_i = (corr_plus[max_lag_idx] - corr[max_lag_idx]) / 1e-5
            gradient[i] = grad_at_i
        
        # Apply gradient descent with learning rate
        h_current = h_current - learning_rate * gradient
        h_current = np.clip(h_current, 0, 1)

    return (list(best_h), best_c5, n_points)
