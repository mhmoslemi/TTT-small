import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Basis function approach
    m = 10  # number of basis functions
    
    # Initialize basis functions (piecewise linear)
    basis = np.zeros((n_points, m))
    x_points = np.linspace(0, 2, n_points)
    
    # Create triangular basis functions
    for i in range(m):
        center = i * 2.0 / (m - 1) if m > 1 else 1.0
        width = 0.5
        basis[:, i] = np.maximum(0, 1 - np.abs(x_points - center) / width)
    
    # Normalize basis functions
    for i in range(m):
        if np.sum(basis[:, i]) > 0:
            basis[:, i] /= np.sum(basis[:, i])
    
    # Start with coefficients that reproduce h_start
    try:
        h_start_normalized = h_start * (n_points / 2) / np.sum(h_start)
        coeffs = np.linalg.lstsq(basis, h_start_normalized, rcond=None)[0]
    except:
        coeffs = np.ones(m) / m
    
    # Optimization parameters
    max_iter = 5000
    trust_radius = 0.1
    patience = 50
    patience_count = 0
    perturbation_magnitude = 0.05
    
    def evaluate_objective(coeffs):
        h = basis @ coeffs
        h = np.clip(h, 0, 1)
        # Renormalize to satisfy constraint
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    def project_to_feasible(coeffs):
        h = basis @ coeffs
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        new_coeffs = np.linalg.lstsq(basis, h, rcond=None)[0]
        return new_coeffs
    
    def compute_gradient(coeffs):
        h = basis @ coeffs
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Approximate gradient via finite differences
        grad = np.zeros(m)
        epsilon = 1e-6
        for i in range(m):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += epsilon
            h_plus = basis @ coeffs_plus
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            coeffs_minus = coeffs.copy()
            coeffs_minus[i] -= epsilon
            h_minus = basis @ coeffs_minus
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            grad[i] = (evaluate_erdos_solution(h_plus, dx) - 
                       evaluate_erdos_solution(h_minus, dx)) / (2 * epsilon)
        return grad
    
    # Trust region optimization
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute objective and gradient
        current_c5 = evaluate_objective(coeffs)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = basis @ coeffs
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            patience_count = 0
        else:
            patience_count += 1
            
        if patience_count > patience:
            # Perturb coefficients
            coeffs += np.random.normal(0, perturbation_magnitude, m)
            perturbation_magnitude *= 0.95
            patience_count = 0
        
        grad = compute_gradient(coeffs)
        
        # Update using gradient descent with trust region
        try:
            # Try small steps
            step = -grad * 0.01
            new_coeffs = coeffs + step
            
            # Project to feasible space
            new_coeffs = project_to_feasible(new_coeffs)
            
            new_c5 = evaluate_objective(new_coeffs)
            
            if new_c5 < current_c5:
                coeffs = new_coeffs
                trust_radius *= 2  # Increase trust region if successful
            else:
                # Try smaller step
                step = -grad * 0.001
                new_coeffs = coeffs + step
                new_coeffs = project_to_feasible(new_coeffs)
                new_c5 = evaluate_objective(new_coeffs)
                if new_c5 < current_c5:
                    coeffs = new_coeffs
                    trust_radius *= 2
                else:
                    trust_radius = max(0.001, trust_radius * 0.5)
        except:
            # Fail safe: smaller random perturbation
            coeffs += np.random.normal(0, 0.01, m)
            coeffs = project_to_feasible(coeffs)
    
    # Final refinement with direct method
    try:
        def objective(c):
            h = basis @ c
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(c):
            h = basis @ c
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            return np.sum(h) - n_points / 2.0
            
        c0 = coeffs.copy()
        bounds = [(None, None) for _ in range(m)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        res = minimize(objective, c0, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success and res.fun < best_c5:
            raw_h = basis @ res.x
            raw_h = np.clip(raw_h, 0, 1)
            final_h = raw_h * (n_points / 2) / np.sum(raw_h)
            best_c5 = evaluate_erdos_solution(final_h, dx)
            best_h = final_h
    except Exception:
        pass
    
    # Final validation
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    
    return (list(final_h), best_c5, n_points)
