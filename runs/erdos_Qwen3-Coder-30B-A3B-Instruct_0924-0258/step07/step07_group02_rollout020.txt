import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing approach
    T = 1.0
    T_min = 1e-6
    alpha = 0.99
    max_iter = 1000
    iter_count = 0
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    while T > T_min and time() - start_time < budget_s * 0.95:
        iter_count += 1
        # Propose new solution
        h_new = current_h + np.random.normal(0, 0.05, n_points)
        # Clip to feasible range
        h_new = np.clip(h_new, 0, 1)
        # Normalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        delta = c5_new - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = h_new.copy()
            current_c5 = c5_new
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down
        T *= alpha
        
        # Occasionally do gradient descent step for local refinement
        if iter_count % 50 == 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add constraint for largest overlap so far
                corr = np.correlate(best_h, 1 - best_h, mode='full')
                max_k_idx = np.argmax(corr)
                max_lag = max_k_idx - (n_points - 1)
                
                # Add constraint on the largest overlap
                if max_lag < 0:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points+max_lag], 1 - h_opt[-max_lag:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[max_lag:], 1 - h_opt[:n_points-max_lag]))
                
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
                
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
    
    # Final refinement with local optimization
    try:
        # Use a subset of the best solution as starting point for more precise optimization
        h_subset = np.array(best_h)
        # Refine using scipy minimize on a restricted set of variables
        def obj(h_var):
            h = h_var.copy()
            h = h * (n_points/2) / np.sum(h)
            return evaluate_erdos_solution(h, dx)
        
        res = minimize(obj, h_subset, method='L-BFGS-B', bounds=[(0,1)]*n_points)
        if res.success:
            h_final = res.x
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final verification and cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
