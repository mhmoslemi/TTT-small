import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a structured approach - sine wave with small amplitude
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.2 * np.sin(np.pi * x)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    # Normalize to correct integral
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds for differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint for sum
    def constraint_sum(x):
        return np.sum(x) - n_points / 2.0
    
    # Wrapper for differential evolution
    def objective_de(x):
        # Normalize to sum constraint
        x_norm = x * (n_points/2) / np.sum(x)
        return evaluate_erdos_solution(x_norm, dx)
    
    # Use differential evolution for global search
    de_options = {'maxiter': 1000, 'popsize': 50, 'mutation': (0.5, 1), 'recombination': 0.9, 'strategy': 'best1bin'}
    try:
        result_de = differential_evolution(
            objective_de,
            bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            **de_options,
            seed=seed,
            maxtime=budget_s * 0.8
        )
        if result_de.success:
            h_de = result_de.x * (n_points/2) / np.sum(result_de.x)
            c5_de = evaluate_erdos_solution(h_de, dx)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    except Exception:
        pass
    
    # Local refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective_slsqp(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum_slsqp(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum_slsqp}
        bounds_slsqp = [(0, 1) for _ in range(n_points)]
        res = minimize(objective_slsqp, h_refined, method='SLSQP', 
                       bounds=bounds_slsqp, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective_slsqp(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
