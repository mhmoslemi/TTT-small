import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a structured function: a sin wave modulated to have sum=1
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)  # Ensure positive
        h_start = np.clip(h_start, 0, 1)
        
        # Normalize to unit integral
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Set up bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Function to constrain the sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Lagrangian implementation with barrier
    def objective_with_barrier(h, barrier_strength=1e3):
        # Penalty for violating sum constraint
        penalty = barrier_strength * (constraint_sum(h))**2
        # The actual objective
        overlap = evaluate_erdos_solution(h, dx)
        return overlap + penalty
    
    # Differential Evolution optimization phase
    de_options = {
        'maxiter': 1000,
        'popsize': 50,
        'recombination': 0.9,
        'mutation': (0.5, 1.0),
        'disp': False,
        'seed': seed,
    }
    
    # Set up DE with proper bounds
    de_result = differential_evolution(
        objective_with_barrier,
        bounds,
        args=(1e3,),  # Barrier strength
        **de_options,
        callback=lambda x, convergence: None  # Placeholder
    )
    
    # Refine with SLSQP if we haven't timed out
    if time() - start_time < budget_s * 0.95:
        try:
            refined_result = minimize(
                objective_with_barrier,
                de_result.x,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 1000, 'ftol': 1e-12}
            )
            if refined_result.success:
                h_final = refined_result.x
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    # Ensure final validation
    if time() - start_time > budget_s * 0.95:
        return (list(best_h), best_c5, n_points)
    
    # Final validation checks
    h_valid = np.clip(best_h, 0, 1)
    h_valid = h_valid * (n_points/2) / np.sum(h_valid)
    final_c5 = evaluate_erdos_solution(h_valid, dx)
    
    return (list(h_valid), final_c5, n_points)
