import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 100
    max_iter = 250
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_count = 10
    
    # Initialize population
    population = []
    fitness = []
    for i in range(pop_size):
        if i == 0:
            # Start with best solution
            h = best_h.copy()
        else:
            # Generate random feasible solution
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Tracks best solutions
    hall_of_fame = [(best_h.copy(), best_c5)]
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        # Update hall of fame
        for i in range(min(elite_count, len(sorted_pop))):
            if sorted_fit[i] < hall_of_fame[-1][1]:
                hall_of_fame.append((sorted_pop[i].copy(), sorted_fit[i]))
                hall_of_fame = sorted(hall_of_fame, key=lambda x: x[1])[:elite_count]
        
        # Selection
        selected_indices = []
        for _ in range(pop_size):
            # Tournament selection
            tournament_size = 3
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected_indices.append(winner_idx)
        
        # Crossover and Mutation
        new_population = []
        new_fitness = []
        
        # Elitism
        for i in range(elite_count):
            if i < len(sorted_pop):
                new_population.append(sorted_pop[i].copy())
                new_fitness.append(sorted_fit[i])
        
        # Generate offspring
        for _ in range(pop_size - elite_count):
            if np.random.rand() < crossover_rate:
                # Select parents
                p1_idx = selected_indices[np.random.randint(0, pop_size)]
                p2_idx = selected_indices[np.random.randint(0, pop_size)]
                p1 = population[p1_idx]
                p2 = population[p2_idx]
                
                # Arithmetic crossover
                alpha = np.random.rand()
                offspring = alpha * p1 + (1 - alpha) * p2
                
                # Mutation
                if np.random.rand() < mutation_rate:
                    # Gaussian mutation
                    std = 0.1 * (1.0 - iter_num/max_iter)
                    noise = np.random.normal(0, std, n_points)
                    offspring = offspring + noise
                else:
                    # Bit-flip for diversity
                    flip_indices = np.random.choice(n_points, int(0.1 * n_points), replace=False)
                    for idx in flip_indices:
                        offspring[idx] = 1 - offspring[idx]
                
                # Repair and normalize
                offspring = np.clip(offspring, 0, 1)
                offspring = offspring * (n_points/2) / np.sum(offspring)
            else:
                # Random mutation
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                offspring = h
            
            c5 = evaluate_erdos_solution(offspring, dx)
            new_population.append(offspring)
            new_fitness.append(c5)
            
            if c5 < best_c5:
                best_h = offspring.copy()
                best_c5 = c5
        
        population = new_population
        fitness = new_fitness
    
    # Final refinement with trust-constr
    try:
        # Apply trust-constr on top solutions
        ranked_solutions = sorted([(population[i], fitness[i]) for i in range(pop_size)], key=lambda x: x[1])[:5]
        for h_candidate, _ in ranked_solutions:
            # Use scipy for final refinement to fine-tune
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint(h):
                return np.sum(h) - n_points/2
            
            constraints = [{'type': 'eq', 'fun': constraint}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            result = minimize(obj, h_candidate, method='trust-constr', bounds=bounds, constraints=constraints, 
                             options={'maxiter': 50, 'disp': False})
            if result.success:
                c5_refined = obj(result.x)
                if c5_refined < best_c5:
                    best_h = result.x.copy()
                    best_c5 = c5_refined
    except:
        pass
    
    # Final normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
