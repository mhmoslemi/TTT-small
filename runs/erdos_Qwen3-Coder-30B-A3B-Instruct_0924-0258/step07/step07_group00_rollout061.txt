import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent approach with adaptive learning rate
    h = h_start.copy()
    learning_rate = 0.01
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Main optimization loop
    iteration = 0
    while time() - start_time < budget_s * 0.95:
        # Compute overlap for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h) - 1)
        
        # Compute gradient for the worst-case lag
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + max_lag) % len(h)
            gradient[i] = 2 * (1 - h[j] - h[i])
        
        # Update step
        h_new = h - learning_rate * gradient
        
        # Enforce constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check if new solution improves
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
            # Reset learning rate when improving
            learning_rate = 0.01
        else:
            no_improvement_count += 1
            # Reduce learning rate if no progress
            learning_rate *= 0.95
            if learning_rate < 1e-6:
                learning_rate = 1e-6
                
        # Update current solution
        h = h_new
        
        # Reset learning rate periodically
        if iteration % 50 == 0 and iteration > 0:
            learning_rate = 0.01
            
        # Early termination if no improvement
        if no_improvement_count >= max_no_improvement:
            break
            
        iteration += 1
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
