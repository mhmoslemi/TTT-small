import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T0 = 0.1
    cooling_rate = 0.001
    perturbation_fraction = 0.1
    max_iter = 1000
    stagnation_limit = 100
    
    h_current = h_start.copy()
    c5_current = best_c5
    
    # Keep track of best solution found so far
    best_h = h_current.copy()
    best_c5 = c5_current
    
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Time-based cooling schedule
        t = iteration / max_iter
        T = T0 * np.exp(-cooling_rate * t)
        
        # Select random indices to perturb
        n_perturb = max(1, int(n_points * perturbation_fraction))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Create new candidate solution
        h_new = h_current.copy()
        for i in indices:
            h_new[i] += np.random.uniform(-0.1, 0.1)
            
        # Clip and re-scale to maintain constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            stagnation_count = 0
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(c5_new - c5_current) / T):
                h_current = h_new
                c5_current = c5_new
                stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Update best solution
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        # Early exit if no improvement
        if stagnation_count > stagnation_limit:
            break
    
    # Final local optimization with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
