import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    step_size = 0.01
    max_iter = 10000
    iter_count = 0
    
    # Gradient-based optimization with focus on max overlap lag
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all correlations for all possible lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Get the maximum overlap lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Compute incremental gradient for the max overlap lag
        # correlation = sum h[i] * (1 - h[i + max_lag])
        # Grad = 1 - 2*h[i + max_lag]
        # But we actually need to compute the gradient for h[i] wrt overlap at lag max_lag
        # For i in range(n_points - abs(max_lag)):
        #   overlap contribute from h[i]: h[i] * (1 - h[i + max_lag]) 
        #   -> d/dh[i] = 1 - h[i + max_lag]
        #   -> d/dh[i + max_lag] = -h[i]
        
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += 1 - h_current[i + max_lag]
                if i + max_lag < n_points:
                    gradient[i + max_lag] -= h_current[i]
        else:
            for i in range(n_points + max_lag):
                gradient[i] += 1 - h_current[i - max_lag]
                if i - max_lag >= 0:
                    gradient[i - max_lag] -= h_current[i]
        
        # Gradient descent step
        h_new = h_current - step_size * gradient
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new C5 value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept step if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # If not improving, reduce step size
        if c5_new >= evaluate_erdos_solution(h_current, dx):
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            h_current = h_new
            step_size *= 1.05  # Increase step size slightly
        
        iter_count += 1
    
    return (list(best_h), best_c5, n_points)
