import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # ALM parameters
    max_iter = 5000
    lambda_lag = 0.0
    mu_penalty = 100.0
    alpha = 0.1
    tol = 1e-6
    tr_radius = 0.1
    tr_radius_max = 1.0
    tr_radius_min = 1e-6
    
    def compute_gradient(h, k_active):
        """Compute gradient of the overlapping integral with respect to given k"""
        n = len(h)
        grad = np.zeros(n)
        
        for k in k_active:
            if k < n:
                # Gradient contribution from h_i (1 - h_{i+k}) 
                # d/dh_i [h_i (1 - h_{i+k})] = 1 - h_{i+k} if i < n-k
                if k == 0:
                    grad += 1 - h
                else:
                    for i in range(n-k):
                        grad[i] += 1 - h[i+k]
        
        return grad / n
    
    def compute_active_k(h):
        """Find the lag k that gives maximum overlap"""
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        return max_k
    
    # Main ALM loop
    prev_c5 = float('inf')
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlap and active k
        current_c5 = evaluate_erdos_solution(h_current, dx)
        k_active = [compute_active_k(h_current)]
        
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            
        # Compute gradient at current h
        grad = compute_gradient(h_current, k_active)
        
        # Augmented Lagrangian update
        lag_grad = grad + lambda_lag * np.ones(n_points) + mu_penalty * (h_current - 1.0)
        
        # Trust-region step selection
        step = -alpha * lag_grad
        
        # Apply step and project back to feasible set
        h_new = h_current + step
        h_new = np.clip(h_new, 0, 1)
        
        # Must scale to satisfy constraint exactly
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new point
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject step
        if new_c5 < current_c5:
            h_current = h_new.copy()
            alpha *= 1.1  # Increase step size
            if alpha > tr_radius_max:
                alpha = tr_radius_max
        else:
            alpha *= 0.5  # Decrease step size
            if alpha < tr_radius_min:
                alpha = tr_radius_min
                
        # Update Lagrange multiplier
        lambda_lag += mu_penalty * (np.sum(h_current) - n_points/2)
        
        # Update penalty parameter
        if abs(np.sum(h_current) - n_points/2) > 1e-3:
            mu_penalty *= 2
        else:
            mu_penalty *= 0.9
            
        # Convergence check
        if abs(current_c5 - prev_c5) < tol:
            break
        prev_c5 = current_c5
    
    # Final polishing with local optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    def projected_h(h):
        h = np.clip(h, 0, 1)
        return h * (n_points/2) / np.sum(h)
    
    h_final = projected_h(best_h)
    try:
        res = minimize(objective, h_final, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            final_c5 = objective(h_final)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except Exception:
        pass
    
    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
