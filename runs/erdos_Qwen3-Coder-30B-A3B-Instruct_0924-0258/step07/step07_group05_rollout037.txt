import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with trust region
    max_iter = 5000
    convergence_tol = 1e-6
    alpha = 0.1
    mu = 100.0
    trust_region = 0.2
    
    h_current = best_h.copy()
    prev_c5 = best_c5
    
    def compute_gradient(h):
        # Compute all overlap integrals to find active constraint
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Gradient of the active overlap (generalized)
        grad = np.zeros_like(h)
        k = max_idx - (len(h) - 1)
        
        # Forward pass
        if k >= 0:
            for i in range(len(h) - k):
                grad[i] += 1 - h[i + k]
        else:
            k = -k
            for i in range(len(h) - k):
                grad[i + k] += -h[i]
        
        return grad / len(h)
    
    def project_and_scale(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum) < 1e-10:
            h = np.ones_like(h) * 0.5
        else:
            h = h * (n_points / 2.0) / h_sum
        return h
    
    def backtracking_line_search(h_old, grad, alpha, c5_old):
        c = 0.5
        rho = 0.5
        max_iter = 15
        
        for _ in range(max_iter):
            h_new = h_old - alpha * grad
            h_new = project_and_scale(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < c5_old:
                return h_new, c5_new, alpha
            alpha *= rho
            
        return h_old, c5_old, alpha
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current gradient
        grad = compute_gradient(h_current)
        
        # Handle numerical errors
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
        
        # Trust region step
        step_size = min(alpha, trust_region)
        h_candidate = h_current - step_size * grad
        
        # Line search
        h_new, c5_new, alpha = backtracking_line_search(
            h_current, grad, alpha, evaluate_erdos_solution(h_current, dx)
        )
        
        # Update trust region
        if c5_new < best_c5:
            h_current = h_new
            best_c5 = c5_new
            trust_region = min(trust_region * 1.1, 0.5)
            if abs(prev_c5 - c5_new) < convergence_tol:
                pass  # continue with same trust region
        else:
            trust_region = max(trust_region * 0.5, 0.01)  # Reduce trust region
            
        prev_c5 = c5_new

    # Final polishing with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        h_final = project_and_scale(h_current)
        res = minimize(objective, h_final, method='SLSQP',

        if res.success:
            h_final = res.x
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final check
    best_h = project_and_scale(best_h)
    actual_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), actual_c5, n_points)
