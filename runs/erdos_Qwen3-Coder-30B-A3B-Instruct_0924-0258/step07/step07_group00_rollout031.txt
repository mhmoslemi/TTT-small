import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization approach
    max_iter = 1000
    step_size = 0.01
    patience = 50
    no_improvement_count = 0
    
    h = h_start.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps for each lag
        corr = np.correlate(h, 1 - h, mode='full')
        # Extract only the valid overlaps (not the full correlation)
        # For lag k, we get indices from n_points-1 to 2*n_points-2
        valid_corr = corr[n_points-1:2*n_points-1]
        max_lag_idx = np.argmax(valid_corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Calculate gradient for l_max
        # Gradient of overlap at lag max_lag with respect to h[i]
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_lag) % n_points
            gradient[i] = 2 * (1 - h[j] - h[i]) / n_points
            
        # Update h using gradient descent
        h_new = h - step_size * gradient
        
        # Ensure feasibility: clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject update
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Reduce step size if no improvement
            if no_improvement_count > 10:
                step_size *= 0.5
                no_improvement_count = 0
                
        # Early termination if no improvement for many iterations
        if no_improvement_count >= patience:
            break
            
        # Occasionally run a local optimization to refine
        if iteration % 50 == 0 and iteration > 0:
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(objective, h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-9})
                
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass

    return (list(h), best_c5, n_points)
