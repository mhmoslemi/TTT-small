import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Basis function construction - triangular pulses
    n_basis = 10
    basis_support = np.linspace(0, 2, n_basis + 1)
    basis_functions = []
    
    for i in range(n_basis):
        x = np.linspace(0, 2, n_points)
        # Create triangular basis function
        basis = np.zeros(n_points)
        left = basis_support[i]
        right = basis_support[i+1]
        mid = (left + right) / 2
        for j in range(n_points):
            x_pos = j * dx
            if left <= x_pos <= right:
                if x_pos <= mid:
                    basis[j] = (x_pos - left) / (mid - left)
                else:
                    basis[j] = (right - x_pos) / (right - mid)
        basis_functions.append(basis)
    
    basis_matrix = np.array(basis_functions).T
    
    # Initialize coefficients
    coeffs = np.random.rand(n_basis) * 0.5
    h_current = basis_matrix @ coeffs
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    c5_current = evaluate_erdos_solution(h_current, dx)
    
    if c5_current < best_c5:
        best_h = h_current.copy()
        best_c5 = c5_current

    # Optimization parameters
    max_iter = 5000
    trust_radius = 0.1
    patience = 0
    patience_limit = 50
    perturbation_magnitude = 0.05
    cooling_factor = 0.5
    
    def compute_gradient(coeffs, dx):
        h_eval = basis_matrix @ coeffs
        h_eval = np.clip(h_eval, 0, 1)
        h_eval = h_eval * (n_points/2) / np.sum(h_eval)
        
        # Compute approximate gradient using finite differences
        grad = np.zeros(n_basis)
        epsilon = 1e-6
        for i in range(n_basis):
            coeffs_pos = coeffs.copy()
            coeffs_pos[i] += epsilon
            h_pos = basis_matrix @ coeffs_pos
            h_pos = np.clip(h_pos, 0, 1)
            h_pos = h_pos * (n_points/2) / np.sum(h_pos)
            corr_pos = np.correlate(h_pos, 1 - h_pos, mode='full')
            c5_pos = np.max(corr_pos) * dx
            
            coeffs_neg = coeffs.copy()
            coeffs_neg[i] -= epsilon
            h_neg = basis_matrix @ coeffs_neg
            h_neg = np.clip(h_neg, 0, 1)
            h_neg = h_neg * (n_points/2) / np.sum(h_neg)
            corr_neg = np.correlate(h_neg, 1 - h_neg, mode='full')
            c5_neg = np.max(corr_neg) * dx
            
            grad[i] = (c5_pos - c5_neg) / (2 * epsilon)
        return grad
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        try:
            grad = compute_gradient(coeffs, dx)
            h_eval = basis_matrix @ coeffs
            h_eval = np.clip(h_eval, 0, 1)
            h_eval = h_eval * (n_points/2) / np.sum(h_eval)
            c5_eval = evaluate_erdos_solution(h_eval, dx)
            
            # Compute trust region step
            step = -grad * trust_radius
            
            # Ensure gradient is meaningful
            if np.linalg.norm(grad) < 1e-10:
                if patience > patience_limit:
                    # Apply perturbation
                    coeffs += np.random.normal(0, perturbation_magnitude, n_basis)
                    patience = 0
                    perturbation_magnitude *= cooling_factor
                else:
                    patience += 1
            else:
                patience = 0
                
            # Update coefficients
            coeffs_new = coeffs + step
            
            h_new = basis_matrix @ coeffs_new
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                
            if c5_new < c5_eval:
                coeffs = coeffs_new
                trust_radius *= 2  # Increase trust radius on success
            else:
                trust_radius /= 2  # Decrease trust radius on failure
                if trust_radius < 1e-6:
                    trust_radius = 1e-6
                    coeffs = coeffs_new
        except Exception:
            pass

    # Final refinement with local optimization if time allows
    if time() - start_time < budget_s * 0.9:
        try:
            def objective(c):
                h_test = basis_matrix @ c
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points/2) / np.sum(h_test)
                return evaluate_erdos_solution(h_test, dx)
            
            def constraint_sum(c):
                h_test = basis_matrix @ c
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points/2) / np.sum(h_test)
                return np.sum(h_test) - n_points / 2.0
            
            # If we're far from the target or have more time, try final optimization
            bounds = [(-100, 100) for _ in range(n_basis)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            res = minimize(objective, coeffs, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                h_refined = basis_matrix @ res.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
