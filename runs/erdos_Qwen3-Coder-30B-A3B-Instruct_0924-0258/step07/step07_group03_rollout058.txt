import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient Descent with Perturbations
    max_iter = 1000
    lr = 0.1
    perturb_freq = 50
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    
    last_improvement = 0
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        # Compute gradient approximation
        # For lag k, the contribution to gradient of h[i] is (1 - 2*h[i]) * indicator
        # This is simplified for practical computation - think of this as gradient descent
        # on the maximum correlation value
        h_grad = np.zeros_like(h)
        
        # Get the k that gives the maximum overlap
        # For simplicity in gradient estimation, we construct a small gradient
        # approximate using shifted versions
        # Let's compute central differences
        eps = 1e-4
        h_plus = h.copy()
        h_minus = h.copy()
        
        # A smarter approach: identify which indices contribute to maximum overlap
        # and hence compute gradient accordingly
        
        # Let's compute what happens at the peak lag:
        # A full correlation at lag k is sum over i of h[i] * (1 - h[i+k])
        # on the valid range
        # Gradient would be d/dh[i] of that sum = (1 - h[i+k]) when i+k is valid
        # and -h[i+k] when i+k is valid, for terms in correlation
        
        # Better approach:
        # The gradient of overlap with respect to h[i] is
        # Sum_{k} (1 - h[i+k]) * delta_{i,j} where j=i+k
        # This can be computed more directly for the peak lag
        
        # For peak lag k_max, what is the gradient?
        # (1 - h[i]) for i+k_max in [0, n_points-1] 
        # (1 - h[i+k_max]) for i+k_max in [0, n_points-1]
        # Note: we probably should discretize and compute the accurate gradient
        
        # Smart way - compute for all terms in correlation, then examine max term 
        # and its gradient
        shifted_h = np.zeros_like(h)
        shifted_h[1:] = h[:-1]
        # Actually compute gradients correctly, this is one approach:
        # Gradients for each element are partial derivatives of the maximum overlap
        
        # We will use a direct gradient estimation via small shifts:
        # Compute the gradient only for the terms that are in the peak lag
        # For h[i], the gradient contribution to max overlap comes from:
        # For each k: if i+k exists, we look at 1 - h[i+k] if k corresponds to max
        
        # We simplify this by gradient estimation using trial shifts
        # For the maximum overlap lag, we compute a gradient using finite differences
        # but in a more precise way:
        # When using convolution corr[i] = sum_{j} h[j] * (1 - h[i - j])
        # For fixed i (peak lag), we have the actual terms.
        # Gradient: d/correlation_dh[i] depends on terms of the convolution
        
        # Compute gradient more exactly:
        grad = np.zeros_like(h)
        k_max = max_lag_idx - (n_points - 1)
        for i in range(n_points):
            j_start = max(0, i + k_max)
            j_end = min(n_points, i + k_max + 1)  
            if j_start < j_end:
                # Actually for the conv: corr[j] = sum_{k} h[k] * (1 - h[j-k])

                for k in range(n_points):
                    j = k - (k_max - i)  
                    if 0 <= j < n_points:
                        grad_i += (1 - 2 * h[j])  
                grad[i] = grad_i

        # Take gradient step (without fixed lag due to conv nature)
        h += lr * grad
        
        # Apply constraints: clip to unit interval and rescale

        # Early stopping if no improvement in a while
        if i - last_improvement > 200:
            break
            
        # Reduce learning rate over time
        if i % 50 == 0 and i > 0:
            lr *= 0.9

    # Phase 2: SQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP',

                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception as e:
        pass

    # Final valid check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
