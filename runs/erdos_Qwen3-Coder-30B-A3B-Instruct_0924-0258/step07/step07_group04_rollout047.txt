import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe algorithm with adaptive learning rate
    max_iter = 1000
    gamma = 0.1  # initial learning rate
    stagnation_count = 0
    
    h_current = h_start.copy()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation and find maximum overlap direction
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag = np.argmax(corr)
        max_val = corr[max_lag]
        
        # Compute gradient direction (approximated)
        # This is a simplified gradient step using the correlation structure
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Compute the derivative w.r.t. h[i]
            # contribution from correlation with shifted h
            total = 0
            for j in range(n_points):
                lag = j - i
                if 0 <= lag < n_points:
                    total += (1 - h_current[j]) * (1 if lag == max_lag else 0)
                elif -n_points < lag <= 0:
                    total += (1 - h_current[j]) * (1 if lag == max_lag - n_points else 0)
            gradient[i] = total
        
        # Update step (Frank-Wolfe style)
        h_new = h_current - gamma * gradient
        
        # Project onto feasible set (box constraints + sum constraint)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive learning rate
        if c5_new < evaluate_erdos_solution(h_current, dx):
            gamma *= 1.1
        else:
            gamma *= 0.9
            
        gamma = max(0.01, min(0.5, gamma))
        
        # Perturbation to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            perturbation = np.random.normal(0, 0.02, n_points)
            h_perturbed = h_current + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            c5_pert = evaluate_erdos_solution(h_perturbed, dx)
            if c5_pert < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_pert
            h_current = h_perturbed
        else:
            h_current = h_new
            
        if stagnation_count > 100:
            break
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
