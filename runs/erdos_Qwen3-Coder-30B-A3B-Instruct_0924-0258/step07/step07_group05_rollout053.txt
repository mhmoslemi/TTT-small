import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for the algorithm
    n_particles = 50
    max_iter = 1000
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    step_size = 0.1
    temperature = 1.0
    cooling_rate = 0.995
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    # Generate initial population
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.02, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        c5 = evaluate_erdos_solution(h, dx)
        pbest.append(h.copy())
        pbest_c5.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Iterative optimization
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Check if stopping condition is met
        if best_c5 < 0.3807:  # Early stopping if target is reached
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (best_h - particles[i]))
            
            # Update position
            particles[i] += velocities[i] * step_size
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Repair constraint
            sum_h = np.sum(particles[i])
            if abs(sum_h - n_points/2) > 1e-6:
                particles[i] = particles[i] * (n_points/2) / sum_h
            
            # Evaluate
            c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
                
                if c5 < best_c5:
                    best_h = particles[i].copy()
                    best_c5 = c5
            
            # Simulated annealing
            if c5 > best_c5 and np.random.rand() < np.exp(-(c5 - best_c5) / temperature):
                # Accept worse solution
                pass
            
        # Update temperature
        temperature *= cooling_rate

    # Global optimization refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run several optimizations with different starting points
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            h_trial = best_h + np.random.normal(0, 0.01, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points/2) / np.sum(h_trial)
            
            res = minimize(objective, h_trial, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    
    except Exception:
        pass

    # Final convex relaxation check
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add a few key constraints for the optimization
        for k in range(1, n_points//2):
            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-5)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
