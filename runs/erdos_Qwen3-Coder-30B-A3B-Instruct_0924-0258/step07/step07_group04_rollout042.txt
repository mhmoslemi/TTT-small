import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a structured guess
        h_start = np.ones(n_points) * 0.5
        # Perturb slightly to break symmetry 
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # Objective function with barrier method for constraints
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # We'll use differential evolution for global optimization first
    # Then refine with SLSQP
    
    # Differential evolution parameters
    maxiter = 500
    popsize = 50
    tol = 1e-6
    mutation = (0.5, 1.0)
    recombination = 0.9
    
    # Apply DE
    result_de = differential_evolution(
        objective, 
        bounds, 
        maxiter=maxiter, 
        popsize=popsize, 
        mutation=mutation, 
        recombination=recombination,
        seed=seed,
        workers=1,
        disp=False,
        atol=tol,
        callback=lambda x, convergence: 
            time() - start_time > budget_s * 0.95
    )
    
    # Get the best solution from DE
    h_de = result_de.x
    c5_de = result_de.fun
    
    # Refine with SLSQP if time allows
    if time() - start_time < budget_s * 0.8:
        try:
            # Initialize with DE result
            h_refined = h_de.copy()
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            
            # Define constraints for SLSQP
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds_slsqp = [(0, 1) for _ in range(n_points)]
            
            # Perform local refinement
            result_slsqp = minimize(
                objective, 
                h_refined, 
                method='SLSQP', 
                bounds=bounds_slsqp, 
                constraints=cons_sum, 
                options={'maxiter': 200, 'ftol': 1e-10}
            )
            
            if result_slsqp.success:
                h_final = result_slsqp.x
                # Ensure sum constraint is met
                h_final = h_final * (n_points/2) / np.sum(h_final)
                c5_final = objective(h_final)
                
                if c5_final < c5_de:
                    c5_de = c5_final
                    h_de = h_final
        except Exception:
            pass
    
    # Final validation and return best
    h_de = np.clip(h_de, 0, 1)
    h_de = h_de * (n_points/2) / np.sum(h_de)
    final_c5 = evaluate_erdos_solution(h_de, dx)
    
    return (list(h_de), final_c5, n_points)
