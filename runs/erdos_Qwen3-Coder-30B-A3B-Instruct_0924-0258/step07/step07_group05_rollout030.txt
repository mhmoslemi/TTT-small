import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA Parameters
    pop_size = 50
    num_generations = 1000
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    elite_size = 5
    convex_relax_interval = 50
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
        fit = evaluate_erdos_solution(h, dx)
        fitness.append(fit)
        if fit < best_c5:
            best_h = h.copy()
            best_c5 = fit
    
    # Genetic Algorithm
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism
        new_population = population[:elite_size]
        new_fitness = fitness[:elite_size]
        
        # Tournament selection and reproduction
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = tournament_selection(population, fitness, tournament_size)
            parent2_idx = tournament_selection(population, fitness, tournament_size)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child1, child2 = crossover(population[parent1_idx], population[parent2_idx])
            else:
                child1, child2 = population[parent1_idx].copy(), population[parent2_idx].copy()
            
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Evaluate children
            fit1 = evaluate_erdos_solution(child1, dx)
            fit2 = evaluate_erdos_solution(child2, dx)
            
            new_population.extend([child1, child2])
            new_fitness.extend([fit1, fit2])
            
            if fit1 < best_c5:
                best_h = child1.copy()
                best_c5 = fit1
            if fit2 < best_c5:
                best_h = child2.copy()
                best_c5 = fit2
        
        # Trim to pop_size
        population = new_population[:pop_size]
        fitness = new_fitness[:pop_size]
        
        # Convex relaxation every convex_relax_interval generations
        if generation % convex_relax_interval == 0 and generation > 0:
            try:
                # Select top candidates
                top_indices = np.argsort(fitness)[:pop_size//2]
                for idx in top_indices:
                    h_candidate = population[idx]
                    # Solve convex relaxation
                    h_refined = convex_relax(h_candidate, n_points, dx, best_c5)
                    if h_refined is not None:
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
            except Exception:
                pass
    
    # Final local optimization on best
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)

def tournament_selection(population, fitness, tournament_size):
    """Select an individual using tournament selection."""
    selected_indices = np.random.choice(len(population), tournament_size)
    best_idx = selected_indices[np.argmin([fitness[i] for i in selected_indices])]
    return best_idx

def crossover(parent1, parent2):
    """Uniform crossover."""
    child1 = parent1.copy()
    child2 = parent2.copy()
    for i in range(len(parent1)):
        if np.random.rand() < 0.5:
            child1[i], child2[i] = child2[i], child1[i]
    return child1, child2

def mutate(h, mutation_rate):
    """Mutate a solution with Gaussian noise."""
    h_mut = h.copy()
    for i in range(len(h)):
        if np.random.rand() < mutation_rate:
            h_mut[i] += np.random.normal(0, 0.03)
    h_mut = np.clip(h_mut, 0, 1)
    # Normalize sum
    h_mut = h_mut * (len(h) / 2) / np.sum(h_mut)
    return h_mut

def convex_relax(h_init, n_points, dx, max_c5):
    """Use convex relaxation to refine a solution.""" 
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add a constraint that the solution is not worse than max_c5
        corr_sum = cp.sum(cp.multiply(h_opt[:-1], 1 - h_opt[1:]))
        constraints.append(corr_sum <= max_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        
        if prob.status == cp.OPTIMAL:
            return h_opt.value
    except Exception:
        pass
    return None
