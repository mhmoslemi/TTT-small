import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimize with a modified Nelder-Mead approach
    # Use smaller steps and focus on the top overlap lags
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
        
    # Split problem into two phases:
    # Phase 1: Use fewer samples for faster exploration
    n_small = min(n_points, 20)
    if n_points > n_small:
        # Sample from original to create smaller problem
        h_small = h_start[::n_points//n_small]
        h_small = h_small * (n_small / 2) / np.sum(h_small)
        
        # Optimizer for smaller problem
        bounds_small = [(0, 1) for _ in range(n_small)]
        result_small = minimize(fitness, h_small, 
                               bounds=bounds_small,
                               method='Nelder-Mead',
                               options={'maxiter': 500, 'xatol': 1e-6, 'fatol': 1e-6})
        if result_small.success:
            h_opt_small = result_small.x
            h_opt_small = h_opt_small * (n_small / 2) / np.sum(h_opt_small)
            
            # Map back to full resolution
            h_opt_full = np.interp(np.linspace(0, 2, n_points), 
                                 np.linspace(0, 2, n_small), h_opt_small)
            h_opt_full = h_opt_full * (n_points / 2) / np.sum(h_opt_full)
            
            c5_opt = evaluate_erdos_solution(h_opt_full, dx)
            if c5_opt < best_c5:
                best_h = h_opt_full
                best_c5 = c5_opt

    # Phase 2: Full optimization with slight perturbations
    max_iter = 200
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try different heuristics with random neighbors
        h_new = best_h.copy()
        # Random perturbations
        indices = np.random.choice(n_points, size=n_points//3, replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.05)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

        # Additionally, fit a pattern based on the best hits
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        top_lags = np.argsort(corr)[-10:][::-1]
        if len(top_lags) > 0:
            # Try to reduce maximum lag overlap
            # Construct highly peaked function temporarily
            h_peak = best_h.copy()
            for k in top_lags[:3]:
                # Reduce at the prominent lag positions
                h_peak = np.roll(h_peak, -k//2)
                # Reduce a small region
                for j in range(max(0, k//2-2), min(n_points, k//2+2)):
                    h_peak[j] *= 0.9
            h_peak = np.clip(h_peak, 0, 1)
            h_peak = h_peak * (n_points/2) / np.sum(h_peak)
            c5_peak = evaluate_erdos_solution(h_peak, dx)
            if c5_peak < best_c5:
                best_h = h_peak
                best_c5 = c5_peak

    # Final refinement with CVXPY
    try:
        # Reduce CVX problem size for speed
        n_refine = min(n_points, 30)
        if n_refine < n_points:
            # Sample a subset of points
            indices = np.arange(0, n_points, n_points//n_refine)
            h_sample = best_h[indices]
            dx_sample = 2.0 / n_refine
        else:
            h_sample = best_h
            dx_sample = dx
            indices = np.arange(n_points)
            
        # Set up convex variables
        h_opt = cp.Variable(n_refine)
        constraints = [
            cp.sum(h_opt) == n_refine / 2,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraint based on the best overlap detected so far
        corr_orig = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag = np.argmax(corr_orig)
        # Focus on the extreme lag values
        active_lags = [max_lag - n_refine + i for i in range(1, min(5, max_lag))]
        active_lags.extend([max_lag + i for i in range(1, min(5, n_points - max_lag))])
        
        # Add key overlap constraints
        if len(active_lags) > 0:
            for k in active_lags:
                if k < n_refine:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_refine-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_refine+1:], 1 - h_opt[:n_refine-(k-n_refine+1)]))
                constraints.append(corr_sum <= best_c5 * dx_sample + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            if n_refine < n_points:
                # Interpolate back to full size
                h_feasible_full = np.interp(np.linspace(0, 2, n_points), 
                                          np.linspace(0, 2, n_refine), h_feasible)
                h_feasible_full = h_feasible_full * (n_points/2) / np.sum(h_feasible_full)
            else:
                h_feasible_full = h_feasible
            c5_feasible = evaluate_erdos_solution(h_feasible_full, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible_full
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final policy steps to ensure feasibility and correctness
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
