import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T0 = 0.1
    cooling_rate = 0.001
    max_iter = 1000
    perturbation_fraction = 0.15
    
    # Initial solution
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Track best solution
    if current_c5 < best_c5:
        best_h = current_h.copy()
        best_c5 = current_c5
    
    # Annealing loop
    stagnation_count = 0
    T = T0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb solution
        perturb_indices = np.random.choice(n_points, 
                                          int(n_points * perturbation_fraction), 
                                          replace=False)
        new_h = current_h.copy()
        
        # Add small random perturbations
        delta = np.random.uniform(-0.1, 0.1, len(perturb_indices))
        new_h[perturb_indices] += delta
        
        # Clip to [0,1]
        new_h = np.clip(new_h, 0, 1)
        
        # Rescale to satisfy constraint
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Cool down
        T = T0 * np.exp(-cooling_rate * iteration)
        
        if stagnation_count > 100:
            break

    # Gradient Descent refinement
    learning_rate = 0.1
    gd_max_iter = 500
    patience = 50
    
    current_h = best_h.copy()
    current_c5 = best_c5
    
    for iteration in range(gd_max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find the max overlap lag
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient for current max overlap
        gradient = np.zeros_like(current_h)
        n = len(current_h)
        if k_max < n:
            # Contributions from h[i] to overlap at lag k_max
            for i in range(n - k_max):
                gradient[i] += (1 - current_h[i + k_max]) * dx
        else:
            # Contributions from h[i] to overlap at lag k_max
            for i in range(k_max, n):
                gradient[i] += (1 - current_h[i - k_max]) * dx
        
        # Update h
        new_h = current_h - learning_rate * gradient
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            patience = 50
        else:
            patience -= 1
            if patience <= 0:
                learning_rate *= 0.5
                patience = 50
                
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
