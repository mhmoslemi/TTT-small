import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a different optimization approach: Mixed-Integer Programming
    # We'll generate a discretized version using a more structured approach
    
    # Initialize with the known good solution or a simple one
    h_current = h_start.copy()
    
    # Iterative refinement using minimize
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Triangular optimization approach
    step_size = 0.01
    max_iter = 1000
    improvement_threshold = 1e-6
    last_c5 = best_c5
    
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try refining with SLSQP starting from current solution
        try:
            res = minimize(objective, h_current, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 20, 'ftol': 1e-8})
            if res.success:
                h_candidate = res.x
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                h_current = h_candidate
            else:
                # If SLSQP fails, try gradient-free: Nelder-Mead
                res = minimize(objective, h_current, method='Nelder-Mead',
                               options={'maxiter': 20, 'fatol': 1e-8})
                if res.success:
                    h_candidate = res.x
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
                    h_current = h_candidate
        except Exception:
            pass
            
        # Occasionally reinitialize to avoid local minima
        if it % 100 == 0 and it > 0:
            h_current = h_start.copy()
            
        # Early stopping criterion
        if abs(last_c5 - best_c5) < improvement_threshold:
            break
        last_c5 = best_c5
    
    # Perform final SLSQP optimization with best solution
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 100, 'ftol': 1e-10})
        if res.success:
            final_h = res.x
            final_c5 = objective(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
