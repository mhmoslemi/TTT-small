import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        # Periodic initialization with 5 cycles over [0,2]
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 0.4))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Multiresolution approach
    resolutions = [20, 40, 80]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
        if res > n_points:
            # Interpolate to higher resolution
            x_coarse = np.linspace(0, 2, n_points)
            x_fine = np.linspace(0, 2, res)
            current_h = np.interp(x_fine, x_coarse, current_h)
            n_points = res
            dx = 2.0 / n_points
        
        # Annealing parameters for current resolution
        max_iter = 200 if res == 80 else 150 if res == 40 else 100
        perturbation_start = 0.2
        perturbation_end = 0.05
        temp_decay = 0.99
        temp_start = 1.0
        temp = temp_start
        
        for iter_num in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Create candidate by perturbing 20% of points
            n_perturb = max(1, n_points // 5)
            indices = np.random.choice(n_points, n_perturb, replace=False)
            
            candidate_h = current_h.copy()
            perturb_scale = perturbation_start + (perturbation_end - perturbation_start) * (iter_num / max_iter)
            
            for i in indices:
                delta = np.random.uniform(-perturb_scale, perturb_scale) * candidate_h[i]
                candidate_h[i] += delta
            
            # Ensure bounds and normalization
            candidate_h = np.clip(candidate_h, 0, 1)
            candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
            
            # Evaluate
            candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
            
            # Accept if better or with acceptance probability
            if candidate_c5 < current_c5:
                current_h = candidate_h.copy()
                current_c5 = candidate_c5
            else:
                # Metropolis acceptance
                delta = candidate_c5 - current_c5
                if delta > 0 and np.random.random() < np.exp(-delta / temp):
                    current_h = candidate_h.copy()
                    current_c5 = candidate_c5
            
            # Update best if needed
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
            
            # Update temperature
            if iter_num % 50 == 0:
                temp *= temp_decay
    
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # SLSQP with multiple restarts
        best_refined = best_h.copy()
        best_refined_c5 = best_c5
        for _ in range(3):
            h_restart = best_h + np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(h_restart, 0, 1) 
            h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_refined_c5:
                    best_refined = h_final
                    best_refined_c5 = c5_final
        
        if best_refined_c5 < best_c5:
            best_h = best_refined
            best_c5 = best_refined_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
