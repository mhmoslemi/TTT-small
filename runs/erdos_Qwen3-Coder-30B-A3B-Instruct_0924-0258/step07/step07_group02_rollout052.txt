import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 50
    max_iter = 500
    elite_size = 5
    
    # Initialize population
    population = []
    fitness = []
    
    # Start with the best known solution
    population.append(best_h)
    fitness.append(best_c5)
    
    # Add random variations
    for i in range(pop_size - 1):
        if time() - start_time > budget_s * 0.95:
            break
        noise = np.random.normal(0, 0.1, n_points)
        h = best_h + noise
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Evolve
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track the best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new population
        new_population = []
        new_fitness = []
        
        # Elite preservation
        for i in range(elite_size):
            new_population.append(population[i].copy())
            new_fitness.append(fitness[i])
            
        # Generate offspring
        for i in range(pop_size - elite_size):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Tournament selection
            tournament_size = 5
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[j] for j in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Crossover with random parent
            parent2_idx = np.random.choice(pop_size, 1)[0]
            
            # Blend crossover
            alpha = np.random.rand()
            child = alpha * population[winner_idx] + (1 - alpha) * population[parent2_idx]
            
            # Mutation
            mutation_strength = max(0.01, 0.1 - (iter_num / max_iter) * 0.09)
            noise = np.random.normal(0, mutation_strength, n_points)
            child += noise
            
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            # Evaluate
            child_fitness = evaluate_erdos_solution(child, dx)
            
            new_population.append(child)
            new_fitness.append(child_fitness)
            
        population = new_population
        fitness = new_fitness
    
    # Final SQP refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add key constraints that are most active
        h_test = best_h.copy()
        corr = np.correlate(h_test, 1 - h_test, mode='full')
        top_k = np.argsort(corr)[-10:]  # top 10 lags
        for k in top_k:
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
