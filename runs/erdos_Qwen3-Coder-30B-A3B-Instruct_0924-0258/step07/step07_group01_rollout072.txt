import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 100
    num_generations = 200
    mutation_rate = 0.3
    selection_rate = 0.2
    stagnation_limit = 50
    
    # Initialize population
    population = []
    population_fitness = []
    
    # Seed with initial solution
    population.append(h_start.copy())
    population_fitness.append(evaluate_erdos_solution(h_start, dx))
    
    # Add random individuals
    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
        population_fitness.append(evaluate_erdos_solution(h, dx))
    
    # Track best
    best_idx = np.argmin(population_fitness)
    g_best = population[best_idx].copy()
    g_best_value = population_fitness[best_idx]
    
    # Stagnation tracking
    stagnation_count = 0
    
    # Evolve
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all
        for i in range(population_size):
            population_fitness[i] = evaluate_erdos_solution(population[i], dx)
            
        # Selection (top 20%)
        sorted_indices = np.argsort(population_fitness)
        selected_indices = sorted_indices[:int(population_size * selection_rate)]
        selected_population = [population[i] for i in selected_indices]
        
        # Create new population
        new_population = selected_population[:]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected_population, 2)
            # Arithmetic crossover
            alpha = random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                
            # Normalize to sum constraint
            child = child * (n_points/2) / np.sum(child)
            new_population.append(child)
            
        population = new_population[:population_size]
        population_fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update global best
        current_best_idx = np.argmin(population_fitness)
        current_best_value = population_fitness[current_best_idx]
        if current_best_value < g_best_value:
            g_best = population[current_best_idx].copy()
            g_best_value = current_best_value
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > stagnation_limit:
            break
    
    # Local refinement with SLSQP on top 5
    top_candidates = np.argsort([evaluate_erdos_solution(h, dx) for h in population])[:5]
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    for i in range(min(5, len(top_candidates))):
        h_refined = population[top_candidates[i]].copy()
        try:
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            continue
    
    return (list(best_h), best_c5, n_points)
