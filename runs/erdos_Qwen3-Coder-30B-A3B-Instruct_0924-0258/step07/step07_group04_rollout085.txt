import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create periodic function initialization
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    T = 0.5
    alpha = 0.995
    T_min = 1e-5
    max_iter = 1000
    tabu_list = []
    tabu_size = 20
    stagnation_count = 0
    max_stagnation = 50
    
    # Initial perturbation size
    perturbation_size = 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Select random indices to perturb (15% of points)
        n_perturb = max(1, int(0.15 * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Create candidate solution
        h_candidate = best_h.copy()
        for i in indices:
            # Perturb by random amount
            delta = np.random.uniform(-perturbation_size, perturbation_size)
            h_candidate[i] += delta
        
        # Clip and normalize
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        delta = c5_candidate - best_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            
            # Add to tabu list
            tabu_list.append(h_candidate.copy())
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
            
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Adaptive perturbation
        if iteration % 10 == 0 and perturbation_size > 0.05:
            perturbation_size *= 0.95
        
        # Temperature update
        T = max(T * alpha, T_min)
        
        # Restart if stagnation
        if stagnation_count >= max_stagnation:
            # Perturb best solution to escape local minimum
            noise = np.random.normal(0, 0.05, n_points)
            h_restart = best_h + noise
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            best_h = h_restart
            best_c5 = evaluate_erdos_solution(best_h, dx)
            stagnation_count = 0
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use best solution as starting point for refinement
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
