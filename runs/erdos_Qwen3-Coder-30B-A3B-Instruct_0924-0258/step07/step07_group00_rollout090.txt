import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Alternating minimization approach
    max_iterations = 1000
    step_size = 0.01
    tolerance = 1e-6
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for the worst case lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Update step
        h_new = h - step_size * gradient
        
        # Enforce constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check improvement
        current_c5 = evaluate_erdos_solution(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            step_size = min(0.01, step_size * 1.1)
        else:
            step_size = max(1e-6, step_size * 0.5)
            
        h = h_new
        
        if step_size < 1e-6:
            break
    
    # Final local refinement with L-BFGS-B
    def objective_and_gradient(h_flat):
        h = h_flat.reshape(n_points)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        return max_overlap, gradient.flatten()
    
    try:
        from scipy.optimize import fmin_l_bfgs_b
        h_flat = best_h.flatten()
        bounds = [(0, 1) for _ in range(n_points)]
        refined_h, _, _ = fmin_l_bfgs_b(objective_and_gradient, h_flat, 
                                         bounds=bounds, maxiter=100)
        refined_h = refined_h.reshape(n_points)
        refined_h = refined_h * (n_points/2) / np.sum(refined_h)
        refined_c5 = evaluate_erdos_solution(refined_h, dx)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
