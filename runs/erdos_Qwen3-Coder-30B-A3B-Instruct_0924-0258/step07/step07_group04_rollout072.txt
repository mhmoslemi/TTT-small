import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a more sophisticated initial guess
        h_start = np.sin(np.linspace(0, 2*np.pi, n_points)) * 0.3 + 0.5
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe style optimization with coordinate-wise updates
    max_iter = 3000
    gamma = 0.1  # Initial learning rate
    stagnation_count = 0
    patience = 100
    perturb_freq = 50
    
    h_current = h_start.copy()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Focus on a reasonable range of lags
        mid = len(corr) // 2
        lag_window = 50
        corr_trimmed = corr[max(0, mid-lag_window):min(len(corr), mid+lag_window)]
        max_lag_idx = np.argmax(corr_trimmed)
        max_lag = max_lag_idx + mid - lag_window
        
        # Direction: in direction of steepest descent (simplified Frank-Wolfe update)
        # We approximate by looking at the direction of maximum correlation
        s = np.zeros(n_points)
        # The idea is to shift h in direction that reduces the correlation peak
        s[max_lag] = 1
        s = np.correlate(s, np.ones(n_points), mode='full')[:n_points]
        s = 1 - h_current - s
        
        # Update by moving along this direction
        h_new = h_current + gamma * s
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > patience:
            # Try to escape local minimum
            h_new = h_current + np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            new_c5 = evaluate_erdos_solution(h_new, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                stagnation_count = 0
            else:
                # Reduce step size
                gamma *= 0.95
                if gamma < 1e-6:
                    break
        
        h_current = h_new.copy()
        
        # Occasionally perturb
        if iteration % perturb_freq == 0 and iteration > 0:
            pert = np.random.normal(0, 0.02, n_points)
            h_current += pert
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Occasionally adapt step size based on progress
        if iteration % 100 == 0 and iteration > 0:
            gamma *= 0.995
        
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
