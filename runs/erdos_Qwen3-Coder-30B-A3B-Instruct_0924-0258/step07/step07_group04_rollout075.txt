import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Initialize with periodic function
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.998
    max_iter = 10000
    tabu_size = 20
    tabu_list = []
    
    # Annealing loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate
        h_candidate = best_h.copy()
        
        # Select indices to perturb (10-20% of points)
        n_perturb = max(1, int(0.15 * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Perturb selected values
        for i in indices:
            perturbation = np.random.uniform(-0.15, 0.15)
            h_candidate[i] += perturbation
        
        # Clip and renormalize
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
        else:
            # Probabilistic acceptance
            delta = c5_candidate - best_c5
            if np.random.rand() < np.exp(-delta / T):
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
                
        # Update temperature
        T *= alpha
        
        # Update tabu list
        tabu_list.append(best_h.copy())
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
        
        # Try to escape local optima with restart
        if iteration % 500 == 0 and iteration > 0:
            h_restart = h_start.copy()
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            c5_restart = evaluate_erdos_solution(h_restart, dx)
            if c5_restart < best_c5:
                best_h = h_restart.copy()
                best_c5 = c5_restart
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
