import numpy as np
from scipy.optimize import fmin_l_bfgs_b
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points/2
    h = h * (n_points/2) / np.sum(h)
    dx = 2.0 / n_points
    
    start_time = time()
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Adaptive gradient descent with line search
    max_iter = 200
    step_size = 0.01
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current overlaps
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for the max overlap lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Update step
        h_new = h - step_size * gradient
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
            step_size = min(0.015, step_size * 1.2)
        else:
            step_size = max(1e-6, step_size * 0.5)
            
        if step_size < 1e-6:
            break
    
    # Final local refinement with L-BFGS-B
    def objective_and_gradient(h_vec):
        h = h_vec.reshape(n_points)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        return max_overlap, gradient.flatten()
    
    # Refine the best solution found
    try:
        best_h_flat, _, _ = fmin_l_bfgs_b(
            objective_and_gradient, 
            best_h.flatten(), 
            bounds=[(0, 1) for _ in range(n_points)], 
            maxiter=100)
        best_h = best_h_flat.reshape(n_points)
        best_h = best_h * (n_points/2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
