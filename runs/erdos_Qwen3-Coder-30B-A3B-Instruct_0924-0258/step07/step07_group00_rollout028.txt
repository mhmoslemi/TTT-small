import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        n_points = 35

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with focus on maximum lag
    max_iter = 10000
    step_size = 0.01
    patience = 100
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)  # Ensure sum constraint
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)  # Convert to actual lag value
        max_overlap = corr[max_lag_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Compute gradient for the most constraining lag
        # For lag k, d/dh_i (h_i * (1 - h_{i+k})) = (1 - h_{i+k}) if i+k < n_points
        # More precisely, the gradient w.r.t. h_i of the overlap with lag k is 
        # the sum of (1 - h_{i+k}) where i+k ranges over valid indices.
        grad = np.zeros_like(h)
        # For positive lag k
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                grad[i] += (1 - h[i + max_lag])
        # For negative lag -k
        else:
            for i in range(-max_lag, n_points):
                grad[i] += (1 - h[i + max_lag])
        
        # New h = h - step_size * grad
        h_new = h - step_size * grad
        
        # Clip to bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy ∫ h = 1 constraint (sum(h) * dx = 1 => sum(h) = n_points / 2)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Try larger step size, or reduce step size if constraint violated
        h_test = h_new.copy()
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points/2) / np.sum(h_test)
        c5_test = evaluate_erdos_solution(h_test, dx)
        
        if c5_test < max_overlap:
            h = h_test
            step_size *= 1.05  # increase step size
        else:
            step_size *= 0.5  # reduce step size
            if step_size < 1e-8:
                break  # give up if step is too small
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
