import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization
    max_iter = 10000
    step_size = 0.01
    tolerance = 1e-9
    
    h_current = h_start.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        # Identify which lag has maximum overlap
        # The correlation has length 2*n_points - 1
        # The maximum overlap is at index max_lag_idx
        # The actual lag is: max_lag_idx - (n_points - 1)
        lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient for the maximum overlap
        # For a given lag k, we want to differentiate:
        # integral h(x)(1 - h(x+k))dx
        # This is sampled as: sum(h[i] * (1 - h[i+k])) * dx (with wraparound)
        
        # Compute gradient: d/dh[i] of integral h(x)(1 - h(x+k))dx
        # The gradient is: (1 - h[i+k]) + (1 - h[i-k]) ... but we need to be more careful
        # Let's compute it correctly with proper indexing
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Contribution from h[i] to the overlap at lag -lag
            if (i + lag) % n_points < n_points:
                grad[i] += 1 - h_current[(i + lag) % n_points]
            if (i - lag) % n_points < n_points:
                grad[i] += 1 - h_current[(i - lag) % n_points]
                
        # Gradient is same for all indices that are affected by the maximum overlap
        # Clipping to make plain gradient descent work better
        # But take a conservative approach: just project onto feasible direction
        
        # Simple and robust approach: step in the direction that reduces the overlap
        # This means, we want to move towards a direction that decreases most
        # We can consider the projected gradient descent step
        
        # Simplified: we move along an estimate of the gradient direction
        new_h = h_current - step_size * grad
        
        # Clip to [0,1]
        new_h = np.clip(new_h, 0, 1)
        
        # Maintain the sum constraint
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Accept the new point if it improves the metric
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            
        h_current = new_h
        
        # Reduce step size if overflow or no improvement within reasonable steps
        if iteration % 50 == 0 and iteration > 0:
            if abs(new_c5 - max_overlap) < tolerance:
                step_size *= 0.9  # Reduce step size

    # Final local refinement using scipy
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try with a variation of the current best
    try:
        res = minimize(objective, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
