import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Basis function decomposition approach
    # 10 basis functions (piecewise linear)
    m = 10
    basis = np.zeros((n_points, m))
    
    # Create triangular basis functions
    x_vals = np.linspace(0, 2, n_points)
    for i in range(m):
        # Place basis functions at different locations
        center = 2.0 * i / (m - 1) if m > 1 else 1.0
        width = 0.4
        basis[:, i] = np.maximum(0, 1 - np.abs(x_vals - center) / width)
        # Normalize each basis to have integral 1
        integral = np.trapz(basis[:, i], x_vals)
        if integral > 0:
            basis[:, i] /= integral
    
    # Scale to enforce ∫h = 1
    basis_sum = np.sum(basis, axis=1)
    basis_sum[basis_sum == 0] = 1
    basis /= basis_sum[:, np.newaxis]
    
    # Coefficients for basis functions
    coeffs = np.ones(m) / m
    
    # Trust-region optimization on coefficients
    def objective(coeffs):
        h = basis @ coeffs
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Enforce integral constraint via scaling
        h = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(coeffs):
        h = basis @ coeffs
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        return sum_h - n_points/2
    
    # Trust region with scipy
    max_iter = 2000
    trust_radius = 0.1
    patience = 0
    max_patience = 50
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution using coefficients
        h_current = basis @ coeffs
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > max_patience:
            # Perturb coefficients to escape local minima
            coeffs += np.random.normal(0, 0.05, m)
            coeffs = np.abs(coeffs)  # Ensure positive
            coeffs = coeffs / np.sum(coeffs) * m  # Rescale to sum to m
            patience = 0
            
        # Compute gradient approximation using finite differences
        eps = 1e-4
        grad = np.zeros(m)
        for i in range(m):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = np.clip(basis @ coeffs_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            coeffs_minus = coeffs.copy()
            coeffs_minus[i] -= eps
            h_minus = np.clip(basis @ coeffs_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
        # Update using trust region step
        # Simple gradient descent with adaptive step size
        step = -0.01 * grad
        new_coeffs = coeffs + step
        
        # Enforce non-negativity and normalization
        new_coeffs = np.abs(new_coeffs)
        new_coeffs = new_coeffs / np.sum(new_coeffs) * m
        
        # Check if new point improves objective
        h_new = basis @ new_coeffs
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < c5_current:
            coeffs = new_coeffs.copy()
        else:
            # Reduce trust region
            trust_radius *= 0.5

    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
