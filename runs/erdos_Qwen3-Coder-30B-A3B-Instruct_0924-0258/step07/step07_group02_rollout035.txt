import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm parameters
    population_size = 100
    generations = 250
    elite_size = 10
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    decay_interval = 50
    mutation_decay = 0.9
    
    # Initialize population
    population = []
    for i in range(population_size):
        if i == 0:
            # Start with initial solution
            h = h_start.copy()
        else:
            # Random initialization
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evolutionary loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Update best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
        
        # Selection and reproduction
        sorted_indices = np.argsort(fitness)
        new_population = [population[i] for i in sorted_indices[:elite_size]]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection for parents
            parent1_idx = sorted_indices[np.random.choice(tournament_size)]
            parent2_idx = sorted_indices[np.random.choice(tournament_size)]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Apply Gaussian noise
                noise = np.random.normal(0, 0.1 * (mutation_decay ** (gen // decay_interval)), n_points)
                child = child + noise
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                
            # Bit-flip mutation for top 30%
            if gen > 0 and np.random.rand() < 0.1 and fitness[parent1_idx] < np.median(fitness[:population_size//3]):
                flip_indices = np.random.choice(n_points, size=max(1, n_points//20), replace=False)
                for idx in flip_indices:
                    child[idx] = 1 - child[idx]
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                
            new_population.append(child)
        
        # Truncate to population size
        population = new_population[:population_size]
    
    # Final refinement with L-BFGS-B
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    # Select top performers for final refinement
    top_performers = []
    fitness = []
    for h in population:
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
        top_performers.append((h, c5))
    
    top_performers.sort(key=lambda x: x[1])
    
    # Try to refine top solutions with local optimization
    for h_start, _ in top_performers[:5]:
        try:
            result = minimize(
                objective,
                h_start,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 100}
            )
            if result.success:
                refined_h = result.x
                refined_h = np.clip(refined_h, 0, 1)
                refined_h = refined_h * (n_points/2) / np.sum(refined_h)
                refined_c5 = evaluate_erdos_solution(refined_h, dx)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
        except:
            continue
    
    # Final CVXPY refinement to ensure feasibility and further improve
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add approximate constraints based on top performers
        top_h = [h for h, _ in top_performers[:3]]
        max_c5 = max([evaluate_erdos_solution(h, dx) for h in top_h])
        h_test = best_h.copy()
        corr = np.correlate(h_test, 1 - h_test, mode='full')
        # Add some constraints to help optimization
        top_k = np.argsort(corr)[-10:]  # Top 10 lags by overlap
        for k in top_k:
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= max_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=300)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception as e:
        pass

    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
