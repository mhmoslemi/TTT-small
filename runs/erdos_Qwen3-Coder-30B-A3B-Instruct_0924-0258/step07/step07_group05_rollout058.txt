import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    pop_size = 50
    generations = 1000
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
    
    # Track best solution
    for h in population:
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Genetic Algorithm
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Elitism
        elites = [population[i] for i in np.argsort(fitness)[:elite_size]]
        
        # Tournament selection
        new_population = elites.copy()
        while len(new_population) < pop_size:
            # Select parents
            parent1_idx = np.argmin([fitness[i] for i in np.random.choice(len(population), tournament_size)])
            parent2_idx = np.argmin([fitness[i] for i in np.random.choice(len(population), tournament_size)])
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([population[parent1_idx][:crossover_point], 
                                       population[parent2_idx][crossover_point:]])
                child2 = np.concatenate([population[parent2_idx][:crossover_point], 
                                       population[parent1_idx][crossover_point:]])
            else:
                child1 = population[parent1_idx].copy()
                child2 = population[parent2_idx].copy()
            
            # Mutation
            for child in [child1, child2]:
                mutation_mask = np.random.rand(n_points) < mutation_rate
                child[mutation_mask] += np.random.normal(0, 0.03, np.sum(mutation_mask))
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                
            new_population.extend([child1, child2][:pop_size - len(new_population)])
        
        population = new_population[:pop_size]
        
        # Periodic convex relaxation every 50 generations
        if gen % 50 == 0 and gen > 0:
            try:
                # Take top 10% for refinement
                top_idx = np.argsort([evaluate_erdos_solution(h, dx) for h in population])[:pop_size//10]
                for i in top_idx:
                    h_candidate = population[i].copy()
                    h_opt = cp.Variable(n_points)
                    constraints = [
                        cp.sum(h_opt) == n_points / 2.0,
                        h_opt >= 0,
                        h_opt <= 1
                    ]
                    
                    # Add some constraints from the current best solution
                    corr_current = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                    max_k = np.argmax(corr_current)
                    k_val = max_k - (n_points-1)  # convert index to actual k offset
                    
                    # Sample small number of k values for convex relaxation
                    k_vals = np.arange(0, n_points//2) * 2
                    k_vals = k_vals[k_vals < n_points]
                    for k in k_vals:
                        if k < n_points and k > 0:
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                    
                    prob = cp.Problem(cp.Minimize(0), constraints)
                    prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                    if prob.status == cp.OPTIMAL:
                        h_feasible = h_opt.value
                        c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                        if c5_feasible < best_c5:
                            best_h = h_feasible
                            best_c5 = c5_feasible
            except Exception:
                pass
        
        # Update best
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
