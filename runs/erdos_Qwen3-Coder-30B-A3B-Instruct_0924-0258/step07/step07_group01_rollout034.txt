import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Attempt to solve with MILP using CVXPY
    max_overlap_initial = 0.38085
    tighten_factor = 0.995
    max_iter = 50
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Linearize using McCormick envelopes
        h_vars = cp.Variable(n_points)
        max_overlap = cp.Variable()
        
        # Objective: minimize max_overlap
        objective = cp.Minimize(max_overlap)
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1,
            max_overlap >= 0
        ]
        
        # Add McCormick envelopes for bilinear terms using heuristics
        # This is a simplified version of the method used in more complex solvers
        # For each k, we want to bound the integral term of h(x) * (1 - h(x+k))
        # We add linear constraints trying to bound each contribution
        
        # Direct computation with trust region
        def obj_func(h_arr):
            h_arr = np.clip(h_arr, 0, 1)
            h_arr = h_arr * (n_points / 2) / np.sum(h_arr)
            return evaluate_erdos_solution(h_arr, dx)
        
        # Local optimization around current solution
        try:
            res = minimize(obj_func, best_h, method='L-BFGS-B', 
                           bounds=[(0, 1) for _ in range(n_points)],
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_local = res.x
                c5_local = obj_func(h_local)
                if c5_local < best_c5:
                    best_h = h_local
                    best_c5 = c5_local
        except Exception:
            pass
            
        # Tightly tighten max_overlap
        max_overlap_initial = max_overlap_initial * tighten_factor
        
        # Early stopping if improvement is negligible
        if abs(max_overlap_initial - 0.3807) < 1e-5:
            break
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
