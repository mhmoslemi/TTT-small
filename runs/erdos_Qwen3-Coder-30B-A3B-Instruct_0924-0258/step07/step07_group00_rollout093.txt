import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy sum constraint
    dx = 2.0 / n_points
    h = h * (n_points/2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient-based optimization with adaptive step size
    max_iterations = 500
    step_size = 0.01
    tolerance = 1e-6
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps and find maximum
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for the maximum overlap lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = (1 - h[j] - h[i]) * 2 * dx
        
        # Update h
        h_new = h - step_size * gradient
        
        # Enforce bounds and sum constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            step_size = min(0.01, step_size * 1.2)
        else:
            step_size = max(1e-6, step_size * 0.5)
        
        h = h_new
        
        if step_size < 1e-6:
            break
    
    # Final local refinement with L-BFGS-B
    def objective_and_gradient(h_vec):
        h = h_vec.reshape(-1)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        return max_overlap, gradient.flatten()
    
    try:
        from scipy.optimize import fmin_l_bfgs_b
        h_flat = best_h.flatten()
        bounds = [(0, 1) for _ in range(n_points)]
        result = fmin_l_bfgs_b(objective_and_gradient, h_flat, bounds=bounds, maxiter=200)
        best_h = result[0].reshape(-1)
        best_h = best_h * (n_points/2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
