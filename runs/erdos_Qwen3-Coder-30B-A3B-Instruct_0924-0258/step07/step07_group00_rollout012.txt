import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with adaptive learning rate
    h = h_start.copy()
    eta = 0.01
    no_improvement_count = 0
    max_no_improvement = 50
    
    for iteration in range(1000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlations for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_overlap = corr[max_k] * dx
        
        # Check if we improved
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break

        # Compute gradient for the max overlapping lag
        # lag = max_k - (n_points - 1)
        lag = max_k - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + lag) % n_points
            gradient[i] = 2 * (1 - h[j] - h[i])

        # Update h
        h = h - eta * gradient
        
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive learning rate
        if np.any(h < 0) or np.any(h > 1):
            eta *= 0.5
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            eta = min(eta * 1.05, 0.1)  # Gradually increase if stable
                
    # Final local optimization using SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 300, 'ftol': 1e-10})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception as e:
        pass

    return (list(best_h), best_c5, n_points)
