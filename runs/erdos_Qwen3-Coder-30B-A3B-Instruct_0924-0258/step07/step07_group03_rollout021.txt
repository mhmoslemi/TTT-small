import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T0 = 0.1
    cooling_rate = 0.001
    max_iter = 1000
    perturb_fraction = 0.15
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    stagnation_count = 0
    T = T0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb
        num_perturb = int(n_points * perturb_fraction)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        delta = np.random.uniform(-0.1, 0.1, num_perturb)
        new_h = current_h.copy()
        new_h[indices] += delta
        new_h = np.clip(new_h, 0, 1)
        
        # Normalize
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Cool down
        T = T0 * np.exp(-cooling_rate * iteration)
        
        if stagnation_count > 100:
            break
    
    # Gradient Descent refinement
    def compute_gradient(h, k_max):
        grad = np.zeros_like(h)
        n = len(h)
        for i in range(n):
            if i + k_max < n:
                grad[i] += (1 - h[i + k_max]) * dx
            if i - k_max >= 0:
                grad[i] += (1 - h[i - k_max]) * dx
        return grad
    
    h_refined = best_h.copy()
    learning_rate = 0.1
    max_iter_gd = 500
    stagnation_gd = 0
    
    for iteration in range(max_iter_gd):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap for all k
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        
        # Compute gradient
        grad = compute_gradient(h_refined, k_max)
        
        # Update
        h_refined = h_refined - learning_rate * grad
        h_refined = np.clip(h_refined, 0, 1)
        
        # Normalize
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_refined, dx)
        
        if new_c5 < best_c5:
            best_h = h_refined.copy()
            best_c5 = new_c5
            stagnation_gd = 0
        else:
            stagnation_gd += 1
            
        if stagnation_gd > 50:
            learning_rate /= 2.0
            stagnation_gd = 0
            
        if learning_rate < 1e-6:
            break
    
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
