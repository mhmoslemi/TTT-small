import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with adaptive cooling
    n_iter = 0
    max_iter = 2000
    T = 1.0
    T_min = 1e-6
    alpha = 0.95
    
    h_current = best_h.copy()
    c5_current = best_c5
    
    while T > T_min and time() - start_time < budget_s * 0.95:
        n_iter += 1
        # Generate neighbor solution
        h_neighbor = h_current + np.random.normal(0, 0.02, n_points)
        h_neighbor = np.clip(h_neighbor, 0, 1)
        h_neighbor = h_neighbor * (n_points/2) / np.sum(h_neighbor)
        
        # Compute overlap
        c5_neighbor = evaluate_erdos_solution(h_neighbor, dx)
        
        # Accept or reject
        delta = c5_neighbor - c5_current
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            h_current = h_neighbor
            c5_current = c5_neighbor
            
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
        
        # Cool down
        if n_iter % 20 == 0:
            T *= alpha
    
    # Final trust-constr optimization
    try:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_integral(h):
            return np.sum(h) - n_points / 2.0
        
        def constraint_bounds(h):
            return np.concatenate([h, 1 - h])
        
        cons = [{'type': 'eq', 'fun': constraint_integral},
                {'type': 'ineq', 'fun': lambda h: constraint_bounds(h)[n_points:]}]
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='trust-constr', bounds=bounds, constraints=cons, 
                      options={'maxiter': 500, 'disp': False})
        
        if res.success:
            h_opt = res.x
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    except Exception:
        pass

    # Final verification and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
