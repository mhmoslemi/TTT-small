import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based approach
    h = h_start.copy()
    learning_rate = 0.01
    no_improvement_count = 0
    max_no_improvement = 50
    
    iteration = 0
    while time() - start_time < budget_s * 0.95:
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_k = max_idx - (len(h) - 1)  # Convert index to actual lag
        
        # Compute gradient for this specific lag
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + max_k) % len(h)
            gradient[i] = 2 * (1 - h[j] - h[i])
        
        # Update with gradient
        h_new = h - learning_rate * gradient
        
        # Ensure feasibility and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
            learning_rate = min(0.01, learning_rate * 1.05)  # Adaptively increase learning rate
        else:
            no_improvement_count += 1
            learning_rate *= 0.5  # Reduce learning rate
            
        h = h_new.copy()
        
        if no_improvement_count >= max_no_improvement:
            # Occasionally try random perturbation
            if iteration % 100 == 0:
                h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                h = h * (n_points/2) / np.sum(h)
            no_improvement_count = 0
        
        iteration += 1
        
        # Occasionally try local refinement
        if iteration % 100 == 0:
            try:
                def objective(h_vec):
                    return evaluate_erdos_solution(h_vec, dx)
                
                def constraint_sum(h_vec):
                    return np.sum(h_vec) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(objective, h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-9})
                
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    return (list(best_h), best_c5, n_points)
