import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Benders decomposition approach
    max_iter = 5000
    convergence_tol = 1e-6
    perturbation_mag = 0.05
    perturbation_decay = 0.95
    
    # Constraints and variables
    h_vars = cp.Variable(n_points)
    
    # Initialize constraints
    constraints = [
        cp.sum(h_vars) == n_points / 2.0,
        h_vars >= 0,
        h_vars <= 1
    ]
    
    # Add initial constraints for active k values
    for k in range(1, n_points):
        if k <= n_points - 1:
            corr_sum = cp.sum(cp.multiply(h_vars[:n_points-k], 1 - h_vars[k:]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
    
    # Main optimization loop
    last_c5 = float('inf')
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try to solve relaxed problem
        try:
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            if prob.status == cp.OPTIMAL:
                h_new = h_vars.value
                if h_new is not None:
                    # Perturb to escape local minima
                    h_perturbed = h_new + perturbation_mag * np.random.randn(n_points)
                    h_perturbed = np.clip(h_perturbed, 0, 1)
                    h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
                    c5_new = evaluate_erdos_solution(h_perturbed, dx)
                    if c5_new < best_c5:
                        best_h = h_perturbed.copy()
                        best_c5 = c5_new
                    perturbation_mag *= perturbation_decay
        except Exception:
            pass
            
        # Identify problematic shifts
        h_current = best_h.copy()
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_k = np.argmax(corr)
        max_overlap = corr[max_k] * dx
        
        # Add constraint for worst case overlap
        k_val = max_k - (n_points - 1)
        if 1 <= abs(k_val) <= n_points - 1:
            k = abs(k_val)
            if k_val > 0:
                corr_constraint = cp.sum(cp.multiply(h_vars[:n_points-k], 1 - h_vars[k:]))
            else:
                corr_constraint = cp.sum(cp.multiply(h_vars[-k:], 1 - h_vars[:n_points+k]))
            constraints.append(corr_constraint <= max_overlap + 1e-6)
            
        # Check for convergence
        if abs(last_c5 - best_c5) < convergence_tol:
            break
        last_c5 = best_c5
    
    # Final polishing step
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final validation
    if best_c5 < 0:
        best_c5 = 0
    
    return (list(best_h), best_c5, n_points)
