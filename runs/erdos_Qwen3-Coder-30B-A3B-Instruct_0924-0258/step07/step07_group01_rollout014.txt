import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a step function with two unit intervals
        # h(x) = 1 on [a, a+0.5] and [b, b+0.5] where a=0.65, b=0.85
        # This ensures integral = 1 and has good overlap properties
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        a = 0.65
        b = 0.85
        a_idx = int(a / dx)
        b_idx = int(b / dx)
        a_end = int((a + 0.5) / dx)
        b_end = int((b + 0.5) / dx)
        h_start[a_idx:a_end] = 1
        h_start[b_idx:b_end] = 1
        # Normalize to ensure sum = n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization for local refinement
    max_iters = 1000
    lr = 0.01
    tolerance = 1e-8
    
    for it in range(max_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        current_c5 = evaluate_erdos_solution(best_h, dx)
        
        # Compute gradient approximation using finite differences
        grad = np.zeros_like(best_h)
        eps = 1e-6
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            # Ensure h_plus stays within bounds
            h_plus = np.clip(h_plus, 0, 1)
            # Renormalize
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            h_minus = best_h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update step
        new_h = best_h - lr * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < current_c5:
            best_h = new_h
            best_c5 = new_c5
        else:
            lr *= 0.999
            
        if abs(new_c5 - current_c5) < tolerance:
            break
    
    return (list(best_h), best_c5, n_points)
