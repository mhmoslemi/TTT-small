import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe style coordinate-wise optimization
    max_iter = 1000
    gamma = 0.1  # Initial step size
    stagnation_count = 0
    last_improvement = 0
    
    h_current = h_start.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute the correlation
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        lag = len(corr) - n_points
        # Focus on relevant lags (centred part)
        corr_trimmed = corr[lag//2:lag//2 + n_points]
        max_lag_idx = np.argmax(corr_trimmed)
        max_lag = max_lag_idx - n_points//2 + 1  # Convert to lag index
        
        # Compute the direction: gradient-like - (1-h) - projected on max overlap direction
        # Actually just using a descent direction based on max overlap direction
        direction = np.zeros(n_points)
        # For a simple gradient direction, based on correlation
        direction = 1 - h_current
        
        # Apply Frank-Wolfe update
        h_new = h_current + gamma * direction
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Normalize to correct integral
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Try a better update with line search or other improvement
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            last_improvement = iteration
        
        # Adaptive step size adjustment
        if new_c5 < evaluate_erdos_solution(h_current, dx):
            gamma *= 1.1  # Increase if successful
        else:
            gamma *= 0.9  # Decrease if failed
        gamma = max(0.005, min(0.5, gamma))
        
        h_current = h_new
        
        # Perturbation to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            perturbation_magnitude = 0.05
            h_current += np.random.normal(0, perturbation_magnitude, n_points)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Check for stagnation
        if iteration - last_improvement > 100:
            break
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
