import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize CVXPY problem
    h_var = cp.Variable(n_points, name='h')
    
    # Create correlation approximation
    def setup_overlap_constraint():
        # We'll use a constraint-based approach to get a handle on max overlap
        # This is approximate but effective in practice
        return h_var
    
    def refine_solution(h_vals):
        # Ensure it's in valid range
        h_vals = np.clip(h_vals, 0, 1)
        # Normalize
        h_vals = h_vals * (n_points/2) / np.sum(h_vals)
        return h_vals
    
    # Iterative optimization - MIP approach for better handling
    max_iter = 20
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:

        try:
            # Create constraints around current best
            h_current = best_h.copy()

            constraints = [

                h_var >= 0,
                h_var <= 1
            ]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value

        try:

                return evaluate_erdos_solution(h, dx)

                return np.sum(h) - n_points / 2.0
                
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_refined, method='SLSQP',

                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            # If SLSQP fails, continue with best known
            pass
            
    return (list(best_h), best_c5, n_points)
