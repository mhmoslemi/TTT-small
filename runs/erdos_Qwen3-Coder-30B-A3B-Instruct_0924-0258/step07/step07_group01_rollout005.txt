import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from the provided construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use SLSQP with refined approach to minimize C5
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a genetic algorithm approach to direct local search
    # Initialize population
    pop_size = 30
    population = []
    for _ in range(pop_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evaluate initial population
    fitness = [objective(h) for h in population]
    best_idx = np.argmin(fitness)
    g_best = population[best_idx].copy()
    g_best_value = fitness[best_idx]
    
    # Evolution parameters
    max_iter = 500
    elite_size = 5
    mutation_rate = 0.1

    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]

        # Keep elite individuals
        elite = population[:elite_size]
        
        # Create new population through crossover and mutation
        new_population = elite[:]
        for _ in range(pop_size - elite_size):
            # Selection (tournament)
            tournament_size = 3
            selected_indices = np.random.choice(len(population), tournament_size)
            winner_idx = selected_indices[np.argmin([fitness[i] for i in selected_indices])]
            
            # Crossover (single-point)
            parent = population[winner_idx]
            if np.random.rand() < 0.7:
                another_idx = np.random.choice(len(population), p=[1.0/len(population)]*len(population))
                another = population[another_idx]
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent[:crossover_point], another[crossover_point:]])
            else:
                child = parent.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.clip(child[i] + np.random.normal(0, 0.05), 0, 1)
            
            # Normalize
            child = child * (n_points/2) / np.sum(child)
            new_population.append(child)
        
        population = new_population
        
        # Re-evaluate
        fitness = [objective(h) for h in population]
        
        # Update global best
        new_best_idx = np.argmin(fitness)
        if fitness[new_best_idx] < g_best_value:
            g_best = population[new_best_idx].copy()
            g_best_value = fitness[new_best_idx]
            
    # Final SLSQP refinement
    def objective_slsqp(h):
        return evaluate_erdos_solution(h, dx)
    
    try:
        res = minimize(objective_slsqp, g_best, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective_slsqp(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
