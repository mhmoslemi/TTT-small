import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Periodic initialization
        n_points = 20
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
        # Add slight perturbations
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multiresolution approach
    resolutions = [20, 40, 80]
    current_h = h_start.copy()
    current_n = n_points
    
    for res_idx, target_n in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to target resolution
        if target_n > current_n:
            x_old = np.linspace(0, 2, current_n)
            x_new = np.linspace(0, 2, target_n)
            current_h = np.interp(x_new, x_old, current_h)
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (target_n / 2) / np.sum(current_h)
        elif target_n < current_n:
            # Downsample
            current_h = current_h[::current_n//target_n]
            current_h = current_h * (target_n / 2) / np.sum(current_h)
        
        current_n = target_n
        dx = 2.0 / current_n
        
        # Simulated Annealing parameters
        max_iter = 100 if res_idx == 0 else 150 if res_idx == 1 else 200
        temp = 0.5
        temp_decay = 0.99
        temp_min = 1e-6
        
        # Tabu list to prevent cycling
        tabu_list = []
        tabu_size = 20
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Generate candidate
            candidate_h = current_h.copy()
            n_perturb = max(1, current_n // 10)
            indices = np.random.choice(current_n, n_perturb, replace=False)
            
            # Perturb selected points
            for i in indices:
                delta = np.random.normal(0, 0.15 * (1 - candidate_h[i]) if candidate_h[i] < 0.5 else 0.15 * candidate_h[i])
                candidate_h[i] += delta
            
            # Enforce bounds
            candidate_h = np.clip(candidate_h, 0, 1)
            
            # Normalize
            candidate_h = candidate_h * (current_n / 2) / np.sum(candidate_h)
            
            # Check tabu list
            candidate_tuple = tuple(np.round(candidate_h, 6))
            if candidate_tuple in tabu_list:
                continue
                
            # Evaluate
            c5_candidate = evaluate_erdos_solution(candidate_h, dx)
            
            # Accept with probability
            if c5_candidate < best_c5:
                current_h = candidate_h.copy()
                best_h = candidate_h.copy()
                best_c5 = c5_candidate
                
                # Update tabu list
                tabu_list.append(candidate_tuple)
                if len(tabu_list) > tabu_size:
                    tabu_list.pop(0)
                
                # Reset temperature
                temp = 0.5
            else:
                # Accept with probability based on temperature
                delta = c5_candidate - best_c5
                if delta > 0 and np.random.random() < math.exp(-delta / temp):
                    current_h = candidate_h.copy()
                    
                    # Update tabu list
                    tabu_list.append(candidate_tuple)
                    if len(tabu_list) > tabu_size:
                        tabu_list.pop(0)
                
                # Cool down temperature
                temp = max(temp_min, temp * temp_decay)
        
        current_h = best_h.copy()
    
    # Final refinement using SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - current_n / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(current_n)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, current_n)
