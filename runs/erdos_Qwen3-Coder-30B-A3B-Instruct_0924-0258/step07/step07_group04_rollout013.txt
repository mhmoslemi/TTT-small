import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a better initial guess using periodic function
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))  # 5 cycles over [0,2]
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    # Parameters
    T = 0.5           # Initial temperature
    alpha = 0.995     # Cooling rate
    T_min = 1e-5      # Minimum temperature
    max_iter = 1000   # Maximum iterations
    tabu_list_size = 20
    perturbation_start = 0.1
    perturbation_end = 0.05
    stagnation_limit = 50
    
    # Tabu list to store recent solutions (as hashes)
    tabu_list = []
    tabu_tolerance = 1e-6
    
    # Track best solution history
    best_history = []
    
    iteration = 0
    stagnation_count = 0
    current_h = best_h.copy()
    current_c5 = best_c5
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iter:
        # Adapt perturbation size
        perturbation_size = perturbation_start + (perturbation_end - perturbation_start) * (iteration / max_iter)
        
        # Select random indices to modify (15% of points)
        num_modify = max(1, int(0.15 * n_points))
        indices = np.random.choice(n_points, num_modify, replace=False)
        
        # Create candidate solution
        candidate_h = current_h.copy()
        for i in indices:
            # Add perturbation
            delta = np.random.uniform(-perturbation_size, perturbation_size)
            candidate_h[i] += delta
        
        # Enforce bounds
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Normalize to satisfy integral constraint
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Accept or reject based on SA criteria
        if candidate_c5 < current_c5:
            # Always accept better solution
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            stagnation_count = 0
        else:
            # Accept with probability based on temperature
            delta = candidate_c5 - current_c5
            prob = math.exp(-delta / T)
            if np.random.random() < prob:
                current_h = candidate_h.copy()
                current_c5 = candidate_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            best_history.append((current_c5, time() - start_time))
            stagnation_count = 0
        
        # Update tabu list
        # Use a hash to represent solution similarity
        sol_hash = hash(tuple(np.round(current_h, decimals=6)))
        tabu_list.append(sol_hash)
        if len(tabu_list) > tabu_list_size:
            tabu_list.pop(0)
        
        # If stagnated, perturb best solution
        if stagnation_count > stagnation_limit:
            # Randomly perturb best solution
            perturb_indices = np.random.choice(n_points, max(1, int(0.1 * n_points)), replace=False)
            for i in perturb_indices:
                delta = np.random.uniform(-0.1, 0.1)
                best_h[i] += delta
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            current_h = best_h.copy()
            current_c5 = evaluate_erdos_solution(current_h, dx)
            stagnation_count = 0
        
        # Cooling
        T = max(T * alpha, T_min)
        iteration += 1
    
    # Final refinement using SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        h_refined = np.clip(best_h, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final verification
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
