import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Initialize with periodic function
        x = np.linspace(0, 2, 35)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (35 / 2) / np.sum(h_start)
        h_start *= normalization
        n_points = 35
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    # Parameters
    max_iter = 10000
    T = 0.5
    alpha = 0.995
    T_min = 1e-5
    tabu_list = []
    tabu_size = 20
    stagnation_threshold = 50
    stagnation_count = 0
    current_h = best_h.copy()
    current_c5 = best_c5
    perturbation_size = 0.1
    
    # Main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate solution (perturb 15% of indices)
        candidate = current_h.copy()
        indices = np.random.choice(n_points, size=int(n_points * 0.15), replace=False)
        for i in indices:
            perturb = np.random.uniform(-perturbation_size, perturbation_size)
            candidate[i] += perturb
        
        # Enforce bounds and normalization
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * (n_points / 2) / np.sum(candidate)
        
        # Check tabu list
        is_tabu = False
        for tabu_h in tabu_list:
            if np.allclose(candidate, tabu_h, atol=1e-6):
                is_tabu = True
                break
        
        # Evaluate candidate
        new_c5 = evaluate_erdos_solution(candidate, dx)
        
        # Acceptance criterion
        accept = False
        if new_c5 < current_c5:
            accept = True
        elif not is_tabu:
            # Accept with probability
            delta = new_c5 - current_c5
            if np.random.rand() < math.exp(-delta / T):
                accept = True
        
        # Update best solution
        if accept and new_c5 < best_c5:
            best_h = candidate.copy()
            best_c5 = new_c5
        
        # Update current solution
        if accept:
            current_h = candidate.copy()
            current_c5 = new_c5
            
            # Update tabu list
            if len(tabu_list) >= tabu_size:
                tabu_list.pop(0)
            tabu_list.append(candidate.copy())
            
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Adaptive cooling
        T = max(T * alpha, T_min)
        
        # Stagnation restart
        if stagnation_count >= stagnation_threshold:
            # Perturb best solution slightly
            restart_h = best_h.copy()
            perturbation = np.random.uniform(-0.05, 0.05, n_points)
            restart_h += perturbation
            restart_h = np.clip(restart_h, 0, 1)
            restart_h = restart_h * (n_points / 2) / np.sum(restart_h)
            
            # Re-evaluate
            restart_c5 = evaluate_erdos_solution(restart_h, dx)
            if restart_c5 < current_c5:
                current_h = restart_h.copy()
                current_c5 = restart_c5
            
            stagnation_count = 0
            perturbation_size = max(0.05, perturbation_size * 0.95)
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
