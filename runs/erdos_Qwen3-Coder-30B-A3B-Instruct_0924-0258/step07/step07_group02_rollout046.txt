import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    pop_size = 50
    max_iter = 200
    inertia = 0.7
    cognitive = 1.4
    social = 1.4
    velocity_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for i in range(pop_size):
        # Perturb initial solution
        noise = np.random.uniform(-0.1, 0.1, n_points)
        h = best_h + noise
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
        c5 = evaluate_erdos_solution(h, dx)
        pbest.append(h.copy())
        pbest_c5.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # PSO iterations
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample critical lags based on recent performance
        if iter_num % 10 == 0:
            # Get top 10% lags based on latest particle overlaps
            sample_lags = set()
            for h in particles[:10]:
                corr = np.correlate(h, 1 - h, mode='full')
                top_k = np.argsort(corr)[-n_points//10:]
                sample_lags.update(top_k)
            sample_lags = list(sample_lags)
        else:
            sample_lags = list(range(1, n_points))
        
        for i in range(pop_size):
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            cognitive_force = cognitive * r1 * (pbest[i] - particles[i])
            social_force = social * r2 * (best_h - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive_force + social_force
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to constraint sum(h) = n_points/2
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
                
                if c5 < best_c5:
                    best_h = particles[i].copy()
                    best_c5 = c5
    
    # Final SQP refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add compatibility constraints based on best solution so far
        h_test = best_h.copy()
        corr = np.correlate(h_test, 1 - h_test, mode='full')
        max_k_idx = np.argmax(corr)
        max_lag = max_k_idx - (n_points - 1)  # actual lag
        # Add key constraints that are most active
        sample_lags = np.sort(np.argsort(corr)[-10:])  # top 10 lags
        for k in sample_lags:
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
