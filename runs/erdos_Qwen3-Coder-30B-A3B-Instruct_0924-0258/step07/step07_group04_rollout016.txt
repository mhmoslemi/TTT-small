import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Define constraints for differential evolution
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # DE optimization
    try:
        result_de = differential_evolution(
            objective,
            bounds,
            constraints=cons_sum,
            maxiter=500,
            popsize=50,
            tol=1e-6,
            recombination=0.9,
            mutation=(0.5, 1.0),
            seed=seed,
            callback=lambda x, convergence: None,
            disp=False
        )
        
        if result_de.success:
            h_de = result_de.x
            c5_de = objective(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    except Exception:
        pass
    
    # Local refinement with SLSQP
    try:
        def objective_slsqp(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum_slsqp(h):
            return np.sum(h) - n_points / 2.0
        
        cons_slsqp = {'type': 'eq', 'fun': constraint_sum_slsqp}
        bounds_slsqp = [(0, 1) for _ in range(n_points)]
        
        # Use best from DE or initial as starting point
        h_refined = np.clip(best_h, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        res = minimize(
            objective_slsqp,
            h_refined,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons_slsqp,
            options={'maxiter': 1000, 'ftol': 1e-10}
        )
        
        if res.success:
            h_final = res.x
            c5_final = objective_slsqp(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Policy-based refinement if needed
    if time() - start_time < budget_s * 0.9 and best_c5 > 0.3807:
        # Try a structured approach: piecewise constant if better
        def try_piecewise():
            # Simple greedy approach: try making a piecewise constant with 4 segments
            n_seg = 4
            seg_len = n_points // n_seg
            remainder = n_points % n_seg
            
            h_try = np.zeros(n_points)
            
            # Distribute mass
            total_mass = n_points / 2.0
            mass_per_seg = total_mass / n_seg
            
            pos = 0
            for i in range(n_seg):
                # Adjust segment size to accommodate remainder (more at start)
                seg_size = seg_len + (1 if i < remainder else 0)
                h_try[pos:pos+seg_size] = mass_per_seg / seg_size
                pos += seg_size
            
            # Normalize
            h_try = np.clip(h_try, 0, 1)
            h_try = h_try * (n_points/2) / np.sum(h_try)
            return h_try
        
        h_piecewise = try_piecewise()
        c5_piecewise = evaluate_erdos_solution(h_piecewise, dx)
        if c5_piecewise < best_c5:
            best_h = h_piecewise
            best_c5 = c5_piecewise
    
    return (list(best_h), best_c5, n_points)
