import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 100
    max_generations = 300
    elite_size = pop_size // 10
    mutation_rate = 0.2
    crossover_rate = 0.7
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Track hall of fame
    hall_of_fame = []
    best_h_all = best_h.copy()
    best_c5_all = best_c5
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            
        # Update hall of fame
        indices = np.argsort(fitness)
        for i in range(min(elite_size, len(indices))):
            idx = indices[i]
            h = population[idx]
            c5 = fitness[idx]
            if len(hall_of_fame) < elite_size or c5 < hall_of_fame[-1][0]:
                hall_of_fame.append((c5, h.copy()))
                hall_of_fame.sort(key=lambda x: x[0])
                hall_of_fame = hall_of_fame[:elite_size]
        
        # Check for improvement in hall of fame
        if hall_of_fame and hall_of_fame[0][0] < best_c5_all:
            best_c5_all = hall_of_fame[0][0]
            best_h_all = hall_of_fame[0][1].copy()
        
        # Selection - tournament
        selected = []
        for _ in range(pop_size):
            tournament_size = 3
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
            
        # Crossover
        offspring = []
        for i in range(0, pop_size, 2):
            if np.random.rand() < crossover_rate and i + 1 < pop_size:
                parent1 = selected[i]
                parent2 = selected[i+1]
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
                offspring.extend([child1, child2])
            else:
                offspring.extend([selected[i], selected[i+1] if i+1 < pop_size else selected[i]])
                
        # Mutation with adaptive variance
        adaptive_variance = 0.1 * (0.9 ** (generation // 50))
        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                # Mutate 20% of the elements
                num_mutate = max(1, int(n_points * 0.2))
                indices = np.random.choice(n_points, num_mutate, replace=False)
                for idx in indices:
                    # Gaussian mutation
                    offspring[i][idx] += np.random.normal(0, adaptive_variance)
                # Ensure bounds and rescale
                offspring[i] = np.clip(offspring[i], 0, 1)
                offspring[i] = offspring[i] * (n_points/2) / np.sum(offspring[i])
        
        # Replace population with offspring
        population = offspring[:pop_size]
    
    # Final refinement using L-BFGS-B
    try:
        # Take top solutions from hall of fame
        for c5, h_refine in hall_of_fame[:5]:
            bounds = [(0, 1) for _ in range(n_points)]
            result = minimize(
                lambda h: evaluate_erdos_solution(h, dx),
                h_refine,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2}),
                options={'maxiter': 100}
            )
            if result.success:
                h_final = result.x
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5_all:
                    best_c5_all = c5_final
                    best_h_all = h_final
    except:
        pass
    
    # Final projection to ensure constraints
    best_h_all = np.clip(best_h_all, 0, 1)
    best_h_all = best_h_all * (n_points/2) / np.sum(best_h_all)
    best_c5_all = evaluate_erdos_solution(best_h_all, dx)
    
    return (list(best_h_all), best_c5_all, n_points)
