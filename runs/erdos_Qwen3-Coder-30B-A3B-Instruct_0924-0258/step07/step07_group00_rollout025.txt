from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient-based optimization with adaptive step size
    max_iter = 10000
    step_size = 0.01
    tolerance = 1e-8
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps and find maximum
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        max_overlap = corr[max_lag_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Compute gradient with respect to h for the max overlap lag
        gradient = np.zeros(n_points)
        # Create shifted versions of h for computing discrete gradient
        if max_lag >= 0:
            # Right shift by max_lag
            shifted_h = np.pad(h[:-max_lag], (max_lag, 0), mode='constant', constant_values=0)
            gradient = 1 - 2 * shifted_h
        else:
            # Left shift by |max_lag|
            shifted_h = np.pad(h[-max_lag:], (0, -max_lag), mode='constant', constant_values=0)
            gradient = 1 - 2 * shifted_h
            
        # Update h in gradient descent direction
        h_new = h - step_size * gradient
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check constraint satisfaction
        if not (0 <= np.min(h_new) and np.max(h_new) <= 1):
            step_size *= 0.5
            continue
            
        # Check that the sum constraint is maintained
        if abs(np.sum(h_new) - n_points / 2) > 1e-6:
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
        # Update h
        h = h_new
        
        if iteration % 100 == 0:
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
