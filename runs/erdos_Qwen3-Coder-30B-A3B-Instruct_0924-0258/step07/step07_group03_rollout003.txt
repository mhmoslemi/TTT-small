import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based optimization
    max_iter = 1000
    lr = 0.1
    stagnation_threshold = 50
    stagnation_count = 0
    
    # Random restarts
    num_restarts = 5
    for restart in range(num_restarts):
        if restart > 0:
            # Perturb the current best solution
            h = best_h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        else:
            h = best_h.copy()
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.8:
                break
                
            # Compute all overlaps
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            max_overlap = corr[max_idx] * dx
            
            # Find which lag gives us the maximum overlap
            lag = max_idx - (n_points - 1)
            
            # Compute gradient for this specific lag
            grad = np.zeros_like(h)
            for i in range(n_points):
                if i + lag >= 0 and i + lag < n_points:
                    grad[i] = 1 - h[i + lag]
                else:
                    grad[i] = 0
            
            # Update h using gradient descent
            h_new = h - lr * grad
            
            # Project back to feasible region
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Update h
            h = h_new
            
            # Evaluate new solution
            new_c5 = evaluate_erdos_solution(h, dx)
            
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
                
            if stagnation_count > stagnation_threshold:
                lr *= 0.5
                stagnation_count = 0
                if lr < 1e-6:
                    break
    
    # Phase 2: Local refinement with L-BFGS-B
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='L-BFGS-B', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
