import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a smoother construction
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(np.pi * x))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 100
    max_gen = 250
    mutation_rate = 0.2
    crossover_rate = 0.7
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if i == 0 and 'initial_h_values' in globals():
            h = h_start.copy()
        elif i < pop_size // 2:
            # Random samples
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        else:
            # Latin hypercube like samples
            h = np.random.rand(n_points)
            idx = np.random.permutation(n_points)
            h[idx[:n_points//2]] = 1
            h[idx[n_points//2:]] = 0
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
        f = evaluate_erdos_solution(h, dx)
        fitness.append(f)
        if f < best_c5:
            best_h = h.copy()
            best_c5 = f
    
    # Track hall of fame
    hall_of_fame = [(best_h.copy(), best_c5)]
    
    # Evolutionary loop
    for gen in range(max_gen):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]
        fitness = [fitness[i] for i in sorted_idx]
        
        # Update hall of fame
        if fitness[0] < hall_of_fame[0][1]:
            hall_of_fame = [(population[0].copy(), fitness[0])]
        
        # Selection: top 20% advance
        elite_count = max(10, pop_size // 10)
        next_gen = population[:elite_count]
        next_fitness = fitness[:elite_count]
        
        # Crossover and Mutation
        while len(next_gen) < pop_size:
            # Tournament selection
            a_idx = np.random.choice(elite_count)
            b_idx = np.random.choice(elite_count)
            parent_a = population[a_idx]
            parent_b = population[b_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.uniform(0.2, 0.8)
                child = alpha * parent_a + (1 - alpha) * parent_b
            else:
                child = parent_a.copy()
            
            # Mutation
            mutation_mask = np.random.rand(n_points) < mutation_rate
            if gen > 50:  # decrease mutation rate
                mutation_scale = 0.05 * (1 - gen / max_gen)
            else:
                mutation_scale = 0.1
            child[mutation_mask] += np.random.normal(0, mutation_scale, np.sum(mutation_mask))
            
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            next_gen.append(child)
            next_fitness.append(evaluate_erdos_solution(child, dx))
        
        population = next_gen
        fitness = next_fitness
    
    # Local refinement using SQP
    try:
        # Start with the best solution
        h_refine = best_h.copy()
        max_iter = 50
        for _ in range(max_iter):
            # Optimize with constraints using cvxpy
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add constraints for key lags
            h_test = h_refine.copy()
            corr = np.correlate(h_test, 1 - h_test, mode='full')
            # Sample key lags from top 30% in corpus
            top_lags = np.argsort(corr)[-n_points//3:]
            
            for k in top_lags[:15]:  # restrict number of constraints
                if k < n_points - 1:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-5)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-7, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    new_c5 = evaluate_erdos_solution(h_feasible, dx)
                    if new_c5 < best_c5:
                        best_h = h_feasible
                        best_c5 = new_c5
                        h_refine = h_feasible.copy()
            except:
                pass
    except:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
