import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive Augmented Lagrangian Method
    max_iter = 5000
    lambda_lag = 0.0
    mu_penalty = 100.0
    alpha = 0.1
    convergence_tol = 1e-6
    trust_region_factor = 1.1
    
    def compute_gradient(h, active_k):
        # Compute gradient with respect to the active constraint
        grad = np.zeros_like(h)
        for k in active_k:
            if k == 0:
                grad += 1 - h
            else:
                # Gradient of h_i * (1 - h_{i+k}) w.r.t h_i
                # = 1 - h_{i+k} for i < n-k, 0 elsewhere
                # And -h_i for h_{i+k} derivative
                if k < len(h):
                    if k == 0:
                        grad += (1 - h)
                    else:
                        # Create sparse gradient for computing max overlap constraint
                        for i in range(len(h) - k):
                            grad[i] += 1 - h[i + k]
                            if i + k < len(h):
                                grad[i + k] -= h[i]
        return grad
    
    def project_and_scale(h, target_sum):
        # Project onto [0,1] then scale to sum=target_sum
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum) > 1e-12:
            h = h * target_sum / h_sum
        else:
            h = np.ones_like(h) * target_sum / len(h)
        return h
    
    def get_active_k(h, dx):
        # Find all active k values
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        active_ks = []
        for i, val in enumerate(corr):
            if abs(val - max_val) < 1e-6:
                # Convert index back to k
                k = i - (len(h) - 1)
                if k >= 0:
                    active_ks.append(k)
                else:
                    active_ks.append(len(h) + i - (len(h) - 1))
        return active_ks
    
    def adaptive_alm_step(h, lambda_lag, mu_penalty, alpha):
        # Compute current overlap
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        
        # Identify active constraints
        active_k = []
        for i, val in enumerate(corr):
            if abs(val - max_val) < 1e-6:
                k = i - (len(h) - 1)
                if k >= 0:
                    active_k.append(k)
                else:
                    active_k.append(len(h) + i - (len(h) - 1))
        
        # Compute gradient
        grad = compute_gradient(h, active_k)
        grad = grad * dx  # scale
        
        # Lagrangian gradient
        lag_grad = grad + lambda_lag
        
        # Line search
        step_size = alpha
        for _ in range(20):
            h_new = h - step_size * lag_grad
            h_new = project_and_scale(h_new, n_points/2)
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            # Accept step if improvement
            if new_c5 < best_c5 or np.random.rand() < 0.1:
                h = h_new
                break
            else:
                step_size *= 0.5
                
        return h, lag_grad
    
    # Main ALM loop
    last_c5 = best_c5 + 1e6  # Initialize to ensure loop starts
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Adaptive update
        h_candidate, lag_grad = adaptive_alm_step(best_h, lambda_lag, mu_penalty, alpha)
        
        # Evaluate new solution
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Update best
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
        
        # Update Lagrange multiplier
        lambda_lag += mu_penalty * (np.sum(h_candidate) - n_points/2)
        
        # Update penalty parameter
        if abs(np.sum(h_candidate) - n_points/2) > 1e-3:
            mu_penalty *= 2.0  # Increase penalty if constraint violated
        else:
            mu_penalty *= 0.9  # Decrease otherwise
        
        # Trust region update
        alpha *= trust_region_factor  # Expand trust region
        if abs(last_c5 - best_c5) < convergence_tol:
            break
        last_c5 = best_c5
        
        # Periodic polishing
        if iteration % 100 == 0 and iteration > 0:
            try:
                h_refined = minimize(
                    lambda x: evaluate_erdos_solution(project_and_scale(x, n_points/2), dx),
                    best_h,
                    method='L-BFGS-B',
                    bounds=[(0, 1) for _ in range(n_points)],
                    options={'maxiter': 10}
                )
                if h_refined.success:
                    h_polished = project_and_scale(h_refined.x, n_points/2)
                    c5_polished = evaluate_erdos_solution(h_polished, dx)
                    if c5_polished < best_c5:
                        best_h = h_polished
                        best_c5 = c5_polished
            except Exception:
                pass
    
    # Final polishing
    try:
        h_final = minimize(
            lambda x: evaluate_erdos_solution(project_and_scale(x, n_points/2), dx),
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 50}
        )
        if h_final.success:
            final_h = project_and_scale(h_final.x, n_points/2)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
