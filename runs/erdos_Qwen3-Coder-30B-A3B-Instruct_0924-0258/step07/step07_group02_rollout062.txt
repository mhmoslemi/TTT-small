import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a balanced starting solution
        h_start = np.ones(n_points) * 0.5
        h_start[:n_points//2] = 1.0
        h_start[n_points//2:] = 0.0
        # Normalize
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm parameters
    pop_size = 100
    max_generations = 250
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = 10
    tournament_size = 5
    
    population = []
    fitness = []
    
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Initialization
    for _ in range(pop_size):
        if np.random.rand() < 0.5:
            # Random uniform with normalization
            h = np.random.rand(n_points)
            h = repair_h(h)
        else:
            # Latin hypercube-like approach
            h = np.random.rand(n_points)
            h = np.sort(h)
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
        f = evaluate_erdos_solution(h, dx)
        fitness.append(f)
    
    # Track best so far
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Adaptive mutation
    mutation_var = 0.1
    generation = 0
    
    # Main EA loop
    no_improve_count = 0
    
    while generation < max_generations and time() - start_time < budget_s * 0.95:
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        # Track best
        if sorted_fit[0] < best_c5:
            best_h = sorted_pop[0].copy()
            best_c5 = sorted_fit[0]
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count > 80:
            break
            
        # Create new population
        new_pop = []
        new_fit = []
        
        # Elitism
        for i in range(elite_size):
            new_pop.append(sorted_pop[i].copy())
            new_fit.append(sorted_fit[i])
        
        # Generation with crossover and mutation
        for _ in range(pop_size - elite_size):
            # Tournament selection
            selected_indices = np.random.choice(pop_size, tournament_size, replace=False)
            selected_fitness = [fitness[i] for i in selected_indices]
            winner_idx = selected_indices[np.argmin(selected_fitness)]
            
            # Clone parent
            child = population[winner_idx].copy()
            
            # Crossover with another random individual
            if np.random.rand() < crossover_rate:
                other_idx = np.random.choice(pop_size, 1)[0]
                alpha = np.random.rand()
                child = alpha * child + (1 - alpha) * population[other_idx]
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Apply Gaussian noise
                child += np.random.normal(0, mutation_var, n_points)
                # Bit-flip mutation on low values
                flip_idx = np.random.choice(n_points, int(0.2 * n_points), replace=False)
                for i in flip_idx:
                    if child[i] < 0.5:
                        child[i] = 1.0 - child[i]
            
            # Ensure feasibility
            child = repair_h(child)
            
            # Evaluation
            c5 = evaluate_erdos_solution(child, dx)
            new_pop.append(child)
            new_fit.append(c5)
            
        population = new_pop
        fitness = new_fit
        
        # Update mutation variance
        generation += 1
        if generation % 50 == 0:
            mutation_var *= 0.9
            
        # Occasionally add a new random individual
        if np.random.rand() < 0.05:
            new_h = np.random.rand(n_points)
            new_h = repair_h(new_h)
            new_c5 = evaluate_erdos_solution(new_h, dx)
            population[-1] = new_h
            fitness[-1] = new_c5
    
    # Final local refinement with L-BFGS
    try:
        best_h = repair_h(best_h)
        def obj_func(h):
            return evaluate_erdos_solution(h, dx)
        
        res = minimize(obj_func, best_h, method='L-BFGS-B', 
                      bounds=[(0, 1)] * n_points,
                      options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-6})
        if res.success:
            refined_h = res.x
            refined_h = repair_h(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    # Final normalization and verification
    best_h = repair_h(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
