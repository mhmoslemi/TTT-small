import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialization
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA Parameters
    pop_size = 50
    max_generations = 200
    mutation_rate = 0.05
    crossover_rate = 0.7
    elite_count = 5
    
    # Population initialization
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evolutionary Loop
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create next generation
        new_population = [population[i] for i in range(elite_count)]
        
        # Crossover and Mutation
        while len(new_population) < pop_size:
            # Tournament selection
            idx1 = np.random.randint(0, pop_size//2)
            idx2 = np.random.randint(0, pop_size//2)
            h1 = population[idx1]
            h2 = population[idx2]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = alpha * h1 + (1 - alpha) * h2
            else:
                child = h1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            new_population.append(child)
        
        population = new_population
        
        # Adapt mutation rate
        if gen > 0 and gen % 20 == 0:
            mutation_rate = max(0.005, mutation_rate * 0.8)
    
    # Final SQP refinement for top solutions
    top_solutions = population[:10]
    for h in top_solutions:
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Direct constraint for key overlaps
            h_test = h.copy()
            corr = np.correlate(h_test, 1 - h_test, mode='full')
            top_k = np.argsort(corr)[-10:]
            for k in top_k:
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
