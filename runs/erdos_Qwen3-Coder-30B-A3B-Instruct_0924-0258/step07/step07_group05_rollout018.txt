import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Benders decomposition approach
    max_iterations = 5000
    convergence_tol = 1e-6
    perturb_mag = 0.05
    perturb_decay = 0.95
    
    h_prev = best_h.copy()
    c5_prev = best_c5
    
    for iter_num in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all possible shifts
        overlaps = np.correlate(best_h, 1 - best_h, mode='full')
        k_indices = np.arange(len(overlaps))
        # We are interested in valid shifts k in [0, n_points-1]
        # But compute only in [-n_points+1, n_points-1]
        # For our case, we consider only positive shifts that make sense
        # The full correlation length is 2*n_points - 1
        # valid k values range from 0 to n_points-1 (but can go up to 2*n_points-2)
        # But for k>0 we look at correlation at shift k, i.e., h[i]* (1-h[i+k])
        # k range from 0 to n_points-1 (inclusive) 
        # To avoid double counting, we'll work with positive only 
        # But we'll only take k from 0 to n_points-1 that exist in correlation
        valid_k = np.arange(0, n_points)
        # Since overlap for k values where i+k >= n_points is zeros due to boundary, 
        # The actual valid overlaps for shift k (0 to n_points-1) are at positions
        # n_points-1+k (or 0 <= k < n_points for valid correlations)
        # Also, because we're computing mode='full', all entries are valid 
        # Normalized overlaps for each k:
        # The overlap at k = sum(h[i] * (1 - h[i+k])) for valid i
        # For each k, number of valid i = n_points - k
        
        # Extract the relevant correlation values for k = 0 to n_points-1
        # Overlaps at full correlation include indices from [-n_points+1, n_points-1]
        # actual_k is shift, i.e., overlap at k = overlaps[n_points-1+k] 
        # which is for shift k=0, pos n_points-1; k=1, pos n_points; etc.
        valid_overlaps = [overlaps[n_points-1+k] * dx for k in range(n_points)]
        k_star = np.argmax(valid_overlaps)
        max_overlap = valid_overlaps[k_star]
        
        # Add a constraint based on the k_star to the master problem (Benders)
        # This is the key idea: add constraints for only the most constraining k
        
        # Solve a convex relaxation with one constraint added:
        try:
            h_relax = cp.Variable(n_points)
            constraints = [
                cp.sum(h_relax) == n_points / 2.0,
                h_relax >= 0,
                h_relax <= 1
            ]
            
            # Add the active constraint based on the current solution
            # For alignment, use the exact form of overlap that we just computed
            # Which is sum_{i=0}^{n_points-1-k_star} h[i] * (1 - h[i+k_star])
            if k_star < n_points:
                # Form the constraint for this k
                expr = cp.sum(cp.multiply(h_relax[:n_points-k_star], 1 - h_relax[k_star:]))
                constraints.append(expr <= max_overlap + 1e-8)  # We add a buffer like before
            
            # Additional strengthening - use previously obtained best
            if best_c5 < np.inf:
                expr = cp.sum(cp.multiply(h_relax[:n_points-k_star], 1 - h_relax[k_star:]))
                constraints.append(expr <= best_c5 * dx + 1e-8)
            
            # Objective: minimize something that's not the max_overlap itself, 
            # but the norm or 0 to allow flexible solver (for actual relaxation to work).
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_relax.value
                # Clip the h to [0,1] for numerical stability
                h_new = np.clip(h_new, 0, 1)
                # Rescale to maintain constraint
                h_new = h_new * (n_points/2) / np.sum(h_new)
                h_new = np.array(h_new)
                
                h_new = h_new + perturb_mag * np.random.randn(n_points)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    # print(f"Improvement: {best_c5:.6f} at iteration {iter_num}")
                
                # Update perturbation magnitude
                perturb_mag *= perturb_decay
                if abs(c5_new - c5_prev) < convergence_tol:
                    pass  # Convergence check could go here but it's noisy
                c5_prev = c5_new
                h_prev = h_new.copy()
                
        except Exception as e:
            # Fallback: simple random perturbation
            h_new = best_h + perturb_mag * np.random.randn(n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            
            perturb_mag *= perturb_decay
            c5_prev = c5_new
    
    # Final polishing with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
