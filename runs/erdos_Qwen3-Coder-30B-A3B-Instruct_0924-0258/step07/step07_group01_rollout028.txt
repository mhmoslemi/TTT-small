import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Make sure h_start satisfies constraint
    h_start = h_start * (n_points/2) / np.sum(h_start)
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Iterative optimization using CVXPY
    max_iter = 20
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Define CVXPY problem
        h = cp.Variable(n_points)
        
        # Objective: upper bound on max overlap
        # We'll use a conservative estimate and iterate to tighten
        max_overlap = cp.Variable()
        
        # Constraint that h satisfies the integral and bounds
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points/2]
        
        # We intentionally simplify; we'll use a different approach 
        # to get the true max overlap from the correlation
        
        # Brute-force compute full correlation max using numpy in a way that is
        # consistent with the evaluation function
        def eval_constraint(h_vals):
            corr = np.correlate(h_vals, 1 - h_vals, mode='full')
            max_corr = np.max(corr)
            return max_corr
        
        # Instead of trying to formulate this in CVXPY directly,
        # use a simpler strategy with SLSQP refinement after CVXPY iterations
        
        # Refine with scipy
        def objective(h_vals):
            return evaluate_erdos_solution(h_vals, dx)
        
        def constraint_sum(h_vals):
            return np.sum(h_vals) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Start with current best point
        h_initial = best_h.copy()
        
        # SLSQP optimization
        try:
            res = minimize(objective, h_initial, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 300, 'ftol': 1e-10})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
                    
        except Exception:
            pass
        
        # Simple local search strategy: perturb and evaluate
        for _ in range(3):
            if time() - start_time > budget_s * 0.95:
                break
            h_perturb = best_h + np.random.normal(0, 0.02, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
            c5_test = evaluate_erdos_solution(h_perturb, dx)
            if c5_test < best_c5:
                best_h = h_perturb.copy()
                best_c5 = c5_test
    
    return (list(best_h), best_c5, n_points)
