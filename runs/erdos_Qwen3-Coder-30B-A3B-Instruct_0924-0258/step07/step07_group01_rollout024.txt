import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 200
    mutation_rate = 0.3
    elite_size = 20
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    fitness = []
    
    # Start with initial solution
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points/2) / np.sum(h_start)
    population.append(h_start.copy())
    
    # Fill rest of population
    for i in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
    
    stagnation_count = 0
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > 50:
            break
            
        # Create new population
        new_population = []
        # Elitism
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Selection (tournament)
            tournament_size = 3
            indices = np.random.choice(population_size, tournament_size)
            best_idx = indices[np.argmin([fitness[i] for i in indices])]
            parent1 = population[best_idx]
            
            if np.random.random() < crossover_rate and len(new_population) + 1 < population_size:
                indices = np.random.choice(population_size, tournament_size)
                best_idx = indices[np.argmin([fitness[i] for i in indices])]
                parent2 = population[best_idx]
                # Arithmetic crossover
                alpha = np.random.random()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points/2) / np.sum(child1)
                new_population.append(child1)
                
                if len(new_population) < population_size:
                    child2 = (1 - alpha) * parent1 + alpha * parent2
                    child2 = np.clip(child2, 0, 1)
                    child2 = child2 * (n_points/2) / np.sum(child2)
                    new_population.append(child2)
            else:
                # Mutation
                child = parent1.copy()
                mutated_indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
                for i in mutated_indices:
                    child[i] += np.random.normal(0, 0.1)
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                new_population.append(child)
                
        population = new_population
    
    # Local refinement using SLSQP on best solutions
    for _ in range(5):  # Try refining top candidates
        if time() - start_time > budget_s * 0.95:
            break
            
        # Get top 5 candidates
        top_candidates = population[:5]
        
        for h_candidate in top_candidates:
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h_candidate, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-10})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_c5 = c5_refined
                        best_h = h_refined
            except Exception:
                pass
    
    return (list(best_h), best_c5, n_points)
