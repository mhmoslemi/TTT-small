import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    # This means we have h(x) = 1 on [0.65, 1.15] and [1.15, 1.65] truncated to [0,2]
    # Actually, better to not let [1.15,1.65] exceed 2, so correct placement
    # Let's make h(x) = 1 on [a, a + 0.5] and [b, b + 0.5] with a,b chosen so that ∫h = 1
    # with overlapping intervals allowed.
    # We'll make a = 0.5, b = 1.0 => lengths 0.5 each, total area = 1.0 → good.
    
    h = np.zeros(n_points)

    a_idx = int(a / dx)
    b_idx = int(b / dx)
    interval_length = int(0.5 / dx)

    start1 = max(0, a_idx)
    end1 = min(n_points, start1 + interval_length)

    start2 = max(0, b_idx)
    end2 = min(n_points, start2 + interval_length)
    h[start2:end2] = 1.0

    integral = np.sum(h) * dx

    def objective(h):
        return evaluate_erdos_solution(h, dx)
