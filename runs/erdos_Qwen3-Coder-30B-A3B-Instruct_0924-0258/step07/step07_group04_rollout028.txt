import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_h(n_points):
    x = np.linspace(0, 2, n_points)
    h_initial = 0.5 * (1 + np.sin(2 * np.pi * x))
    h_initial = np.clip(h_initial, 0, 1)
    normalization = (n_points / 2) / np.sum(h_initial)
    h_initial *= normalization
    return h_initial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = initialize_h(n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.998
    min_T = 1e-8
    max_iter = 5000
    tabu_size = 20
    tabu_list = []
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate
        candidate_h = current_h.copy()
        num_changes = max(1, int(0.15 * n_points))
        indices = np.random.choice(n_points, num_changes, replace=False)
        
        for i in indices:
            perturbation = np.random.normal(0, 0.1)
            candidate_h[i] += perturbation
            candidate_h[i] = np.clip(candidate_h[i], 0, 1)
        
        # Renormalize
        candidate_h = candidate_h * (n_points/2) / np.sum(candidate_h)
        
        # Check tabu
        if any(np.allclose(candidate_h, h, atol=1e-6) for h in tabu_list):
            # Accept with low probability
            if np.random.random() > 0.95:
                pass
            else:
                continue
                
        # Evaluate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5:
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
        else:
            delta = candidate_c5 - current_c5
            prob = np.exp(-delta / T)
            if np.random.random() < prob:
                current_h = candidate_h.copy()
                current_c5 = candidate_c5
        
        # Update best
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Update tabu list
        tabu_list.append(current_h.copy())
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
            
        # Cool down
        T = max(min_T, T * alpha)
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        h_refined = np.clip(best_h, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
