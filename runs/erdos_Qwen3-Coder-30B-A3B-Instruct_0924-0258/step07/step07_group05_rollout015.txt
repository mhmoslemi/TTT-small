import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Benders decomposition parameters
    max_iter = 5000
    convergence_tol = 1e-6
    perturb_mag = 0.05
    perturb_decay = 0.95
    
    # Initialize
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    previous_c5 = float('inf')
    iter_count = 0
    
    # Constraint set for Benders cuts
    constraints = []
    
    while iter_count < max_iter and time() - start_time < budget_s * 0.95:
        # Step 1: Find the current worst-case overlap
        corr = np.correlate(h, 1 - h, mode='full')
        k_indices = np.arange(len(corr))
        max_idx = np.argmax(corr)
        worst_k = k_indices[max_idx]
        max_overlap = corr[max_idx] * dx
        
        # Step 2: Update Benders constraint
        if worst_k not in [c[0] for c in constraints]:
            constraints.append((worst_k, max_overlap))
        
        # Step 3: Solve convex relaxation using CVX
        try:
            h_var = cp.Variable(n_points)
            obj = cp.Minimize(0)  # dummy objective
            cons = [cp.sum(h_var) == n_points / 2.0,
                    h_var >= 0,
                    h_var <= 1]
            
            # Add Benders constraints
            for k, bound in constraints:
                if k < n_points:
                    # Compute correlation contribution for this k
                    corr_term = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
                    cons.append(corr_term <= bound * dx + 1e-8)
            
            prob = cp.Problem(obj, cons)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            
            if prob.status == cp.OPTIMAL:
                h = np.array(h_var.value).flatten()
        except Exception:
            pass
        
        # Step 4: Perturb and project
        h_perturbed = h + perturb_mag * np.random.randn(n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Step 5: Evaluate and update
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Step 6: Update convergence check
        if abs(c5 - previous_c5) < convergence_tol:
            break
        previous_c5 = c5
        iter_count += 1
        perturb_mag *= perturb_decay
    
    # Final polishing with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
