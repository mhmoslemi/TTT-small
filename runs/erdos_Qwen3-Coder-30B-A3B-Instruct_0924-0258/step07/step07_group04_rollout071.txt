import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(2 * np.pi * x)
        h_start = np.clip(h_start, 0, 1)
    
    # Normalize to correct integral
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe style optimization with periodic perturbations
    max_iter = 1000
    gamma = 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag = np.argmax(corr)
        max_corr_val = corr[max_lag]
        
        # Construct descent direction (simplified) - only focus on max_lag
        s = np.zeros_like(best_h)
        # Make sure we're not going to an extreme value
        s = 1 - best_h  # Note: direction to decrease correlation
        
        # Frank-Wolfe step
        h_new = best_h + gamma * s
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Normalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Perturb periodically to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            perturb_strength = 0.05
            perturbation = np.random.normal(0, perturb_strength, n_points)
            best_h = best_h + perturbation
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            
        # Rate decay
        gamma *= 0.995
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
