import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm settings
    pop_size = 100
    num_parents = pop_size // 5
    mutation_rate = 0.3
    max_generations = 200
    no_improvement_limit = 50
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    best_fitness_history = []
    
    generation = 0
    no_improvement = 0
    
    while generation < max_generations and (time() - start_time) < budget_s * 0.95:
        # Evaluate fitness
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        best_fitness_history.append(fitness[sorted_indices[0]])
        
        # Check for improvement
        if len(best_fitness_history) > 1:
            if best_fitness_history[-1] >= best_fitness_history[-2]:
                no_improvement += 1
            else:
                no_improvement = 0
        if no_improvement >= no_improvement_limit:
            break
        
        # Select parents (best 20%)
        parents = [population[i] for i in sorted_indices[:num_parents]]
        
        # Generate offspring through crossover and mutation
        offspring = []
        while len(offspring) < pop_size - num_parents:
            # Tournament selection
            parent1 = parents[np.random.randint(0, len(parents))]
            parent2 = parents[np.random.randint(0, len(parents))]
            
            # Arithmetic crossover
            alpha = np.random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            
            # Normalize
            child = child * (n_points/2) / np.sum(child)
            offspring.append(child)
        
        # Replace worst individuals with offspring
        worst_indices = sorted_indices[num_parents:]
        for i, idx in enumerate(worst_indices):
            if i < len(offspring):
                population[idx] = offspring[i]
        
        # Keep track of the best individual
        current_best_idx = sorted_indices[0]
        current_best = population[current_best_idx]
        if fitness[current_best_idx] < best_c5:
            best_h = current_best.copy()
            best_c5 = fitness[current_best_idx]
        
        generation += 1
    
    # Refinement with SLSQP on top candidates
    top_candidates = [population[i] for i in sorted_indices[:5]]
    for h_refined in top_candidates:
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
