import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # SLSQP optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    current_h = h_start.copy()
    current_h = np.clip(current_h, 0, 1)
    current_h = current_h * (n_points/2) / np.sum(current_h)
    
    # Try multiple restarts with different perturbations
    for restart in range(5):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Add small random noise to start
        h_perturbed = current_h + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        try:
            res = minimize(objective, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
        
        # Also try a CVXPY approach for better convergence
        try:
            h_var = cp.Variable(n_points, bounds=(0, 1))
            constraints = [cp.sum(h_var) == n_points/2]
            
            # We minimize an upper bound on the maximum overlap
            # Compute all correlations and get maximum
            corr_exprs = []
            for lag in range(2 * n_points - 1):
                corr_val = 0
                for i in range(n_points):
                    if 0 <= i - lag < n_points:
                        corr_val += h_var[i] * (1 - h_var[i - lag])
                corr_exprs.append(corr_val)
            
            max_corr = cp.max(corr_exprs)
            objective_cvx = cp.Minimize(max_corr)
            prob = cp.Problem(objective_cvx, constraints)
            
            # Try to solve with different solvers
            prob.solve(solver=cp.ECOS, verbose=False, max_iter=500)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_sol = h_var.value
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points/2) / np.sum(h_sol)
                    c5_sol = evaluate_erdos_solution(h_sol, dx)
                    if c5_sol < best_c5:
                        best_h = h_sol
                        best_c5 = c5_sol
                        
        except Exception:
            pass
    
    # Final SLSQP refinement
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
