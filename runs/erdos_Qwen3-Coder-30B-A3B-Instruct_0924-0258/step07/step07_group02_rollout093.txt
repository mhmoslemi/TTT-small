import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a good starting point using a known construction
        x = np.linspace(0, 2, n_points)
        # Use piecewise constant function with two segments
        h_start = np.zeros(n_points)
        h_start[x < 1] = 0.6
        h_start[x >= 1] = 0.4
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with trust-constr
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    cons = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': lambda h: constraint_bounds(h)[n_points:]},
        {'type': 'ineq', 'fun': lambda h: constraint_bounds(h)[:n_points]}
    ]
    
    # Additional gradient-based optimization
    max_iter = 200
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Use scipy minimize with trust-constr for better convergence
        try:
            result = minimize(
                objective,
                best_h,
                method='trust-constr',
                constraints=cons,
                options={'maxiter': 10, 'disp': False}
            )
            if result.success:
                h_test = result.x
                # Project to [0,1] and normalize
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points/2) / np.sum(h_test)
                c5_test = evaluate_erdos_solution(h_test, dx)
                if c5_test < best_c5:
                    best_h = h_test
                    best_c5 = c5_test
        except:
            pass
            
        # Add some random perturbation every 10 iterations
        if iter_num % 10 == 0 and iter_num > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h_perturb = best_h + noise
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
            c5_perturb = evaluate_erdos_solution(h_perturb, dx)
            if c5_perturb < best_c5:
                best_h = h_perturb
                best_c5 = c5_perturb
    
    # Final run with convex optimization if we have time
    if time() - start_time < budget_s * 0.9:
        try:
            h_opt = cp.Variable(n_points)
            c5_var = cp.Variable()
            
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1,
                c5_var >= 0
            ]
            
            # For a few top lags, add constraints
            h_test = best_h.copy()
            corr = np.correlate(h_test, 1 - h_test, mode='full')
            top_lags = np.argsort(corr)[-10:]  # Top 10 lags mean largest overlaps
            
            for k in top_lags:
                if k < n_points:
                    # Compute correlation for this lag
                    if k == 0:
                        corr_val = cp.sum(cp.multiply(h_opt, 1 - h_opt))
                    elif k < n_points:
                        corr_val = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    else:
                        corr_val = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                    constraints.append(corr_val <= c5_var * dx + 1e-8)
            
            prob = cp.Problem(cp.Minimize(c5_var), constraints)
            
            # Use a more robust solver
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass
    
    # Final verification and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
