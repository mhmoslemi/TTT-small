import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 200
    elite_size = population_size // 5
    mutation_rate = 0.3
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    for i in range(population_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    stagnant = 0
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            stagnant = 0
        else:
            stagnant += 1
            
        if stagnant > 50:
            break
            
        # Create new population
        new_population = population[:elite_size]  # Elitism
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if np.random.random() < crossover_rate and len(new_population) + 1 < population_size:
                # Tournament selection
                parent1_idx = np.random.choice(elite_size)
                parent2_idx = np.random.choice(elite_size)
                parent1 = population[parent1_idx]
                parent2 = population[parent2_idx]
                
                # Arithmetic crossover
                alpha = np.random.random()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
                
                # Mutation
                if np.random.random() < mutation_rate:
                    child1 += np.random.normal(0, 0.05, n_points)
                    child1 = np.clip(child1, 0, 1)
                    child1 = child1 * (n_points/2) / np.sum(child1)
                
                if np.random.random() < mutation_rate:
                    child2 += np.random.normal(0, 0.05, n_points)
                    child2 = np.clip(child2, 0, 1)
                    child2 = child2 * (n_points/2) / np.sum(child2)
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Random individual with mutation
                h = population[np.random.randint(elite_size)].copy()
                h += np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                new_population.append(h)
        
        population = new_population
    
    # Final local refinement with SLSQP
    top_candidates = population[:5]
    best_refined = best_h.copy()
    best_refined_c5 = best_c5
    
    for h_candidate in top_candidates:
        h_refined = h_candidate.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_refined_c5:
                    best_refined = h_final
                    best_refined_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_refined), best_refined_c5, n_points)
