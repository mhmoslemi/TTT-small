import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based descent with adaptive learning rate
    h = h_start.copy()
    learning_rate = 0.1
    max_iter = 1000
    patience = 50
    patience_counter = 0
    best_iter = 0
    target_sum = n_points / 2.0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute current overlap and find worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Compute gradient at the worst lag
        # Gradient of correlation h * (1-h) at the lag that maximizes it
        # We approximate gradient of max overlap w.r.t h
        grad = np.zeros_like(h)
        half_len = len(h)
        for i in range(len(h)):
            if max_idx - i >= 0 and max_idx - i < len(h):
                grad[i] += (1 - 2 * h[i]) * (1 - h[max_idx - i])
            # Symmetric case
            if max_idx - (len(h) - 1 - i) >= 0 and max_idx - (len(h) - 1 - i) < len(h):
                offset = max_idx - (len(h) - 1 - i)
                # This partial gradient is tricky to compute directly
                # Let's approximate with finite differences
                grad[i] += (1 - 2 * h[i]) * (1 - h[offset])
        
        # Project to feasible region before gradient update
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h = h * target_sum / current_sum
        
        # Update step with learning rate
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        
        # Project back to unit sum
        h = h * target_sum / np.sum(h)
        
        # Perturb occasionally to escape plateaus
        if iteration % 50 == 0 and iteration > 0:
            perturb_idx = np.random.choice(len(h), size=int(len(h) * 0.1), replace=False)
            h[perturb_idx] += np.random.uniform(-0.1, 0.1, size=len(perturb_idx))
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            best_iter = iteration
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter > patience:
                learning_rate *= 0.5
                patience_counter = 0
                
    # Phase 2: Refinement with SQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12, 'eps': 1e-6})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final verification and return
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
