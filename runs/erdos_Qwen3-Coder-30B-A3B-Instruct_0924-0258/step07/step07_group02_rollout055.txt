import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Simple construction as fallback
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.4
        h_start[-1] = 0.6
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA Parameters
    pop_size = 100
    max_generations = 300
    elite_size = pop_size // 10
    mutation_rate = 0.2
    crossover_rate = 0.7
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = np.random.rand(n_points) * 0.8 + 0.1  # Random values in [0.1, 0.9]
            h = h * (n_points/2) / np.sum(h)  # Normalize to sum to n_points/2
        population.append(h)
    
    # Hall of fame
    hall_of_fame = []
    
    # EA loop
    no_improvement_count = 0
    
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Update hall of fame
        sorted_indices = np.argsort(fitness)
        top_fitness = [fitness[i] for i in sorted_indices[:elite_size]]
        top_solutions = [population[i] for i in sorted_indices[:elite_size]]
        
        for h, f in zip(top_solutions, top_fitness):
            if len(hall_of_fame) < elite_size or f < min([x[0] for x in hall_of_fame]):
                hall_of_fame.append((f, h.copy()))
                if len(hall_of_fame) > elite_size:
                    hall_of_fame.sort()
                    hall_of_fame = hall_of_fame[:elite_size]
        
        # Check for improvement
        if fitness[sorted_indices[0]] < best_c5:
            best_h = population[sorted_indices[0]].copy()
            best_c5 = fitness[sorted_indices[0]]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 80:
                break
        
        # Selection: tournament
        def tournament_selection(k=5):
            indices = np.random.choice(pop_size, k)
            best_idx = indices[np.argmin([fitness[i] for i in indices])]
            return population[best_idx]
        
        # Crossover
        new_pop = []
        for _ in range(pop_size):
            if np.random.rand() < crossover_rate:
                parent1 = tournament_selection()
                parent2 = tournament_selection()
                alpha = np.random.rand()
                offspring = alpha * parent1 + (1 - alpha) * parent2
            else:
                offspring = tournament_selection()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Adaptive mutation
                decay = max(0.05, 0.1 * (1 - gen / max_generations))
                for i in range(n_points):
                    if np.random.rand() < 0.2:  # 20% chance to mutate
                        offspring[i] += np.random.normal(0, decay)
                
                # Repair to keep within [0,1] and sum constraint
                offspring = np.clip(offspring, 0, 1)
                offspring = offspring * (n_points/2) / np.sum(offspring)
            
            new_pop.append(offspring)
        
        population = new_pop
    
    # Final L-BFGS refinement
    try:
        # Select top solutions for final refinement
        top_candidates = [x[1] for x in hall_of_fame]
        if not top_candidates:
            top_candidates = [best_h]
        
        for h_candidate in top_candidates:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            cons = [{'type': 'eq', 'fun': constraint_sum},
                    {'type': 'ineq', 'fun': lambda h: 1 - h},
                    {'type': 'ineq', 'fun': lambda h: h}]
            
            h0 = h_candidate.copy()
            res = minimize(objective, h0, method='L-BFGS-B', 
                          bounds=[(0, 1) for _ in range(n_points)],
                          constraints=cons, options={'ftol': 1e-8, 'gtol': 1e-6})
            
            if res.success:
                refined_result = evaluate_erdos_solution(res.x, dx)
                if refined_result < best_c5:
                    best_h = res.x.copy()
                    best_c5 = refined_result
    except Exception:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
