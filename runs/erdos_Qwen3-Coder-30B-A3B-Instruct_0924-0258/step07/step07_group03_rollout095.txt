import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T0 = 0.1
    cooling_rate = 0.001
    max_iterations = 1000
    perturbation_fraction = 0.15
    
    h_current = h_start.copy()
    c5_current = best_c5
    T = T0
    stagnation_count = 0
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Perturb solution
        h_new = h_current.copy()
        n_perturb = int(perturbation_fraction * n_points)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        deltas = np.random.uniform(-0.1, 0.1, n_perturb)
        h_new[indices] += deltas
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept with simulated annealing probability
        delta = c5_new - c5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_current = h_new
            c5_current = c5_new
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                stagnation_count = 0
        else:
            stagnation_count += 1
            
        T = T0 * np.exp(-cooling_rate * iteration)
        
        if stagnation_count > 100:
            break
    
    # Gradient Descent refinement
    def compute_gradient(h, k_max):
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i + k_max < len(h):
                grad[i] += (1 - h[i + k_max]) * dx
            if i - k_max >= 0:
                grad[i] += (1 - h[i - k_max]) * dx
        return grad
    
    h_refined = best_h.copy()
    learning_rate = 0.1
    max_iter_gd = 500
    stagnation_gd = 0
    
    for iteration in range(max_iter_gd):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find maximum overlap lag
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient
        grad = compute_gradient(h_refined, k_max)
        
        # Update
        h_refined = h_refined - learning_rate * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        # Check improvement
        c5_new = evaluate_erdos_solution(h_refined, dx)
        if c5_new < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_new
            stagnation_gd = 0
        else:
            stagnation_gd += 1
            
        if stagnation_gd > 50:
            learning_rate *= 0.5
            stagnation_gd = 0
            
        if learning_rate < 1e-6:
            break
    
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
