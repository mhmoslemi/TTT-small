import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with adaptive learning rate
    h = h_start.copy()
    learning_rate = 0.01
    max_iter = 1000
    no_improvement_count = 0
    max_no_improvement = 50
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_k = max_idx - (len(h) - 1)  # Convert from full correlation index
        
        # Compute gradient for the most constraining shift
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + max_k) % len(h)
            gradient[i] = 2 * (1 - h[j] - h[i])
        
        # Update h
        h_new = h - learning_rate * gradient
        
        # Clip to [0,1] and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement_count = 0
            learning_rate = 0.01  # Reset learning rate
        else:
            no_improvement_count += 1
            # Reduce learning rate if no improvement
            learning_rate *= 0.9
            
        h = h_new.copy()
        
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
