import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

    # Basis function approach
    m = 10  # number of basis functions
    if 'initial_h_values' in globals():
        # Initialize basis functions aligned with initial solution
        basis_functions = np.zeros((n_points, m))
        # Sample key points from the initial solution
        key_indices = np.linspace(0, n_points-1, m, dtype=int)
        for i in range(m):
            basis_functions[:, i] = np.zeros(n_points)
            basis_functions[key_indices[i], i] = 1.0
        # Normalize to have area 1
        basis_functions = basis_functions * (n_points/2) / np.sum(basis_functions, axis=0)
    else:
        # Initialize uniform basis functions
        basis_functions = np.zeros((n_points, m))
        for i in range(m):
            basis_functions[:, i] = np.zeros(n_points)
            start_idx = int(i * n_points / m)
            end_idx = int((i + 1) * n_points / m)
            if i == m - 1:
                end_idx = n_points
            basis_functions[start_idx:end_idx, i] = 1.0 / (end_idx - start_idx)
    
    # Initialize coefficients
    coeffs = np.ones(m) / m
    coeffs = coeffs * (n_points/2) / np.sum(coeffs)
    
    # Trust region parameters
    trust_radius = 0.1
    max_iter = 10000
    patience = 0
    max_patience = 50
    
    # Iteration loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current h
        h_current = basis_functions @ coeffs
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Compute current C5
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1
            
        # Gradient approximation using finite differences
        grad = np.zeros(m)
        eps = 1e-5
        for i in range(m):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = basis_functions @ coeffs_plus
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            coeffs_minus = coeffs.copy()
            coeffs_minus[i] -= eps
            h_minus = basis_functions @ coeffs_minus
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Trust region subproblem solution
        # Simplified: perform projected gradient step with trust region
        delta = coeffs - coeffs + grad  # Delta computed from gradient
        norm_delta = np.linalg.norm(delta)
        
        if norm_delta > trust_radius:
            delta = delta * trust_radius / norm_delta
            
        # New coefficients in trust region
        new_coeffs = coeffs - delta

        if patience > max_patience:

            h = basis_functions @ c
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(c):
            h = basis_functions @ c
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            return np.sum(h) - n_points / 2.0

        res = minimize(objective, coeffs, method='SLSQP',

            h_refined = basis_functions @ coeffs_refined
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    # Ensure returned values are valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
