import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
        h_start += np.random.uniform(-0.05, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive multiresolution strategy
    resolutions = [20, 40, 80]
    max_iterations = [100, 150, 200]
    perturbation_sizes = [0.15, 0.10, 0.05]
    
    for res_idx, (n_points, max_iter, pert_size) in enumerate(zip(resolutions, max_iterations, perturbation_sizes)):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize
        if res_idx == 0:
            h_current = h_start.copy()
        else:
            # Interpolate from previous resolution
            h_prev = h_current
            h_current = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_prev)), h_prev)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
        
        dx = 2.0 / n_points
        current_h = h_current.copy()
        current_c5 = evaluate_erdos_solution(current_h, dx)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Main optimization loop
        temperature = 1.0
        tabu_list = []
        tabu_size = 20
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate candidate
            candidate_h = current_h.copy()
            indices = np.random.choice(n_points, size=max(1, int(n_points * 0.2)), replace=False)
            for i in indices:
                delta = np.random.uniform(-pert_size, pert_size) * candidate_h[i]
                candidate_h[i] += delta
            candidate_h = np.clip(candidate_h, 0, 1)
            candidate_h = candidate_h * (n_points/2) / np.sum(candidate_h)
            
            # Evaluate
            candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
            
            # Accept or reject
            if candidate_c5 < current_c5:
                current_h = candidate_h.copy()
                current_c5 = candidate_c5
                if candidate_c5 < best_c5:
                    best_h = candidate_h.copy()
                    best_c5 = candidate_c5
            else:
                # Accept with probability
                delta = candidate_c5 - current_c5
                if np.random.random() < np.exp(-delta / (temperature + 1e-8)):
                    current_h = candidate_h.copy()
                    current_c5 = candidate_c5
                    
            # Update temperature
            if iteration % 50 == 0 and iteration > 0:
                temperature *= 0.99
                
            # Tabu list management
            if len(tabu_list) >= tabu_size:
                tabu_list.pop(0)
                
    # Final refinement
    try:
        h_final = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
