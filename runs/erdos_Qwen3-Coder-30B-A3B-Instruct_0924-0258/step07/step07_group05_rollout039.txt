import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ALM parameters
    max_iter = 5000
    convergence_tol = 1e-6
    alpha = 0.1
    mu = 100.0
    trust_region_factor = 1.1
    backtracking_patience = 10
    
    h = best_h.copy()
    prev_c5 = best_c5
    lagrange_multiplier = 0.0
    
    # Precompute indices to improve performance
    k_indices = np.arange(1, n_points)
    k_ranges = [np.arange(n_points - k) for k in k_indices]
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient for all k
        current_c5 = evaluate_erdos_solution(h, dx)
        
        # Find active k
        corr = np.correlate(h, 1 - h, mode='full')
        active_k = np.argmax(corr)
        active_value = corr[active_k]
        
        # Compute gradient for active k
        grad = np.zeros(n_points)
        if active_k < n_points:
            # h_i(1 - h_{i+k}) contribution
            for i in range(n_points - active_k):
                grad[i] += (1 - h[i + active_k])
        else:
            # h_{i+k}(1 - h_i) contribution
            for i in range(active_k - n_points + 1, n_points):
                grad[i] += -h[i - (active_k - n_points + 1)]
        
        # ALM gradient
        al_grad = grad * dx + lagrange_multiplier + mu * (h - 1)
        
        # Apply trust region
        al_grad = np.clip(al_grad, -1.0, 1.0)
        
        # Line search with backtracking
        step_size = alpha
        for _ in range(backtracking_patience):
            h_trial = h - step_size * al_grad
            h_trial = np.clip(h_trial, 0, 1)
            # Scale to meet constraint
            scale_factor = (n_points / 2.0) / np.sum(h_trial)
            h_trial = h_trial * scale_factor
            
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            
            if c5_trial < current_c5 or step_size < 1e-10:
                h = h_trial
                break
            step_size *= 0.5
            
        # Update Lagrange multiplier and penalty
        constraint_violation = np.sum(h) - n_points / 2.0
        lagrange_multiplier += mu * constraint_violation
        
        # Adaptive penalty parameter
        if abs(constraint_violation) > 1e-3:
            mu *= 2.0
        else:
            mu *= 0.9
            
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        if abs(prev_c5 - current_c5) < convergence_tol:
            break
        prev_c5 = current_c5
        
        # Trust region adaptive step
        if abs(prev_c5 - current_c5) / (prev_c5 + 1e-8) < 1e-3:
            alpha *= trust_region_factor
        
        if alpha > 1.0:
            alpha = 1.0
            
    # Final polishing with gradient descent
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        mu_final = 1e4
        def augmented_lagrangian(h):
            obj = objective(h)
            penalty = 0.5 * mu_final * (constraint_sum(h))**2
            return obj + penalty
        
        def augmented_lagrangian_grad(h):
            corr = np.correlate(h, 1 - h, mode='full')
            active_k = np.argmax(corr)
            active_value = corr[active_k]
            
            grad = np.zeros(n_points)
            if active_k < n_points:
                for i in range(n_points - active_k):
                    grad[i] += (1 - h[i + active_k])
            else:
                for i in range(active_k - n_points + 1, n_points):
                    grad[i] += -h[i - (active_k - n_points + 1)]
            return grad * dx + mu_final * constraint_sum(h)
        
        res = minimize(augmented_lagrangian, best_h, method='L-BFGS-B', 
                       jac=augmented_lagrangian_grad, bounds=bounds, 
                       options={'maxiter': 100, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
