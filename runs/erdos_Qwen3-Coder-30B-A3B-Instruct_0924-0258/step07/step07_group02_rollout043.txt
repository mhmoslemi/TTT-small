import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create smooth starting point
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use only convex optimization with iterative refinement
    max_iterations = 100
    for iter_num in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            # Formulate as convex optimization problem
            h_vars = cp.Variable(n_points)
            constraints = [
                cp.sum(h_vars) == n_points / 2.0,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # For optimization, sample key overlaps
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            # Use top 10% of lags
            top_lags = np.argsort(corr)[-n_points//10:]
            
            # Add constraints for the most important lags
            for k in top_lags:
                if k < n_points:
                    # Add constraint for this lag
                    corr_sum = cp.sum(cp.multiply(h_vars[:n_points-k], 1 - h_vars[k:]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-8)
            
            # This is just to create a valid problem - objective will be handled by constraints
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=200)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_vars.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass
            
        # Local search refinement
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        # Simple trust region approach with random perturbations
        noise = np.random.normal(0, 0.05, n_points)
        h_trial = best_h + noise
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        
        c5_trial = obj(h_trial)
        if c5_trial < best_c5:
            best_h = h_trial
            best_c5 = c5_trial
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
