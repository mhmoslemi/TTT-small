import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Genetic algorithm iterations
    gen = 0
    stagnation_count = 0
    prev_best = best_c5
    
    while time() - start_time < budget_s * 0.95:
        gen += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism: keep best individuals
        new_population = population[:elite_size]
        
        # Tournament selection and reproduction
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = min(np.random.choice(pop_size, tournament_size, replace=False), 
                             key=lambda i: fitness[i])
            parent2_idx = min(np.random.choice(pop_size, tournament_size, replace=False), 
                             key=lambda i: fitness[i])
            
            parent1 = population[parent1_idx].copy()
            parent2 = population[parent2_idx].copy()
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    mutation_indices = np.random.choice(n_points, 
                                                       size=int(mutation_rate * n_points), 
                                                       replace=False)
                    for idx in mutation_indices:
                        child[idx] += np.random.normal(0, 0.03)
                
                # Repair constraints
                child = np.clip(child, 0, 1)
                sum_h = np.sum(child)
                if abs(sum_h - n_points/2) > 1e-6:
                    child = child * (n_points/2) / sum_h
                    
                new_population.append(child)
        
        # Keep only pop_size individuals
        population = new_population[:pop_size]
        
        # Evaluate new population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        
        # Check for stagnation
        if abs(prev_best - best_c5) < 1e-6:
            stagnation_count += 1
        else:
            prev_best = best_c5
            stagnation_count = 0
            
        # Periodic convex relaxation
        if gen % 20 == 0:
            try:
                # Select top individuals for refinement
                top_idx = int(pop_size * 0.2)
                for i in range(top_idx):
                    h_refine = population[i]
                    h_opt = cp.Variable(n_points)
                    constraints = [
                        cp.sum(h_opt) == n_points / 2.0,
                        h_opt >= 0,
                        h_opt <= 1
                    ]
                    
                    # Add constraints from promising values
                    for k in range(1, n_points//4):
                        if n_points - k > 0:
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                    
                    prob = cp.Problem(cp.Minimize(0), constraints)
                    prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=300)
                    if prob.status == cp.OPTIMAL:
                        h_feasible = h_opt.value
                        c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                        if c5_feasible < best_c5:
                            best_h = h_feasible
                            best_c5 = c5_feasible
            except Exception:
                pass
        
        # Restart if stuck
        if stagnation_count > 50:
            h_start = best_h.copy()
            population = []
            fitness = []
            for i in range(pop_size):
                h = h_start + np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                population.append(h.copy())
                c5 = evaluate_erdos_solution(h, dx)
                fitness.append(c5)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
            stagnation_count = 0
    
    # Final local optimization on best solution
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
