import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a simple step function construction
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place two intervals of length 0.5 each
        # First interval: from 0.65 to 1.15
        # Second interval: from 1.35 to 1.85  
        start1 = int(0.65 / dx)
        end1 = int(1.15 / dx)
        start2 = int(1.35 / dx)
        end2 = int(1.85 / dx)
        h_start[start1:end1] = 1.0
        h_start[start2:end2] = 1.0
        # Adjust to make integral = 1
        h_start = h_start * (n_points/2) / np.sum(h_start)
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-free optimization using Nelder-Mead on a subset of parameters
    # Instead of full optimization, we try to make final adjustments to the step locations
    h_current = h_start.copy()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Use simple local search with random perturbations
    n_iterations = 2000
    for i in range(n_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation
        h_perturbed = h_current.copy()
        idx = np.random.randint(0, n_points)
        # Slightly modify one point
        h_perturbed[idx] += np.random.normal(0, 0.02)
        h_perturbed[idx] = max(0, min(1, h_perturbed[idx]))
        
        # Normalize
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        c5_new = evaluate_erdos_solution(h_perturbed, dx)
        if c5_new < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_new
            
        if np.random.rand() < 0.5:
            h_current = h_perturbed
        else:
            h_current = h_current
            
    # Final validation and cleanup
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
