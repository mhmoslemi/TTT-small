import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Benders decomposition with constraint tightening
    max_iter = 5000
    convergence_tol = 1e-6
    perturbation_magnitude = 0.05
    perturbation_decay = 0.95
    
    # Initialize with current solution
    h_current = h_start.copy()
    prev_c5 = best_c5
    
    # Benders decomposition loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Extract relevant overlaps (k values 0 to n_points-1)
        overlaps = corr[n_points-1:]  # k from 0 to n_points-1
        
        # Identify worst-case k
        worst_k_idx = np.argmax(overlaps)
        worst_overlap = overlaps[worst_k_idx]
        
        # Convex relaxation to improve h
        try:
            h_new = cp.Variable(n_points)
            constraints = [
                cp.sum(h_new) == n_points / 2.0,
                h_new >= 0,
                h_new <= 1
            ]
            
            # Add the most constraining constraint
            if worst_k_idx < n_points:
                # For shift k=worst_k_idx, the overlap is sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k])
                k = worst_k_idx
                corr_sum = cp.sum(cp.multiply(h_new[:n_points-k], 1 - h_new[k:]))
                constraints.append(corr_sum <= worst_overlap * dx + 1e-8)
            
            # Add additional constraints for stability
            k_vals = [k for k in range(0, min(20, n_points//3)) if k > 0]
            for k_val in k_vals:
                if k_val < n_points:
                    k = k_val
                    corr_sum = cp.sum(cp.multiply(h_new[:n_points-k], 1 - h_new[k:]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-8)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            
            if prob.status == cp.OPTIMAL:
                h_solution = h_new.value
                # Perturb
                h_perturbed = h_solution + perturbation_magnitude * np.random.randn(n_points)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
                
                c5_new = evaluate_erdos_solution(h_perturbed, dx)
                
                if c5_new < best_c5:
                    best_h = h_perturbed.copy()
                    best_c5 = c5_new
                
                h_current = h_perturbed.copy()
                prev_c5 = c5_new
                perturbation_magnitude *= perturbation_decay
                
                if abs(prev_c5 - c5_new) < convergence_tol:
                    break
            
        except Exception:
            pass
            
        # Local polishing with SLSQP
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Try multiple initializations to escape local optima
            for _ in range(3):
                # Add small random perturbation to current solution
                h_pert = h_current + np.random.normal(0, 0.02, n_points)
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points/2) / np.sum(h_pert)
                
                res = minimize(objective, h_pert, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-8})
                
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
        except Exception:
            pass
            
    # Final check and refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
