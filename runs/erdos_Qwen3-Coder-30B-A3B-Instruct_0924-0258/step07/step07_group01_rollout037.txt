import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Try a construction with two equal intervals
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place two intervals of length 0.5 each
        interval_length = 0.5
        interval_steps = int(interval_length / dx)
        # Place at symmetric positions
        a = 0.65
        b = 0.85
        a_idx = int(a / dx)
        b_idx = int(b / dx)
        h_start[a_idx:a_idx+interval_steps] = 1
        h_start[b_idx:b_idx+interval_steps] = 1
        # Normalize to integral = 1
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # Re-evaluate to get the actual value
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_h = h_start.copy()
    
    # Small local optimization with gradient-based method
    # Simple hill-climbing on a small neighborhood
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Perturb in small steps
    step_size = 0.01
    for _ in range(5000):
        if time() - start_time > budget_s * 0.95:
            break
        # Generate small random change to h
        change = np.random.normal(0, step_size, n_points)
        new_h = current_h + change
        # Clip to [0,1] and renormalize
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    
    return (list(best_h), best_c5, n_points)
