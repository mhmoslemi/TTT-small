import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    convex_interval = 50
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Best solution tracking
    best_in_generation = float('inf')
    stagnation_counter = 0
    stagnation_limit = 100
    
    for generation in range(10000):  # Large number of iterations
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Track best in this generation
        gen_best = min(fitness)
        if gen_best < best_in_generation:
            best_in_generation = gen_best
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Check for stagnation
        if stagnation_counter > stagnation_limit:
            # Reset with new random population
            population = []
            for _ in range(pop_size):
                if 'initial_h_values' in globals():
                    h = h_start + np.random.normal(0, 0.05, n_points)
                else:
                    h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                population.append(h)
            best_in_generation = float('inf')
            stagnation_counter = 0
        
        # Select parents using tournament selection
        parents = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            parents.append(population[winner].copy())
        
        # Create new population through crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            if i + 1 >= pop_size:
                new_population.append(parents[i].copy())
                break
                
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, parents[i], parents[i+1])
                child2 = np.where(mask, parents[i+1], parents[i])
            else:
                child1, child2 = parents[i].copy(), parents[i+1].copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate_indices = np.random.choice(n_points, int(mutation_rate * n_points), replace=False)
                for idx in mutate_indices:
                    child1[idx] += np.random.normal(0, 0.03)
            
            if np.random.rand() < mutation_rate:
                mutate_indices = np.random.choice(n_points, int(mutation_rate * n_points), replace=False)
                for idx in mutate_indices:
                    child2[idx] += np.random.normal(0, 0.03)
            
            # Ensure feasibility
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        # Trim if necessary
        if len(new_population) > pop_size:
            new_population = new_population[:pop_size]
        
        population = new_population[:]
        
        # Periodic convex relaxation
        if generation % convex_interval == 0 and generation > 0:
            try:
                # Select top performers
                top_indices = np.argsort(fitness)[:pop_size//4]
                for idx in top_indices:
                    h_candidate = population[idx]
                    # Formulate convex problem
                    h_opt = cp.Variable(n_points)
                    constraints = [
                        cp.sum(h_opt) == n_points / 2.0,
                        h_opt >= 0,
                        h_opt <= 1
                    ]
                    # Add constraints on overlap for a few sample shifts
                    sample_shifts = np.linspace(0.1, 1.9, min(15, n_points//4))
                    for k_val in sample_shifts:
                        k = int(k_val * n_points / 2)
                        if k < n_points and k > 0:
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                            constraints.append(corr_sum <= (best_c5 + 1e-6) * dx)
                    
                    prob = cp.Problem(cp.Minimize(0), constraints)
                    prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                    if prob.status == cp.OPTIMAL:
                        h_feasible = h_opt.value
                        c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                        if c5_feasible < best_c5:
                            best_h = h_feasible
                            best_c5 = c5_feasible
            except Exception:
                pass
        
        # Update global best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
    
    # Final local optimization of the best solution
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
