import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial setup
    n_points_list = [20, 40, 80]
    current_n = 20
    best_c5 = float('inf')
    best_h = None
    
    # Multiresolution optimization
    for n_points in n_points_list:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize with periodic function
        x = np.linspace(0, 2, n_points)
        h_initial = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_initial = np.clip(h_initial, 0, 1)
        normalization = (n_points / 2) / np.sum(h_initial)
        h_initial *= normalization
        
        # Add random perturbations
        perturbation = 0.05 * np.random.random(n_points)
        h_initial += perturbation * h_initial
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # If restoring from previous solution, interpolate
        if best_h is not None and n_points > len(best_h):
            h_initial = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(best_h)), best_h)
            # Add small perturbations
            perturbation = 0.05 * np.random.random(n_points)
            h_initial += perturbation * h_initial
            h_initial = np.clip(h_initial, 0, 1)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        dx = 2.0 / n_points
        h_current = h_initial.copy()
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        # Local optimization with simulated annealing
        max_iter = 200 if n_points == 80 else 150 if n_points == 40 else 100
        temp = 0.1
        cooling_rate = 0.99
        tabu_list = []
        tabu_size = 20
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate neighbor
            candidate = h_current.copy()
            indices = np.random.choice(n_points, size=max(1, n_points // 5), replace=False)
            perturb_mag = temp * 0.5  # Adaptive perturbation
            for i in indices:
                delta = np.random.uniform(-perturb_mag, perturb_mag) * candidate[i]
                candidate[i] += delta
            candidate = np.clip(candidate, 0, 1)
            candidate = candidate * (n_points / 2) / np.sum(candidate)
            
            # Evaluate
            c5_candidate = evaluate_erdos_solution(candidate, dx)
            
            # Accept or reject
            if c5_candidate < c5_current:
                h_current = candidate
                c5_current = c5_candidate
            else:
                # Metropolis criterion
                delta = c5_candidate - c5_current
                if np.random.random() < np.exp(-delta / temp):
                    h_current = candidate
                    c5_current = c5_candidate
                    
            # Track best
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
            
            # Update temperature
            temp *= cooling_rate
            
            # Tabu list
            if len(tabu_list) >= tabu_size:
                tabu_list.pop(0)
            tabu_list.append(candidate.copy())
        
        # Final polish if this is the finest resolution
        if n_points == 80:
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(objective, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 1000, 'ftol': 1e-10})
                if res.success:
                    h_final = res.x
                    c5_final = objective(h_final)
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
            except Exception:
                pass
                
        # Update best solution
        if best_h is None or c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
    
    # Return the best result
    if best_h is None:
        best_h = h_initial.copy()
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
    return (list(best_h), best_c5, len(best_h))
