import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing Parameters
    T0 = 0.1
    cooling_rate = 0.001
    max_iterations = 10000
    perturbation_fraction = 0.15
    
    h_current = h_start.copy()
    c5_current = best_c5
    T = T0
    
    stagnation_count = 0
    max_stagnation = 100
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Perturb h
        h_new = h_current.copy()
        num_perturb = max(1, int(n_points * perturbation_fraction))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
        
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
            stagnation_count = 0
        else:
            # Accept with probability
            delta = c5_new - c5_current
            prob = np.exp(-delta / T)
            if np.random.rand() < prob:
                h_current = h_new.copy()
                c5_current = c5_new
            stagnation_count += 1
            
        # Update best
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        # Cool down
        T = T0 * np.exp(-cooling_rate * iteration)
        
        if stagnation_count >= max_stagnation:
            break
    
    # Gradient Descent refinement
    h_refined = best_h.copy()
    max_iter_gd = 500
    learning_rate = 0.1
    stagnation_gd = 0
    min_improvement = 1e-10
    
    for iteration in range(max_iter_gd):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find k_max
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient for k_max
        grad = np.zeros_like(h_refined)
        n = len(h_refined)
        if k_max < n:
            for i in range(n - k_max):
                grad[i] += (1 - h_refined[i + k_max]) * dx
        else:
            for i in range(k_max, n):
                grad[i] += (1 - h_refined[i - k_max]) * dx
                
        # Update
        h_new = h_refined - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            h_refined = h_new.copy()
            best_c5 = c5_new
            stagnation_gd = 0
        else:
            stagnation_gd += 1
            
        if stagnation_gd > 50:
            learning_rate *= 0.5
            stagnation_gd = 0
            if learning_rate < 1e-6:
                break
    
    # Final SLSQP optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
