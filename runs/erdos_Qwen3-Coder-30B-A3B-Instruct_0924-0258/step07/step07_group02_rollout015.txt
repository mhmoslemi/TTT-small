import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Use initial construction if available

        n_points = len(h_start)
    else:

        h_start = np.zeros(n_points)

        segments = 3
        seg_width = n_points // segments
        for i in range(segments):
            start_idx = i * seg_width
            end_idx = (i+1) * seg_width if i < segments-1 else n_points
            if i % 2 == 0:
                h_start[start_idx:end_idx] = 0.8
            else:
                h_start[start_idx:end_idx] = 0.2
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient based refinement with trust region approach
    # Use a multi-start strategy
    n_starts = 10
    for start_idx in range(n_starts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb starting point
        if start_idx == 0:
            h_pert = h_start.copy()
        else:
            # Add more significant perturbation for diversity
            h_pert = h_start + np.random.normal(0, 0.05, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            
        # Project to constraint sum(h) = n_points/2
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        # Use a simple gradient-based optimizer trying to minimize C5
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)

            return np.sum(h_vec) - n_points/2

            return np.concatenate([h_vec, 1-h_vec])
        
        # Bound constraint bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Simple bound-constrained optimization
        try:
            result = minimize(
                objective, 
                h_pert, 
                method='L-BFGS-B', 
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            if result.success:
                h_opt = result.x
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
        except:
            pass

    # Perturb around the best solution
    # Also try to fit a simple trigonometric basis
    
    # Use small cosine expansion for more structured approach
    n_terms = min(8, n_points//2)
    
    def cosine_basis(x, k):
        return np.cos(np.pi * k * x)
    
    # Create design matrix for basis
    x = np.linspace(0, 2, n_points)
    Phi = np.zeros((n_points, n_terms))

    best_h = np.clip(best_h, 0, 1.0)

    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
