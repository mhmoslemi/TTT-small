import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a periodic function as initialization
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    T = 0.5  # Initial temperature
    alpha = 0.995  # Cooling rate
    T_min = 1e-5  # Minimum temperature
    max_iter = 1000
    tabu_list = []
    tabu_size = 20
    stagnation_count = 0
    max_stagnation = 50
    
    # Adaptive perturbation size
    perturbation_size = 0.1
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    iter_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate solution
        candidate_h = current_h.copy()
        
        # Select random indices to perturb (15% of points)
        n_perturb = max(1, int(0.15 * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        for idx in indices:
            # Perturb by Gaussian noise
            delta = np.random.normal(0, perturbation_size)
            candidate_h[idx] += delta
            
        # Enforce bounds
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Normalize to sum constraint
        candidate_h = candidate_h * (n_points/2) / np.sum(candidate_h)
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Acceptance criteria
        if candidate_c5 < current_c5:
            # Always accept better solutions
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            if candidate_c5 < best_c5:
                best_h = candidate_h.copy()
                best_c5 = candidate_c5
            stagnation_count = 0
        else:
            # Probabilistic acceptance based on temperature
            delta = candidate_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                current_h = candidate_h.copy()
                current_c5 = candidate_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Update tabu list
        if len(tabu_list) >= tabu_size:
            tabu_list.pop(0)
        tabu_list.append(current_h.copy())
        
        # Cooling schedule
        T = max(T * alpha, T_min)
        
        # Adaptive perturbation size
        if T < 0.05:
            perturbation_size = max(0.05, perturbation_size * 0.99)
        
        # Stagnation detection and restart
        if stagnation_count > max_stagnation:
            # Restart with a small perturbation of best solution
            current_h = best_h.copy()
            noise = np.random.normal(0, 0.05, n_points)
            current_h += noise
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points/2) / np.sum(current_h)
            current_c5 = evaluate_erdos_solution(current_h, dx)
            stagnation_count = 0
            
        iter_count += 1
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
