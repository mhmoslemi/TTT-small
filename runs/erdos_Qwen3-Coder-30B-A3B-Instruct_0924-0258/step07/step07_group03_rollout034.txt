import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based refinement
    h = h_start.copy()
    learning_rate = 0.1
    max_iter = 1000
    
    # Track best solution during gradient descent
    current_c5 = best_c5
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation to find worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        l_max = np.argmax(corr)
        max_overlap = corr[l_max] * dx
        
        # Compute gradient contribution from worst lag
        # We want to reduce max_overlap by modifying h
        # Approximate gradient by finite differences
        
        gradient = np.zeros_like(h)
        eps = 1e-6
        
        # For each position, compute gradient using finite difference
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_overlap_plus = corr_plus[l_max] * dx
            
            gradient[i] = (max_overlap_plus - max_overlap) / eps
        
        # Update h
        h += learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Occasionally add perturbations to escape local minima
        if iteration % 50 == 0:
            perturb_idx = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            for i in perturb_idx:
                h[i] += np.random.uniform(-0.1, 0.1)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        
        # Check for improvement
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Reduce learning rate adaptively
        if iteration % 50 == 0 and iteration > 0:
            learning_rate *= 0.9
            
    # Phase 2: Local SQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
