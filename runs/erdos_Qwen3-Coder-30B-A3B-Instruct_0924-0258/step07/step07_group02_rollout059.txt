import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a good initial solution
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive differential evolution with adaptive parameters
    pop_size = 40
    max_iter = 200
    F = 0.8
    CR = 0.9
    
    # Initialize population
    population = []
    fitness = []
    
    # Create initial population
    for i in range(pop_size):
        if i == 0:
            # Use the best candidate
            h = best_h.copy()
        else:
            # Perturb the best candidate
            h = best_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fit = evaluate_erdos_solution(h, dx)
        fitness.append(fit)
        if fit < best_c5:
            best_h = h.copy()
            best_c5 = fit
    
    # Evolution loop
    no_improvement_count = 0
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 10% (elitism)
        elite = population[:pop_size//10]
        
        # Generate new population
        new_population = elite.copy()
        for i in range(pop_size - len(elite)):
            # DE mutation and crossover
            a, b, c = population[np.random.choice(len(population), 3, replace=False)]
            
            # Mutation
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (n_points/2) / np.sum(mutant)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < CR, mutant, population[i])
            trial = np.clip(trial, 0, 1)
            trial = trial * (n_points/2) / np.sum(trial)
            
            # Evaluate trial
            trial_fit = evaluate_erdos_solution(trial, dx)
            new_population.append(trial)
            fitness.append(trial_fit)
            
            # Update best
            if trial_fit < best_c5:
                best_h = trial.copy()
                best_c5 = trial_fit
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Check for early stopping
        if no_improvement_count > 80:
            break
            
        # Adapt parameters
        if iter_num % 20 == 0:
            CR = max(0.1, CR * 0.95)
            F = max(0.1, F * 0.95)
    
    # Final local refinement with scipy minimize
    try:
        # Get top 10 candidates for local refinement
        indices = np.argsort([evaluate_erdos_solution(h, dx) for h in population])[:10]
        candidates = [population[i] for i in indices]
        
        for h in candidates:
            # Wrap objective function for scipy
            def obj(h_vals):
                h_array = np.array(h_vals)
                h_array = h_array * (n_points/2) / np.sum(h_array)
                return evaluate_erdos_solution(h_array, dx)
            
            # Define constraints
            def constraint_sum(h_vals):
                return np.sum(h_vals) - n_points/2
            
            def constraint_bounds(h_vals):
                return np.concatenate([h_vals, 1-h_vals])
                
            cons = [{'type': 'eq', 'fun': constraint_sum},
                   {'type': 'ineq', 'fun': lambda h_vals: h_vals},
                   {'type': 'ineq', 'fun': lambda h_vals: 1 - h_vals}]
            
            # Optimize with bounds and constraints
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, h, bounds=bounds, constraints=cons,
                          method='SLSQP', options={'maxiter': 100})
            
            if res.success:
                refined_h = res.x * (n_points/2) / np.sum(res.x)
                refined_c5 = evaluate_erdos_solution(refined_h, dx)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
                    
    except Exception:
        pass

    # Final check and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
