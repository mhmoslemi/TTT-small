import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        # Initialize with periodic function
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
        # Add small random perturbations
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multiresolution approach
    resolutions = [20, 40, 80]
    current_n = 20
    current_h = h_start.copy()
    
    # Increase resolution progressively
    for i, new_n in enumerate(resolutions):
        if new_n > current_n:
            # Interpolate solution to new size
            old_x = np.linspace(0, 2, current_n)
            new_x = np.linspace(0, 2, new_n)
            current_h = np.interp(new_x, old_x, current_h)
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (new_n/2) / np.sum(current_h)
            current_n = new_n
        
        # Local optimization at this resolution
        # Simulated Annealing with adaptive cooling
        T = 1.0
        T_min = 1e-6
        alpha = 0.995
        max_iter = 1000 if i == len(resolutions) - 1 else 500
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate candidate by perturbing 10% of points
            candidate = current_h.copy()
            indices = np.random.choice(current_n, size=int(current_n * 0.1), replace=False)
            for idx in indices:
                delta = np.random.normal(0, 0.1 * T)
                candidate[idx] += delta
            candidate = np.clip(candidate, 0, 1)
            candidate = candidate * (current_n/2) / np.sum(candidate)
            
            # Evaluate
            c5_candidate = evaluate_erdos_solution(candidate, 2.0 / current_n)
            
            # Accept or reject
            if c5_candidate < best_c5:
                best_h = candidate.copy()
                best_c5 = c5_candidate
                current_h = candidate.copy()
            else:
                # Accept with probability based on temperature
                delta = c5_candidate - best_c5
                if np.random.random() < np.exp(-delta / T):
                    current_h = candidate.copy()
            
            # Update annealing temperature
            T = max(T * alpha, T_min)
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (current_n/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / current_n)
        
        def constraint_sum(h):
            return np.sum(h) - current_n / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(current_n)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, current_n)
