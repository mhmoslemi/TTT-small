from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Basis function approach: represent h as linear combination of basis functions
    m_basis = 10  # number of basis functions
    if 'initial_h_values' in globals():
        # Initialize basis functions to match features of initial construction
        basis_positions = np.linspace(0, 2, m_basis + 2)[1:-1]  # internal points
        basis_width = 0.3
    else:
        basis_positions = np.linspace(0, 2, m_basis + 2)[1:-1]
        basis_width = 0.3
        
    def triangular_basis(x, pos, width):
        # Create triangular basis function with peak at pos
        return np.maximum(0, 1 - np.abs(x - pos) / width)
    
    # Generate basis matrix
    x_vals = np.linspace(0, 2, n_points)
    basis_matrix = np.zeros((n_points, m_basis))
    for i in range(m_basis):
        basis_matrix[:, i] = triangular_basis(x_vals, basis_positions[i], basis_width)
    
    # Initialize coefficients
    coeffs = np.ones(m_basis) * 0.1
    
    # Trust region optimization
    max_iter = 10000
    tr_radius = 0.1
    patience = 0
    patience_limit = 50
    
    def compute_h_from_coeffs(coeffs):
        h = basis_matrix @ coeffs
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Adjust to meet sum constraint
        sum_h = np.sum(h)
        if abs(sum_h) > 1e-10:
            h = h * (n_points / 2) / sum_h
        return h
    
    def objective(coeffs):
        h = compute_h_from_coeffs(coeffs)
        return evaluate_erdos_solution(h, dx)
    
    def project_coeffs(coeffs):
        # Keep coeffs positive and within reasonable bounds
        coeffs = np.clip(coeffs, -10, 10)
        return coeffs
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current h and objective
        h_current = compute_h_from_coeffs(coeffs)
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1
            
        # Adaptive perturbation
        if patience > patience_limit:
            coeffs += np.random.normal(0, 0.05, m_basis)
            coeffs = project_coeffs(coeffs)
            patience = 0
            
        # Trust region step
        try:
            # Simple gradient approximation via finite differences
            eps = 1e-4
            grad = np.zeros(m_basis)

            for i in range(m_basis):
                coeffs_plus = coeffs.copy()
                coeffs_plus[i] += eps
                obj_plus = objective(coeffs_plus)
                grad[i] = (obj_plus - base_obj) / eps

            # If gradient is small, use random perturbation
            if np.linalg.norm(grad) < 1e-8:
                coeffs += np.random.normal(0, 0.05, m_basis)
            else:
                step = -grad * tr_radius / np.linalg.norm(grad)
                coeffs += step
                coeffs = project_coeffs(coeffs)
                
            # Update trust radius
            if current_c5 <= best_c5 + 1e-8:
                tr_radius *= 1.2  # Increase if successful

        except Exception:
            # Fall back to random search
            coeffs += np.random.normal(0, 0.05, m_basis)
            coeffs = project_coeffs(coeffs)
    
    # Final refinement

            return evaluate_erdos_solution(h, dx)

        h_seed = compute_h_from_coeffs(coeffs)

            return np.sum(h) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint_func}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective_full, h_seed, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        
        if res.success:
            h_final = res.x
            c5_final = objective_full(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
                
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
