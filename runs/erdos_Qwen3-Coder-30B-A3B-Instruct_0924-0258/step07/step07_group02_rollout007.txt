import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point
        x = np.linspace(0, 2, n_points)
        h_start = np.zeros(n_points)
        # Create piecewise constant function
        h_start[:n_points//3] = 0.8
        h_start[n_points//3:2*n_points//3] = 0.3
        h_start[2*n_points//3:] = 0.6
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 100
    generations = 300
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = pop_size // 10
    max_no_improve = 80

    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Calculate initial fitness
    fitness = []
    for h in population:
        fit = evaluate_erdos_solution(h, dx)
        fitness.append(fit)
    
    best_fitness = min(fitness)
    no_improve_count = 0
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break

        # Sort population by fitness
        sorted_idx = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_idx]
        sorted_fit = [fitness[i] for i in sorted_idx]
        
        # Update best solution
        if sorted_fit[0] < best_c5:
            best_h = sorted_pop[0].copy()
            best_c5 = sorted_fit[0]
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        if no_improve_count >= max_no_improve:
            break
            
        # Create new population
        new_pop = []
        
        # Elitism
        for i in range(elite_size):
            new_pop.append(sorted_pop[i].copy())
        
        # Crossover and mutation
        while len(new_pop) < pop_size:
            # Selection (tournament)
            tourn_size = 3
            idx1 = np.random.choice(elite_size, tourn_size, replace=False)
            idx2 = np.random.choice(elite_size, tourn_size, replace=False)
            parent1 = sorted_pop[np.argmin([sorted_fit[i] for i in idx1])]
            parent2 = sorted_pop[np.argmin([sorted_fit[i] for i in idx2])]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child[i] += np.random.normal(0, 0.05)
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
            
            new_pop.append(child)
            
        population = new_pop[:pop_size]
        
        # Recalculate fitness
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
    
    # Final L-BFGS refinement with boundary constraints
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        # Try several starting points
        best_final = best_h.copy()
        best_final_c5 = best_c5
        
        h_refine = best_h.copy()
        result = minimize(
            objective,
            h_refine,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2},
            options={'ftol': 1e-10, 'gtol': 1e-10}
        )
        
        if result.success:
            h_final = result.x
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_final_c5:
                best_final = h_final
                best_final_c5 = c5_final
                
        # Final check with exact optimization
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add constraint matrix to make it exact
            corr = np.correlate(best_final, 1 - best_final, mode='full')
            max_k_idx = np.argmax(corr)
            max_lag = max_k_idx - (n_points - 1)
            
            prob = cp.Problem(cp.Minimize(cp.sum(cp.multiply(h_opt, 1 - h_opt))), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-8, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                h_feasible = np.clip(h_feasible, 0, 1)
                h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_final_c5:
                    best_final = h_feasible
                    best_final_c5 = c5_feasible
        except Exception:
            pass
            
        best_h = best_final
        best_c5 = best_final_c5
            
    except:
        pass
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
