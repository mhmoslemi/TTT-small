import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # construct a reasonable starting point
        x = np.linspace(0, 2, n_points)
        h_start = np.ones(n_points) * 0.5
        # make a more structured one
        h_start[:n_points//2] = 1.0
        h_start[n_points//2:] = 0.0
        # normalize
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a hybrid approach: gradient-based with local restarts
    # Simple local search with random perturbations
    max_iter = 1000
    perturbation_scale = 0.05
    n_restart = 20
    patience = 50
    
    for restart in range(n_restart):
        if time() - start_time > budget_s * 0.95:
            break
        
        h_current = best_h.copy()
        c5_current = best_c5
        no_improvement_count = 0
        
        for iter_num in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate perturbation around h_current
            perturbation = np.random.normal(0, perturbation_scale, n_points)
            h_trial = h_current + perturbation
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            
            if c5_trial < c5_current:
                h_current = h_trial.copy()
                c5_current = c5_trial
                no_improvement_count = 0
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
            else:
                no_improvement_count += 1
                if no_improvement_count > patience:
                    break

    # Final refinement using convex optimization on key constraints
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraints from best solution so far
        h_test = best_h.copy()
        corr = np.correlate(h_test, 1 - h_test, mode='full')
        # use top lags to define constraints
        top_lags = np.argsort(corr)[-15:]
        for k in top_lags:
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final verification and return
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
