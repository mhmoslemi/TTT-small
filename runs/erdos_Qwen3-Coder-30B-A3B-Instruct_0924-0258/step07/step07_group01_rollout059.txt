import numpy as np
from time import time
from scipy.optimize import minimize
from cvxpy import *

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    n_points = 35
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    max_iter = 50
    tighten_factor = 0.995
    max_overlap_initial = 0.38085
    max_overlap = max_overlap_initial
    
    # Helper function to create and solve MILP
    def solve_milp(h0, max_overlap_bound):
        h_vars = Variable(n_points)
        overlap_vars = Variable(2 * n_points - 1)
        
        # Constraints for h
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            sum(h_vars) == n_points / 2
        ]
        
        # Enforce overlap constraint using McCormick envelopes
        for k in range(2 * n_points - 1):
            # Create bounding box for h[i] * (1 - h[i + k])
            i_start = max(0, k - n_points + 1)
            i_end = min(n_points - 1, k)
            for i in range(i_start, i_end + 1):
                j = i + k - (n_points - 1)
                if 0 <= j < n_points:
                    # McCormick envelopes
                    w = Variable()
                    constraints.append(w >= h_vars[i] - h_vars[i] * h_vars[j])
                    constraints.append(w >= h_vars[j] - h_vars[i] * h_vars[j])
                    constraints.append(w <= h_vars[i] * (1 - h_vars[j]))
                    constraints.append(w >= 0)
                    
                    # Accumulate overlap terms (simplified approximation)
                    if k < n_points:
                        overlap_vars[k] += w
                    
        # Maximize overlap constraint
        constraints.append(overlap_vars[0] <= max_overlap_bound * dx)
        
        # Objective: minimize sum of overlap terms
        objective = Minimize(sum(overlap_vars))
        
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=GUROBI, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                return h_vars.value
        except:
            pass
        return None

    # Iterative tightening with MILP
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try to find solution with current max_overlap
        h_solution = solve_milp(best_h, max_overlap)
        
        if h_solution is not None:
            h_new = np.array(h_solution).flatten()
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                
        # Tighten max_overlap
        max_overlap *= tighten_factor

    # Final refinement with SLSQP
    h_refined = best_h.copy()
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, h_refined, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
