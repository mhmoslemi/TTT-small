import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    generations = 1000
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    elite_size = 5
    convex_interval = 50
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Evolutionary loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        new_population = population[:elite_size]
        new_fitness = fitness[:elite_size]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitness, tournament_size)
            if np.random.rand() < crossover_rate:
                parent2 = tournament_selection(population, fitness, tournament_size)
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            child = mutate(child, mutation_rate)
            child = repair_constraint(child, n_points, dx)
            
            # Evaluate
            c5 = evaluate_erdos_solution(child, dx)
            new_population.append(child)
            new_fitness.append(c5)
            
            if c5 < best_c5:
                best_h = child.copy()
                best_c5 = c5
        
        population = new_population
        fitness = new_fitness
        
        # Convex relaxation every convex_interval generations
        if gen % convex_interval == 0 and gen > 0:
            try:
                top_candidates = population[:min(5, len(population))]
                for h_candidate in top_candidates:
                    try:
                        h_opt = cp.Variable(n_points)
                        constraints = [
                            cp.sum(h_opt) == n_points / 2.0,
                            h_opt >= 0,
                            h_opt <= 1
                        ]
                        
                        # Add constraints for a few k values
                        k_vals = np.linspace(0.05, 1.95, min(10, n_points//3))
                        for i, k_val in enumerate(k_vals):
                            k = int(k_val * n_points / 2)
                            if k < n_points and k > 0:
                                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                                # Use a conservative bound
                                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                        
                        prob = cp.Problem(cp.Minimize(0), constraints)
                        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                        if prob.status == cp.OPTIMAL:
                            h_feasible = h_opt.value
                            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                            if c5_feasible < best_c5:
                                best_h = h_feasible
                                best_c5 = c5_feasible
                    except Exception:
                        pass
            except Exception:
                pass
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)

def tournament_selection(population, fitness, k):
    indices = np.random.choice(len(population), k)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx]

def crossover(parent1, parent2):
    mask = np.random.rand(len(parent1)) < 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(individual, mutation_rate):
    mask = np.random.rand(len(individual)) < mutation_rate
    individual += np.random.normal(0, 0.03, len(individual)) * mask
    return individual

def repair_constraint(h, n_points, dx):
    h = np.clip(h, 0, 1)
    h = h * (n_points/2) / np.sum(h)
    return h
