import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        # Try to create a more structured function that avoids local optima
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a hybrid approach: differential evolution for global search + local refinement
    # First, try optimization with bounds
    try:
        # Use scipy minimize with SLSQP for local refinement
        from scipy.optimize import differential_evolution
        
        def objective(h):
            # Normalize the function
            h_norm = h * (n_points / 2) / np.sum(h)
            return evaluate_erdos_solution(h_norm, dx)
        
        # Bounds for each variable
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run differential evolution
        result = differential_evolution(
            objective,
            bounds,
            maxiter=100,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            tol=1e-6,
            disp=False
        )
        
        if result.success:
            h_candidate = result.x * (n_points/2) / np.sum(result.x)
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
    except Exception:
        pass

    # Local refinement using Powell method
    try:
        def objective_local(h):
            h_norm = h * (n_points / 2) / np.sum(h)
            return evaluate_erdos_solution(h_norm, dx)
        
        # Random starting point near current best
        h_init = best_h + np.random.normal(0, 0.01, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points/2) / np.sum(h_init)
        
        # Optimize using Powell's method
        result = minimize(
            objective_local,
            h_init,
            method='Powell',
            tol=1e-8,
            options={'maxiter': 500, 'disp': False}
        )
        
        if result.success:
            h_candidate = result.x * (n_points/2) / np.sum(result.x)
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
                
    except Exception:
        pass

    # Final convex optimization pass to refine further
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Solve the problem using convex optimization with our best solution
        # as initial guess
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            # Project to meet the sum constraint exactly
            h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
