import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Benders decomposition approach
    max_iter = 5000
    convergence_tol = 1e-6
    perturbation_mag = 0.05
    perturbation_decay = 0.95
    
    # Initialize
    h_current = best_h.copy()
    prev_c5 = best_c5
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all shifts
        corr_full = np.correlate(h_current, 1 - h_current, mode='full')
        # Only consider valid shifts (0 to n_points-1)
        corr_valid = corr_full[n_points-1:] * dx  # Only the overlap terms
        max_k_idx = np.argmax(corr_valid)
        max_k = max_k_idx
        
        # Add constraint from worst case shift
        # For this constraint, we set: sum_{i=0}^{n_points-1-k} h_i * (1 - h_{i+k}) <= C5
        # which corresponds to corr_valid[max_k] <= C5 * dx
        
        # Solve Convex Relaxation
        try:
            h_new = cp.Variable(n_points)
            constraints = [
                cp.sum(h_new) == n_points / 2.0,
                h_new >= 0,
                h_new <= 1
            ]
            
            # Add a constraint from the current worst shift
            k = max_k
            if k > 0:
                # Constraint: sum_{i=0}^{n_points-1-k} h_i * (1 - h_{i+k}) <= C5 * dx
                # This is approximated by a linear constraint via first-order approximation
                # or use exact formulation if possible
                # For now, use direct constraint with underlying logic
                overlap_sum = cp.sum(cp.multiply(h_new[:n_points-k], 1 - h_new[k:]))
                constraints.append(overlap_sum <= best_c5 * dx + 1e-8)
            
            # Objective: minimize objective (but we're already at best_c5)
            # This approach: we just try to minimize h while respecting the constraint
            # We modify the approach to use actual convex optimization more directly
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_new.value
                # compute overlap for this feasible solution
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass
    
        # Perturb h_current and add some exploration
        h_current = h_current + perturbation_mag * np.random.randn(n_points)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Evaluate
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        # Update perturbation magnitude
        perturbation_mag *= perturbation_decay
        
        # Check for convergence
        if abs(c5_current - prev_c5) < convergence_tol:
            break
        prev_c5 = c5_current
        
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
