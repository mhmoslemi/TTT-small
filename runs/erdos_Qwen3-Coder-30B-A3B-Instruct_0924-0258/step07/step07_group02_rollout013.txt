import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Population-based evolutionary algorithm
    pop_size = 100
    n_points = 35
    max_generations = 250
    tournament_size = 5
    elite_size = 10
    
    # Initialize population
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
        for _ in range(pop_size - 1):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Start with a simple construction
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        population = [h_start]
        for _ in range(pop_size - 1):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Calculate initial fitness
    fitness = []
    for h in population:
        c5 = evaluate_erdos_solution(h, 2.0 / n_points)
        fitness.append(c5)
    
    # Initialize best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Hall of fame for top solutions
    hall_of_fame = []
    for i in np.argsort(fitness)[:elite_size]:
        hall_of_fame.append((population[i].copy(), fitness[i]))
    
    # Evolutionary loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        # Update hall of fame
        for i in range(min(elite_size, len(sorted_fit))):
            h, c5 = hall_of_fame[i]
            if c5 > sorted_fit[i]:
                hall_of_fame[i] = (sorted_pop[i].copy(), sorted_fit[i])
                
        # Create next generation
        new_population = []
        new_fitness = []
        
        # Elitism: keep top performers
        for i in range(elite_size):
            new_population.append(sorted_pop[i].copy())
            new_fitness.append(sorted_fit[i])
            
        # Generate offspring through crossover and mutation
        for i in range(pop_size - elite_size):
            # Tournament selection
            selected_indices = np.random.choice(len(sorted_pop), tournament_size, replace=False)
            selected_fitness = [sorted_fit[j] for j in selected_indices]
            winner_idx = selected_indices[np.argmin(selected_fitness)]
            
            # Crossover with a second parent
            parent1 = sorted_pop[winner_idx]
            parent2_idx = np.random.choice(len(sorted_pop), 1, replace=False)[0]
            parent2 = sorted_pop[parent2_idx]
            
            # Arithmetic crossover
            alpha = np.random.uniform(0.2, 0.8)
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation: Gaussian and bit-flip
            mutation_rate = 0.2
            if np.random.rand() < 0.5:  # Gaussian mutation
                # Decrease variance over time
                var = 0.1 * (1 - generation / max_generations)
                noise = np.random.normal(0, var, n_points)
                child += noise
            else:  # Bit-flip mutation for top performers
                if generation < max_generations // 2:
                    top_indices = sorted_indices[:pop_size // 3]
                else:
                    top_indices = sorted_indices[:pop_size // 6]
                if np.random.rand() < 0.3 and len(top_indices) > 0:
                    idx_to_flip = np.random.choice(top_indices)
                    child[idx_to_flip] = 1 - child[idx_to_flip]
            
            # Ensure feasibility
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            c5 = evaluate_erdos_solution(child, 2.0 / n_points)
            new_fitness.append(c5)
            
        population = new_population
        fitness = new_fitness
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
    
    # Final refinement with local optimization
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add constraints from top solutions
        for h_sol, _ in hall_of_fame:
            corr = np.correlate(h_sol, 1 - h_sol, mode='full')
            max_k_idx = np.argmax(corr)
            max_lag = max_k_idx - (n_points - 1)
            # Add constraint for the most active lags
            if max_lag > 0:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-max_lag], 1 - h_opt[max_lag:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[-max_lag:], 1 - h_opt[:n_points+max_lag]))
            # Use slightly larger threshold to allow small improvements
            constraints.append(corr_sum <= best_c5 * (2.0 / n_points) + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, 2.0 / n_points)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final verification to make sure we're feasible
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    return (list(best_h), best_c5, n_points)
