import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    n_points = 35
    pop_size = 100
    population = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        # Create a good starting point
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
    
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # Generate initial population
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Perturb the initial solution
            h = h_start + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # Evolutionary Algorithm
    max_generations = 250
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_count = 10
    tournament_size = 5
    adaptive_mutation_decay = 0.9
    
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, 2.0/n_points)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new population
        new_population = [population[i] for i in range(elite_count)]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            selected = []
            for _ in range(2):
                tournament_indices = np.random.choice(range(pop_size), 
                                                  size=tournament_size, replace=False)
                tournament_fitness = [fitness[i] for i in tournament_indices]
                winner_index = tournament_indices[np.argmin(tournament_fitness)]
                selected.append(population[winner_index])
            
            # Crossover
            if np.random.rand() < crossover_rate and len(selected) == 2:
                alpha = np.random.uniform(0.2, 0.8)
                child = alpha * selected[0] + (1 - alpha) * selected[1]
            else:
                child = selected[0] if len(selected) > 0 else population[0].copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation_strength = 0.1 * (adaptive_mutation_decay ** (gen // 50))
                for i in range(n_points):
                    if np.random.rand() < 0.1:  # Bit-flip
                        child[i] = 1 - child[i]
                    else:  # Gaussian perturbation
                        child[i] += np.random.normal(0, mutation_strength)
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:pop_size]
    
    # Final local optimization on top solutions
    top_solutions = population[:5]
    dx = 2.0 / n_points
    
    for h in top_solutions:
        # Try to refine with cvxpy
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            continue
    
    # Ensure final solution meets constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
