import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 100
    generations = 300
    elite_size = pop_size // 10
    mutation_rate = 0.2
    crossover_rate = 0.7
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Track hall of fame
    hall_of_fame = [(h_start.copy(), best_c5)]
    
    # EA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Update hall of fame
        for i in range(pop_size):
            if fitness[i] < hall_of_fame[0][1]:
                hall_of_fame.insert(0, (population[i].copy(), fitness[i]))
        hall_of_fame = hall_of_fame[:elite_size]
        
        # Sort population
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check convergence
        if gen > 0 and abs(fitness[0] - best_c5) < 1e-6:
            if gen > 80:
                break
        else:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Create new population
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Selection and reproduction
        while len(new_population) < pop_size:
            # Tournament selection
            p1 = population[np.random.randint(0, pop_size//2)]
            p2 = population[np.random.randint(0, pop_size//2)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                offspring = alpha * p1 + (1 - alpha) * p2
            else:
                offspring = p1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Adaptive mutation based on generation
                mutation_strength = 0.1 * (1 - gen / generations) + 0.02
                noise = np.random.normal(0, mutation_strength, n_points)
                offspring += noise
                offspring = np.clip(offspring, 0, 1)
            
            # Repair
            offspring = offspring * (n_points/2) / np.sum(offspring)
            
            new_population.append(offspring)
        
        population = new_population[:pop_size]
    
    # Final SQP refinement on top candidates
    best_h = hall_of_fame[0][0].copy()
    best_c5 = hall_of_fame[0][1]
    
    try:
        # Try refining with CVXOPT for constrained optimization
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Create a rough approximation of key constraints
        sample_indices = np.linspace(0, n_points-1, min(20, n_points), dtype=int)
        
        for k in sample_indices:
            if k < n_points // 2:
                term = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                term = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(term <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(cp.sum(h_opt)), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass
    except Exception:
        pass
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
