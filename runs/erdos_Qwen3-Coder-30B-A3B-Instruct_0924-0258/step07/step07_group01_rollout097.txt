import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 200
    elite_size = int(0.2 * population_size)
    mutation_rate = 0.3
    stagnation_limit = 50
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    best_gen = np.min(fitness)
    stagnation_count = 0
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check if we improved
        if fitness[0] < best_gen:
            best_gen = fitness[0]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > stagnation_limit:
            break
            
        # Create new population
        new_population = population[:elite_size]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(population[:elite_size], 2)
            # Arithmetic crossover
            alpha = random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            # Mutation
            if random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
            # Clip and normalize
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            new_population.append(child)
            
        population = new_population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Final refinement with SLSQP
    best_idx = np.argmin(fitness)
    h_refined = population[best_idx].copy()
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
