import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T0 = 0.1
    cooling_rate = 0.001
    max_iter = 1000
    perturbation_fraction = 0.15
    
    current_h = h_start.copy()
    current_c5 = best_c5
    T = T0
    
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb a fraction of the points
        n_perturb = int(n_points * perturbation_fraction)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Apply perturbation
        delta = np.random.uniform(-0.1, 0.1, n_perturb)
        new_h = current_h.copy()
        new_h[indices] += delta
        
        # Clip to feasible region
        new_h = np.clip(new_h, 0, 1)
        
        # Normalize to satisfy constraint
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject based on SA criterion
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.random() < np.exp(-delta_c5 / T):
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Cool down temperature
        T = T0 * np.exp(-cooling_rate * iteration)
        
        if stagnation_count > 100:
            break
    
    # Gradient Descent refinement
    def gradient_descent_refinement(h0, max_iter=500, lr=0.1):
        h = h0.copy()
        best_h = h.copy()
        best_c5 = evaluate_erdos_solution(h, dx)
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.98:
                break
                
            # Find the maximum overlap lag
            corr = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(corr)
            max_overlap = corr[k_max]
            
            # Compute gradient (focus on the max overlap lag)
            grad = np.zeros_like(h)
            if k_max < len(h):
                # Gradient contribution to h[i] for h[i+k_max]
                for i in range(len(h) - k_max):
                    grad[i] += (1 - h[i + k_max]) * dx
            else:
                # Gradient contribution to h[i] for h[i-k_max]
                for i in range(k_max, len(h)):
                    grad[i] += (1 - h[i - k_max]) * dx
            
            # Update step with gradient
            h = h - lr * grad
            
            # Clip and normalize
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            
            c5 = evaluate_erdos_solution(h, dx)
            
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            else:
                # Reduce learning rate if no improvement
                lr *= 0.95
        
        return best_h, best_c5
    
    # Apply GD refinement
    try:
        h_gd, c5_gd = gradient_descent_refinement(best_h)
        if c5_gd < best_c5:
            best_h = h_gd
            best_c5 = c5_gd
    except Exception:
        pass
    
    # Final SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Take a small sample of the boundary
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
