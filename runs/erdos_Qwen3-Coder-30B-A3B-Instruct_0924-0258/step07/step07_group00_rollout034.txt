import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent approach
    max_iter = 1000
    step_size = 0.01
    patience = 50
    no_improvement = 0
    
    h = h_start.copy()
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        # If we found a better solution, update
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement >= patience:
            break
            
        # Compute gradient for the lag that gives maximum overlap
        # We need to handle the circular nature of correlation
        lag = max_lag_idx - (n_points - 1)
        grad = np.zeros(n_points)
        
        # Gradient computation for the specific lag (this is an approximation)
        for j in range(n_points):
            k = (j + lag) % n_points
            if 0 <= k < n_points:
                grad[j] = 2.0 * (1.0 - h[k] - h[j]) / n_points
        
        # Update using gradient descent
        h_new = h - step_size * grad
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to meet constraint
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        # Check if the new solution is valid (sometimes gradient might cause issues)
        if np.sum(h_new) > 0:
            h = h_new.copy()
        else:
            # Reset on failure
            h = h_start.copy()
            
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
