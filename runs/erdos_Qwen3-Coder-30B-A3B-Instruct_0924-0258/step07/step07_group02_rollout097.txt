import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a simple ramp function
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing approach with adaptive cooling
    max_iter = 1000
    temp = 1.0
    cooling_rate = 0.995
    min_temp = 1e-6
    accept_prob = 0.8
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor solution
        h_new = best_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05)
        h_new[idx] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Metropolis acceptance criterion
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        else:
            # Accept with probability based on temperature
            if np.random.rand() < accept_prob * np.exp(-(c5_new - best_c5) / temp):
                best_h = h_new.copy()
                best_c5 = c5_new
                
        # Cool down
        temp = max(temp * cooling_rate, min_temp)
        
        # Occasionally refine with local quadratic optimization
        if iter_num % 50 == 0 and time() - start_time < budget_s * 0.9:
            try:
                # Local optimization using L-BFGS on a small subset
                indices = np.random.choice(n_points, size=min(15, n_points), replace=False)
                h_local = best_h.copy()
                h_local[indices] = np.clip(h_local[indices] + np.random.normal(0, 0.01, len(indices)), 0, 1)
                h_local = h_local * (n_points/2) / np.sum(h_local)
                
                c5_local = evaluate_erdos_solution(h_local, dx)
                if c5_local < best_c5:
                    best_h = h_local.copy()
                    best_c5 = c5_local
            except Exception:
                pass
    
    # Final SQP refinement using CVXPY
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add a few key constraints to guide the optimization
        h_ref = best_h.copy()
        corr = np.correlate(h_ref, 1 - h_ref, mode='full')
        top_k = np.argsort(corr)[-5:]  # top 5 lags
        for k in top_k:
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception as e:
        pass

    # Final verification and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
