import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with trust region
    max_iter = 5000
    convergence_tol = 1e-6
    
    # Initialize h
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Trust region parameters
    trust_radius = 0.1
    alpha = 0.1
    trust_radius_factor = 1.1
    backtracking_factor = 0.5
    max_backtrack = 10
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update current h to satisfy constraint
        sum_h = np.sum(h_current)
        if abs(sum_h - n_points/2) > 1e-6:
            h_current = h_current * (n_points/2) / sum_h
            
        # Compute current value
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        # Compute gradient for the max overlap constraint
        # Find which k is active
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        active_k = np.argmax(corr)
        
        # Compute gradient of the active constraint
        grad = np.zeros(n_points)
        if active_k < n_points:
            # Constraint S_k = sum h_i (1 - h_{i+k}) for i = 0..n-1-k
            # grad = 1 - 2*h_{i+k} for i=0..n-1-k, 
            #        -h_i for i=0..n-1-k, 
            #        0 otherwise
            for i in range(n_points - active_k):
                grad[i] += 1 - 2 * h_current[i + active_k]
                if i + active_k < n_points:
                    grad[i + active_k] -= h_current[i]
        else:
            # for large k, the correlation depends on later indices
            k = active_k - n_points + 1
            for i in range(n_points - k):
                grad[i + k] += 1 - 2 * h_current[i]
                if i < n_points - k:
                    grad[i] -= h_current[i + k]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-12:
            grad = grad / grad_norm
            
        # Try a step
        h_trial = h_current - alpha * grad
        
        # Project to [0,1]
        h_trial = np.clip(h_trial, 0, 1)
        
        # Scale to correct sum
        sum_trial = np.sum(h_trial)
        if abs(sum_trial - n_points/2) > 1e-8:
            h_trial = h_trial * (n_points/2) / sum_trial
            
        # Check if this step reduces the objective
        trial_c5 = evaluate_erdos_solution(h_trial, dx)
        
        if trial_c5 < current_c5 and trial_c5 < best_c5:
            h_current = h_trial.copy()
            best_c5 = trial_c5
            trust_radius = min(1.0, trust_radius * trust_radius_factor)
            if best_c5 < 0.3807:
                break
        else:
            # Backtracking line search
            for _ in range(max_backtrack):
                alpha *= backtracking_factor
                h_trial = h_current - alpha * grad
                h_trial = np.clip(h_trial, 0, 1)
                sum_trial = np.sum(h_trial)
                if abs(sum_trial - n_points/2) > 1e-8:
                    h_trial = h_trial * (n_points/2) / sum_trial
                    
                trial_c5 = evaluate_erdos_solution(h_trial, dx)
                if trial_c5 < current_c5:
                    h_current = h_trial.copy()
                    best_c5 = trial_c5
                    trust_radius = min(1.0, trust_radius * trust_radius_factor)
                    break
            else:
                trust_radius *= backtracking_factor
                
        if abs(current_c5 - best_c5) < convergence_tol:
            break
            
    # Final polishing with scipy
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try to improve with local optimization
        res = minimize(objective, h_current, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    # Ensure final constraint
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
