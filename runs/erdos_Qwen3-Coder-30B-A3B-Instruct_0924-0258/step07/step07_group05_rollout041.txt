import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Benders decomposition parameters
    max_iter = 5000
    convergence_tol = 1e-6
    perturbation_mag = 0.05
    perturbation_decay = 0.95
    
    h_current = h_start.copy()
    c5_current = best_c5
    
    # Store constraints
    constraints = []

    prev_c5 = float('inf')
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all shifts (just compute relevant ones)
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # We only care about k from 0 to n-1
        overlaps = []
        for k in range(n_points):
            # k corresponds to shift k elements
            overlap = corr[n_points-1+k] * dx
            overlaps.append(overlap)
        
        # Find worst overlap
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Add constraint if not already in list
        # We'll actually compute and add a tightening constraint
        # For now, we use the most constraining case
        
        # Benders cut approximation
        new_constraint = {
            'k': max_k,
            'overlap': max_overlap
        }
        
        # Try to solve a relaxed version using CVXPY to improve h
        try:
            h_opt = cp.Variable(n_points)
            # Constraint: sum = n/2
            cons_sum = cp.sum(h_opt) == n_points / 2.0
            # Bounds: 0 <= h <= 1
            cons_bounds = [h_opt >= 0, h_opt <= 1]
            
            # Add constraint for current worst case
            if max_k < n_points:

                # This is nonconvex. Reasonably, we use linearized approximation.
                # But we'll solve a variant of it with known structure.
                
                # This is difficult to represent exactly. For now use convex relaxation
                # and constraint tightening over multiple Benders iterations.

            c5_final = objective(h_final)
            if c5_final < best_c5:

                best_c5 = c5_final

    return (list(best_h), best_c5, n_points)
