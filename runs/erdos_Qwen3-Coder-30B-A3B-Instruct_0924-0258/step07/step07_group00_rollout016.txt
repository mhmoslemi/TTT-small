import numpy as np
from scipy.optimize import fmin_l_bfgs_b
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Set up the problem
    n_points = 35
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Main loop with gradient descent
    max_iterations = 500
    step_size = 0.01
    tolerance = 1e-6
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient for identified lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Update h
        h_new = h - step_size * gradient
        
        # Enforce bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check improvement
        current_c5 = evaluate_erdos_solution(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            step_size = min(0.01, step_size * 1.2)
        else:
            step_size = max(1e-6, step_size * 0.5)
            
        h = h_new
        
        if step_size < 1e-6:
            break
    
    # Final local refinement with L-BFGS-B
    def objective_and_gradient(h_vec):
        h = h_vec.reshape(n_points)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        return max_overlap, gradient.flatten()
    
    bounds = [(0, 1) for _ in range(n_points)]
    refined_h, _, _ = fmin_l_bfgs_b(
        objective_and_gradient, 
        best_h, 
        bounds=bounds, 
        maxiter=200
    )
    
    # Renormalize after refinement
    refined_h = refined_h * (n_points/2) / np.sum(refined_h)
    final_c5 = evaluate_erdos_solution(refined_h, dx)
    
    # Return the best solution found
    if final_c5 < best_c5:
        return (list(refined_h), final_c5, n_points)
    else:
        return (list(best_h), best_c5, n_points)
