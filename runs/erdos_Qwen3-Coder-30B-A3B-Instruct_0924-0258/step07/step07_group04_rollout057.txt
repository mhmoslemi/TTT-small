import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))  # periodic guess
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
        # Add small random perturbations
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multiresolution approach
    resolutions = [20, 40, 80]  # coarse to fine
    perturbation_sizes = [0.15, 0.10, 0.05]  # adaptive perturbations
    iterations_per_res = [100, 150, 200]
    
    for res_idx, (n_points_new, pert_size, iters) in enumerate(zip(resolutions, perturbation_sizes, iterations_per_res)):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate or use best solution from previous resolution
        if res_idx == 0:
            h_current = best_h.copy()
        else:
            # Resample to new resolution
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_points_new)
            h_current = np.interp(x_new, x_old, h_current)
            # Add perturbations
            perturb = np.random.uniform(-pert_size, pert_size, n_points_new)
            h_current *= (1 + perturb)
            h_current = np.clip(h_current, 0, 1)
            
        n_points = n_points_new
        dx = 2.0 / n_points
        
        # Normalize
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Optimization loop
        for iteration in range(iters):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Select random points to perturb (10-20%)
            n_perturb = max(1, int(0.15 * n_points))
            indices = np.random.choice(n_points, n_perturb, replace=False)
            
            # Create candidate
            h_candidate = h_current.copy()
            perturb = np.random.uniform(-pert_size, pert_size, n_perturb)
            for i, idx in enumerate(indices):
                h_candidate[idx] *= (1 + perturb[i])
            
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
            
            # Evaluate
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept if better or with probability
            if c5_candidate < best_c5:
                h_current = h_candidate.copy()
                best_c5 = c5_candidate
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
            elif np.random.random() < 0.05:  # 5% chance to accept worse
                h_current = h_candidate.copy()
        
        if res_idx > 0:
            # Try to refine further
            try:
                h_refined = h_current.copy()
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(objective, h_refined, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    h_final = res.x
                    c5_final = objective(h_final)
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
            except:
                pass
    
    # Final polish with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
