import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        n_points = len(h_start)
    else:

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 200
    elite_size = population_size // 5
    mutation_rate = 0.3
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    stagnation_count = 0
    last_improvement = 0
    
    for generation in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            last_improvement = generation
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > 50:
            break
            
        # Create new population
        new_population = []
        
        # Elite preservation
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Create offspring
        while len(new_population) < population_size:
            # Selection (tournament)
            tournament_size = 5
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[winner_idx]
            
            if np.random.random() < crossover_rate:
                # Select second parent
                tournament_indices = np.random.choice(population_size, tournament_size)
                tournament_fitness = [fitness[i] for i in tournament_indices]
                winner_idx = tournament_indices[np.argmin(tournament_fitness)]
                parent2 = population[winner_idx]
                
                # Arithmetic crossover
                alpha = np.random.random()
                offspring = alpha * parent1 + (1 - alpha) * parent2
            else:
                offspring = parent1.copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                mutation_strength = 0.1
                for i in range(n_points):
                    if np.random.random() < 0.1:  # 10% chance to mutate each gene
                        offspring[i] += np.random.normal(0, mutation_strength)
                
                # Clip to bounds
                offspring = np.clip(offspring, 0, 1)
                
                # Renormalize
                offspring = offspring * (n_points/2) / np.sum(offspring)
            
            new_population.append(offspring)
        
        population = new_population[:population_size]
    
    # Local refinement with SLSQP on top solutions
    top_candidates = population[:5]
    for h_refined in top_candidates:
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
