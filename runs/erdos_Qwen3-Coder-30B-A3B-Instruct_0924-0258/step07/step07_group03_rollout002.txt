import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def gradient_descent_step(h, dx, learning_rate):
    n = len(h)
    # Compute correlations for all lags
    corr = np.correlate(h, 1 - h, mode='full')
    max_lag_idx = np.argmax(corr)
    max_lag = max_lag_idx - (n - 1)
    
    # Compute gradient for this specific lag
    grad = np.zeros(n)
    if max_lag >= 0:
        # h[i] contributes to correlation with (1-h[i+max_lag])
        for i in range(n - max_lag):
            grad[i] += (1 - h[i + max_lag])
        # h[i+max_lag] contributes to correlation with (1-h[i])
        for i in range(n - max_lag):
            grad[i + max_lag] -= h[i]
    else:
        # max_lag < 0
        # h[i] contributes to correlation with (1-h[i+max_lag])
        for i in range(-max_lag, n):
            grad[i] += (1 - h[i + max_lag])
        # h[i-max_lag] contributes to correlation with (1-h[i])
        for i in range(-max_lag, n):
            grad[i - max_lag] -= h[i]
    
    # Update h
    h_new = h - learning_rate * grad
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Scale to maintain constraint
    scale = (n/2) / np.sum(h_new)
    h_new = h_new * scale
    return h_new

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient Descent with adaptive learning rate
    iterations = 0
    learning_rate = 0.1
    stagnation_count = 0
    max_stagnation = 100
    
    while time() - start_time < budget_s * 0.8 and iterations < 10000:
        h_new = gradient_descent_step(best_h, dx, learning_rate)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            stagnation_count = 0
            # Reduce learning rate occasionally to allow fine-tuning
            if iterations % 100 == 0 and learning_rate > 1e-6:
                learning_rate *= 0.5
        else:
            stagnation_count += 1
            
        iterations += 1
        
        if stagnation_count > max_stagnation:
            break
    
    # Phase 2: Local refinement using L-BFGS-B
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Try with current best as starting point
    try:
        res = minimize(objective, best_h, method='L-BFGS-B',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 500})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
