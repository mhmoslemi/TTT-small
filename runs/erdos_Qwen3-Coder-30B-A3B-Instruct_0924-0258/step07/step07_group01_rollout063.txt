import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Simple construction: two intervals of equal size
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place two intervals of length 0.5 each
        # First at [0.5, 1.0], second at [1.5, 2.0]
        # In discrete terms, these should be around the center
        interval_length = 0.5  # Length of each interval
        interval1_start = 0.65
        interval2_start = 0.85
        interval1_end = interval1_start + interval_length
        interval2_end = interval2_start + interval_length
        
        # Convert to discrete indices
        start1_idx = int(interval1_start / dx)
        end1_idx = int(interval1_end / dx)
        start2_idx = int(interval2_start / dx)
        end2_idx = int(interval2_end / dx)
        
        # Make sure we're within bounds
        end1_idx = min(end1_idx, n_points)
        end2_idx = min(end2_idx, n_points)
        
        h_start[start1_idx:end1_idx] = 1.0
        h_start[start2_idx:end2_idx] = 1.0
        
        # Normalize so that sum(h) * dx = 1
        total_sum = np.sum(h_start)
        if total_sum > 0:
            h_start = h_start * (n_points / 2) / total_sum
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try slightly perturbing to see if we can improve
    best_c5 = best_c5
    best_h = best_h.copy()
    
    # Use a gradient-free optimization approach with simulated annealing ideas
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Perturb at random
    max_perturb_iter = 5000
    temp_start = 1.0
    temp_end = 0.001
    anneal_rate = (temp_end / temp_start) ** (1.0 / max_perturb_iter)
    
    temp = temp_start
    for i in range(max_perturb_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Make a small change
        h_trial = current_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.1, 0.1)
        h_trial[idx] += delta
        h_trial[idx] = np.clip(h_trial[idx], 0, 1)
        
        # Normalize
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        
        # Evaluate
        trial_c5 = evaluate_erdos_solution(h_trial, dx)
        
        # Accept or reject
        if trial_c5 < current_c5 or np.random.rand() < np.exp(-(trial_c5 - current_c5) / temp):
            current_h = h_trial
            current_c5 = trial_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        temp *= anneal_rate
        
    return (list(best_h), best_c5, n_points)
