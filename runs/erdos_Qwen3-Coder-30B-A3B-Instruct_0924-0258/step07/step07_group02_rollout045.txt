import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a smooth starting point
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(np.pi * x))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with local refinement
    T = 1.0
    T_min = 1e-5
    alpha = 0.99
    max_iter = 1000
    stagnation_limit = 100
    stagnation_count = 0
    
    h_current = best_h.copy()
    c5_current = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor
        h_new = h_current.copy()
        
        # Random perturbation
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05)
        h_new[idx] += delta
        
        # Enforce bounds and integral constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            stagnation_count = 0
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(c5_new - c5_current) / T):
                h_current = h_new
                c5_current = c5_new
            stagnation_count += 1
            
        # Cool down
        T = max(T_min, T * alpha)
        
        # Early stopping if stagnation
        if stagnation_count > stagnation_limit:
            break
    
    # Local refinement with scipy
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_integral(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    cons = [{'type': 'eq', 'fun': constraint_integral}, 
            {'type': 'ineq', 'fun': lambda h: constraint_bounds(h)[n_points:]}]
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                        options={'maxiter': 100, 'ftol': 1e-6})
        if result.success:
            refined_h = result.x
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
