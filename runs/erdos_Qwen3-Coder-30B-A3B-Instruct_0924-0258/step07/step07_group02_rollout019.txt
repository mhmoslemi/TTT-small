import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a starting point with better structure
        x = np.linspace(0, 2, n_points)
        # Try a smooth piecewise linear construction
        h_start = np.zeros(n_points)
        for i in range(n_points):
            if x[i] < 0.5:
                h_start[i] = 1.5 * x[i]
            elif x[i] < 1.5:
                h_start[i] = 1.0
            else:
                h_start[i] = 2.0 - 1.5 * x[i]
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with bounds
    max_iter = 2000
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Use scipy to minimize the max correlation
        def objective(h_vec):
            # Reshape and clip
            h_clipped = np.clip(h_vec, 0, 1)
            h_normalized = h_clipped * (n_points/2) / np.sum(h_clipped)
            return evaluate_erdos_solution(h_normalized, dx)
        
        # Constraints
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
            
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                            options={'maxiter': 10, 'ftol': 1e-6}, callback=None)
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass
    
    # Additional random search refinement
    for _ in range(200):
        if time() - start_time > budget_s * 0.95:
            break
        h_trial = best_h + np.random.normal(0, 0.01, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        c5_trial = evaluate_erdos_solution(h_trial, dx)
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
