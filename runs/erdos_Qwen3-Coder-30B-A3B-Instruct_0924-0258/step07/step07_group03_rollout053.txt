import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    max_iter = 2000
    lr = 0.1
    patience = 0
    patience_limit = 50
    last_improvement = 0
    
    # For random restarts
    restarts = 5
    best_solution = best_h.copy()
    best_value = best_c5
    
    for restart in range(restarts):
        if time() - start_time > budget_s * 0.8:
            break
            
        h_current = best_solution.copy() if restart > 0 else h_start.copy()
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.8:

            # Compute all correlations (overlaps)
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)

            # The maximum overlap occurs at lag = max_idx - (n_points - 1)

            # This is a simplified finite difference approximation of gradient
            gradient = np.zeros_like(h_current)

                temp_h[i] += 1e-6

            new_value = evaluate_erdos_solution(h_new, dx)
            
            if new_value < best_value:

                patience = 0
                
                if new_value < best_c5:
                    best_h = h_new.copy()
                    best_c5 = new_value
            else:
                patience += 1
                
            h_current = h_new

                patience = 0
                if lr < 1e-6:
                    break
    
    # Phase 2: Final refinement with L-BFGS
    def objective(h):
        return evaluate_erdos_solution(h, dx)

    except Exception:
        pass
