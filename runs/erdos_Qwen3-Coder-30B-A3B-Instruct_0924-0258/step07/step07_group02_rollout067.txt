import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct better starting point: ramp with sinusoidal perturbations
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(2 * np.pi * x)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a different optimization approach: direct NLP-based optimization
    # with simple gradient-aided method
    
    # Initialize parameters
    max_iter = 1000
    lr = 1e-3
    decay = 0.99
    patience = 10
    patience_counter = 0
    best_c5_last = best_c5
    
    h_current = best_h.copy()
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution and gradients
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Simple gradient estimation via finite differences
        # Compute gradient w.r.t. each variable
        grad = np.zeros_like(h_current)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (c5_plus - c5_current) / eps
        
        # Gradient descent step
        h_new = h_current - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        h_current = h_new
        
        # Adaptive learning rate
        if patience_counter > patience:
            lr *= decay
            patience_counter = 0
        
        # Early stopping based on improvement
        if abs(best_c5_last - best_c5) < 1e-8:
            break
        best_c5_last = best_c5
    
    # Refine final solution using local optimization
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add a few representative constraints related to key overlaps
        h_test = best_h.copy()
        corr = np.correlate(h_test, 1 - h_test, mode='full')
        top_lags = np.argsort(corr)[-15:]  # Top 15 lags
        for k in top_lags:
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-8)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
