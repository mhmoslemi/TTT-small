import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    max_iter = 1000
    initial_temp = 0.1
    cooling_rate = 0.001
    perturb_fraction = 0.1
    stagnation_limit = 100
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Track stagnation
    stagnation_count = 0
    
    # SA loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Calculate current temperature
        t = iteration / max_iter
        temp = initial_temp * np.exp(-cooling_rate * t)
        
        # Perturb solution
        perturb_indices = np.random.choice(n_points, 
                                          size=int(n_points * perturb_fraction), 
                                          replace=False)
        new_h = current_h.copy()
        
        for i in perturb_indices:
            # Add noise
            new_h[i] += np.random.uniform(-0.1, 0.1)
            # Clip to [0, 1]
            new_h[i] = np.clip(new_h[i], 0, 1)
            
        # Scale to maintain sum constraint
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Compute new objective
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            stagnation_count = 0
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(new_c5 - current_c5) / temp):
                current_h = new_h
                current_c5 = new_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Check for stagnation
        if stagnation_count > stagnation_limit:
            break
            
    # SLSQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
