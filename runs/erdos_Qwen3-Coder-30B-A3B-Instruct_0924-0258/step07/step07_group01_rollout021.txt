import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Load initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 100
    num_generations = 200
    elite_size = 20
    mutation_rate = 0.3
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    fitness = []
    
    # Start with initial solution
    population.append(h_start.copy())
    fitness.append(evaluate_erdos_solution(h_start, dx))
    
    # Fill rest with random samples
    for i in range(pop_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Track best solution
    best_idx = np.argmin(fitness)
    g_best = population[best_idx].copy()
    g_best_value = fitness[best_idx]
    
    stagnation_count = 0
    
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        fitness = np.array(fitness)
        # Sort by fitness
        sort_idx = np.argsort(fitness)
        population = [population[i] for i in sort_idx]
        fitness = fitness[sort_idx]
        
        # Update global best
        if fitness[0] < g_best_value:
            g_best = population[0].copy()
            g_best_value = fitness[0]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > 50:
            break
            
        # Create offspring
        new_population = population[:elite_size].copy()  # Elitism
        
        while len(new_population) < pop_size:
            # Tournament selection
            idx1 = np.random.randint(0, elite_size)
            idx2 = np.random.randint(0, elite_size)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Crossover
            if np.random.random() < crossover_rate:
                alpha = np.random.random()
                offspring = alpha * parent1 + (1 - alpha) * parent2
            else:
                offspring = parent1.copy()
                
            # Mutation
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.02, n_points)
                offspring += noise
                offspring = np.clip(offspring, 0, 1)
                
            # Normalize to correct integral
            offspring = offspring * (n_points/2) / np.sum(offspring)
            
            new_population.append(offspring)
            
        population = new_population[:pop_size]
    
    # Local refinement with SLSQP
    h_refined = g_best.copy()
    h_refined = np.clip(h_refined, 0, 1)
    h_refined = h_refined * (n_points/2) / np.sum(h_refined)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Refine top solutions from GA
    top_candidates = population[:5]
    for candidate in top_candidates:
        try:
            res = minimize(objective, candidate, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200})
            if res.success:
                candidate_refined = res.x
                c5_refined = objective(candidate_refined)
                if c5_refined < best_c5:
                    best_h = candidate_refined
                    best_c5 = c5_refined
        except Exception:
            continue
            
    return (list(best_h), best_c5, n_points)
