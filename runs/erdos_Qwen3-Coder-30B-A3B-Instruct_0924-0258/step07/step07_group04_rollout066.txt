import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a smooth, structured function that should perform well
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.2 * np.sin(3 * np.pi * x)  # Sine wave to avoid sharp edges
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    
    # Normalize to satisfy constraint sum(h) * dx = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Create bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define the Lagrangian with penalty terms for constraints
    def lagrangian(h):
        # Compute the objective
        c5 = evaluate_erdos_solution(h, dx)
        
        # Add penalty for sum constraint
        sum_penalty = 1000 * (np.sum(h) - n_points / 2.0)**2
        
        # Add penalties for bounds
        bound_penalty = 1000 * (np.sum(np.maximum(0, -h)) + 
                               np.sum(np.maximum(0, h - 1)))
        
        return c5 + sum_penalty + bound_penalty
    
    # Differential evolution to search for good initial solution
    de_start_time = time()
    de_result = differential_evolution(lagrangian, bounds, maxiter=200, popsize=50,
                                       mutation=(0.5, 1), recombination=0.9,
                                       seed=seed, atol=1e-6, tol=1e-6)
    
    # If we run out of time, return best solution found so far
    if time() - start_time > budget_s * 0.9:
        return (list(h_start), evaluate_erdos_solution(h_start, dx), n_points)
    
    # Now, refine with SLSQP for further improvement
    h_refined = de_result.x.copy()
    h_refined = np.clip(h_refined, 0, 1)
    h_refined = h_refined * (n_points/2) / np.sum(h_refined)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, h_refined, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < evaluate_erdos_solution(h_start, dx):
                return (list(h_final), c5_final, n_points)
    except Exception:
        pass
    
    # Return best solution found
    return (list(h_start), evaluate_erdos_solution(h_start, dx), n_points)
