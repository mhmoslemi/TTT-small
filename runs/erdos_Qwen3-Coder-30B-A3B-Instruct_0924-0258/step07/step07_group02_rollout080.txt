from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        h_start = np.zeros(n_points)
        h_start[0:n_points//2] = 0.5
        h_start[n_points//2:] = 0.75
        h_start = np.clip(h_start, 0, 1)
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Two-phase optimization to ensure convergence:
    # 1. Differential Evolution to explore global space
    # 2. L-BFGS-B to fine-tune
    
    # First phase: DE-based population search
    population_size = 50
    max_generations = 150
    
    # Initialize population
    population = []
    for i in range(population_size):
        h = best_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # DE parameters
    F = 0.8
    CR = 0.9
    mutation_factor = 0.5
    
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(population_size):
            # Select three different individuals
            idxs = [idx for idx in range(population_size) if idx != i]
            a, b, c = np.random.choice(idxs, 3, replace=False)
            
            # Mutation
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)

            trial = np.where(np.random.rand(n_points) < CR, mutant, population[i])

            c5_new = evaluate_erdos_solution(trial, dx)

            # Track best solution
            if c5_new < best_c5:
                best_h = trial.copy()
                best_c5 = c5_new
    
    # Second phase: Refinement using L-BFGS-B
    try:
        # Get top solutions for refinement
        candidates = []
        for i in range(min(10, len(population))):  # Top 10
            c5_val = evaluate_erdos_solution(population[i], dx)
            candidates.append((c5_val, population[i]))
        
        candidates.sort(key=lambda x: x[0])
        
        for _, h_refine in candidates[:3]:  # Try 3 best candidates
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0

                return np.concatenate([h, 1 - h])

            result = minimize(obj, h_refine, method='L-BFGS-B',

    except Exception:

    # Final verification and return

    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
