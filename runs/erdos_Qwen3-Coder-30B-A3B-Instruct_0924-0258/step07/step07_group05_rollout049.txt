import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Basis function approach with piecewise linear functions
    n_basis = 10
    # Initialize basis functions as triangular pulses
    x_vals = np.linspace(0, 2, n_points)
    basis_functions = []
    
    # Create basis: triangular pulses
    for i in range(n_basis):
        center = i * 2.0 / (n_basis - 1) if n_basis > 1 else 1.0
        width = 0.4
        basis = np.zeros(n_points)
        for j in range(n_points):
            x = x_vals[j]
            if abs(x - center) <= width:
                basis[j] = 1.0 - abs(x - center) / width
        basis_functions.append(basis)
    
    basis_matrix = np.array(basis_functions).T  # n_points x n_basis
    
    # Initial coefficients
    try:
        coeffs = np.linalg.lstsq(basis_matrix, h_start, rcond=None)[0]
    except:
        coeffs = np.random.rand(n_basis) * 0.1
    
    # Optimization parameters
    max_iter = 5000
    patience = 0
    max_patience = 50
    perturbation_magnitude = 0.05
    
    # Optimization loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Construct h from coefficients
        h = basis_matrix @ coeffs
        h = np.clip(h, 0, 1)
        
        # Enforce constraint ∫ h = 1
        h_sum = np.sum(h) * dx
        if abs(h_sum - 1.0) > 1e-8:
            h = h * (1.0 / h_sum)
        
        # Evaluate current C5
        current_c5 = evaluate_erdos_solution(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1
        
        # Gradient estimation
        grad = np.zeros(n_basis)
        eps = 1e-5
        
        for i in range(n_basis):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = basis_matrix @ coeffs_plus
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus) * dx
            if abs(h_plus_sum - 1.0) > 1e-8:
                h_plus = h_plus * (1.0 / h_plus_sum)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / eps

        # Simple gradient step
        step_size = 0.001
        coeffs_new = coeffs - step_size * grad
        coeffs = coeffs_new

        # Project coefficients to ensure h stays valid
        h_test = basis_matrix @ coeffs
        h_test = np.clip(h_test, 0, 1)
        
        h_sum = np.sum(h_test) * dx
        if abs(h_sum - 1.0) > 1e-8:
            coeffs = coeffs * (1.0 / h_sum)  # Renormalize

        # Perturbation if no improvement
        if patience > max_patience:
            coeffs += np.random.normal(0, perturbation_magnitude, n_basis)
            perturbation_magnitude *= 0.95
            patience = 0

        # Occasionally use convex relaxation to improve solution
        if iter_num % 100 == 0 and iter_num > 0:
            try:
                c = cp.Variable(n_basis)
                h_expr = basis_matrix @ c
                obj = cp.Minimize(cp.sum(cp.multiply(h_expr, 1 - h_expr[:n_points-1])))
                constraints = [
                    cp.sum(h_expr) == 1.0,
                    c >= 0,
                    h_expr >= 0,
                    h_expr <= 1
                ]
                prob = cp.Problem(obj, constraints)
                prob.solve(solver=cp.ECOS, verbose=False, eps=1e-6, max_iters=100)
                if prob.status == cp.OPTIMAL:
                    c_val = c.value
                    if c_val is not None:
                        coeffs = c_val
            except Exception:
                pass

    # Final evaluation with precise method
    h_final = basis_matrix @ coeffs
    h_final = np.clip(h_final, 0, 1)
    h_final_sum = np.sum(h_final) * dx
    if abs(h_final_sum - 1.0) > 1e-8:
        h_final = h_final * (1.0 / h_final_sum)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    if final_c5 < best_c5:
        best_h = h_final
        best_c5 = final_c5

    return (list(best_h), best_c5, n_points)
