import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based descent with adaptive learning rate
    h = best_h.copy()
    learning_rate = 0.1
    max_iter = 1000
    stagnation_counter = 0
    min_improvement = 1e-8
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute current correlations and identify worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        l_max = np.argmax(corr)
        current_c5 = corr[l_max] * dx
        
        # Compute gradient in terms of full correlation
        # This is a rough gradient estimation via finite differences
        # We want to reduce the maximum correlation
        grad = np.zeros_like(h)
        
        # Approximate gradient for each position
        epsilon = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1-h_plus, mode='full')
            grad[i] = (corr_plus[l_max] - corr[l_max]) / epsilon
            
        # Apply gradient descent with adaptive learning rate
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Perturb occasionally to avoid local minima
        if iteration % 50 == 0 and iteration > 0:
            perturb_indices = np.random.choice(n_points, size=int(n_points*0.1), replace=False)
            for idx in perturb_indices:
                h[idx] += np.random.uniform(-0.1, 0.1)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        
        # Check for improvement
        new_c5 = evaluate_erdos_solution(h, dx)
        if new_c5 + min_improvement < current_c5:
            stagnation_counter = 0
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            stagnation_counter += 1
            if stagnation_counter > 50:
                learning_rate *= 0.5
                stagnation_counter = 0
    
    # Phase 2: SQP refinement with constraints
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use the best solution from phase 1 as initial guess
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
