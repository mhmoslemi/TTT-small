import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimistic starting bound
    max_overlap_initial = 0.38085
    tighten_factor = 0.995
    max_iter = 50
    max_slsqp_iter = 500
    
    current_bound = max_overlap_initial
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Create optimization problem
        h = cp.Variable(n_points)
        z = cp.Variable((n_points, n_points), name="z")
        max_overlap = cp.Variable(name="max_overlap")
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1,
            max_overlap >= 0
        ]
        
        # McCormick envelopes for bilinear terms: z[i,j] >= h[i]*(1-h[j]) if i+k=j
        for i in range(n_points):
            for k in range(-n_points+1, n_points):
                j = i + k
                if 0 <= j < n_points:
                    # z[i,j] >= h[i] - h[i]*h[j] 
                    # z[i,j] >= h[j] - h[i]*h[j]
                    # z[i,j] >= 0
                    # z[i,j] >= h[i] + h[j] - 1
                    constraints += [
                        z[i, j] >= h[i] - h[i]*h[j],
                        z[i, j] >= h[j] - h[i]*h[j],
                        z[i, j] >= 0,
                        z[i, j] >= h[i] + h[j] - 1
                    ]
        
        # Set the boundary constraints using correlation
        # Manually identify relevant correlation terms
        for i in range(2*n_points - 1):
            k = i - (n_points - 1)
            if k >= 0:
                range_i = range(n_points - k)
                corr_terms = [z[i2, i2+k] for i2 in range_i]
                if corr_terms:
                    constraints += [cp.sum(corr_terms) * dx <= max_overlap]
            else:
                range_i = range(-k, n_points)
                corr_terms = [z[i2-k, i2] for i2 in range_i]
                if corr_terms:
                    constraints += [cp.sum(corr_terms) * dx <= max_overlap]
        
        # Objective: minimize max_overlap
        objective = cp.Minimize(max_overlap)
        
        # Solve with Gurobi or MOSEK if available, otherwise fall back to ECOS
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GUROBI, verbose=False)
        except:
            try:
                prob.solve(solver=cp.MOSEK, verbose=False)
            except:
                try:
                    prob.solve(solver=cp.CVXOPT, verbose=False)
                except:
                    try:
                        prob.solve(solver=cp.ECOS, verbose=False)
                    except:
                        break  # If all solvers fail, go to refinement
        
        if prob.status == cp.OPTIMAL:
            h_val = h.value
            new_c5 = evaluate_erdos_solution(h_val, dx)
            
            if new_c5 < best_c5:
                best_h = h_val.copy()
                best_c5 = new_c5
                
            current_bound *= tighten_factor
        else:
            # Fallback to local refinement if solver fails
            current_bound *= tighten_factor

        iter_count += 1

    # Refinement
    h_refined = best_h.copy()
    h_refined = np.clip(h_refined, 0, 1)
    h_refined = h_refined * (n_points/2) / np.sum(h_refined)

    def objective(h):
        return evaluate_erdos_solution(h, dx)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    try:
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': max_slsqp_iter, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
