import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Initialize with periodic-like function
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Annealing parameters
    max_iter = 10000
    T = 1.0
    alpha = 0.998
    tabu_size = 20
    tabu_list = []
    
    # Simulated Annealing with Tabu Search
    current_h = h_start.copy()
    current_c5 = best_c5
    stagnation_counter = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate
        candidate_h = current_h.copy()
        indices = np.random.choice(n_points, size=max(1, int(n_points * 0.15)), replace=False)
        for i in indices:
            # Random perturbation (±10% to ±20%)
            perturbation = np.random.uniform(0.05, 0.2)
            if np.random.random() > 0.5:
                perturbation = -perturbation
            candidate_h[i] += perturbation * candidate_h[i]
        
        # Clip and renormalize
        candidate_h = np.clip(candidate_h, 0, 1)
        normalization = (n_points / 2) / np.sum(candidate_h)
        candidate_h *= normalization
        
        # Check tabu
        if tuple(candidate_h.round(8)) in [tuple(h.round(8)) for h in tabu_list]:
            # Accept with low probability
            if np.random.random() > 0.95:
                pass
            else:
                continue
                
        # Evaluate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Acceptance criterion
        if candidate_c5 < current_c5 or np.random.random() < np.exp(-(candidate_c5 - current_c5) / T):
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            
            # Add to tabu list
            tabu_list.append(current_h.copy())
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
                
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                stagnation_counter = 0
            else:
                stagnation_counter += 1
        else:
            stagnation_counter += 1
            
        # Cooling schedule
        T = max(0.0001, T * alpha)
        
        # Reset occasionally to escape local minima
        if stagnation_counter > 300:
            # Reset to best solution so far
            current_h = best_h.copy()
            current_c5 = best_c5
            stagnation_counter = 0
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
