import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on the overlap bound
    max_iter = 20
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Define CVXPY problem
        h_var = cp.Variable(n_points)
        max_overlap = cp.Variable()
        
        # Objective: minimize max_overlap
        objective = cp.Minimize(max_overlap)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2.0,
            max_overlap >= 0
        ]
        
        # Add overlap constraint - approximate maximum overlap constraint
        # We'll use a heuristic which is to linearize the non-convex part
        # by forming a large matrix that computes all correlations via a sum
        # and impose an upper bound on the maximum correlation as a constraint
        # This is a simplification of a more accurate approach but will work as
        # a first-order approximation
        
        # Compute all non-zero correlations using sliding window approach
        h_conc = cp.hstack([h_var, h_var])
        
        # Convolution-like constraints: for each valid lag
        max_lag = 2 * n_points - 1
        for i in range(max_lag):
            corr_val = cp.sum(cp.multiply(h_var, cp.roll(1 - h_var, i - n_points + 1))) * dx
            constraints.append(corr_val <= max_overlap)
        
        # The above is a theoretically naive way to handle the max,
        # but given the practical limits and the optimizations below
        # it approximates the bound correctly for good enough solutions
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=300)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_var.value
                if h_new is not None:
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points/2) / np.sum(h_new)
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
        except:
            pass
            
        # Local refinement using SLSQP if needed
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            # Start from previous solution to avoid rock-bottom problems
            h_refined = best_h.copy()
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
        # Stopping condition based on iteration count or convergence
        if iteration > 10 and iteration % 5 == 0:
            if abs(best_c5 - evaluate_erdos_solution(best_h, dx)) < 1e-5:
                break
    
    # Final strict check
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
