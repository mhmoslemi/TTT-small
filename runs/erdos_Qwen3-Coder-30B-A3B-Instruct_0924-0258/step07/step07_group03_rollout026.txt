import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based descent
    max_iter = 1000
    lr = 0.1
    patience = 50
    no_improvement = 0
    target_sum = n_points / 2.0
    
    h_current = h_start.copy()
    h_current = h_current * target_sum / np.sum(h_current)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute correlations and find worst case
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        max_lag = max_idx - (len(h_current) - 1)
        max_val = corr[max_idx]
        
        # Compute gradient for the worst case
        grad = np.zeros_like(h_current)
        lag = max_lag
        if lag >= 0:
            # h[i] contributes to lag l if l = i - j, so for each h[i], 
            # we add -1 for every position where 1-h[j] has contribution
            for i in range(len(h_current)):
                j = i - lag
                if 0 <= j < len(h_current):
                    grad[i] -= 1
        else:
            # lag is negative, so i = j - lag, which means j = i + |lag|
            for i in range(len(h_current)):
                j = i + abs(lag)
                if 0 <= j < len(h_current):
                    grad[i] -= 1

        # Line search to adapt step size
        step_size = lr
        new_h = h_current - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            h_current = new_h.copy()
            best_c5 = new_c5
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > patience:
                lr *= 0.5
                no_improvement = 0
                if lr < 1e-6:
                    break

    # Phase 2: Local refinement with L-BFGS-B
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try multiple random restarts
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_init = h_current + np.random.normal(0, 0.01, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * target_sum / np.sum(h_init)
            
            res = minimize(objective, h_init, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 500})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
