import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Periodic initialization
        n_points = 20
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))  # 5 cycles over [0,2]
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
        
        # Add small random perturbations
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multiresolution approach: start with coarse grid and refine
    resolutions = [20, 40, 80]
    current_n = 20
    current_h = h_start.copy()
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to new resolution
        if res > current_n:
            current_h = np.interp(np.linspace(0, 2, res), 
                                np.linspace(0, 2, current_n), current_h)
            current_n = res
        elif res < current_n:
            # Downsample
            current_h = current_h[::current_n//res]
            current_n = res
        
        dx = 2.0 / current_n
        # Reverse previous best if it's already better
        temp_h = current_h.copy()
        temp_c5 = evaluate_erdos_solution(temp_h, dx)
        if temp_c5 < best_c5:
            best_h = temp_h.copy()
            best_c5 = temp_c5
        
        # Local optimization with simulated annealing
        current_h = temp_h.copy()
        temp_c5 = temp_c5
        
        # Temperature control
        T = 0.5
        T_min = 1e-6
        alpha = 0.995
        iteration = 0
        max_iter = 1000
        
        while T > T_min and iteration < max_iter:
            if time() - start_time > budget_s * 0.95:
                break
            
            # Create candidate
            candidate_h = current_h.copy()
            idxs = np.random.choice(current_n, size=max(1, current_n//10), replace=False)
            perturbation_size = T * 0.2
            for i in idxs:
                delta = np.random.normal(0, perturbation_size)
                candidate_h[i] += delta
            candidate_h = np.clip(candidate_h, 0, 1)
            candidate_h = candidate_h * (current_n/2) / np.sum(candidate_h)
            
            # Evaluate
            candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
            
            # Accept or reject
            if candidate_c5 < temp_c5:
                current_h = candidate_h
                temp_c5 = candidate_c5
                if temp_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = temp_c5
            else:
                # Accept with some probability
                p = np.exp(-(candidate_c5 - temp_c5) / T)
                if np.random.rand() < p:
                    current_h = candidate_h
                    temp_c5 = candidate_c5
            
            T = T * alpha
            iteration += 1
        
        if current_n == 80:
            break
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (current_n/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - current_n / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(current_n)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, current_n)
