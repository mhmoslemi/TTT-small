import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use existing initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient computation with trust region
    def compute_gradient(h, dx):
        n = len(h)
        grad = np.zeros(n)
        corr = np.correlate(h, 1-h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Gradient for active constraint
        # corr[k] = sum_{i=0}^{n-1-k} h[i] * (1 - h[i+k])
        # For each i: d/dh[i] = (1 - h[i+k]) 
        # For each (i+k): d/dh[i+k] = -h[i]  
        for k in range(n):
            if k < n:
                grad[k] = 0.0
                for i in range(n-k):
                    grad[k] += (1 - h[i+k])
                if k > 0:
                    for i in range(k, n):
                        grad[i] -= h[i-k]
        return grad * dx
    
    # Trust region ALM with gradient-based updates
    max_iter = 5000
    alpha = 0.1
    mu = 100.0
    tolerance = 1e-6
    trust_region = 0.1
    
    h_current = best_h.copy()
    prev_c5 = best_c5
    
    # Project to feasible region
    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-8:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Use existing method only if there are not many points
    if n_points * n_points < 20000:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_start, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_current = res.x
                current_c5 = objective(h_current)
                if current_c5 < best_c5:
                    best_h = h_current.copy()
                    best_c5 = current_c5
        except Exception:
            pass
    
    # ALM loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current gradient
        try:
            grad = compute_gradient(h_current, dx)
            # Alm update with penalty
            grad_lagrange = grad + mu * (h_current - 0.5)  # Lagrange term for bounds (simplified)
            h_update = h_current - alpha * grad_lagrange
            h_update = project_h(h_update)
            
            current_c5 = evaluate_erdos_solution(h_update, dx)
            
            # Accept if improvement
            if current_c5 < best_c5:
                h_current = h_update.copy()
                best_c5 = current_c5
                if abs(prev_c5 - current_c5) < tolerance:
                    alpha *= 0.9
                else:
                    alpha = min(alpha * 1.1, 0.5)
                prev_c5 = current_c5
            
            # Adjust penalty
            if current_c5 > best_c5:
                mu *= 1.1
            else:
                mu *= 0.95
            
            # Decrease step size if not good enough
            if (current_c5 > prev_c5 and iter_num % 10 == 0):
                alpha *= 0.5
            
        except Exception as e:
            # Fallback to project most recent
            h_current = project_h(h_current)
            
    # Final polish
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_current, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass

    # Ensure constraints
    best_h = project_h(best_h)
    return (list(best_h), best_c5, n_points)
