import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a structured initial guess
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Differential Evolution phase
    de_bounds = [(0, 1) for _ in range(n_points)]
    
    # Define an augmented objective with penalty for constraint violation
    def augmented_objective(h):
        penalty = 1e6 * abs(constraint_sum(h)) if abs(constraint_sum(h)) > 1e-3 else 0
        return objective(h) + penalty
    
    try:
        result_de = differential_evolution(
            augmented_objective,
            de_bounds,
            maxiter=1000,
            popsize=50,
            mutation=(0.5, 1.0),
            recombination=0.9,
            seed=seed,
            atol=1e-8,
            polish=False  # Disable local optimization to save time
        )
        
        if result_de.success:
            h_best = result_de.x
            h_best = np.clip(h_best, 0, 1)
            h_best = h_best * (n_points/2) / np.sum(h_best)
            c5_best = objective(h_best)
        else:
            h_best = h_start.copy()
            h_best = h_best * (n_points/2) / np.sum(h_best)
            c5_best = objective(h_best)
    except:
        h_best = h_start.copy()
        h_best = h_best * (n_points/2) / np.sum(h_best)
        c5_best = objective(h_best)
    
    # Local refinement with SLSQP if time permits
    if time() - start_time < budget_s * 0.9:
        try:
            res = minimize(objective, h_best, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < c5_best:
                    h_best = h_final
                    c5_best = c5_final
        except:
            pass
    
    return (list(h_best), c5_best, n_points)
