import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a smooth, low-amplitude sine wave
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.1 * np.sin(2 * np.pi * x)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    # Normalize to sum(h) = n_points / 2
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint for sum(h) = n_points/2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Add the constraint to the optimization
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # DE parameters
    max_iter = 500
    popsize = 50
    mutation = (0.5, 1)
    crossover = 0.9
    seed = 42
    
    # Define the objective function for DE
    def objective_de(h):
        return evaluate_erdos_solution(h, dx)
    
    # DE for global search
    try:
        result = differential_evolution(
            objective_de, 
            bounds, 
            args=(),
            maxiter=max_iter,
            popsize=popsize,
            mutation=mutation,
            crossover=crossover,
            seed=seed,
            disp=False,
            tol=1e-10,
            callback=lambda x, convergence: None
        )
        if result.success:
            h_de = result.x
            c5_de = result.fun
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    except Exception as e:
        pass

    # SLSQP refinement
    try:
        def objective_slsqp(h):
            return evaluate_erdos_solution(h, dx)
        
        res = minimize(
            objective_slsqp, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons_sum, 
            options={'maxiter': 1000, 'ftol': 1e-10}
        )
        if res.success:
            h_final = res.x
            c5_final = objective_slsqp(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final validation
    if time() - start_time > budget_s * 0.95:
        return (list(best_h), best_c5, n_points)
    
    # Clamp and normalize final solution
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
