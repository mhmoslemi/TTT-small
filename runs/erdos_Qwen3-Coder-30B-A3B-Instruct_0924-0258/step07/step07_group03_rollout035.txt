import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Convex relaxation with SDP to get better starting point
    if budget_s > 200:
        # Create SDP formulation
        h_sdp = cp.Variable(n_points)
        Z = cp.Variable((n_points, n_points), symmetric=True)
        t = cp.Variable()
        
        # Constraints for SDP relaxation
        constraints = [
            cp.diag(Z) == h_sdp,
            Z >> 0,  # Positive semidefinite
            cp.trace(Z) == n_points / 2.0,  # Ensures sum(h) = n_points/2
            t >= cp.max(cp.multiply(h_sdp, 1 - h_sdp) @ np.eye(n_points) + 
                        cp.conv(cp.multiply(h_sdp, 1 - h_sdp), np.ones(n_points))[:n_points])
        ]
        
        # Try to solve as SDP
        try:
            prob = cp.Problem(cp.Minimize(t), constraints)
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sdp_vals = h_sdp.value
                if h_sdp_vals is not None:
                    h_sdp_vals = np.clip(h_sdp_vals, 0, 1)
                    # Normalize to sum to n_points/2
                    h_sdp_vals = h_sdp_vals * (n_points / 2.0) / np.sum(h_sdp_vals)
                    objective_sdp = evaluate_erdos_solution(h_sdp_vals, dx)
                    if objective_sdp < best_c5:
                        best_h = h_sdp_vals
                        best_c5 = objective_sdp
        except:
            pass
    
    # Phase 2: Local optimization (LSQP)
    max_local_iter = 1000
    if time() - start_time < budget_s * 0.8:
        # Local optimization using SLSQP
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use best h so far as starting point
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': max_local_iter, 'ftol': 1e-10},
                       callback=lambda x: None)
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    
    # Final refinement with simple gradient descent if needed
    if time() - start_time < budget_s * 0.95:
        h_refine = best_h.copy()
        lr = 0.01
        eps = 1e-8
        max_iter_refine = 1000
        
        def gradient_descent_step(h):
            # Evaluate current value
            val = evaluate_erdos_solution(h, dx)
            
            # Approximate gradient by finite differences
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                val_plus = evaluate_erdos_solution(h_plus, dx)
                grad[i] = (val_plus - val) / eps
            
            # Update using gradient descent
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            return h_new
        
        for i in range(max_iter_refine):
            if time() - start_time > budget_s * 0.95:
                break
            h_new = gradient_descent_step(h_refine)
            val_new = evaluate_erdos_solution(h_new, dx)
            if val_new < best_c5:
                best_h = h_new
                best_c5 = val_new
            h_refine = h_new
    
    return (list(best_h), best_c5, n_points)
