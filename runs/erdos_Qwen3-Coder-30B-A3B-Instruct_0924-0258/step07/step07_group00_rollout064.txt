import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization
    h = h_start.copy()
    learning_rate = 0.01
    stagnation_count = 0
    max_stagnation = 100
    
    # Main optimization loop
    iteration = 0
    while time() - start_time < budget_s * 0.95:
        # Compute all correlations to find maximum
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_lag = max_idx - (len(h) - 1)
        
        # Compute gradient w.r.t. the maximum overlap
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + max_lag) % len(h)
            gradient[i] = 2.0 * (1.0 - h[j] - h[i])
        
        # Update with gradient descent
        h_new = h - learning_rate * gradient
        
        # Apply constraints: clip and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            stagnation_count = 0
            learning_rate = 0.01  # Reset learning rate
        else:
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                learning_rate *= 0.5
                stagnation_count = 0
                if learning_rate < 1e-6:
                    break
        
        h = h_new
        iteration += 1
    
    # Final refinement with SLSQP if needed and time allows
    if time() - start_time < budget_s * 0.9:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
