import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Start with a good upper bound for C5
    max_overlap_initial = 0.38085
    
    # Try to tighten linearly with iterations
    tighten_factor = 0.995
    max_iter = 50
    iter_count = 0
    
    # Iterative tightening approach
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        # Create optimization variables
        h = cp.Variable(n_points)
        max_overlap = cp.Variable()
        
        # Objective: minimize max_overlap
        objective = cp.Minimize(max_overlap)
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2.0,  # Integral constraint
            h >= 0,  # Non-negativity
            h <= 1,  # Boundedness
            max_overlap >= 0  # Non-negativity of overlap
        ]
        
        # Add bilinear constraint using McCormick envelopes
        # For k in range(2*n_points - 1), we compute sum over i: h[i] * (1 - h[i+k])
        # Since this is not directly linear, we use a sample approach
        
        # Sample overlap related constraints based on full correlation
        val = np.correlate(h_start, 1 - h_start, mode='full')
        max_correlation = np.max(val)
        
        # Approximate: we want to limit the overlap from above
        # Create a simpler form of constraints
        # The overlap integral (which we want minimized) is: max_k integral h(x) * (1 - h(x+k)) dx
        # We approximate this instead by ensuring the correlation doesn't exceed max_overlap
        
        # For now, we just solve the convex problem with approximate entry way:
        
        # Assuming we can model this somewhat through a squared constraint or constraint on
        # maximum correlation value - this model is more simplified but effective
        # in a MILP setting, we'd build proper McCormick envelopes. Since we can't
        # easily do full MILP here, we perform refined local search.
        
        # Skip kitchen sink MILP for now and directly optimize with SLSQP like before but from
        # better initialization
        
        # Refinement with SLSQP from current best
        def objective(h_array):
            h_array = np.clip(h_array, 0, 1)
            h_array = h_array * (n_points/2) / np.sum(h_array)
            return evaluate_erdos_solution(h_array, dx)
        
        def constraint_sum(h_array):
            return np.sum(h_array) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        h_init = best_h.copy()
        try:
            result = minimize(objective, h_init, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 500, 'ftol': 1e-10})
            if result.success:
                h_new = result.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
        except Exception:
            pass
            
        # Now offer a bit more systematic placement to possibly improve
        # Using gradient-based refinement and better initialization
        
        # With a slightly larger neighborhood around best found so far:
        # Perturb the best solution a bit and restart optimization
            
        # For robustness, do a multi-start random process as well
        h_perturbed = best_h + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        try:
            result = minimize(objective, h_perturbed, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 500, 'ftol': 1e-10})
            if result.success:
                h_new = result.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
        except Exception:
            pass

        iter_count += 1
        if iter_count < max_iter and (time() - start_time) > budget_s * 0.95:
            break
        
        max_overlap_initial *= tighten_factor
    
    return (list(best_h), best_c5, n_points)
