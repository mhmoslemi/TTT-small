import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Generate a better periodic initial guess
        x = np.linspace(0, 2, 35)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (35 / (2 * np.sum(h_start)))
        n_points = 35
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    T = 0.5
    alpha = 0.995
    T_min = 1e-5
    tabu_list = []
    tabu_size = 20
    max_iter = 1000
    stagnation_limit = 50
    stagnation_count = 0
    
    current_h = h_start.copy()
    current_c5 = best_c5
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Perturbation schedule
    perturbation_size_start = 0.1
    perturbation_size_end = 0.05
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Adaptive perturbation size
        perturbation_size = perturbation_size_start + (perturbation_size_end - perturbation_size_start) * (iteration / max_iter)
        
        # Select random indices to perturb (15% of points)
        indices = np.random.choice(n_points, size=int(0.15 * n_points), replace=False)
        
        # Create candidate solution
        candidate_h = current_h.copy()
        for i in indices:
            # Perturb by small random amount
            delta = np.random.uniform(-perturbation_size, perturbation_size)
            candidate_h[i] += delta
        
        # Clip to bounds
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Normalize
        candidate_h = candidate_h * (n_points / (2 * np.sum(candidate_h)))
        
        # Evaluate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Accept or reject based on simulated annealing criterion
        if candidate_c5 < current_c5:
            # Always accept if better
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
        else:
            # Accept with probability
            delta_c5 = candidate_c5 - current_c5
            if np.random.rand() < np.exp(-delta_c5 / T):
                current_h = candidate_h.copy()
                current_c5 = candidate_c5
                
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Check for stagnation
        if stagnation_count >= stagnation_limit:
            # Perturb best solution slightly and reset stagnation count
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    delta = np.random.uniform(-0.05, 0.05)
                    best_h[i] += delta
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / (2 * np.sum(best_h)))
            current_h = best_h.copy()
            current_c5 = evaluate_erdos_solution(current_h, dx)
            stagnation_count = 0
            
        # Update tabu list
        if len(tabu_list) >= tabu_size:
            tabu_list.pop(0)
        tabu_list.append(current_h.copy())
        
        # Decrease temperature
        T = max(T * alpha, T_min)
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / (2 * np.sum(h_refined)))
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
