import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based descent
    max_iter = 1000
    lr = 0.1
    stagnation_threshold = 50
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute overlaps
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag = np.argmax(corr)
        max_value = corr[max_lag]
        
        # Compute gradient for the max lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Count how many times h[i] contributes to the correlation
            for k in range(n_points):
                if 0 <= i+k < n_points:
                    grad[i] += (1 - best_h[i+k]) * (1 if k == max_lag - n_points + 1 else 0)
                if 0 <= i-k < n_points:
                    grad[i] += (1 - best_h[i-k]) * (1 if k == max_lag - n_points + 1 else 0)
        
        # Apply gradient descent update
        new_h = best_h - lr * grad
        
        # Apply constraint: clip values and scale
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count > stagnation_threshold:
                lr *= 0.5
                stagnation_count = 0
    
    # Phase 2: Local refinement with L-BFGS-B
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try multiple random restarts for better local search
    for _ in range(5):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_init = np.clip(best_h + np.random.normal(0, 0.01, n_points), 0, 1)
        h_init = h_init * (n_points/2) / np.sum(h_init)
        
        try:
            res = minimize(objective, h_init, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 300, 'ftol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            continue
    
    return (list(best_h), best_c5, n_points)
