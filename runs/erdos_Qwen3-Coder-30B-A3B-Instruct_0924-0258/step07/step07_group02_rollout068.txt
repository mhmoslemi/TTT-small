import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse optimization with gradient-based method
    opt_iters = 0
    max_opt_iters = 50
    
    for iter_num in range(max_opt_iters):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Regularized gradient descent with projection
        # Compute gradient approximation using finite differences
        h_current = best_h.copy()
        c5_current = best_c5

        # Compute correlations for current point
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_k_idx = np.argmax(corr)
        max_lag = max_k_idx - (n_points - 1)  # this point's max lag

        # Simple gradient descent step with small learning rate
        learning_rate = 0.01
        
        # Perturb in direction that reduces max overlap
        h_new = h_current.copy()
        
        # Gradient approximation using finite difference
        epsilon = 1e-4
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)

            h_minus = h_minus * (n_points/2) / np.sum(h_minus)

            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)

        # Update with gradient step (but ensure feasibility)
        h_new = h_current - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_new = evaluate_erdos_solution(h_new, dx)

            best_h = h_new.copy()
            best_c5 = c5_new

    best_c5 = evaluate_erdos_solution(best_h, dx)

    return (list(best_h), best_c5, n_points)
