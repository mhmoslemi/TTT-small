import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based descent
    max_iter = 1000
    learning_rate = 0.1
    h = h_start.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        l_max = np.argmax(corr)
        max_overlap = corr[l_max] * dx
        
        # Compute gradient (approximate)
        # The gradient w.r.t. h[i] is influenced by the correlation at l_max
        # We approximate the effect of changing h[i] on max_overlap
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # Approximate gradient by forward difference with small perturbation
            h_pert = h.copy()
            h_pert[i] += 1e-6
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
            grad[i] = (corr_pert[l_max] - corr[l_max]) / 1e-6
        
        # Update step and project to feasible region
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Perturb occasionally
        if iteration % 50 == 0:
            perturb_idx = np.random.choice(len(h), size=int(0.1*len(h)), replace=False)
            h[perturb_idx] += np.random.uniform(-0.1, 0.1, len(perturb_idx))
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        
        # Update best solution
        c5_current = evaluate_erdos_solution(h, dx)
        if c5_current < best_c5:
            best_h = h.copy()
            best_c5 = c5_current
            
        # Adaptive learning rate
        if iteration % 100 == 0:
            learning_rate *= 0.8
    
    # Phase 2: Refine using SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
