import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

    else:

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe optimization with perturbations
    max_iter = 1000
    gamma = 0.1
    perturb_freq = 50
    stagnation_limit = 100
    stagnation_count = 0
    
    h = h_start.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points/2) / np.sum(h)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > stagnation_limit:
            break
            
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h) - 1)
        
        # Create direction vector for Frank-Wolfe step
        # We want to move towards reducing the maximum overlap
        direction = np.zeros_like(h)
        # Perturb only at positions that contribute to max overlap
        for i in range(len(h)):
            # Calculate impact of decreasing h[i] on maximum overlap
            # This is a simplified approximation - for the actual optimization we
            # want to compute gradient of the overlap term
            if max_lag >= 0:
                # If positive lag, we're looking at h[i+k] terms
                if i + max_lag < len(h):
                    direction[i] = -1
            else:
                # If negative lag, we're looking at h[i+k] where k < 0
                if i >= -max_lag:
                    direction[i] = -1
        
        # Simplified approach: use a gradient-like direction more carefully
        # Since this is a nonconvex problem, we estimate a gradient
        # by examining local correlations
        direction = np.ones_like(h) * -1.0
        
        # Compute dotted products to get update direction  
        # Use manual small steps on coordinates
        h_new = h + gamma * direction
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check if moving in this direction improves the objective
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < current_c5:
            h = h_new.copy()
        else:
            # Try orthogonal direction in case the previous is bad
            direction = np.ones_like(h) * 1.0
            h_new = h + gamma * direction
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            new_c5 = evaluate_erdos_solution(h_new, dx)
            if new_c5 < current_c5:
                h = h_new.copy()
        
        # Apply random perturbation periodically to escape local minima
        if iteration % perturb_freq == 0:
            perturbation = np.random.normal(0, 0.02, n_points)

            h = h * (n_points/2) / np.sum(h)
    
    # Final refinement with SLSQP if we're close to good solution
    if time() - start_time < budget_s * 0.7:

                return evaluate_erdos_solution(h, dx)

            res = minimize(objective, h_refined, method='SLSQP',

            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
