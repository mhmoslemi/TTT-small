import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a symmetric construction
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe parameters
    max_iter = 1000
    gamma = 0.1  # Initial learning rate
    perturbation_freq = 50
    perturbation_strength = 0.05
    
    h_current = h_start.copy()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    stagnation_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlations
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Focus on relevant lags (around the center)
        center = len(corr) // 2
        relevant_corr = corr[max(0, center-50):min(len(corr), center+50)]
        max_idx = np.argmax(relevant_corr)
        max_lag = max_idx - 50 + center
        
        # Create the direction vector for Frank-Wolfe
        # We want to decrease the maximum correlation term
        # So we update in direction that reduces the correlation
        # Let's compute a gradient-like direction
        direction = np.zeros(n_points)
        for k in range(n_points):
            # The effect of changing h[k] on correlation term at lag max_lag
            # We are roughly estimating how changing h[k] affects correlation
            # by looking at correlation computation
            # correlation term at lag max_lag is sum(h[i] * (1 - h[i + max_lag]))
            if 0 <= k + max_lag < n_points:
                direction[k] += -(1 - h_current[k + max_lag])
            if 0 <= k - max_lag < n_points:
                direction[k] += h_current[k - max_lag]
        
        # Project direction to feasible set [0,1]
        # The Frank-Wolfe step: h_new = h_current + gamma * (direction_vec - h_current)
        # But, since h is constrained between 0 and 1, we need to make it feasible.
        # For convex constraints, the projection is done as h = clip(h + gamma * d, 0, 1)
        # But we do not want to just step in direction d, we want to project to feasible set.
        # Frank-Wolfe step = h_current + gamma * (s - h_current) where s is feasible
        # s = min(1, max(0, h_current + gamma*direction))
        s = np.clip(h_current + gamma * direction, 0, 1)
        h_next = np.clip(h_current + gamma * (s - h_current), 0, 1)
        
        # Projection to sum constraint
        h_next = h_next * (n_points/2) / np.sum(h_next)
        
        # Evaluate
        c5 = evaluate_erdos_solution(h_next, dx)
        
        if c5 < best_c5:
            best_h = h_next.copy()
            best_c5 = c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        h_current = h_next.copy()
        
        # Perturbation to escape local minima
        if iteration % perturbation_freq == 0 and iteration > 0:
            perturbation = np.random.normal(0, perturbation_strength, n_points)
            h_current = np.clip(h_current + perturbation, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
        
        if stagnation_count > 100:
            break
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
