import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create initial good starting point
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 100
    max_generations = 250
    elite_size = 10
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Mix of uniform random and Latin hypercube sampling
            if i % 2 == 0:
                h = np.random.rand(n_points)
                h = h * (n_points/2) / np.sum(h)
                h = np.clip(h, 0, 1)
            else:
                # Latin hypercube sampling
                h = np.linspace(0, 1, n_points+1)[1:] + np.random.rand(n_points) * (1.0/n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
        population.append(h)
        f = evaluate_erdos_solution(h, dx)
        fitness.append(f)
        if f < best_c5:
            best_h = h.copy()
            best_c5 = f
    
    # Hall of fame for best individuals
    hall_of_fame = list(zip(fitness, population))
    hall_of_fame.sort()
    hall_of_fame = hall_of_fame[:elite_size]
    
    # EA loop
    last_improvement = 0
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness
        indexed_pop = list(zip(fitness, population))
        indexed_pop.sort()
        population = [ind[1] for ind in indexed_pop]
        fitness = [ind[0] for ind in indexed_pop]
        
        # Check for stagnation
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            if last_improvement > 80:
                break
                
        # Select parents (tournament selection)
        parents = []
        for i in range(pop_size // 2):
            # Tournament
            k = 5
            tournament_indices = np.random.choice(pop_size, k)
            tournament_fitness = [fitness[idx] for idx in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            parents.append(population[winner_idx].copy())
        
        # Create offspring
        offspring = []
        crossover_rate = 0.7 if generation < 150 else 0.3
        mutate_rate = 0.2
        
        for i in range(0, len(parents), 2):
            p1 = parents[i]
            p2 = parents[min(i+1, len(parents)-1)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child1 = alpha * p1 + (1 - alpha) * p2
                child2 = (1 - alpha) * p1 + alpha * p2
            else:
                child1, child2 = p1.copy(), p2.copy()
                
            # Mutation
            if np.random.rand() < mutate_rate:
                # Gaussian mutation
                sigma = 0.1 * (1 - generation / max_generations)
                noise = np.random.normal(0, sigma, n_points)
                child1 += noise
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points/2) / np.sum(child1)
                
            if np.random.rand() < mutate_rate:
                # Bit-flip mutation for top 30%
                if i < len(parents) * 0.3:
                    flip_indices = np.random.choice(n_points, int(0.2 * n_points))
                    child2[flip_indices] = 1 - child2[flip_indices]
                    child2 = np.clip(child2, 0, 1)
                    child2 = child2 * (n_points/2) / np.sum(child2)
            
            offspring.append(child1)
            offspring.append(child2)
        
        # Replace population
        population = offspring
        fitness = [evaluate_erdos_solution(h, dx) for h in offspring]
        
        # Update hall of fame
        extended_hof = list(zip(fitness, population)) + hall_of_fame
        extended_hof.sort()
        hall_of_fame = extended_hof[:elite_size]
    
    # Final refinement with local optimization
    h_refined = best_h.copy()
    refined_c5 = best_c5
    
    # Try L-BFGS-B refinement on top candidates
    for i in range(min(5, len(hall_of_fame))):
        h_candidate = hall_of_fame[i][1]
        try:
            # Use bounds and constraints
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
            
            result = minimize(
                lambda x: evaluate_erdos_solution(x, dx),
                h_candidate,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 500}
            )
            
            if result.success:
                c5_candidate = evaluate_erdos_solution(result.x, dx)
                if c5_candidate < refined_c5:
                    refined_c5 = c5_candidate
                    h_refined = result.x.copy()
                    
        except:
            continue

    # Final feasibility check and normalization
    h_refined = np.clip(h_refined, 0, 1)
    h_refined = h_refined * (n_points/2) / np.sum(h_refined)
    final_c5 = evaluate_erdos_solution(h_refined, dx)
    
    return (list(h_refined), final_c5, n_points)
