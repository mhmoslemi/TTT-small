import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_h(n_points):
    x = np.linspace(0, 2, n_points)
    h_initial = 0.5 * (1 + np.sin(2 * np.pi * x))
    h_initial = np.clip(h_initial, 0, 1)
    normalization = (n_points / 2) / np.sum(h_initial)
    h_initial *= normalization
    return h_initial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = initialize_h(n_points)
        # Add some random perturbation
        noise = np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start + noise, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    T = 1.0
    alpha = 0.998
    tabu_size = 20
    tabu_list = []
    max_iter = 10000
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate
        candidate = best_h.copy()
        indices = np.random.choice(n_points, size=max(1, int(0.15 * n_points)), replace=False)
        
        for i in indices:
            perturbation = np.random.uniform(-0.15, 0.15)
            candidate[i] += perturbation
            candidate[i] = np.clip(candidate[i], 0, 1)
            
        # Normalize
        candidate = candidate * (n_points/2) / np.sum(candidate)
        
        # Check tabu
        candidate_tuple = tuple(candidate.round(6))
        if candidate_tuple in tabu_list:
            # Accept with probability
            delta = evaluate_erdos_solution(candidate, dx) - best_c5
            if delta > 0 and np.random.rand() > np.exp(-delta / T):
                continue
                
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(candidate, dx)
        
        # Accept or reject
        delta = c5_candidate - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = candidate.copy()
            best_c5 = c5_candidate
            
            # Update tabu list
            tabu_list.append(candidate_tuple)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        
        # Cool down
        T = max(0.001, T * alpha)
    
    # Final refinement with SLSQP if better than initial
    if best_c5 < evaluate_erdos_solution(h_start, dx):
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 500, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
