import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Multiresolution approach
    resolutions = [20, 40, 80]
    current_h = None
    current_n = 0
    best_h = None
    best_c5 = float('inf')
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize
        if current_h is None:
            if 'initial_h_values' in globals():
                h_start = np.array(initial_h_values)
                current_h = h_start.copy()
            else:
                # Periodic initialization
                x = np.linspace(0, 2, n_points)
                h_initial = 0.5 * (1 + np.sin(2 * np.pi * x / 5))  # 5 cycles
                h_initial = np.clip(h_initial, 0, 1)
                normalization = (n_points / 2) / np.sum(h_initial)
                h_initial *= normalization
                current_h = h_initial + np.random.normal(0, 0.05, n_points) * h_initial
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * (n_points / 2) / np.sum(current_h)
        else:
            # Interpolate and perturb
            old_h = current_h.copy()
            old_n = current_n
            current_h = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, old_n), old_h)
            current_h += np.random.normal(0, 0.05, n_points) * current_h
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            
        current_n = n_points
        dx = 2.0 / n_points
        
        # Local search
        n_iter = 200 if n_points == 80 else 100
        perturbation_rate = 0.2
        temp = 1.0
        tabu_list = []
        tabu_size = 20
        
        for iteration in range(n_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate candidate
            candidate = current_h.copy()
            n_perturb = max(1, int(perturbation_rate * n_points))
            indices = np.random.choice(n_points, size=n_perturb, replace=False)
            
            for i in indices:
                delta = np.random.uniform(-0.1, 0.1) * candidate[i]
                candidate[i] += delta
                candidate[i] = np.clip(candidate[i], 0, 1)
                
            # Normalize
            candidate = candidate * (n_points / 2) / np.sum(candidate)
            
            # Evaluate
            c5_candidate = evaluate_erdos_solution(candidate, dx)
            
            # Accept with simulated annealing criteria
            if c5_candidate < best_c5:
                best_h = candidate.copy()
                best_c5 = c5_candidate
                current_h = candidate.copy()
            else:
                # Accept with probability based on temperature
                if np.random.random() < math.exp(-(c5_candidate - best_c5) / temp):
                    current_h = candidate.copy()
                    
            # Temperature decay
            if iteration % 50 == 0 and iteration > 0:
                temp *= 0.99
                
            # Tabu list update
            if len(tabu_list) >= tabu_size:
                tabu_list.pop(0)
            tabu_list.append(current_h.copy())
    
    # Final refinement with optimization
    if best_h is not None and time() - start_time < budget_s - 10:
        try:
            dx = 2.0 / len(best_h)
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - len(h) / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(len(best_h))]
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 1000, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    if best_h is None:
        # Fallback
        if 'initial_h_values' in globals():
            h_start = np.array(initial_h_values)
        else:
            x = np.linspace(0, 2, 35)
            h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))  
            h_start = np.clip(h_start, 0, 1)
            normalization = (35 / 2) / np.sum(h_start)
            h_start *= normalization
        dx = 2.0 / 35
        best_c5 = evaluate_erdos_solution(h_start, dx)
        best_h = h_start
        
    return (list(best_h), best_c5, len(best_h))
