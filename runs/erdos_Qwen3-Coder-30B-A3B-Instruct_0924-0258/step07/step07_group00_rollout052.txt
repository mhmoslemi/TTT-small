import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with focus on max overlap
    max_iter = 10000
    step_size = 0.01
    tolerance = 1e-9
    min_step_size = 1e-6
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps and find maximum
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # The maximum overlap comes from lag (max_idx - (n_points - 1))
        lag = max_idx - (n_points - 1)
        
        # Compute gradient w.r.t. h for this specific lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + lag < n_points:
                grad[i] = 1 - 2 * best_h[i + lag]
        
        # Gradient descent step
        h_new = best_h - step_size * grad
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject the step
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            step_size = min(step_size * 1.2, 0.1)  # Increase step size if successful
        else:
            step_size = max(step_size * 0.5, min_step_size)  # Reduce step size if unsuccessful
            
        if abs(best_c5 - max_overlap) < tolerance:
            break
    
    # Final cleanup and validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
