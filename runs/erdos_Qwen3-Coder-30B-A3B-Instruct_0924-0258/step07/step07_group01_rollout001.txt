import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # CVXPY based optimization
    max_iterations = 20
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create CVXPY variables
        h_var = cp.Variable(n_points, bounds=(0, 1))
        
        # Create correlation constraint using auxiliary variables
        # For maximum overlap, we need to model max_lag correlation
        # We'll use a semi-infinite programming approach by sampling lags
        lags = np.arange(-n_points + 1, n_points)
        max_corr = cp.Variable()
        
        # Create constraints for maximum correlation
        corr_constraints = []
        for lag in lags:
            # Create correlation expression for this lag
            if lag >= 0:
                corr_expr = cp.sum(cp.multiply(h_var[lag:], 1 - h_var[:n_points-lag]))
            else:
                corr_expr = cp.sum(cp.multiply(h_var[:n_points+lag], 1 - h_var[-lag:]))
            corr_constraints.append(cp.constraints._cast_as_constraint(max_corr >= corr_expr * dx))
        
        # Add sum constraint
        sum_constraint = cp.sum(h_var) == n_points / 2.0
        
        # Objective
        objective = cp.Minimize(max_corr)
        prob = cp.Problem(objective, [sum_constraint] + corr_constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    # Renormalize to sum constraint
                    h_sol = h_sol * (n_points/2) / np.sum(h_sol)
                    c5_sol = evaluate_erdos_solution(h_sol, dx)
                    if c5_sol < best_c5:
                        best_h = h_sol.copy()
                        best_c5 = c5_sol
        except:
            pass
            
        # SLSQP refinement on best solution found so far
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
