import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 100
    max_generations = 250
    elite_size = 10
    mutation_rate = 0.2
    crossover_rate = 0.7
    
    # Initialize population
    population = []
    fitness = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
        population.append(h)
        f = evaluate_erdos_solution(h, dx)
        fitness.append(f)
    
    # Hall of Fame
    hall_of_fame = []
    for i in np.argsort(fitness)[:elite_size]:
        hall_of_fame.append(population[i].copy())
    
    # Track best solution
    best_idx = np.argmin(fitness)
    if fitness[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    # Evolutionary loop
    generation = 0
    no_improvement = 0
    max_no_improvement = 80
    
    while generation < max_generations and time() - start_time < budget_s * 0.95:
        # Sort by fitness
        sorted_idx = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_idx]
        sorted_fit = [fitness[i] for i in sorted_idx]
        
        # Create new population
        new_pop = []
        new_fit = []
        
        # Keep elite
        for i in range(elite_size):
            new_pop.append(sorted_pop[i].copy())
            new_fit.append(sorted_fit[i])
        
        # Generate rest through crossover and mutation
        while len(new_pop) < pop_size:
            # Tournament selection
            tour_size = 4
            idx1 = np.random.choice(sorted_idx[:pop_size//2], tour_size, replace=False)
            idx2 = np.random.choice(sorted_idx[:pop_size//2], tour_size, replace=False)
            sel1 = sorted_pop[np.argmin([fitness[i] for i in idx1])]
            sel2 = sorted_pop[np.argmin([fitness[i] for i in idx2])]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = alpha * sel1 + (1 - alpha) * sel2
            else:
                child = sel1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Gaussian perturbation
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                
                # Bit flip (flip 20% of elements)
                flip_idx = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
                for idx in flip_idx:
                    child[idx] = 1 - child[idx]
                child = child * (n_points / 2) / np.sum(child)
            
            # Feasibility repair
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_pop.append(child)
            new_fit.append(evaluate_erdos_solution(child, dx))
        
        population = new_pop
        fitness = new_fit
        
        # Update hall of fame
        for i in range(pop_size):
            if fitness[i] < sorted_fit[0]:
                hall_of_fame = [population[i].copy()] + hall_of_fame[:-1]
        
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement >= max_no_improvement:
            break
            
        generation += 1
    
    # Final local refinement with L-BFGS
    def obj_func(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Try local optimization on top candidates
    for i in range(min(5, len(hall_of_fame))):
        h_init = hall_of_fame[i]
        try:
            res = minimize(obj_func, h_init, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                           constraints={'type': 'eq', 'fun': constraint_sum},
                           options={'maxiter': 100})
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = res.x.copy()
        except:
            pass
    
    # Final CVXPY validation
    try:
        h_cvx = cp.Variable(n_points)
        constraints = [cp.sum(h_cvx) == n_points / 2.0, h_cvx >= 0, h_cvx <= 1]
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_cvx.value
            best_h = h_feasible.copy()
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass
    
    # Ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
