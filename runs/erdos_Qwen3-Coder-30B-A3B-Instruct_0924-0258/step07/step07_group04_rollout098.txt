import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a smooth starting function
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)  # Sine wave with offset
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe optimization
    max_iter = 1000
    gamma = 0.1  # Initial step size
    stagnation_count = 0
    last_best = best_c5
    
    h_current = h_start.copy()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            last_best = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Compute correlation with 1-h
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Focus on relevant lags (approximately ±50)
        mid = len(corr) // 2
        corr_window = corr[max(0, mid-50):min(len(corr), mid+50)]
        max_lag_idx = np.argmax(corr_window) + max(0, mid-50)
        max_lag = max_lag_idx - mid
        
        # Build direction vector based on maximum overlap
        # Create a vector that points toward reducing the maximum correlation
        direction = np.zeros_like(h_current)
        # Shift 1-h to align with the maximum correlation lag
        if max_lag >= 0:
            direction[max_lag:] = (1 - h_current[:-max_lag])
        else:
            direction[:max_lag] = (1 - h_current[-max_lag:])
        
        # Apply Frank-Wolfe step: move in the descent direction
        new_h = h_current + gamma * direction
        # Project onto feasible set: clip between 0,1
        new_h = np.clip(new_h, 0, 1)
        # Normalize to satisfy sum constraint
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Adaptive step size: reduce if no improvement
        h_current = new_h
        if iteration % 50 == 0:
            # Occasionally add perturbations
            perturbation = np.random.normal(0, 0.02, n_points)
            h_current = np.clip(h_current + perturbation, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Reduce step size if stagnating
        if stagnation_count > 50:
            gamma *= 0.9
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
